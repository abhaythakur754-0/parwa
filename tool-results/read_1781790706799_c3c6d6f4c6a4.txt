    43→def _format_notification(ntype: str, ev: Dict) -> tuple:
    44→    """Format notification title and description by type."""
    45→    signal = ev.get("signal", {})
    46→    score = ev.get("priority_score", 0)
    47→
    48→    if ntype == TYPE_STUCK_TICKET:
    49→        tid = signal.get("ticket_id", "unknown")
    50→        reason = signal.get("reason", "unknown")
    51→        quality = signal.get("quality_score", "?")
    52→        escalation = ev.get("escalation_tier", "soft_reminder")
    53→        hours = signal.get("hours_stuck", 0)
    54→        title = f"Stuck Ticket [{escalation.upper()}]: {tid}"
    55→        desc = (f"Ticket {tid} could not be resolved automatically. "
    56→                f"Reason: {reason}. Quality achieved: {quality}. "
    57→                f"Loops used: {signal.get('loops_used', 0)}. "
    58→                f"Hours stuck: {hours}. Escalation: {escalation}. "
    59→                f"Requires manual review.")
    60→
    61→    elif ntype == TYPE_QUOTA_LOW:
    62→        data = list(signal.values())[0] if signal else {}
    63→        tier = list(signal.keys())[0] if signal else "unknown"
    64→        title = f"Quota {data.get('status', 'warning').upper()}: {tier}"
    65→        desc = (f"{tier.capitalize()} plan quota is {data.get('burn_pct', 0)}% used "
    66→                f"({data.get('remaining', 0)} of {data.get('total', 0)} remaining). "
    67→                f"Consider upgrading or requesting additional quota.")
    68→
    69→    elif ntype == TYPE_ACCURACY_DROP:
    70→        trend = signal.get("trend", "unknown")
    71→        severity = signal.get("severity", "warning")
    72→        trigger = signal.get("trigger", "unknown")
    73→        acc_7d = signal.get("accuracy_7d", "?")
    74→        acc_today = signal.get("accuracy_today", "?")
    75→        title = f"Accuracy Drift [{severity.upper()}]: {trend}"
    76→        desc = (f"Resolution accuracy trend is '{trend}' (severity: {severity}). "
    77→                f"7-day avg: {acc_7d}, Today: {acc_today}. "
    78→                f"Trigger: {trigger}. "
    79→                f"This may indicate KB gaps, policy changes, or model drift. "
    80→                f"Review recent resolutions and update knowledge base if needed.")
    81→
    82→    elif ntype == TYPE_INTEGRATION_DOWN:
    83→        degraded = signal.get("degraded_services", [])
    84→        worst_uptime = signal.get("worst_uptime_pct", 0)
    85→        names = ", ".join(f"{d['name']} ({d['status']}, uptime={d.get('uptime_pct', '?')}%)" for d in degraded)
    86→        title = f"Integration Issues: {signal.get('total_degraded', len(degraded))} service(s) degraded"
    87→        desc = (f"Degraded integrations: {names}. "
    88→                f"Worst uptime: {worst_uptime}%. "
    89→                f"Check API credentials and service status.")
    90→
    91→    elif ntype == "load_bottleneck":
    92→        at_cap = signal.get("at_capacity", [])
    93→        high_load = signal.get("high_load", [])
    94→        vip = signal.get("vip_overflow_risk", False)
    95→        bottlenecks = ", ".join(v["name"] for v in at_cap + high_load)
    96→        title = f"Load Bottleneck: {bottlenecks or 'All variants'}"
    97→        desc = (f"Variant load issue detected. "
    98→                f"At capacity: {', '.join(v['name'] for v in at_cap) or 'none'}. "
    99→                f"High load: {', '.join(v['name'] for v in high_load) or 'none'}. "
   100→                f"VIP overflow risk: {vip}. "
   101→                f"Consider scaling up or routing overflow to available variants.")
   102→
   103→    else:
   104→        title = f"Signal: {ntype}"
   105→        desc = str(signal)
   106→
   107→    return title, desc

   110→async def _create_notifications_from_evals(
   111→    tenant_id: str,
   112→    evaluations: List[Dict[str, Any]],
   113→    db,
   114→) -> List[Dict[str, Any]]:
   115→    """Convert evaluations into notifications via DB. Filter LOW priority."""
   116→    notifications = []
   117→    for ev in evaluations:
   118→        score = ev.get("priority_score", 0)
   119→        if score < 0.40:
   120→            logger.info("LOW priority signal filtered: %s (score=%.3f)", ev.get("type"), score)
   121→            continue
   122→
   123→        ntype = ev.get("type", "unknown")
   124→        title, desc = _format_notification(ntype, ev)
   125→        related = ev.get("related_tickets", [])
   126→
   127→        nf = await db.create_notification(
   128→            tenant_id=tenant_id,
   129→            ntype=ntype,
   130→            priority_score=score,
   131→            title=title,
   132→            description=desc,
   133→            related_tickets=related,
   134→            batch_key=ev.get("type", ""),
   135→            source_data=ev.get("signal", {}),
   136→        )
   137→        notifications.append(nf)
   138→
   139→    return notifications

   142→# ── Command Execution (NEW — Wave 1) ─────────────────────────
   143→
   144→async def _execute_command(
   145→    intent_result: Dict[str, Any],
   146→    auth_result: AuthResult,
   147→    tenant_id: str,
   148→    signals: Dict[str, Any],
   149→    db,
   150→) -> str:
   151→    """Execute a classified command and return the response text.
   152→
   153→    This is the core "Jarvis does things" function.
   154→    Reads/writes DB based on intent. Returns human-readable response.
   155→    """
   156→    intent = intent_result["intent"]
   157→    target = intent_result["target"]
   158→
   159→    # ── QUERY INTENTS ────────────────────────────────────
   160→    if is_query_intent(intent):
   161→        response = await _handle_query(intent, target, tenant_id, signals, db)
   162→        # Log to audit
   163→        await db.create_audit_entry(
   164→            tenant_id=tenant_id,
   165→            action=f"query:{intent}",
   166→            actor_email=auth_result.email,
   167→            target_type="query",
   168→            payload={"intent": intent, "target": target},
   169→        )
   170→        return response
   171→
   172→    # ── CONTROL INTENTS ──────────────────────────────────
   173→    if is_control_intent(intent):
   174→        response = await _handle_control(intent, target, tenant_id, auth_result, db)
   175→        return response
   176→
   177→    # ── APPROVAL INTENTS ─────────────────────────────────
   178→    if is_approval_intent(intent):
   179→        response = await _handle_approval(intent, target, tenant_id, auth_result, db)
   180→        return response
   181→
   182→    # ── EMERGENCY INTENTS ────────────────────────────────
   183→    if is_emergency_intent(intent):
   184→        response = await _handle_emergency(intent, target, tenant_id, auth_result, db)
   185→        return response
   186→
   187→    # ── EXPLAIN / TEACH / AGENT ──────────────────────────
   188→    if intent == "explain_ticket":
   189→        return await _handle_explain_ticket(target, tenant_id, signals, db)
   190→    if intent == "explain_flag":
   191→        return await _handle_explain_flags(tenant_id, db)
   192→    if intent == "teach_skill":
   193→        return await _handle_teach_skill(target, tenant_id, auth_result, db)
   194→    if intent == "create_agent":
   195→        return await _handle_create_agent(target, tenant_id, auth_result, db)
   196→
   197→    return f"I understood your command but don't have a handler for '{intent}' yet. This will be available in a future wave."
   198→

   202→async def _handle_query(intent: str, target: str, tenant_id: str, signals: Dict, db) -> str:
   203→    """Handle all query intents — read from DB, format response."""
   204→
   205→    if intent == "query_status":
   206→        health = await db.health_check()
   207→        active_flags = await db.get_active_flags(tenant_id)
   208→        paused = [f for f in active_flags if f["flag_type"] == "pause_action"]
   209→        return (
   210→            f"**System Status**: {health.get('status', 'unknown')}\n"
   211→            f"Backend: {health.get('backend', 'unknown')}\n"
   212→            f"Active Flags: {len(active_flags)}\n"
   213→            f"Paused Actions: {len(paused)} {', '.join(f['flag_value'] for f in paused) if paused else 'none'}\n"
   214→            f"Mode: Supervised (default)"
   215→        )
   216→
   217→    if intent == "query_errors":
   218→        errors = signals.get("errors", [])
   219→        if not errors:
   220→            return "No errors detected in the current session."
   221→        return f"**Recent Errors** ({len(errors)}):\n" + "\n".join(f"- {e}" for e in errors[-5:])
   222→
   223→    if intent == "query_tickets":
   224→        flow = signals.get("ticket_flow", {})
   225→        summary = flow.get("summary", {}) if isinstance(flow, dict) else {}
   226→        current = flow.get("current_ticket", {}) if isinstance(flow, dict) else {}
   227→        drift = signals.get("drift_status", {})
   228→        return (
   229→            f"**Ticket Flow Summary**:\n"
   230→            f"Total Processed: {summary.get('total', 0)}\n"
   231→            f"Auto-Resolved: {summary.get('auto_resolved', 0)}\n"
   232→            f"Batched: {summary.get('batched', 0)}\n"
   233→            f"Escalated: {summary.get('escalated', 0)}\n"
   234→            f"Stuck: {summary.get('stuck', 0)}\n"
   235→            f"Avg Quality: {summary.get('avg_quality', 'N/A')}\n"
   236→            f"Avg LLM Calls: {summary.get('avg_llm_calls', 'N/A')}\n"
   237→            f"By Path: {summary.get('by_type', {})}\n"
   238→            f"Accuracy Trend: {drift.get('trend_direction', 'no_data')}"
   239→        )
   240→
   241→    if intent == "query_quality":
   242→        stats = await db.get_quality_stats(tenant_id)
   243→        drift = signals.get("drift_status", {})
   244→        drift_line = f"\nDrift: {drift.get('trend_direction', 'no_data')}" \
   245→                    f" (detected={drift.get('drift_detected')}, severity={drift.get('drift_severity')})" \
   246→                    if drift.get('total_scores', 0) > 0 else "\nDrift: insufficient data"
   247→        return (
   248→            f"**Quality Metrics**:\n"
   249→            f"Total Tickets: {stats['total_tickets']}\n"
   250→            f"Average Quality: {stats['avg_quality']:.2%}\n"
   251→            f"Average Confidence: {stats['avg_confidence']:.2%}\n"
   252→            f"Auto-Resolved: {stats['auto_resolved']}\n"
   253→            f"Escalated: {stats['escalated']}\n"
   254→            f"Stuck: {stats['stuck']}\n"
   255→            f"By Path: {stats.get('by_path', {})}"
   256→            f"{drift_line}"
   257→        )
   258→
   259→    if intent == "query_quota":
   260→        quota = signals.get("quota_status", {})
   261→        if not quota:
   262→            return "No quota data available. Connect to billing system for live quota tracking."
   263→        parts = []
   264→        for tier, data in quota.items():
   265→            parts.append(
   266→                f"  {tier}: {data.get('remaining', 0)}/{data.get('total', 0)} "
   267→                f"remaining ({data.get('burn_pct', 0)}% used, status: {data.get('status', 'unknown')})"
   268→            )
   269→        return f"**Quota Status**:\n" + "\n".join(parts)
   270→
   271→    if intent == "query_notifications":
   272→        if target and target.startswith("PARWA-NFY"):
   273→            # Specific notification lookup
   274→            nf = await db.get_notification(target)
   275→            if nf:
   276→                return (
   277→                    f"**{nf['title']}** (Key: {nf['notification_key']})\n\n"
   278→                    f"{nf['description']}\n\n"
   279→                    f"Priority: {nf['priority']} (score: {nf['priority_score']})\n"
   280→                    f"Status: {'Resolved' if nf['is_resolved'] else 'Unresolved'}\n"
   281→                    f"Type: {nf['type']}\n"
   282→                    f"Related tickets: {', '.join(nf['related_tickets']) or 'None'}"
   283→                )

   285→        else:
   286→            # List all notifications
   287→            nfs = await db.get_notifications(tenant_id)
   288→            if not nfs:
   289→                return "No unresolved notifications. All clear."

   291→            for nf in nfs[:10]:
   292→                lines.append(f"  [{nf['priority']}] {nf['notification_key']}: {nf['title']}")

   294→                lines.append(f"  ... and {len(nfs) - 10} more")

   296→
   297→    if intent == "query_flags":
   298→        flags = await db.get_active_flags(tenant_id)
   299→        if not flags:
   300→            return "No active flags/rules. System is running with default behavior."
   301→        lines = [f"**Active Flags** ({len(flags)}):"]
   302→        for f in flags:
   303→            lines.append(
   304→                f"  [{f['flag_type']}] {f['flag_value']} "
   305→                f"(scope={f['scope']}, set by {f['set_by']})"
   306→            )
   307→        return "\n".join(lines)
   308→
   309→    if intent == "query_audit":
   310→        trail = await db.get_audit_trail(tenant_id, limit=20)
   311→        if not trail:
   312→            return "No audit trail entries yet."
   313→        lines = [f"**Recent Activity** ({len(trail)} entries):"]
   314→        for entry in trail[:10]:
   315→            lines.append(
   316→                f"  [{entry['created_at'][:19]}] {entry['action']} "
   317→                f"by {entry['actor_email']}"
   318→            )
   319→        if len(trail) > 10:
   320→            lines.append(f"  ... and {len(trail) - 10} more")
   321→        return "\n".join(lines)
   322→
   323→    # ── Wave 2 Query Handlers ─────────────────────────────
   324→
   325→    if intent == "query_health":
   326→        health = signals.get("integration_health", {})
   327→        services = health.get("services", {})
   328→        if not services:
   329→            return "No integration health data yet. Run a poll cycle first to collect ping data."
   330→        lines = [f"**Integration Health** ({health.get('healthy_count', 0)} healthy, "
   331→                 f"{health.get('degraded_count', 0)} degraded):"]
   332→        for svc_name, svc_data in services.items():
   333→            status_icon = "OK" if svc_data.get("status") == "healthy" else "!!"
   334→            lines.append(
   335→                f"  [{status_icon}] {svc_name}: {svc_data.get('status', '?')} "
   336→                f"(uptime={svc_data.get('uptime_pct', '?')}%, "
   337→                f"avg={svc_data.get('avg_response_ms', '?')}ms" +
   338→                (f", error: {svc_data.get('last_error', 'none')}"
   339→                 if svc_data.get('last_error') else ")")
   340→            )
   341→        return "\n".join(lines)
   342→
   343→    if intent == "query_cost":
   344→        costs = signals.get("llm_costs", {})
   345→        persisted = costs.get("persisted", {})
   346→        live = costs.get("live_session", {})

   348→            f"**LLM Cost Summary**:",

   350→            f"  Total Calls (persisted): {persisted.get('total_calls', 0)}",

   356→        if by_model:
   357→            lines.append("  By Model:")
   358→            for model, mdata in by_model.items():
   359→                lines.append(f"    {model}: ${mdata['cost']:.4f} ({mdata['calls']} calls, {mdata['tokens']} tokens)")
   360→        return "\n".join(lines)
   361→
   362→    if intent == "query_flow":
   363→        flow = signals.get("ticket_flow", {})
   364→        summary = flow.get("summary", {}) if isinstance(flow, dict) else {}
   365→        by_node = summary.get("by_node", {})
   366→        lines = [
   367→            f"**Ticket Flow Metrics**:",
   368→            f"  Total: {summary.get('total', 0)}",
   369→            f"  Auto-Resolved: {summary.get('auto_resolved', 0)}",
   370→            f"  Batched: {summary.get('batched', 0)}",
   371→            f"  Escalated: {summary.get('escalated', 0)}",
   372→            f"  Stuck: {summary.get('stuck', 0)}",

   397→
   398→    if intent == "query_stuck":
   399→        stuck = signals.get("stuck_tickets", [])

   406→            lines.append(
   407→                f"  [{tier.upper()}] {s.get('ticket_id', '?')}: "
   408→                f"reason={s.get('reason', '?')}, "
   409→                f"quality={s.get('quality_score', '?')}, "
   410→                f"{hours}h stuck, loops={s.get('loops_used', 0)}"
   411→            )
   412→        return "\n".join(lines)
   413→
   414→    return f"Query '{intent}' not yet implemented."

   417→# ── Control Handlers ──────────────────────────────────────────
   418→
   419→async def _handle_control(intent: str, target: str, tenant_id: str, auth: AuthResult, db) -> str:
   420→    """Handle control commands — writes to system_flags table."""
   421→
   422→    if intent == "control_pause":
   423→        flag = await db.set_flag(
   424→            tenant_id=tenant_id,
   425→            flag_type="pause_action",
   426→            flag_value=target,
   427→            set_by=auth.email,
   428→            reason=f"Paused via Jarvis by {auth.email}",
   429→        )
   430→        # Audit
   431→        await db.create_audit_entry(
   432→            tenant_id=tenant_id, action="control_pause",
   433→            actor_email=auth.email, target_type="flag", target_id=flag["id"],
   434→            payload={"target": target, "intent": intent},
   435→        )
   436→        return f"[OK] Paused '{target}'. PARWA will stop processing {target} requests. Use 'resume {target}' to re-enable."
   437→
   438→    if intent == "control_resume":
   439→        # Revoke the most recent pause flag for this target
   440→        flags = await db.get_active_flags(tenant_id, flag_type="pause_action")
   441→        revoked = 0
   442→        for f in flags:
   443→            if f["flag_value"] == target or target == "all":
   444→                await db.revoke_flag(f["id"], auth.email)
   445→                revoked += 1
   446→        await db.create_audit_entry(
   447→            tenant_id=tenant_id, action="control_resume",
   448→            actor_email=auth.email, target_type="flag", target_id=target,
   449→            payload={"target": target, "revoked_count": revoked},
   450→        )
   451→        if revoked > 0:
   452→            return f"[OK] Resumed '{target}'. Revoked {revoked} pause flag(s)."
   453→        return f"No active pause flag found for '{target}'. Already running."
   454→
   455→    if intent == "control_route":
   456→        # Parse "handle Instagram DMs" → channel=instagram, route_to=ai
   457→        route_to = "ai"  # default
   458→        if "human" in target or "take" in target or "i'll" in target:
   459→            route_to = "human"
   460→        flag = await db.set_flag(
   461→            tenant_id=tenant_id,
   462→            flag_type="redirect_channel",
   463→            flag_value=f"{target}:{route_to}",
   464→            set_by=auth.email,
   465→            reason=f"Redirected {target} to {route_to} by {auth.email}",
   466→        )
   467→        await db.create_audit_entry(
   468→            tenant_id=tenant_id, action="control_route",
   469→            actor_email=auth.email, target_type="flag", target_id=flag["id"],
   470→            payload={"channel": target, "route_to": route_to},
   471→        )
   472→        return f"[OK] Workflow Redirected: {target} → {route_to.upper()}. PARWA will {'handle' if route_to == 'ai' else 'skip'} {target} requests."
   473→
   474→    if intent == "control_mode":
   475→        valid_modes = {"shadow", "supervised", "graduated"}
   476→        mode = target.lower() if target.lower() in valid_modes else "supervised"
   477→        flag = await db.set_flag(
   478→            tenant_id=tenant_id,
   479→            flag_type="force_mode",
   480→            flag_value=mode,
   481→            set_by=auth.email,
   482→            reason=f"Mode changed to {mode} by {auth.email}",
   483→        )
   484→        await db.create_audit_entry(
   485→            tenant_id=tenant_id, action="control_mode",
   486→            actor_email=auth.email, target_type="flag", target_id=flag["id"],
   487→            payload={"mode": mode},
   488→        )
   489→        return f"[OK] System mode set to **{mode.upper()}**. PARWA will operate in {mode} mode."
   490→
   491→    if intent == "control_disable_rule":
   492→        # Revoke the most recent non-expired flag
   493→        flags = await db.get_active_flags(tenant_id)
   494→        if flags:
   495→            last = flags[-1]
   496→            await db.revoke_flag(last["id"], auth.email)
   497→            await db.create_audit_entry(
   498→                tenant_id=tenant_id, action="control_disable_rule",
   499→                actor_email=auth.email, target_type="flag", target_id=last["id"],
   500→                payload={"revoked_flag": last["flag_type"], "revoked_value": last["flag_value"]},
   501→            )
   502→            return f"[OK] Disabled last rule: {last['flag_type']}={last['flag_value']}. System reverted to default behavior."
   503→        return "No active rules to disable."
   504→
   505→    if intent == "control_skill_assign":
   506→        return "[OK] Skill re-assignment noted. This requires variant config updates (coming in Wave 3)."
   507→
   508→    return f"Control '{intent}' received but not yet fully implemented."

   623→# ── Main Node Function ────────────────────────────────────────
   624→
   625→
   626→async def jarvis_notify(state: dict) -> dict:
   627→    """Jarvis Node 3: NOTIFY — Act + Verify.
   628→
   629→    Wave 1: Now wired with command_parser + jarvis_auth + jarvis_db.
   630→    Full pipeline: chat → parse → auth → execute → DB → response.
   631→    """
   632→    start = time.time()
   633→    tenant_id = state.get("tenant_id", "")
   634→    trigger = state.get("trigger", "poll")
   635→    evaluations = state.get("evaluations", [])
   636→    signals = state.get("signals", {})
   637→    admin_question = state.get("admin_question", "")
   638→    user_context = state.get("user_context", {})
   639→    logs = []
   640→    llm_calls = 0
   641→
   642→    db = get_db()
   643→
   644→    # 1. Create notifications from evaluations (poll/monitor mode)
   645→    notifications = []
   646→    if trigger in ("poll", "stuck_ticket") and evaluations:
   647→        notifications = await _create_notifications_from_evals(tenant_id, evaluations, db)
   648→        logs.append({"node": "J3", "technique": "CreateNotifications", "duration_ms": 0,
   649→                     "result_summary": f"created={len(notifications)}"})
   650→
   651→    # 2. Handle admin chat (full pipeline: parse → auth → execute)
   652→    chat_response = ""
   653→    if trigger == "admin_chat" and admin_question:
   654→        # Step 1: Classify intent
   655→        intent_result = await classify_command(admin_question)
   656→        if intent_result.get("classification_method") == "llm":
   657→            llm_calls += 1
   658→        logs.append({"node": "J3", "technique": "CommandParser",
   659→                     "duration_ms": 0, "result_summary": f"intent={intent_result['intent']} "
   660→                     f"method={intent_result['classification_method']} "
   661→                     f"confidence={intent_result['confidence']}"} )
   662→
   663→        # Step 2: Authorize
   664→        auth_result = await authorize_command(
   665→            intent=intent_result["intent"],
   666→            user_context=user_context,
   667→            tenant_id=tenant_id,
   668→        )
   669→        logs.append({"node": "J3", "technique": "AuthCheck",
   670→                     "duration_ms": 0, "result_summary": f"authorized={auth_result.authorized} "
   671→                     f"role={auth_result.role}"})
   672→
   673→        # Step 3: Execute (or deny)
   674→        if auth_result.authorized:
   675→            chat_response = await _execute_command(
   676→                intent_result=intent_result,
   677→                auth_result=auth_result,
   678→                tenant_id=tenant_id,
   679→                signals=signals,
   680→                db=db,
   681→            )
   682→            logs.append({"node": "J3", "technique": "ExecuteCommand",
   683→                         "duration_ms": 0, "result_summary": f"executed={intent_result['intent']}"})
   684→        else:
   685→            chat_response = f"[DENIED] {auth_result.reason}"
   686→            logs.append({"node": "J3", "technique": "AuthDenied",
   687→                         "duration_ms": 0, "result_summary": auth_result.reason})
   688→
   689→    # 3. Build quota feedback for PARWA Node 2
   690→    quota_feedback = {
   691→        "tenant_id": tenant_id,
   692→        "quota_status": signals.get("quota_status", {}),
   693→        "variant_tier": list(signals.get("quota_status", {}).keys())[0] if signals.get("quota_status") else "parwa",
   694→    }
   695→
   696→    # 4. Summary
   697→    keys = [n["notification_key"] for n in notifications]
   698→    elapsed = int((time.time() - start) * 1000)
   699→    logger.info("Jarvis NOTIFY complete: tenant=%s notifications=%d chat=%s llm=%d auth=%s [%dms]",
   700→                tenant_id, len(notifications), bool(chat_response), llm_calls,
   701→                bool(user_context.get("role")), elapsed)
   702→
   703→    return {
   704→        "notifications": notifications,
   705→        "notification_keys": keys,
   706→        "chat_response": chat_response,
   707→        "quota_feedback": quota_feedback,
   708→        "notify_log": logs,
   709→        "total_token_usage": state.get("total_token_usage", 0) + llm_calls,
   710→        "intent_result": intent_result if trigger == "admin_chat" else None,
   711→        "auth_result": {"authorized": auth_result.authorized, "role": auth_result.role}
   712→                        if trigger == "admin_chat" else None,
   713→    }

    61→def _rule_based_check(
    62→    action: str, amount: float, tier: str
    63→) -> Dict[str, Any]:
    64→    """Check if action can be executed based on variant rules."""
    65→    caps = _CAPABILITY_MATRIX.get(tier, _CAPABILITY_MATRIX["mini"])
    66→    limits = _EXEC_LIMITS.get(tier, _EXEC_LIMITS["mini"])
    67→
    68→    can_execute = True
    69→    reason = ""
    70→
    71→    if action == "execute_refund":
    72→        if not caps["execute_refund"]:
    73→            can_execute = False
    74→            reason = f"Tier '{tier}' cannot execute refunds — recommend only"
    75→        elif amount > limits["max_refund"]:
    76→            can_execute = False
    77→            reason = f"Amount ${amount} exceeds tier '{tier}' limit of ${limits['max_refund']}"
    78→
    79→    elif action == "execute_credit":
    80→        if not caps["execute_credit"]:
    81→            can_execute = False
    82→            reason = f"Tier '{tier}' cannot execute credits — recommend only"
    83→        elif amount > limits["max_credit"]:
    84→            can_execute = False
    85→            reason = f"Amount ${amount} exceeds tier '{tier}' limit of ${limits['max_credit']}"
    86→
    87→    elif action == "account_change":
    88→        if not caps["account_change"]:
    89→            can_execute = False
    90→            reason = f"Tier '{tier}' cannot execute account changes — recommend only"
    91→
    92→    elif action == "provide_info":
    93→        can_execute = True
    94→        reason = "Information provision — no execution needed"
    95→
    96→    else:
    97→        can_execute = True
    98→        reason = f"Action '{action}' — no restrictions apply"
    99→
   100→    return {"can_execute": can_execute, "reason": reason}

   151→# ── ReAct: Think-Act-Observe loop (LLM) ───────────────────────────
   152→
   153→
   154→async def _execute_real_refund(
   155→    tenant_id: str, details: Dict, crm_data: Dict,
   156→) -> Dict[str, Any]:
   157→    """Execute a REAL refund via Stripe API using the tenant's stored credentials.
   158→
   159→    This calls the Stripe Refunds API to actually move money back to the
   160→    customer's card. The tenant must have Stripe connected via the
   161→    integration catalog (credentials stored in the integrations table).
   162→
   163→    Returns:
   164→        Dict with: success (bool), amount (float), refund_id (str), error (str)
   165→    """
   166→    import httpx
   167→    from database.base import SessionLocal
   168→    from app.services.integration_service import IntegrationService
   169→
   170→    # Extract refund details from the action details or CRM data
   171→    amount = details.get("amount", 0)
   172→    payment_intent_id = details.get("payment_intent_id", "")
   173→    order_id = details.get("order_id", "")
   174→    charge_id = details.get("charge_id", "")
   175→
   176→    # Try to find the payment_intent_id from CRM/ecommerce data
   177→    if not payment_intent_id and crm_data:
   178→        orders = crm_data.get("ecommerce_orders")
   179→        if orders:
   180→            if isinstance(orders, list) and len(orders) > 0:
   181→                order = orders[0] if isinstance(orders[0], dict) else {}
   182→                payment_intent_id = order.get("payment_intent_id", "") or order.get("transaction_id", "")
   183→                if not amount:
   184→                    amount = order.get("total_price", 0) or order.get("amount", 0)
   185→
   186→    if not amount or float(amount) <= 0:
   187→        return {"success": False, "amount": 0, "refund_id": "", "error": "No refund amount specified"}
   188→
   189→    # Load the tenant's Stripe credentials
   190→    db = SessionLocal()
   191→    try:
   192→        service = IntegrationService(db)
   193→        creds = service.get_credential_config(tenant_id, "stripe")
   194→        if not creds or not creds.get("api_key"):
   195→            return {
   196→                "success": False,
   197→                "amount": amount,
   198→                "refund_id": "",
   199→                "error": "Stripe not connected — tenant has not provided Stripe API key",
   200→            }
   201→
   202→        stripe_key = creds["api_key"]
   203→
   204→        # Call Stripe Refunds API
   205→        # https://stripe.com/docs/api/refunds/create
   206→        refund_payload = {
   207→            "amount": int(float(amount) * 100),  # Stripe uses cents
   208→            "reason": "requested_by_customer",
   209→        }
   210→        if payment_intent_id:
   211→            refund_payload["payment_intent"] = payment_intent_id
   212→        elif charge_id:
   213→            refund_payload["charge"] = charge_id
   214→
   215→        async with httpx.AsyncClient(timeout=30.0) as client:
   216→            response = await client.post(
   217→                "https://api.stripe.com/v1/refunds",
   218→                headers={
   219→                    "Authorization": f"Bearer {stripe_key}",
   220→                    "Content-Type": "application/x-www-form-urlencoded",
   221→                },
   222→                data=refund_payload,
   223→            )
   224→
   225→        if response.status_code == 200:
   226→            refund_data = response.json()
   227→            logger.info(
   228→                "stripe_refund_executed tenant=%s amount=%s refund_id=%s",
   229→                tenant_id, amount, refund_data.get("id", ""),
   230→            )
   231→            return {
   232→                "success": True,
   233→                "amount": amount,
   234→                "refund_id": refund_data.get("id", ""),
   235→                "status": refund_data.get("status", "pending"),
   236→            }
   237→        else:
   238→            error_msg = response.json().get("error", {}).get("message", f"Stripe API returned {response.status_code}")
   239→            logger.warning("stripe_refund_failed tenant=%s error=%s", tenant_id, error_msg[:200])
   240→            return {
   241→                "success": False,
   242→                "amount": amount,
   243→                "refund_id": "",
   244→                "error": error_msg,
   245→            }
   246→    except Exception as exc:
   247→        logger.warning("stripe_refund_exception tenant=%s error=%s", tenant_id, str(exc)[:200])
   248→        return {
   249→            "success": False,
   250→            "amount": amount,
   251→            "refund_id": "",
   252→            "error": f"Refund execution error: {str(exc)[:150]}",
   253→        }
   254→    finally:
   255→        db.close()

   258→async def _react_execute(
   259→    action: str, details: Dict, knowledge: str, crm_data: Dict,
   260→    tenant_id: str = "",
   261→) -> Dict[str, Any]:
   262→    """Think-Act-Observe loop for complex action execution.
   263→
   264→    Previously this was a stub that returned "Action simulated successfully"
   265→    without calling any real tool. Now it dispatches to the ReActToolRegistry
   266→    which has 7 real tools (CRMTool→HubSpot, BillingTool, OrderTool→Shopify,
   267→    TicketTool, ServiceHealthChecker, KnownIssueDetector, ConfigValidator).
   268→
   269→    The LLM generates a THOUGHT about what tool to call, then we map common
   270→    action keywords to tool actions and execute them. If no tool matches,
   271→    we fall back to the LLM-generated thought + a honest "no_tool_executed"
   272→    observation so the AI knows the action wasn't actually performed.
   273→    """
   274→    prompt = f"""You are executing a customer support action.
   275→
   276→Action: {action}
   277→Details: {details}
   278→Knowledge: {knowledge[:1500]}
   279→Customer Data: {str(crm_data)[:500]}
   280→
   281→Think about what needs to happen:
   282→THOUGHT:"""
   283→
   284→    try:
   285→        thought = await llm_call(prompt, max_tokens=200, temperature=0.2)
   286→    except Exception as e:
   287→        thought = f"Action execution failed: {e}"
   288→
   289→    # ── Dispatch to real tools via ReActToolRegistry ──────────────
   290→    # Map common action keywords to tool calls. The ReActToolRegistry tools
   291→    # I rewrote earlier call real external APIs (HubSpot, Shopify, etc.).
   292→    observation = None
   293→    tool_executed = None
   294→
   295→    try:
   296→        from app.core.react_tools import ReActToolRegistry
   297→        registry = ReActToolRegistry()
   298→        # initialize_defaults registers all 7 tools.
   299→        try:
   300→            import asyncio
   301→            asyncio.get_event_loop()
   302→            # If we're in an async context, can't call the async init —
   303→            # use the sync variant.
   304→            registry.register_tool_sync = registry.register_tool_sync  # noqa
   305→        except RuntimeError:
   306→            pass
   307→        # Initialize synchronously (the tools' __init__ are sync).
   308→        try:
   309→            await registry.initialize_defaults()
   310→        except Exception:
   311→            # initialize_defaults might be async; fall back to manual sync registration.
   312→            try:
   313→                from app.core.react_tools.order_tool import OrderTool
   314→                from app.core.react_tools.billing_tool import BillingTool
   315→                from app.core.react_tools.crm_tool import CRMTool
   316→                from app.core.react_tools.ticket_tool import TicketTool
   317→                registry.register_tool_sync(OrderTool())
   318→                registry.register_tool_sync(BillingTool())
   319→                registry.register_tool_sync(CRMTool())
   320→                registry.register_tool_sync(TicketTool())
   321→            except Exception:
   322→                pass
   323→
   324→        action_lower = action.lower()
   325→        details_str = str(details)
   326→
   327→        # Map action → tool call based on keywords.
   328→        if any(kw in action_lower for kw in ("customer", "contact", "crm", "lookup")):
   329→            customer_id = details.get("customer_id", "") or details.get("id", "")
   330→            if customer_id and tenant_id:
   331→                result = await registry.execute("crm_integration", "get_customer", tenant_id, customer_id=customer_id)
   332→                if result.success:
   333→                    observation = f"Customer lookup succeeded: {str(result.data)[:300]}"
   334→                    tool_executed = "crm_integration.get_customer"
   335→                else:
   336→                    observation = f"Customer lookup failed: {result.error}"
   337→                    tool_executed = "crm_integration.get_customer (failed)"
   338→
   339→        elif any(kw in action_lower for kw in ("order", "ecommerce", "shopify")):
   340→            order_id = details.get("order_id", "") or details.get("id", "")
   341→            if order_id and tenant_id:
   342→                result = await registry.execute("order_management", "get_order", tenant_id, order_id=order_id)
   343→                if result.success:
   344→                    observation = f"Order lookup succeeded: {str(result.data)[:300]}"
   345→                    tool_executed = "order_management.get_order"
   346→                else:
   347→                    observation = f"Order lookup failed: {result.error}"
   348→                    tool_executed = "order_management.get_order (failed)"
   349→
   350→        elif any(kw in action_lower for kw in ("billing", "invoice", "payment", "refund")):
   351→            if tenant_id:
   352→                # ── REAL STRIPE REFUND EXECUTION ──────────────────────
   353→                # When the action is execute_refund AND the tenant has Stripe
   354→                # connected, actually call the Stripe API to process the refund.
   355→                # This updates the company's Stripe account (real money movement)
   356→                # and Node 6.5 will push the result to their CRM.
   357→                if action == "execute_refund":
   358→                    refund_result = await _execute_real_refund(
   359→                        tenant_id, details, crm_data,
   360→                    )
   361→                    if refund_result["success"]:
   362→                        observation = (
   363→                            f"REFUND EXECUTED via Stripe: ${refund_result['amount']} "
   364→                            f"refund ID: {refund_result['refund_id']}. "
   365→                            f"The customer's payment has been refunded."
   366→                        )
   367→                        tool_executed = "stripe.execute_refund"
   368→                    else:
   369→                        observation = (
   370→                            f"Refund execution failed: {refund_result['error']}. "
   371→                            f"Falling back to recommendation only."
   372→                        )
   373→                        tool_executed = "stripe.execute_refund (failed)"
   374→                else:
   375→                    # Non-refund billing action — use the billing tool (lookup)
   376→                    result = await registry.execute("billing_system", "get_billing_summary", tenant_id)
   377→                    if result.success:
   378→                        observation = f"Billing lookup succeeded: {str(result.data)[:300]}"
   379→                        tool_executed = "billing_system.get_billing_summary"
   380→                    else:
   381→                        observation = f"Billing lookup failed: {result.error}"
   382→                        tool_executed = "billing_system.get_billing_summary (failed)"
   383→
   384→        elif any(kw in action_lower for kw in ("ticket", "escalate", "resolve")):
   385→            ticket_id = details.get("ticket_id", "") or details.get("id", "")
   386→            if ticket_id and tenant_id:
   387→                result = await registry.execute("ticket_system", "get_ticket", tenant_id, ticket_id=ticket_id)
   388→                if result.success:
   389→                    observation = f"Ticket lookup succeeded: {str(result.data)[:300]}"
   390→                    tool_executed = "ticket_system.get_ticket"
   391→                else:
   392→                    observation = f"Ticket lookup failed: {result.error}"
   393→                    tool_executed = "ticket_system.get_ticket (failed)"
   394→
   395→    except Exception as exc:
   396→        observation = f"Tool dispatch failed: {str(exc)[:200]}"
   397→        tool_executed = None
   398→
   399→    if observation is None:
   400→        # No tool matched the action — honest observation so the AI knows.
   401→        observation = f"No tool executed for action '{action}'. Thought: {thought[:200]}"
   402→        tool_executed = None
   403→
   404→    return {
   405→        "action": action,
   406→        "thought": thought,
   407→        "observation": observation,
   408→        "tool_executed": tool_executed,
   409→        "status": "completed",
   410→    }

   413→# ── L4: RuleBasedAction Enhanced — time/frequency/category rules (non-LLM) ─
   414→
   415→
   416→def _rule_based_check_enhanced(
   417→    action: str, details: Dict, tier: str, dynamic_ctx: Dict,
   418→) -> Dict[str, Any]:
   419→    """Enhanced rule-based check beyond tier+amount.
   420→
   421→    The basic _rule_based_check only validates tier permissions and
   422→    amount limits. This enhanced check adds:
   423→    - Time window violations (refund on 45-day-old order)
   424→    - Frequency violations (3rd refund this week)
   425→    - Category restrictions (digital downloads non-refundable)
   426→
   427→    Returns can_execute + all reasons if blocked.
   428→    """
   429→    # Start with the basic check result
   430→    basic = _rule_based_check(action, details.get("amount", 0), tier)
   431→
   432→    # If basic check already blocks, no need to check further
   433→    if not basic["can_execute"]:
   434→        return basic
   435→
   436→    # Check dynamic context blocks (time, frequency, category)
   437→    blocks = dynamic_ctx.get("blocks", [])
   438→    if blocks:
   439→        return {
   440→            "can_execute": False,
   441→            "reason": f"Rule violation: {'; '.join(blocks)}",
   442→            "blocks": blocks,
   443→        }
   444→
   445→    return {
   446→        "can_execute": True,
   447→        "reason": basic.get("reason", ""),
   448→        "blocks": [],
   449→    }

   485→def _zero_shot_flag_action(action: str, details: Dict, knowledge: str) -> Dict[str, Any]:
   486→    """Flag statistically unusual actions."""
   487→    flags = []
   488→
   489→    amount = details.get("amount", 0)
   490→    if amount > 5000:
   491→        flags.append(f"High-value action: ${amount}")
   492→    if amount < 0:
   493→        flags.append("Negative amount detected")
   494→
   495→    # Check if action type matches knowledge
   496→    action_in_kb = action.replace("_", " ") in knowledge.lower()
   497→    if not action_in_kb and action != "provide_info":
   498→        flags.append(f"Action type '{action}' not found in knowledge base")
   499→
   500→    return {
   501→        "flagged": len(flags) > 0,
   502→        "flags": flags,
   503→        "severity": "high" if amount > 5000 else "low",
   504→    }

   532→
   533→def _dynamic_context(
   534→    action: str, details: Dict, crm_data: Dict, state: Dict,
   535→) -> Dict[str, Any]:
   536→    """Enrich action context with real-time data.
   537→
   538→    Adds three things ReAct doesn't have:
   539→    1. Today's date — for "within 30 days" policy checks
   540→    2. Recent action frequency — "3rd refund this week" → block
   541→    3. Order category — "digital download" → non-refundable
   542→
   543→    Returns enriched context + any time/frequency/category blocks.
   544→    """
   545→    now = datetime.now(timezone.utc)
   546→    context = {
   547→        "current_date": now.isoformat(),
   548→        "blocks": [],
   549→        "warnings": [],
   550→    }
   551→
   552→    # 1. Time-window check: is this order too old for a refund?
   553→    if action in ("execute_refund", "execute_credit"):
   554→        order_date = None
   555→        if isinstance(crm_data, dict):
   556→            orders = crm_data.get("ecommerce_orders", [])
   557→            if isinstance(orders, list) and len(orders) > 0:
   558→                order = orders[0] if isinstance(orders[0], dict) else {}
   559→                raw_date = order.get("created_at", "") or order.get("order_date", "")
   560→                if raw_date:
   561→                    try:
   562→                        # Parse common date formats
   563→                        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%Y-%m-%d %H:%M:%S"):
   564→                            try:
   565→                                order_date = datetime.strptime(raw_date[:19], fmt).replace(tzinfo=timezone.utc)
   566→                                break
   567→                            except ValueError:
   568→                                continue
   569→                    except Exception:
   570→                        pass
   571→
   572→        if order_date:
   573→            days_since_order = (now - order_date).days
   574→            # Determine category for time window
   575→            category = "default"
   576→            if isinstance(crm_data, dict):
   577→                orders = crm_data.get("ecommerce_orders", [])
   578→                if isinstance(orders, list) and len(orders) > 0:
   579→                    order = orders[0] if isinstance(orders[0], dict) else {}
   580→                    product_type = str(order.get("product_type", "")).lower()
   581→                    if "digital" in product_type or "download" in product_type:
   582→                        category = "digital"
   583→                    elif "subscription" in product_type or "recurring" in product_type:
   584→                        category = "subscription"
   585→
   586→            window = _REFUND_TIME_WINDOWS.get(category, _REFUND_TIME_WINDOWS["default"])
   587→            if days_since_order > window:
   588→                context["blocks"].append(
   589→                    f"Order {days_since_order}d old — exceeds {category} refund window of {window}d"
   590→                )
   591→            else:
   592→                context["warnings"].append(
   593→                    f"Order {days_since_order}d old (within {window}d {category} window)"
   594→                )
   595→
   596→    # 2. Frequency check: too many actions for this customer recently?
   597→    freq_limits = _FREQUENCY_LIMITS.get(action)
   598→    if freq_limits:
   599→        recent_actions = state.get("recent_customer_actions", [])
   600→        if isinstance(recent_actions, list):
   601→            this_week = 0
   602→            this_month = 0
   603→            for ra in recent_actions:
   604→                if not isinstance(ra, dict) or ra.get("action") != action:
   605→                    continue
   606→                ra_time = ra.get("timestamp", 0)
   607→                if ra_time:
   608→                    try:
   609→                        ra_dt = datetime.fromtimestamp(ra_time, tz=timezone.utc)
   610→                        if (now - ra_dt).days <= 7:
   611→                            this_week += 1
   612→                        if (now - ra_dt).days <= 30:
   613→                            this_month += 1
   614→                    except Exception:
   615→                        pass
   616→
   617→            if this_week >= freq_limits["max_per_week"]:
   618→                context["blocks"].append(
   619→                    f"{this_week} {action} actions this week (limit: {freq_limits['max_per_week']})"
   620→                )
   621→            elif this_month >= freq_limits["max_per_month"]:
   622→                context["blocks"].append(
   623→                    f"{this_month} {action} actions this month (limit: {freq_limits['max_per_month']})"
   624→                )
   625→            else:
   626→                context["warnings"].append(
   627→                    f"{action} frequency: {this_week}/wk {this_month}/mo"
   628→                )
   629→
   630→    # 3. Category check: non-refundable product types
   631→    if action == "execute_refund":
   632→        if isinstance(crm_data, dict):
   633→            orders = crm_data.get("ecommerce_orders", [])
   634→            if isinstance(orders, list) and len(orders) > 0:
   635→                order = orders[0] if isinstance(orders[0], dict) else {}
   636→                product_category = str(order.get("product_category", "")).lower()
   637→                if product_category in _NON_REFUNDABLE_CATEGORIES:
   638→                    context["blocks"].append(
   639→                        f"Product category '{product_category}' is non-refundable"
   640→                    )
   641→
   642→    context["blocked"] = len(context["blocks"]) > 0
   643→    return context

   648→
   649→def _idempotency_check(action: str, ticket_id: str, state: Dict) -> Dict[str, Any]:
   650→    """Check if this exact action was already executed for this ticket.
   651→
   652→    Prevents double-refund, double-credit on pipeline retries or

   654→    prior-run technique_log entries.

   657→    actions_taken = state.get("actions_taken", [])

   660→            return {
   661→                "already_done": True,
   662→                "previous_status": prev.get("status", "unknown"),
   663→            }

   665→    # Check prior Node 5 runs (retry scenario — technique_log may have ReAct entries)
   666→    prior_log = state.get("technique_log", [])
   667→    if isinstance(prior_log, list):
   668→        for entry in prior_log:
   669→            if isinstance(entry, dict) and entry.get("node") == 5 and entry.get("technique") == "ReAct":
   670→                if action in entry.get("result_summary", ""):
   671→                    return {"already_done": True, "previous_status": "prior_run"}
   672→
   673→    return {"already_done": False}

   698→# ── L1: NearDedup — flag similar actions for same customer (non-LLM)

   700→
   701→def _near_dedup(action: str, details: Dict, crm_data: Dict, state: Dict) -> Dict[str, Any]:
   702→    """Check if a similar action was recently executed for same customer+order.
   703→
   704→    Catches cross-ticket double execution: same customer, same order,
   705→    different ticket number. Does NOT block — only flags for review.
   706→    """
   707→    customer_id = details.get("customer_id", "") or (crm_data.get("customer_id", "") if isinstance(crm_data, dict) else "")

   718→        if not isinstance(ra, dict):
   719→            continue
   720→        if ra.get("action") == action:

   722→                return {
   723→                    "duplicate_suspect": True,

   727→    return {"duplicate_suspect": False}

   730→# ── L2: NumericalConsistencyCheck — verify amounts vs CRM (non-LLM)
   731→
   732→
   733→def _numerical_consistency_check(action: str, details: Dict, crm_data: Dict) -> Dict[str, Any]:
   734→    """Cross-check action amount against CRM order data.
   735→
   736→    Prevents wrong-amount refunds: if the action says refund $500 but
   737→    the CRM order shows $200, something is wrong. Blocks the action
   738→    before it reaches Stripe.
   739→    """
   740→    amount = float(details.get("amount", 0))
   741→    if amount == 0 or action not in ("execute_refund", "execute_credit"):
   742→        return {"consistent": True, "reason": "No amount to verify or non-monetary action"}
   743→
   744→    # Try to find order amount from CRM
   745→    crm_amount = None
   746→    if isinstance(crm_data, dict):
   747→        orders = crm_data.get("ecommerce_orders", [])
   748→        if isinstance(orders, list) and len(orders) > 0:
   749→            order = orders[0] if isinstance(orders[0], dict) else {}
   750→            raw_amount = order.get("total_price", 0) or order.get("amount", 0)
   751→            if raw_amount:
   752→                crm_amount = float(raw_amount)
   753→
   754→    if crm_amount is None or crm_amount == 0:
   755→        return {"consistent": True, "reason": "No CRM amount to compare against"}
   756→
   757→    # Refund amount should not exceed order amount
   758→    if action == "execute_refund" and amount > crm_amount:
   759→        return {
   760→            "consistent": False,
   761→            "reason": f"Refund ${amount} exceeds order total ${crm_amount}",
   762→        }
   763→
   764→    # Credit amount should not exceed 2x order amount (generous buffer)
   765→    if action == "execute_credit" and amount > crm_amount * 2:
   766→        return {
   767→            "consistent": False,
   768→            "reason": f"Credit ${amount} is >2x order total ${crm_amount}",
   769→        }
   770→
   771→    return {"consistent": True, "reason": f"Amount ${amount} within expected range of ${crm_amount}"}

   774→# ── L2: SafetyNet — scrub PII from observations (non-LLM) ────────
   775→
   776→
   777→_PII_PATTERNS = [
   778→    re.compile(r'\b[\w.+-]+@[\w-]+\.[\w.-]+\b'),                           # email
   779→    re.compile(r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b'),                          # phone (US-like)
   780→    re.compile(r'\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b'),            # card number (16 digits)
   781→    re.compile(r'\b(?:SSN|social)\s*[:=]?\s*\d{3}-?\d{2}-?\d{4}\b', re.I), # SSN
   782→    re.compile(r'\b\d{16,19}\b'),                                           # long numeric IDs
   783→]
   784→
   785→
   786→def _safety_net_scrub(text: str) -> Dict[str, Any]:
   787→    """Scrub PII from text (observations, logs, verification output).
   788→
   789→    Catches email addresses, phone numbers, card numbers, SSNs, and
   790→    long numeric identifiers that could be payment cards. Replaces
   791→    each match with [REDACTED] to preserve readability.
   792→    """
   793→    if not text:
   794→        return {"scrubbed": text, "pii_found": False, "count": 0}
   795→
   796→    scrubbed = text
   797→    count = 0
   798→    for pattern in _PII_PATTERNS:
   799→        matches = pattern.findall(text)
   800→        if matches:
   801→            count += len(matches)
   802→            scrubbed = pattern.sub("[REDACTED]", scrubbed)
   803→
   804→    return {"scrubbed": scrubbed, "pii_found": count > 0, "count": count}

   807→# ── L3: ContradictionCheck — ReAct vs ReverseThinking (non-LLM) ──

   810→def _contradiction_check(react_result: Dict, reverse_result: Dict) -> Dict[str, Any]:
   811→    """Check if execution result contradicts verification result.
   812→
   813→    Catches: ReAct says 'completed' but Reverse says 'high risk' or
   814→    'not verified'. These disagreements mean the action needs human
   815→    review before shipping.
   816→    """
   817→    contradictions = []
   818→
   819→    if not react_result or not reverse_result:
   820→        return {"has_contradiction": False, "contradictions": []}
   821→
   822→    # ReAct succeeded but Reverse says not verified
   823→    if react_result.get("status") == "completed" and not reverse_result.get("verified"):
   824→        contradictions.append("Action completed but verification failed")

   826→    # ReAct failed but Reverse says verified (suspicious)
   827→    if react_result.get("status") not in ("completed", "recommended") and reverse_result.get("verified"):
   828→        contradictions.append("Action failed but verification passed — suspicious")
   829→
   830→    # ReAct success but Reverse says high risk
   831→    if react_result.get("status") == "completed" and reverse_result.get("risk") == "high":
   832→        contradictions.append("Action completed but high risk flagged")
   833→
   834→    return {"has_contradiction": len(contradictions) > 0, "contradictions": contradictions}

   837→# ── L3: Escalation — auto-escalate high risk (non-LLM) ────────────
   838→
   839→
   840→def _should_escalate(
   841→    action: str, reverse_result: Dict, zsv_result: Dict,
   842→    contradiction: Dict, tier: str,
   843→) -> Dict[str, Any]:
   844→    """Determine if this action should be escalated to human review.
   845→
   846→    Checks multiple signals: reverse verification risk, ZSV flags,
   847→    contradictions, and tier-specific rules. Returns escalate=True
   848→    with all reasons so the human knows WHY.
   849→    """
   850→    reasons = []
   851→
   852→    if not reverse_result:
   853→        pass  # No reverse result (skipped action)
   854→    else:
   855→        if reverse_result.get("risk") == "high":
   856→            reasons.append("Reverse verification: HIGH risk")
   857→        if reverse_result.get("verified") is False:
   858→            reasons.append("Reverse verification: NOT verified")
   859→
   860→    if zsv_result.get("flagged") and zsv_result.get("severity") == "high":
   861→        reasons.append(f"ZeroShotValidator high-severity: {zsv_result.get('flags', [])}")
   862→
   863→    if contradiction.get("has_contradiction"):
   864→        reasons.append(f"Contradiction: {contradiction.get('contradictions', [])}")
   865→
   866→    # Growth tier: always flag refunds >$400 (close to $500 cap)
   867→    if tier == "parwa" and action in ("execute_refund", "execute_credit"):
   868→        for flag in zsv_result.get("flags", []):
   869→            if "High-value" in str(flag):
   870→                reasons.append("High-value action on Growth tier — near limit")
   871→
   872→    return {"escalate": len(reasons) > 0, "reasons": reasons}

   875→# ── L4: SufficiencyCheck — did we solve the problem? (non-LLM) ────
   876→
   877→
   878→def _sufficiency_check(
   879→    action: str, actions_taken: List[Dict], original_query: str,
   880→) -> Dict[str, Any]:
   881→    """Check if the actions taken actually address the original problem.
   882→
   883→    Catches: action 'succeeded' but solved the WRONG problem, or
   884→    execution completed but no actual result was produced.
   885→    """
   886→    if not actions_taken:
   887→        return {"sufficient": False, "reason": "No actions were taken"}
   888→
   889→    # Map action types to expected outcome signals
   890→    expected_outcomes = {
   891→        "execute_refund": ["refund", "money back", "credited", "reimburse"],
   892→        "execute_credit": ["credit", "applied", "adjusted", "compensation"],
   893→        "account_change": ["updated", "changed", "modified", "confirmed"],
   894→        "provide_info": ["information", "provided", "answered", "explained"],
   895→    }
   896→
   897→    expected = expected_outcomes.get(action, [])
   898→
   899→    if not expected:
   900→        # No expected pattern — trust the status
   901→        any_completed = any(a.get("status") == "completed" for a in actions_taken if isinstance(a, dict))
   902→        return {
   903→            "sufficient": any_completed,
   904→            "reason": "No expected outcome pattern — using completion status",
   905→        }
   906→
   907→    # Check if any action result contains expected outcome signals
   908→    all_observations = " ".join(
   909→        (a.get("observation", "") + " " + a.get("thought", ""))
   910→        for a in actions_taken if isinstance(a, dict)
   911→    ).lower()
   912→
   913→    matched = [e for e in expected if e in all_observations]
   914→
   915→    if matched:
   916→        return {"sufficient": True, "reason": f"Expected outcomes found: {matched}"}
   917→
   918→    # Even without keyword match, completed status = sufficient
   919→    any_completed = any(a.get("status") == "completed" for a in actions_taken if isinstance(a, dict))
   920→    if any_completed:
   921→        return {"sufficient": True, "reason": "Action completed (status=completed)"}
   922→
   923→    return {"sufficient": False, "reason": f"Expected {expected} but no match in results"}

   926→# ── L4: MAKER FinalCheck + StrictMode (non-LLM) ──────────────────
   927→
   928→
   929→def _maker_final_check(
   930→    action: str, knowledge: str, bridge_result: str,
   931→    execution_ok: bool, verification_ok: bool,
   932→) -> Dict[str, Any]:
   933→    """Final safety gate: any critical gap → BLOCK the action.
   934→
   935→    StrictMode: 2+ critical gaps = hard block (no override possible).
   936→    Single gap = soft block (human override allowed).
   937→    This is the last check before the action result ships downstream.
   938→    """
   939→    critical_gaps = []
   940→
   941→    # No knowledge at all for this action
   942→    if not knowledge.strip():
   943→        critical_gaps.append("No knowledge base content available")
   944→
   945→    # Bridge found nothing relevant
   946→    if "No direct knowledge bridge" in bridge_result:
   947→        critical_gaps.append("MAKER bridge: no relevant knowledge found")
   948→
   949→    # Execution failed
   950→    if not execution_ok:
   951→        critical_gaps.append("Action execution did not succeed")
   952→
   953→    # Verification failed
   954→    if not verification_ok:
   955→        critical_gaps.append("Action verification did not pass")
   956→
   957→    should_block = len(critical_gaps) > 0
   958→    strict_mode = len(critical_gaps) >= 2  # 2+ gaps = strict block
   959→
   960→    return {
   961→        "should_block": should_block,
   962→        "strict_mode": strict_mode,
   963→        "critical_gaps": critical_gaps,
   964→    }

  1027→# ── L4: MetaLearner — adjust confidence from past patterns (non-LLM)
  1028→
  1029→
  1030→def _meta_learner_adjust(action: str, tier: str, state: Dict) -> Dict[str, Any]:
  1031→    """Adjust confidence based on past action success rates.
  1032→
  1033→    If refunds for this tenant have failed 60% of the time, reduce
  1034→    confidence. If they succeed 95% of the time, boost it. Uses
  1035→    technique_log history — no extra LLM call.
  1036→    """
  1037→    past_log = state.get("technique_log", [])
  1038→    if not isinstance(past_log, list):
  1039→        return {"confidence_adjustment": 0.0, "reason": "No past log data"}
  1040→
  1041→    action_type_count = 0
  1042→    action_success_count = 0
  1043→
  1044→    for entry in past_log:
  1045→        if not isinstance(entry, dict):
  1046→            continue
  1047→        if entry.get("technique") == "ReAct" and action in entry.get("result_summary", ""):
  1048→            action_type_count += 1
  1049→            if "status=completed" in entry.get("result_summary", ""):
  1050→                action_success_count += 1
  1051→
  1052→    if action_type_count == 0:
  1053→        return {"confidence_adjustment": 0.0, "reason": "No past data for this action"}
  1054→
  1055→    success_rate = action_success_count / action_type_count
  1056→

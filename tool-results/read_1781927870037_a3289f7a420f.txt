   315→    """Check text for emergency signals using keyword matching (FREE)."""

   317→    matched: Dict[str, List[str]] = {}

   320→        for keyword in keywords:
   321→            pattern = r'\b' + re.escape(keyword) + r'\b'

   323→                matched.setdefault(emergency_type, []).append(keyword)

   325→    if matched:
   326→        priority_order = ["safety", "legal_threat", "compliance", "media"]
   327→        for etype in priority_order:
   328→            if etype in matched:
   329→                return {
   330→                    "emergency_flag": True,
   331→                    "emergency_type": etype,
   332→                    "matched_keywords": matched,
   333→                }
   334→
   335→    return {
   336→        "emergency_flag": False,

   370→def _keyword_empathy_check(text: str) -> Dict[str, Any]:

   372→    text_lower = text.lower()
   373→    flags: List[str] = []
   374→    total_matches = 0
   375→
   376→    for flag_name, keywords in EMPATHY_PATTERNS.items():
   377→        for keyword in keywords:
   378→            if " " in keyword:
   379→                if keyword in text_lower:
   380→                    flags.append(flag_name)
   381→                    total_matches += 1
   382→                    break
   383→            else:
   384→                pattern = r'\b' + re.escape(keyword) + r'\b'
   385→                if re.search(pattern, text_lower):
   386→                    flags.append(flag_name)
   387→                    total_matches += 1
   388→                    break
   389→
   390→    if not flags:
   391→        empathy_score = 0.7
   392→    elif len(flags) == 1:
   393→        empathy_score = 0.4
   394→    elif len(flags) == 2:
   395→        empathy_score = 0.25
   396→    else:
   397→        empathy_score = 0.1
   398→
   399→    return {
   400→        "empathy_score": empathy_score,
   401→        "empathy_flags": flags,
   402→    }

   504→def _apply_crp_compression(text: str) -> Dict[str, Any]:
   505→    """Apply CRP (Concise Response Protocol) compression.
   506→
   507→    Target: 30-40% token reduction with >95% information retention.
   508→    """
   509→    if not text:
   510→        return {
   511→            "compressed_text": text,
   512→            "tokens_removed": 0,
   513→            "compression_ratio": 1.0,
   514→            "phrases_removed": [],
   515→        }
   516→
   517→    original_tokens = len(text.split())
   518→    compressed = text
   519→    phrases_removed: List[str] = []
   520→
   521→    for phrase in CRP_FILLER_PHRASES:
   522→        if phrase.lower() in compressed.lower():
   523→            pattern = re.compile(re.escape(phrase), re.IGNORECASE)
   524→            compressed = pattern.sub("", compressed)
   525→            phrases_removed.append(phrase)
   526→
   527→    compressed = re.sub(r'\s{2,}', ' ', compressed)
   528→    compressed = re.sub(r'\.\s*\.', '.', compressed)
   529→    compressed = re.sub(r'^\s+|\s+$', '', compressed, flags=re.MULTILINE)
   530→    compressed = compressed.strip()
   531→
   532→    thank_you_count = len(re.findall(r'\bthank you\b', compressed, re.IGNORECASE))
   533→    if thank_you_count > 1:
   534→        first = True
   535→        parts = re.split(r'(\bThank you\b|\bthank you\b)', compressed, flags=re.IGNORECASE)
   536→        result_parts = []
   537→        for part in parts:
   538→            if re.match(r'\bthank you\b', part, re.IGNORECASE):
   539→                if first:
   540→                    result_parts.append(part)
   541→                    first = False
   542→                else:
   543→                    phrases_removed.append("duplicate thank you")
   544→            else:
   545→                result_parts.append(part)
   546→        compressed = "".join(result_parts)
   547→
   548→    final_tokens = len(compressed.split())
   549→    tokens_removed = max(0, original_tokens - final_tokens)
   550→    compression_ratio = final_tokens / original_tokens if original_tokens > 0 else 1.0
   551→
   552→    return {
   553→        "compressed_text": compressed,
   554→        "tokens_removed": tokens_removed,
   555→        "compression_ratio": round(compression_ratio, 3),
   556→        "phrases_removed": phrases_removed,
   557→    }

   569→    empathy_score: float,
   570→    reasoning_output: str = "",
   571→    technique_used: str = "",
   572→    signals: Optional[Dict[str, Any]] = None,
   573→) -> Dict[str, Any]:
   574→    """Run the CLARA quality gate pipeline (High: threshold 95, 8-check — strictest).
   575→
   576→    CLARA = Concise Logical Adaptive Response Architecture.
   577→    Validates: Structure -> Logic -> Brand -> Tone -> Delivery
   578→               -> Reasoning Alignment -> Consistency -> Completeness
   579→
   580→    High version: Highest threshold (95), 8 checks (strictest).
   581→    """
   582→    if not response:
   583→        return {
   584→            "passed": False,
   585→            "score": 0.0,
   586→            "issues": ["empty_response"],
   587→            "adjusted_response": response,
   588→            "checks": {},
   589→        }
   590→
   591→    checks: Dict[str, Dict[str, Any]] = {}
   592→    issues: List[str] = []
   593→    score = 100.0
   594→
   595→    # Check 1: STRUCTURE
   596→    has_greeting = bool(re.search(
   597→        r'\b(thank you|hello|hi|dear|greetings)\b',
   598→        response, re.IGNORECASE,
   599→    ))
   600→    has_acknowledgment = bool(re.search(
   601→        r'\b(understand|sorry|apologize|acknowledge|we see|I see)\b',
   602→        response, re.IGNORECASE,
   603→    ))
   604→    has_action = bool(re.search(
   605→        r'\b(will|can|let me|our team|we\'ll|we will|I\'ll|I will)\b',
   606→        response, re.IGNORECASE,
   607→    ))
   608→    has_steps = bool(re.search(
   609→        r'\b(first|then|next|finally|step \d)\b',
   610→        response, re.IGNORECASE,
   611→    ))
   612→
   613→    structure_score = 0
   614→    if has_greeting or has_acknowledgment:
   615→        structure_score += 25
   616→    if has_action:
   617→        structure_score += 25
   618→    if has_steps:
   619→        structure_score += 25
   620→    if len(response.split('.')) >= 3:
   621→        structure_score += 25
   622→
   623→    if structure_score < 50:
   624→        issues.append("poor_structure")
   625→        score -= 20
   626→
   627→    checks["structure"] = {
   628→        "score": structure_score,
   629→        "has_greeting": has_greeting,
   630→        "has_acknowledgment": has_acknowledgment,
   631→        "has_action": has_action,
   632→        "has_steps": has_steps,
   633→    }
   634→
   635→    # Check 2: LOGIC
   636→    query_tokens = set(re.findall(r'\b\w{4,}\b', query.lower()))
   637→    response_tokens = set(re.findall(r'\b\w{4,}\b', response.lower()))
   638→    stop_words = {
   639→        "that", "this", "with", "have", "will", "been", "from",
   640→        "they", "would", "could", "should", "there", "their",
   641→        "about", "which", "when", "where", "your", "please",
   642→    }
   643→    query_content = query_tokens - stop_words
   644→    overlap = query_content & (response_tokens - stop_words) if query_content else set()
   645→    logic_score = int(len(overlap) / max(len(query_content), 1) * 100)
   646→
   647→    if logic_score < 30:
   648→        issues.append("off_topic")
   649→        score -= 25
   650→    elif logic_score < 50:
   651→        issues.append("partial_topic_coverage")
   652→        score -= 15
   653→
   654→    checks["logic"] = {
   655→        "score": logic_score,
   656→        "overlap_count": len(overlap),
   657→    }
   658→
   659→    # Check 3: BRAND
   660→    inappropriate_words = {
   661→        "dude", "bro", "lol", "lmao", "rofl", "idk", "tbh",
   662→        "ngl", "smh", "bruh", "yolo",
   663→    }
   664→    response_lower = response.lower()
   665→    brand_violations = [w for w in inappropriate_words if w in response_lower.split()]
   666→    brand_score = 100 - (len(brand_violations) * 25)
   667→
   668→    if brand_violations:
   669→        issues.append("brand_violation")
   670→        score -= len(brand_violations) * 15
   671→
   672→    checks["brand"] = {
   673→        "score": max(0, brand_score),
   674→        "violations": brand_violations,
   675→    }
   676→
   677→    # Check 4: TONE
   678→    tone_score = 80
   679→    if empathy_score < 0.3 and not has_acknowledgment:
   680→        tone_score = 30
   681→        issues.append("insufficient_empathy")
   682→        score -= 20
   683→    elif empathy_score < 0.5 and not has_acknowledgment:
   684→        tone_score = 55
   685→        issues.append("moderate_empathy_gap")
   686→        score -= 10
   687→    elif empathy_score > 0.6:
   688→        tone_score = 90
   689→
   690→    checks["tone"] = {
   691→        "score": tone_score,
   692→        "empathy_score": empathy_score,
   693→        "has_acknowledgment": has_acknowledgment,
   694→    }
   695→
   696→    # Check 5: DELIVERY
   697→    delivery_score = 80
   698→    if len(response.strip()) < 30:
   699→        delivery_score = 25
   700→        issues.append("response_too_short")
   701→        score -= 25
   702→    elif len(response.strip()) < 50:
   703→        delivery_score = 50
   704→        issues.append("response_brief")
   705→        score -= 10
   706→    elif len(response.strip()) > 2000:
   707→        delivery_score = 65
   708→        issues.append("response_verbose")
   709→        score -= 5
   710→
   711→    placeholders = re.findall(r'\{+\w+\}+', response)
   712→    if placeholders:
   713→        issues.append("unresolved_placeholders")
   714→        score -= 10
   715→        delivery_score -= 20
   716→
   717→    checks["delivery"] = {
   718→        "score": delivery_score,
   719→        "response_length": len(response),
   720→        "has_placeholders": bool(placeholders),
   721→    }
   722→
   723→    # Check 6: REASONING ALIGNMENT
   724→    reasoning_score = 70
   725→    if reasoning_output and technique_used:
   726→        reasoning_tokens = set(re.findall(r'\b\w{4,}\b', reasoning_output.lower()))
   727→        resp_tokens_r = set(re.findall(r'\b\w{4,}\b', response.lower()))
   728→        reasoning_overlap = reasoning_tokens & resp_tokens_r
   729→        if len(reasoning_overlap) > 3:
   730→            reasoning_score = 90
   731→        elif len(reasoning_overlap) > 0:
   732→            reasoning_score = 70
   733→        else:
   734→            reasoning_score = 40
   735→            issues.append("reasoning_not_reflected")
   736→            score -= 15
   737→
   738→    checks["reasoning_alignment"] = {
   739→        "score": reasoning_score,
   740→        "technique_used": technique_used,
   741→    }
   742→
   743→    # Check 7: CONSISTENCY (High-specific)
   744→    consistency_score = 85
   745→    # Check for contradiction markers
   746→    contradiction_patterns = [
   747→        (r'\byes\b.*\bno\b', "yes_no_contradiction"),
   748→        (r'\bwill\b.*\bcannot\b', "will_cannot_contradiction"),
   749→        (r'\bavailable\b.*\bunavailable\b', "availability_contradiction"),
   750→    ]
   751→    for pattern, issue_name in contradiction_patterns:
   752→        if re.search(pattern, response, re.IGNORECASE):
   753→            consistency_score -= 20
   754→            issues.append(issue_name)
   755→            score -= 10
   756→            break
   757→
   758→    checks["consistency"] = {
   759→        "score": consistency_score,
   760→    }
   761→
   762→    # Check 8: COMPLETENESS (High-specific)
   763→    completeness_score = 70
   764→    query_aspects = query.split(" and ")
   765→    if len(query_aspects) > 1:
   766→        addressed = sum(
   767→            1 for aspect in query_aspects
   768→            if any(w in response_lower for w in aspect.lower().split() if len(w) > 3)
   769→        )
   770→        completeness_score = int(addressed / len(query_aspects) * 100)
   771→        if completeness_score < 50:
   772→            issues.append("incomplete_address")
   773→            score -= 15
   774→
   775→    # Check if response has a closing/follow-up
   776→    has_closing = bool(re.search(
   777→        r'\b(let me know|feel free|don\'t hesitate|reach out|contact us)\b',
   778→        response, re.IGNORECASE,
   779→    ))
   780→    if not has_closing:
   781→        completeness_score -= 10
   782→
   783→    checks["completeness"] = {
   784→        "score": completeness_score,
   785→        "has_closing": has_closing,
   786→    }
   787→
   788→    # Compute final score
   789→    final_score = max(0.0, min(100.0, score))
   790→    passed = final_score >= 75.0  # High threshold: 75 (agentic-ready)
   791→
   792→    adjusted_response = response
   793→    if issues and "unresolved_placeholders" in issues:
   794→        adjusted_response = re.sub(r'\{+\w+\}+', '', adjusted_response)
   795→        adjusted_response = re.sub(r'\s{2,}', ' ', adjusted_response).strip()
   796→
   797→    return {
   798→        "passed": passed,
   799→        "score": round(final_score, 2),
   800→        "issues": issues,
   801→        "adjusted_response": adjusted_response,
   802→        "checks": checks,
   803→    }

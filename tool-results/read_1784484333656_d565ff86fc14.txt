   387→    # ── Gap 6: Auto-resume pending escalations every 5 minutes ──

   394→
   395→    async def _auto_resume_escalations_loop():
   396→        # Wait 60s after startup before first run (let other services init)
   397→        await asyncio.sleep(60)
   398→        while True:
   399→            try:
   400→                from database.base import SessionLocal
   401→                from database.models.core import Company
   402→                from app.core.escalation_vault.resume_pipeline import auto_resume_pending
   403→
   404→                db = SessionLocal()
   405→                try:
   406→                    companies = db.query(Company).filter(
   407→                        Company.subscription_status == "active",
   408→                    ).all()
   409→                    total_resumed = 0
   410→                    for company in companies:
   411→                        try:
   412→                            result = await auto_resume_pending(str(company.id))
   413→                            resumed = result.get("resolved", 0)
   414→                            if resumed > 0:
   415→                                total_resumed += resumed
   416→                                logger.info(
   417→                                    "auto_resume_loop: company=%s resumed=%d",
   418→                                    str(company.id)[:8], resumed,
   419→                                )
   420→                        except Exception as company_exc:
   421→                            logger.warning(
   422→                                "auto_resume_company_failed company=%s err=%s",
   423→                                str(company.id)[:8], str(company_exc)[:200],
   424→                            )
   425→                    if total_resumed > 0:
   426→                        logger.info(
   427→                            "auto_resume_loop_complete: total_resumed=%d companies=%d",
   428→                            total_resumed, len(companies),
   429→                        )
   430→                finally:
   431→                    db.close()
   432→            except Exception as loop_exc:
   433→                logger.warning(
   434→                    "auto_resume_loop_error err=%s",
   435→                    str(loop_exc)[:200],
   436→                )
   437→            # Sleep 5 minutes before next run
   438→            await asyncio.sleep(300)
   439→
   440→    asyncio.create_task(_auto_resume_escalations_loop())
   441→    logger.info("auto_resume_escalations_loop_started (runs every 5 min)")

   454→    # ── Stuck-ticket recovery loop ─────────────────────────────────
   455→    # If a ticket's background dispatch thread fails to start (crash, OOM,
   456→    # server restart), the ticket stays "open" forever with no AI response.
   457→    # This loop checks every 3 minutes for tickets stuck in "open" status
   458→    # for more than 15 minutes and re-dispatches them.
   459→    #
   460→    # IMPORTANT: The cutoff is 15 minutes (not 5) because with 7 concurrent
   461→    # pipelines, a queue of 25 tickets takes ~4 minutes to drain. A 5-min
   462→    # cutoff would re-dispatch tickets that are just waiting in the queue,
   463→    # creating duplicate threads and causing OOM crashes.
   464→    async def _stuck_ticket_recovery_loop():
   465→        await asyncio.sleep(180)  # Wait 3 min after startup
   466→        while True:
   467→            try:
   468→                from database.base import SessionLocal
   469→                from database.models.tickets import Ticket, TicketMessage
   470→                from datetime import datetime, timezone, timedelta
   471→                from app.services.pipeline_dispatcher import dispatch_pipeline_for_ticket
   472→
   473→                db = SessionLocal()
   474→                try:
   475→                    # Re-dispatch tickets stuck in 'open' for >15 min
   476→                    # (truly orphaned — never picked up by workers)
   477→                    cutoff = datetime.now(timezone.utc) - timedelta(minutes=15)
   478→                    stuck_tickets = db.query(Ticket).filter(
   479→                        Ticket.status == "open",
   480→                        Ticket.created_at < cutoff,
   481→                    ).limit(5).all()
   482→
   483→                    # Also re-dispatch tickets stuck in 'processing' for >5 min
   484→                    # (worker crashed mid-pipeline — reset to open so workers
   485→                    # pick them up again)
   486→                    processing_cutoff = datetime.now(timezone.utc) - timedelta(minutes=5)
   487→                    stuck_processing = db.query(Ticket).filter(
   488→                        Ticket.status == "processing",
   489→                        Ticket.updated_at < processing_cutoff,
   490→                    ).limit(5).all()
   491→
   492→                    all_stuck = stuck_tickets + stuck_processing
   493→
   494→                    if all_stuck:
   495→                        logger.info(
   496→                            "stuck_ticket_recovery: %d open + %d processing stuck",
   497→                            len(stuck_tickets), len(stuck_processing),
   498→                        )
   499→
   500→                    for ticket in all_stuck:
   501→                        try:
   502→                            # If ticket is stuck in 'processing', reset to 'open'
   503→                            # so the DB-backed workers pick it up again
   504→                            if ticket.status == "processing":
   505→                                ticket.status = "open"
   506→                                db.commit()
   507→                                logger.info(
   508→                                    "stuck_ticket_recovery: reset processing→open ticket %s",
   509→                                    str(ticket.id)[:8],
   510→                                )
   511→                                continue
   512→
   513→                            # For 'open' tickets: check if AI message exists
   514→                            has_ai = db.query(TicketMessage).filter(
   515→                                TicketMessage.ticket_id == str(ticket.id),
   516→                                TicketMessage.role == "ai",
   517→                            ).first()
   518→
   519→                            if has_ai:
   520→                                # Pipeline ran but didn't update status — mark as awaiting_human
   521→                                ticket.status = "awaiting_human"
   522→                                ticket.awaiting_human = True
   523→                                db.commit()
   524→                                logger.info(
   525→                                    "stuck_ticket_recovery: ticket %s had AI msg, marking awaiting_human",
   526→                                    str(ticket.id)[:8],
   527→                                )
   528→                            else:
   529→                                # No AI message — pipeline never ran. Re-dispatch.
   530→                                logger.info(
   531→                                    "stuck_ticket_recovery: re-dispatching ticket %s",
   532→                                    str(ticket.id)[:8],
   533→                                )
   534→                                dispatch_pipeline_for_ticket(
   535→                                    ticket_id=str(ticket.id),
   536→                                    company_id=ticket.company_id,
   537→                                    priority=ticket.priority or "medium",
   538→                                    channel="email",
   539→                                    sync=True,
   540→                                )
   541→                        except Exception as ticket_exc:
   542→                            logger.warning(
   543→                                "stuck_ticket_recovery: failed for ticket %s: %s",
   544→                                str(ticket.id)[:8], str(ticket_exc)[:150],
   545→                            )
   546→                finally:
   547→                    db.close()
   548→            except Exception as recovery_exc:
   549→                logger.warning(
   550→                    "stuck_ticket_recovery_loop_error: %s",
   551→                    str(recovery_exc)[:200],
   552→                )
   553→            # Check every 3 minutes (was 2 — increased to reduce thread pile-up)
   554→            await asyncio.sleep(180)
   555→
   556→    asyncio.create_task(_stuck_ticket_recovery_loop())
   557→    logger.info("stuck_ticket_recovery_loop_started (checks every 3 min, cutoff 15 min)")
   558→
   559→    yield
   560→
   561→    # Shutdown: flush Sentry events
   562→    try:
   563→        from app.core.sentry import flush as sentry_flush
   564→        sentry_flush(timeout=2.0)
   565→    except Exception as exc:
   566→        logger.warning("sentry_flush_error", error=str(exc))
   567→
   568→    # Shutdown: close Redis pool
   569→    try:
   570→        from app.core.redis import close_redis
   571→        await close_redis()
   572→        logger.info("redis_closed")
   573→    except Exception as exc:
   574→        logger.warning("redis_close_error", error=str(exc))
   575→
   576→    logger.info("parwa_shutdown")
   577→

   873→@app.get("/api/events/since", tags=["Events"])
   874→async def get_events_since_endpoint(
   875→    request: Request,
   876→    last_seen: float = Query(
   877→        ..., description="Epoch timestamp of last received event"
   878→    ),
   879→    # M-08 FIX: Require explicit authentication on the events endpoint.
   880→    # Previously relied only on middleware-level tenant scoping, allowing
   881→    # unauthenticated requests to reach the handler.
   882→    current_user: User = Depends(get_current_user),
   883→):
   884→    """Fetch events missed during disconnection (BC-005).
   885→
   886→    On reconnect, the client calls this endpoint with their last_seen
   887→    timestamp to fetch all events that occurred while disconnected.
   888→
   889→    BC-001: Events are scoped to the requesting tenant.
   890→    BC-005: Event buffer stores events for 24 hours.
   891→    M-08: Requires explicit JWT authentication.
   892→    """
   893→    company_id = getattr(request.state, "company_id", None)
   894→    if not company_id:
   895→        return JSONResponse(
   896→            status_code=403,
   897→            content={
   898→                "error": {
   899→                    "code": "AUTHORIZATION_ERROR",
   900→                    "message": "Tenant identification required",
   901→                    "details": None,
   902→                }
   903→            },
   904→        )
   905→
   906→    from app.core.event_buffer import get_events_since
   907→    events = await get_events_since(
   908→        company_id=company_id,
   909→        last_seen=last_seen,
   910→    )
   911→
   912→    return {
   913→        "events": events,
   914→        "count": len(events),
   915→        "last_seen": last_seen,
   916→    }

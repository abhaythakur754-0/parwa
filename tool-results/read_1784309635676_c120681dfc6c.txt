    87→interface Installment {
    88→  number: number;
    89→  amount: number;
    90→  status: 'pending' | 'processing' | 'paid' | 'failed' | 'skipped';
    91→  processed_at?: string;
    92→  is_extra?: boolean;

    94→
    95→interface FlexPlanStatus {
    96→  plan_id: string;
    97→  variant_tier: string;
    98→  status: 'pending' | 'active' | 'paused' | 'completed' | 'cancelled' | 'failed';
    99→  total_amount: number;
   100→  collected_amount: number;
   101→  remaining_amount: number;
   102→  progress_percent: number;
   103→  installments: {
   104→    total: number;
   105→    completed: number;
   106→    pending: number;
   107→    failed: number;
   108→  };
   109→  period: {
   110→    start: string | null;
   111→    end: string | null;
   112→  };
   113→  failure_info: {
   114→    consecutive_failures: number;
   115→    last_reason: string | null;
   116→    last_failure_at: string | null;
   117→  } | null;
   118→  recent_installments: Installment[];
   119→}

   122→// ─── Helper Functions ─────────────────────────────────────────────
   123→
   124→function formatCurrency(amount: number): string {
   125→  return new Intl.NumberFormat('en-US', {
   126→    style: 'currency',
   127→    currency: 'USD',
   128→    minimumFractionDigits: 2,
   129→  }).format(amount);
   130→}
   131→
   132→function formatDate(dateStr: string): string {
   133→  if (!dateStr) return 'N/A';
   134→  const date = new Date(dateStr);
   135→  return date.toLocaleDateString('en-US', {
   136→    month: 'short',
   137→    day: 'numeric',
   138→    hour: '2-digit',
   139→    minute: '2-digit',
   140→  });
   141→}
   142→
   143→function getStatusColor(status: string): string {
   144→  switch (status) {
   145→    case 'active':
   146→      return 'bg-green-100 text-green-800 border-green-200';
   147→    case 'paused':
   148→      return 'bg-yellow-100 text-yellow-800 border-yellow-200';
   149→    case 'completed':
   150→      return 'bg-blue-100 text-blue-800 border-blue-200';
   151→    case 'cancelled':
   152→      return 'bg-gray-100 text-gray-800 border-gray-200';
   153→    case 'failed':
   154→      return 'bg-red-100 text-red-800 border-red-200';
   155→    case 'pending':
   156→      return 'bg-purple-100 text-purple-800 border-purple-200';
   157→    default:
   158→      return 'bg-gray-100 text-gray-800 border-gray-200';
   159→  }
   160→}
   161→
   162→function getInstallmentStatusIcon(status: string) {
   163→  switch (status) {
   164→    case 'paid':
   165→      return <CheckCircleIcon className="h-4 w-4 text-green-600" />;
   166→    case 'failed':
   167→      return <XCircleIcon className="h-4 w-4 text-red-600" />;
   168→    case 'processing':
   169→      return <ClockIcon className="h-4 w-4 text-blue-600 animate-spin" />;
   170→    default:
   171→      return <ClockIcon className="h-4 w-4 text-gray-400" />;
   172→  }
   173→}
   174→
   175→function getVariantDisplayName(variant: string): string {
   176→  const names: Record<string, string> = {
   177→    mini: 'Mini PARWA',
   178→    parwa: 'PARWA',
   179→    high: 'PARWA High',
   180→  };
   181→  return names[variant] || variant.toUpperCase();
   182→}

   185→// ─── Main Component ───────────────────────────────────────────────
   186→
   187→interface FlexPayDashboardProps {
   188→  /** The FlexPay plan ID to display */
   189→  planId?: string;
   190→  /** Optional callback when action is taken */
   191→  onAction?: (action: string, data?: any) => void;
   192→  /** Whether to show admin controls (pause/resume/cancel) */
   193→  showControls?: boolean;
   194→  /** Auto-refresh interval in milliseconds (0 = no auto-refresh) */
   195→  refreshInterval?: number;
   196→}
   197→
   198→export function FlexPayDashboard({
   199→  planId,
   200→  onAction,
   201→  showControls = true,
   202→  refreshInterval = 30000, // Refresh every 30 seconds by default
   203→}: FlexPayDashboardProps) {
   204→  // State
   205→  const [planStatus, setPlanStatus] = useState<FlexPlanStatus | null>(null);
   206→  const [loading, setLoading] = useState(true);
   207→  const [error, setError] = useState<string | null>(null);
   208→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   209→  
   210→  // Dialog states
   211→  const [cancelDialogOpen, setCancelDialogOpen] = useState(false);
   212→  const [pauseDialogOpen, setPauseDialogOpen] = useState(false);
   213→  const [resumeDialogOpen, setResumeDialogOpen] = useState(false);
   214→  
   215→  // Fetch plan status
   216→  const fetchStatus = useCallback(async () => {
   217→    if (!planId) {
   218→      setLoading(false);
   219→      return;
   220→    }
   221→
   222→    try {
   223→      setError(null);
   224→      const response = await fetch(`/api/flexpay/${planId}/status`, {
   225→        headers: {
   226→          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
   227→        },
   228→      });
   229→
   230→      if (!response.ok) {
   231→        throw new Error(`Failed to fetch status: ${response.statusText}`);
   232→      }
   233→
   234→      const data: FlexPlanStatus = await response.json();
   235→      setPlanStatus(data);
   236→    } catch (err) {
   237→      setError(err instanceof Error ? err.message : 'Failed to load plan status');
   238→    } finally {
   239→      setLoading(false);
   240→    }
   241→  }, [planId]);
   242→
   243→  // Initial fetch and auto-refresh
   244→  useEffect(() => {
   245→    fetchStatus();
   246→
   247→    if (refreshInterval > 0) {
   248→      const interval = setInterval(fetchStatus, refreshInterval);
   249→      return () => clearInterval(interval);
   250→    }
   251→  }, [fetchStatus, refreshInterval]);
   252→
   253→  // Action handlers
   254→  const handleAction = async (action: string, payload?: any) => {
   255→    if (!planId) return;
   256→
   257→    setActionLoading(action);
   258→    try {
   259→      const response = await fetch(`/api/flexpay/${planId}/${action}`, {
   260→        method: 'POST',
   261→        headers: {
   262→          'Content-Type': 'application/json',
   263→          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
   264→        },
   265→        body: JSON.stringify(payload || {}),
   266→      });
   267→
   268→      if (!response.ok) {
   269→        throw new Error(`Action failed: ${response.statusText}`);
   270→      }
   271→
   272→      // Refresh status after action
   273→      await fetchStatus();
   274→      
   275→      // Close dialogs
   276→      setCancelDialogOpen(false);
   277→      setPauseDialogOpen(false);
   278→      setResumeDialogOpen(false);
   279→
   280→      // Notify parent
   281→      onAction?.(action, payload);
   282→    } catch (err) {
   283→      setError(err instanceof Error ? err.message : `Failed to ${action}`);
   284→    } finally {
   285→      setActionLoading(null);
   286→    }
   287→  };
   288→
   289→  // Loading state
   290→  if (loading) {
   291→    return (
   292→      <Card className="w-full max-w-4xl mx-auto">
   293→        <CardContent className="p-8">
   294→          <div className="flex items-center justify-center space-x-4">
   295→            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-primary"></div>
   296→            <span className="text-muted-foreground">Loading your FlexPay progress...</span>
   297→          </div>
   298→        </CardContent>
   299→      </Card>
   300→    );
   301→  }
   302→
   303→  // Error state
   304→  if (error && !planStatus) {
   305→    return (
   306→      <Card className="w-full max-w-4xl mx-auto border-destructive">
   307→        <CardContent className="p-6">
   308→          <div className="flex items-center space-x-3 text-destructive">
   309→            <AlertTriangleIcon className="h-6 w-6" />
   310→            <div>
   311→              <h3 className="font-semibold">Unable to Load FlexPay Data</h3>
   312→              <p className="text-sm mt-1">{error}</p>
   313→              <Button variant="outline" size="sm" className="mt-3" onClick={fetchStatus}>
   314→                Try Again
   315→              </Button>
   316→            </div>
   317→          </div>
   318→        </CardContent>
   319→      </Card>
   320→    );
   321→  }
   322→
   323→  // No plan selected
   324→  if (!planStatus) {
   325→    return (
   326→      <Card className="w-full max-w-4xl mx-auto">
   327→        <CardContent className="p-8 text-center">
   328→          <ClockIcon className="h-12 w-12 mx-auto text-muted-foreground mb-4" />
   329→          <h3 className="text-lg font-semibold mb-2">No Active FlexPay Plan</h3>
   330→          <p className="text-muted-foreground">
   331→            Select a plan or create a new FlexPay installment plan to get started.
   332→          </p>
   333→        </CardContent>
   334→      </Card>
   335→    );
   336→  }
   337→
   338→  // Calculate day info
   339→  const currentDay = planStatus.installments.completed + 1;
   340→  const totalDays = Math.ceil(planStatus.installments.total / 1.33); // Approximate (accounts for extra charges)
   341→
   342→  return (
   343→    <div className="w-full max-w-4xl mx-auto space-y-6">
   344→      {/* Header Card */}
   345→      <Card>
   346→        <CardHeader className="pb-4">
   347→          <div className="flex items-start justify-between">
   348→            <div className="space-y-1">
   349→              <CardTitle className="text-2xl font-bold flex items-center gap-3">
   350→                FlexPay Dashboard
   351→                <Badge variant="outline" className={getStatusColor(planStatus.status)}>
   352→                  {planStatus.status.charAt(0).toUpperCase() + planStatus.status.slice(1)}
   353→                </Badge>
   354→              </CardTitle>
   355→              <CardDescription className="text-base">
   356→                {getVariantDisplayName(planStatus.variant_tier)} • {formatCurrency(planStatus.total_amount)} total
   357→              </CardDescription>
   358→            </div>
   359→            
   360→            {showControls && (
   361→              <div className="flex gap-2">
   362→                {(planStatus.status === 'active' || planStatus.status === 'pending') && (
   363→                  <Button
   364→                    variant="outline"
   365→                    size="sm"
   366→                    onClick={() => setPauseDialogOpen(true)}
   367→                    disabled={!!actionLoading}
   368→                  >
   369→                    <PauseIcon className="h-4 w-4 mr-2" />
   370→                    Pause
   371→                  </Button>
   372→                )}
   373→                
   374→                {planStatus.status === 'paused' && (
   375→                  <Button
   376→                    variant="outline"
   377→                    size="sm"
   378→                    onClick={() => setResumeDialogOpen(true)}
   379→                    disabled={!!actionLoading}
   380→                  >
   381→                    <PlayIcon className="h-4 w-4 mr-2" />
   382→                    Resume
   383→                  </Button>
   384→                )}
   385→                
   386→                {!['completed', 'cancelled'].includes(planStatus.status) && (
   387→                  <Dialog open={cancelDialogOpen} onOpenChange={setCancelDialogOpen}>
   388→                    <DialogTrigger asChild>
   389→                      <Button variant="destructive" size="sm" disabled={!!actionLoading}>
   390→                        Cancel Plan
   391→                      </Button>
   392→                    </DialogTrigger>
   393→                    <DialogContent>
   394→                      <DialogHeader>
   395→                        <DialogTitle>Cancel FlexPay Plan?</DialogTitle>
   396→                        <DialogDescription>
   397→                          This will stop all future payments. You'll keep access for the portion you've already paid ({formatCurrency(planStatus.collected_amount)}).
   398→                          <br /><br />
   399→                          <strong>This action cannot be undone.</strong>
   400→                        </DialogDescription>
   401→                      </DialogHeader>
   402→                      <DialogFooter>
   403→                        <Button variant="outline" onClick={() => setCancelDialogOpen(false)}>
   404→                          Keep Plan
   405→                        </Button>
   406→                        <Button
   407→                          variant="destructive"
   408→                          onClick={() => handleAction('cancel', { reason: 'Customer requested cancellation' })}
   409→                          disabled={actionLoading === 'cancel'}
   410→                        >
   411→                          {actionLoading === 'cancel' ? 'Cancelling...' : 'Yes, Cancel Plan'}
   412→                        </Button>
   413→                      </DialogFooter>
   414→                    </DialogContent>
   415→                  </Dialog>
   416→                )}
   417→              </div>
   418→            )}
   419→          </div>
   420→        </CardHeader>
   421→
   422→        <CardContent className="space-y-6">
   423→          {/* Day X of Y Progress */}
   424→          <div className="space-y-3">
   425→            <div className="flex items-baseline justify-between">
   426→              <span className="text-sm font-medium text-muted-foreground">Payment Progress</span>
   427→              <span className="text-3xl font-bold text-primary">
   428→                Day {Math.min(currentDay, totalDays)} of {totalDays}
   429→              </span>
   430→            </div>
   431→            
   432→            <Progress value={planStatus.progress_percent} className="h-3" />
   433→            
   434→            <div className="flex justify-between text-sm text-muted-foreground">
   435→              <span>{formatCurrency(planStatus.collected_amount)} collected</span>
   436→              <span>{planStatus.progress_percent.toFixed(1)}% complete</span>
   437→              <span>{formatCurrency(planStatus.remaining_amount)} remaining</span>
   438→            </div>
   439→          </div>
   440→
   441→          <Separator />
   442→
   443→          {/* Quick Stats Grid */}
   444→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   445→            <div className="text-center p-3 bg-green-50 dark:bg-green-950 rounded-lg">
   446→              <div className="text-2xl font-bold text-green-700 dark:text-green-300">
   447→                {planStatus.installments.completed}
   448→              </div>
   449→              <div className="text-xs text-green-600 dark:text-green-400 mt-1">Paid</div>
   450→            </div>
   451→            
   452→            <div className="text-center p-3 bg-blue-50 dark:bg-blue-950 rounded-lg">
   453→              <div className="text-2xl font-bold text-blue-700 dark:text-blue-300">
   454→                {planStatus.installments.pending}
   455→              </div>
   456→              <div className="text-xs text-blue-600 dark:text-blue-400 mt-1">Pending</div>
   457→            </div>
   458→            
   459→            <div className="text-center p-3 bg-yellow-50 dark:bg-yellow-950 rounded-lg">
   460→              <div className="text-2xl font-bold text-yellow-700 dark:text-yellow-300">
   461→                {formatCurrency(planStatus.total_amount / totalDays)}
   462→              </div>
   463→              <div className="text-xs text-yellow-600 dark:text-yellow-400 mt-1">Daily Avg</div>
   464→            </div>
   465→            
   466→            <div className="text-center p-3 bg-purple-50 dark:bg-purple-950 rounded-lg">
   467→              <div className="text-2xl font-bold text-purple-700 dark:text-purple-300">
   468→                {planStatus.installments.failed}
   469→              </div>
   470→              <div className="text-xs text-purple-600 dark:text-purple-400 mt-1">Failed</div>
   471→            </div>
   472→          </div>
   473→
   474→          {/* Failure Warning */}
   475→          {planStatus.failure_info && planStatus.failure_info.consecutive_failures > 0 && (
   476→            <Alert variant="destructive">
   477→              <AlertTriangleIcon className="h-4 w-4" />
   478→              <AlertDescription>
   479→                <strong>Payment Issues Detected</strong><br />
   480→                Last failure: {planStatus.failure_info.last_reason || 'Unknown reason'}<br />
   481→                Consecutive failures: {planStatus.failure_info.consecutive_failures}/3 before auto-pause
   482→              </AlertDescription>
   483→            </Alert>
   484→          )}
   485→        </CardContent>
   486→      </Card>
   487→
   488→      {/* Recent Installments */}
   489→      <Card>
   490→        <CardHeader>
   491→          <CardTitle className="text-lg">Recent Payments</CardTitle>
   492→          <CardDescription>Last 5 installment transactions</CardDescription>
   493→        </CardHeader>
   494→        <CardContent>
   495→          {planStatus.recent_installments.length === 0 ? (
   496→            <div className="text-center py-8 text-muted-foreground">
   497→              No payments processed yet. Your first charge will happen soon!
   498→            </div>
   499→          ) : (
   500→            <div className="space-y-3">
   501→              {planStatus.recent_installments.map((inst) => (
   502→                <div
   503→                  key={inst.number}
   504→                  className="flex items-center justify-between p-3 rounded-lg bg-muted/50 hover:bg-muted transition-colors"
   505→                >
   506→                  <div className="flex items-center gap-3">
   507→                    {getInstallmentStatusIcon(inst.status)}
   508→                    <div>
   509→                      <div className="font-medium">
   510→                        Installment #{inst.number}
   511→                        {inst.is_extra && (
   512→                          <Badge variant="secondary" className="ml-2 text-xs">
   513→                            Extra Charge
   514→                          </Badge>
   515→                        )}
   516→                      </div>
   517→                      <div className="text-sm text-muted-foreground">
   518→                        {inst.processed_at ? formatDate(inst.processed_at) : 'Scheduled'}
   519→                      </div>
   520→                    </div>
   521→                  </div>
   522→                  
   523→                  <div className="text-right">
   524→                    <div className={`font-semibold ${
   525→                      inst.status === 'paid' ? 'text-green-600' :
   526→                      inst.status === 'failed' ? 'text-red-600' :
   527→                      'text-foreground'
   528→                    }`}>
   529→                      {formatCurrency(inst.amount)}
   530→                    </div>
   531→                    <Badge variant="outline" className="text-xs mt-1 capitalize">
   532→                      {inst.status}
   533→                    </Badge>
   534→                  </div>
   535→                </div>
   536→              ))}
   537→            </div>
   538→          )}
   539→        </CardContent>
   540→      </Card>
   541→
   542→      {/* How It Works Info Card */}
   543→      {planStatus.status === 'pending' || planStatus.status === 'active' ? (
   544→        <Card className="border-dashed">
   545→          <CardHeader>
   546→            <CardTitle className="text-lg flex items-center gap-2">
   547→              <ClockIcon className="h-5 w-5 text-primary" />
   548→              How FlexPay Works
   549→            </CardTitle>
   550→          </CardHeader>
   551→          <CardContent className="space-y-3 text-sm text-muted-foreground">
   552→            <p>
   553→              Instead of paying <strong className="text-foreground">{formatCurrency(planStatus.total_amount)}</strong> upfront,
   554→              we split it into smaller daily installments that fit within payment limits.
   555→            </p>
   556→            <ul className="list-disc list-inside space-y-1 ml-2">
   557→              <li><strong>Daily base charge:</strong> $100 every morning (9 AM UTC)</li>
   558→              <li><strong>Extra charge:</strong> Additional $100 every 3rd day (11 AM UTC)</li>
   559→              <li><strong>Total duration:</strong> {totalDays} days to collect full amount</li>
   560→              <li><strong>Auto-retry:</strong> Failed payments retry automatically after 24 hours</li>
   561→              <li><strong>Auto-pause:</strong> Plan pauses after 3 consecutive failures</li>
   562→            </ul>
   563→            <p className="pt-2">
   564→              You can pause or cancel anytime from this dashboard.
   565→            </p>
   566→          </CardContent>
   567→        </Card>
   568→      ) : null}
   569→
   570→      {/* Pause Dialog */}
   571→      <Dialog open={pauseDialogOpen} onOpenChange={setPauseDialogOpen}>
   572→        <DialogContent>
   573→          <DialogHeader>
   574→            <DialogTitle>Pause FlexPay Plan?</DialogTitle>
   575→            <DialogDescription>
   576→              This will temporarily stop all future payments until you resume.
   577→              Your service access will continue for already-paid portions.
   578→            </DialogDescription>
   579→          </DialogHeader>
   580→          <DialogFooter>
   581→            <Button variant="outline" onClick={() => setPauseDialogOpen(false)}>
   582→              Cancel
   583→            </Button>
   584→            <Button
   585→              variant="default"
   586→              onClick={() => handleAction('pause', { reason: 'Customer paused manually' })}
   587→              disabled={actionLoading === 'pause'}
   588→            >
   589→              {actionLoading === 'pause' ? 'Pausing...' : 'Yes, Pause Plan'}
   590→            </Button>
   591→          </DialogFooter>
   592→        </DialogContent>
   593→      </Dialog>
   594→
   595→      {/* Resume Dialog */}
   596→      <Dialog open={resumeDialogOpen} onOpenChange={setResumeDialogOpen}>
   597→        <DialogContent>
   598→          <DialogHeader>
   599→            <DialogTitle>Resume FlexPay Plan?</DialogTitle>
   600→            <DialogDescription>
   601→              This will restart automatic payments from where they left off.
   602→              Make sure your payment method is up to date before resuming.
   603→            </DialogDescription>
   604→          </DialogHeader>
   605→          <DialogFooter>
   606→            <Button variant="outline" onClick={() => setResumeDialogOpen(false)}>
   607→              Cancel
   608→            </Button>
   609→            <Button
   610→              variant="default"
   611→              onClick={() => handleAction('resume', { notes: 'Customer resumed manually' })}
   612→              disabled={actionLoading === 'resume'}
   613→            >
   614→              {actionLoading === 'resume' ? 'Resuming...' : 'Yes, Resume Plan'}
   615→            </Button>
   616→          </DialogFooter>
   617→        </DialogContent>
   618→      </Dialog>
   619→
   620→      {/* Error Toast */}
   621→      {error && (
   622→        <Alert variant="destructive" className="animate-in slide-in-from-top-2">
   623→          <AlertTriangleIcon className="h-4 w-4" />
   624→          <AlertDescription className="flex items-center justify-between">
   625→            <span>{error}</span>
   626→            <Button variant="ghost" size="sm" className="h-auto p-1" onClick={() => setError(null)}>
   627→              ✕
   628→            </Button>
   629→          </AlertDescription>
   630→        </Alert>
   631→      )}
   632→    </div>
   633→  );
   634→}

   644→interface CompanyPlansListProps {
   645→  /** Callback when a plan is selected */
   646→  onSelectPlan: (planId: string) => void;
   647→  /** Show completed/cancelled plans too */
   648→  includeCompleted?: boolean;
   649→}
   650→
   651→export function FlexPayCompanyPlansList({ 
   652→  onSelectPlan, 
   653→  includeCompleted = false 
   654→}: CompanyPlansListProps) {
   655→  const [plans, setPlans] = useState<any[]>([]);
   656→  const [loading, setLoading] = useState(true);
   657→  const [error, setError] = useState<string | null>(null);
   658→
   659→  useEffect(() => {
   660→    async function fetchPlans() {
   661→      try {
   662→        const response = await fetch(
   663→          `/api/flexpay/company-plans?include_completed=${includeCompleted}`,
   664→          {
   665→            headers: {
   666→              'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
   667→            },
   668→          }
   669→        );
   670→
   671→        if (!response.ok) {
   672→          throw new Error('Failed to fetch plans');
   673→        }
   674→
   675→        const data = await response.json();
   676→        setPlans(data);
   677→      } catch (err) {
   678→        setError(err instanceof Error ? err.message : 'Failed to load plans');
   679→      } finally {
   680→        setLoading(false);
   681→      }
   682→    }
   683→
   684→    fetchPlans();
   685→  }, [includeCompleted]);
   686→
   687→  if (loading) {
   688→    return (
   689→      <div className="space-y-3">
   690→        {[1, 2, 3].map((i) => (
   691→          <div key={i} className="h-20 bg-muted rounded-lg animate-pulse" />
   692→        ))}
   693→      </div>
   694→    );
   695→  }
   696→
   697→  if (error) {
   698→    return (
   699→      <Alert variant="destructive">
   700→        <AlertTriangleIcon className="h-4 w-4" />
   701→        <AlertDescription>{error}</AlertDescription>
   702→      </Alert>
   703→    );
   704→  }
   705→
   706→  if (plans.length === 0) {
   707→    return (
   708→      <Card>
   709→        <CardContent className="py-8 text-center text-muted-foreground">
   710→          No FlexPay plans found for your company.
   711→        </CardContent>
   712→      </Card>
   713→    );
   714→  }
   715→
   716→  return (
   717→    <div className="space-y-3">
   718→      {plans.map((plan) => (
   719→        <Card
   720→          key={plan.plan_id}
   721→          className="cursor-pointer hover:shadow-md transition-shadow"
   722→          onClick={() => onSelectPlan(plan.plan_id)}
   723→        >
   724→          <CardContent className="p-4">
   725→            <div className="flex items-center justify-between">
   726→              <div className="space-y-1">
   727→                <div className="flex items-center gap-2">
   728→                  <span className="font-semibold">{getVariantDisplayName(plan.variant_tier)}</span>
   729→                  <Badge variant="outline" className={getStatusColor(plan.status)}>
   730→                    {plan.status}
   731→                  </Badge>
   732→                </div>
   733→                <div className="text-sm text-muted-foreground">
   734→                  {plan.current_day && plan.total_days
   735→                    ? `Day ${plan.current_day} of ${plan.total_days}`
   736→                    : `Created ${formatDate(plan.created_at)}`}
   737→                </div>
   738→              </div>
   739→              
   740→              <div className="text-right space-y-1">
   741→                <div className="font-semibold">
   742→                  {formatCurrency(plan.collected_amount)} / {formatCurrency(plan.total_amount)}
   743→                </div>
   744→                <Progress value={plan.progress_percent} className="w-24 h-2" />
   745→                <div className="text-xs text-muted-foreground">
   746→                  {plan.progress_percent.toFixed(1)}%
   747→                </div>
   748→              </div>
   749→            </div>
   750→          </CardContent>
   751→        </Card>
   752→      ))}
   753→    </div>
   754→  );
   755→}

   133→   133→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   134→   134→
   135→   135→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   136→   136→  agentsReplaced: number;
   137→   137→  humanCost: number;
   138→   138→  savings: number;
   139→   139→  savingsPercent: number;
   140→   140→} {
   141→   141→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   142→   142→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   143→   143→  const savings = Math.max(0, humanCost - monthlyCost);
   144→   144→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   145→   145→  return { agentsReplaced, humanCost, savings, savingsPercent };
   146→   146→}

   150→   150→function UsageBar({
   151→   151→  used,
   152→   152→  total,
   153→   153→  label,
   154→   154→  unit,
   155→   155→  overageRate,
   156→   156→}: {
   157→   157→  used: number;
   158→   158→  total: number;
   159→   159→  label: string;
   160→   160→  unit: string;
   161→   161→  overageRate?: string;
   162→   162→}) {
   163→   163→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
   164→   164→  const isOverLimit = used > total;
   165→   165→
   166→   166→  return (
   167→   167→    <div className="space-y-1.5">
   168→   168→      <div className="flex items-center justify-between">
   169→   169→        <span className="text-[10px] text-orange-200/40">{label}</span>
   170→   170→        <span className={cn(
   171→   171→          'text-[10px] font-medium',
   172→   172→          isOverLimit ? 'text-red-400' : 'text-orange-200/50'
   173→   173→        )}>
   174→   174→          {used.toLocaleString()} / {total.toLocaleString()} {unit}
   175→   175→        </span>
   176→   176→      </div>
   177→   177→      <div className="h-1.5 bg-white/5 rounded-full overflow-hidden">
   178→   178→        <div
   179→   179→          className={cn(
   180→   180→            'h-full rounded-full transition-all duration-700',
   181→   181→            isOverLimit ? 'bg-gradient-to-r from-red-500 to-red-400' : 'bg-gradient-to-r from-emerald-500 to-emerald-400'
   182→   182→          )}
   183→   183→          style={{ width: `${pct}%` }}
   184→   184→        />
   185→   185→      </div>
   186→   186→      {isOverLimit && overageRate && (
   187→   187→        <p className="text-[9px] text-red-400 flex items-center gap-1">
   188→   188→          <AlertTriangle className="w-2.5 h-2.5" />
   189→   189→          Over limit — {(used - total).toLocaleString()} overage at {overageRate}/ticket
   190→   190→        </p>
   191→   191→      )}
   192→   192→    </div>
   193→   193→  );
   194→   194→}

   312→   312→function CouponCodeInput({
   313→   313→  appliedCoupon,
   314→   314→  onApply,
   315→   315→  onRemove,
   316→   316→}: {
   317→   317→  appliedCoupon: Coupon | null;
   318→   318→  onApply: (coupon: Coupon) => void;
   319→   319→  onRemove: () => void;
   320→   320→}) {
   321→   321→  const [code, setCode] = useState('');
   322→   322→  const [isValidating, setIsValidating] = useState(false);
   323→   323→  const [error, setError] = useState<string | null>(null);
   324→   324→
   325→   325→  const handleApply = () => {
   326→   326→    if (!code.trim()) return;
   327→   327→    setIsValidating(true);
   328→   328→    setError(null);
   329→   329→
   330→   330→    // Simulate brief validation delay
   331→   331→    setTimeout(() => {
   332→   332→      const coupon = validateCoupon(code);
   333→   333→      if (coupon) {
   334→   334→        onApply(coupon);
   335→   335→        toast.success(`Coupon applied: ${formatDiscount(coupon)}`);
   336→   336→      } else {
   337→   337→        setError('Invalid coupon code. Please check and try again.');
   338→   338→        toast.error('Invalid coupon code');
   339→   339→      }
   340→   340→      setIsValidating(false);
   341→   341→    }, 300);
   342→   342→  };
   343→   343→
   344→   344→  const handleKeyDown = (e: React.KeyboardEvent) => {
   345→   345→    if (e.key === 'Enter') {
   346→   346→      e.preventDefault();
   347→   347→      handleApply();
   348→   348→    }
   349→   349→  };
   350→   350→
   351→   351→  if (appliedCoupon) {
   352→   352→    return (
   353→   353→      <div className="rounded-xl border border-emerald-500/30 p-4" style={{ background: 'rgba(16,185,129,0.06)' }}>
   354→   354→        <div className="flex items-center justify-between">

   358→   358→            </div>

   361→   361→                <span className="text-sm font-semibold text-emerald-400">
   362→   362→                  {appliedCoupon.code}
   363→   363→                </span>
   364→   364→                <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   365→   365→                  {formatDiscount(appliedCoupon)}

   369→   369→                {appliedCoupon.description}
   370→   370→              </p>
   371→   371→            </div>
   372→   372→          </div>
   373→   373→          <button
   374→   374→            onClick={onRemove}
   375→   375→            className="p-1.5 rounded-lg hover:bg-white/[0.06] text-orange-200/30 hover:text-red-400 transition-all"
   376→   376→            title="Remove coupon"
   377→   377→          >
   378→   378→            <X className="w-4 h-4" />
   379→   379→          </button>
   380→   380→        </div>
   381→   381→        {appliedCoupon.discountPercent === 100 && (
   382→   382→          <div className="mt-3 flex items-start gap-2 rounded-lg p-2.5" style={{ background: 'rgba(16,185,129,0.08)' }}>

   384→   384→            <div>
   385→   385→              <p className="text-[10px] font-medium text-emerald-300">
   386→   386→                Free checkout — $0.00 total

   394→   394→      </div>
   395→   395→    );
   396→   396→  }
   397→   397→
   398→   398→  return (
   399→   399→    <div className="rounded-xl border border-white/[0.08] p-4 space-y-3" style={{ background: 'rgba(255,255,255,0.03)' }}>
   400→   400→      <div className="flex items-center gap-2">
   401→   401→        <Tag className="w-4 h-4 text-orange-400" />
   402→   402→        <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   403→   403→          Have a coupon code?
   404→   404→        </span>
   405→   405→      </div>
   406→   406→      <div className="flex gap-2">
   407→   407→        <input
   408→   408→          type="text"
   409→   409→          value={code}
   410→   410→          onChange={(e) => {
   411→   411→            setCode(e.target.value.toUpperCase());
   412→   412→            setError(null);
   413→   413→          }}
   414→   414→          onKeyDown={handleKeyDown}
   415→   415→          placeholder="Enter coupon code"
   416→   416→          className={cn(
   417→   417→            'flex-1 h-10 rounded-lg bg-white/[0.04] border px-3 text-sm text-white placeholder-orange-200/20',
   418→   418→            'focus:outline-none focus:border-orange-500/50 transition-all',
   419→   419→            error ? 'border-red-500/40' : 'border-white/[0.08]'

   426→   426→        >
   427→   427→          {isValidating ? (
   428→   428→            <Loader2 className="w-3.5 h-3.5 animate-spin" />
   429→   429→          ) : (
   430→   430→            <Check className="w-3.5 h-3.5" />

   432→   432→          Apply
   433→   433→        </button>

   435→   435→      {error && (

   439→   439→        </p>

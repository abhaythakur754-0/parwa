   199→   199→   199→def _detect_capability(query: str, ticket_type: str) -> str | None:
   200→   200→   200→    """Detect which capability this query needs.
   201→   201→   201→
   202→   202→   202→    First checks the direct ticket_type → capability mapping (fast path).
   203→   203→   203→    Then checks CAPABILITY_PATTERNS for capabilities without a built-in
   204→   204→   204→    ticket_type (legal_review, fraud_security, etc.).
   205→   205→   205→
   206→   206→   206→    Returns the capability key, or None if no capability matched.
   207→   207→   207→    """
   208→   208→   208→    # Fast path: built-in ticket_type maps to a capability
   209→   209→   209→    if ticket_type in TICKET_TYPE_TO_CAPABILITY:
   210→   210→   210→        return TICKET_TYPE_TO_CAPABILITY[ticket_type]
   211→   211→   211→
   212→   212→   212→    # Direct pattern match for capabilities without a built-in ticket_type
   213→   213→   213→    query_lower = query.lower()
   214→   214→   214→    for capability, patterns in CAPABILITY_PATTERNS.items():
   215→   215→   215→        for pat in patterns:
   216→   216→   216→            if re.search(pat, query_lower):
   217→   217→   217→                return capability
   218→   218→   218→
   219→   219→   219→    return None

   222→   222→   222→async def _llm_match_capability(query: str, tenant_id: str) -> str | None:
   223→   223→   223→    """LLM-based capability matcher — fallback when rule-based matcher returns None.
   224→   224→   224→
   225→   225→   225→    Uses NVIDIA Llama 3.1 8B (cheap, ~500ms, ~$0.0001/call) to classify the
   226→   226→   226→    query against the 24-capability vocabulary. Also considers any custom
   227→   227→   227→    capabilities the tenant has defined on their agents.
   228→   228→   228→
   229→   229→   229→    Args:
   230→   230→   230→        query: The customer's ticket text (truncated to 500 chars to keep
   231→   231→   231→               token usage low).
   232→   232→   232→        tenant_id: The tenant's company_id — used to fetch any custom
   233→   233→   233→                   capabilities the tenant has defined on their agents.
   234→   234→   234→
   235→   235→   235→    Returns:
   236→   236→   236→        Capability key string, or None if the LLM call fails or returns
   237→   237→   237→        an unrecognized capability.
   238→   238→   238→    """
   239→   239→   239→    # Fetch tenant's custom capabilities (if any) to extend the vocabulary
   240→   240→   240→    tenant_capabilities: list[str] = []
   241→   241→   241→    try:
   242→   242→   242→        from database.base import SessionLocal
   243→   243→   243→        from database.models.variant_engine import AIAgentAssignment
   244→   244→   244→        import json as _json
   245→   245→   245→        db = SessionLocal()
   246→   246→   246→        try:
   247→   247→   247→            rows = db.query(AIAgentAssignment).filter(
   248→   248→   248→                AIAgentAssignment.company_id == tenant_id,
   249→   249→   249→                AIAgentAssignment.status == "active",
   250→   250→   250→            ).all()
   251→   251→   251→            for row in rows:
   252→   252→   252→                try:
   253→   253→   253→                    caps = _json.loads(row.capabilities or "[]")
   254→   254→   254→                    if isinstance(caps, list):
   255→   255→   255→                        for c in caps:
   256→   256→   256→                            if c and c not in tenant_capabilities and c not in ALL_CAPABILITIES:
   257→   257→   257→                                tenant_capabilities.append(c)
   258→   258→   258→                except (_json.JSONDecodeError, TypeError):
   259→   259→   259→                    pass
   260→   260→   260→        finally:
   261→   261→   261→            db.close()
   262→   262→   262→    except Exception as exc:
   263→   263→   263→        logger.warning(
   264→   264→   264→            "llm_capability_matcher_tenant_caps_failed tenant=%s err=%s",
   265→   265→   265→            tenant_id, str(exc)[:150],
   266→   266→   266→        )
   267→   267→   267→
   268→   268→   268→    # Build the capability vocabulary for the LLM
   269→   269→   269→    vocab = ALL_CAPABILITIES + tenant_capabilities
   270→   270→   270→    vocab_str = ", ".join(vocab)
   271→   271→   271→
   272→   272→   272→    # Truncate query to keep token usage low
   273→   273→   273→    query_trunc = query[:500]
   274→   274→   274→
   275→   275→   275→    messages = [
   276→   276→   276→        {
   277→   277→   277→            "role": "system",
   278→   278→   278→            "content": (
   279→   279→   279→                "You are a ticket classifier. Given a customer query, return "
   280→   280→   280→                "the single best matching capability key from this list. "
   281→   281→   281→                "Return ONLY the capability key, nothing else. "
   282→   282→   282→                f"Capabilities: {vocab_str}"
   283→   283→   283→            ),
   284→   284→   284→        },
   285→   285→   285→        {
   286→   286→   286→            "role": "user",
   287→   287→   287→            "content": query_trunc,
   288→   288→   288→        },
   289→   289→   289→    ]
   290→   290→   290→
   291→   291→   291→    try:
   292→   292→   292→        from app.core.parwa_pipeline.pipeline_config import (
   293→   293→   293→            NVIDIA_API_KEY, NVIDIA_API_BASE, NVIDIA_MODEL,
   294→   294→   294→        )
   295→   295→   295→        if not NVIDIA_API_KEY:
   296→   296→   296→            logger.warning("llm_capability_matcher_no_nvidia_key")
   297→   297→   297→            return None
   298→   298→   298→
   299→   299→   299→        import httpx
   300→   300→   300→        import asyncio
   301→   301→   301→        url = f"{NVIDIA_API_BASE}/chat/completions"
   302→   302→   302→        payload = {
   303→   303→   303→            "model": NVIDIA_MODEL,
   304→   304→   304→            "messages": messages,
   305→   305→   305→            "temperature": 0.0,  # deterministic — same query = same capability
   306→   306→   306→            "max_tokens": 20,    # capability keys are short
   307→   307→   307→        }
   308→   308→   308→        headers = {
   309→   309→   309→            "Authorization": f"Bearer {NVIDIA_API_KEY}",
   310→   310→   310→            "Content-Type": "application/json",
   311→   311→   311→        }
   312→   312→   312→
   313→   313→   313→        async with httpx.AsyncClient(timeout=10.0) as client:
   314→   314→   314→            r = await client.post(url, json=payload, headers=headers)
   315→   315→   315→
   316→   316→   316→        if r.status_code != 200:
   317→   317→   317→            logger.warning(
   318→   318→   318→                "llm_capability_matcher_http_%d: %s",
   319→   319→   319→                r.status_code, r.text[:200],
   320→   320→   320→            )
   321→   321→   321→            return None
   322→   322→   322→
   323→   323→   323→        content = r.json()["choices"][0]["message"]["content"].strip().lower()
   324→   324→   324→        # Normalize: strip quotes, whitespace, trailing punctuation
   325→   325→   325→        content = content.strip("`'\".,;: \n\r\t")
   326→   326→   326→
   327→   327→   327→        # Validate the LLM returned a real capability
   328→   328→   328→        if content in vocab:
   329→   329→   329→            logger.info(
   330→   330→   330→                "llm_capability_matcher_hit tenant=%s capability=%s",
   331→   331→   331→                tenant_id, content,
   332→   332→   332→            )
   333→   333→   333→            return content
   334→   334→   334→        else:
   335→   335→   335→            logger.warning(
   336→   336→   336→                "llm_capability_matcher_unknown_response tenant=%s response=%s",
   337→   337→   337→                tenant_id, content[:80],
   338→   338→   338→            )
   339→   339→   339→            return None
   340→   340→   340→
   341→   341→   341→    except Exception as exc:
   342→   342→   342→        logger.warning(
   343→   343→   343→            "llm_capability_matcher_failed tenant=%s err=%s",
   344→   344→   344→            tenant_id, str(exc)[:150],
   345→   345→   345→        )
   346→   346→   346→        return None

   373→   373→   373→                ),

   375→   375→   375→            ).fetchall()

   382→   382→   382→                except (ValueError, TypeError):
   383→   383→   383→                    continue
   384→   384→   384→                if not isinstance(caps, list):
   385→   385→   385→                    continue

   387→   387→   387→                    if not isinstance(c, str):
   388→   388→   388→                        continue
   389→   389→   389→                    c_lower = c.lower().strip()
   390→   390→   390→                    if not c_lower:
   391→   391→   391→                        continue

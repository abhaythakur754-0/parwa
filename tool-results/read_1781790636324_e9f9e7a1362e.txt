    43→    43→def _format_notification(ntype: str, ev: Dict) -> tuple:
    44→    44→    """Format notification title and description by type."""
    45→    45→    signal = ev.get("signal", {})
    46→    46→    score = ev.get("priority_score", 0)
    47→    47→
    48→    48→    if ntype == TYPE_STUCK_TICKET:
    49→    49→        tid = signal.get("ticket_id", "unknown")
    50→    50→        reason = signal.get("reason", "unknown")
    51→    51→        quality = signal.get("quality_score", "?")
    52→    52→        escalation = ev.get("escalation_tier", "soft_reminder")
    53→    53→        hours = signal.get("hours_stuck", 0)
    54→    54→        title = f"Stuck Ticket [{escalation.upper()}]: {tid}"
    55→    55→        desc = (f"Ticket {tid} could not be resolved automatically. "
    56→    56→                f"Reason: {reason}. Quality achieved: {quality}. "
    57→    57→                f"Loops used: {signal.get('loops_used', 0)}. "
    58→    58→                f"Hours stuck: {hours}. Escalation: {escalation}. "
    59→    59→                f"Requires manual review.")
    60→    60→
    61→    61→    elif ntype == TYPE_QUOTA_LOW:
    62→    62→        data = list(signal.values())[0] if signal else {}
    63→    63→        tier = list(signal.keys())[0] if signal else "unknown"
    64→    64→        title = f"Quota {data.get('status', 'warning').upper()}: {tier}"
    65→    65→        desc = (f"{tier.capitalize()} plan quota is {data.get('burn_pct', 0)}% used "
    66→    66→                f"({data.get('remaining', 0)} of {data.get('total', 0)} remaining). "
    67→    67→                f"Consider upgrading or requesting additional quota.")
    68→    68→
    69→    69→    elif ntype == TYPE_ACCURACY_DROP:
    70→    70→        trend = signal.get("trend", "unknown")
    71→    71→        severity = signal.get("severity", "warning")
    72→    72→        trigger = signal.get("trigger", "unknown")
    73→    73→        acc_7d = signal.get("accuracy_7d", "?")
    74→    74→        acc_today = signal.get("accuracy_today", "?")
    75→    75→        title = f"Accuracy Drift [{severity.upper()}]: {trend}"
    76→    76→        desc = (f"Resolution accuracy trend is '{trend}' (severity: {severity}). "
    77→    77→                f"7-day avg: {acc_7d}, Today: {acc_today}. "
    78→    78→                f"Trigger: {trigger}. "
    79→    79→                f"This may indicate KB gaps, policy changes, or model drift. "
    80→    80→                f"Review recent resolutions and update knowledge base if needed.")
    81→    81→
    82→    82→    elif ntype == TYPE_INTEGRATION_DOWN:
    83→    83→        degraded = signal.get("degraded_services", [])
    84→    84→        worst_uptime = signal.get("worst_uptime_pct", 0)
    85→    85→        names = ", ".join(f"{d['name']} ({d['status']}, uptime={d.get('uptime_pct', '?')}%)" for d in degraded)
    86→    86→        title = f"Integration Issues: {signal.get('total_degraded', len(degraded))} service(s) degraded"
    87→    87→        desc = (f"Degraded integrations: {names}. "
    88→    88→                f"Worst uptime: {worst_uptime}%. "
    89→    89→                f"Check API credentials and service status.")
    90→    90→
    91→    91→    elif ntype == "load_bottleneck":
    92→    92→        at_cap = signal.get("at_capacity", [])
    93→    93→        high_load = signal.get("high_load", [])
    94→    94→        vip = signal.get("vip_overflow_risk", False)
    95→    95→        bottlenecks = ", ".join(v["name"] for v in at_cap + high_load)
    96→    96→        title = f"Load Bottleneck: {bottlenecks or 'All variants'}"
    97→    97→        desc = (f"Variant load issue detected. "
    98→    98→                f"At capacity: {', '.join(v['name'] for v in at_cap) or 'none'}. "
    99→    99→                f"High load: {', '.join(v['name'] for v in high_load) or 'none'}. "
   100→   100→                f"VIP overflow risk: {vip}. "
   101→   101→                f"Consider scaling up or routing overflow to available variants.")
   102→   102→
   103→   103→    else:
   104→   104→        title = f"Signal: {ntype}"
   105→   105→        desc = str(signal)
   106→   106→
   107→   107→    return title, desc

   110→   110→async def _create_notifications_from_evals(
   111→   111→    tenant_id: str,
   112→   112→    evaluations: List[Dict[str, Any]],
   113→   113→    db,
   114→   114→) -> List[Dict[str, Any]]:
   115→   115→    """Convert evaluations into notifications via DB. Filter LOW priority."""
   116→   116→    notifications = []
   117→   117→    for ev in evaluations:
   118→   118→        score = ev.get("priority_score", 0)
   119→   119→        if score < 0.40:
   120→   120→            logger.info("LOW priority signal filtered: %s (score=%.3f)", ev.get("type"), score)
   121→   121→            continue
   122→   122→
   123→   123→        ntype = ev.get("type", "unknown")
   124→   124→        title, desc = _format_notification(ntype, ev)
   125→   125→        related = ev.get("related_tickets", [])
   126→   126→
   127→   127→        nf = await db.create_notification(
   128→   128→            tenant_id=tenant_id,
   129→   129→            ntype=ntype,
   130→   130→            priority_score=score,
   131→   131→            title=title,
   132→   132→            description=desc,
   133→   133→            related_tickets=related,
   134→   134→            batch_key=ev.get("type", ""),
   135→   135→            source_data=ev.get("signal", {}),
   136→   136→        )
   137→   137→        notifications.append(nf)
   138→   138→
   139→   139→    return notifications

   142→   142→# ── Command Execution (NEW — Wave 1) ─────────────────────────
   143→   143→
   144→   144→async def _execute_command(
   145→   145→    intent_result: Dict[str, Any],
   146→   146→    auth_result: AuthResult,
   147→   147→    tenant_id: str,
   148→   148→    signals: Dict[str, Any],
   149→   149→    db,
   150→   150→) -> str:
   151→   151→    """Execute a classified command and return the response text.
   152→   152→
   153→   153→    This is the core "Jarvis does things" function.
   154→   154→    Reads/writes DB based on intent. Returns human-readable response.
   155→   155→    """
   156→   156→    intent = intent_result["intent"]
   157→   157→    target = intent_result["target"]
   158→   158→
   159→   159→    # ── QUERY INTENTS ────────────────────────────────────
   160→   160→    if is_query_intent(intent):
   161→   161→        response = await _handle_query(intent, target, tenant_id, signals, db)
   162→   162→        # Log to audit
   163→   163→        await db.create_audit_entry(
   164→   164→            tenant_id=tenant_id,
   165→   165→            action=f"query:{intent}",
   166→   166→            actor_email=auth_result.email,
   167→   167→            target_type="query",
   168→   168→            payload={"intent": intent, "target": target},
   169→   169→        )
   170→   170→        return response
   171→   171→
   172→   172→    # ── CONTROL INTENTS ──────────────────────────────────
   173→   173→    if is_control_intent(intent):
   174→   174→        response = await _handle_control(intent, target, tenant_id, auth_result, db)
   175→   175→        return response
   176→   176→
   177→   177→    # ── APPROVAL INTENTS ─────────────────────────────────
   178→   178→    if is_approval_intent(intent):
   179→   179→        response = await _handle_approval(intent, target, tenant_id, auth_result, db)
   180→   180→        return response
   181→   181→
   182→   182→    # ── EMERGENCY INTENTS ────────────────────────────────
   183→   183→    if is_emergency_intent(intent):
   184→   184→        response = await _handle_emergency(intent, target, tenant_id, auth_result, db)
   185→   185→        return response
   186→   186→
   187→   187→    # ── EXPLAIN / TEACH / AGENT ──────────────────────────
   188→   188→    if intent == "explain_ticket":
   189→   189→        return await _handle_explain_ticket(target, tenant_id, signals, db)
   190→   190→    if intent == "explain_flag":
   191→   191→        return await _handle_explain_flags(tenant_id, db)
   192→   192→    if intent == "teach_skill":
   193→   193→        return await _handle_teach_skill(target, tenant_id, auth_result, db)
   194→   194→    if intent == "create_agent":
   195→   195→        return await _handle_create_agent(target, tenant_id, auth_result, db)
   196→   196→
   197→   197→    return f"I understood your command but don't have a handler for '{intent}' yet. This will be available in a future wave."
   198→   198→

   202→   202→async def _handle_query(intent: str, target: str, tenant_id: str, signals: Dict, db) -> str:
   203→   203→    """Handle all query intents — read from DB, format response."""
   204→   204→
   205→   205→    if intent == "query_status":
   206→   206→        health = await db.health_check()
   207→   207→        active_flags = await db.get_active_flags(tenant_id)
   208→   208→        paused = [f for f in active_flags if f["flag_type"] == "pause_action"]
   209→   209→        return (
   210→   210→            f"**System Status**: {health.get('status', 'unknown')}\n"
   211→   211→            f"Backend: {health.get('backend', 'unknown')}\n"
   212→   212→            f"Active Flags: {len(active_flags)}\n"
   213→   213→            f"Paused Actions: {len(paused)} {', '.join(f['flag_value'] for f in paused) if paused else 'none'}\n"
   214→   214→            f"Mode: Supervised (default)"
   215→   215→        )
   216→   216→
   217→   217→    if intent == "query_errors":
   218→   218→        errors = signals.get("errors", [])
   219→   219→        if not errors:
   220→   220→            return "No errors detected in the current session."
   221→   221→        return f"**Recent Errors** ({len(errors)}):\n" + "\n".join(f"- {e}" for e in errors[-5:])
   222→   222→
   223→   223→    if intent == "query_tickets":
   224→   224→        flow = signals.get("ticket_flow", {})
   225→   225→        summary = flow.get("summary", {}) if isinstance(flow, dict) else {}
   226→   226→        current = flow.get("current_ticket", {}) if isinstance(flow, dict) else {}
   227→   227→        drift = signals.get("drift_status", {})
   228→   228→        return (
   229→   229→            f"**Ticket Flow Summary**:\n"
   230→   230→            f"Total Processed: {summary.get('total', 0)}\n"
   231→   231→            f"Auto-Resolved: {summary.get('auto_resolved', 0)}\n"
   232→   232→            f"Batched: {summary.get('batched', 0)}\n"
   233→   233→            f"Escalated: {summary.get('escalated', 0)}\n"
   234→   234→            f"Stuck: {summary.get('stuck', 0)}\n"
   235→   235→            f"Avg Quality: {summary.get('avg_quality', 'N/A')}\n"
   236→   236→            f"Avg LLM Calls: {summary.get('avg_llm_calls', 'N/A')}\n"
   237→   237→            f"By Path: {summary.get('by_type', {})}\n"
   238→   238→            f"Accuracy Trend: {drift.get('trend_direction', 'no_data')}"
   239→   239→        )
   240→   240→
   241→   241→    if intent == "query_quality":
   242→   242→        stats = await db.get_quality_stats(tenant_id)
   243→   243→        drift = signals.get("drift_status", {})
   244→   244→        drift_line = f"\nDrift: {drift.get('trend_direction', 'no_data')}" \
   245→   245→                    f" (detected={drift.get('drift_detected')}, severity={drift.get('drift_severity')})" \
   246→   246→                    if drift.get('total_scores', 0) > 0 else "\nDrift: insufficient data"
   247→   247→        return (
   248→   248→            f"**Quality Metrics**:\n"
   249→   249→            f"Total Tickets: {stats['total_tickets']}\n"
   250→   250→            f"Average Quality: {stats['avg_quality']:.2%}\n"
   251→   251→            f"Average Confidence: {stats['avg_confidence']:.2%}\n"
   252→   252→            f"Auto-Resolved: {stats['auto_resolved']}\n"
   253→   253→            f"Escalated: {stats['escalated']}\n"
   254→   254→            f"Stuck: {stats['stuck']}\n"
   255→   255→            f"By Path: {stats.get('by_path', {})}"
   256→   256→            f"{drift_line}"
   257→   257→        )
   258→   258→
   259→   259→    if intent == "query_quota":
   260→   260→        quota = signals.get("quota_status", {})
   261→   261→        if not quota:
   262→   262→            return "No quota data available. Connect to billing system for live quota tracking."
   263→   263→        parts = []
   264→   264→        for tier, data in quota.items():
   265→   265→            parts.append(
   266→   266→                f"  {tier}: {data.get('remaining', 0)}/{data.get('total', 0)} "
   267→   267→                f"remaining ({data.get('burn_pct', 0)}% used, status: {data.get('status', 'unknown')})"
   268→   268→            )
   269→   269→        return f"**Quota Status**:\n" + "\n".join(parts)
   270→   270→
   271→   271→    if intent == "query_notifications":
   272→   272→        if target and target.startswith("PARWA-NFY"):
   273→   273→            # Specific notification lookup
   274→   274→            nf = await db.get_notification(target)
   275→   275→            if nf:
   276→   276→                return (
   277→   277→                    f"**{nf['title']}** (Key: {nf['notification_key']})\n\n"
   278→   278→                    f"{nf['description']}\n\n"
   279→   279→                    f"Priority: {nf['priority']} (score: {nf['priority_score']})\n"

   281→   281→                    f"Type: {nf['type']}\n"
   282→   282→                    f"Related tickets: {', '.join(nf['related_tickets']) or 'None'}"
   283→   283→                )
   284→   284→            return f"Notification {target} not found."
   285→   285→        else:
   286→   286→            # List all notifications
   287→   287→            nfs = await db.get_notifications(tenant_id)
   288→   288→            if not nfs:
   289→   289→                return "No unresolved notifications. All clear."
   290→   290→            lines = [f"**Notifications** ({len(nfs)} unresolved):"]
   291→   291→            for nf in nfs[:10]:
   292→   292→                lines.append(f"  [{nf['priority']}] {nf['notification_key']}: {nf['title']}")
   293→   293→            if len(nfs) > 10:
   294→   294→                lines.append(f"  ... and {len(nfs) - 10} more")
   295→   295→            return "\n".join(lines)
   296→   296→
   297→   297→    if intent == "query_flags":
   298→   298→        flags = await db.get_active_flags(tenant_id)
   299→   299→        if not flags:
   300→   300→            return "No active flags/rules. System is running with default behavior."
   301→   301→        lines = [f"**Active Flags** ({len(flags)}):"]
   302→   302→        for f in flags:
   303→   303→            lines.append(
   304→   304→                f"  [{f['flag_type']}] {f['flag_value']} "
   305→   305→                f"(scope={f['scope']}, set by {f['set_by']})"
   306→   306→            )
   307→   307→        return "\n".join(lines)
   308→   308→
   309→   309→    if intent == "query_audit":
   310→   310→        trail = await db.get_audit_trail(tenant_id, limit=20)
   311→   311→        if not trail:
   312→   312→            return "No audit trail entries yet."
   313→   313→        lines = [f"**Recent Activity** ({len(trail)} entries):"]
   314→   314→        for entry in trail[:10]:
   315→   315→            lines.append(
   316→   316→                f"  [{entry['created_at'][:19]}] {entry['action']} "
   317→   317→                f"by {entry['actor_email']}"
   318→   318→            )
   319→   319→        if len(trail) > 10:
   320→   320→            lines.append(f"  ... and {len(trail) - 10} more")
   321→   321→        return "\n".join(lines)
   322→   322→
   323→   323→    # ── Wave 2 Query Handlers ─────────────────────────────
   324→   324→
   325→   325→    if intent == "query_health":
   326→   326→        health = signals.get("integration_health", {})
   327→   327→        services = health.get("services", {})
   328→   328→        if not services:
   329→   329→            return "No integration health data yet. Run a poll cycle first to collect ping data."
   330→   330→        lines = [f"**Integration Health** ({health.get('healthy_count', 0)} healthy, "
   331→   331→                 f"{health.get('degraded_count', 0)} degraded):"]
   332→   332→        for svc_name, svc_data in services.items():
   333→   333→            status_icon = "OK" if svc_data.get("status") == "healthy" else "!!"
   334→   334→            lines.append(
   335→   335→                f"  [{status_icon}] {svc_name}: {svc_data.get('status', '?')} "
   336→   336→                f"(uptime={svc_data.get('uptime_pct', '?')}%, "
   337→   337→                f"avg={svc_data.get('avg_response_ms', '?')}ms" +
   338→   338→                (f", error: {svc_data.get('last_error', 'none')}"
   339→   339→                 if svc_data.get('last_error') else ")")
   340→   340→            )
   341→   341→        return "\n".join(lines)
   342→   342→
   343→   343→    if intent == "query_cost":
   344→   344→        costs = signals.get("llm_costs", {})
   345→   345→        persisted = costs.get("persisted", {})
   346→   346→        live = costs.get("live_session", {})
   347→   347→        lines = [
   348→   348→            f"**LLM Cost Summary**:",
   349→   349→            f"  Total Cost (persisted): ${persisted.get('total_cost_usd', 0):.4f}",
   350→   350→            f"  Total Calls (persisted): {persisted.get('total_calls', 0)}",
   351→   351→            f"  Total Tokens (persisted): {persisted.get('total_tokens', 0)}",
   352→   352→            f"  Combined Calls (incl. live): {costs.get('total_calls_combined', 0)}",
   353→   353→            f"  Combined Tokens (incl. live): {costs.get('total_tokens_combined', 0)}",
   354→   354→        ]
   355→   355→        by_model = persisted.get("by_model", {})
   356→   356→        if by_model:
   357→   357→            lines.append("  By Model:")
   358→   358→            for model, mdata in by_model.items():
   359→   359→                lines.append(f"    {model}: ${mdata['cost']:.4f} ({mdata['calls']} calls, {mdata['tokens']} tokens)")
   360→   360→        return "\n".join(lines)
   361→   361→
   362→   362→    if intent == "query_flow":
   363→   363→        flow = signals.get("ticket_flow", {})
   364→   364→        summary = flow.get("summary", {}) if isinstance(flow, dict) else {}
   365→   365→        by_node = summary.get("by_node", {})
   366→   366→        lines = [
   367→   367→            f"**Ticket Flow Metrics**:",
   368→   368→            f"  Total: {summary.get('total', 0)}",
   369→   369→            f"  Auto-Resolved: {summary.get('auto_resolved', 0)}",
   370→   370→            f"  Batched: {summary.get('batched', 0)}",
   371→   371→            f"  Escalated: {summary.get('escalated', 0)}",
   372→   372→            f"  Stuck: {summary.get('stuck', 0)}",
   373→   373→            f"  Avg Quality: {summary.get('avg_quality', 0)}",
   374→   374→            f"  Avg LLM Calls/Ticket: {summary.get('avg_llm_calls', 0)}",
   375→   375→        ]
   376→   376→        if by_node:
   377→   377→            lines.append("  Node Distribution:")
   378→   378→            for node, count in sorted(by_node.items()):
   379→   379→                lines.append(f"    {node}: {count} tickets")
   380→   380→        return "\n".join(lines)
   381→   381→
   382→   382→    if intent == "query_load":
   383→   383→        load = signals.get("load_status", {})
   384→   384→        variants = load.get("variants", [])

   387→   387→            return "No variant load data. Configure agent_configs with max_concurrent to enable load monitoring."

   393→   393→                f"  [{status_icon}] {v['name']}: {v.get('concurrent', 0)}/{v.get('max_concurrent', 5)} "

   395→   395→            )
   396→   396→        return "\n".join(lines)
   397→   397→
   398→   398→    if intent == "query_stuck":

   623→   623→# ── Main Node Function ────────────────────────────────────────
   624→   624→
   625→   625→
   626→   626→async def jarvis_notify(state: dict) -> dict:
   627→   627→    """Jarvis Node 3: NOTIFY — Act + Verify.
   628→   628→
   629→   629→    Wave 1: Now wired with command_parser + jarvis_auth + jarvis_db.
   630→   630→    Full pipeline: chat → parse → auth → execute → DB → response.
   631→   631→    """
   632→   632→    start = time.time()
   633→   633→    tenant_id = state.get("tenant_id", "")
   634→   634→    trigger = state.get("trigger", "poll")
   635→   635→    evaluations = state.get("evaluations", [])
   636→   636→    signals = state.get("signals", {})
   637→   637→    admin_question = state.get("admin_question", "")
   638→   638→    user_context = state.get("user_context", {})
   639→   639→    logs = []
   640→   640→    llm_calls = 0
   641→   641→
   642→   642→    db = get_db()
   643→   643→
   644→   644→    # 1. Create notifications from evaluations (poll/monitor mode)
   645→   645→    notifications = []
   646→   646→    if trigger in ("poll", "stuck_ticket") and evaluations:
   647→   647→        notifications = await _create_notifications_from_evals(tenant_id, evaluations, db)
   648→   648→        logs.append({"node": "J3", "technique": "CreateNotifications", "duration_ms": 0,
   649→   649→                     "result_summary": f"created={len(notifications)}"})
   650→   650→
   651→   651→    # 2. Handle admin chat (full pipeline: parse → auth → execute)
   652→   652→    chat_response = ""
   653→   653→    if trigger == "admin_chat" and admin_question:
   654→   654→        # Step 1: Classify intent
   655→   655→        intent_result = await classify_command(admin_question)
   656→   656→        if intent_result.get("classification_method") == "llm":
   657→   657→            llm_calls += 1
   658→   658→        logs.append({"node": "J3", "technique": "CommandParser",
   659→   659→                     "duration_ms": 0, "result_summary": f"intent={intent_result['intent']} "
   660→   660→                     f"method={intent_result['classification_method']} "
   661→   661→                     f"confidence={intent_result['confidence']}"} )
   662→   662→
   663→   663→        # Step 2: Authorize
   664→   664→        auth_result = await authorize_command(
   665→   665→            intent=intent_result["intent"],
   666→   666→            user_context=user_context,
   667→   667→            tenant_id=tenant_id,
   668→   668→        )
   669→   669→        logs.append({"node": "J3", "technique": "AuthCheck",
   670→   670→                     "duration_ms": 0, "result_summary": f"authorized={auth_result.authorized} "
   671→   671→                     f"role={auth_result.role}"})
   672→   672→
   673→   673→        # Step 3: Execute (or deny)
   674→   674→        if auth_result.authorized:
   675→   675→            chat_response = await _execute_command(
   676→   676→                intent_result=intent_result,
   677→   677→                auth_result=auth_result,
   678→   678→                tenant_id=tenant_id,
   679→   679→                signals=signals,
   680→   680→                db=db,
   681→   681→            )
   682→   682→            logs.append({"node": "J3", "technique": "ExecuteCommand",
   683→   683→                         "duration_ms": 0, "result_summary": f"executed={intent_result['intent']}"})
   684→   684→        else:
   685→   685→            chat_response = f"[DENIED] {auth_result.reason}"
   686→   686→            logs.append({"node": "J3", "technique": "AuthDenied",
   687→   687→                         "duration_ms": 0, "result_summary": auth_result.reason})
   688→   688→
   689→   689→    # 3. Build quota feedback for PARWA Node 2
   690→   690→    quota_feedback = {
   691→   691→        "tenant_id": tenant_id,
   692→   692→        "quota_status": signals.get("quota_status", {}),
   693→   693→        "variant_tier": list(signals.get("quota_status", {}).keys())[0] if signals.get("quota_status") else "parwa",
   694→   694→    }
   695→   695→
   696→   696→    # 4. Summary
   697→   697→    keys = [n["notification_key"] for n in notifications]
   698→   698→    elapsed = int((time.time() - start) * 1000)
   699→   699→    logger.info("Jarvis NOTIFY complete: tenant=%s notifications=%d chat=%s llm=%d auth=%s [%dms]",
   700→   700→                tenant_id, len(notifications), bool(chat_response), llm_calls,
   701→   701→                bool(user_context.get("role")), elapsed)
   702→   702→
   703→   703→    return {
   704→   704→        "notifications": notifications,
   705→   705→        "notification_keys": keys,
   706→   706→        "chat_response": chat_response,
   707→   707→        "quota_feedback": quota_feedback,
   708→   708→        "notify_log": logs,
   709→   709→        "total_token_usage": state.get("total_token_usage", 0) + llm_calls,
   710→   710→        "intent_result": intent_result if trigger == "admin_chat" else None,
   711→   711→        "auth_result": {"authorized": auth_result.authorized, "role": auth_result.role}
   712→   712→                        if trigger == "admin_chat" else None,
   713→   713→    }

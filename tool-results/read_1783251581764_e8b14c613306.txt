   130→

   135→    result = await llm_call(prompt, max_tokens=150)
   136→    return {
   137→        "relevant_knowledge": result,

   139→        "knowledge_contradictory": False,
   140→    }

   143→# ── Knowledge Retrieval (type-based, non-LLM) ────────────────────
   144→
   145→
   146→def _retrieve_knowledge(ticket_type: str, query: str = "", tenant_id: str = "") -> List[Dict[str, Any]]:
   147→    """Retrieve knowledge documents for THIS tenant only.
   148→
   149→    Two-tier retrieval:
   150→      1. PER-TENANT KB: query DocumentChunk table filtered by company_id
   151→         (the tenant's own uploaded docs — real multi-tenant isolation)
   152→      2. FALLBACK: shared KNOWLEDGE_BASE dict (platform default docs)
   153→         only used when the tenant has no uploaded KB docs
   154→
   155→    BC-001: tenant_id scopes ALL queries. Company A NEVER sees Company B's docs.
   156→    """
   157→    # ── Tier 1: Per-tenant KB from PostgreSQL ────────────────────
   158→    if tenant_id:
   159→        try:
   160→            from database.base import SessionLocal
   161→            from database.models.onboarding import KnowledgeDocument, DocumentChunk
   162→
   163→            db = SessionLocal()
   164→            try:
   165→                # Get completed documents for THIS tenant only
   166→                docs_query = db.query(KnowledgeDocument).filter(
   167→                    KnowledgeDocument.company_id == tenant_id,
   168→                    KnowledgeDocument.status == "completed",
   169→                )
   170→                tenant_docs = docs_query.all()
   171→
   172→                if tenant_docs:
   173→                    # Get chunks for those documents — scoped by company_id
   174→                    doc_ids = [d.id for d in tenant_docs]
   175→                    chunks = db.query(DocumentChunk).filter(
   176→                        DocumentChunk.company_id == tenant_id,
   177→                        DocumentChunk.document_id.in_(doc_ids),
   178→                    ).all()
   179→
   180→                    if chunks:
   181→                        result = []
   182→                        for chunk in chunks:
   183→                            result.append({
   184→                                "content": chunk.content,
   185→                                "source": f"tenant_kb:{chunk.document_id}",
   186→                                "section": "C",
   187→                            })
   188→                        logger.info(
   189→                            "Node 3: Retrieved %d tenant KB chunks for company_id=%s",
   190→                            len(result), tenant_id,
   191→                        )
   192→                        return result
   193→                    else:
   194→                        logger.info("Node 3: Tenant has docs but no chunks — falling back to default KB")
   195→                else:
   196→                    logger.info("Node 3: No tenant KB docs for company_id=%s — falling back to default KB", tenant_id)
   197→            finally:
   198→                db.close()
   199→        except Exception as exc:
   200→            logger.warning("Node 3: Tenant KB query failed: %s — falling back to default KB", str(exc)[:200])
   201→
   202→    # ── Tier 2: Fallback to shared default KB (platform docs) ────
   203→    # Primary: exact type match
   204→    docs = KNOWLEDGE_BASE.get(ticket_type, [])
   205→
   206→    # Secondary: also pull from related types
   207→    related_types = {
   208→        "refund_request": ["billing"],
   209→        "billing": ["refund_request", "faq"],
   210→        "account_change": ["technical", "faq"],
   211→        "technical": ["account_change", "faq"],
   212→        "complaint": ["billing", "faq"],
   213→        "faq": ["billing", "account_change", "refund_request"],
   214→    }
   215→    for rt in related_types.get(ticket_type, []):
   216→        docs.extend(KNOWLEDGE_BASE.get(rt, []))
   217→
   218→    # Phase 7: T2→T1 Cross-type pattern detection
   219→    # If the query contains strong signals for a different type that isn't
   220→    # already covered by primary or related_types, pull those docs too.
   221→    if query:
   222→        query_lower = query.lower()
   223→        # Define cross-type signal patterns (query signals → should pull this type)
   224→        cross_type_signals = {
   225→            "refund_request": [r"\brefund\b", r"\bmoney\s+back\b", r"\breturn\s+my\b", r"\bprorated?\b", r"\beligibl", r"\btermination\s+fee\b"],
   226→            "billing": [r"\bcharg(?:ed|e|es|ing)\b", r"\binvoice\b", r"\bpayment\b", r"\bovercharge", r"\bdouble\s+charge\b", r"\bpricing\b"],
   227→            "technical": [r"\blogin\b", r"\bss[so]\b", r"\bpassword\b", r"\berror\b", r"\bcrash\b", r"\bnot\s+working\b", r"\bcan'?t\s+access\b", r"\bsession\b", r"\bcache\b"],
   228→            "faq": [r"\bwhat\s+(?:is|are)\b", r"\bhow\s+(?:do|does|much|many)\b", r"\bfeature", r"\b(?:plan|plans?)\s*(?:pricing|cost|include)\b"],
   229→            "complaint": [r"\bterrible\b", r"\bworst\b", r"\bunacceptab", r"\bfrustrat", r"\bdisappoint", r"\bnever\s+again\b"],
   230→            "account_change": [r"\bchange\s+(?:email|password)\b", r"\bupgrade\b", r"\bdowngrade\b", r"\bswitch\s+plan\b", r"\bworkspace\b", r"\bteam\s+member\b"],
   231→        }
   232→        # Determine which types are already covered
   233→        covered_types = {ticket_type} | set(related_types.get(ticket_type, []))
   234→        for cross_type, patterns in cross_type_signals.items():
   235→            if cross_type in covered_types:
   236→                continue
   237→            # If 2+ patterns match, this type is likely relevant
   238→            match_count = sum(1 for pat in patterns if re.search(pat, query_lower))
   239→            if match_count >= 2:
   240→                extra_docs = KNOWLEDGE_BASE.get(cross_type, [])
   241→                docs.extend(extra_docs)
   242→
   243→    # Deduplicate by source
   244→    seen = set()
   245→    unique_docs = []
   246→    for d in docs:
   247→        if d["source"] not in seen:
   248→            seen.add(d["source"])
   249→            unique_docs.append({
   250→                "content": d["content"],
   251→                "source": d["source"],
   252→                "section": d.get("section", "C"),
   253→            })
   254→
   255→    return unique_docs

   258→# ── Phase 4: Smart knowledge filtering (non-LLM) ─────────────────
   259→
   260→
   261→def _filter_relevant_docs(
   262→    documents: List[Dict[str, Any]], query: str, ticket_type: str
   263→) -> List[Dict[str, Any]]:
   264→    """Score each doc by LLM-based semantic relevance to the query.
   265→
   266→    Uses Llama 3.1 8B to rate each KB doc's relevance (0-10) to the
   267→    customer's query. This replaces the old keyword-overlap scoring
   268→    which couldn't understand semantic meaning.
   269→
   270→    Falls back to keyword scoring if LLM is unavailable (BC-008).
   271→    """
   272→    if not documents:
   273→        return documents
   274→
   275→    # ── LLM-based semantic relevance scoring ─────────────────────
   276→    # Ask Llama 3.1 to rate each doc's relevance to the query (0-10).
   277→    # This is the "knowledge base" — the LLM reads the query + doc title
   278→    # and scores how relevant it is. Much better than keyword matching.
   279→    try:
   280→        import asyncio
   281→        from app.core.parwa_pipeline.llm_client import llm_call
   282→
   283→        # Build a single prompt that scores ALL docs at once (1 LLM call, not N)
   284→        doc_list = []
   285→        for i, doc in enumerate(documents):
   286→            title = doc.get("source", "unknown")
   287→            content_preview = doc.get("content", "")[:200]
   288→            doc_list.append(f"DOC {i}: [{title}] {content_preview}")
   289→
   290→        prompt = f"""Rate the relevance of each document to the customer's question (0-10).
   291→
   292→Customer question: "{query}"
   293→
   294→Documents:
   295→{chr(10).join(doc_list)}
   296→
   297→Reply with ONLY the doc numbers and scores, one per line:
   298→0: <score>
   299→1: <score>
   300→2: <score>
   301→..."""
   302→
   303→        async def _score_docs():
   304→            return await llm_call(prompt, max_tokens=100, temperature=0.1)
   305→
   306→        try:
   307→            loop = asyncio.get_event_loop()
   308→            if loop.is_running():
   309→                # Already in async context — run in thread
   310→                import concurrent.futures
   311→                with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   312→                    result = pool.submit(asyncio.run, _score_docs).result(timeout=30)
   313→            else:
   314→                result = asyncio.run(_score_docs())
   315→        except Exception:
   316→            result = asyncio.run(_score_docs())
   317→
   318→        # Parse scores from LLM response
   319→        scores: Dict[int, int] = {}
   320→        for line in result.strip().split("\n"):
   321→            parts = line.strip().split(":")
   322→            if len(parts) == 2:
   323→                try:
   324→                    doc_idx = int(parts[0].strip())
   325→                    score = int(parts[1].strip())
   326→                    scores[doc_idx] = max(0, min(10, score))
   327→                except ValueError:
   328→                    continue
   329→
   330→        # If we got scores, sort docs by LLM relevance score
   331→        if scores and len(scores) >= len(documents) // 2:
   332→            scored_docs = []
   333→            for i, doc in enumerate(documents):
   334→                score = scores.get(i, 0)
   335→                scored_docs.append((score, doc))
   336→            scored_docs.sort(key=lambda x: x[0], reverse=True)
   337→
   338→            # Keep top 8 docs with score >= 3
   339→            filtered = [doc for score, doc in scored_docs if score >= 3][:8]
   340→            if filtered:
   341→                return filtered
   342→            # If all scored low, keep top 5 anyway
   343→            return [doc for _, doc in scored_docs[:5]]
   344→
   345→        # If LLM scoring failed, fall through to keyword scoring below
   346→        logger.warning("LLM relevance scoring failed — falling back to keyword filter")
   347→
   348→    except Exception as exc:
   349→        logger.warning("LLM relevance scoring error: %s — falling back to keyword filter", str(exc)[:200])
   350→
   351→    # ── Fallback: keyword-based relevance scoring (original logic) ──
   352→    query_lower = query.lower()
   353→    query_terms = set(w.lower() for w in query_lower.split() if len(w) > 3)
   354→    type_keywords = {
   355→        "refund_request": {"refund", "prorated", "termination", "fee", "credit", "eligible", "days", "annual", "monthly"},
   356→        "billing": {"charge", "invoice", "payment", "billed", "subscription", "duplicate", "price", "plan", "rate"},
   357→        "technical": {"login", "sso", "password", "access", "error", "cache", "sync", "session", "lockout", "certificate"},
   358→        "faq": {"plan", "pricing", "feature", "channel", "agent", "ticket", "onboarding", "integration"},
   359→        "complaint": {"complaint", "compensation", "credit", "escalate", "resolution", "priority", "tenure"},
   360→        "account_change": {"email", "password", "plan", "upgrade", "downgrade", "workspace", "member", "sso"},
   361→    }
   362→    query_terms |= type_keywords.get(ticket_type, set())
   363→
   364→    filler = {"that", "this", "have", "been", "will", "would", "could", "should",
   365→              "their", "there", "about", "which", "where", "when", "what", "with",
   366→              "from", "your", "just", "also", "than", "them", "they", "some"}
   367→    query_terms -= filler
   368→
   369→    scored_docs = []
   370→    for doc in documents:
   371→        text = doc.get("content", "").lower()
   372→        doc_words = set(text.split())
   373→        overlap = len(query_terms & doc_words)
   374→        query_bigrams = set()
   375→        words = query_lower.split()
   376→        for i in range(len(words) - 1):
   377→            bg = f"{words[i]} {words[i+1]}"
   378→            if len(bg) > 6:
   379→                query_bigrams.add(bg)
   380→        bigram_hits = sum(1 for bg in query_bigrams if bg in text)
   381→        score = overlap + (bigram_hits * 2)
   382→        scored_docs.append((score, doc))
   383→
   384→    scored_docs.sort(key=lambda x: x[0], reverse=True)
   385→    if len(scored_docs) > 8:
   386→        scored_docs = scored_docs[:8]
   387→
   388→    return [doc for _, doc in scored_docs]

   391→# ── Phase 4: Non-LLM CLARA sufficiency check (replaces 1 LLM call) ─
   392→
   393→
   394→def _check_knowledge_sufficiency(
   395→    documents: List[Dict[str, Any]], query: str, ticket_type: str
   396→) -> Dict[str, bool]:
   397→    """Non-LLM heuristic: do we have KB docs for this ticket type?
   398→    Replaces the LLM-based CLARA re-evaluate from Phase 2."""
   399→    has_primary = any(d.get("source", "") in {
   400→        "refund_policy_v2", "billing_policy_v2", "tech_troubleshooting_v2",
   401→        "general_faq_v2", "complaint_handling_v2", "account_policy_v2",
   402→    } for d in documents)
   403→
   404→    # If we have docs for the primary type, knowledge is likely sufficient
   405→    type_doc_counts = {
   406→        "refund_request": 4, "billing": 3, "technical": 2,
   407→        "faq": 3, "complaint": 1, "account_change": 2,
   408→    }
   409→    expected = type_doc_counts.get(ticket_type, 2)
   410→    sufficient = len(documents) >= max(expected - 1, 1) and has_primary
   411→
   412→    return {
   413→        "knowledge_sufficient": sufficient,
   414→        "knowledge_contradictory": False,  # our KB is internally consistent
   415→    }

   421→def _check_contradictions(documents: List[Dict[str, Any]]) -> bool:
   422→    """Quick keyword-level contradiction check."""
   423→    if len(documents) < 2:
   424→        return False
   425→    all_numbers = {}
   426→    for doc in documents:
   427→        text = doc.get("content", "")
   428→        for match in re.finditer(r"(\d+)%", text):
   429→            num = int(match.group(1))
   430→            context = text[max(0, match.start()-30):match.end()+30].lower()
   431→            for keyword in ["refund", "fee", "credit", "discount"]:
   432→                if keyword in context:
   433→                    if keyword in all_numbers and all_numbers[keyword] != num:
   434→                        return True
   435→                    all_numbers[keyword] = num
   436→    return False

   439→def _read_ai_wiki(tenant_id: str, ticket_type: str, query: str, tier: str = "parwa") -> tuple:
   440→    """Phase 6: Read AI Wiki Sections A, B, C from real store.
   441→    
   442→    Returns (section_a_entries, section_b_entries, section_c_entries)
   443→    as lists of dicts in node format (same shape as KB docs).
   444→    All non-LLM — keyword search only.
   445→    """
   446→    from app.core.parwa_pipeline.ai_wiki_store import get_wiki_store
   447→    
   448→    wiki = get_wiki_store()
   449→    wiki_a_raw = []
   450→    wiki_b_raw = []
   451→    wiki_c_raw = []
   452→    
   453→    try:
   454→        # Section A: Search for similar ticket patterns (top 3)
   455→        wiki_a_raw = wiki.search(
   456→            tenant_id=tenant_id, section="A", query=query,
   457→            ticket_type=ticket_type, tier=tier, max_results=3,
   458→        )
   459→        
   460→        # Section B: Read admin behavior patterns (all, max 2)
   461→        wiki_b_raw = wiki.read(tenant_id=tenant_id, section="B", tier=tier)
   462→        wiki_b_raw = wiki_b_raw[:2]  # limit for context window
   463→        
   464→        # Section C: Read company knowledge (all, max 3)
   465→        wiki_c_raw = wiki.read(tenant_id=tenant_id, section="C", tier=tier)
   466→        wiki_c_raw = wiki_c_raw[:3]
   467→    except Exception as e:
   468→        logger.warning("Wiki read error (non-fatal): %s", e)
   469→    
   470→    # Convert to node format (same shape as KB docs)
   471→    wiki_a = [e.to_node_format() for e in wiki_a_raw]
   472→    wiki_b = [e.to_node_format() for e in wiki_b_raw]
   473→    wiki_c = [e.to_node_format() for e in wiki_c_raw]
   474→    
   475→    # Build wiki summary for similar patterns found
   476→    wiki_patterns = []
   477→    for entry in wiki_a_raw:

   481→            "techniques_that_worked": c.get("techniques_that_worked", []),
   482→            "quality_achieved": c.get("quality_achieved", 0),
   483→            "answer_summary": c.get("answer_summary", "")[:200],
   484→            "historical_success_rate": round(entry.success_count / max(entry.usage_count, 1), 3),
   485→        })
   486→    
   487→    return wiki_a, wiki_b, wiki_c, wiki_patterns

   684→async def node_3_knowledge_fetch(state: PipelineV2State) -> dict:
   685→    """Node 3: Knowledge Fetch — Phase 4 optimized.
   686→    LLM calls: 1 (was 5) — removed HyDE, MultiQuery, StepBack, CLARA re-evaluate."""
   687→    start = time.time()
   688→    query = state.get("query", "")
   689→    tenant_id = state.get("tenant_id", "")
   690→    ticket_type = state.get("ticket_type", "general")
   691→    logs = []
   692→    llm_calls = 0
   693→
   694→    # ── Wave 4: Inject Jarvis guidance for this ticket ─────
   695→    system_flags = state.get("system_flags")
   696→    if not system_flags:
   697→        try:
   698→            from app.core.parwa_pipeline.parwa_bridge import load_system_flags
   699→            system_flags = await load_system_flags(tenant_id)
   700→        except Exception:
   701→            system_flags = {}
   702→    guidance_map = system_flags.get("guidance", {})
   703→    ticket_id = state.get("ticket_id", "")
   704→    jarvis_guidance = guidance_map.get(ticket_id, "")
   705→    if jarvis_guidance:
   706→        logs.append({"node": 3, "technique": "JARVIS_GUIDANCE", "duration_ms": 0,
   707→                     "result_summary": f"guidance_len={len(jarvis_guidance)}"})
   708→
   709→    # 1. CLARA: Gatekeeper (LLM) — the ONLY LLM call in Node 3 now
   710→    clara_result = await _clara_gatekeep(query, ticket_type)
   711→    logs.append({"node": 3, "technique": "CLARA", "duration_ms": 0, "result_summary": "gatekeep_done"})
   712→    llm_calls += 1
   713→
   714→    # 2. RAG Retrieval (per-tenant KB first, fallback to shared default)
   715→    documents = _retrieve_knowledge(ticket_type, query, tenant_id=tenant_id)
   716→    logs.append({"node": 3, "technique": "RAG", "duration_ms": 0, "result_summary": f"{len(documents)} docs"})
   717→
   718→    # 3. Smart knowledge filtering (Phase 4: non-LLM relevance ranking)
   719→    filtered = _filter_relevant_docs(documents, query, ticket_type)
   720→    logs.append({"node": 3, "technique": "SmartFilter", "duration_ms": 0, "result_summary": f"{len(documents)}→{len(filtered)}"})
   721→
   722→    # 4. Knowledge sufficiency check (Phase 4: non-LLM, replaces CLARA re-evaluate LLM call)
   723→    sufficiency = _check_knowledge_sufficiency(filtered, query, ticket_type)
   724→    clara_result["knowledge_sufficient"] = sufficiency["knowledge_sufficient"]
   725→    clara_result["knowledge_contradictory"] = sufficiency["knowledge_contradictory"]
   726→    logs.append({"node": 3, "technique": "SufficiencyCheck", "duration_ms": 0,
   727→                 "result_summary": f"sufficient={sufficiency['knowledge_sufficient']}"})
   728→
   729→    # ── P1 Notification: emit ticket:knowledge_gap ─────────────────
   730→    # When the AI can't find enough knowledge to answer confidently, tell the
   731→    # human so they know the AI is working blind. This helps the human decide
   732→    # whether to intervene (provide more KB docs) or let the AI try anyway.
   733→    if not sufficiency["knowledge_sufficient"]:
   734→        # ── APPROACH A: Pause + ask for guidance ───────────────────
   735→        # Node 3 has doubt (KB insufficient). Instead of continuing blind,
   736→        # interrupt the pipeline and ask for guidance. When resumed,
   737→        # the answer is injected into the knowledge context.
   738→        #
   739→        # IMPORTANT: interrupt() works by raising a special exception that
   740→        # LangGraph catches internally to pause the graph. We must NOT
   741→        # catch that exception — let it propagate so the graph pauses.
   742→        from langgraph.types import interrupt
   743→        guidance = interrupt({
   744→            "node": 3,
   745→            "question": (
   746→                f"I don't have enough knowledge base documents to answer "
   747→                f"this {ticket_type} ticket confidently. "
   748→                f"Only {len(filtered)} relevant doc(s) found. "
   749→                f"Can you provide guidance on how to handle this?"
   750→            ),
   751→            "ticket_id": state.get("ticket_id", ""),
   752→            "ticket_type": ticket_type,
   753→            "docs_found": len(filtered),
   754→            "query_summary": query[:200],
   755→        })
   756→        # ── When resumed, execution continues HERE ─────────────────
   757→        # `guidance` contains the human/variant's answer.
   758→        # Inject it as a knowledge doc so downstream nodes can use it.
   759→        logger.info(
   760→            "Node 3 resumed with guidance: %s (ticket=%s)",
   761→            str(guidance)[:100], state.get("ticket_id", ""),
   762→        )
   763→        filtered.append({
   764→            "content": f"Human/Agent Guidance: {guidance}",
   765→            "source": "human_guidance",
   766→            "score": 1.0,
   767→        })
   768→        clara_result["knowledge_sufficient"] = True
   769→        logs.append({
   770→            "node": 3, "technique": "HumanGuidanceInjection",
   771→            "duration_ms": 0,
   772→            "result_summary": f"guidance injected ({len(str(guidance))} chars)",
   773→        })
   774→
   775→        try:
   776→            from app.core.event_emitter import emit_ticket_event
   777→            await emit_ticket_event(
   778→                company_id=tenant_id,
   779→                event_type="ticket:knowledge_gap",
   780→                payload={
   781→                    "company_id": tenant_id,
   782→                    "ticket_id": state.get("ticket_id", ""),
   783→                    "ticket_type": ticket_type,
   784→                    "query": query[:500],
   785→                    "docs_found": len(filtered),
   786→                    "contradictory": sufficiency["knowledge_contradictory"],
   787→                    "node": 3,
   788→                },
   789→                correlation_id=state.get("ticket_id", ""),
   790→            )
   791→        except Exception as exc:
   792→            logger.warning("node_3_knowledge_gap_notification_failed: %s", str(exc)[:200])
   793→
   794→    # 5. Contradiction check (non-LLM)
   795→    has_contradiction = _check_contradictions(filtered)
   796→    if has_contradiction:
   797→        clara_result["knowledge_contradictory"] = True
   798→        logs.append({"node": 3, "technique": "ContradictionCheck", "duration_ms": 0, "result_summary": "CONTRADICTION_FOUND"})
   799→
   800→    # 6. DynamicContext (non-LLM)
   801→    dynamic_ctx = state.get("customer_context", {})
   802→    logs.append({"node": 3, "technique": "DynamicContext", "duration_ms": 0, "result_summary": "context_pulled"})
   803→
   804→    # 7. AI Wiki (Phase 6: real store reads)
   805→    tier = state.get("variant_tier", "parwa")
   806→    wiki_a, wiki_b, wiki_c, wiki_patterns = _read_ai_wiki(tenant_id, ticket_type, query, tier)
   807→    wiki_log_msg = f"A={len(wiki_a)} B={len(wiki_b)} C={len(wiki_c)}"
   808→    if wiki_patterns:
   809→        wiki_log_msg += f" patterns_found={len(wiki_patterns)}"
   810→    logs.append({"node": 3, "technique": "AIWiki", "duration_ms": 0,
   811→                 "result_summary": wiki_log_msg})
   812→    
   813→    # 7b. Policy sync check (Phase 6: detect version changes)
   814→    sync_status = {"synced": True, "version": "v2.0", "previous_version": None}
   815→    try:
   816→        from app.core.parwa_pipeline.ai_wiki_store import get_wiki_store
   817→        wiki_store = get_wiki_store()
   818→        current_policy_version = state.get("policy_version", "v2.0")
   819→        sync_status = wiki_store.check_policy_sync(tenant_id, current_policy_version)
   820→        if not sync_status["synced"]:
   821→            logs.append({
   822→                "node": 3, "technique": "PolicySyncCheck",
   823→                "duration_ms": 0,
   824→                "result_summary": f"POLICY_CHANGED {sync_status['previous_version']} → {sync_status['version']} ({sync_status.get('patterns_invalidated', 0)} invalidated)",
   825→            })
   826→    except Exception as e:
   827→        logger.warning("Policy sync check failed (non-fatal): %s", e)
   828→
   829→    # 8. CRM + E-commerce + Carrier data (real, via crm_actions/ecommerce_actions/carrier_api_connector)
   830→    crm_data = await _fetch_crm_data(tenant_id, dynamic_ctx)
   831→    crm_summary = "crm_fetched"
   832→    if crm_data.get("crm_contact"):
   833→        crm_summary += "+hubspot"
   834→    if crm_data.get("ecommerce_orders"):
   835→        crm_summary += "+shopify"
   836→    if crm_data.get("carrier_tracking"):
   837→        crm_summary += "+carrier"
   838→    logs.append({"node": 3, "technique": "UCB", "duration_ms": 0, "result_summary": crm_summary})
   839→
   840→    elapsed = int((time.time() - start) * 1000)
   841→    logger.info(
   842→        "Node 3 complete: ticket=%s docs=%d sufficient=%s llm=%d [%dms]",
   843→        state["ticket_id"], len(filtered), clara_result["knowledge_sufficient"], llm_calls, elapsed,
   844→    )
   845→
   846→    # Wave 4: Inject Jarvis guidance as additional knowledge
   847→    if jarvis_guidance:
   848→        filtered.append({
   849→            "source": "jarvis_guidance",

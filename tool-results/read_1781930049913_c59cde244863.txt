     1→"""
     2→Variant Service: Config resolution for the Variant Engine.
     3→
     4→Given a variant_tier + industry (or variant_instance_id), this service
     5→returns the FULL pipeline config — which nodes to run, with what settings,
     6→what model to use, what system prompt, what tools are available.
     7→
     8→This is the SINGLE SOURCE OF TRUTH for all variant×industry configurations.
     9→No other module should hardcode variant configs — they all ask this service.
    10→
    11→Architecture:
    12→  resolve(tier, industry) → VariantConfig
    13→  resolve_by_instance(instance_id) → VariantConfig (looks up tier + industry)
    14→
    15→  VariantConfig contains EVERYTHING the pipeline needs:
    16→    - Which steps to run
    17→    - What LLM model each step should use
    18→    - Token budgets per step
    19→    - System prompts (industry-aware)
    20→    - Available tools (industry-aware)
    21→    - Quality thresholds
    22→    - Cost limits
    23→    - Timeout settings
    24→
    25→Design:
    26→  - Configs are built from a 2D matrix: VARIANT_CONFIGS[tier][industry]
    27→  - Industry-specific overrides layer on top of tier defaults
    28→  - Instance-level overrides layer on top of industry configs
    29→  - Priority: instance > industry > tier > global default
    30→
    31→BC-001: company_id first parameter on public methods.
    32→BC-008: Every public method wrapped in try/except — never crash.
    33→BC-012: All timestamps UTC.
    34→"""

   375→# VARIANT SERVICE
   376→# ══════════════════════════════════════════════════════════════════
   377→
   378→
   379→class VariantService:
   380→    """Single source of truth for variant×industry configuration.
   381→
   382→    Usage:
   383→        service = VariantService()
   384→
   385→        # Resolve by tier + industry
   386→        config = service.resolve("mini_parwa", "ecommerce")
   387→
   388→        # Resolve by instance (looks up tier + industry from DB)
   389→        config = service.resolve_by_instance("inst_abc123")
   390→
   391→        # Access config fields
   392→        config.steps              → ['pii_check', 'empathy_check', ...]
   393→        config.generation_model   → 'gpt-4o-mini'
   394→        config.system_prompt      → 'You are a customer service...'
   395→        config.available_tools    → ['order_tracker', ...]
   396→    """
   397→
   398→    def __init__(self) -> None:
   399→        """Initialize the variant service."""
   400→        logger.info(
   401→            "VariantService initialized — %d tiers, %d industries",
   402→            len(TIER_DEFAULTS),
   403→            len(INDUSTRY_OVERRIDES),
   404→        )
   405→
   406→    def resolve(
   407→        self,
   408→        variant_tier: str,
   409→        industry: str = "general",
   410→    ) -> VariantConfig:
   411→        """Resolve full config for a variant_tier + industry combination.
   412→
   413→        Priority stack (later overrides earlier):
   414→          1. Global tier defaults (TIER_DEFAULTS)
   415→          2. Industry overrides (INDUSTRY_OVERRIDES)
   416→          3. Instance-level overrides (future: from DB)
   417→
   418→        Args:
   419→            variant_tier: 'mini_parwa' | 'parwa' | 'parwa_high'.
   420→            industry: 'ecommerce' | 'logistics' | 'saas' | 'general'.
   421→
   422→        Returns:
   423→            Fully resolved VariantConfig.
   424→        """
   425→        try:
   426→            # 1. Start with tier defaults
   427→            tier_config = TIER_DEFAULTS.get(
   428→                variant_tier,
   429→                TIER_DEFAULTS["parwa"],  # safe fallback
   430→            )
   431→
   432→            # 2. Apply industry overrides
   433→            industry_overrides = INDUSTRY_OVERRIDES.get(industry, {})
   434→            tier_industry_overrides = industry_overrides.get(variant_tier, {})
   435→
   436→            # 3. Merge: tier defaults + industry overrides
   437→            merged = {**tier_config, **tier_industry_overrides}
   438→
   439→            # 4. Resolve steps (it's a callable in tier defaults)
   440→            steps_fn = merged.get("steps", get_pro_pipeline_steps)
   441→            if callable(steps_fn):
   442→                steps = steps_fn()
   443→            else:
   444→                steps = list(steps_fn)
   445→
   446→            # 5. Build step configs (disable steps not in this pipeline)
   447→            step_configs: Dict[str, StepConfig] = {}
   448→            for step_id in ALL_NODES:
   449→                default = STEP_DEFAULTS.get(step_id, StepConfig(step_id=step_id))
   450→                step_configs[step_id] = StepConfig(
   451→                    step_id=step_id,
   452→                    enabled=(step_id in steps),
   453→                    model=merged.get(
   454→                        f"{step_id}_model",
   455→                        default.model,
   456→                    ),
   457→                    max_tokens=default.max_tokens,
   458→                    timeout_seconds=default.timeout_seconds,
   459→                    cost_weight=default.cost_weight,
   460→                )
   461→
   462→            # Override classification/generation/quality models from merged config
   463→            if "classification_model" in merged:
   464→                step_configs["classify"].model = merged["classification_model"]
   465→            if "generation_model" in merged:
   466→                step_configs["generate"].model = merged["generation_model"]
   467→            if "quality_model" in merged:
   468→                step_configs["quality_gate"].model = merged["quality_model"]
   469→
   470→            # 6. Get industry-specific settings
   471→            system_prompt = get_industry_prompt(industry)
   472→            available_tools = get_industry_tools(industry)
   473→            response_tone = get_industry_tone(industry)
   474→
   475→            # 7. Build the final config
   476→            config = VariantConfig(
   477→                variant_tier=variant_tier,
   478→                industry=industry,
   479→                steps=steps,
   480→                step_configs=step_configs,
   481→                max_total_tokens=merged.get("max_total_tokens", 1000),
   482→                max_total_latency_ms=merged.get("max_total_latency_ms", 20000.0),
   483→                max_cost_usd=merged.get("max_cost_usd", 0.01),
   484→                classification_model=merged.get("classification_model", "gpt-4o-mini"),
   485→                generation_model=merged.get("generation_model", "gpt-4o-mini"),
   486→                quality_model=merged.get("quality_model", "gpt-4o-mini"),
   487→                quality_threshold=merged.get("quality_threshold", 0.7),
   488→                quality_max_retries=merged.get("quality_max_retries", 1),
   489→                system_prompt=system_prompt,
   490→                available_tools=available_tools,
   491→                response_tone=response_tone,
   492→                techniques_allowed=merged.get("techniques_allowed", []),
   493→                technique_tier_max=merged.get("technique_tier_max", 1),
   494→                enable_context_compression=merged.get("enable_context_compression", False),
   495→                enable_context_health=merged.get("enable_context_health", False),
   496→                enable_dedup=merged.get("enable_dedup", False),
   497→                cost_per_query_estimate=merged.get("cost_per_query_estimate", 0.003),
   498→                billing_rate_per_1k_tokens=merged.get("billing_rate_per_1k_tokens", 0.00015),
   499→            )
   500→
   501→            logger.info(
   502→                "VariantConfig resolved: tier=%s, industry=%s, "
   503→                "steps=%d, model=%s, cost_est=$%.4f",
   504→                variant_tier, industry, len(steps),
   505→                config.generation_model,
   506→                config.cost_per_query_estimate,
   507→            )
   508→
   509→            return config
   510→
   511→        except Exception:
   512→            logger.exception(
   513→                "VariantService.resolve failed for tier=%s, industry=%s — "
   514→                "returning safe fallback config",
   515→                variant_tier, industry,
   516→            )
   517→            # BC-008: Return a safe minimal config
   518→            return VariantConfig(
   519→                variant_tier=variant_tier or "mini_parwa",
   520→                industry=industry or "general",
   521→                steps=get_mini_pipeline_steps(),
   522→                max_total_tokens=500,
   523→                generation_model="gpt-4o-mini",
   524→                cost_per_query_estimate=0.003,
   525→                system_prompt=get_industry_prompt("general"),
   526→                available_tools=get_industry_tools("general"),
   527→                response_tone="professional_adaptable",
   528→            )
   529→
   530→    def resolve_by_instance(
   531→        self,
   532→        company_id: str,
   533→        instance_id: str,
   534→        db: Any = None,
   535→    ) -> VariantConfig:
   536→        """Resolve config by looking up instance details from DB.
   537→
   538→        1. Look up instance_id → variant_tier, industry
   539→        2. Call resolve(tier, industry) for base config
   540→        3. Apply instance-level overrides (from DB)
   541→
   542→        If a db session is provided, looks up the VariantInstance
   543→        from the database. Otherwise, falls back to tier+industry
   544→        resolution using a heuristic from the instance_id naming
   545→        convention (inst_{variant_type}_{company_id}).
   546→
   547→        Args:
   548→            company_id: Tenant identifier (BC-001).
   549→            instance_id: Variant instance identifier.
   550→            db: Optional SQLAlchemy session for DB lookups.
   551→
   552→        Returns:
   553→            Fully resolved VariantConfig with instance overrides.
   554→        """
   555→        try:
   556→            variant_tier = "parwa"
   557→            industry = "general"
   558→
   559→            # ── DB-backed lookup ──
   560→            if db is not None:
   561→                try:
   562→                    from database.models.variant_engine import VariantInstance
   563→                    instance = (
   564→                        db.query(VariantInstance)
   565→                        .filter(
   566→                            VariantInstance.id == instance_id,
   567→                            VariantInstance.company_id == company_id,
   568→                        )
   569→                        .first()
   570→                    )
   571→                    if instance is not None:
   572→                        variant_tier = instance.variant_type
   573→                        # Try to derive industry from company settings
   574→                        try:
   575→                            from database.models.core import CompanySetting
   576→                            setting = (
   577→                                db.query(CompanySetting)
   578→                                .filter(
   579→                                    CompanySetting.company_id == company_id,
   580→                                )
   581→                                .first()
   582→                            )
   583→                            if setting and hasattr(setting, "industry"):
   584→                                industry = setting.industry or "general"
   585→                        except Exception:
   586→                            pass  # Industry resolution is best-effort
   587→
   588→                        logger.info(
   589→                            "resolve_by_instance: DB lookup succeeded — "
   590→                            "instance_id=%s, company_id=%s, tier=%s, industry=%s",
   591→                            instance_id, company_id, variant_tier, industry,
   592→                        )
   593→                    else:
   594→                        logger.warning(
   595→                            "resolve_by_instance: instance_id=%s not found "
   596→                            "in DB for company_id=%s — using heuristic",
   597→                            instance_id, company_id,
   598→                        )
   599→                except ImportError:
   600→                    logger.warning(
   601→                        "resolve_by_instance: VariantInstance model not "
   602→                        "available — using heuristic fallback",
   603→                    )
   604→
   605→            # ── Heuristic fallback from instance_id naming convention ──
   606→            if db is None or variant_tier == "parwa":
   607→                # Try to extract variant type from instance_id pattern:
   608→                # inst_{variant_type}_{company_id} or inst_onboarding_{variant_type}_{company_id}
   609→                if "mini_parwa" in instance_id or "mini" in instance_id:
   610→                    variant_tier = "mini_parwa"
   611→                elif "parwa_high" in instance_id or "high" in instance_id:
   612→                    variant_tier = "parwa_high"
   613→                elif "parwa" in instance_id:
   614→                    variant_tier = "parwa"
   615→
   616→            # Resolve base config
   617→            config = self.resolve(variant_tier, industry)
   618→            config.instance_id = instance_id
   619→
   620→            # ── Apply instance-level overrides from DB ──
   621→            if db is not None:
   622→                try:
   623→                    from database.models.variant_engine import VariantAICapability
   624→                    import json
   625→
   626→                    overrides = (
   627→                        db.query(VariantAICapability)
   628→                        .filter(
   629→                            VariantAICapability.company_id == company_id,
   630→                            VariantAICapability.instance_id == instance_id,
   631→                            VariantAICapability.is_enabled == True,
   632→                        )
   633→                        .all()
   634→                    )
   635→
   636→                    for cap in overrides:
   637→                        if cap.config_json:
   638→                            try:
   639→                                override_config = json.loads(cap.config_json)
   640→                                # Apply model overrides
   641→                                if "generation_model" in override_config:
   642→                                    config.generation_model = override_config["generation_model"]
   643→                                if "classification_model" in override_config:
   644→                                    config.classification_model = override_config["classification_model"]
   645→                                if "quality_model" in override_config:
   646→                                    config.quality_model = override_config["quality_model"]
   647→                                if "quality_threshold" in override_config:
   648→                                    config.quality_threshold = float(
   649→                                        override_config["quality_threshold"]
   650→                                    )
   651→                                if "max_total_tokens" in override_config:
   652→                                    config.max_total_tokens = int(
   653→                                        override_config["max_total_tokens"]
   654→                                    )
   655→                            except (json.JSONDecodeError, TypeError, ValueError):
   656→                                pass
   657→
   658→                    if overrides:
   659→                        logger.info(
   660→                            "resolve_by_instance: applied %d instance-level "
   661→                            "overrides for instance_id=%s",
   662→                            len(overrides), instance_id,
   663→                        )
   664→                except ImportError:
   665→                    pass  # VariantAICapability not available — skip overrides
   666→
   667→            return config
   668→
   669→        except Exception:
   670→            logger.exception(
   671→                "resolve_by_instance failed for company_id=%s, "
   672→                "instance_id=%s — returning safe fallback",
   673→                company_id, instance_id,
   674→            )
   675→            return self.resolve("mini_parwa", "general")

   677→    def resolve_for_shadow(
   678→        self,
   679→        company_id: str,
   680→        live_variant: str,
   681→        shadow_variant: str,
   682→        industry: str = "general",
   683→    ) -> Tuple[VariantConfig, VariantConfig]:
   684→        """Resolve configs for both live and shadow variants.
   685→
   686→        Used by the Shadow Mode pipeline bridge to get both
   687→        configurations simultaneously for comparison processing.
   688→
   689→        Args:
   690→            company_id: Tenant identifier (BC-001).
   691→            live_variant: Current live variant tier.
   692→            shadow_variant: Variant being tested in shadow.
   693→            industry: Industry vertical.
   694→
   695→        Returns:
   696→            Tuple of (live_config, shadow_config).
   697→        """
   698→        try:
   699→            live_config = self.resolve(live_variant, industry)
   700→            shadow_config = self.resolve(shadow_variant, industry)
   701→
   702→            logger.info(
   703→                "resolve_for_shadow: company_id=%s, live=%s, shadow=%s, "
   704→                "industry=%s — live_steps=%d, shadow_steps=%d",
   705→                company_id, live_variant, shadow_variant, industry,
   706→                len(live_config.steps), len(shadow_config.steps),
   707→            )
   708→
   709→            return (live_config, shadow_config)
   710→
   711→        except Exception:
   712→            logger.exception(
   713→                "resolve_for_shadow failed for company_id=%s — "
   714→                "returning safe fallback configs",
   715→                company_id,
   716→            )
   717→            return (self.resolve("mini_parwa", "general"),
   718→                    self.resolve("parwa", "general"))

   720→    def get_all_configs(self) -> Dict[str, Dict[str, VariantConfig]]:
   721→        """Get all variant×industry configurations.

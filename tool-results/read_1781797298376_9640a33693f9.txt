   428→# IN-MEMORY BACKEND (works immediately, no deps)
   429→# ═══════════════════════════════════════════════════════════════
   430→
   431→class InMemoryBackend(StorageBackend):
   432→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   433→
   434→    def __init__(self):
   435→        self._notifications: Dict[str, Dict] = {}
   436→        self._flags: List[Dict] = []
   437→        self._audit: List[Dict] = []
   438→        self._quality_scores: List[Dict] = []
   439→        self._batches: Dict[str, List[Dict]] = {}
   440→        self._batch_metas: Dict[str, Dict] = {}
   441→        self._lock = threading.Lock()
   442→        self._next_nfy_id = 1
   443→        # Wave 2 stores
   444→        self._integration_pings: List[Dict] = []
   445→        self._llm_costs: List[Dict] = []
   446→        self._stuck_ticket_events: List[Dict] = []
   447→        self._load_status: Dict[str, Dict] = {}  # tenant_id -> {variant -> {concurrent, max_concurrent}}
   448→        self._agent_configs: List[Dict] = []  # Wave 3: agent config store
   449→        self._outbox: List[Dict] = []  # Wave 3: outbox queue (recall/void)
   450→        self._inbox: List[Dict] = []  # Wave 4: PARWA → Jarvis inbox
   451→        self._training_data_records: List[Dict] = []  # Wave 4: training data
   452→
   453→    # ── Helpers ───────────────────────────────────────────────
   454→
   455→    def _next_notification_number(self) -> int:
   456→        with self._lock:
   457→            n = self._next_nfy_id
   458→            self._next_nfy_id += 1
   459→            return n
   460→
   461→    @staticmethod
   462→    def _priority_from_score(score: float) -> str:
   463→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:
   464→            return PRIORITY_CRITICAL
   465→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_HIGH]:
   466→            return PRIORITY_HIGH
   467→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_MEDIUM]:
   468→            return PRIORITY_MEDIUM
   469→        return PRIORITY_LOW
   470→
   471→    # ── Notifications ────────────────────────────────────────
   472→
   473→    async def create_notification(
   474→        self,
   475→        tenant_id: str,
   476→        ntype: str,
   477→        priority_score: float,
   478→        title: str,
   479→        description: str,
   480→        related_tickets: Optional[List[str]] = None,
   481→        batch_key: Optional[str] = None,
   482→        source_data: Optional[Dict] = None,
   483→    ) -> Dict[str, Any]:
   484→        num = self._next_notification_number()
   485→        key = f"PARWA-NFY-{num:03d}"
   486→        now = time.time()
   487→        nf = {
   488→            "id": str(uuid.uuid4()),
   489→            "tenant_id": tenant_id,
   490→            "notification_key": key,
   491→            "type": ntype,
   492→            "priority": self._priority_from_score(priority_score),
   493→            "priority_score": round(priority_score, 4),
   494→            "title": title,
   495→            "description": description,
   496→            "related_tickets": related_tickets or [],
   497→            "batch_key": batch_key,
   498→            "source_data": source_data or {},
   499→            "is_read": False,
   500→            "is_resolved": False,
   501→            "created_at": now,
   502→            "resolved_at": None,
   503→            "backend": "memory",
   504→        }
   505→        self._notifications[key] = nf
   506→        logger.info("NFY created: %s type=%s score=%.3f [%s]", key, ntype, priority_score, tenant_id)
   507→        return nf
   508→
   509→    async def get_notification(self, key: str) -> Optional[Dict[str, Any]]:
   510→        return self._notifications.get(key)
   511→
   512→    async def get_notifications(
   513→        self,
   514→        tenant_id: str,
   515→        include_resolved: bool = False,
   516→        min_priority: Optional[str] = None,
   517→    ) -> List[Dict[str, Any]]:
   518→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   519→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   520→        results = []
   521→        for n in self._notifications.values():
   522→            if n["tenant_id"] != tenant_id:
   523→                continue
   524→            if not include_resolved and n["is_resolved"]:
   525→                continue
   526→            if priority_order.get(n["priority"], 99) > min_rank:
   527→                continue
   528→            results.append(n)
   529→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   530→        return results
   531→
   532→    async def resolve_notification(self, key: str) -> bool:
   533→        n = self._notifications.get(key)
   534→        if n and not n["is_resolved"]:
   535→            n["is_resolved"] = True
   536→            n["resolved_at"] = time.time()
   537→            return True
   538→        return False
   539→
   540→    async def dismiss_notification(self, key: str) -> bool:
   541→        n = self._notifications.get(key)
   542→        if n:
   543→            n["is_read"] = True
   544→            return True
   545→        return False
   546→
   547→    # ── System Flags ─────────────────────────────────────────
   548→
   549→    async def set_flag(
   550→        self,
   551→        tenant_id: str,
   552→        flag_type: str,
   553→        flag_value: str,
   554→        set_by: str,
   555→        scope: str = "global",
   556→        target_id: Optional[str] = None,
   557→        reason: Optional[str] = None,
   558→        expires_at: Optional[str] = None,
   559→    ) -> Dict[str, Any]:
   560→        flag = {
   561→            "id": str(uuid.uuid4()),
   562→            "tenant_id": tenant_id,
   563→            "flag_type": flag_type,
   564→            "flag_value": flag_value,
   565→            "scope": scope,
   566→            "target_id": target_id,
   567→            "set_by": set_by,
   568→            "reason": reason,
   569→            "expires_at": expires_at,
   570→            "created_at": datetime.now(timezone.utc).isoformat(),
   571→            "revoked_at": None,
   572→            "backend": "memory",
   573→        }
   574→        self._flags.append(flag)
   575→        logger.info("FLAG set: %s=%s scope=%s [%s]", flag_type, flag_value, scope, tenant_id)
   576→        return flag
   577→
   578→    async def get_active_flags(
   579→        self, tenant_id: str, flag_type: Optional[str] = None
   580→    ) -> List[Dict[str, Any]]:
   581→        now = datetime.now(timezone.utc)
   582→        active = []
   583→        for f in self._flags:
   584→            if f["tenant_id"] != tenant_id:
   585→                continue
   586→            if f["revoked_at"] is not None:
   587→                continue
   588→            if f["expires_at"]:
   589→                exp = datetime.fromisoformat(f["expires_at"]) if isinstance(f["expires_at"], str) else f["expires_at"]
   590→                if exp < now:
   591→                    continue
   592→            if flag_type and f["flag_type"] != flag_type:
   593→                continue
   594→            active.append(f)
   595→        return active
   596→
   597→    async def revoke_flag(self, flag_id: str, revoked_by: str) -> bool:
   598→        for f in self._flags:

   709→    # ── Batch Queue ──────────────────────────────────────────
   710→
   711→    async def add_to_batch(
   712→        self,
   713→        tenant_id: str,
   714→        batch_key: str,
   715→        ticket_id: str,
   716→        confidence: float,
   717→    ) -> Optional[Dict[str, Any]]:
   718→        BATCH_WINDOW_S = 300
   719→        now = time.time()
   720→
   721→        if tenant_id not in self._batch_metas:
   722→            self._batch_metas[tenant_id] = {}
   723→
   724→        # Find existing batch with same key within window
   725→        existing_key = None
   726→        for bk, meta in self._batch_metas[tenant_id].items():
   727→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   728→                existing_key = bk
   729→                break
   730→
   731→        if existing_key:
   732→            meta = self._batch_metas[tenant_id][existing_key]
   733→            meta["ticket_ids"].append(ticket_id)
   734→            meta["signal_count"] = len(meta["ticket_ids"])
   735→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   736→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   737→            # Don't flush yet — still within window
   738→            return None
   739→        else:
   740→            meta = {
   741→                "batch_key": batch_key,
   742→                "tenant_id": tenant_id,
   743→                "batch_start": now,
   744→                "ticket_ids": [ticket_id],
   745→                "signal_count": 1,
   746→                "confidence_min": confidence,
   747→                "confidence_max": confidence,
   748→                "status": "pending",
   749→            }
   750→            self._batch_metas[tenant_id][batch_key] = meta
   751→            return None

   881→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
   882→        """Compute health from recent pings (last 30 per service)."""
   883→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
   884→        services: Dict[str, List[Dict]] = {}
   885→        for p in tenant_pings:
   886→            services.setdefault(p["service_name"], []).append(p)
   887→
   888→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
   889→        for svc, pings in services.items():
   890→            recent = pings[-30:]  # last 30 pings
   891→            if not recent:
   892→                continue
   893→            healthy = sum(1 for p in recent if p["is_healthy"])
   894→            total = len(recent)
   895→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
   896→            last_ping = recent[-1]
   897→            avg_ms = (sum(p["response_ms"] or 0 for p in recent) / total) if total else 0
   898→            last_error = None
   899→            last_error_time = None
   900→            for p in reversed(recent):
   901→                if not p["is_healthy"]:
   902→                    last_error = p.get("error_detail")
   903→                    last_error_time = p["created_at"]
   904→                    break
   905→
   906→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
   907→            if status != "healthy":
   908→                result["degraded_count"] += 1
   909→            else:
   910→                result["healthy_count"] += 1
   911→
   912→            result["services"][svc] = {
   913→                "status": status,
   914→                "uptime_pct": uptime_pct,
   915→                "total_pings": total,
   916→                "healthy_pings": healthy,
   917→                "avg_response_ms": round(avg_ms, 1),
   918→                "last_ping_at": last_ping["created_at"],
   919→                "last_healthy_at": next(
   920→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
   921→                    None,
   922→                ),
   923→                "last_error": last_error,
   924→                "last_error_at": last_error_time,
   925→                "last_status_code": last_ping.get("status_code"),
   926→            }
   927→        return result

  1009→    # ── Drift Detection (Wave 2) ─────────────────────────────
  1010→
  1011→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1012→        """Analyze quality_scores for drift using day-over-day comparison."""
  1013→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
  1014→        if len(tenant_scores) < 3:
  1015→            return {
  1016→                "tenant_id": tenant_id, "drift_detected": False,
  1017→                "drift_severity": "none", "accuracy_7d": None,
  1018→                "accuracy_yesterday": None, "accuracy_today": None,
  1019→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
  1020→                "total_scores": len(tenant_scores),
  1021→            }
  1022→        # Group by date (ISO date portion)
  1023→        by_date: Dict[str, List[float]] = {}
  1024→        for s in tenant_scores:
  1025→            date_str = s["created_at"][:10]  # YYYY-MM-DD
  1026→            by_date.setdefault(date_str, []).append(s["overall_score"])
  1027→
  1028→        dates_sorted = sorted(by_date.keys())
  1029→
  1030→        def _avg(scores: List[float]) -> float:
  1031→            return round(sum(scores) / len(scores), 4) if scores else 0.0
  1032→
  1033→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
  1034→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1035→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
  1036→
  1037→        # Drift detection: >5% drop for 3+ days
  1038→        drift_detected = False
  1039→        drift_severity = "none"
  1040→        trigger_reason = None
  1041→        trend_direction = "stable"
  1042→
  1043→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
  1044→            recent_3 = dates_sorted[-3:]
  1045→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
  1046→            # Check if declining 3+ consecutive days
  1047→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
  1048→            drop_7d = accuracy_7d - accuracy_today
  1049→
  1050→            if declining and drop_7d > 0.05:
  1051→                drift_detected = True
  1052→                trend_direction = "declining"
  1053→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
  1054→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
  1055→            elif drop_7d > 0.03:
  1056→                drift_detected = True
  1057→                trend_direction = "slight_decline"
  1058→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
  1059→                drift_severity = "warning"
  1060→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
  1061→                trend_direction = "improving"
  1062→            else:
  1063→                trend_direction = "stable"
  1064→
  1065→        # Check for repeated errors (same low score pattern)
  1066→        if not drift_detected and tenant_scores:
  1067→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
  1068→            if len(low_scores) >= 3:
  1069→                # Check if same resolution_path failing
  1070→                paths: Dict[str, int] = {}
  1071→                for s in low_scores:
  1072→                    p = s.get("resolution_path", "unknown")
  1073→                    paths[p] = paths.get(p, 0) + 1
  1074→                worst_path = max(paths.items(), key=lambda x: x[1])
  1075→                if worst_path[1] >= 3:
  1076→                    drift_detected = True
  1077→                    drift_severity = "warning"
  1078→                    trend_direction = "declining"
  1079→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
  1080→
  1081→        return {
  1082→            "tenant_id": tenant_id,
  1083→            "drift_detected": drift_detected,
  1084→            "drift_severity": drift_severity,

  1284→class SupabaseBackend(StorageBackend):
  1285→    """Supabase storage via REST API (PostgREST). Uses httpx.
  1286→
  1287→    Requires env vars: SUPABASE_URL, SUPABASE_ANON_KEY
  1288→    """
  1289→
  1290→    def __init__(self):
  1291→        if not SUPABASE_URL or not SUPABASE_ANON_KEY:
  1292→            raise ValueError("SUPABASE_URL and SUPABASE_ANON_KEY must be set")
  1293→        self._url = SUPABASE_URL.rstrip("/")
  1294→        self._headers = {
  1295→            "apikey": SUPABASE_ANON_KEY,
  1296→            "Authorization": f"Bearer {SUPABASE_ANON_KEY}",
  1297→            "Content-Type": "application/json",
  1298→            "Prefer": "return=representation",
  1299→        }
  1300→        self._client = None  # Lazy init on first use
  1301→
  1302→    async def _get_client(self):
  1303→        if self._client is None or self._client.is_closed:
  1304→            import httpx
  1305→            self._client = httpx.AsyncClient(
  1306→                base_url=self._url,
  1307→                headers=self._headers,
  1308→                timeout=15.0,
  1309→            )
  1310→        return self._client
  1311→
  1312→    async def _rest_get(self, table: str, query: str = "") -> List[Dict]:
  1313→        client = await self._get_client()
  1314→        url = f"/rest/v1/{table}"
  1315→        if query:
  1316→            url += f"?{query}"
  1317→        r = await client.get(url)
  1318→        r.raise_for_status()
  1319→        return r.json() if r.text else []
  1320→
  1321→    async def _rest_post(self, table: str, data: Dict) -> Dict:
  1322→        client = await self._get_client()
  1323→        r = await client.post(f"/rest/v1/{table}", json=data)
  1324→        r.raise_for_status()
  1325→        result = r.json()
  1326→        if isinstance(result, list) and result:
  1327→            return result[0]
  1328→        return result
  1329→
  1330→    async def _rest_patch(self, table: str, filter_query: str, data: Dict) -> List[Dict]:
  1331→        client = await self._get_client()
  1332→        r = await client.patch(f"/rest/v1/{table}?{filter_query}", json=data)
  1333→        r.raise_for_status()
  1334→        return r.json() if r.text else []
  1335→
  1336→    async def close(self):
  1337→        if self._client and not self._client.is_closed:
  1338→            await self._client.aclose()

  1763→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1764→        # Fetch scores + client-side drift analysis (same logic as InMemory)
  1765→        scores = await self._rest_get(
  1766→            TABLE_QUALITY_SCORES,
  1767→            f"tenant_id=eq.{tenant_id}&select=overall_score,resolution_path,created_at&order=created_at.asc&limit=5000"
  1768→        )
  1769→        if len(scores) < 3:
  1770→            return {"tenant_id": tenant_id, "drift_detected": False, "drift_severity": "none",
  1771→                    "accuracy_7d": None, "accuracy_yesterday": None, "accuracy_today": None,
  1772→                    "trend_direction": "stable", "trigger_reason": "insufficient_data", "total_scores": len(scores)}
  1773→        by_date: Dict[str, List[float]] = {}
  1774→        for s in scores:
  1775→            d = (s.get("created_at") or "")[:10]
  1776→            by_date.setdefault(d, []).append(float(s.get("overall_score", 0)))
  1777→        dates_sorted = sorted(by_date.keys())
  1778→        def _avg(vals): return round(sum(vals) / len(vals), 4) if vals else 0.0
  1779→        accuracy_7d = _avg([v for d in dates_sorted[-7:] for v in by_date[d]])
  1780→        accuracy_today = _avg(by_date[dates_sorted[-1]])
  1781→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1782→        drift_detected = False; drift_severity = "none"; trigger_reason = None; trend_direction = "stable"
  1783→        if len(dates_sorted) >= 3:
  1784→            daily_avgs = [_avg(by_date[d]) for d in dates_sorted[-3:]]
  1785→            declining = all(daily_avgs[i] > daily_avgs[i+1] for i in range(len(daily_avgs)-1))
  1786→            drop = accuracy_7d - accuracy_today
  1787→            if declining and drop > 0.05:
  1788→                drift_detected = True; trend_direction = "declining"
  1789→                trigger_reason = f"accuracy_dropped_{drop:.1%}_over_3_days"
  1790→                drift_severity = "critical" if drop > 0.10 else "warning"
  1791→            elif drop > 0.03:
  1792→                drift_detected = True; trend_direction = "slight_decline"
  1793→                trigger_reason = f"accuracy_dropped_{drop:.1%}_from_7d_avg"; drift_severity = "warning"
  1794→            elif accuracy_today > accuracy_7d + 0.03:
  1795→                trend_direction = "improving"
  1796→        return {"tenant_id": tenant_id, "drift_detected": drift_detected, "drift_severity": drift_severity,
  1797→                "accuracy_7d": accuracy_7d, "accuracy_yesterday": accuracy_yesterday,
  1798→                "accuracy_today": accuracy_today, "trend_direction": trend_direction,
  1799→                "trigger_reason": trigger_reason, "total_scores": len(scores)}

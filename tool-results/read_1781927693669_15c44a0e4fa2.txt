     1→"""
     2→ParwaGraphState — Shared State for the Multi-Agent LangGraph System
     3→
     4→This TypedDict defines the single shared state object that flows through
     5→ALL 24+ agent nodes in the PARWA LangGraph graph. Every node reads from
     6→and writes to this state.
     7→
     8→Design Principles:
     9→  1. Every field is Optional — nodes only read what they need, write what they produce
    10→  2. Annotated[list, operator.add] for append-only lists (errors, audit trail)
    11→  3. Annotated[dict, operator.or_] for merge-accumulated dicts (node_outputs)
    12→  4. variant_tier is the KING field — it drives agent availability, MAKER mode,
    13→     technique access, channel availability, and approval requirements
    14→  5. Groups are logical, not enforced by Python — they help developers understand
    15→     which fields belong together
    16→
    17→Groups (24):
    18→  1. INPUT              — Raw query + metadata
    19→  2. PII_REDACTION      — PII-scrubbed message + detected entities
    20→  3. EMPATHY_ENGINE     — Sentiment, urgency, threat detection
    21→  4. ROUTER_AGENT       — Intent, complexity, target agent selection
    22→  5. DOMAIN_AGENT       — Agent response, confidence, proposed action
    23→  6. MAKER_VALIDATOR    — K-solution generation, selection, red flags
    24→  7. CONTROL_SYSTEM     — Approval decisions, confidence breakdowns
    25→  8. DSPY_OPTIMIZER     — Prompt optimization metadata
    26→  9. GUARDRAILS         — Safety checks, flags, blocks
    27→  10. CHANNEL_DELIVERY   — Delivery status, confirmations
    28→  11. STATE_UPDATE       — Persistence, audit, metrics
    29→  12. GSD_STATE          — Guided Support Dialogue state machine
    30→  13. METADATA           — Processing timing, token usage, execution log
    31→  14. JARVIS_AWARENESS   — System-wide awareness for Jarvis Command Center
    32→  15. EMERGENCY_CONTROLS — AI pause, circuit breakers, emergency state
    33→  16. ANTI_ARBITRAGE     — Gaming detection, risk scoring
    34→  17. BRAND_VOICE_RAG    — Brand voice application, RAG retrieval
    35→  18. COLLECTIVE_INTEL   — Collective intelligence, reward signals
    36→  19. SELF_HEALING_TRUST — API recovery, circuit breakers, trust scoring
    37→  20. JARVIS_COMMAND     — NL parsing, Co-Pilot, Feed
    38→  21. INTEGRATION        — External system health, connector data
    39→  22. VOICE_CALL         — Call tracking, compliance
    40→  23. SMS_COMPLIANCE     — TCPA, rate limits
    41→  24. DYNAMIC_INSTRUCTION— Policy, undo, real-time updates
    42→
    43→BC-001: company_id (tenant_id) is always present for multi-tenant isolation.
    44→BC-008: Never crash — all nodes wrap in try/except, write errors to state.errors.
    45→BC-012: All timestamps UTC.

    56→# ══════════════════════════════════════════════════════════════════
    57→# Append-only list reducer: new items are appended, never replaced
    58→# ══════════════════════════════════════════════════════════════════
    59→
    60→def _merge_lists(existing: List[Any], new: List[Any]) -> List[Any]:
    61→    """Reducer: append new items to existing list."""
    62→    return existing + new
    63→
    64→
    65→def _merge_dicts(existing: Dict[str, Any], new: Dict[str, Any]) -> Dict[str, Any]:
    66→    """Reducer: merge new dict into existing dict (new keys override)."""
    67→    merged = {**existing, **new}
    68→    return merged
    69→
    70→
    71→def _max_float(existing: float, new: float) -> float:
    72→    """Reducer: keep the maximum value (for scores)."""
    73→    return max(existing, new)
    74→
    75→
    76→def _replace(existing: Any, new: Any) -> Any:
    77→    """Reducer: last write wins (default behavior)."""
    78→    return new

   940→# STATE TRANSITION VALIDATION & SANITIZATION
   941→# ══════════════════════════════════════════════════════════════════
   942→
   943→
   944→def validate_and_sanitize_node_output(
   945→    node_name: str,
   946→    update: Dict[str, Any],
   947→) -> Dict[str, Any]:
   948→    """
   949→    Validate and sanitize a node's output before it is written to state.
   950→
   951→    This is the recommended entry point for all nodes to call before
   952→    returning their state update dict. It:
   953→
   954→      1. Validates the update against known field constraints
   955→      2. Logs any validation errors with the node name for traceability
   956→      3. Sanitizes invalid values to safe defaults
   957→      4. Logs each correction with the node name
   958→      5. Returns the sanitized update dict
   959→
   960→    This function NEVER raises exceptions (BC-008). If anything goes
   961→    wrong internally, the original update is returned unchanged.
   962→
   963→    Args:
   964→        node_name: Name of the node producing this update (for logging)
   965→        update: Partial state update dict that the node wants to return
   966→
   967→    Returns:
   968→        Sanitized update dict with invalid values corrected to safe defaults.
   969→
   970→    Example usage in a node::
   971→
   972→        def my_agent_node(state: ParwaGraphState) -> Dict[str, Any]:
   973→            result = {
   974→                "variant_tier": "pro",
   975→                "agent_confidence": 0.85,
   976→                "action_type": "informational",
   977→            }
   978→            return validate_and_sanitize_node_output("my_agent_node", result)
   979→    """
   980→    from app.core.langgraph.validators import (
   981→        validate_state_transition,
   982→        sanitize_state_update,
   983→    )
   984→    from app.logger import get_logger
   985→
   986→    _logger = get_logger("langgraph_state")
   987→
   988→    try:
   989→        # Step 1: Validate — collect errors for logging
   990→        errors = validate_state_transition(update)
   991→        if errors:
   992→            _logger.warning(
   993→                "node_output_validation_errors",
   994→                node=node_name,
   995→                errors=errors,
   996→                fields=list(update.keys()),
   997→            )
   998→
   999→        # Step 2: Sanitize — correct invalid values to safe defaults
  1000→        sanitized = sanitize_state_update(update)
  1001→
  1002→        # Step 3: Log summary if any corrections were made
  1003→        if sanitized != update:
  1004→            corrected_fields = [
  1005→                key for key in update if update[key] != sanitized.get(key)
  1006→            ]
  1007→            _logger.info(
  1008→                "node_output_sanitized",
  1009→                node=node_name,
  1010→                corrected_fields=corrected_fields,
  1011→                original_values={k: update[k] for k in corrected_fields},
  1012→                sanitized_values={k: sanitized[k] for k in corrected_fields},
  1013→            )
  1014→
  1015→        return sanitized
  1016→
  1017→    except Exception as exc:
  1018→        # BC-008: Never crash — if validation itself fails, return original
  1019→        _logger.error(
  1020→            "node_output_validation_failed_returning_original",
  1021→            node=node_name,
  1022→            error=str(exc),
  1023→        )
  1024→        return update
  1025→

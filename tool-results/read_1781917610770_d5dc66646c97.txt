   586→class InMemoryBackend(StorageBackend):
   587→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   588→
   589→    def __init__(self):
   590→        self._notifications: Dict[str, Dict] = {}
   591→        self._flags: List[Dict] = []
   592→        self._audit: List[Dict] = []
   593→        self._quality_scores: List[Dict] = []
   594→        self._batches: Dict[str, List[Dict]] = {}
   595→        self._batch_metas: Dict[str, Dict] = {}
   596→        self._lock = threading.Lock()
   597→        self._next_nfy_id = 1
   598→        # Wave 2 stores
   599→        self._integration_pings: List[Dict] = []
   600→        self._llm_costs: List[Dict] = []

   682→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99

   689→            if priority_order.get(n["priority"], 99) > min_rank:
   690→                continue
   691→            results.append(n)
   692→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   693→        return results

   872→    # ── Batch Queue ──────────────────────────────────────────
   873→
   874→    async def add_to_batch(
   875→        self,
   876→        tenant_id: str,
   877→        batch_key: str,
   878→        ticket_id: str,
   879→        confidence: float,
   880→    ) -> Optional[Dict[str, Any]]:
   881→        BATCH_WINDOW_S = 300
   882→        now = time.time()
   883→
   884→        if tenant_id not in self._batch_metas:
   885→            self._batch_metas[tenant_id] = {}
   886→
   887→        # Find existing batch with same key within window
   888→        existing_key = None
   889→        for bk, meta in self._batch_metas[tenant_id].items():
   890→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   891→                existing_key = bk
   892→                break
   893→
   894→        if existing_key:
   895→            meta = self._batch_metas[tenant_id][existing_key]
   896→            meta["ticket_ids"].append(ticket_id)
   897→            meta["signal_count"] = len(meta["ticket_ids"])
   898→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   899→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   900→            # Don't flush yet — still within window
   901→            return None
   902→        else:
   903→            meta = {
   904→                "batch_key": batch_key,
   905→                "tenant_id": tenant_id,
   906→                "batch_start": now,
   907→                "ticket_ids": [ticket_id],
   908→                "signal_count": 1,
   909→                "confidence_min": confidence,
   910→                "confidence_max": confidence,
   911→                "status": "pending",
   912→            }
   913→            self._batch_metas[tenant_id][batch_key] = meta
   914→            return None

  1044→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
  1045→        """Compute health from recent pings (last 30 per service)."""
  1046→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
  1047→        services: Dict[str, List[Dict]] = {}
  1048→        for p in tenant_pings:
  1049→            services.setdefault(p["service_name"], []).append(p)
  1050→
  1051→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
  1052→        for svc, pings in services.items():
  1053→            recent = pings[-30:]  # last 30 pings
  1054→            if not recent:
  1055→                continue
  1056→            healthy = sum(1 for p in recent if p["is_healthy"])
  1057→            total = len(recent)
  1058→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
  1059→            last_ping = recent[-1]
  1060→            avg_ms = (sum(p["response_ms"] or 0 for p in recent) / total) if total else 0
  1061→            last_error = None
  1062→            last_error_time = None
  1063→            for p in reversed(recent):
  1064→                if not p["is_healthy"]:
  1065→                    last_error = p.get("error_detail")
  1066→                    last_error_time = p["created_at"]
  1067→                    break
  1068→
  1069→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
  1070→            if status != "healthy":
  1071→                result["degraded_count"] += 1
  1072→            else:
  1073→                result["healthy_count"] += 1
  1074→
  1075→            result["services"][svc] = {
  1076→                "status": status,
  1077→                "uptime_pct": uptime_pct,
  1078→                "total_pings": total,
  1079→                "healthy_pings": healthy,
  1080→                "avg_response_ms": round(avg_ms, 1),
  1081→                "last_ping_at": last_ping["created_at"],
  1082→                "last_healthy_at": next(
  1083→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
  1084→                    None,
  1085→                ),
  1086→                "last_error": last_error,
  1087→                "last_error_at": last_error_time,
  1088→                "last_status_code": last_ping.get("status_code"),
  1089→            }
  1090→        return result

  1172→    # ── Drift Detection (Wave 2) ─────────────────────────────
  1173→
  1174→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1175→        """Analyze quality_scores for drift using day-over-day comparison."""
  1176→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
  1177→        if len(tenant_scores) < 3:
  1178→            return {
  1179→                "tenant_id": tenant_id, "drift_detected": False,
  1180→                "drift_severity": "none", "accuracy_7d": None,
  1181→                "accuracy_yesterday": None, "accuracy_today": None,
  1182→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
  1183→                "total_scores": len(tenant_scores),
  1184→            }
  1185→        # Group by date (ISO date portion)
  1186→        by_date: Dict[str, List[float]] = {}
  1187→        for s in tenant_scores:
  1188→            date_str = s["created_at"][:10]  # YYYY-MM-DD
  1189→            by_date.setdefault(date_str, []).append(s["overall_score"])
  1190→
  1191→        dates_sorted = sorted(by_date.keys())
  1192→
  1193→        def _avg(scores: List[float]) -> float:
  1194→            return round(sum(scores) / len(scores), 4) if scores else 0.0
  1195→
  1196→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
  1197→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1198→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
  1199→
  1200→        # Drift detection: >5% drop for 3+ days
  1201→        drift_detected = False
  1202→        drift_severity = "none"
  1203→        trigger_reason = None
  1204→        trend_direction = "stable"
  1205→
  1206→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
  1207→            recent_3 = dates_sorted[-3:]
  1208→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
  1209→            # Check if declining 3+ consecutive days
  1210→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
  1211→            drop_7d = accuracy_7d - accuracy_today
  1212→
  1213→            if declining and drop_7d > 0.05:
  1214→                drift_detected = True
  1215→                trend_direction = "declining"
  1216→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
  1217→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
  1218→            elif drop_7d > 0.03:
  1219→                drift_detected = True
  1220→                trend_direction = "slight_decline"
  1221→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
  1222→                drift_severity = "warning"
  1223→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
  1224→                trend_direction = "improving"
  1225→            else:
  1226→                trend_direction = "stable"
  1227→
  1228→        # Check for repeated errors (same low score pattern)
  1229→        if not drift_detected and tenant_scores:
  1230→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
  1231→            if len(low_scores) >= 3:
  1232→                # Check if same resolution_path failing
  1233→                paths: Dict[str, int] = {}
  1234→                for s in low_scores:
  1235→                    p = s.get("resolution_path", "unknown")
  1236→                    paths[p] = paths.get(p, 0) + 1
  1237→                worst_path = max(paths.items(), key=lambda x: x[1])
  1238→                if worst_path[1] >= 3:
  1239→                    drift_detected = True
  1240→                    drift_severity = "warning"
  1241→                    trend_direction = "declining"
  1242→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
  1243→
  1244→        return {
  1245→            "tenant_id": tenant_id,
  1246→            "drift_detected": drift_detected,
  1247→            "drift_severity": drift_severity,
  1248→            "accuracy_7d": accuracy_7d,
  1249→            "accuracy_yesterday": accuracy_yesterday,
  1250→            "accuracy_today": accuracy_today,
  1251→            "trend_direction": trend_direction,
  1252→            "trigger_reason": trigger_reason,
  1253→            "total_scores": len(tenant_scores),
  1254→        }

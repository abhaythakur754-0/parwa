     5→ * - Instance list, status, capacity, orchestration controls (Phase 5)
     6→ * - Variant Router section (Phase 14)
     7→ * - Route Ticket test panel (Phase 14)

    11→

    13→
    14→import React, { useState, useEffect, useCallback } from 'react';
    15→import { cn } from '@/lib/utils';
    16→import { useRouter } from 'next/navigation';
    17→import { VariantInstanceCard, type VariantInstanceData } from '@/components/jarvis-cc/VariantInstanceCard';
    18→import { MetricCard } from '@/components/jarvis-cc/MetricCard';
    19→import { get, post, del } from '@/lib/api';
    20→import { toast } from 'sonner';
    21→import {
    22→  Loader2,
    23→  Route,
    24→  Plus,
    25→  Trash2,
    26→  BarChart3,
    27→  Compass,
    28→  ArrowRight,
    29→} from 'lucide-react';
    30→
    31→// ── Types ───────────────────────────────────────────────────────────

    34→  id: string;
    35→  variant_tier: 'mini_parwa' | 'parwa' | 'parwa_high';
    36→  status: string;
    37→  active_tickets: number;
    38→  capacity: number;
    39→  quality_score: number | null;
    40→  latency_ms: number | null;
    41→  company_id: string;

    43→}
    44→

    46→interface VariantInfo {
    47→  id: string;
    48→  variant_type: 'mini' | 'parwa' | 'parwa_high';
    49→  status: string;
    50→  config?: Record<string, unknown>;
    51→  created_at?: string;
    52→}
    53→

    55→  variant_id: string;
    56→  variant_type: string;
    57→  ticket_count: number;
    58→  avg_quality_score?: number;
    59→  avg_latency_ms?: number;
    60→}
    61→
    62→interface RouteTicketResult {
    63→  variant_id: string;
    64→  variant_type: string;
    65→  reason?: string;
    66→}

    71→  <svg className="w-6 h-6 text-orange-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
    72→    <path strokeLinecap="round" strokeLinejoin="round" d="M8.25 3v1.5M4.5 8.25H3m18 0h-1.5M4.5 12H3m18 0h-1.5M4.5 15.75H3m18 0h-1.5M8.25 19.5V21M12 3v1.5m0 15V21m3.75-18v1.5m0 15V21m-9-1.5h10.5a2.25 2.25 0 0 0 2.25-2.25V6.75a2.25 2.25 0 0 0-2.25-2.25H6.75A2.25 2.25 0 0 0 4.5 6.75v10.5a2.25 2.25 0 0 0 2.25 2.25Zm.75-12h9v9h-9v-9Z" />
    73→  </svg>
    74→);

    76→const tierNames: Record<string, string> = {
    77→  mini_parwa: 'Mini Parwa',
    78→  mini: 'Mini Parwa',
    79→  parwa: 'Parwa Standard',
    80→  parwa_high: 'Parwa High',
    81→};
    82→
    83→const tierDescriptions: Record<string, string> = {
    84→  mini_parwa: 'Lightweight agent for simple queries and FAQ handling',
    85→  mini: 'Lightweight agent for simple queries and FAQ handling',
    86→  parwa: 'Standard agent with full technique suite and RAG support',
    87→  parwa_high: 'Premium agent with advanced reasoning and escalation capabilities',
    88→};
    89→
    90→// ── Variants Page ───────────────────────────────────────────────────
    91→
    92→export default function VariantsPage() {
    93→  const router = useRouter();
    94→  const [instances, setInstances] = useState<VariantInstance[]>([]);
    95→  const [isLoading, setIsLoading] = useState(true);
    96→  const [error, setError] = useState<string | null>(null);
    97→
    98→  // Phase 14 state
    99→  const [phase14Variants, setPhase14Variants] = useState<VariantInfo[]>([]);
   100→  const [variantUsage, setVariantUsage] = useState<VariantUsageEntry[]>([]);
   101→  const [routeTicketIntent, setRouteTicketIntent] = useState('');
   102→  const [routeTicketComplexity, setRouteTicketComplexity] = useState(5);
   103→  const [routeTicketResult, setRouteTicketResult] = useState<RouteTicketResult | null>(null);
   104→  const [isRouting, setIsRouting] = useState(false);
   105→  const [isAddingVariant, setIsAddingVariant] = useState(false);
   106→  const [addVariantType, setAddVariantType] = useState<'mini' | 'parwa' | 'parwa_high'>('mini');
   107→  const [isRemovingVariant, setIsRemovingVariant] = useState<string | null>(null);
   108→  const [isLoadingVariants, setIsLoadingVariants] = useState(true);
   109→  const [isLoadingUsage, setIsLoadingUsage] = useState(true);
   110→
   111→  // ── Fetch instances (Phase 5 existing) ──────────────────────────────
   112→  const fetchInstances = useCallback(async () => {
   113→    setIsLoading(true);
   114→    setError(null);
   115→    try {
   116→      const result = await get<VariantInstance[]>('/api/ai/instances');
   117→      setInstances(Array.isArray(result) ? result : []);
   118→    } catch {
   119→      try {
   120→        const sessionId = localStorage.getItem('jarvis_cc_session_id');
   121→        if (sessionId) {
   122→          await get(`/api/jarvis/cc/awareness/snapshot?session_id=${sessionId}`);
   123→          setInstances([]);
   124→        }
   125→      } catch {
   126→        setInstances([]);
   127→      }
   128→    } finally {
   129→      setIsLoading(false);
   130→    }
   131→  }, []);
   132→
   133→  // ── Phase 14: Fetch variants from /api/variants/list ───────────────
   134→  const fetchPhase14Variants = useCallback(async () => {
   135→    setIsLoadingVariants(true);
   136→    try {
   137→      const result = await get<VariantInfo[]>('/api/variants/list');
   138→      setPhase14Variants(Array.isArray(result) ? result : []);
   139→    } catch {
   140→      // Phase 14 endpoint unavailable — non-critical
   141→      setPhase14Variants([]);
   142→    } finally {
   143→      setIsLoadingVariants(false);
   144→    }
   145→  }, []);
   146→
   147→  // ── Phase 14: Fetch usage from /api/variants/usage ─────────────────
   148→  const fetchVariantUsage = useCallback(async () => {
   149→    setIsLoadingUsage(true);
   150→    try {
   151→      const result = await get<VariantUsageEntry[]>('/api/variants/usage');
   152→      setVariantUsage(Array.isArray(result) ? result : []);
   153→    } catch {
   154→      setVariantUsage([]);
   155→    } finally {
   156→      setIsLoadingUsage(false);
   157→    }
   158→  }, []);
   159→
   160→  useEffect(() => {
   161→    fetchInstances();
   162→  }, [fetchInstances]);
   163→
   164→  useEffect(() => {
   165→    fetchPhase14Variants();
   166→    fetchVariantUsage();
   167→  }, [fetchPhase14Variants, fetchVariantUsage]);
   168→
   169→  // ── Phase 14: Route Ticket ──────────────────────────────────────────
   170→  const handleRouteTicket = async () => {
   171→    if (!routeTicketIntent.trim()) {
   172→      toast.error('Please enter a ticket intent');
   173→      return;
   174→    }
   175→
   176→    setIsRouting(true);
   177→    setRouteTicketResult(null);
   178→    try {
   179→      const result = await post<RouteTicketResult>('/api/variants/route-ticket', {
   180→        intent: routeTicketIntent,
   181→        complexity_score: routeTicketComplexity,
   182→      });
   183→      setRouteTicketResult(result);
   184→      toast.success(`Routed to ${tierNames[result.variant_type] || result.variant_type}`);
   185→    } catch (err) {
   186→      toast.error('Failed to route ticket');
   187→    } finally {
   188→      setIsRouting(false);
   189→    }
   190→  };
   191→
   192→  // ── Phase 14: Add Variant ───────────────────────────────────────────
   193→  const handleAddVariant = async () => {
   194→    setIsAddingVariant(true);
   195→    try {
   196→      await post('/api/variants/add', {
   197→        variant_type: addVariantType,
   198→      });
   199→      toast.success(`${tierNames[addVariantType]} variant added`);
   200→      await fetchPhase14Variants();
   201→    } catch (err) {
   202→      toast.error('Failed to add variant');
   203→    } finally {
   204→      setIsAddingVariant(false);
   205→    }
   206→  };
   207→
   208→  // ── Phase 14: Remove Variant ────────────────────────────────────────
   209→  const handleRemoveVariant = async (variantId: string) => {
   210→    if (!confirm('Are you sure you want to remove this variant?')) return;
   211→
   212→    setIsRemovingVariant(variantId);
   213→    try {
   214→      await del('/api/variants/remove', {
   215→        data: { variant_id: variantId },
   216→      } as Parameters<typeof del>[1]);
   217→      toast.success('Variant removed');
   218→      await fetchPhase14Variants();
   219→    } catch (err) {
   220→      toast.error('Failed to remove variant');
   221→    } finally {
   222→      setIsRemovingVariant(null);
   223→    }
   224→  };
   225→
   226→  // Compute metrics
   227→  const totalActive = instances.filter(i => i.status === 'active').length;
   228→  const totalCapacity = instances.reduce((sum, i) => sum + (i.capacity || 0), 0);
   229→  const totalTickets = instances.reduce((sum, i) => sum + (i.active_tickets || 0), 0);
   230→  const avgQuality = instances.length > 0
   231→    ? instances.reduce((sum, i) => sum + (i.quality_score || 0), 0) / instances.filter(i => i.quality_score !== null).length
   232→    : null;
   233→  const utilization = totalCapacity > 0 ? totalTickets / totalCapacity : 0;
   234→
   235→  // Map to card data
   236→  const cardInstances: VariantInstanceData[] = instances.map(i => ({
   237→    id: i.id,
   238→    name: tierNames[i.variant_tier] || i.variant_tier,
   239→    tier: i.variant_tier,
   240→    status: (i.status === 'active' ? 'active' : i.status === 'idle' ? 'idle' : i.status === 'paused' ? 'paused' : 'error') as VariantInstanceData['status'],
   241→    capacity: i.capacity || 0,
   242→    activeTickets: i.active_tickets || 0,
   243→    qualityScore: i.quality_score,
   244→    latencyMs: i.latency_ms,
   245→  }));
   246→
   247→  // Group by tier
   248→  const groupedInstances = cardInstances.reduce((acc, inst) => {
   249→    if (!acc[inst.tier]) acc[inst.tier] = [];
   250→    acc[inst.tier].push(inst);
   251→    return acc;
   252→  }, {} as Record<string, VariantInstanceData[]>);
   253→
   254→  const handleEscalate = (instanceId: string) => {
   255→    const instance = instances.find((i) => i.id === instanceId);
   256→    if (!instance) return;
   257→
   258→    const tierOrder = ['mini_parwa', 'parwa', 'parwa_high'] as const;
   259→    const currentIdx = tierOrder.indexOf(instance.variant_tier as typeof tierOrder[number]);
   260→    const nextTier = currentIdx < tierOrder.length - 1 ? tierOrder[currentIdx + 1] : null;
   261→
   262→    setInstances((prev) =>
   263→      prev.map((inst) =>
   264→        inst.id === instanceId
   265→          ? {
   266→              ...inst,
   267→              status: 'active',
   268→              capacity: nextTier === 'parwa' ? 5000 : nextTier === 'parwa_high' ? 99999 : inst.capacity,
   269→            }
   270→          : inst
   271→      )
   272→    );
   273→
   274→    try {
   275→      const escalations = JSON.parse(localStorage.getItem('parwa_variant_escalations') || '[]');
   276→      escalations.push({
   277→        instanceId,
   278→        fromTier: instance.variant_tier,
   279→        toTier: nextTier || instance.variant_tier,
   280→        escalatedAt: new Date().toISOString(),
   281→      });
   282→      localStorage.setItem('parwa_variant_escalations', JSON.stringify(escalations));
   283→    } catch {
   284→      // localStorage unavailable — non-critical
   285→    }
   286→
   287→    const tierName = tierNames[instance.variant_tier] || instance.variant_tier;
   288→    if (nextTier) {
   289→      toast.success(`Escalated ${tierName} instance`, {
   290→        description: `Instance promoted to ${tierNames[nextTier]} tier. Active tickets will be re-routed.`,
   291→      });
   292→    } else {
   293→      toast.info(`Escalation requested for ${tierName}`, {
   294→        description: 'Already at highest tier. A human agent has been notified.',
   295→      });
   296→    }
   297→  };
   298→
   299→  const handleShadowMode = (instanceId: string, tier: string) => {
   300→    const tierOrder = ['mini_parwa', 'parwa', 'parwa_high'] as const;
   301→    const currentIdx = tierOrder.indexOf(tier as typeof tierOrder[number]);
   302→    const shadowTier = currentIdx < tierOrder.length - 1 ? tierOrder[currentIdx + 1] : null;
   303→
   304→    if (shadowTier) {
   305→      toast.info('Opening Shadow Mode', {
   306→        description: `Test ${tierNames[shadowTier]} against ${tierNames[tier]} in Shadow Mode`,
   307→      });
   308→    } else {
   309→      toast.info('Opening Shadow Mode', {
   310→        description: `Configure shadow testing for ${tierNames[tier]}`,
   311→      });
   312→    }
   313→
   314→    router.push('/dashboard/shadow-mode');
   315→  };
   316→
   317→  const handleRebalance = (instanceId: string) => {
   318→    const instance = instances.find((i) => i.id === instanceId);
   319→    if (!instance) return;
   320→
   321→    const sameTierInstances = instances.filter(
   322→      (i) => i.variant_tier === instance.variant_tier && i.id !== instanceId && i.status === 'active'
   323→    );
   324→
   325→    if (sameTierInstances.length === 0) {
   326→      toast.info('No available instances for rebalancing', {
   327→        description: 'Add more instances or escalate to a higher tier.',
   328→      });
   329→      return;
   330→    }
   331→
   332→    const overload = instance.active_tickets - Math.floor(instance.capacity * 0.6);
   333→    if (overload <= 0) {
   334→      toast.info('Instance is within healthy load', {
   335→        description: `Utilization at ${Math.round((instance.active_tickets / instance.capacity) * 100)}% — no rebalance needed.`,
   336→      });
   337→      return;
   338→    }
   339→
   340→    const perInstance = Math.floor(overload / sameTierInstances.length);
   341→    const remainder = overload % sameTierInstances.length;
   342→
   343→    setInstances((prev) =>
   344→      prev.map((inst) => {
   345→        if (inst.id === instanceId) {
   346→          return { ...inst, active_tickets: inst.active_tickets - overload };
   347→        }
   348→        const idx = sameTierInstances.findIndex((s) => s.id === inst.id);
   349→        if (idx >= 0) {
   350→          return {
   351→            ...inst,
   352→            active_tickets: inst.active_tickets + perInstance + (idx < remainder ? 1 : 0),
   353→          };
   354→        }
   355→        return inst;
   356→      })
   357→    );
   358→
   359→    try {
   360→      const rebalances = JSON.parse(localStorage.getItem('parwa_variant_rebalances') || '[]');
   361→      rebalances.push({
   362→        instanceId,
   363→        tier: instance.variant_tier,
   364→        ticketsRedistributed: overload,
   365→        targetCount: sameTierInstances.length,
   366→        rebalancedAt: new Date().toISOString(),
   367→      });
   368→      localStorage.setItem('parwa_variant_rebalances', JSON.stringify(rebalances));
   369→    } catch {
   370→      // localStorage unavailable — non-critical
   371→    }
   372→
   373→    toast.success(`Rebalanced ${tierNames[instance.variant_tier] || instance.variant_tier}`, {
   374→      description: `${overload} ticket${overload !== 1 ? 's' : ''} redistributed across ${sameTierInstances.length} instance${sameTierInstances.length !== 1 ? 's' : ''}.`,
   375→    });
   376→  };
   377→
   378→  return (
   379→    <div className="space-y-6">
   380→      {/* Header */}
   381→      <div className="flex items-center justify-between pb-4 border-b border-white/[0.06]">
   382→        <div>
   383→          <h1 className="text-xl font-bold text-white">Variant Engine</h1>
   384→          <p className="text-sm text-zinc-500 mt-0.5">
   385→            Configure and monitor your AI variant instances
   386→          </p>
   387→        </div>
   388→        <button
   389→          onClick={() => { fetchInstances(); fetchPhase14Variants(); fetchVariantUsage(); }}
   390→          disabled={isLoading}
   391→          className="text-xs px-3 py-1.5 rounded-lg bg-white/[0.04] text-zinc-400 hover:text-white hover:bg-white/[0.08] transition-colors disabled:opacity-50"
   392→        >
   393→          {isLoading ? 'Loading...' : 'Refresh'}
   394→        </button>
   395→      </div>
   396→
   397→      {/* Summary Metrics */}
   398→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   399→        <MetricCard
   400→          label="Active Instances"
   401→          value={`${totalActive}/${instances.length}`}
   402→          variant={totalActive === 0 && instances.length > 0 ? 'warning' : 'default'}
   403→        />
   404→        <MetricCard
   405→          label="Total Capacity"
   406→          value={totalCapacity}
   407→          subtitle="tickets"
   408→        />
   409→        <MetricCard
   410→          label="Active Tickets"
   411→          value={totalTickets}
   412→          variant={utilization >= 0.9 ? 'danger' : utilization >= 0.7 ? 'warning' : 'default'}
   413→        />
   414→        <MetricCard
   415→          label="Avg Quality"
   416→          value={avgQuality !== null ? `${Math.round(avgQuality * 100)}%` : '--'}
   417→          variant={avgQuality !== null ? (avgQuality >= 0.7 ? 'success' : avgQuality >= 0.5 ? 'warning' : 'danger') : 'default'}
   418→        />
   419→      </div>
   420→
   421→      {/* Utilization Overview */}
   422→      <div className="rounded-xl border border-white/[0.06] bg-[#1A1A1A] p-4">
   423→        <h3 className="text-xs font-medium text-zinc-500 uppercase tracking-wider mb-3">Pool Utilization</h3>
   424→        <div className="h-3 bg-white/5 rounded-full overflow-hidden">
   425→          <div
   426→            className={cn(
   427→              'h-full rounded-full transition-all duration-700',
   428→              utilization >= 0.9 ? 'bg-red-500' : utilization >= 0.7 ? 'bg-gradient-to-r from-amber-500 to-orange-500' : 'bg-gradient-to-r from-emerald-500 to-emerald-400'
   429→            )}
   430→            style={{ width: `${Math.min(utilization * 100, 100)}%` }}
   431→          />
   432→        </div>
   433→        <div className="flex items-center justify-between mt-2 text-xs text-zinc-500">
   434→          <span>{totalTickets} active tickets</span>
   435→          <span>{Math.round(utilization * 100)}% utilized</span>
   436→        </div>
   437→      </div>
   438→
   439→      {/* ══════════════════════════════════════════════════════════════════
   440→          Phase 14: Variant Router Section
   441→         ══════════════════════════════════════════════════════════════════ */}
   442→      <div className="rounded-xl border border-orange-500/10 bg-[#1A1A1A] overflow-hidden">
   443→        <div className="px-4 py-3 border-b border-white/[0.06] flex items-center gap-2">
   444→          <Compass className="w-4 h-4 text-orange-400" />
   445→          <h3 className="text-sm font-semibold text-white">Variant Router</h3>
   446→          <span className="text-[10px] px-1.5 py-0.5 rounded bg-orange-500/10 text-orange-400 uppercase tracking-wider">Phase 14</span>
   447→        </div>
   448→        <div className="p-4">
   449→          {/* Active variants list */}
   450→          {isLoadingVariants ? (
   451→            <div className="flex items-center gap-2 py-4 justify-center">
   452→              <Loader2 className="w-4 h-4 animate-spin text-orange-400" />
   453→              <span className="text-xs text-zinc-500">Loading variants...</span>
   454→            </div>
   455→          ) : phase14Variants.length === 0 ? (
   456→            <div className="text-center py-6">
   457→              <p className="text-sm text-zinc-500 mb-2">No active variants configured</p>
   458→              <p className="text-xs text-zinc-600">Add a variant below to get started</p>
   459→            </div>
   460→          ) : (
   461→            <div className="space-y-2 mb-4">
   462→              {phase14Variants.map((v) => (
   463→                <div
   464→                  key={v.id}
   465→                  className="flex items-center justify-between px-3 py-2.5 rounded-lg bg-white/[0.02] border border-white/[0.04] hover:border-white/[0.08] transition-colors"
   466→                >
   467→                  <div className="flex items-center gap-3">
   468→                    <span className={cn(
   469→                      'w-2 h-2 rounded-full',
   470→                      v.status === 'active' ? 'bg-emerald-400' : v.status === 'idle' ? 'bg-zinc-500' : 'bg-red-400'
   471→                    )} />
   472→                    <span className="text-sm text-white font-medium">
   473→                      {tierNames[v.variant_type] || v.variant_type}
   474→                    </span>
   475→                    <span className="text-[10px] px-1.5 py-0.5 rounded bg-white/5 text-zinc-500 uppercase">
   476→                      {v.status}
   477→                    </span>
   478→                  </div>
   479→                  <button
   480→                    onClick={() => handleRemoveVariant(v.id)}
   481→                    disabled={isRemovingVariant === v.id}
   482→                    className="inline-flex items-center gap-1 text-[11px] px-2.5 py-1 rounded-lg bg-red-500/10 text-red-400 hover:bg-red-500/20 transition-colors disabled:opacity-50"
   483→                  >
   484→                    {isRemovingVariant === v.id ? (
   485→                      <Loader2 className="w-3 h-3 animate-spin" />
   486→                    ) : (
   487→                      <Trash2 className="w-3 h-3" />
   488→                    )}
   489→                    Remove
   490→                  </button>
   491→                </div>
   492→              ))}
   493→            </div>
   494→          )}
   495→
   496→          {/* Add Variant */}
   497→          <div className="flex items-center gap-2 pt-3 border-t border-white/[0.04]">
   498→            <select
   499→              value={addVariantType}
   500→              onChange={(e) => setAddVariantType(e.target.value as 'mini' | 'parwa' | 'parwa_high')}
   501→              className="bg-white/[0.04] border border-white/[0.06] rounded-lg px-3 py-1.5 text-xs text-zinc-300 focus:outline-none focus:border-orange-500/30"
   502→            >
   503→              <option value="mini">Mini Parwa</option>
   504→              <option value="parwa">Parwa Standard</option>
   505→              <option value="parwa_high">Parwa High</option>
   506→            </select>
   507→            <button
   508→              onClick={handleAddVariant}
   509→              disabled={isAddingVariant}
   510→              className="inline-flex items-center gap-1.5 text-xs px-3 py-1.5 rounded-lg bg-gradient-to-r from-orange-600 to-orange-500 text-white hover:from-orange-500 hover:to-orange-400 transition-all disabled:opacity-50 shadow-sm"
   511→            >
   512→              {isAddingVariant ? (
   513→                <Loader2 className="w-3 h-3 animate-spin" />

   149→// ── Helper Functions ──────────────────────────────────────────────────

   151→function formatNumber(n: number): string {
   152→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`;
   153→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
   154→  return n.toLocaleString();
   155→}
   156→
   157→function formatHours(hours: number): string {
   158→  if (hours < 1) return `${Math.round(hours * 60)}m`;
   159→  return `${hours.toFixed(1)}h`;
   160→}
   161→
   162→function formatPercent(n: number): string {
   163→  return `${n.toFixed(1)}%`;
   164→}
   165→
   166→function formatCurrency(n: number): string {
   167→  if (n >= 1) return `$${n.toFixed(2)}`;
   168→  return `$${n.toFixed(4)}`;
   169→}

   233→  // show what they selected on the Models page so the dashboard feels connected.

   235→    industry?: string;
   236→    isAnnual?: boolean;
   237→    variants?: Array<{ id: string; name: string; quantity: number; monthlyPrice: number }>;

   249→        if (ctx && ctx.variants && ctx.variants.length > 0) {
   250→          setPricingContext(ctx);
   251→        }
   252→      }
   253→    } catch { /* ignore */ }
   254→  }, []);
   255→
   256→  // ── Check onboarding status (non-blocking) ──

   258→  // appropriate prompts, but we do NOT force-redirect users.
   259→  // Users should be free to explore the dashboard before starting onboarding.
   260→
   261→  useEffect(() => {
   262→    if (onboardingChecked) return;

   264→    async function checkOnboarding() {
   265→      try {
   266→        const res = await fetch('/api/onboarding/state');
   267→        if (res.ok) {
   268→          const state = await res.json();
   269→          if (state.status === 'completed' || state.first_victory_completed) {
   270→            // Onboarding IS completed — cache it locally
   271→            localStorage.setItem('parwa_onboarding_completed', 'true');
   272→          }
   273→        }
   274→        // Whether onboarding is completed or not, let the user see the dashboard.
   275→        // The dashboard will show appropriate prompts for incomplete onboarding.
   276→      } catch {
   277→        // Network error — check localStorage for onboarding completion flag
   278→        // but don't redirect regardless. Let user explore.
   279→        const hasCompletedOnboarding = localStorage.getItem('parwa_onboarding_completed') === 'true';
   280→        if (hasCompletedOnboarding) {
   281→          // already completed, nothing to do
   282→        }
   283→        // Not completed but still let them see dashboard
   284→      }
   285→      setOnboardingChecked(true);
   286→    }
   287→
   288→    checkOnboarding();
   289→  }, [onboardingChecked]);

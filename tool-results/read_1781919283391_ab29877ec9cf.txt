   626→   626→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:

   675→   675→    async def get_notifications(
   676→   676→        self,
   677→   677→        tenant_id: str,
   678→   678→        include_resolved: bool = False,
   679→   679→        min_priority: Optional[str] = None,
   680→   680→    ) -> List[Dict[str, Any]]:
   681→   681→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   682→   682→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   683→   683→        results = []
   684→   684→        for n in self._notifications.values():
   685→   685→            if n["tenant_id"] != tenant_id:
   686→   686→                continue
   687→   687→            if not include_resolved and n["is_resolved"]:
   688→   688→                continue
   689→   689→            if priority_order.get(n["priority"], 99) > min_rank:
   690→   690→                continue
   691→   691→            results.append(n)
   692→   692→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   693→   693→        return results

   699→   699→            n["resolved_at"] = time.time()
   700→   700→            return True
   701→   701→        return False

   872→   872→    # ── Batch Queue ──────────────────────────────────────────
   873→   873→
   874→   874→    async def add_to_batch(
   875→   875→        self,
   876→   876→        tenant_id: str,
   877→   877→        batch_key: str,
   878→   878→        ticket_id: str,
   879→   879→        confidence: float,
   880→   880→    ) -> Optional[Dict[str, Any]]:
   881→   881→        BATCH_WINDOW_S = 300
   882→   882→        now = time.time()
   883→   883→
   884→   884→        if tenant_id not in self._batch_metas:
   885→   885→            self._batch_metas[tenant_id] = {}
   886→   886→
   887→   887→        # Find existing batch with same key within window
   888→   888→        existing_key = None
   889→   889→        for bk, meta in self._batch_metas[tenant_id].items():
   890→   890→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   891→   891→                existing_key = bk
   892→   892→                break
   893→   893→
   894→   894→        if existing_key:
   895→   895→            meta = self._batch_metas[tenant_id][existing_key]
   896→   896→            meta["ticket_ids"].append(ticket_id)
   897→   897→            meta["signal_count"] = len(meta["ticket_ids"])
   898→   898→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   899→   899→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   900→   900→            # Don't flush yet — still within window
   901→   901→            return None
   902→   902→        else:
   903→   903→            meta = {
   904→   904→                "batch_key": batch_key,
   905→   905→                "tenant_id": tenant_id,
   906→   906→                "batch_start": now,
   907→   907→                "ticket_ids": [ticket_id],
   908→   908→                "signal_count": 1,
   909→   909→                "confidence_min": confidence,
   910→   910→                "confidence_max": confidence,
   911→   911→                "status": "pending",
   912→   912→            }
   913→   913→            self._batch_metas[tenant_id][batch_key] = meta
   914→   914→            return None

   917→   917→        results = list(self._batch_metas.get(tenant_id, {}).values())

   919→   919→        return results

  1174→  1174→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1175→  1175→        """Analyze quality_scores for drift using day-over-day comparison."""
  1176→  1176→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
  1177→  1177→        if len(tenant_scores) < 3:
  1178→  1178→            return {
  1179→  1179→                "tenant_id": tenant_id, "drift_detected": False,
  1180→  1180→                "drift_severity": "none", "accuracy_7d": None,
  1181→  1181→                "accuracy_yesterday": None, "accuracy_today": None,
  1182→  1182→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
  1183→  1183→                "total_scores": len(tenant_scores),
  1184→  1184→            }
  1185→  1185→        # Group by date (ISO date portion)
  1186→  1186→        by_date: Dict[str, List[float]] = {}
  1187→  1187→        for s in tenant_scores:
  1188→  1188→            date_str = s["created_at"][:10]  # YYYY-MM-DD
  1189→  1189→            by_date.setdefault(date_str, []).append(s["overall_score"])
  1190→  1190→
  1191→  1191→        dates_sorted = sorted(by_date.keys())
  1192→  1192→
  1193→  1193→        def _avg(scores: List[float]) -> float:
  1194→  1194→            return round(sum(scores) / len(scores), 4) if scores else 0.0
  1195→  1195→
  1196→  1196→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
  1197→  1197→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1198→  1198→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
  1199→  1199→
  1200→  1200→        # Drift detection: >5% drop for 3+ days
  1201→  1201→        drift_detected = False
  1202→  1202→        drift_severity = "none"
  1203→  1203→        trigger_reason = None
  1204→  1204→        trend_direction = "stable"
  1205→  1205→
  1206→  1206→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
  1207→  1207→            recent_3 = dates_sorted[-3:]
  1208→  1208→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
  1209→  1209→            # Check if declining 3+ consecutive days
  1210→  1210→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
  1211→  1211→            drop_7d = accuracy_7d - accuracy_today
  1212→  1212→
  1213→  1213→            if declining and drop_7d > 0.05:
  1214→  1214→                drift_detected = True
  1215→  1215→                trend_direction = "declining"
  1216→  1216→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
  1217→  1217→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
  1218→  1218→            elif drop_7d > 0.03:
  1219→  1219→                drift_detected = True
  1220→  1220→                trend_direction = "slight_decline"
  1221→  1221→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
  1222→  1222→                drift_severity = "warning"
  1223→  1223→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
  1224→  1224→                trend_direction = "improving"
  1225→  1225→            else:
  1226→  1226→                trend_direction = "stable"
  1227→  1227→
  1228→  1228→        # Check for repeated errors (same low score pattern)
  1229→  1229→        if not drift_detected and tenant_scores:

  1237→  1237→                worst_path = max(paths.items(), key=lambda x: x[1])
  1238→  1238→                if worst_path[1] >= 3:
  1239→  1239→                    drift_detected = True
  1240→  1240→                    drift_severity = "warning"
  1241→  1241→                    trend_direction = "declining"
  1242→  1242→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
  1243→  1243→
  1244→  1244→        return {
  1245→  1245→            "tenant_id": tenant_id,
  1246→  1246→            "drift_detected": drift_detected,
  1247→  1247→            "drift_severity": drift_severity,
  1248→  1248→            "accuracy_7d": accuracy_7d,
  1249→  1249→            "accuracy_yesterday": accuracy_yesterday,
  1250→  1250→            "accuracy_today": accuracy_today,
  1251→  1251→            "trend_direction": trend_direction,
  1252→  1252→            "trigger_reason": trigger_reason,

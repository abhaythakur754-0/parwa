   266→   266→   266→    def list_tickets(
   267→   267→   267→        self,
   268→   268→   268→        status: Optional[List[str]] = None,
   269→   269→   269→        priority: Optional[List[str]] = None,

   277→   277→   277→        date_from: Optional[datetime] = None,
   278→   278→   278→        date_to: Optional[datetime] = None,
   279→   279→   279→        search: Optional[str] = None,
   280→   280→   280→        page: int = 1,
   281→   281→   281→        page_size: int = 20,
   282→   282→   282→        sort_by: str = "created_at",
   283→   283→   283→        sort_order: str = "desc",
   284→   284→   284→    ) -> Tuple[List[Ticket], int]:
   285→   285→   285→        """List tickets with filters and pagination.
   286→   286→   286→
   287→   287→   287→        Args:
   288→   288→   288→            status: Filter by status list
   289→   289→   289→            priority: Filter by priority list
   290→   290→   290→            category: Filter by category list
   291→   291→   291→            assigned_to: Filter by assignee ID
   292→   292→   292→            channel: Filter by channel
   293→   293→   293→            customer_id: Filter by customer ID
   294→   294→   294→            tags: Filter by tags (matches any)
   295→   295→   295→            is_spam: Filter by spam status
   296→   296→   296→            is_frozen: Filter by frozen status
   297→   297→   297→            date_from: Filter by creation date (from)
   298→   298→   298→            date_to: Filter by creation date (to)
   299→   299→   299→            search: Full-text search query
   300→   300→   300→            page: Page number (1-based)
   301→   301→   301→            page_size: Items per page
   302→   302→   302→            sort_by: Sort field
   303→   303→   303→            sort_order: Sort direction (asc/desc)
   304→   304→   304→
   305→   305→   305→        Returns:
   306→   306→   306→            Tuple of (tickets list, total count)
   307→   307→   307→        """
   308→   308→   308→        query = self.db.query(Ticket).filter(
   309→   309→   309→            Ticket.company_id == self.company_id
   310→   310→   310→        )
   311→   311→   311→
   312→   312→   312→        # Apply filters
   313→   313→   313→        if status:
   314→   314→   314→            query = query.filter(Ticket.status.in_(status))
   315→   315→   315→
   316→   316→   316→        if priority:
   317→   317→   317→            query = query.filter(Ticket.priority.in_(priority))
   318→   318→   318→
   319→   319→   319→        if category:
   320→   320→   320→            query = query.filter(Ticket.category.in_(category))
   321→   321→   321→
   322→   322→   322→        if assigned_to:
   323→   323→   323→            query = query.filter(Ticket.assigned_to == assigned_to)
   324→   324→   324→
   325→   325→   325→        if channel:
   326→   326→   326→            query = query.filter(Ticket.channel == channel)
   327→   327→   327→
   328→   328→   328→        if customer_id:
   329→   329→   329→            query = query.filter(Ticket.customer_id == customer_id)
   330→   330→   330→
   331→   331→   331→        if is_spam is not None:
   332→   332→   332→            query = query.filter(Ticket.is_spam == is_spam)
   333→   333→   333→
   334→   334→   334→        if is_frozen is not None:
   335→   335→   335→            query = query.filter(Ticket.frozen == is_frozen)
   336→   336→   336→
   337→   337→   337→        if date_from:
   338→   338→   338→            query = query.filter(Ticket.created_at >= date_from)
   339→   339→   339→
   340→   340→   340→        if date_to:
   341→   341→   341→            query = query.filter(Ticket.created_at <= date_to)
   342→   342→   342→
   343→   343→   343→        # Tags filter (JSON contains)
   344→   344→   344→        if tags:
   345→   345→   345→            # Simple tag matching - check if any tag exists in the JSON
   346→   346→   346→            for tag in tags:
   347→   347→   347→                query = query.filter(Ticket.tags.contains(f'"{tag}"'))
   348→   348→   348→
   349→   349→   349→        # Full-text search
   350→   350→   350→        if search:
   351→   351→   351→            # M-33: Escape ILIKE wildcards to prevent SQL injection
   352→   352→   352→            escaped = search.replace("%", r"\\%").replace("_", r"\\_")
   353→   353→   353→            search_pattern = f"%{escaped}%"
   354→   354→   354→            query = query.filter(
   355→   355→   355→                or_(
   356→   356→   356→                    Ticket.subject.ilike(search_pattern, escape="\\"),
   357→   357→   357→                    Ticket.metadata_json.ilike(search_pattern, escape="\\"),
   358→   358→   358→                )
   359→   359→   359→            )
   360→   360→   360→
   361→   361→   361→        # Count total
   362→   362→   362→        total = query.count()
   363→   363→   363→
   364→   364→   364→        # Sort
   365→   365→   365→        sort_column = getattr(Ticket, sort_by, Ticket.created_at)
   366→   366→   366→        if sort_order == "desc":
   367→   367→   367→            query = query.order_by(desc(sort_column))
   368→   368→   368→        else:
   369→   369→   369→            query = query.order_by(sort_column)
   370→   370→   370→
   371→   371→   371→        # Paginate
   372→   372→   372→        offset = (page - 1) * page_size
   373→   373→   373→        tickets = query.offset(offset).limit(page_size).all()
   374→   374→   374→
   375→   375→   375→        return tickets, total

   390→   390→   390→    ) -> Ticket:
   391→   391→   391→        """Update ticket fields.

   394→   394→   394→            ticket_id: Ticket ID

   406→   406→   406→

   419→   419→   419→                ticket_id, old_status, status, user_id, reason
   420→   420→   420→            )
   421→   421→   421→
   422→   422→   422→        # Update fields
   423→   423→   423→        if priority is not None:
   424→   424→   424→            ticket.priority = priority
   425→   425→   425→
   426→   426→   426→        if category is not None:
   427→   427→   427→            ticket.category = category
   428→   428→   428→

   430→   430→   430→            ticket.tags = json.dumps(tags)
   431→   431→   431→
   432→   432→   432→        if status is not None:
   433→   433→   433→            ticket.status = status

   813→   813→   813→    def _calculate_similarity(self, str1: str, str2: str) -> float:
   814→   814→   814→        """Calculate similarity between two strings.
   815→   815→   815→
   816→   816→   816→        Simple Jaccard similarity on words.
   817→   817→   817→
   818→   818→   818→        Args:
   819→   819→   819→            str1: First string
   820→   820→   820→            str2: Second string
   821→   821→   821→
   822→   822→   822→        Returns:
   823→   823→   823→            Similarity score between 0 and 1
   824→   824→   824→        """
   825→   825→   825→        words1 = set(str1.split())
   826→   826→   826→        words2 = set(str2.split())
   827→   827→   827→
   828→   828→   828→        if not words1 or not words2:
   829→   829→   829→            return 0.0
   830→   830→   830→
   831→   831→   831→        intersection = words1 & words2
   832→   832→   832→        union = words1 | words2
   833→   833→   833→
   834→   834→   834→        return len(intersection) / len(union)

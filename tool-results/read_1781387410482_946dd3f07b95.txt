   120→   120→// ── Savings Calculation ──────────────────────────────────────────────────
   121→   121→
   122→   122→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   123→   123→  agentsReplaced: number;
   124→   124→  humanCost: number;
   125→   125→  savings: number;
   126→   126→  savingsPercent: number;
   127→   127→} {
   128→   128→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   129→   129→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   130→   130→  const savings = Math.max(0, humanCost - monthlyCost);
   131→   131→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   132→   132→  return { agentsReplaced, humanCost, savings, savingsPercent };
   133→   133→}

   242→   242→// ── Component ──────────────────────────────────────────────────────────
   243→   243→
   244→   244→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   245→   245→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   246→   246→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   247→   247→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   248→   248→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   249→   249→  const [isSubmitting, setIsSubmitting] = useState(false);
   250→   250→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   251→   251→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   252→   252→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   253→   253→
   254→   254→  // Restore add-ons and multi-variant selections from localStorage
   255→   255→  useEffect(() => {
   256→   256→    try {
   257→   257→      const stored = localStorage.getItem('parwa_pricing_context');
   258→   258→      if (stored) {
   259→   259→        const ctx = JSON.parse(stored) as PricingContext & {
   260→   260→          selectedVariants?: string[];
   261→   261→          variantQuantities?: Record<string, number>;
   262→   262→        };
   263→   263→        if (ctx.addOns) {
   264→   264→          setAddOns(ctx.addOns);
   265→   265→        }
   266→   266→        // Restore multiple variants if selected on Pricing page
   267→   267→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   268→   268→          const tiers: VariantTier[] = ctx.selectedVariants
   269→   269→            .map((v: string) => {
   270→   270→              const normalized = normalizeTier(v);
   271→   271→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   272→   272→            })
   273→   273→            .filter(Boolean) as VariantTier[];
   274→   274→          if (tiers.length > 0) {
   275→   275→            setActiveVariants(tiers);
   276→   276→          }
   277→   277→        }
   278→   278→        // Restore variant quantities
   279→   279→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   280→   280→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   281→   281→          const newQuantities = { ...variantQuantities };
   282→   282→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   283→   283→            const tier = normalizeTier(variantKey);
   284→   284→            if (['starter', 'growth', 'high'].includes(tier)) {
   285→   285→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   286→   286→            }
   287→   287→          }
   288→   288→          setVariantQuantities(newQuantities);
   289→   289→        }
   290→   290→      }
   291→   291→    } catch {
   292→   292→      // ignore
   293→   293→    }
   294→   294→  }, []);
   295→   295→
   296→   296→  // Check if Paddle.js is available
   297→   297→  useEffect(() => {
   298→   298→    getPaddleInstance().then((paddle) => {
   299→   299→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   300→   300→    }).catch(() => {
   301→   301→      setPaddleStatus('unavailable');
   302→   302→    });
   303→   303→  }, []);
   304→   304→
   305→   305→  // ── Calculations ──────────────────────────────────────────────────
   306→   306→
   307→   307→  const totalTicketLimit = useMemo(
   308→   308→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   309→   309→    [activeVariants, variantQuantities]
   310→   310→  );
   311→   311→
   312→   312→  const baseSubscription = useMemo(
   313→   313→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   314→   314→    [activeVariants, variantQuantities]
   315→   315→  );
   316→   316→
   317→   317→  const addOnTotal = useMemo(() => {
   318→   318→    let total = 0;
   319→   319→    if (addOns.voice) {
   320→   320→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   321→   321→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   322→   322→        total += voiceAddOn.price;
   323→   323→      }
   324→   324→    }
   325→   325→    if (addOns.customApi) {
   326→   326→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   327→   327→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   328→   328→        total += customApiAddOn.price;
   329→   329→      }
   330→   330→    }
   331→   331→    return total;
   332→   332→  }, [addOns, activeVariants]);
   333→   333→
   334→   334→  const totalMonthly = baseSubscription + addOnTotal;
   335→   335→
   336→   336→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   337→   337→
   338→   338→  const toggleVariant = (tier: VariantTier) => {
   339→   339→    setActiveVariants((prev) => {
   340→   340→      if (prev.includes(tier)) {
   341→   341→        if (prev.length <= 1) {
   342→   342→          toast.error('You must have at least one active variant');
   343→   343→          return prev;
   344→   344→        }
   345→   345→        return prev.filter((t) => t !== tier);
   346→   346→      }
   347→   347→      return [...prev, tier];
   348→   348→    });
   349→   349→  };
   350→   350→
   351→   351→  const updateQuantity = (tier: VariantTier, qty: number) => {
   352→   352→    const newQty = Math.max(1, Math.min(qty, 10));
   353→   353→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   354→   354→  };
   355→   355→
   356→   356→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   357→   357→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   358→   358→  };
   359→   359→
   360→   360→  // ── Payment success handler — ONLY called after Paddle confirms ──
   361→   361→  const handlePaymentSuccess = () => {
   362→   362→    setPaymentConfirmed(true);
   363→   363→    localStorage.removeItem('parwa_payment_pending');
   364→   364→    toast.success('Payment successful! Welcome to PARWA!');
   365→   365→    // Only trigger onboarding completion AFTER Paddle confirms
   366→   366→    onComplete();
   367→   367→  };
   368→   368→
   369→   369→  const handleProceed = async () => {
   370→   370→    setIsSubmitting(true);
   371→   371→    setCheckoutError(null);
   372→   372→
   373→   373→    try {
   374→   374→      // Update pricing context in localStorage
   375→   375→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   376→   376→      const context: PricingContext = {
   377→   377→        industry: 'other',
   378→   378→        variant: primaryOnboardingVariant,
   379→   379→        addOns,
   380→   380→        totalMonthly,
   381→   381→        timestamp: new Date().toISOString(),
   382→   382→      };
   383→   383→
   384→   384→      // Preserve industry from existing context
   385→   385→      try {
   386→   386→        const stored = localStorage.getItem('parwa_pricing_context');
   387→   387→        if (stored) {
   388→   388→          const existing = JSON.parse(stored) as PricingContext;
   389→   389→          context.industry = existing.industry;
   390→   390→        }
   391→   391→      } catch {
   392→   392→        // ignore
   393→   393→      }
   394→   394→
   395→   395→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   396→   396→
   397→   397→      // ── Paddle Checkout Flow ──────────────────────────────────────
   398→   398→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   399→   399→
   400→   400→      // Add each active variant as a Paddle line item (with quantity)
   401→   401→      for (const tier of activeVariants) {
   402→   402→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   403→   403→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   404→   404→        const qty = variantQuantities[tier] || 1;
   405→   405→        if (variantPriceId) {
   406→   406→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   407→   407→        }
   408→   408→      }
   409→   409→
   410→   410→      // Voice add-on (only for starter where it's not included)
   411→   411→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   412→   412→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   413→   413→      }
   414→   414→
   415→   415→      // Custom API add-on (only for variants where it's not included)
   416→   416→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   417→   417→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });
   418→   418→      }
   419→   419→
   420→   420→      const customData = {
   421→   421→        source: 'parwa_onboarding',
   422→   422→        variant: primaryOnboardingVariant,
   423→   423→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   424→   424→        variantQuantities,
   425→   425→        addOns,
   426→   426→        industry: context.industry,
   427→   427→        totalMonthly,
   428→   428→      };
   429→   429→
   430→   430→      // ── Step 1: Try server-side checkout (backend creates Paddle transaction) ──
   431→   431→      let checkoutUrl: string | null = null;
   432→   432→
   433→   433→      try {
   434→   434→        const res = await fetch('/api/onboarding/checkout', {
   435→   435→          method: 'POST',
   436→   436→          headers: { 'Content-Type': 'application/json' },
   437→   437→          body: JSON.stringify({
   438→   438→            variant: primaryOnboardingVariant,
   439→   439→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   440→   440→            variantQuantities,
   441→   441→            addOns,
   442→   442→            totalMonthly,
   443→   443→            industry: context.industry,
   444→   444→          }),
   445→   445→        });
   446→   446→
   447→   447→        if (res.ok) {
   448→   448→          const checkoutData = await res.json();

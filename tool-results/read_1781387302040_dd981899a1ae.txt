   149→// ── Helper Functions ──────────────────────────────────────────────────

   151→function formatNumber(n: number): string {
   152→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`;
   153→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
   154→  return n.toLocaleString();
   155→}
   156→
   157→function formatHours(hours: number): string {
   158→  if (hours < 1) return `${Math.round(hours * 60)}m`;
   159→  return `${hours.toFixed(1)}h`;
   160→}
   161→
   162→function formatPercent(n: number): string {
   163→  return `${n.toFixed(1)}%`;
   164→}
   165→
   166→function formatCurrency(n: number): string {
   167→  if (n >= 1) return `$${n.toFixed(2)}`;
   168→  return `$${n.toFixed(4)}`;
   169→}

   230→
   231→  // ── Check onboarding status (non-blocking) ──
   232→  // We silently check onboarding state so the dashboard can show
   233→  // appropriate prompts, but we do NOT force-redirect users.
   234→  // Users should be free to explore the dashboard before starting onboarding.
   235→
   236→  useEffect(() => {
   237→    if (onboardingChecked) return;
   238→
   239→    async function checkOnboarding() {
   240→      try {
   241→        const res = await fetch('/api/onboarding/state');
   242→        if (res.ok) {
   243→          const state = await res.json();
   244→          if (state.status === 'completed' || state.first_victory_completed) {
   245→            // Onboarding IS completed — cache it locally
   246→            localStorage.setItem('parwa_onboarding_completed', 'true');
   247→          }
   248→        }
   249→        // Whether onboarding is completed or not, let the user see the dashboard.
   250→        // The dashboard will show appropriate prompts for incomplete onboarding.
   251→      } catch {
   252→        // Network error — check localStorage for onboarding completion flag
   253→        // but don't redirect regardless. Let user explore.
   254→        const hasCompletedOnboarding = localStorage.getItem('parwa_onboarding_completed') === 'true';
   255→        if (hasCompletedOnboarding) {
   256→          // already completed, nothing to do
   257→        }
   258→        // Not completed but still let them see dashboard
   259→      }
   260→      setOnboardingChecked(true);
   261→    }
   262→
   263→    checkOnboarding();
   264→  }, [onboardingChecked]);
   265→
   266→  // Show loading while checking onboarding
   267→  if (!onboardingChecked) {

   287→    }

   312→    get<ROISnapshot>('/api/v1/admin/roi')
   313→      .then((roi) => setRoiState({ status: 'success', data: roi }))
   314→      .catch(() => setRoiState({ status: 'error', data: null }));
   315→  }, []);

   342→    else if (range.start_date === sevenDaysAgo) setDatePreset('7d');
   343→    else if (range.start_date === thirtyDaysAgo) setDatePreset('30d');

   386→            <KPICard
   387→              title="AI vs Human Cost"

   133→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   134→
   135→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   136→  agentsReplaced: number;
   137→  humanCost: number;
   138→  savings: number;
   139→  savingsPercent: number;
   140→} {
   141→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   142→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   143→  const savings = Math.max(0, humanCost - monthlyCost);
   144→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   145→  return { agentsReplaced, humanCost, savings, savingsPercent };
   146→}

   453→// ── Component ──────────────────────────────────────────────────────────
   454→
   455→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   456→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   457→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   458→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   459→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   460→  const [isSubmitting, setIsSubmitting] = useState(false);
   461→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   462→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   463→  const [appliedCoupon, setAppliedCoupon] = useState<Coupon | null>(null);
   464→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   465→
   466→  // Restore add-ons from localStorage
   467→  useEffect(() => {
   468→    try {
   469→      const stored = localStorage.getItem('parwa_pricing_context');
   470→      if (stored) {
   471→        const ctx = JSON.parse(stored) as PricingContext;
   472→        if (ctx.addOns) {
   473→          setAddOns(ctx.addOns);
   474→        }
   475→      }
   476→    } catch {
   477→      // ignore
   478→    }
   479→  }, []);
   480→
   481→  // Check if Paddle.js is available
   482→  useEffect(() => {
   483→    getPaddleInstance().then((paddle) => {
   484→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   485→    }).catch(() => {
   486→      setPaddleStatus('unavailable');
   487→    });
   488→  }, []);
   489→
   490→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   491→
   492→  const totalTicketLimit = useMemo(
   493→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   494→    [activeVariants, variantQuantities]
   495→  );
   496→
   497→  const baseSubscription = useMemo(
   498→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   499→    [activeVariants, variantQuantities]
   500→  );
   501→
   502→  const addOnTotal = useMemo(() => {
   503→    let total = 0;
   504→    if (addOns.voice) {
   505→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   506→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   507→        total += voiceAddOn.price;
   508→      }
   509→    }
   510→    if (addOns.customApi) {
   511→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   512→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   513→        total += customApiAddOn.price;
   514→      }
   515→    }
   516→    return total;
   517→  }, [addOns, activeVariants]);
   518→
   519→  const totalMonthly = baseSubscription + addOnTotal;
   520→
   521→  // ── Coupon-discounted total ──────────────────────────────────────
   522→  const discountedTotal = useMemo(
   523→    () => applyCouponDiscount(totalMonthly, appliedCoupon),
   524→    [totalMonthly, appliedCoupon]
   525→  );
   526→  const isFreeCheckout = discountedTotal === 0;
   527→
   528→  // Per-variant discounted prices
   529→  const variantDiscountedPrices = useMemo(() => {
   530→    const prices: Record<VariantTier, number> = { starter: 0, growth: 0, high: 0 };
   531→    for (const tier of activeVariants) {
   532→      const variantPrice = VARIANT_PRICES[tier] * (variantQuantities[tier] || 1);

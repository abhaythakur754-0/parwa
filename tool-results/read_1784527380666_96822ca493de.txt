   121→def _load_tenant_variants(tenant_id: str) -> Dict[str, Any]:
   122→    """Load the tenant's active variant tiers and quota from the DB.
   123→
   124→    Data sources (in priority order):
   125→      1. _TEST_REGISTRY — for unit tests (no DB needed)
   126→      2. VariantInstance table — active instances for this company
   127→      3. Subscription table — fallback if no instances found
   128→      4. pricing_config defaults — if nothing in DB yet
   129→
   130→    Returns dict with:
   131→      - available_tiers: list of short tier keys the tenant owns
   132→        e.g. ["mini"] or ["parwa"] or ["mini", "parwa", "high"]
   133→      - highest_tier_short: "mini" | "parwa" | "high"
   134→      - highest_tier_db: "mini_parwa" | "parwa" | "parwa_high"
   135→      - quota: {tier_short: {total, used, remaining}} for each owned tier
   136→      - source: where the data came from (for logging)
   137→
   138→    BC-008: Never crashes — returns safe defaults on any error.
   139→    """
   140→    # ── 1. Test override ──
   141→    test_entry = _TEST_REGISTRY.get(tenant_id)
   142→    if test_entry:
   143→        short = test_entry["tier_short"]
   144→        return {
   145→            "available_tiers": [short],
   146→            "highest_tier_short": short,
   147→            "highest_tier_db": test_entry["tier_db"],
   148→            "quota": {
   149→                short: {
   150→                    "total": test_entry["quota_total"],
   151→                    "used": test_entry["quota_total"] - test_entry["quota_remaining"],
   152→                    "remaining": test_entry["quota_remaining"],
   153→                },
   154→            },
   155→            "source": "test_registry",
   156→        }
   157→
   158→    # ── 2. Real DB lookup ──
   159→    try:
   160→        from database.base import SessionLocal
   161→        from database.models.variant_engine import VariantInstance
   162→        from database.models.billing_extended import UsageRecord
   163→        from sqlalchemy import func
   164→
   165→        with SessionLocal() as db:
   166→            # 2a. Get all active variant instances for this tenant
   167→            instances = (
   168→                db.query(VariantInstance)
   169→                .filter(
   170→                    VariantInstance.company_id == tenant_id,
   171→                    VariantInstance.status == "active",
   172→                )
   173→                .all()
   174→            )
   175→
   176→            if instances:
   177→                # Collect unique variant types this tenant owns
   178→                owned_db_tiers: List[str] = []
   179→                seen_types: set = set()
   180→                for inst in instances:
   181→                    vt = inst.variant_type.strip().lower() if inst.variant_type else ""
   182→                    if vt and vt not in seen_types:
   183→                        seen_types.add(vt)
   184→                        owned_db_tiers.append(vt)
   185→
   186→                # Map to short keys
   187→                owned_short = []
   188→                for db_tier in owned_db_tiers:
   189→                    short = TIER_DB_TO_SHORT.get(db_tier, db_tier)
   190→                    if short in CAPABILITY_MATRIX:
   191→                        owned_short.append(short)
   192→
   193→                if not owned_short:
   194→                    # DB had instances but none mapped to known tiers
   195→                    raise ValueError("no_mappable_tiers")
   196→
   197→                # Sort by priority (mini first)
   198→                owned_short = [t for t in TIER_ORDER if t in owned_short]
   199→                highest_short = owned_short[-1]  # last = highest
   200→
   201→                # 2b. Get monthly ticket usage
   202→                current_month = datetime.now(timezone.utc).strftime("%Y-%m")
   203→                total_used = (
   204→                    db.query(func.coalesce(func.sum(UsageRecord.tickets_used), 0))
   205→                    .filter(
   206→                        UsageRecord.company_id == tenant_id,
   207→                        UsageRecord.record_month == current_month,
   208→                    )
   209→                    .scalar()
   210→                ) or 0
   211→                total_used = int(total_used)
   212→
   213→                # 2c. Get quota limits from pricing_config
   214→                quota = _build_quota_from_limits(owned_short, total_used)
   215→
   216→                return {
   217→                    "available_tiers": owned_short,
   218→                    "highest_tier_short": highest_short,
   219→                    "highest_tier_db": TIER_SHORT_TO_DB.get(highest_short, highest_short),
   220→                    "quota": quota,
   221→                    "source": "db_variant_instances",
   222→                }
   223→
   224→            # ── 3. Fallback: Subscription table ──
   225→            from database.models.billing import Subscription
   226→
   227→            sub = (
   228→                db.query(Subscription)
   229→                .filter(
   230→                    Subscription.company_id == tenant_id,
   231→                    Subscription.status == "active",
   232→                )
   233→                .order_by(Subscription.created_at.desc())
   234→                .first()
   235→            )
   236→
   237→            if sub and sub.tier:
   238→                tier_raw = sub.tier.strip().lower()
   239→                # sub.tier stores canonical names: starter, growth, high
   240→                tier_map = {"starter": "mini", "growth": "parwa", "high": "high"}
   241→                # Also support old DB names
   242→                tier_map.update(TIER_DB_TO_SHORT)
   243→
   244→                short = tier_map.get(tier_raw, "mini")
   245→                if short not in CAPABILITY_MATRIX:
   246→                    short = "mini"
   247→
   248→                # Get usage
   249→                current_month = datetime.now(timezone.utc).strftime("%Y-%m")
   250→                total_used = (
   251→                    db.query(func.coalesce(func.sum(UsageRecord.tickets_used), 0))
   252→                    .filter(
   253→                        UsageRecord.company_id == tenant_id,
   254→                        UsageRecord.record_month == current_month,
   255→                    )
   256→                    .scalar()
   257→                ) or 0
   258→                total_used = int(total_used)
   259→
   260→                quota = _build_quota_from_limits([short], total_used)
   261→
   262→                return {
   263→                    "available_tiers": [short],
   264→                    "highest_tier_short": short,
   265→                    "highest_tier_db": TIER_SHORT_TO_DB.get(short, short),
   266→                    "quota": quota,
   267→                    "source": "db_subscription",
   268→                }
   269→
   270→    except Exception as exc:
   271→        logger.warning(
   272→            "Node 2 DB lookup failed for tenant %s: %s — using default mini_parwa",
   273→            tenant_id, exc,
   274→        )
   275→
   276→    # ── 4. Ultimate default: mini_parwa ──
   277→    return {
   278→        "available_tiers": ["mini"],
   279→        "highest_tier_short": "mini",
   280→        "highest_tier_db": "mini_parwa",
   281→        "quota": {"mini": {"total": 2000, "used": 0, "remaining": 2000}},
   282→        "source": "default_mini_parwa",
   283→    }

   285→
   286→def _build_quota_from_limits(
   287→    owned_short: List[str],
   288→    total_used: int,
   289→) -> Dict[str, Dict[str, int]]:
   290→    """Build quota dict from pricing_config limits.
   291→
   292→    For multi-variant tenants, the monthly ticket limit is shared
   293→    across ALL their variants (it's a company-level limit, not
   294→    per-variant).  So ``total_used`` is the same for every tier.
   295→    """
   296→    try:
   297→        from app.core.pricing_config import VARIANT_LIMITS, VariantType
   298→    except ImportError:
   299→        # Fallback if pricing_config can't be imported
   300→        return {t: {"total": 2000, "used": total_used, "remaining": max(0, 2000 - total_used)}
   301→                for t in owned_short}
   302→
   303→    SHORT_TO_VARIANT_TYPE = {
   304→        "mini": VariantType.MINI,
   305→        "parwa": VariantType.PARWA,
   306→        "high": VariantType.HIGH,
   307→    }
   308→
   309→    quota: Dict[str, Dict[str, int]] = {}
   310→
   311→    for short in owned_short:
   312→        vt = SHORT_TO_VARIANT_TYPE.get(short)
   313→        if vt and vt in VARIANT_LIMITS:
   314→            limits = VARIANT_LIMITS[vt]
   315→            ticket_limit = int(limits.get("monthly_tickets", 2000))
   316→        else:
   317→            ticket_limit = 2000
   318→
   319→        remaining = max(0, ticket_limit - total_used)
   320→        quota[short] = {
   321→            "total": ticket_limit,
   322→            "used": total_used,
   323→            "remaining": remaining,
   324→        }
   325→
   326→    return quota

   329→# ── Keyword-Based Complexity Correction ──────────────────────────
   330→
   331→
   332→def _correct_complexity(
   333→    query: str, complexity: str, action: str, ticket_type: str,
   334→) -> Tuple[str, str]:
   335→    """Post-LLM correction: blend LLM classification with keyword heuristics.
   336→
   337→    Catches cases where the LLM misclassifies:
   338→      - Simple account changes tagged as complex
   339→      - Technical integration / billing disputes tagged as simple/medium
   340→    """
   341→    query_lower = query.lower()
   342→
   343→    # Strong indicators that should ALWAYS be complex regardless of LLM
   344→    strong_complex_patterns = [
   345→        r'\bintegrate\b.*\b(api|crm|erp|system|webhook)\b',
   346→        r'\bapi\b.*\b(endpoint|connect|integration)\b',
   347→        r'\b(overcharg|wrong charge|double charge|unauthorized charge)\b',
   348→        r'\brefund\b.*\b(subscription|plan|enterprise|difference)\b',
   349→        r'\b(chargeback|file\s+a\s+dispute)\b',
   350→        r'\b(error|crash|500|403)\b.*\b(urgent|deadline|team)\b',
   351→        r'\b(manager|supervisor)\b.*\b(complaint|terrible|worst)\b',
   352→        r'\b(enterprise|plan)\b.*\b(charg|refund|credit)\b.*\b\d+',
   353→    ]
   354→
   355→    for pattern in strong_complex_patterns:
   356→        if re.search(pattern, query_lower):
   357→            if complexity in ("simple", "medium"):
   358→                logger.info(
   359→                    "Complexity correction: %s -> complex (keyword override)",
   360→                    complexity,
   361→                )
   362→                return "complex", action
   363→
   364→    # Strong indicators that should ALWAYS be simple regardless of LLM
   365→    strong_simple_patterns = [
   366→        r'^\s*I\s+(need\s+)?(want\s+)?to\s+(change|update)\s+my\s+(billing\s+)?(email|address|password|name|phone)\b',
   367→        r'^\s*I\s+(need\s+)?(want\s+)?to\s+(change|update)\s+(my\s+)?billing\s+email\b',
   368→    ]
   369→
   370→    for pattern in strong_simple_patterns:
   371→        if re.search(pattern, query_lower):
   372→            if complexity == "complex":
   373→                logger.info(
   374→                    "Complexity correction: %s -> simple (keyword override)",
   375→                    complexity,
   376→                )
   377→                return "simple", "provide_info"
   378→
   379→    return complexity, action

   382→# ── Route Decision Logic (3-dimensional) ──────────────────────────
   383→
   384→
   385→def _check_capability(
   386→    tier: str,
   387→    ticket_type: str,
   388→    complexity: str,
   389→    action: str,
   390→    action_details: Dict,
   391→) -> bool:
   392→    """Dimension 1: Can this tier HANDLE this ticket?"""
   393→    caps = CAPABILITY_MATRIX.get(tier)
   394→    if not caps:
   395→        return False
   396→
   397→    # Simple/medium info: all tiers can handle
   398→    if action == "provide_info":
   399→        return True
   400→
   401→    # Phase 7: investigate_billing needs complex reasoning
   402→    if action == "investigate_billing":
   403→        return caps.get("complex_reasoning", False)
   404→
   405→    # Complex reasoning: only parwa and high
   406→    if complexity in ("complex", "hard") and not caps["complex_reasoning"]:
   407→        return False
   408→
   409→    # Execution actions: check tier limits
   410→    if action == "execute_refund":
   411→        if not caps["execute_refund"]:
   412→            return False
   413→        amount = action_details.get("amount", 0)
   414→        limits = EXECUTION_LIMITS.get(tier, {})
   415→        if amount > limits.get("max_refund", 0):
   416→            return False
   417→
   418→    if action == "execute_credit":
   419→        if not caps["execute_credit"]:
   420→            return False
   421→        amount = action_details.get("amount", 0)
   422→        limits = EXECUTION_LIMITS.get(tier, {})
   423→        if amount > limits.get("max_credit", 0):
   424→            return False
   425→
   426→    if action == "account_change" and not caps["account_change"]:
   427→        return False
   428→
   429→    return True
   430→
   431→
   432→def _check_quota(
   433→    tenant_variants: Dict[str, Any],
   434→    tier: str,
   435→) -> bool:
   436→    """Dimension 2: Does this tier have tickets remaining?"""
   437→    quota = tenant_variants.get("quota", {})
   438→    tier_quota = quota.get(tier)
   439→    if not tier_quota:
   440→        return False
   441→    return tier_quota["remaining"] > 0

   443→
   444→def _needs_complex_path(query: str) -> bool:
   445→    """Check if the query content requires complex path regardless of
   446→    complexity label.
   447→
   448→    Some medium-complexity queries need the full reasoning pipeline
   449→    (technical integrations, multi-system questions, detailed how-tos).
   450→    """
   451→    query_lower = query.lower()
   452→    complex_content_patterns = [
   453→        r'\b(integrate|connect)\b.*\b(api|crm|erp|system|service)\b',
   454→        r'\bendpoint\b',
   455→        r'\bhow\s+do\s+i\s+(set\s+up|configure|implement)\b.*\b(api|webhook|integration)\b',
   456→        r'\b(ssl|oauth|authentication|sso)\b.*\b(integrate|setup|configure)\b',
   457→    ]
   458→    return any(re.search(p, query_lower) for p in complex_content_patterns)

   460→
   461→def _route_decision(
   462→    tenant_variants: Dict[str, Any],
   463→    ticket_type: str,
   464→    complexity: str,
   465→    action: str,
   466→    action_details: Dict,
   467→    query: str = "",
   468→) -> Tuple[str, str, List[str]]:
   469→    """3-Dimensional Routing: capability + quota + efficiency.
   470→
   471→    Returns (selected_tier_short, path, capabilities_list).
   472→    Uses the LOWEST eligible tier to preserve higher quotas for
   473→    harder tickets.
   474→
   475→    For multi-variant tenants, it picks the cheapest tier that can
   476→    handle the ticket and still has quota remaining.
   477→    """
   478→    # Apply keyword-based complexity correction first
   479→    corrected_complexity, corrected_action = _correct_complexity(
   480→        query, complexity, action, ticket_type,
   481→    )
   482→
   483→    # Determine which path
   484→    # FORCE_COMPLEX_PATH: when True, all tickets go through the complex path
   485→    # (Node 4 with 7 LLM calls + Node 6 quality gate). Used for testing.
   486→    if os.environ.get("FORCE_COMPLEX_PATH", "").lower() in ("1", "true", "yes"):
   487→        target_path = "complex_path"
   488→        logger.info("Force complex path enabled — all tickets through Node 4 + Node 6")
   489→    elif corrected_complexity in ("simple", "medium") and corrected_action == "provide_info":
   490→        target_path = "simple_path"
   491→    elif corrected_complexity in ("simple", "medium") and corrected_action == "recommend":
   492→        target_path = "simple_path"
   493→    elif corrected_complexity == "simple" and corrected_action == "account_change":
   494→        target_path = "simple_path"
   495→    elif corrected_action == "investigate_billing":
   496→        target_path = "complex_path"
   497→    else:
   498→        target_path = "complex_path"
   499→
   500→    # Override: query content needs complex path
   501→    if target_path == "simple_path" and _needs_complex_path(query):
   502→        logger.info("Path override: simple -> complex (query content requires full reasoning)")
   503→        target_path = "complex_path"
   504→
   505→    # Find lowest eligible tier from what the tenant actually owns
   506→    available_tiers = tenant_variants.get("available_tiers", [])
   507→
   508→    selected_tier: Optional[str] = None
   509→    for tier in TIER_ORDER:
   510→        if tier not in available_tiers:
   511→            continue  # tenant doesn't own this tier
   512→        if _check_capability(tier, ticket_type, complexity, action, action_details):
   513→            if _check_quota(tenant_variants, tier):
   514→                selected_tier = tier
   515→                break  # lowest eligible tier found
   516→
   517→    # Fallback: if no tier has quota or can handle it, use highest owned tier
   518→    if selected_tier is None:
   519→        if available_tiers:
   520→            selected_tier = available_tiers[-1]  # highest
   521→        else:
   522→            selected_tier = "mini"
   523→
   524→    # If mini can't handle the action, try to find a higher tier
   525→    # that can, even if quota is exhausted (better to handle than to fail)
   526→    if not _check_capability(selected_tier, ticket_type, complexity, action, action_details):
   527→        for tier in reversed(TIER_ORDER):
   528→            if tier in available_tiers and _check_capability(tier, ticket_type, complexity, action, action_details):
   529→                selected_tier = tier
   530→                break
   531→
   532→    # Get capabilities for selected tier
   533→    caps = CAPABILITY_MATRIX.get(selected_tier, {})
   534→    capabilities = [k for k, v in caps.items() if v is True or isinstance(v, str)]
   535→
   536→    return selected_tier, target_path, capabilities

   542→async def node_2_smart_route(state: PipelineV2State) -> dict:
   543→    """Node 2: Smart Route — WHO handles this + WHERE does it go?
   544→
   545→    0 LLM calls. Pure logic + DB reads.
   546→    Reads from VariantInstance / Subscription / UsageRecord to
   547→    determine what the tenant actually purchased and how much
   548→    quota remains.
   549→    """
   550→    start = time.time()
   551→    tenant_id = state.get("tenant_id", "")
   552→    ticket_type = state.get("ticket_type", "general")
   553→    complexity = state.get("complexity", "simple")
   554→    action = state.get("required_action", "provide_info")
   555→    action_details = state.get("action_details", {})
   556→    logs: List[Dict] = []
   557→
   558→    # ── Wave 4: Check Jarvis system flags ──────────────────
   559→    system_flags = state.get("system_flags")
   560→    if not system_flags:
   561→        try:
   562→            from app.core.parwa_pipeline.parwa_bridge import load_system_flags

   585→            "technique_log": logs,

   593→        route_to = redirected[channel]
   594→        logs.append({"node": 2, "technique": "JARVIS_REDIRECT_CHECK", "duration_ms": 0, "result_summary": f"channel={channel} redirected to {route_to}"})
   595→        logger.info("Node 2: Channel '%s' redirected to '%s' by Jarvis", channel, route_to)

   597→            return {
   598→                "route_decision": "simple_path",
   599→                "current_path": "simple_path",
   600→                "variant_tier": "mini_parwa",
   601→                "quota_remaining": {},
   602→                "variant_capabilities": [],
   603→                "status": "escalated",
   604→                "escalation_context": {"reason": f"Channel '{channel}' redirected to human by Jarvis"},

   606→                "total_token_usage": state.get("total_token_usage", 0),

   608→
   609→    # Check force_mode: if Jarvis forced a mode, log it for downstream
   610→    force_mode = system_flags.get("force_mode")

   612→        logs.append({"node": 2, "technique": "JARVIS_MODE_CHECK", "duration_ms": 0, "result_summary": f"mode={force_mode}"})
   613→
   614→    # ── 1. Load tenant variants from DB ──
   615→    tenant_variants = _load_tenant_variants(tenant_id)

   618→
   619→    logs.append({
   620→        "node": 2,
   621→        "technique": "VariantRegistry",
   622→        "duration_ms": 0,
   623→        "result_summary": (
   624→            f"tiers={tenant_variants['available_tiers']} "
   625→            f"highest={highest_db} source={source}"

   627→    })
   628→
   629→    # ── 2. Quota snapshot ──
   630→    quota_remaining = {}
   631→    for tier_short, q in tenant_variants.get("quota", {}).items():
   632→        quota_remaining[tier_short] = q["remaining"]

   635→        "node": 2,

   638→        "result_summary": f"quota={quota_remaining}",

   641→    # ── 3. 3-Dimensional route decision ──
   642→    selected_tier_short, path, capabilities = _route_decision(
   643→        tenant_variants,
   644→        ticket_type, complexity, action, action_details,
   645→        query=state.get("query", ""),
   646→    )
   647→    logs.append({
   648→        "node": 2,
   649→        "technique": "RouteDecision",
   650→        "duration_ms": 0,
   651→        "result_summary": f"tier={selected_tier_short} path={path}",
   652→    })
   653→
   654→    # ── 4. Map short tier back to DB name for downstream nodes ──
   655→    selected_tier_db = TIER_SHORT_TO_DB.get(selected_tier_short, selected_tier_short)
   656→
   657→    # ── Layer 1: Non-LLM enrichment techniques (0 LLM calls) ─────
   658→    # CLARA: validate route decision is sound

   671→        n2_sc1 = "REVIEW: complex path for simple ticket (intentional?)"

   673→
   674→    # MAKER: ground decision in actual ticket data

   678→    # CoVe: verify tier has capability for this action

   681→
   682→    # RuleBasedAction: complex tickets always go to complex_path
   683→    n2_rba1 = "ok"
   684→    if complexity == "complex" and path != "complex_path":
   685→        n2_rba1 = "VIOLATION: complex ticket not on complex_path"
   686→    logs.append({"node": 2, "technique": "RuleBasedAction", "duration_ms": 0, "result_summary": n2_rba1})
   687→
   688→    # SafetyNet: check if route would bypass safety
   689→    n2_sn1 = "safe"
   690→    if action == "refund" and selected_tier_short in ("mini", "free"):
   691→        n2_sn1 = "BLOCK: refund on low tier"
   692→    logs.append({"node": 2, "technique": "SafetyNet", "duration_ms": 0, "result_summary": n2_sn1})
   693→
   694→    # ContradictionCheck: route vs action
   695→    n2_cc1 = "ok"

   699→
   700→    # IdempotencyCheck: route decision hash
   701→    import hashlib as _n2_hl
   702→    n2_route_hash = _n2_hl.md5(f"{tenant_id}:{ticket_type}:{action}:{selected_tier_short}".encode()).hexdigest()[:10]
   703→    logs.append({"node": 2, "technique": "IdempotencyCheck", "duration_ms": 0, "result_summary": f"hash={n2_route_hash}"})
   704→
   705→    # Escalation: tier downgrade may force escalation
   706→    n2_esc1 = "no_escalation"
   707→    if not tenant_variants.get("available_tiers"):
   708→        n2_esc1 = "NO_VARIANTS_FORCE_ESCALATE"
   709→    logs.append({"node": 2, "technique": "Escalation", "duration_ms": 0, "result_summary": n2_esc1})
   710→
   711→    # DynamicContext: route context enrichment
   712→    n2_dc1 = f"context tier={selected_tier_short} caps={len(capabilities) if isinstance(capabilities, list) else 0}"
   713→    logs.append({"node": 2, "technique": "DynamicContext", "duration_ms": 0, "result_summary": n2_dc1})
   714→
   715→    # MetaLearner: route pattern analysis
   716→    n2_ml1_features = []
   717→    if path == "complex_path":
   718→        n2_ml1_features.append("needs_reasoning")
   719→    if complexity == "complex":
   720→        n2_ml1_features.append("complex_input")
   721→    if action in ("refund", "escalate"):
   722→        n2_ml1_features.append("high_stakes")
   723→    n2_ml1 = ",".join(n2_ml1_features) if n2_ml1_features else "routine"
   724→    logs.append({"node": 2, "technique": "MetaLearner", "duration_ms": 0, "result_summary": f"pattern={n2_ml1}"})
   725→
   726→    # ── Layer 2: Deeper non-LLM analysis (0 LLM calls) ───────────
   727→    # SmartRouter.depth2: route quality score
   728→    n2_sr2 = "optimal"
   729→    if path == "complex_path" and complexity == "simple":
   730→        n2_sr2 = "OVER_ENGINEERED: complex path for simple ticket"
   731→    elif path == "simple_path" and complexity == "complex":
   732→        n2_sr2 = "UNDER_ENGINEERED: simple path for complex ticket"
   733→    logs.append({"node": 2, "technique": "SmartRouter.depth2", "duration_ms": 0, "result_summary": n2_sr2})
   734→
   735→    # ZeroShotValidator.depth2: cross-check tier vs ticket type
   736→    n2_zsv2 = "ok"
   737→    if ticket_type == "legal_review" and selected_tier_short not in ("full", "flagship"):
   738→        n2_zsv2 = "REVIEW: legal_review on non-flagship tier"

   740→        n2_zsv2 = "REVIEW: fraud on low tier"
   741→    logs.append({"node": 2, "technique": "ZeroShotValidator.depth2", "duration_ms": 0, "result_summary": n2_zsv2})
   742→
   743→    # SmartFilter.depth2: filter redundant capabilities
   744→    n2_caps_list = capabilities if isinstance(capabilities, list) else []
   745→    n2_unique_caps = list(set(n2_caps_list))
   746→    n2_sf2 = f"unique_caps={len(n2_unique_caps)} total={len(n2_caps_list)}"
   747→    logs.append({"node": 2, "technique": "SmartFilter.depth2", "duration_ms": 0, "result_summary": n2_sf2})
   748→
   749→    # CLARA.depth2: validate quota snapshot
   750→    n2_clara2 = "valid" if isinstance(quota_remaining, dict) and quota_remaining else "EMPTY: no quota data"
   751→    logs.append({"node": 2, "technique": "CLARA.depth2", "duration_ms": 0, "result_summary": n2_clara2})
   752→
   753→    # GSD.depth2: state transition logged
   754→    n2_gsd2 = "ROUTING_COMMITTED"
   755→    logs.append({"node": 2, "technique": "GSD.depth2", "duration_ms": 0, "result_summary": n2_gsd2})
   756→
   757→    # SelfConsistency.depth2: 3-way check
   758→    n2_sc2 = "3_way_consistent"
   759→    if action == "refund" and "refund" not in str(n2_caps_list).lower():
   760→        n2_sc2 = "MISALIGNED: refund action without refund capability"
   761→    logs.append({"node": 2, "technique": "SelfConsistency.depth2", "duration_ms": 0, "result_summary": n2_sc2})
   762→
   763→    # MAKER.depth2: ground in tier source
   764→    n2_maker2 = f"source={source} highest={highest_db}"
   765→    logs.append({"node": 2, "technique": "MAKER.depth2", "duration_ms": 0, "result_summary": n2_maker2})

   782→

   823→

   830→    logs.append({"node": 2, "technique": "CLARA.depth3", "duration_ms": 0, "result_summary": n2_clara3})

   832→    # GSD.depth3: state = routing complete
   833→    n2_gsd3 = "ROUTING_COMPLETE → NEXT_NODE"

   836→    # SelfConsistency.depth3: final 3-way

   838→    logs.append({"node": 2, "technique": "SelfConsistency.depth3", "duration_ms": 0, "result_summary": n2_sc3})

   840→    # MAKER.depth3: final grounding

   843→

   847→

   850→    logs.append({"node": 2, "technique": "RuleBasedAction.depth3", "duration_ms": 0, "result_summary": n2_rba3})
   851→
   852→    # SafetyNet.depth3: final safety check
   853→    n2_sn3 = "safe" if n2_sn1 == "safe" and n2_sn2 == "ok" else "flagged"
   854→    logs.append({"node": 2, "technique": "SafetyNet.depth3", "duration_ms": 0, "result_summary": n2_sn3})
   855→
   856→    # ContradictionCheck.depth3: final contradiction check
   857→    n2_cc3 = "ok" if n2_cc1 == "ok" and n2_cc2 == "ok" else "FLAGGED"
   858→    logs.append({"node": 2, "technique": "ContradictionCheck.depth3", "duration_ms": 0, "result_summary": n2_cc3})
   859→
   860→    # IdempotencyCheck.depth3: final idempotency
   861→    n2_ic3 = f"final_key={n2_route_hash}_{path}"
   862→    logs.append({"node": 2, "technique": "IdempotencyCheck.depth3", "duration_ms": 0, "result_summary": n2_ic3})
   863→
   864→    # SufficiencyCheck: enough info to proceed?
   865→    n2_suff_inputs = [bool(selected_tier_short), bool(path), bool(ticket_type), bool(action)]
   866→    n2_suff_count = sum(n2_suff_inputs)
   867→    n2_suff1 = "sufficient" if n2_suff_count == 4 else f"insufficient: {n2_suff_count}/4"
   868→    logs.append({"node": 2, "technique": "SufficiencyCheck", "duration_ms": 0, "result_summary": n2_suff1})
   869→
   870→    # Escalation.depth3: route-based escalation summary
   871→    n2_esc3 = "no_escalation" if n2_esc1 == "no_escalation" and n2_esc2 == "no_escalation" else "escalation_flagged"
   872→    logs.append({"node": 2, "technique": "Escalation.depth3", "duration_ms": 0, "result_summary": n2_esc3})
   873→
   874→    elapsed = int((time.time() - start) * 1000)
   875→    logger.info(
   876→        "Node 2 complete: ticket=%s tier=%s(%s) path=%s source=%s [%dms]",
   877→        state.get("ticket_id", "?"),
   878→        selected_tier_short, selected_tier_db,
   879→        path, source, elapsed,
   880→    )
   881→
   882→    # ── P2 Notification: emit ticket:routed when routing is surprising ──
   883→    # Fires when the routing decision is unusual:
   884→    # - Path was overridden from simple → complex (query needed full reasoning)
   885→    # - Fallback tier was used (no tier had quota or could handle the ticket)
   886→    # This helps the human understand why a ticket went to a higher tier than expected.
   887→    try:
   888→        from app.core.event_emitter import emit_ticket_event
   889→        tenant_id = state.get("tenant_id", "")
   890→        ticket_id = state.get("ticket_id", "")
   891→        is_surprising = False
   892→        surprise_reason = ""
   893→
   894→        # Check if FORCE_COMPLEX_PATH was used (all tickets forced to complex)
   895→        if os.environ.get("FORCE_COMPLEX_PATH", "").lower() in ("1", "true", "yes"):
   896→            is_surprising = True
   897→            surprise_reason = "force_complex_path_enabled"
   898→
   899→        # Check if quota is exhausted (quota_remaining is a dict per tier)
   900→        # If ANY tier has 0 remaining, that's surprising
   901→        if isinstance(quota_remaining, dict):
   902→            any_exhausted = any(v <= 0 for v in quota_remaining.values() if isinstance(v, (int, float)))
   903→            if any_exhausted and not is_surprising:
   904→                is_surprising = True
   905→                surprise_reason = "quota_exhausted_fallback_to_highest_tier"
   906→
   907→        if is_surprising:
   908→            await emit_ticket_event(
   909→                company_id=tenant_id,
   910→                event_type="ticket:routed",
   911→                payload={
   912→                    "company_id": tenant_id,
   913→                    "ticket_id": ticket_id,
   914→                    "ticket_type": ticket_type,
   915→                    "complexity": complexity,
   916→                    "selected_tier": selected_tier_db,
   917→                    "path": path,
   918→                    "surprise_reason": surprise_reason,
   919→                    "quota_remaining": quota_remaining,
   920→                    "node": 2,
   921→                },
   922→                correlation_id=ticket_id,
   923→            )
   924→    except Exception as exc:
   925→        logger.warning("node_2_routed_notification_failed: %s", str(exc)[:200])
   926→
   927→    return {
   928→        "variant_tier": selected_tier_db,
   929→        "variant_tier_short": selected_tier_short,
   930→        "quota_remaining": quota_remaining,
   931→        "route_decision": path,
   932→        "current_path": path,
   933→        "variant_capabilities": capabilities,
   934→        "technique_log": logs,
   935→    }

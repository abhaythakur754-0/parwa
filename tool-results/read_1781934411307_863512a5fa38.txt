   145→
   146→def _retrieve_knowledge(ticket_type: str, query: str = "") -> List[Dict[str, Any]]:
   147→    """Retrieve knowledge documents based on ticket type.
   148→    Phase 4: Cleaner, no unused query parameters.
   149→    Phase 7: Added T2→T1 cross-type detection — scans query for patterns
   150→    that indicate the ticket spans multiple types, and pulls those KB docs."""
   151→    # Primary: exact type match
   152→    docs = KNOWLEDGE_BASE.get(ticket_type, [])
   153→
   154→    # Secondary: also pull from related types
   155→    related_types = {
   156→        "refund_request": ["billing"],
   157→        "billing": ["refund_request", "faq"],
   158→        "account_change": ["technical", "faq"],
   159→        "technical": ["account_change", "faq"],
   160→        "complaint": ["billing", "faq"],
   161→        "faq": ["billing", "account_change", "refund_request"],
   162→    }
   163→    for rt in related_types.get(ticket_type, []):
   164→        docs.extend(KNOWLEDGE_BASE.get(rt, []))
   165→
   166→    # Phase 7: T2→T1 Cross-type pattern detection
   167→    # If the query contains strong signals for a different type that isn't
   168→    # already covered by primary or related_types, pull those docs too.
   169→    if query:
   170→        query_lower = query.lower()
   171→        # Define cross-type signal patterns (query signals → should pull this type)
   172→        cross_type_signals = {
   173→            "refund_request": [r"\brefund\b", r"\bmoney\s+back\b", r"\breturn\s+my\b", r"\bprorated?\b", r"\beligibl", r"\btermination\s+fee\b"],
   174→            "billing": [r"\bcharg(?:ed|e|es|ing)\b", r"\binvoice\b", r"\bpayment\b", r"\bovercharge", r"\bdouble\s+charge\b", r"\bpricing\b"],
   175→            "technical": [r"\blogin\b", r"\bss[so]\b", r"\bpassword\b", r"\berror\b", r"\bcrash\b", r"\bnot\s+working\b", r"\bcan'?t\s+access\b", r"\bsession\b", r"\bcache\b"],
   176→            "faq": [r"\bwhat\s+(?:is|are)\b", r"\bhow\s+(?:do|does|much|many)\b", r"\bfeature", r"\b(?:plan|plans?)\s*(?:pricing|cost|include)\b"],
   177→            "complaint": [r"\bterrible\b", r"\bworst\b", r"\bunacceptab", r"\bfrustrat", r"\bdisappoint", r"\bnever\s+again\b"],
   178→            "account_change": [r"\bchange\s+(?:email|password)\b", r"\bupgrade\b", r"\bdowngrade\b", r"\bswitch\s+plan\b", r"\bworkspace\b", r"\bteam\s+member\b"],
   179→        }
   180→        # Determine which types are already covered
   181→        covered_types = {ticket_type} | set(related_types.get(ticket_type, []))
   182→        for cross_type, patterns in cross_type_signals.items():
   183→            if cross_type in covered_types:
   184→                continue
   185→            # If 2+ patterns match, this type is likely relevant
   186→            match_count = sum(1 for pat in patterns if re.search(pat, query_lower))
   187→            if match_count >= 2:
   188→                extra_docs = KNOWLEDGE_BASE.get(cross_type, [])
   189→                docs.extend(extra_docs)
   190→
   191→    # Deduplicate by source
   192→    seen = set()
   193→    unique_docs = []
   194→    for d in docs:
   195→        if d["source"] not in seen:
   196→            seen.add(d["source"])
   197→            unique_docs.append({
   198→                "content": d["content"],
   199→                "source": d["source"],
   200→                "section": d.get("section", "C"),
   201→            })
   202→
   203→    return unique_docs

   206→# ── Phase 4: Smart knowledge filtering (non-LLM) ─────────────────
   207→
   208→
   209→def _filter_relevant_docs(
   210→    documents: List[Dict[str, Any]], query: str, ticket_type: str
   211→) -> List[Dict[str, Any]]:
   212→    """Score each doc by keyword relevance to query, keep top results.
   213→    This ensures only the most relevant KB chunks go to downstream nodes."""
   214→    if not documents:
   215→        return documents
   216→
   217→    query_lower = query.lower()
   218→    # Extract significant query terms
   219→    query_terms = set(w.lower() for w in query_lower.split() if len(w) > 3)
   220→    # Also add ticket-type-specific key terms
   221→    type_keywords = {
   222→        "refund_request": {"refund", "prorated", "termination", "fee", "credit", "eligible", "days", "annual", "monthly"},
   223→        "billing": {"charge", "invoice", "payment", "billed", "subscription", "duplicate", "price", "plan", "rate"},
   224→        "technical": {"login", "sso", "password", "access", "error", "cache", "sync", "session", "lockout", "certificate"},
   225→        "faq": {"plan", "pricing", "feature", "channel", "agent", "ticket", "onboarding", "integration"},
   226→        "complaint": {"complaint", "compensation", "credit", "escalate", "resolution", "priority", "tenure"},
   227→        "account_change": {"email", "password", "plan", "upgrade", "downgrade", "workspace", "member", "sso"},
   228→    }
   229→    query_terms |= type_keywords.get(ticket_type, set())
   230→
   231→    # Remove common filler
   232→    filler = {"that", "this", "have", "been", "will", "would", "could", "should",
   233→              "their", "there", "about", "which", "where", "when", "what", "with",
   234→              "from", "your", "just", "also", "than", "them", "they", "some"}
   235→    query_terms -= filler
   236→
   237→    # Score each document
   238→    scored_docs = []
   239→    for doc in documents:
   240→        text = doc.get("content", "").lower()
   241→        doc_words = set(text.split())
   242→        overlap = len(query_terms & doc_words)
   243→        # Also check for query phrase presence (bigrams)
   244→        query_bigrams = set()
   245→        words = query_lower.split()
   246→        for i in range(len(words) - 1):
   247→            bg = f"{words[i]} {words[i+1]}"
   248→            if len(bg) > 6:
   249→                query_bigrams.add(bg)
   250→        bigram_hits = sum(1 for bg in query_bigrams if bg in text)
   251→
   252→        score = overlap + (bigram_hits * 2)  # bigrams worth more
   253→        scored_docs.append((score, doc))
   254→
   255→    # Sort by relevance, keep all (but order matters for downstream truncation)
   256→    scored_docs.sort(key=lambda x: x[0], reverse=True)
   257→
   258→    # If we have more than 8 docs, trim to top 8 (reduces downstream token waste)
   259→    if len(scored_docs) > 8:
   260→        scored_docs = scored_docs[:8]
   261→
   262→    return [doc for _, doc in scored_docs]

   265→# ── Phase 4: Non-LLM CLARA sufficiency check (replaces 1 LLM call) ─
   266→
   267→
   268→def _check_knowledge_sufficiency(
   269→    documents: List[Dict[str, Any]], query: str, ticket_type: str
   270→) -> Dict[str, bool]:
   271→    """Non-LLM heuristic: do we have KB docs for this ticket type?
   272→    Replaces the LLM-based CLARA re-evaluate from Phase 2."""
   273→    has_primary = any(d.get("source", "") in {
   274→        "refund_policy_v2", "billing_policy_v2", "tech_troubleshooting_v2",
   275→        "general_faq_v2", "complaint_handling_v2", "account_policy_v2",
   276→    } for d in documents)
   277→
   278→    # If we have docs for the primary type, knowledge is likely sufficient
   279→    type_doc_counts = {
   280→        "refund_request": 4, "billing": 3, "technical": 2,
   281→        "faq": 3, "complaint": 1, "account_change": 2,
   282→    }
   283→    expected = type_doc_counts.get(ticket_type, 2)
   284→    sufficient = len(documents) >= max(expected - 1, 1) and has_primary
   285→
   286→    return {
   287→        "knowledge_sufficient": sufficient,
   288→        "knowledge_contradictory": False,  # our KB is internally consistent
   289→    }

   292→# ── Helpers ────────────────────────────────────────────────────────

   294→
   295→def _check_contradictions(documents: List[Dict[str, Any]]) -> bool:
   296→    """Quick keyword-level contradiction check."""
   297→    if len(documents) < 2:
   298→        return False
   299→    all_numbers = {}
   300→    for doc in documents:
   301→        text = doc.get("content", "")
   302→        for match in re.finditer(r"(\d+)%", text):
   303→            num = int(match.group(1))
   304→            context = text[max(0, match.start()-30):match.end()+30].lower()
   305→            for keyword in ["refund", "fee", "credit", "discount"]:
   306→                if keyword in context:
   307→                    if keyword in all_numbers and all_numbers[keyword] != num:
   308→                        return True
   309→                    all_numbers[keyword] = num
   310→    return False

   313→def _read_ai_wiki(tenant_id: str, ticket_type: str, query: str, tier: str = "parwa") -> tuple:
   314→    """Phase 6: Read AI Wiki Sections A, B, C from real store.
   315→    
   316→    Returns (section_a_entries, section_b_entries, section_c_entries)
   317→    as lists of dicts in node format (same shape as KB docs).
   318→    All non-LLM — keyword search only.
   319→    """
   320→    from app.core.parwa_pipeline.ai_wiki_store import get_wiki_store
   321→    
   322→    wiki = get_wiki_store()
   323→    wiki_a_raw = []
   324→    wiki_b_raw = []
   325→    wiki_c_raw = []
   326→    
   327→    try:
   328→        # Section A: Search for similar ticket patterns (top 3)
   329→        wiki_a_raw = wiki.search(
   330→            tenant_id=tenant_id, section="A", query=query,
   331→            ticket_type=ticket_type, tier=tier, max_results=3,
   332→        )
   333→        
   334→        # Section B: Read admin behavior patterns (all, max 2)
   335→        wiki_b_raw = wiki.read(tenant_id=tenant_id, section="B", tier=tier)
   336→        wiki_b_raw = wiki_b_raw[:2]  # limit for context window
   337→        
   338→        # Section C: Read company knowledge (all, max 3)
   339→        wiki_c_raw = wiki.read(tenant_id=tenant_id, section="C", tier=tier)
   340→        wiki_c_raw = wiki_c_raw[:3]
   341→    except Exception as e:
   342→        logger.warning("Wiki read error (non-fatal): %s", e)
   343→    
   344→    # Convert to node format (same shape as KB docs)
   345→    wiki_a = [e.to_node_format() for e in wiki_a_raw]
   346→    wiki_b = [e.to_node_format() for e in wiki_b_raw]
   347→    wiki_c = [e.to_node_format() for e in wiki_c_raw]
   348→    
   349→    # Build wiki summary for similar patterns found
   350→    wiki_patterns = []
   351→    for entry in wiki_a_raw:
   352→        c = entry.content
   353→        wiki_patterns.append({
   354→            "entry_key": entry.entry_key,
   355→            "techniques_that_worked": c.get("techniques_that_worked", []),
   356→            "quality_achieved": c.get("quality_achieved", 0),
   357→            "answer_summary": c.get("answer_summary", "")[:200],
   358→            "historical_success_rate": round(entry.success_count / max(entry.usage_count, 1), 3),
   359→        })
   360→    
   361→    return wiki_a, wiki_b, wiki_c, wiki_patterns

   133→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   134→
   135→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   136→  agentsReplaced: number;
   137→  humanCost: number;
   138→  savings: number;
   139→  savingsPercent: number;
   140→} {
   141→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   142→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   143→  const savings = Math.max(0, humanCost - monthlyCost);
   144→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   145→  return { agentsReplaced, humanCost, savings, savingsPercent };
   146→}

   455→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   456→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   457→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   458→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   459→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   460→  const [isSubmitting, setIsSubmitting] = useState(false);
   461→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   462→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   463→  const [appliedCoupon, setAppliedCoupon] = useState<Coupon | null>(null);
   464→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   465→
   466→  // Restore add-ons and multi-variant selections from localStorage
   467→  useEffect(() => {
   468→    try {
   469→      const stored = localStorage.getItem('parwa_pricing_context');
   470→      if (stored) {
   471→        const ctx = JSON.parse(stored) as PricingContext & {
   472→          selectedVariants?: string[];
   473→          variantQuantities?: Record<string, number>;
   474→        };
   475→        if (ctx.addOns) {
   476→          setAddOns(ctx.addOns);
   477→        }
   478→        // Restore multiple variants if selected on ModelsPage
   479→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   480→          const tiers: VariantTier[] = ctx.selectedVariants
   481→            .map((v: string) => {
   482→              const normalized = normalizeTier(v);
   483→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   484→            })
   485→            .filter(Boolean) as VariantTier[];
   486→          if (tiers.length > 0) {
   487→            setActiveVariants(tiers);
   488→          }
   489→        }
   490→        // Restore variant quantities
   491→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   492→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   493→          const newQuantities = { ...variantQuantities };
   494→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   495→            const tier = normalizeTier(variantKey);
   496→            if (['starter', 'growth', 'high'].includes(tier)) {
   497→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   498→            }
   499→          }
   500→          setVariantQuantities(newQuantities);
   501→        }
   502→        // Restore coupon from ModelsPage
   503→        if (ctx.couponCode && typeof ctx.couponCode === 'string') {
   504→          const coupon = validateCoupon(ctx.couponCode);
   505→          if (coupon) {
   506→            setAppliedCoupon(coupon);
   507→          }
   508→        }
   509→      }
   510→    } catch {
   511→      // ignore
   512→    }
   513→  }, []);
   514→
   515→  // Check if Paddle.js is available
   516→  useEffect(() => {
   517→    getPaddleInstance().then((paddle) => {
   518→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   519→    }).catch(() => {
   520→      setPaddleStatus('unavailable');
   521→    });
   522→  }, []);
   523→
   524→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   525→
   526→  const totalTicketLimit = useMemo(
   527→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   528→    [activeVariants, variantQuantities]
   529→  );
   530→
   531→  const baseSubscription = useMemo(
   532→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   533→    [activeVariants, variantQuantities]
   534→  );
   535→
   536→  const addOnTotal = useMemo(() => {

   538→    if (addOns.voice) {
   539→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   540→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   541→        total += voiceAddOn.price;
   542→      }
   543→    }
   544→    if (addOns.customApi) {
   545→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   546→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   547→        total += customApiAddOn.price;
   548→      }
   549→    }
   550→    return total;
   551→  }, [addOns, activeVariants]);
   552→
   553→  const totalMonthly = baseSubscription + addOnTotal;

   555→  // ── Coupon-discounted total ──────────────────────────────────────
   556→  const discountedTotal = useMemo(
   557→    () => applyCouponDiscount(totalMonthly, appliedCoupon),
   558→    [totalMonthly, appliedCoupon]

   561→

   563→  const variantDiscountedPrices = useMemo(() => {

   567→      prices[tier] = applyCouponDiscount(variantPrice, appliedCoupon);
   568→    }
   569→    return prices;
   570→  }, [activeVariants, variantQuantities, appliedCoupon]);
   571→
   572→  const savings = estimateSavings(totalTicketLimit, discountedTotal);
   573→
   574→  // Overage projection (estimates at 80% utilization)
   575→  const projectedTickets = Math.round(totalTicketLimit * 0.8);
   576→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);
   577→  const overageCost = overageTickets * OVERAGE_PRICE_PER_TICKET;
   578→

   580→    setActiveVariants((prev) => {

   582→        if (prev.length <= 1) {
   583→          toast.error('You must have at least one active variant');
   584→          return prev;
   585→        }
   586→        return prev.filter((t) => t !== tier);
   587→      }
   588→      return [...prev, tier];
   589→    });
   590→  };
   591→
   592→  const updateQuantity = (tier: VariantTier, qty: number) => {
   593→    const newQty = Math.max(1, Math.min(qty, 10));
   594→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   595→  };
   596→
   597→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   598→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   599→  };
   600→
   601→  const handleApplyCoupon = (coupon: Coupon) => {
   602→    setAppliedCoupon(coupon);
   603→    setCheckoutError(null);
   604→  };
   605→
   606→  const handleRemoveCoupon = () => {
   607→    setAppliedCoupon(null);
   608→    toast('Coupon removed');
   609→  };
   610→
   611→  // ── Payment success handler — ONLY called after Paddle confirms ──
   612→  const handlePaymentSuccess = () => {
   613→    setPaymentConfirmed(true);
   614→    localStorage.removeItem('parwa_payment_pending');
   615→    if (isFreeCheckout) {
   616→      toast.success('Subscription activated! Welcome to PARWA!');
   617→    } else {
   618→      toast.success('Payment successful! Welcome to PARWA!');
   619→    }
   620→    // Only trigger onboarding completion AFTER Paddle confirms
   621→    onComplete();
   622→  };
   623→
   624→  const handleProceed = async () => {
   625→    setIsSubmitting(true);
   626→    setCheckoutError(null);
   627→
   628→    try {
   629→      // Update pricing context in localStorage
   630→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   631→      const context: PricingContext = {
   632→        industry: 'other',
   633→        variant: primaryOnboardingVariant,
   634→        addOns,
   635→        totalMonthly,
   636→        timestamp: new Date().toISOString(),

   639→      // Preserve industry from existing context
   640→      try {
   641→        const stored = localStorage.getItem('parwa_pricing_context');
   642→        if (stored) {
   643→          const existing = JSON.parse(stored) as PricingContext;
   644→          context.industry = existing.industry;
   645→        }

   647→        // ignore
   648→      }
   649→
   650→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   651→
   652→      // ── Paddle Checkout Flow ──────────────────────────────────────
   653→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   654→
   655→      // Add each active variant as a Paddle line item (with quantity)

   657→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   658→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   659→        const qty = variantQuantities[tier] || 1;
   660→        if (variantPriceId) {
   661→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   662→        }

   664→
   665→      // Voice add-on (only for starter where it's not included)
   666→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   667→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   668→      }
   669→
   670→      // Custom API add-on (only for variants where it's not included)
   671→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   672→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });

   674→
   675→      const customData = {
   676→        source: 'parwa_onboarding',
   677→        variant: primaryOnboardingVariant,
   678→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   679→        variantQuantities,
   680→        addOns,
   681→        industry: context.industry,
   682→        totalMonthly,
   683→        couponCode: appliedCoupon?.code || null,

   685→        isFreeCheckout,

   690→      const paddleDiscountId = getPaddleDiscountId(appliedCoupon);

   693→      let checkoutUrl: string | null = null;
   694→
   695→      try {
   696→        const res = await fetch('/api/onboarding/checkout', {
   697→          method: 'POST',
   698→          headers: { 'Content-Type': 'application/json' },
   699→          body: JSON.stringify({
   700→            variant: primaryOnboardingVariant,
   701→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   702→            variantQuantities,
   703→            addOns,
   704→            totalMonthly,
   705→            industry: context.industry,

   715→          const checkoutData = await res.json();
   716→          checkoutUrl = checkoutData.checkout_url || null;

   719→        // API unavailable — continue with client-side Paddle checkout
   720→      }
   721→
   722→      // If we got a checkout_url from backend, redirect to it
   723→      if (checkoutUrl) {
   724→        localStorage.setItem('parwa_payment_pending', JSON.stringify({
   725→          activeVariants, addOns, totalMonthly, variantQuantities,
   726→          couponCode: appliedCoupon?.code || null,
   727→          pendingAt: new Date().toISOString(),

   732→
   733→      // ── Step 2: Client-side Paddle checkout (items-based overlay) ──

   737→      if (paddle && checkoutItems.length > 0) {

   740→          checkoutItems,
   741→          customData,
   742→          // onPaymentSuccess — ONLY triggers after Paddle confirms the transaction

   744→          // onCheckoutClosed — user closed without paying

   751→          paddleDiscountId,

   761→      try {
   762→        const retryPaddle = await getPaddleInstance();
   763→        if (retryPaddle && checkoutItems.length > 0) {
   764→          setPaddleStatus('ready');
   765→          const opened = await openCheckoutWithItems(
   766→            checkoutItems,
   767→            customData,
   768→            handlePaymentSuccess,
   769→            () => {
   770→              toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   771→              setIsSubmitting(false);
   772→            },
   773→            paddleDiscountCode,
   774→            paddleDiscountId,
   775→          );
   776→          if (opened) return;
   777→        }
   778→      } catch {
   779→        // Paddle truly unavailable
   780→      }
   781→
   782→      // ── Step 4: Paddle unavailable — handle based on checkout type ──
   783→      setPaddleStatus('unavailable');
   784→      localStorage.setItem('parwa_payment_pending', JSON.stringify({
   785→        activeVariants, addOns, totalMonthly, variantQuantities,
   786→        couponCode: appliedCoupon?.code || null,
   787→        pendingAt: new Date().toISOString(),
   788→      }));
   789→
   790→      if (isFreeCheckout) {
   791→        // $0 checkout (100% coupon) — Paddle gateway is unavailable but the total is $0.
   792→        // We still want to confirm the subscription through Paddle if possible, but if
   793→        // Paddle is down, we can safely activate the free subscription since no money
   794→        // is changing hands. The user applied a valid 100% coupon (e.g. "durga754").
   795→        console.log('[cost-breakdown] Free checkout ($0) with Paddle unavailable — activating subscription');
   796→        toast.success('Free plan activated! Welcome to PARWA!');
   797→        handlePaymentSuccess();
   798→      } else {
   799→        // Paid checkout — payment gateway is REQUIRED. Block and show error.
   800→        setCheckoutError(
   801→          'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   802→          'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'
   803→        );
   804→        toast.error('Payment gateway unavailable — please try again or contact support.');
   805→      }
   806→    } catch (err) {
   807→      console.error('[cost-breakdown] Error:', err);
   808→      setCheckoutError('Something went wrong. Please try again.');
   809→      toast.error('Something went wrong. Please try again.');
   810→    } finally {
   811→      setIsSubmitting(false);
   812→    }
   813→  };
   814→
   815→  return (
   816→    <div className="space-y-6">
   817→      {/* Header */}
   818→      <div className="text-center space-y-2">
   819→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   820→          <Receipt className="w-7 h-7 text-emerald-400" />
   821→        </div>
   822→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   823→        <p className="text-orange-200/40 text-sm">
   824→          Configure your plan and complete payment to activate your AI agents.
   825→        </p>
   826→      </div>
   827→
   828→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   829→      {paddleStatus === 'unavailable' && (
   830→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   831→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   832→          <div>
   833→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   834→            <p className="text-[10px] text-orange-200/30 mt-1">
   835→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   836→            </p>
   837→          </div>
   838→        </div>
   839→      )}
   840→      {paddleStatus === 'ready' && (
   841→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   842→          <Shield className="w-4 h-4 text-emerald-400" />
   843→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   844→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   845→        </div>
   846→      )}
   847→
   848→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   849→      <div className="space-y-3">
   850→        <div className="flex items-center gap-2">
   851→          <Brain className="w-4 h-4 text-orange-400" />
   852→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   853→            Active Variants
   854→          </label>
   855→          <span className="text-[10px] text-orange-200/25">
   856→            Each variant adds its own ticket allocation and AI pipeline
   857→          </span>
   858→        </div>
   859→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   860→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   861→            <VariantMixerCard
   862→              key={tier}
   863→              tier={tier}
   864→              isActive={activeVariants.includes(tier)}
   865→              quantity={variantQuantities[tier] || 1}
   866→              onToggle={() => toggleVariant(tier)}
   867→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   868→              discountedPrice={activeVariants.includes(tier) ? variantDiscountedPrices[tier] : undefined}
   869→            />
   870→          ))}
   871→        </div>
   872→      </div>
   873→
   874→      {/* ── 2. Live Usage Bar & Overage Projection ──────────────────── */}
   875→      <div className="rounded-xl border border-white/[0.06] p-4 space-y-4" style={{ background: 'rgba(255,255,255,0.03)' }}>
   876→        <div className="flex items-center gap-2">
   877→          <BarChart3 className="w-4 h-4 text-orange-400" />
   878→          <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   879→            Live Usage & Overage Projection
   880→          </span>
   881→        </div>
   882→        <UsageBar
   883→          used={projectedTickets}
   884→          total={totalTicketLimit}
   885→          label="Ticket Usage (projected at 80%)"
   886→          unit="tickets"
   887→          overageRate={`$${OVERAGE_PRICE_PER_TICKET}`}
   888→        />
   889→        {overageCost > 0 && (
   890→          <div className="rounded-lg border border-red-500/20 p-3 flex items-start gap-2" style={{ background: 'rgba(239,68,68,0.05)' }}>
   891→            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0 mt-0.5" />
   892→            <div>
   893→              <p className="text-xs font-medium text-red-400">Overage Projected</p>
   894→              <p className="text-[10px] text-orange-200/30 mt-0.5">
   895→                At 80% utilization, projected overage: ${overageCost.toFixed(2)} ({overageTickets.toLocaleString()} tickets × ${OVERAGE_PRICE_PER_TICKET}/ticket). Overage charged at end of billing cycle.
   896→              </p>
   897→            </div>
   898→          </div>
   899→        )}
   900→        {overageCost === 0 && (
   901→          <div className="rounded-lg border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   902→            <Shield className="w-3.5 h-3.5 text-emerald-400" />
   903→            <p className="text-[10px] text-emerald-400">Within limits at projected usage. No overage charges.</p>
   904→          </div>
   905→        )}
   906→      </div>
   907→
   908→      {/* ── 3. Add-Ons ───────────────────────────────────────────────── */}
   909→      <div className="space-y-3">
   910→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   911→          Optional Add-Ons
   912→        </label>
   913→        {ADD_ONS.map((addOn) => {
   914→          const Icon = addOn.icon;
   915→          const isSelected = addOns[addOn.key];
   916→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   917→          const showPrice = !isIncluded;
   918→
   919→          return (
   920→            <button
   921→              key={addOn.key}
   922→              type="button"
   923→              onClick={() => toggleAddOn(addOn.key)}
   924→              className={cn(
   925→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',
   926→                isSelected
   927→                  ? 'border-orange-500/30 bg-orange-500/5'
   928→                  : 'border-white/[0.06] hover:border-orange-500/15'
   929→              )}
   930→              style={!isSelected ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   931→            >
   932→              <div className={cn(
   933→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   934→                isSelected ? 'bg-orange-500/10' : 'bg-white/[0.04]'
   935→              )}>
   936→                <Icon className={cn(
   937→                  'w-5 h-5',
   938→                  isSelected ? 'text-orange-400' : 'text-zinc-500'
   939→                )} />
   940→              </div>
   941→              <div className="flex-1 min-w-0">
   942→                <div className="flex items-center gap-2">
   943→                  <p className={cn(
   944→                    'text-sm font-medium',
   945→                    isSelected ? 'text-orange-400' : 'text-white'
   946→                  )}>
   947→                    {addOn.name}
   948→                  </p>

   950→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">

   961→                    isFreeCheckout ? 'text-emerald-400' : 'text-white'

   963→                    {isFreeCheckout ? '$0/mo' : `$${addOn.price}/mo`}
   964→                  </p>

   967→                  'w-8 h-5 rounded-full flex items-center px-0.5 transition-colors duration-200',
   968→                  isSelected || isIncluded
   969→                    ? 'bg-gradient-to-r from-orange-500 to-amber-400'
   970→                    : 'bg-white/10'
   971→                )}>
   972→                  <div className={cn(
   973→                    'w-4 h-4 rounded-full bg-white shadow-sm transition-transform duration-200',
   974→                    (isSelected || isIncluded) ? 'translate-x-3' : 'translate-x-0'

   976→                </div>
   977→              </div>

   980→        })}

   984→      <CouponCodeInput
   985→        appliedCoupon={appliedCoupon}

   988→      />
   989→
   990→      {/* ── 5. Cost Summary (pure math — D7) ──────────────────────────── */}

   992→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   993→        style={{ background: 'rgba(255,255,255,0.03)' }}

   999→          const originalPrice = VARIANT_PRICES[tier] * qty;

  1006→                  'w-2 h-2 rounded-full',

  1020→                )}
  1021→                ${discounted.toLocaleString()}/mo
  1022→              </span>
  1023→            </div>

  1026→
  1027→        {/* Add-ons */}
  1028→        {addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t)) && (

     1→"""
     2→Payment Failure Service (F-027, BG-16)
     3→
     4→Netflix-style payment failure handling:
     5→1. Payment fails → IMMEDIATE service stop
     6→2. No grace period
     7→3. No dunning emails
     8→4. Single notification: "Payment failed, update payment method"
     9→5. Service resumes immediately on successful payment
    10→
    11→ENHANCED (2025-01-19) - RAM-to-DB Migration:
    12→6. Auto-block after configurable hours if payment not received
    13→7. Data cleanup for old resolved failures
    14→
    15→This is deliberately strict to reduce churn from forgotten cards.
    16→
    17→BC-001: All operations validate company_id
    18→BC-002: All money calculations use Decimal
    19→CLAUDE.md Rule 2: Uses EXISTING payment_failures table (no duplication)
    20→"""

    73→    async def handle_payment_failure(
    74→        self,
    75→        company_id: UUID,
    76→        paddle_transaction_id: str,
    77→        failure_code: str,
    78→        failure_reason: str,
    79→        amount_attempted: Decimal,
    80→        paddle_subscription_id: Optional[str] = None,
    81→        currency: str = "USD",
    82→    ) -> Dict[str, Any]:
    83→        """
    84→        Handle a payment failure with immediate service stop.
    85→
    86→        Steps:
    87→        1. Log to payment_failures table
    88→        2. Set company.subscription_status = 'payment_failed'
    89→        3. Stop all AI agents immediately
    90→        4. Block new ticket creation
    91→        5. Freeze existing open tickets (status → 'frozen')
    92→        6. Send payment_failed email notification
    93→        7. Create Socket.io event for real-time UI update
    94→
    95→        Args:
    96→            company_id: Company UUID
    97→            paddle_transaction_id: Paddle transaction ID
    98→            failure_code: Paddle failure code
    99→            failure_reason: Human-readable failure reason
   100→            amount_attempted: Amount that failed to charge
   101→            paddle_subscription_id: Optional Paddle subscription ID
   102→            currency: Currency code (default USD)
   103→
   104→        Returns:
   105→            Dict with failure record details and actions taken
   106→
   107→        Raises:
   108→            PaymentFailureError: If failure cannot be processed
   109→        """
   110→
   111→        def _db_work():
   112→            with SessionLocal() as db:
   113→                # Get company with row lock to prevent race conditions
   114→                company = db.query(Company).filter(
   115→                    Company.id == str(company_id)
   116→                ).with_for_update().first()
   117→
   118→                if not company:
   119→                    raise PaymentFailureError(f"Company {company_id} not found")
   120→
   121→                now = datetime.now(timezone.utc)
   122→
   123→                # Check if there's already an unresolved payment failure
   124→                existing_failure = db.query(PaymentFailure).filter(
   125→                    PaymentFailure.company_id == str(company_id),
   126→                    PaymentFailure.resolved == False,
   127→                ).first()
   128→
   129→                if existing_failure:
   130→                    # Already has active payment failure - just log this one
   131→                    logger.warning(
   132→                        "payment_failure_already_stopped "
   133→                        "company_id=%s existing_failure_id=%s "
   134→                        "new_transaction_id=%s",
   135→                        company_id,
   136→                        existing_failure.id,
   137→                        paddle_transaction_id,
   138→                    )
   139→                    return {
   140→                        "status": "already_stopped",
   141→                        "existing_failure_id": existing_failure.id,
   142→                        "message": "Service already stopped due to previous payment failure",
   143→                    }
   144→
   145→                # Create payment failure record
   146→                failure = PaymentFailure(
   147→                    company_id=str(company_id),
   148→                    paddle_subscription_id=paddle_subscription_id,
   149→                    paddle_transaction_id=paddle_transaction_id,
   150→                    failure_code=failure_code,
   151→                    failure_reason=failure_reason,
   152→                    amount_attempted=amount_attempted,
   153→                    currency=currency,
   154→                    service_stopped_at=now,
   155→                    notification_sent=False,
   156→                    resolved=False,
   157→                )
   158→                db.add(failure)
   159→
   160→                # Update company subscription status
   161→                old_status = company.subscription_status
   162→                company.subscription_status = self.PAYMENT_FAILED_STATUS
   163→
   164→                # Update subscription record if exists
   165→                subscription = db.query(Subscription).filter(
   166→                    Subscription.company_id == str(company_id),
   167→                    Subscription.status == self.ACTIVE_STATUS,
   168→                ).first()
   169→
   170→                if subscription:
   171→                    subscription.status = self.PAYMENT_FAILED_STATUS
   172→
   173→                db.commit()
   174→                db.refresh(failure)
   175→
   176→                logger.info(
   177→                    "payment_failure_handled "
   178→                    "company_id=%s failure_id=%s "
   179→                    "old_status=%s new_status=%s "
   180→                    "amount=%s code=%s",
   181→                    company_id,
   182→                    failure.id,
   183→                    old_status,
   184→                    self.PAYMENT_FAILED_STATUS,
   185→                    amount_attempted,
   186→                    failure_code,
   187→                )
   188→
   189→                # Trigger side-effects for payment failure
   190→                try:
   191→                    # Send payment failure notification email
   192→                    from app.services.email_service import send_email
   193→                    from database.models.core import User
   194→                    company_owner = db.query(User).filter(
   195→                        User.company_id == str(company_id),
   196→                        User.role == "owner",
   197→                    ).first()
   198→                    if company_owner:
   199→                        try:
   200→                            send_email(
   201→                                to=company_owner.email,
   202→                                subject="PARWA: Payment Failed — Action Required",
   203→                                html_content=f"""
   204→                                <html><body>
   205→                                <h2>Payment Failed</h2>
   206→                                <p>Hello {company_owner.full_name or 'there'},</p>
   207→                                <p>We were unable to process your recent payment for your PARWA subscription.</p>
   208→                                <p><strong>Company:</strong> {company.name}</p>
   209→                                <p><strong>Reason:</strong> {failure_reason or 'Payment declined by payment provider'}</p>
   210→                                <p>Your service has been paused. Update your payment method to restore access immediately.</p>
   211→                                </body></html>
   212→                                """,
   213→                            )
   214→                            # Mark notification as sent after successful email delivery
   215→                            failure.notification_sent = True
   216→                            db.commit()
   217→                        except Exception as email_err:
   218→                            logger.error(
   219→                                "payment_failure_email_failed",
   220→                                company_id=str(company_id),
   221→                                error=str(email_err),
   222→                            )
   223→
   224→                    # Notify via notification service
   225→                    try:
   226→                        from app.services.notification_service import NotificationService
   227→                        notif_svc = NotificationService(db)
   228→                        notif_svc.create_notification(
   229→                            user_id=str(company_owner.id) if company_owner else None,
   230→                            company_id=str(company_id),
   231→                            event_type="payment_failed",
   232→                            title="Payment Failed",
   233→                            message=f"Payment processing failed for {company.name}. Please update your billing information.",
   234→                            priority="high",
   235→                        )
   236→                    except Exception as notif_err:
   237→                        logger.error(
   238→                            "payment_failure_notification_failed",
   239→                            company_id=str(company_id),
   240→                            error=str(notif_err),
   241→                        )
   242→
   243→                except Exception as e:
   244→                    logger.error(
   245→                        "payment_failure_side_effects_failed",
   246→                        company_id=str(company_id),
   247→                        error=str(e),
   248→                    )
   249→
   250→                return {
   251→                    "status": "stopped",
   252→                    "failure_id": failure.id,
   253→                    "company_id": str(company_id),
   254→                    "old_status": old_status,
   255→                    "new_status": self.PAYMENT_FAILED_STATUS,
   256→                    "service_stopped_at": now.isoformat(),
   257→                    "actions_taken": [
   258→                        "logged_failure",
   259→                        "updated_subscription_status",
   260→                        "service_stopped",
   261→                    ],
   262→                }
   263→
   264→        return await asyncio.to_thread(_db_work)

   270→        Args:
   271→            company_id: Company UUID

   274→            True if service is stopped, False otherwise

   277→        def _db_work():

   284→
   285→                if failure:

   288→                # Also check company subscription status

   292→
   293→                if company and company.subscription_status == self.PAYMENT_FAILED_STATUS:
   294→                    return True
   295→
   296→                return False
   297→

   300→    async def resume_service(
   301→        self,
   302→        company_id: UUID,
   303→        paddle_transaction_id: str,
   304→    ) -> Dict[str, Any]:
   305→        """
   306→        Resume service after successful payment.
   307→
   308→        Called when payment succeeds after failure:
   309→        1. Update subscription_status → 'active'
   310→        2. Resume AI agents
   311→        3. Unfreeze tickets
   312→        4. Send service_resumed notification
   313→
   314→        Args:
   315→            company_id: Company UUID
   316→            paddle_transaction_id: Successful Paddle transaction ID
   317→
   318→        Returns:
   319→            Dict with resume details
   320→
   321→        Raises:
   322→            PaymentFailureError: If no active failure to resume from
   323→        """
   324→
   325→        def _db_work():
   326→            with SessionLocal() as db:
   327→                # Get company with row lock
   328→                company = db.query(Company).filter(
   329→                    Company.id == str(company_id)
   330→                ).with_for_update().first()
   331→
   332→                if not company:
   333→                    raise PaymentFailureError(f"Company {company_id} not found")
   334→
   335→                # Get active payment failure
   336→                failure = db.query(PaymentFailure).filter(
   337→                    PaymentFailure.company_id == str(company_id),
   338→                    PaymentFailure.resolved == False,
   339→                ).first()
   340→
   341→                if not failure:
   342→                    logger.info(
   343→                        "payment_resume_no_failure company_id=%s", company_id
   344→                    )
   345→                    # No active failure - just ensure status is active
   346→                    if company.subscription_status != self.ACTIVE_STATUS:
   347→                        company.subscription_status = self.ACTIVE_STATUS
   348→                        db.commit()
   349→                    return {
   350→                        "status": "no_failure",
   351→                        "message": "No active payment failure to resume from",
   352→                    }
   353→
   354→                now = datetime.now(timezone.utc)
   355→
   356→                # Mark failure as resolved
   357→                failure.resolved = True
   358→                failure.service_resumed_at = now
   359→
   360→                # Update company status
   361→                old_status = company.subscription_status
   362→                company.subscription_status = self.ACTIVE_STATUS
   363→
   364→                # Update subscription record
   365→                subscription = db.query(Subscription).filter(
   366→                    Subscription.company_id == str(company_id),
   367→                ).order_by(Subscription.created_at.desc()).first()
   368→
   369→                if subscription:
   370→                    subscription.status = self.ACTIVE_STATUS
   371→
   372→                db.commit()
   373→
   374→                logger.info(
   375→                    "payment_service_resumed "
   376→                    "company_id=%s failure_id=%s "
   377→                    "old_status=%s new_status=%s",
   378→                    company_id,
   379→                    failure.id,
   380→                    old_status,
   381→                    self.ACTIVE_STATUS,
   382→                )
   383→
   384→                # Trigger resume side-effects (synchronous only)
   385→                try:
   386→                    # 1. Resume AI agents (unfreeze them)
   387→                    try:
   388→                        # TODO: Replace with actual agent service when available.
   389→                        # Expected API: AgentService.resume_all_for_company(company_id)
   390→                        from app.services.agent_service import AgentService  # type: ignore[import-untyped]
   391→                        agent_svc = AgentService(db)
   392→                        agent_svc.resume_all_for_company(str(company_id))
   393→                        logger.info(
   394→                            "payment_resume_agents company_id=%s", company_id
   395→                        )
   396→                    except ImportError:
   397→                        # Agent service not yet implemented — log and continue
   398→                        logger.warning(
   399→                            "payment_resume_agents_unavailable company_id=%s "
   400→                            "agent_service module not found; agents remain frozen",
   401→                            company_id,
   402→                        )
   403→                    except Exception as agent_err:
   404→                        logger.error(
   405→                            "payment_resume_agents_failed company_id=%s error=%s",
   406→                            company_id,
   407→                            str(agent_err),
   408→                        )
   409→
   410→                    # 2. Unfreeze tickets
   411→                    try:
   412→                        from app.services.ticket_lifecycle_service import TicketLifecycleService
   413→                        ticket_svc = TicketLifecycleService(db, str(company_id))
   414→                        thaw_result = ticket_svc.thaw_tickets_for_account()
   415→                        logger.info(
   416→                            "payment_resume_thawed_tickets company_id=%s thawed=%s",
   417→                            company_id,
   418→                            thaw_result.get("thawed_count", 0),
   419→                        )
   420→                    except Exception as ticket_err:
   421→                        logger.error(
   422→                            "payment_resume_thaw_failed company_id=%s error=%s",
   423→                            company_id,
   424→                            str(ticket_err),
   425→                        )
   426→
   427→                    # 3. Send service_resumed email
   428→                    try:
   429→                        from app.services.email_service import send_email
   430→                        from database.models.core import User
   431→                        company_owner = db.query(User).filter(
   432→                            User.company_id == str(company_id),
   433→                            User.role == "owner",
   434→                        ).first()
   435→                        if company_owner:
   436→                            send_email(
   437→                                to=company_owner.email,
   438→                                subject="PARWA: Service Restored",
   439→                                html_content=f"""
   440→                                <html><body>
   441→                                <h2>Service Restored</h2>
   442→                                <p>Hello {company_owner.full_name or 'there'},</p>
   443→                                <p>Your PARWA subscription payment has been processed successfully.</p>
   444→                                <p><strong>Company:</strong> {company.name}</p>
   445→                                <p>Your AI agents and ticket service have been restored. Thank you for updating your payment method.</p>
   446→                                </body></html>
   447→                                """,
   448→                            )
   449→                            logger.info(
   450→                                "payment_resume_email_sent company_id=%s", company_id
   451→                            )
   452→                    except Exception as email_err:
   453→                        logger.error(
   454→                            "payment_resume_email_failed company_id=%s error=%s",
   455→                            company_id,
   456→                            str(email_err),
   457→                        )
   458→
   459→                except Exception as e:
   460→                    logger.error(
   461→                        "payment_resume_side_effects_failed company_id=%s error=%s",
   462→                        company_id,
   463→                        str(e),
   464→                    )
   465→
   466→                return {
   467→                    "status": "resumed",
   468→                    "failure_id": failure.id,
   469→                    "company_id": str(company_id),
   470→                    "old_status": old_status,
   471→                    "new_status": self.ACTIVE_STATUS,
   472→                    "service_resumed_at": now.isoformat(),
   473→                    "actions_taken": [
   474→                        "marked_failure_resolved",
   475→                        "updated_subscription_status",
   476→                        "resumed_agents",
   477→                        "thawed_tickets",
   478→                        "sent_service_resumed_email",
   479→                        "emitted_socket_event",
   480→                    ],
   481→                }
   482→
   483→        result = await asyncio.to_thread(_db_work)
   484→
   485→        # 4. Emit Socket.io event for real-time UI update (async — cannot run inside _db_work)
   486→        if result.get("status") == "resumed":
   487→            try:
   488→                from app.core.event_emitter import emit_event
   489→                await emit_event(
   490→                    company_id=result["company_id"],
   491→                    event_type="billing:service_resumed",
   492→                    payload={
   493→                        "company_id": result["company_id"],
   494→                        "old_status": result["old_status"],
   495→                        "new_status": result["new_status"],
   496→                        "service_resumed_at": result["service_resumed_at"],
   497→                    },
   498→                )
   499→                logger.info(
   500→                    "payment_resume_socket_event company_id=%s", result["company_id"]
   501→                )
   502→            except Exception as socket_err:
   503→                logger.error(
   504→                    "payment_resume_socket_failed company_id=%s error=%s",
   505→                    result["company_id"],
   506→                    str(socket_err),
   507→                )
   508→
   509→        return result

   639→        Schedule auto-block for a company if payment isn't received.
   640→        
   641→        This is used when you want to give the customer X hours to update
   642→        their payment method before permanently blocking access.
   643→        
   644→        Args:
   645→            company_id: Company UUID
   646→            block_after_hours: Hours to wait before auto-blocking (default 24)
   647→            
   648→        Returns:
   649→            Dict with scheduling details
   650→        """
   651→        
   652→        def _db_work():
   653→            with SessionLocal() as db:
   654→                failure = db.query(PaymentFailure).filter(
   655→                    PaymentFailure.company_id == str(company_id),
   656→                    PaymentFailure.resolved == False,
   657→                ).first()
   658→                
   659→                if not failure:
   660→                    return {
   661→                        "status": "no_failure",
   662→                        "message": "No active payment failure found for this company"
   663→                    }
   664→                
   665→                now = datetime.now(timezone.utc)
   666→                block_at = now + __import__('datetime').timedelta(hours=block_after_hours)
   667→                
   668→                # Store auto-block schedule in extended_metadata (JSON)
   669→                # This avoids schema changes - follows CLAUDE.md Rule 3
   670→                import json as _json
   671→                
   672→                metadata = {}
   673→                if hasattr(failure, 'extended_metadata') and failure.extended_metadata:
   674→                    try:
   675→                        if isinstance(failure.extended_metadata, dict):
   676→                            metadata = failure.extended_metadata
   677→                        else:
   678→                            metadata = _json.loads(failure.extended_metadata)
   679→                    except (TypeError, ValueError):
   680→                        metadata = {}
   681→                
   682→                metadata['auto_block_scheduled'] = True
   683→                metadata['auto_block_scheduled_at'] = now.isoformat()
   684→                metadata['auto_block_due_at'] = block_at.isoformat()
   685→                metadata['block_after_hours'] = block_after_hours
   686→                
   687→                # Try to save to extended_metadata column if it exists
   688→                if hasattr(failure, 'extended_metadata'):
   689→                    failure.extended_metadata = metadata
   690→                    db.commit()
   691→                    
   692→                    logger.info(
   693→                        "auto_block_scheduled company_id=%s due_at=%s hours=%d",
   694→                        str(company_id)[:8],
   695→                        block_at.isoformat(),
   696→                        block_after_hours,
   697→                    )
   698→                    
   699→                    return {
   700→                        "status": "scheduled",
   701→                        "failure_id": failure.id,
   702→                        "company_id": str(company_id),
   703→                        "auto_block_due_at": block_at.isoformat(),
   704→                        "block_after_hours": block_after_hours,
   705→                        "scheduled_at": now.isoformat(),
   706→                    }
   707→                else:
   708→                    # Column doesn't exist yet - return schedule info
   709→                    # The cleanup loop will handle blocking based on created_at + block_after_hours
   710→                    logger.info(
   711→                        "auto_block_scheduled_no_column company_id=%s due_at=%s",
   712→                        str(company_id)[:8],
   713→                        block_at.isoformat(),
   714→                    )
   715→                    
   716→                    return {
   717→                        "status": "scheduled_fallback",
   718→                        "company_id": str(company_id),
   719→                        "auto_block_due_at": block_at.isoformat(),
   720→                        "block_after_hours": block_after_hours,
   721→                        "note": "extended_metadata column not yet available - using fallback logic"
   722→                    }
   723→
   724→        return await asyncio.to_thread(_db_work)

   726→    async def check_and_execute_auto_blocks(self) -> Dict[str, Any]:
   727→        """
   728→        Check for accounts that should be auto-blocked and execute blocking.
   729→        
   730→        Called by periodic task in main.py (every hour).
   731→        Finds payment failures where:
   732→        - resolved = False (still active)
   733→        - Created more than block_after_hours ago
   734→        
   735→        Returns:
   736→            Dict with count of accounts checked/blocked
   737→        """
   738→        
   739→        def _db_work():
   740→            from datetime import timedelta
   741→            
   742→            with SessionLocal() as db:
   743→                now = datetime.now(timezone.utc)
   744→                blocked_count = 0
   745→                checked_count = 0
   746→                
   747→                # Find all unresolved payment failures
   748→                failures = db.query(PaymentFailure).filter(
   749→                    PaymentFailure.resolved == False,
   750→                ).all()
   751→                
   752→                for failure in failures:
   753→                    checked_count += 1
   754→                    
   755→                    # Default: block after 24 hours
   756→                    block_after_hours = 24
   757→                    
   758→                    # Check if custom schedule was set in extended_metadata
   759→                    if hasattr(failure, 'extended_metadata') and failure.extended_metadata:
   760→                        try:
   761→                            import json as _json
   762→                            meta = failure.extended_metadata
   763→                            if isinstance(meta, str):
   764→                                meta = _json.loads(meta)
   765→                            
   766→                            if meta.get('auto_block_scheduled'):
   767→                                block_after_hours = meta.get('block_after_hours', 24)
   768→                        except (TypeError, ValueError, AttributeError):
   769→                            pass
   770→                    
   771→                    # Calculate when this should have been blocked
   772→                    created_at = failure.created_at or failure.service_stopped_at or now
   773→                    due_time = created_at + timedelta(hours=block_after_hours)
   774→                    
   775→                    # If past due time, ensure account is blocked
   776→                    if now >= due_time:
   777→                        company = db.query(Company).filter(
   778→                            Company.id == failure.company_id
   779→                        ).first()
   780→                        
   781→                        if company and company.subscription_status != 'payment_failed':
   782→                            # Ensure status is payment_failed (should already be set)
   783→                            company.subscription_status = self.PAYMENT_FAILED_STATUS
   784→                            
   785→                            # Update failure record metadata
   786→                            if hasattr(failure, 'extended_metadata'):
   787→                                import json as _json
   788→                                try:
   789→                                    meta = failure.extended_metadata
   790→                                    if isinstance(meta, str):
   791→                                        meta = _json.loads(meta) if meta else {}
   792→                                    elif not meta:
   793→                                        meta = {}
   794→                                    meta['auto_blocked_at'] = now.isoformat()
   795→                                    meta['auto_blocked_reason'] = f'Auto-blocked after {block_after_hours}h'
   796→                                    failure.extended_metadata = meta
   797→                                except Exception:
   798→                                    pass
   799→                            
   800→                            blocked_count += 1
   801→                            logger.info(
   802→                                "auto_block_executed company_id=%s failure_id=%s hours=%d",
   803→                                str(failure.company_id)[:8],
   804→                                str(failure.id)[:8],
   805→                                block_after_hours,
   806→                            )
   807→                
   808→                if blocked_count > 0:
   809→                    db.commit()
   810→                
   811→                logger.info(
   812→                    "auto_block_check completed checked=%d blocked=%d",
   813→                    checked_count,
   814→                    blocked_count,
   815→                )
   816→                
   817→                return {
   818→                    "status": "completed",
   819→                    "checked": checked_count,
   820→                    "blocked": blocked_count,
   821→                    "ran_at": now.isoformat(),
   822→                }
   823→
   824→        return await asyncio.to_thread(_db_work)

   826→    @staticmethod
   827→    def cleanup_old_failures(retention_days: int = 90) -> int:
   828→        """
   829→        Clean up old RESOLVED payment failures to prevent table bloat.
   830→        
   831→        Deletes resolved (paid/fixed) payment failure records older than
   832→        retention_days. Keeps unresolved records forever until resolved.
   833→        
   834→        Args:
   835→            retention_days: Days to keep resolved records (default 90)
   836→            
   837→        Returns:
   838→            Number of records deleted
   839→        """
   840→        from datetime import timedelta
   841→        
   842→        with SessionLocal() as db:
   843→            cutoff = datetime.now(timezone.utc) - timedelta(days=retention_days)
   844→            
   845→            result = db.query(PaymentFailure).filter(
   846→                PaymentFailure.resolved == True,
   847→                PaymentFailure.service_resumed_at < cutoff,
   848→            ).delete()
   849→            
   850→            db.commit()
   851→            
   852→            logger.info(
   853→                "payment_failures_cleanup deleted=%d older_than=%d_days",
   854→                result,
   855→                retention_days,
   856→            )
   857→            
   858→            return result

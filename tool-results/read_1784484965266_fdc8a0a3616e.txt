    74→    74→        self,
    75→    75→        company_id: UUID,
    76→    76→        paddle_transaction_id: str,
    77→    77→        failure_code: str,
    78→    78→        failure_reason: str,
    79→    79→        amount_attempted: Decimal,
    80→    80→        paddle_subscription_id: Optional[str] = None,
    81→    81→        currency: str = "USD",
    82→    82→    ) -> Dict[str, Any]:
    83→    83→        """
    84→    84→        Handle a payment failure with immediate service stop.
    85→    85→
    86→    86→        Steps:
    87→    87→        1. Log to payment_failures table
    88→    88→        2. Set company.subscription_status = 'payment_failed'
    89→    89→        3. Stop all AI agents immediately
    90→    90→        4. Block new ticket creation
    91→    91→        5. Freeze existing open tickets (status → 'frozen')
    92→    92→        6. Send payment_failed email notification
    93→    93→        7. Create Socket.io event for real-time UI update
    94→    94→
    95→    95→        Args:
    96→    96→            company_id: Company UUID
    97→    97→            paddle_transaction_id: Paddle transaction ID
    98→    98→            failure_code: Paddle failure code
    99→    99→            failure_reason: Human-readable failure reason
   100→   100→            amount_attempted: Amount that failed to charge
   101→   101→            paddle_subscription_id: Optional Paddle subscription ID
   102→   102→            currency: Currency code (default USD)
   103→   103→
   104→   104→        Returns:
   105→   105→            Dict with failure record details and actions taken
   106→   106→
   107→   107→        Raises:
   108→   108→            PaymentFailureError: If failure cannot be processed
   109→   109→        """
   110→   110→
   111→   111→        def _db_work():
   112→   112→            with SessionLocal() as db:
   113→   113→                # Get company with row lock to prevent race conditions
   114→   114→                company = db.query(Company).filter(
   115→   115→                    Company.id == str(company_id)
   116→   116→                ).with_for_update().first()
   117→   117→
   118→   118→                if not company:
   119→   119→                    raise PaymentFailureError(f"Company {company_id} not found")
   120→   120→
   121→   121→                now = datetime.now(timezone.utc)
   122→   122→
   123→   123→                # Check if there's already an unresolved payment failure
   124→   124→                existing_failure = db.query(PaymentFailure).filter(
   125→   125→                    PaymentFailure.company_id == str(company_id),
   126→   126→                    PaymentFailure.resolved == False,
   127→   127→                ).first()
   128→   128→
   129→   129→                if existing_failure:
   130→   130→                    # Already has active payment failure - just log this one
   131→   131→                    logger.warning(
   132→   132→                        "payment_failure_already_stopped "
   133→   133→                        "company_id=%s existing_failure_id=%s "
   134→   134→                        "new_transaction_id=%s",
   135→   135→                        company_id,
   136→   136→                        existing_failure.id,
   137→   137→                        paddle_transaction_id,
   138→   138→                    )
   139→   139→                    return {
   140→   140→                        "status": "already_stopped",
   141→   141→                        "existing_failure_id": existing_failure.id,
   142→   142→                        "message": "Service already stopped due to previous payment failure",
   143→   143→                    }
   144→   144→
   145→   145→                # Create payment failure record
   146→   146→                failure = PaymentFailure(
   147→   147→                    company_id=str(company_id),
   148→   148→                    paddle_subscription_id=paddle_subscription_id,
   149→   149→                    paddle_transaction_id=paddle_transaction_id,
   150→   150→                    failure_code=failure_code,
   151→   151→                    failure_reason=failure_reason,
   152→   152→                    amount_attempted=amount_attempted,
   153→   153→                    currency=currency,
   154→   154→                    service_stopped_at=now,
   155→   155→                    notification_sent=False,
   156→   156→                    resolved=False,
   157→   157→                )
   158→   158→                db.add(failure)
   159→   159→
   160→   160→                # Update company subscription status
   161→   161→                old_status = company.subscription_status
   162→   162→                company.subscription_status = self.PAYMENT_FAILED_STATUS
   163→   163→
   164→   164→                # Update subscription record if exists
   165→   165→                subscription = db.query(Subscription).filter(
   166→   166→                    Subscription.company_id == str(company_id),
   167→   167→                    Subscription.status == self.ACTIVE_STATUS,
   168→   168→                ).first()
   169→   169→
   170→   170→                if subscription:
   171→   171→                    subscription.status = self.PAYMENT_FAILED_STATUS
   172→   172→
   173→   173→                db.commit()
   174→   174→                db.refresh(failure)
   175→   175→
   176→   176→                logger.info(
   177→   177→                    "payment_failure_handled "
   178→   178→                    "company_id=%s failure_id=%s "
   179→   179→                    "old_status=%s new_status=%s "
   180→   180→                    "amount=%s code=%s",
   181→   181→                    company_id,
   182→   182→                    failure.id,
   183→   183→                    old_status,
   184→   184→                    self.PAYMENT_FAILED_STATUS,
   185→   185→                    amount_attempted,
   186→   186→                    failure_code,
   187→   187→                )
   188→   188→
   189→   189→                # Trigger side-effects for payment failure
   190→   190→                try:
   191→   191→                    # Send payment failure notification email
   192→   192→                    from app.services.email_service import send_email
   193→   193→                    from database.models.core import User
   194→   194→                    company_owner = db.query(User).filter(
   195→   195→                        User.company_id == str(company_id),
   196→   196→                        User.role == "owner",
   197→   197→                    ).first()
   198→   198→                    if company_owner:
   199→   199→                        try:
   200→   200→                            send_email(
   201→   201→                                to=company_owner.email,
   202→   202→                                subject="PARWA: Payment Failed — Action Required",
   203→   203→                                html_content=f"""
   204→   204→                                <html><body>
   205→   205→                                <h2>Payment Failed</h2>
   206→   206→                                <p>Hello {company_owner.full_name or 'there'},</p>
   207→   207→                                <p>We were unable to process your recent payment for your PARWA subscription.</p>
   208→   208→                                <p><strong>Company:</strong> {company.name}</p>
   209→   209→                                <p><strong>Reason:</strong> {failure_reason or 'Payment declined by payment provider'}</p>
   210→   210→                                <p>Your service has been paused. Update your payment method to restore access immediately.</p>
   211→   211→                                </body></html>
   212→   212→                                """,
   213→   213→                            )
   214→   214→                            # Mark notification as sent after successful email delivery
   215→   215→                            failure.notification_sent = True
   216→   216→                            db.commit()
   217→   217→                        except Exception as email_err:
   218→   218→                            logger.error(
   219→   219→                                "payment_failure_email_failed",
   220→   220→                                company_id=str(company_id),
   221→   221→                                error=str(email_err),
   222→   222→                            )
   223→   223→
   224→   224→                    # Notify via notification service
   225→   225→                    try:
   226→   226→                        from app.services.notification_service import NotificationService
   227→   227→                        notif_svc = NotificationService(db)
   228→   228→                        notif_svc.create_notification(
   229→   229→                            user_id=str(company_owner.id) if company_owner else None,
   230→   230→                            company_id=str(company_id),
   231→   231→                            event_type="payment_failed",
   232→   232→                            title="Payment Failed",
   233→   233→                            message=f"Payment processing failed for {company.name}. Please update your billing information.",
   234→   234→                            priority="high",
   235→   235→                        )
   236→   236→                    except Exception as notif_err:
   237→   237→                        logger.error(
   238→   238→                            "payment_failure_notification_failed",
   239→   239→                            company_id=str(company_id),
   240→   240→                            error=str(notif_err),
   241→   241→                        )
   242→   242→
   243→   243→                except Exception as e:
   244→   244→                    logger.error(
   245→   245→                        "payment_failure_side_effects_failed",
   246→   246→                        company_id=str(company_id),
   247→   247→                        error=str(e),
   248→   248→                    )
   249→   249→
   250→   250→                return {
   251→   251→                    "status": "stopped",
   252→   252→                    "failure_id": failure.id,
   253→   253→                    "company_id": str(company_id),
   254→   254→                    "old_status": old_status,
   255→   255→                    "new_status": self.PAYMENT_FAILED_STATUS,
   256→   256→                    "service_stopped_at": now.isoformat(),
   257→   257→                    "actions_taken": [
   258→   258→                        "logged_failure",
   259→   259→                        "updated_subscription_status",
   260→   260→                        "service_stopped",
   261→   261→                    ],
   262→   262→                }
   263→   263→
   264→   264→        return await asyncio.to_thread(_db_work)

   300→   300→    async def resume_service(
   301→   301→        self,
   302→   302→        company_id: UUID,

   304→   304→    ) -> Dict[str, Any]:
   305→   305→        """
   306→   306→        Resume service after successful payment.
   307→   307→
   308→   308→        Called when payment succeeds after failure:
   309→   309→        1. Update subscription_status → 'active'
   310→   310→        2. Resume AI agents
   311→   311→        3. Unfreeze tickets
   312→   312→        4. Send service_resumed notification
   313→   313→
   314→   314→        Args:
   315→   315→            company_id: Company UUID
   316→   316→            paddle_transaction_id: Successful Paddle transaction ID
   317→   317→

   319→   319→            Dict with resume details
   320→   320→

   322→   322→            PaymentFailureError: If no active failure to resume from
   323→   323→        """
   324→   324→
   325→   325→        def _db_work():
   326→   326→            with SessionLocal() as db:
   327→   327→                # Get company with row lock
   328→   328→                company = db.query(Company).filter(
   329→   329→                    Company.id == str(company_id)
   330→   330→                ).with_for_update().first()
   331→   331→
   332→   332→                if not company:
   333→   333→                    raise PaymentFailureError(f"Company {company_id} not found")
   334→   334→
   335→   335→                # Get active payment failure
   336→   336→                failure = db.query(PaymentFailure).filter(
   337→   337→                    PaymentFailure.company_id == str(company_id),
   338→   338→                    PaymentFailure.resolved == False,
   339→   339→                ).first()
   340→   340→
   341→   341→                if not failure:
   342→   342→                    logger.info(
   343→   343→                        "payment_resume_no_failure company_id=%s", company_id
   344→   344→                    )

   346→   346→                    if company.subscription_status != self.ACTIVE_STATUS:
   347→   347→                        company.subscription_status = self.ACTIVE_STATUS
   348→   348→                        db.commit()

   350→   350→                        "status": "no_failure",
   351→   351→                        "message": "No active payment failure to resume from",
   352→   352→                    }
   353→   353→
   354→   354→                now = datetime.now(timezone.utc)
   355→   355→

   369→   369→                if subscription:

   380→   380→                    old_status,

   633→   633→    async def schedule_auto_block(
   634→   634→        self,
   635→   635→        company_id: UUID,
   636→   636→        block_after_hours: int = 24,
   637→   637→    ) -> Dict[str, Any]:
   638→   638→        """
   639→   639→        Schedule auto-block for a company if payment isn't received.
   640→   640→        
   641→   641→        This is used when you want to give the customer X hours to update
   642→   642→        their payment method before permanently blocking access.
   643→   643→        
   644→   644→        Args:
   645→   645→            company_id: Company UUID
   646→   646→            block_after_hours: Hours to wait before auto-blocking (default 24)
   647→   647→            
   648→   648→        Returns:
   649→   649→            Dict with scheduling details
   650→   650→        """
   651→   651→        
   652→   652→        def _db_work():
   653→   653→            with SessionLocal() as db:
   654→   654→                failure = db.query(PaymentFailure).filter(
   655→   655→                    PaymentFailure.company_id == str(company_id),
   656→   656→                    PaymentFailure.resolved == False,
   657→   657→                ).first()
   658→   658→                
   659→   659→                if not failure:
   660→   660→                    return {
   661→   661→                        "status": "no_failure",
   662→   662→                        "message": "No active payment failure found for this company"
   663→   663→                    }
   664→   664→                
   665→   665→                now = datetime.now(timezone.utc)
   666→   666→                block_at = now + __import__('datetime').timedelta(hours=block_after_hours)
   667→   667→                
   668→   668→                # Store auto-block schedule in extended_metadata (JSON)
   669→   669→                # This avoids schema changes - follows CLAUDE.md Rule 3
   670→   670→                import json as _json
   671→   671→                
   672→   672→                metadata = {}
   673→   673→                if hasattr(failure, 'extended_metadata') and failure.extended_metadata:
   674→   674→                    try:
   675→   675→                        if isinstance(failure.extended_metadata, dict):
   676→   676→                            metadata = failure.extended_metadata
   677→   677→                        else:
   678→   678→                            metadata = _json.loads(failure.extended_metadata)
   679→   679→                    except (TypeError, ValueError):
   680→   680→                        metadata = {}
   681→   681→                
   682→   682→                metadata['auto_block_scheduled'] = True
   683→   683→                metadata['auto_block_scheduled_at'] = now.isoformat()
   684→   684→                metadata['auto_block_due_at'] = block_at.isoformat()
   685→   685→                metadata['block_after_hours'] = block_after_hours
   686→   686→                
   687→   687→                # Try to save to extended_metadata column if it exists
   688→   688→                if hasattr(failure, 'extended_metadata'):
   689→   689→                    failure.extended_metadata = metadata
   690→   690→                    db.commit()
   691→   691→                    
   692→   692→                    logger.info(
   693→   693→                        "auto_block_scheduled company_id=%s due_at=%s hours=%d",
   694→   694→                        str(company_id)[:8],
   695→   695→                        block_at.isoformat(),
   696→   696→                        block_after_hours,
   697→   697→                    )
   698→   698→                    
   699→   699→                    return {
   700→   700→                        "status": "scheduled",
   701→   701→                        "failure_id": failure.id,
   702→   702→                        "company_id": str(company_id),
   703→   703→                        "auto_block_due_at": block_at.isoformat(),
   704→   704→                        "block_after_hours": block_after_hours,
   705→   705→                        "scheduled_at": now.isoformat(),
   706→   706→                    }
   707→   707→                else:
   708→   708→                    # Column doesn't exist yet - return schedule info
   709→   709→                    # The cleanup loop will handle blocking based on created_at + block_after_hours
   710→   710→                    logger.info(
   711→   711→                        "auto_block_scheduled_no_column company_id=%s due_at=%s",
   712→   712→                        str(company_id)[:8],
   713→   713→                        block_at.isoformat(),
   714→   714→                    )
   715→   715→                    
   716→   716→                    return {
   717→   717→                        "status": "scheduled_fallback",
   718→   718→                        "company_id": str(company_id),
   719→   719→                        "auto_block_due_at": block_at.isoformat(),
   720→   720→                        "block_after_hours": block_after_hours,
   721→   721→                        "note": "extended_metadata column not yet available - using fallback logic"
   722→   722→                    }
   723→   723→
   724→   724→        return await asyncio.to_thread(_db_work)

   726→   726→    async def check_and_execute_auto_blocks(self) -> Dict[str, Any]:
   727→   727→        """
   728→   728→        Check for accounts that should be auto-blocked and execute blocking.
   729→   729→        
   730→   730→        Called by periodic task in main.py (every hour).
   731→   731→        Finds payment failures where:
   732→   732→        - resolved = False (still active)
   733→   733→        - Created more than block_after_hours ago
   734→   734→        
   735→   735→        Returns:
   736→   736→            Dict with count of accounts checked/blocked
   737→   737→        """
   738→   738→        
   739→   739→        def _db_work():
   740→   740→            from datetime import timedelta
   741→   741→            
   742→   742→            with SessionLocal() as db:
   743→   743→                now = datetime.now(timezone.utc)
   744→   744→                blocked_count = 0
   745→   745→                checked_count = 0
   746→   746→                
   747→   747→                # Find all unresolved payment failures
   748→   748→                failures = db.query(PaymentFailure).filter(
   749→   749→                    PaymentFailure.resolved == False,
   750→   750→                ).all()
   751→   751→                
   752→   752→                for failure in failures:
   753→   753→                    checked_count += 1
   754→   754→                    
   755→   755→                    # Default: block after 24 hours
   756→   756→                    block_after_hours = 24
   757→   757→                    
   758→   758→                    # Check if custom schedule was set in extended_metadata
   759→   759→                    if hasattr(failure, 'extended_metadata') and failure.extended_metadata:
   760→   760→                        try:
   761→   761→                            import json as _json
   762→   762→                            meta = failure.extended_metadata
   763→   763→                            if isinstance(meta, str):
   764→   764→                                meta = _json.loads(meta)
   765→   765→                            
   766→   766→                            if meta.get('auto_block_scheduled'):
   767→   767→                                block_after_hours = meta.get('block_after_hours', 24)
   768→   768→                        except (TypeError, ValueError, AttributeError):
   769→   769→                            pass
   770→   770→                    
   771→   771→                    # Calculate when this should have been blocked
   772→   772→                    created_at = failure.created_at or failure.service_stopped_at or now
   773→   773→                    due_time = created_at + timedelta(hours=block_after_hours)
   774→   774→                    
   775→   775→                    # If past due time, ensure account is blocked
   776→   776→                    if now >= due_time:
   777→   777→                        company = db.query(Company).filter(
   778→   778→                            Company.id == failure.company_id
   779→   779→                        ).first()
   780→   780→                        
   781→   781→                        if company and company.subscription_status != 'payment_failed':
   782→   782→                            # Ensure status is payment_failed (should already be set)
   783→   783→                            company.subscription_status = self.PAYMENT_FAILED_STATUS
   784→   784→                            
   785→   785→                            # Update failure record metadata
   786→   786→                            if hasattr(failure, 'extended_metadata'):
   787→   787→                                import json as _json
   788→   788→                                try:
   789→   789→                                    meta = failure.extended_metadata
   790→   790→                                    if isinstance(meta, str):
   791→   791→                                        meta = _json.loads(meta) if meta else {}
   792→   792→                                    elif not meta:
   793→   793→                                        meta = {}
   794→   794→                                    meta['auto_blocked_at'] = now.isoformat()
   795→   795→                                    meta['auto_blocked_reason'] = f'Auto-blocked after {block_after_hours}h'
   796→   796→                                    failure.extended_metadata = meta
   797→   797→                                except Exception:
   798→   798→                                    pass
   799→   799→                            
   800→   800→                            blocked_count += 1
   801→   801→                            logger.info(
   802→   802→                                "auto_block_executed company_id=%s failure_id=%s hours=%d",
   803→   803→                                str(failure.company_id)[:8],
   804→   804→                                str(failure.id)[:8],
   805→   805→                                block_after_hours,
   806→   806→                            )
   807→   807→                
   808→   808→                if blocked_count > 0:
   809→   809→                    db.commit()
   810→   810→                
   811→   811→                logger.info(
   812→   812→                    "auto_block_check completed checked=%d blocked=%d",
   813→   813→                    checked_count,
   814→   814→                    blocked_count,
   815→   815→                )
   816→   816→                
   817→   817→                return {
   818→   818→                    "status": "completed",
   819→   819→                    "checked": checked_count,
   820→   820→                    "blocked": blocked_count,
   821→   821→                    "ran_at": now.isoformat(),
   822→   822→                }
   823→   823→
   824→   824→        return await asyncio.to_thread(_db_work)

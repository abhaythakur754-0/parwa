    35→/**
    36→ * Try to proxy a request to the backend FastAPI server.
    37→ * Returns the Response on success, or null if backend is unavailable / returned an error.
    38→ */
    39→async function proxyToBackend(request: NextRequest, pathSegments: string[]): Promise<Response | null> {
    40→  if (!BACKEND_URL) return null;
    41→
    42→  const backendPath = `${BACKEND_URL}/api/jarvis/${pathSegments.join('/')}`;
    43→  const url = new URL(request.url);
    44→  const searchParams = url.searchParams.toString();
    45→  const fullUrl = searchParams ? `${backendPath}?${searchParams}` : backendPath;
    46→
    47→  // Determine proxy origin (must match an allowed origin on the backend CSRF middleware)
    48→  // FRONTEND_URL takes priority (set to https://parwa.buzz in production).
    49→  // VERCEL_URL is only used for preview deployments — in production it's
    50→  // a random deployment URL that the backend would reject via CORS.
    51→  const proxyOrigin =
    52→    process.env.FRONTEND_URL ||
    53→    (process.env.NODE_ENV === 'production' ? 'https://parwa.buzz' : null) ||
    54→    (process.env.VERCEL_URL ? `https://${process.env.VERCEL_URL}` : null) ||
    55→    'http://localhost:3000';
    56→
    57→  try {
    58→    const body = ['POST', 'PATCH', 'PUT'].includes(request.method)
    59→      ? await request.arrayBuffer()
    60→      : undefined;
    61→
    62→    const headers: Record<string, string> = {
    63→      'Content-Type': request.headers.get('Content-Type') || 'application/json',
    64→      'Origin': proxyOrigin,
    65→      'Referer': `${proxyOrigin}/`,
    66→    };
    67→
    68→    // Forward auth token from parwa_at cookie as Bearer token
    69→    const authHeader = request.headers.get('authorization');
    70→    if (authHeader) {
    71→      headers['Authorization'] = authHeader;
    72→    } else {
    73→      const cookieToken = request.cookies.get('parwa_at')?.value;
    74→      if (cookieToken) {
    75→        headers['Authorization'] = `Bearer ${cookieToken}`;
    76→      }
    77→    }
    78→
    79→    const response = await fetch(fullUrl, {
    80→      method: request.method,
    81→      headers,
    82→      body,
    83→      signal: AbortSignal.timeout(90000),  // 90s — AI pipeline can take 30-60s
    84→    });
    85→
    86→    if (response.status >= 200 && response.status < 300) {
    87→      // Don't return the raw response — Vercel's edge adds content-encoding
    88→      // on top of any backend compression, causing "Failed to fetch" in the
    89→      // browser. Parse the body and return a clean Response instead.
    90→      const text = await response.text();
    91→      return new Response(text, {
    92→        status: response.status,
    93→        headers: {
    94→          'Content-Type': response.headers.get('Content-Type') || 'application/json',
    95→        },
    96→      });
    97→    }
    98→
    99→    // For /cc/* paths (handled by my proxy-first check), return the actual
   100→    // error response instead of falling back to local handlers. The backend
   101→    // 401/403/500 should be visible to the client.
   102→    if (pathSegments[0] === 'cc') {
   103→      const text = await response.text();
   104→      try {
   105→        const data = JSON.parse(text);
   106→        return new Response(JSON.stringify(data), {
   107→          status: response.status,
   108→          headers: { 'Content-Type': 'application/json' },
   109→        });
   110→      } catch {
   111→        return new Response(text, {
   112→          status: response.status,
   113→          headers: { 'Content-Type': 'application/json' },
   114→        });
   115→      }
   116→    }
   117→
   118→    // Backend returned an error — fall back to local handling
   119→    return null;
   120→  } catch (err) {
   121→    // Backend unreachable — fall back to local handling
   122→    console.warn('[Jarvis] Backend proxy failed:', (err instanceof Error ? err.message : String(err))?.slice(0, 150));
   123→    return null;
   124→  }
   125→}

   129→let ZAI: any = null;

   131→async function getZAI() {
   132→  if (!ZAI) {
   133→    try {
   134→      const mod = await import('z-ai-web-dev-sdk');
   135→      const ZAIClass = (mod as any).default;
   136→      if (ZAIClass && typeof ZAIClass.create === 'function') {
   137→        ZAI = await ZAIClass.create();
   138→      }
   139→    } catch (err) {
   140→      console.warn('[Jarvis] z-ai-web-dev-sdk not available:', (err instanceof Error ? err.message : String(err))?.slice(0, 100));
   141→    }
   142→  }
   143→  return ZAI;
   144→}

   146→async function callZAISDK(messages: Array<{role: string, content: string}>): Promise<string | null> {
   147→  try {
   148→    const zai = await getZAI();
   149→    if (!zai || !zai.chat || !zai.chat.completions) return null;
   150→
   151→    const completion = await zai.chat.completions.create({
   152→      messages: messages.map(m => ({
   153→        role: m.role === 'assistant' ? 'assistant' : m.role,
   154→        content: m.content,
   155→      })),
   156→      temperature: 0.8,
   157→      max_tokens: 800,
   158→    });
   159→
   160→    const text = completion?.choices?.[0]?.message?.content;
   161→    if (text && text.trim().length > 10) return text.trim();
   162→    return null;
   163→  } catch (err) {
   164→    console.warn('[Jarvis] z-ai-web-dev-sdk failed:', (err instanceof Error ? err.message : String(err))?.slice(0, 150));
   165→    return null;
   166→  }
   167→}

   177→function getGoogleProvider(): any {
   178→  return {
   179→    name: 'google',
   180→    apiKey: GOOGLE_AI_KEY,
   181→    model: 'gemini-2.0-flash',
   182→    apiUrl: `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${GOOGLE_AI_KEY}`,
   183→    buildHeaders: () => ({ 'Content-Type': 'application/json' }),
   184→    buildBody: (messages: any[]) => {
   185→      const systemMsg = messages.find(m => m.role === 'system');
   186→      const chatMsgs = messages.filter(m => m.role !== 'system');
   187→      const contents = chatMsgs.map(m => ({
   188→        role: m.role === 'assistant' ? 'model' : 'user',
   189→        parts: [{ text: m.content }],
   190→      }));
   191→      return JSON.stringify({
   192→        systemInstruction: systemMsg ? { parts: [{ text: systemMsg.content }] } : undefined,
   193→        contents,
   194→        generationConfig: { temperature: 0.7, maxOutputTokens: 800 },
   195→      });
   196→    },
   197→    parseResponse: (data: any) => {
   198→      return data?.candidates?.[0]?.content?.parts?.[0]?.text || null;
   199→    },

   247→function getProvider(name: string): any | null {
   248→  switch (name) {
   249→    case 'google': return GOOGLE_AI_KEY ? getGoogleProvider() : null;
   250→    case 'cerebras': return CEREBRAS_KEY ? getCerebrasProvider() : null;
   251→    case 'groq': return GROQ_KEY ? getGroqProvider() : null;
   252→    default: return null;
   253→  }

   256→async function callProvider(provider: any, messages: Array<{role: string, content: string}>): Promise<string | null> {
   257→  if (!provider.apiKey) return null;
   258→
   259→  const response = await fetch(provider.apiUrl, {
   260→    method: 'POST',
   261→    headers: provider.buildHeaders(provider.apiKey),
   262→    body: provider.buildBody(messages, provider.model),
   263→    signal: AbortSignal.timeout(15000),
   264→  });
   265→
   266→  if (!response.ok) {
   267→    const errorText = await response.text().catch(() => 'Unknown error');
   268→    throw new Error(`HTTP ${response.status}: ${errorText.slice(0, 200)}`);
   269→  }
   270→
   271→  const data = await response.json();
   272→  const text = provider.parseResponse(data);
   273→
   274→  if (!text || text.trim().length < 10) {
   275→    throw new Error('Empty or too-short response from provider');
   276→  }
   277→
   278→  return text.trim();
   279→}

   283→async function callAI(messages: Array<{role: string, content: string}>): Promise<string | null> {
   284→  // 1. Try z-ai-web-dev-sdk FIRST (most reliable in production)
   285→  try {
   286→    const result = await callZAISDK(messages);
   287→    if (result) return result;
   288→  } catch (error) {
   289→    console.warn('[Jarvis] z-ai-web-dev-sdk error:', (error instanceof Error ? error.message : String(error))?.slice(0, 100));
   290→  }
   291→
   292→  // 2. Try free providers in order: Google → Cerebras → Groq
   293→  const providerList = ['google', 'cerebras', 'groq'];
   294→  for (const name of providerList) {
   295→    const provider = getProvider(name);
   296→    if (provider) {
   297→      try {
   298→        const result = await callProvider(provider, messages);
   299→        if (result) return result;
   300→      } catch (error) {
   301→        console.warn(`[Jarvis] Provider "${name}" failed:`, (error instanceof Error ? error.message : String(error))?.slice(0, 100));
   302→      }
   303→    }
   304→  }
   305→
   306→  // All providers failed — keyword fallback
   307→  console.warn('[Jarvis] All AI providers failed, using keyword fallback');
   308→  return null;
   309→}

   791→const BULLET_EMOJIS = ['🤖', '💰', '📡', '🛒', '💻', '🚛', '🏥', '🔒', '🚀', '✅', '🎯', '📊', '⚡', '💡', '🔧', '🔗', '📦', '📈', '⭐', '🎯', '🧠', '💬', '📁', '🔑', '🎉', '✨', '🔔', '💪', '🥊', '🛡️'];

   793→function isEmojiChar(ch: string): boolean {
   794→  // Must use codePointAt for emoji (surrogate pairs in JS)
   795→  const code = ch.codePointAt(0) || 0;
   796→  return (code >= 0x1F300 && code <= 0x1FAFF) || (code >= 0x2600 && code <= 0x27BF) || (code >= 0xFE00 && code <= 0xFE0F);
   797→}

   799→function forceBulletFormat(text: string): string {
   800→  const lines = text.split('\n');
   801→  if (lines.length === 0) return text;
   802→
   803→  // Check if text already has proper bullet formatting
   804→  const nonEmpty = lines.filter(l => l.trim());
   805→  if (nonEmpty.length === 0) return text;
   806→
   807→  // Count bullet-formatted lines (lines starting with bullet markers or emojis)
   808→  const bulletCount = nonEmpty.filter(l => {
   809→    const t = l.trim();
   810→    return /^[\u2022\-*•]\s/.test(t) || /^[0-9]+[.)]\s/.test(t) || isEmojiChar(t);
   811→  }).length;
   812→
   813→  // If 40%+ lines are already bullets, return as-is
   814→  if (nonEmpty.length > 2 && bulletCount / nonEmpty.length >= 0.4) return text;
   815→
   816→  // If only 1-2 non-empty lines total AND none are long paragraphs, keep as-is
   817→  if (nonEmpty.length <= 2) {
   818→    const hasLongLine = nonEmpty.some(l => l.trim().length > 150);
   819→    if (!hasLongLine) return text;
   820→  }
   821→
   822→  // Convert paragraph text into bullet-point format
   823→  const result: string[] = [];
   824→  let openerUsed = false;
   825→
   826→  for (const line of lines) {
   827→    const trimmed = line.trim();
   828→    if (!trimmed) {
   829→      result.push('');
   830→      continue;
   831→    }
   832→
   833→    // Already a bullet or numbered list item — keep
   834→    if (/^[\u2022\-*•]\s/.test(trimmed) || /^[0-9]+[.)]\s/.test(trimmed)) {
   835→      result.push(trimmed);
   836→      continue;
   837→    }
   838→
   839→    // Starts with emoji — keep
   840→    if (isEmojiChar(trimmed)) {
   841→      result.push(trimmed);
   842→      continue;
   843→    }
   844→
   845→    // Very short line (< 50 chars) — treat as opener or question
   846→    if (trimmed.length < 50) {
   847→      if (!openerUsed) {
   848→        result.push(trimmed);
   849→        openerUsed = true;
   850→      } else {
   851→        // Subsequent short lines get bullet treatment
   852→        const emoji = pickEmoji(trimmed);
   853→        result.push(`${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`);
   854→      }
   855→      continue;
   856→    }
   857→
   858→    // Long line: split into sentences and convert to bullets
   859→    const sentences = trimmed.match(/[^.!?]*[.!?]+/g) || [trimmed];
   860→
   861→    if (sentences.length === 1 && trimmed.length < 80) {
   862→      // Single short sentence — keep as is
   863→      result.push(trimmed);
   864→      continue;
   865→    }
   866→
   867→    if (sentences.length === 1) {
   868→      // Single long sentence — break on commas if possible
   869→      const parts = trimmed.split(/,/g).map(s => s.trim()).filter(Boolean);
   870→      if (parts.length >= 2) {
   871→        // First part as opener
   872→        result.push(parts[0]);
   873→        result.push('');
   874→        for (let i = 1; i < parts.length; i++) {
   875→          const p = parts[i].trim();
   876→          if (!p) continue;
   877→          const emoji = pickEmoji(p);
   878→          result.push(`${emoji} ${p.charAt(0).toUpperCase() + p.slice(1)}`);
   879→        }
   880→      } else {
   881→        const emoji = pickEmoji(trimmed);
   882→        result.push(`${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`);
   883→      }
   884→      continue;
   885→    }
   886→
   887→    // Multiple sentences — first as opener, rest as bullets
   888→    if (!openerUsed) {
   889→      result.push(sentences[0].trim());
   890→      result.push('');
   891→      openerUsed = true;
   892→    }
   893→
   894→    for (let i = 1; i < sentences.length; i++) {
   895→      const s = sentences[i].trim();
   896→      if (!s) continue;
   897→      const emoji = pickEmoji(s);
   898→      result.push(`${emoji} ${s.charAt(0).toUpperCase() + s.slice(1)}`);
   899→    }
   900→  }
   901→
   902→  return result.join('\n');
   903→}

  1116→function detectStage(message: string, session: any): string {
  1117→  const lower = message.toLowerCase();
  1118→  const ctx = session.context;
  1119→  const prevStage = session.detected_stage || ctx.detected_stage || 'welcome';
  1120→
  1121→  // Track stage history for transition detection
  1122→  if (!session.stage_history) session.stage_history = [];
  1123→
  1124→  // Phase 8a: Enhanced stage detection with history and nuanced stages
  1125→
  1126→  // Welcome — only in first messages
  1127→  if (session.message_count_today <= 1 && prevStage === 'welcome') return 'welcome';
  1128→
  1129→  // Verification — OTP/confirm (highest priority for active flows)
  1130→  if (lower.includes('verify') || lower.includes('otp') || lower.includes('confirm email')) return 'verification';
  1131→
  1132→  // Payment — active checkout intent
  1133→  if (lower.includes('pay') || lower.includes('checkout') || lower.includes('subscribe now') || lower.includes('complete purchase')) return 'payment';
  1134→
  1135→  // Bill review — checking invoice/bill details
  1136→  if (lower.includes('bill') || lower.includes('invoice') || lower.includes('receipt') || lower.includes('charge')) return 'bill_review';
  1137→
  1138→  // Handoff — requesting human transfer
  1139→  if (lower.includes('handoff') || lower.includes('transfer') || lower.includes('speak to human') || lower.includes('real person')) return 'handoff';
  1140→
  1141→  // Objection handling — user raising concerns
  1142→  const objectionPatterns = /(?:too (?:expensive|costly|pricey|much)|not sure|concern|worried|hesitat|risk|what if|scam|trust|reliable|safe)/i;
  1143→  if (objectionPatterns.test(lower)) return 'objection_handling';
  1144→
  1145→  // Variant selection — discussing specific features/variants
  1146→  const variantPatterns = /(?:variant|which (?:plan|one)|compare|difference between|mini parwa|parwa high|mini vs|parwa vs)/i;
  1147→  if (variantPatterns.test(lower) && (ctx.selected_variants?.length > 0 || lower.includes('variant') || lower.includes('compare'))) return 'variant_selection';
  1148→
  1149→  // Onboarding questions — asking about business specifics
  1150→  const onboardingPatterns = /(?:how many|team size|employees|tickets? (?:per day|daily|monthly)|channels?|current (?:setup|tool|system)|what (?:crm|helpdesk|platform))/i;
  1151→  if (onboardingPatterns.test(lower) && !ctx.industry) return 'onboarding_questions';
  1152→
  1153→  // Pricing — discussing plans/costs
  1154→  if (lower.includes('price') || lower.includes('pricing') || lower.includes('cost') || lower.includes('plan') || lower.includes('package') || lower.includes('how much')) return 'pricing';
  1155→
  1156→  // Demo — wanting to try/see
  1157→  if (lower.includes('demo') || lower.includes('try') || lower.includes('see it') || lower.includes('show me') || lower.includes('experience')) return 'demo';
  1158→
  1159→  // Discovery — learning about industry
  1160→  if (!ctx.industry && (lower.includes('ecommerce') || lower.includes('e-commerce') || lower.includes('saas') || lower.includes('logistics') || lower.includes('healthcare') || lower.includes('retail') || lower.includes('industry'))) return 'discovery';
  1161→
  1162→  // Default: maintain previous stage unless it was welcome (which we should advance from)
  1163→  if (prevStage === 'welcome') return 'discovery';
  1164→  return prevStage;
  1165→}

  1169→export async function POST(request: NextRequest, { params }: { params: Promise<{ path: string[] }> }) {
  1170→  const { path } = await params;
  1171→  const endpoint = path.join('/');
  1172→

  1175→    // The backend has /api/jarvis/cc/* mounted (jarvis_cc.py). The local
  1176→    // in-memory handlers below were built for the older /api/jarvis/session
  1177→    // style and don't know about /cc/session, /cc/message, etc.
  1178→    if (path[0] === 'cc') {
  1179→      const proxyResult = await proxyToBackend(request, path);

  1182→      return NextResponse.json(
  1183→        { error: { code: 'backend_unavailable', message: 'Jarvis CC backend is unavailable. Please retry.', details: null } },
  1184→        { status: 502 }
  1185→      );
  1186→    }
  1187→
  1188→    // ── POST /session — Create Session ──────────────────────────
  1189→    if (endpoint === 'session') {
  1190→      const body = await request.json();
  1191→      const session = createDefaultSession(body.entry_source, body.entry_params);
  1192→
  1193→      // Phase 8b: Context-aware welcome based on entry_source
  1194→      const welcomeContent = getContextAwareWelcome(session.context.entry_source, session.context);

  1199→        role: 'jarvis',
  1200→        content: welcomeContent,
  1201→        message_type: 'text',

  1203→        timestamp: new Date().toISOString(),
  1204→      };

  1209→
  1210→    // ── POST /message — Send Message & Get AI Reply ────────────
  1211→    if (endpoint === 'message') {
  1212→      // ── Try backend proxy first (LangGraph 13-stage pipeline + RAG + PostgreSQL) ──
  1213→      const proxyResult = await proxyToBackend(request, path);
  1214→      console.log(`[Jarvis] Backend proxy ${proxyResult ? 'succeeded' : 'failed, using local fallback'}`);
  1215→      if (proxyResult) return proxyResult;
  1216→
  1217→      // ── Local fallback: in-memory handling ──
  1218→      const body = await request.json();
  1219→      const { content, session_id, context: incomingContext } = body;
  1220→
  1221→      let session = session_id ? sessions.get(session_id) : undefined;
  1222→      if (!session) {
  1223→        session = createDefaultSession('direct');
  1224→        sessions.set(session.id, session);
  1225→      }

  1227→      // ── Merge incoming context from frontend BEFORE building AI response ──
  1228→      // This is how Jarvis "knows" what the user did on other pages (ROI, models, etc.)
  1229→      if (incomingContext && typeof incomingContext === 'object') {
  1230→        for (const [key, value] of Object.entries(incomingContext)) {
  1231→          if (value !== null && value !== undefined) {
  1232→            session.context[key] = value;
  1233→          }
  1234→        }
  1235→        session.updated_at = new Date().toISOString();
  1236→        sessions.set(session.id, session);
  1237→      }
  1238→
  1239→      if (!content || typeof content !== 'string') {
  1240→        return NextResponse.json({ error: { code: 'bad_request', message: 'Message content is required', details: null } }, { status: 400 });
  1241→      }
  1242→
  1243→      // Auto-extract demo_topics and concerns from user message
  1244→      const lower = content.toLowerCase();

  1246→      for (const [topic, keywords] of Object.entries(topicKeywords)) {
  1247→        if (keywords.some(kw => lower.includes(kw)) && !(session.context.demo_topics || []).includes(topic)) {
  1248→          if (!session.context.demo_topics) session.context.demo_topics = [];
  1249→          session.context.demo_topics.push(topic);
  1250→        }
  1251→      }
  1252→      const concernKeywords = { expensive: ['expensive', 'too much', 'costly', 'overpriced'], quality: ['wrong answer', 'dumb', 'mistake', 'inaccurate'], security: ['data breach', 'hack', 'privacy', 'unsafe'], setup: ['how long', 'setup time', 'complicated'] };
  1253→      for (const [concern, keywords] of Object.entries(concernKeywords)) {
  1254→        if (keywords.some(kw => lower.includes(kw)) && !(session.context.concerns_raised || []).includes(concern)) {
  1255→          if (!session.context.concerns_raised) session.context.concerns_raised = [];

  1700→      }

  1708→      const offset = parseInt(url.searchParams.get('offset') || '0', 10);
  1709→
  1710→      if (!sessionId || !sessions.has(sessionId)) {
  1711→        return NextResponse.json({ messages: [], total: 0, limit, offset, has_more: false });
  1712→      }
  1713→
  1714→      const session = sessions.get(sessionId)!;
  1715→      const allMessages = session.messages;
  1716→      const paged = allMessages.slice(offset, offset + limit);
  1717→      return NextResponse.json({ messages: paged, total: allMessages.length, limit, offset, has_more: offset + limit < allMessages.length });
  1718→    }
  1719→
  1720→    // ── GET /demo-pack/status ─────────────────────────────────
  1721→    if (endpoint === 'demo-pack/status') {
  1722→      const sessionId = url.searchParams.get('session_id');
  1723→      if (!sessionId || !sessions.has(sessionId)) {
  1724→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1725→      }
  1726→      const session = sessions.get(sessionId)!;
  1727→      return NextResponse.json({ pack_type: session.pack_type, remaining_today: session.remaining_today, total_allowed: session.pack_type === 'demo' ? 50 : 20, pack_expiry: session.pack_type === 'demo' ? new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString() : null, demo_call_remaining: !session.context.demo_call_used });
  1728→    }
  1729→
  1730→    // ── GET /payment/status — Payment Status Check ───────────────
  1731→    if (endpoint === 'payment/status') {
  1732→      const sessionId = url.searchParams.get('session_id');
  1733→      if (!sessionId || !sessions.has(sessionId)) {
  1734→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1735→      }
  1736→      const session = sessions.get(sessionId)!;
  1737→      const paymentData = session.context.payment_data;
  1738→      return NextResponse.json({
  1739→        payment_status: session.payment_status,
  1740→        transaction_id: paymentData?.transaction_id || null,
  1741→        checkout_url: paymentData?.checkout_url || null,
  1742→        amount: paymentData?.total || 0,
  1743→        currency: paymentData?.currency || 'USD',
  1744→        items: paymentData?.items || [],
  1745→        subtotal: paymentData?.subtotal || 0,
  1746→        tax: paymentData?.tax || 0,
  1747→        created_at: paymentData?.created_at || null,
  1748→        completed_at: paymentData?.completed_at || null,
  1749→        bill_summary: session.context.bill_summary || null,
  1750→      });
  1751→    }
  1752→
  1753→    // ── GET /tickets — List Session Tickets ──────────────────────
  1754→    if (endpoint === 'tickets') {
  1755→      const sessionId = url.searchParams.get('session_id');
  1756→      if (!sessionId || !sessions.has(sessionId)) {
  1757→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1758→      }
  1759→      const session = sessions.get(sessionId)!;
  1760→      const tickets = session.context.action_tickets || [];
  1761→      const typeFilter = url.searchParams.get('type');
  1762→      const statusFilter = url.searchParams.get('status');
  1763→      const filtered = tickets.filter((t: any) => {
  1764→        if (typeFilter && t.type !== typeFilter) return false;
  1765→        if (statusFilter && t.status !== statusFilter) return false;
  1766→        return true;
  1767→      });
  1768→      return NextResponse.json({ tickets: filtered, total: filtered.length });
  1769→    }
  1770→
  1771→    // ── GET /tickets/:id — Get Specific Ticket ───────────────────
  1772→    if (endpoint.startsWith('tickets/') && endpoint.split('/').length === 2) {
  1773→      const ticketId = endpoint.split('/')[1];
  1774→      const sessionId = url.searchParams.get('session_id');
  1775→      if (!sessionId || !sessions.has(sessionId)) {
  1776→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1777→      }
  1778→      const session = sessions.get(sessionId)!;
  1779→      const tickets = session.context.action_tickets || [];
  1780→      const ticket = tickets.find((t: any) => t.id === ticketId);
  1781→      if (!ticket) {
  1782→        return NextResponse.json({ error: { code: 'not_found', message: 'Ticket not found', details: null } }, { status: 404 });
  1783→      }
  1784→      return NextResponse.json(ticket);
  1785→    }
  1786→
  1787→    return NextResponse.json({ error: { code: 'not_found', message: `Unknown GET endpoint: /${endpoint}`, details: null } }, { status: 404 });
  1788→  } catch (error: unknown) {
  1789→    console.error('Jarvis API GET error:', error);
  1790→    const message = error instanceof Error ? error.message : 'Internal server error';
  1791→    return NextResponse.json({ error: { code: 'internal_error', message, details: null } }, { status: 500 });
  1792→  }
  1793→}

  1795→export async function PATCH(request: NextRequest, { params }: { params: Promise<{ path: string[] }> }) {
  1796→  const { path } = await params;
  1797→  const endpoint = path.join('/');
  1798→  const url = new URL(request.url);
  1799→
  1800→  try {
  1801→    // ── PROXY FIRST: any /cc/* path goes to backend jarvis_cc router ──
  1802→    if (path[0] === 'cc') {
  1803→      const proxyResult = await proxyToBackend(request, path);
  1804→      if (proxyResult) return proxyResult;
  1805→      return NextResponse.json(
  1806→        { error: { code: 'backend_unavailable', message: 'Jarvis CC backend is unavailable. Please retry.', details: null } },
  1807→        { status: 502 }
  1808→      );
  1809→    }
  1810→
  1811→    // ── PATCH /context ────────────────────────────────────────
  1812→    if (endpoint === 'context') {
  1813→      const sessionId = url.searchParams.get('session_id');
  1814→      if (!sessionId || !sessions.has(sessionId)) {
  1815→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1816→      }
  1817→      const body = await request.json();
  1818→      const session = sessions.get(sessionId)!;
  1819→      session.context = { ...session.context, ...body };
  1820→      session.updated_at = new Date().toISOString();
  1821→      sessions.set(sessionId, session);
  1822→      return NextResponse.json(session);
  1823→    }
  1824→
  1825→    // ── PATCH /tickets/:id/status — Update Ticket Status ────────
  1826→    if (endpoint.startsWith('tickets/') && endpoint.endsWith('/status') && endpoint.split('/').length === 3) {
  1827→      const parts = endpoint.split('/');
  1828→      const ticketId = parts[1];
  1829→      const sessionId = url.searchParams.get('session_id');
  1830→      if (!sessionId || !sessions.has(sessionId)) {
  1831→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1832→      }
  1833→      const body = await request.json();
  1834→      const session = sessions.get(sessionId)!;
  1835→      const updated = updateActionTicket(session, ticketId, { status: body.status, metadata: body.metadata });
  1836→      if (!updated) {
  1837→        return NextResponse.json({ error: { code: 'not_found', message: 'Ticket not found', details: null } }, { status: 404 });
  1838→      }
  1839→      session.updated_at = new Date().toISOString();
  1840→      sessions.set(sessionId, session);
  1841→      return NextResponse.json(updated);
  1842→    }
  1843→
  1844→    return NextResponse.json({ error: { code: 'not_found', message: `Unknown PATCH endpoint: /${endpoint}`, details: null } }, { status: 404 });
  1845→  } catch (error: unknown) {
  1846→    console.error('Jarvis API PATCH error:', error);
  1847→    const message = error instanceof Error ? error.message : 'Internal server error';
  1848→    return NextResponse.json({ error: { code: 'internal_error', message, details: null } }, { status: 500 });
  1849→  }
  1850→}

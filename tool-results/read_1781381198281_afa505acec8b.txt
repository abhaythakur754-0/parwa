   579→        return prev.filter((t) => t !== tier);

   583→  };

   589→
   590→  const toggleAddOn = (key: 'voice' | 'customApi') => {

   596→    setCheckoutError(null);
   597→  };

   609→      toast.success('Subscription activated! Welcome to PARWA!');

   637→          context.industry = existing.industry;

   640→        // ignore
   641→      }
   642→
   643→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   644→
   645→      // ── Paddle Checkout Flow ──────────────────────────────────────

   647→
   648→      // Add each active variant as a Paddle line item (with quantity)

   690→          method: 'POST',
   691→          headers: { 'Content-Type': 'application/json' },

   693→            variant: primaryOnboardingVariant,
   694→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),

   718→          activeVariants, addOns, totalMonthly, variantQuantities,
   719→          couponCode: appliedCoupon?.code || null,

   801→      setCheckoutError('Something went wrong. Please try again.');

   807→
   808→  return (

   810→      {/* Header */}
   811→      <div className="text-center space-y-2">
   812→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   813→          <Receipt className="w-7 h-7 text-emerald-400" />
   814→        </div>
   815→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   816→        <p className="text-orange-200/40 text-sm">
   817→          Configure your plan and complete payment to activate your AI agents.
   818→        </p>
   819→      </div>
   820→
   821→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   822→      {paddleStatus === 'unavailable' && (
   823→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   824→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   825→          <div>
   826→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   827→            <p className="text-[10px] text-orange-200/30 mt-1">
   828→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   829→            </p>
   830→          </div>
   831→        </div>
   832→      )}
   833→      {paddleStatus === 'ready' && (
   834→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   835→          <Shield className="w-4 h-4 text-emerald-400" />
   836→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   837→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   838→        </div>
   839→      )}
   840→
   841→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   842→      <div className="space-y-3">
   843→        <div className="flex items-center gap-2">
   844→          <Brain className="w-4 h-4 text-orange-400" />
   845→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   846→            Active Variants
   847→          </label>
   848→          <span className="text-[10px] text-orange-200/25">
   849→            Each variant adds its own ticket allocation and AI pipeline
   850→          </span>
   851→        </div>
   852→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   853→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   854→            <VariantMixerCard
   855→              key={tier}
   856→              tier={tier}
   857→              isActive={activeVariants.includes(tier)}
   858→              quantity={variantQuantities[tier] || 1}
   859→              onToggle={() => toggleVariant(tier)}
   860→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   861→              discountedPrice={activeVariants.includes(tier) ? variantDiscountedPrices[tier] : undefined}
   862→            />
   863→          ))}
   864→        </div>
   865→      </div>
   866→
   867→      {/* ── 2. Live Usage Bar & Overage Projection ──────────────────── */}
   868→      <div className="rounded-xl border border-white/[0.06] p-4 space-y-4" style={{ background: 'rgba(255,255,255,0.03)' }}>
   869→        <div className="flex items-center gap-2">
   870→          <BarChart3 className="w-4 h-4 text-orange-400" />
   871→          <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   872→            Live Usage & Overage Projection
   873→          </span>
   874→        </div>
   875→        <UsageBar
   876→          used={projectedTickets}
   877→          total={totalTicketLimit}
   878→          label="Ticket Usage (projected at 80%)"
   879→          unit="tickets"
   880→          overageRate={`$${OVERAGE_PRICE_PER_TICKET}`}
   881→        />
   882→        {overageCost > 0 && (
   883→          <div className="rounded-lg border border-red-500/20 p-3 flex items-start gap-2" style={{ background: 'rgba(239,68,68,0.05)' }}>
   884→            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0 mt-0.5" />
   885→            <div>
   886→              <p className="text-xs font-medium text-red-400">Overage Projected</p>
   887→              <p className="text-[10px] text-orange-200/30 mt-0.5">
   888→                At 80% utilization, projected overage: ${overageCost.toFixed(2)} ({overageTickets.toLocaleString()} tickets × ${OVERAGE_PRICE_PER_TICKET}/ticket). Overage charged at end of billing cycle.
   889→              </p>
   890→            </div>
   891→          </div>
   892→        )}
   893→        {overageCost === 0 && (
   894→          <div className="rounded-lg border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   895→            <Shield className="w-3.5 h-3.5 text-emerald-400" />
   896→            <p className="text-[10px] text-emerald-400">Within limits at projected usage. No overage charges.</p>

   899→      </div>
   900→
   901→      {/* ── 3. Add-Ons ───────────────────────────────────────────────── */}
   902→      <div className="space-y-3">
   903→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   904→          Optional Add-Ons
   905→        </label>
   906→        {ADD_ONS.map((addOn) => {
   907→          const Icon = addOn.icon;
   908→          const isSelected = addOns[addOn.key];
   909→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   910→          const showPrice = !isIncluded;
   911→

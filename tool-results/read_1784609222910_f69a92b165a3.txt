    43→    43→        self,
    44→    44→        company_id: str,
    45→    45→    ) -> Dict[str, Any]:
    46→    46→        """Main entry point: Analyze company's CRM and return recommendations.
    47→    47→
    48→    48→        Steps:
    49→    49→        1. Get all connected integrations for this company
    50→    50→        2. Fetch sample data from each connected integration
    51→    51→        3. Analyze data patterns to detect business type and gaps
    52→    52→        4. Use LLM to generate personalized recommendations
    53→    53→        
    54→    54→        Returns:
    55→    55→            Dict with:
    56→    56→            - connected_integrations: list of currently connected
    57→    57→            - data_profile: summary of what data was found
    58→    58→            - recommendations: list of recommended integrations with reasons
    59→    59→            - analysis_summary: plain text explanation
    60→    60→        """
    61→    61→        logger.info("Starting CRM analysis for company %s", company_id)
    62→    62→
    63→    63→        # Step 1: Get connected integrations
    64→    64→        connected = await self._get_connected_integrations(company_id)
    65→    65→        
    66→    66→        # Step 2: Gather data from each integration
    67→    67→        data_profile = await self._gather_data_from_integrations(company_id, connected)
    68→    68→        
    69→    69→        # Step 3: Detect patterns and gaps
    70→    70→        detected_gaps = await self._detect_gaps(data_profile, connected)
    71→    71→        
    72→    72→        # Step 4: Generate LLM-powered recommendations
    73→    73→        recommendations = await self._generate_recommendations(
    74→    74→            company_id=company_id,
    75→    75→            data_profile=data_profile,
    76→    76→            connected=connected,
    77→    77→            gaps=detected_gaps,
    78→    78→        )
    79→    79→
    80→    80→        result = {
    81→    81→            "company_id": company_id,
    82→    82→            "analyzed_at": datetime.now(timezone.utc).isoformat(),
    83→    83→            "connected_integrations": connected,
    84→    84→            "data_profile": data_profile,
    85→    85→            "detected_gaps": detected_gaps,
    86→    86→            "recommendations": recommendations,
    87→    87→            "analysis_summary": self._build_summary(data_profile, recommendations),
    88→    88→        }
    89→    89→
    90→    90→        # Save to database for persistence (onboarding → dashboard transition)
    91→    91→        saved_result = self._save_analysis_result(result)

    93→    93→            result["analysis_id"] = str(saved_result.id)
    94→    94→            result["is_saved"] = True
    95→    95→        else:

    97→    97→
    98→    98→        logger.info("CRM analysis complete for company %s: %d recommendations (saved=%s)", 
    99→    99→                    company_id, len(recommendations), result["is_saved"])
   100→   100→        return result

   130→   130→    async def _gather_data_from_integrations(
   131→   131→        self,
   132→   132→        company_id: str,
   133→   133→        connected: List[Dict[str, Any]],
   134→   134→    ) -> Dict[str, Any]:
   135→   135→        """Gather sample data from each connected integration."""
   136→   136→        profile = {
   137→   137→            "total_contacts": 0,
   138→   138→            "total_orders": 0,
   139→   139→            "total_deals": 0,
   140→   140→            "has_products": False,
   141→   141→            "has_shipping_addresses": False,
   142→   142→            "has_payment_data": False,
   143→   143→            "has_email_campaigns": False,
   144→   144→            "has_ticket_data": False,
   145→   145→            "industries_detected": [],
   146→   146→            "data_points": [],
   147→   147→        }
   148→   148→
   149→   149→        for integration in connected:
   150→   150→            int_type = integration["type"]
   151→   151→            
   152→   152→            try:
   153→   153→                if int_type in ("hubspot", "salesforce", "pipedrive"):
   154→   154→                    crm_data = await self._fetch_crm_data(company_id, int_type)
   155→   155→                    profile["total_contacts"] += crm_data.get("contact_count", 0)
   156→   156→                    profile["total_deals"] += crm_data.get("deal_count", 0)
   157→   157→                    profile["has_products"] = profile["has_products"] or crm_data.get("has_products", False)
   158→   158→                    
   159→   159→                elif int_type in ("shopify", "woocommerce", "bigcommerce"):
   160→   160→                    ecommerce_data = await self._fetch_ecommerce_data(company_id, int_type)
   161→   161→                    profile["total_orders"] += ecommerce_data.get("order_count", 0)
   162→   162→                    profile["has_shipping_addresses"] = ecommerce_data.get("has_shipping_addresses", False)
   163→   163→                    profile["has_payment_data"] = profile.has_payment_data or ecommerce_data.get("has_payment_data", False)
   164→   164→                    profile["has_products"] = profile["has_products"] or ecommerce_data.get("has_products", False)
   165→   165→                    if ecommerce_data.get("industry"):
   166→   166→                        profile["industries_detected"].append("ecommerce")
   167→   167→                        
   168→   168→                elif int_type in ("stripe", "paddle", "paypal"):
   169→   169→                    payment_data = await self._fetch_payment_data(company_id, int_type)
   170→   170→                    profile["has_payment_data"] = True
   171→   171→                    profile["data_points"].append({
   172→   172→                        "source": int_type,
   173→   173→                        "type": "payment",
   174→   174→                        "count": payment_data.get("transaction_count", 0),
   175→   175→                    })
   176→   176→                    
   177→   177→                elif int_type in ("mailchimp", "klaviyo", "brevo"):
   178→   178→                    marketing_data = await self._fetch_marketing_data(company_id, int_type)
   179→   179→                    profile["has_email_campaigns"] = True
   180→   180→                    profile["data_points"].append({
   181→   181→                        "source": int_type,
   182→   182→                        "type": "marketing",
   183→   183→                        "subscriber_count": marketing_data.get("subscriber_count", 0),
   184→   184→                    })
   185→   185→                    
   186→   186→                elif int_type in ("zendesk", "freshdesk", "intercom", "gorgias"):
   187→   187→                    profile["has_ticket_data"] = True
   188→   188→                    
   189→   189→                elif int_type in ("shipstation", "aftership", "easypost", "fedex", "ups", "dhl"):
   190→   190→                    profile["has_shipping_addresses"] = True
   191→   191→
   192→   192→            except Exception as e:
   193→   193→                logger.warning("Failed to gather data from %s: %s", int_type, str(e)[:100])
   194→   194→                profile["data_points"].append({
   195→   195→                    "source": int_type,
   196→   196→                    "error": str(e)[:200],
   197→   197→                })
   198→   198→
   199→   199→        # Deduplicate industries
   200→   200→        profile["industries_detected"] = list(set(profile["industries_detected"]))
   201→   201→        return profile

   203→   203→    async def _fetch_crm_data(
   204→   204→        self, company_id: str, crm_type: str
   205→   205→    ) -> Dict[str, Any]:
   206→   206→        """Fetch data summary from CRM integrations."""
   207→   207→        import httpx
   208→   208→
   209→   209→        creds = self.integration_service.get_credential_config(company_id, crm_type)
   210→   210→        if not creds:
   211→   211→            return {"contact_count": 0, "deal_count": 0, "has_products": False}
   212→   212→
   213→   213→        try:
   214→   214→            if crm_type == "hubspot":
   215→   215→                base_url = "https://api.hubapi.com"
   216→   216→                headers = {"Authorization": f"Bearer {creds.get('api_key', '')}"}
   217→   217→                
   218→   218→                async with httpx.AsyncClient(timeout=10.0) as client:
   219→   219→                    # Get contacts count
   220→   220→                    contacts_resp = await client.get(
   221→   221→                        f"{base_url}/crm/v3/objects/contacts",
   222→   222→                        params={"limit": 1},
   223→   223→                        headers=headers,
   224→   224→                    )
   225→   225→                    contact_count = contacts_resp.json().get("total", 0) if contacts_resp.status_code == 200 else 0
   226→   226→                    
   227→   227→                    # Get deals count
   228→   228→                    deals_resp = await client.get(
   229→   229→                        f"{base_url}/crm/v3/objects/deals",
   230→   230→                        params={"limit": 1},
   231→   231→                        headers=headers,
   232→   232→                    )
   233→   233→                    deal_count = deals_resp.json().get("total", 0) if deals_resp.status_code == 200 else 0
   234→   234→                    
   235→   235→                    # Check for products
   236→   236→                    products_resp = await client.get(
   237→   237→                        f"{base_url}/crm/v3/objects/products",
   238→   238→                        params={"limit": 1},
   239→   239→                        headers=headers,
   240→   240→                    )
   241→   241→                    has_products = products_resp.status_code == 200 and products_resp.json().get("total", 0) > 0
   242→   242→
   243→   243→                return {
   244→   244→                    "contact_count": contact_count,
   245→   245→                    "deal_count": deal_count,
   246→   246→                    "has_products": has_products,
   247→   247→                }
   248→   248→
   249→   249→            elif crm_type == "shopify":
   250→   250→                return await self._fetch_shopify_summary(creds)
   251→   251→
   252→   252→        except Exception as e:
   253→   253→            logger.warning("CRM data fetch failed for %s: %s", crm_type, e)
   254→   254→
   255→   255→        return {"contact_count": 0, "deal_count": 0, "has_products": False}

   283→   283→    async def _fetch_shopify_summary(self, creds: Dict[str, Any]) -> Dict[str, Any]:
   284→   284→        """Fetch summary data from Shopify store."""
   285→   285→        import httpx
   286→   286→
   287→   287→        store_url = creds.get("store_url", "").rstrip("/")
   288→   288→        token = creds.get("access_token", "")
   289→   289→        
   290→   290→        if not store_url or not token:
   291→   291→            return {"order_count": 0, "has_shipping_addresses": False, "has_products": False}
   292→   292→
   293→   293→        headers = {"X-Shopify-Access-Token": token}
   294→   294→        
   295→   295→        async with httpx.AsyncClient(timeout=10.0) as client:
   296→   296→            # Get order count
   297→   297→            orders_resp = await client.get(
   298→   298→                f"{store_url}/admin/api/2024-01/orders/count.json",
   299→   299→                headers=headers,
   300→   300→            )
   301→   301→            order_count = orders_resp.json().get("count", 0) if orders_resp.status_code == 200 else 0
   302→   302→            
   303→   303→            # Get product count
   304→   304→            products_resp = await client.get(
   305→   305→                f"{store_url}/admin/api/2024-01/products/count.json",
   306→   306→                headers=headers,
   307→   307→            )
   308→   308→            has_products = products_resp.json().get("count", 0) > 0 if products_resp.status_code == 200 else False
   309→   309→            
   310→   310→            # Check recent orders for shipping addresses
   311→   311→            shipping_check = await client.get(
   312→   312→                f"{store_url}/admin/api/2024-01/orders.json?limit=5&status=any",
   313→   313→                headers=headers,
   314→   314→            )
   315→   315→            has_shipping = False
   316→   316→            if shipping_check.status_code == 200:
   317→   317→                orders = shipping_check.json().get("orders", [])
   318→   318→                has_shipping = any(o.get("shipping_address") for o in orders)
   319→   319→
   320→   320→        return {
   321→   321→            "order_count": order_count,
   322→   322→            "has_shipping_addresses": has_shipping,
   323→   323→            "has_products": has_products,
   324→   324→        }

   394→   394→    async def _detect_gaps(
   395→   395→        self,
   396→   396→        data_profile: Dict[str, Any],
   397→   397→        connected: List[Dict[str, Any]],
   398→   398→    ) -> List[Dict[str, Any]]:
   399→   399→        """Detect gaps between current setup and potential needs."""
   400→   400→        connected_types = {c["type"] for c in connected}
   401→   401→        gaps = []
   402→   402→
   403→   403→        # Gap detection rules
   404→   404→        gap_rules = [
   405→   405→            {
   406→   406→                "id": "shipping_missing",
   407→   407→                "condition": data_profile.get("total_orders", 0) > 0 and not data_profile.get("has_shipping_addresses"),
   408→   408→                "category": "shipping",
   409→   409→                "severity": "high",
   410→   410→                "message": "You have orders but no shipping integration for tracking",
   411→   411→                "recommended": ["shipstation", "aftership", "easypost"],
   412→   412→            },
   413→   413→            {
   414→   414→                "id": "payment_missing",
   415→   415→                "condition": data_profile.get("has_products") and not data_profile.get("has_payment_data"),
   416→   416→                "category": "payments",
   417→   417→                "severity": "high",
   418→   418→                "message": "You sell products but no payment processor is connected",
   419→   419→                "recommended": ["stripe", "paddle"],
   420→   420→            },
   421→   421→            {
   422→   422→                "id": "marketing_missing",
   423→   423→                "condition": data_profile.get("total_contacts", 0) > 100 and not data_profile.get("has_email_campaigns"),
   424→   424→                "category": "marketing",
   425→   425→                "severity": "medium",
   426→   426→                "message": f"You have {data_profile.get('total_contacts', 0)} contacts but no email marketing tool",
   427→   427→                "recommended": ["mailchimp", "klaviyo", "brevo"],
   428→   428→            },
   429→   429→            {
   430→   430→                "id": "helpdesk_missing",
   431→   431→                "condition": data_profile.get("total_contacts", 0) > 50 and not data_profile.get("has_ticket_data"),
   432→   432→                "category": "helpdesk",
   433→   433→                "severity": "medium",
   434→   434→                "message": "Growing customer base but no dedicated helpdesk system",
   435→   435→                "recommended": ["zendesk", "freshdesk", "intercom", "gorgias"],
   436→   436→            },
   437→   437→            {
   438→   438→                "id": "analytics_missing",
   439→   439→                "condition": data_profile.get("total_orders", 0) > 10 or data_profile.get("total_contacts", 0) > 100,
   440→   440→                "category": "analytics",
   441→   441→                "severity": "low",
   442→   442→                "message": "Significant activity but no analytics integration for insights",
   443→   443→                "recommended": ["mixpanel", "amplitude", "google_analytics"],
   444→   444→            },
   445→   445→            {
   446→   446→                "id": "communication_missing",
   447→   447→                "condition": data_profile.get("total_contacts", 0) > 20,
   448→   448→                "category": "communication",
   449→   449→                "severity": "low",
   450→   450→                "message": "Team communication could be improved with Slack integration",
   451→   451→                "recommended": ["slack"],
   452→   452→            },
   453→   453→        ]
   454→   454→
   455→   455→        for rule in gap_rules:
   456→   456→            if rule["condition"]:
   457→   457→                # Filter out already-connected recommendations
   458→   458→                available_recs = [r for r in rule["recommended"] if r not in connected_types]
   459→   459→                if available_recs:
   460→   460→                    gaps.append({

   467→   467→    async def _call_nvidia_glm(self, prompt: str, max_tokens: int = 800, temperature: float = 0.3) -> str:
   468→   468→        """Call NVIDIA GLM 5.2 directly for CRM analysis.
   469→   469→        
   470→   470→        Uses the same NVIDIA API key and endpoint as the main pipeline.
   471→   471→        Model: z-ai/glm-5.2 (your preferred LLM)
   472→   472→        
   473→   473→        This bypasses the generic llm_call to ensure we use NVIDIA GLM specifically
   474→   474→        for integration recommendations - giving more consistent results.
   475→   475→        """
   476→   476→        import httpx
   477→   477→        
   478→   478→        api_key = os.environ.get("NVIDIA_API_KEY", "")
   479→   479→        if not api_key:
   480→   480→            logger.warning("NVIDIA_API_KEY not set for CRM analysis")
   481→   481→            return ""
   482→   482→        
   483→   483→        messages = [
   484→   484→            {
   485→   485→                "role": "system",
   486→   486→                "content": "You are Parwa's intelligent integration advisor. You analyze business data and recommend specific third-party integrations that would improve their workflow. Always respond in valid JSON format."
   487→   487→            },
   488→   488→            {
   489→   489→                "role": "user", 
   490→   490→                "content": prompt
   491→   491→            }
   492→   492→        ]
   493→   493→        
   494→   494→        payload = {
   495→   495→            "model": "z-ai/glm-5.2",  # Your preferred NVIDIA GLM model
   496→   496→            "messages": messages,
   497→   497→            "temperature": temperature,
   498→   498→            "max_tokens": max_tokens,
   499→   499→        }
   500→   500→        headers = {
   501→   501→            "Authorization": f"Bearer {api_key}",
   502→   502→            "Content-Type": "application/json",
   503→   503→        }
   504→   504→        
   505→   505→        try:
   506→   506→            async with httpx.AsyncClient(timeout=60.0) as client:
   507→   507→                r = await client.post(
   508→   508→                    "https://integrate.api.nvidia.com/v1/chat/completions",
   509→   509→                    json=payload,
   510→   510→                    headers=headers,
   511→   511→                )
   512→   512→            
   513→   513→            if r.status_code == 200:
   514→   514→                data = r.json()
   515→   515→                content = data.get("choices", [{}])[0].get("message", {}).get("content", "")
   516→   516→                usage = data.get("usage", {})
   517→   517→                logger.info(
   518→   518→                    "NVIDIA GLM CRM analysis complete: tokens=%s (prompt=%s, completion=%s)",
   519→   519→                    usage.get("total_tokens", "?"),
   520→   520→                    usage.get("prompt_tokens", "?"),
   521→   521→                    usage.get("completion_tokens", "?"),
   522→   522→                )
   523→   523→                return content.strip()
   524→   524→            else:
   525→   525→                logger.error("NVIDIA GLM API error %d: %s", r.status_code, r.text[:200])
   526→   526→                return ""
   527→   527→                
   528→   528→        except Exception as e:
   529→   529→            logger.error("NVIDIA GLM call failed: %s", str(e)[:200])
   530→   530→            return ""

   121→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   122→
   123→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   124→  agentsReplaced: number;
   125→  humanCost: number;
   126→  savings: number;
   127→  savingsPercent: number;
   128→} {
   129→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   130→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   131→  const savings = Math.max(0, humanCost - monthlyCost);
   132→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   133→  return { agentsReplaced, humanCost, savings, savingsPercent };
   134→}

   136→// ── Usage Bar Component ────────────────────────────────────────────────
   137→
   138→function UsageBar({
   139→  used,
   140→  total,
   141→  label,
   142→  unit,
   143→  overageRate,
   144→}: {
   145→  used: number;
   146→  total: number;
   147→  label: string;
   148→  unit: string;
   149→  overageRate?: string;
   150→}) {
   151→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
   152→  const isOverLimit = used > total;
   153→
   154→  return (
   155→    <div className="space-y-1.5">
   156→      <div className="flex items-center justify-between">
   157→        <span className="text-[10px] text-orange-200/40">{label}</span>
   158→        <span className={cn(
   159→          'text-[10px] font-medium',
   160→          isOverLimit ? 'text-red-400' : 'text-orange-200/50'
   161→        )}>
   162→          {used.toLocaleString()} / {total.toLocaleString()} {unit}
   163→        </span>
   164→      </div>
   165→      <div className="h-1.5 bg-white/5 rounded-full overflow-hidden">
   166→        <div
   167→          className={cn(
   168→            'h-full rounded-full transition-all duration-700',
   169→            isOverLimit ? 'bg-gradient-to-r from-red-500 to-red-400' : 'bg-gradient-to-r from-emerald-500 to-emerald-400'
   170→          )}
   171→          style={{ width: `${pct}%` }}
   172→        />
   173→      </div>
   174→      {isOverLimit && overageRate && (
   175→        <p className="text-[9px] text-red-400 flex items-center gap-1">
   176→          <AlertTriangle className="w-2.5 h-2.5" />
   177→          Over limit — {(used - total).toLocaleString()} overage at {overageRate}/ticket
   178→        </p>
   179→      )}
   180→    </div>
   181→  );
   182→}

   268→// ── Component ──────────────────────────────────────────────────────────
   269→
   270→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   271→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   272→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   273→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   274→  const [isSubmitting, setIsSubmitting] = useState(false);
   275→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   276→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   277→
   278→  // Restore add-ons from localStorage
   279→  useEffect(() => {
   280→    try {
   281→      const stored = localStorage.getItem('parwa_pricing_context');
   282→      if (stored) {
   283→        const ctx = JSON.parse(stored) as PricingContext;
   284→        if (ctx.addOns) {
   285→          setAddOns(ctx.addOns);
   286→        }
   287→      }
   288→    } catch {
   289→      // ignore
   290→    }
   291→  }, []);
   292→
   293→  // Check if Paddle.js is available
   294→  useEffect(() => {
   295→    getPaddleInstance().then((paddle) => {
   296→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   297→    }).catch(() => {
   298→      setPaddleStatus('unavailable');
   299→    });
   300→  }, []);
   301→
   302→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   303→
   304→  const totalTicketLimit = useMemo(
   305→    () => activeVariants.reduce((sum, tier) => sum + VARIANT_LIMITS[tier].monthlyTickets, 0),
   306→    [activeVariants]
   307→  );
   308→
   309→  const baseSubscription = useMemo(
   310→    () => activeVariants.reduce((sum, tier) => sum + VARIANT_PRICES[tier], 0),
   311→    [activeVariants]
   312→  );
   313→
   314→  const addOnTotal = useMemo(() => {
   315→    let total = 0;
   316→    if (addOns.voice) {
   317→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   318→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   319→        total += voiceAddOn.price;
   320→      }
   321→    }
   322→    if (addOns.customApi) {
   323→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   324→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   325→        total += customApiAddOn.price;
   326→      }
   327→    }
   328→    return total;
   329→  }, [addOns, activeVariants]);
   330→
   331→  const totalMonthly = baseSubscription + addOnTotal;
   332→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   333→
   334→  // Overage projection (estimates at 80% utilization)
   335→  const projectedTickets = Math.round(totalTicketLimit * 0.8);
   336→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);
   337→  const overageCost = overageTickets * OVERAGE_PRICE_PER_TICKET;
   338→
   339→  const toggleVariant = (tier: VariantTier) => {
   340→    setActiveVariants((prev) => {
   341→      if (prev.includes(tier)) {
   342→        if (prev.length <= 1) {
   343→          toast.error('You must have at least one active variant');
   344→          return prev;
   345→        }
   346→        return prev.filter((t) => t !== tier);
   347→      }
   348→      return [...prev, tier];
   349→    });
   350→  };
   351→
   352→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   353→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   354→  };
   355→
   356→  const handleProceed = async () => {
   357→    setIsSubmitting(true);
   358→    setCheckoutError(null);
   359→
   360→    try {
   361→      // Update pricing context in localStorage
   362→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   363→      const context: PricingContext = {
   364→        industry: 'other',
   365→        variant: primaryOnboardingVariant,
   366→        addOns,
   367→        totalMonthly,
   368→        timestamp: new Date().toISOString(),
   369→      };
   370→
   371→      // Preserve industry from existing context
   372→      try {
   373→        const stored = localStorage.getItem('parwa_pricing_context');
   374→        if (stored) {
   375→          const existing = JSON.parse(stored) as PricingContext;
   376→          context.industry = existing.industry;
   377→        }
   378→      } catch {
   379→        // ignore
   380→      }
   381→
   382→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   383→
   384→      // ── Paddle Checkout Flow ──────────────────────────────────────
   385→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   386→
   387→      // Add each active variant as a Paddle line item
   388→      for (const tier of activeVariants) {
   389→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   390→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   391→        if (variantPriceId) {
   392→          checkoutItems.push({ priceId: variantPriceId, quantity: 1 });
   393→        }
   394→      }
   395→
   396→      // Voice add-on (only for starter where it's not included)
   397→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   398→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   399→      }
   400→
   401→      // Custom API add-on (only for variants where it's not included)
   402→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   403→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });
   404→      }
   405→
   406→      const customData = {
   407→        source: 'parwa_onboarding',
   408→        variant: primaryOnboardingVariant,
   409→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   410→        addOns,
   411→        industry: context.industry,
   412→        totalMonthly,
   413→      };
   414→
   415→      // Try server-side transaction first
   416→      let transactionId: string | null = null;
   417→      let checkoutUrl: string | null = null;
   418→
   419→      try {
   420→        const res = await fetch('/api/onboarding/checkout', {
   421→          method: 'POST',
   422→          headers: { 'Content-Type': 'application/json' },
   423→          body: JSON.stringify({
   424→            variant: primaryOnboardingVariant,
   425→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   426→            addOns,
   427→            totalMonthly,
   428→            industry: context.industry,
   429→          }),
   430→        });
   431→
   432→        if (res.ok) {
   433→          const checkoutData = await res.json();
   434→          transactionId = checkoutData.transaction_id || null;
   435→          checkoutUrl = checkoutData.checkout_url || null;
   436→        }
   437→      } catch {
   438→        // API unavailable — continue with client-side Paddle checkout
   439→      }
   440→
   441→      // If we got a checkout_url, redirect to it
   442→      if (checkoutUrl) {
   443→        window.location.href = checkoutUrl;
   444→        return;
   445→      }
   446→
   447→      // Client-side Paddle checkout (items-based)
   448→      if (paddleStatus === 'ready' && checkoutItems.length > 0) {
   449→        const opened = await openCheckoutWithItems(
   450→          checkoutItems,
   451→          customData,
   452→          () => {
   453→            toast.success('Payment successful! Welcome to PARWA!');
   454→            onComplete();
   455→          },
   456→          () => {
   457→            toast('Checkout closed. You can try again anytime.', { icon: 'ℹ️' });
   458→            setIsSubmitting(false);
   459→          },
   460→        );
   461→        if (opened) {
   462→          return;
   463→        }
   464→      }
   465→
   466→      // Fallback: No Paddle available — save configuration and proceed
   467→      toast.success('Configuration saved! Complete payment to activate your plan.');
   468→      onComplete();
   469→    } catch (err) {
   470→      console.error('[cost-breakdown] Error:', err);
   471→      setCheckoutError('Something went wrong. Please try again.');
   472→      toast.error('Something went wrong. Please try again.');
   473→    } finally {
   474→      setIsSubmitting(false);
   475→    }
   476→  };
   477→
   478→  return (
   479→    <div className="space-y-6">
   480→      {/* Header */}
   481→      <div className="text-center space-y-2">
   482→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   483→          <Receipt className="w-7 h-7 text-emerald-400" />
   484→        </div>
   485→        <h2 className="text-2xl font-bold text-white">Review Your Plan</h2>
   486→        <p className="text-orange-200/40 text-sm">
   487→          Configure variants, add-ons, and review costs before checkout.
   488→        </p>
   489→      </div>
   490→
   491→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   492→      {paddleStatus === 'unavailable' && (
   493→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   494→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   495→          <div>
   496→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   497→            <p className="text-[10px] text-orange-200/30 mt-1">
   498→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   499→            </p>
   500→          </div>
   501→        </div>
   502→      )}
   503→      {paddleStatus === 'ready' && (
   504→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   505→          <Shield className="w-4 h-4 text-emerald-400" />
   506→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   507→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   508→        </div>
   509→      )}
   510→
   511→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   512→      <div className="space-y-3">
   513→        <div className="flex items-center gap-2">
   514→          <Brain className="w-4 h-4 text-orange-400" />
   515→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   516→            Active Variants
   517→          </label>
   518→          <span className="text-[10px] text-orange-200/25">
   519→            Each variant adds its own ticket allocation and AI pipeline
   520→          </span>
   521→        </div>
   522→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   523→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   524→            <VariantMixerCard
   525→              key={tier}
   526→              tier={tier}
   527→              isActive={activeVariants.includes(tier)}
   528→              onToggle={() => toggleVariant(tier)}
   529→            />
   530→          ))}
   531→        </div>
   532→      </div>
   533→
   534→      {/* ── 2. Live Usage Bar & Overage Projection ──────────────────── */}
   535→      <div className="rounded-xl border border-white/[0.06] p-4 space-y-4" style={{ background: 'rgba(255,255,255,0.03)' }}>
   536→        <div className="flex items-center gap-2">
   537→          <BarChart3 className="w-4 h-4 text-orange-400" />
   538→          <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   539→            Live Usage & Overage Projection
   540→          </span>
   541→        </div>
   542→        <UsageBar
   543→          used={projectedTickets}
   544→          total={totalTicketLimit}
   545→          label="Ticket Usage (projected at 80%)"
   546→          unit="tickets"
   547→          overageRate={`$${OVERAGE_PRICE_PER_TICKET}`}
   548→        />
   549→        {overageCost > 0 && (
   550→          <div className="rounded-lg border border-red-500/20 p-3 flex items-start gap-2" style={{ background: 'rgba(239,68,68,0.05)' }}>
   551→            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0 mt-0.5" />
   552→            <div>
   553→              <p className="text-xs font-medium text-red-400">Overage Projected</p>
   554→              <p className="text-[10px] text-orange-200/30 mt-0.5">
   555→                At 80% utilization, projected overage: ${overageCost.toFixed(2)} ({overageTickets.toLocaleString()} tickets × ${OVERAGE_PRICE_PER_TICKET}/ticket). Overage charged at end of billing cycle.
   556→              </p>
   557→            </div>
   558→          </div>
   559→        )}
   560→        {overageCost === 0 && (
   561→          <div className="rounded-lg border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   562→            <Shield className="w-3.5 h-3.5 text-emerald-400" />
   563→            <p className="text-[10px] text-emerald-400">Within limits at projected usage. No overage charges.</p>
   564→          </div>
   565→        )}
   566→      </div>
   567→
   568→      {/* ── 3. Add-Ons ───────────────────────────────────────────────── */}
   569→      <div className="space-y-3">
   570→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   571→          Optional Add-Ons
   572→        </label>
   573→        {ADD_ONS.map((addOn) => {
   574→          const Icon = addOn.icon;
   575→          const isSelected = addOns[addOn.key];
   576→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   577→          const showPrice = !isIncluded;
   578→
   579→          return (
   580→            <button
   581→              key={addOn.key}
   582→              type="button"
   583→              onClick={() => toggleAddOn(addOn.key)}
   584→              className={cn(
   585→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',
   586→                isSelected
   587→                  ? 'border-orange-500/30 bg-orange-500/5'
   588→                  : 'border-white/[0.06] hover:border-orange-500/15'
   589→              )}
   590→              style={!isSelected ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   591→            >
   592→              <div className={cn(
   593→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   594→                isSelected ? 'bg-orange-500/10' : 'bg-white/[0.04]'
   595→              )}>
   596→                <Icon className={cn(
   597→                  'w-5 h-5',
   598→                  isSelected ? 'text-orange-400' : 'text-zinc-500'
   599→                )} />
   600→              </div>
   601→              <div className="flex-1 min-w-0">
   602→                <div className="flex items-center gap-2">
   603→                  <p className={cn(
   604→                    'text-sm font-medium',
   605→                    isSelected ? 'text-orange-400' : 'text-white'
   606→                  )}>
   607→                    {addOn.name}
   608→                  </p>
   609→                  {isIncluded && (
   610→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   611→                      Included
   612→                    </span>
   613→                  )}
   614→                </div>
   615→                <p className="text-[10px] text-orange-200/30 mt-0.5">{addOn.description}</p>
   616→              </div>
   617→              <div className="flex flex-col items-end shrink-0 gap-1">
   618→                {showPrice && (
   619→                  <p className="text-sm font-semibold text-white">${addOn.price}/mo</p>
   620→                )}
   621→                <div className={cn(
   622→                  'w-8 h-5 rounded-full flex items-center px-0.5 transition-colors duration-200',
   623→                  isSelected || isIncluded
   624→                    ? 'bg-gradient-to-r from-orange-500 to-amber-400'
   625→                    : 'bg-white/10'
   626→                )}>
   627→                  <div className={cn(
   628→                    'w-4 h-4 rounded-full bg-white shadow-sm transition-transform duration-200',
   629→                    (isSelected || isIncluded) ? 'translate-x-3' : 'translate-x-0'
   630→                  )} />
   631→                </div>
   632→              </div>
   633→            </button>
   634→          );
   635→        })}
   636→      </div>
   637→
   638→      {/* ── 4. Cost Summary (pure math — D7) ──────────────────────────── */}
   639→      <div
   640→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   641→        style={{ background: 'rgba(255,255,255,0.03)' }}
   642→      >
   643→        {/* Per-variant breakdown */}
   644→        {activeVariants.map((tier) => (
   645→          <div key={tier} className="flex items-center justify-between">
   646→            <span className="text-sm text-orange-200/50 flex items-center gap-1.5">
   647→              <div className={cn(
   648→                'w-2 h-2 rounded-full',
   649→                tier === 'starter' ? 'bg-emerald-400' : tier === 'growth' ? 'bg-orange-400' : 'bg-purple-400'
   650→              )} />
   651→              {VARIANT_DISPLAY[tier].name}
   652→            </span>
   653→            <span className="text-sm text-white">${VARIANT_PRICES[tier].toLocaleString()}/mo</span>
   654→          </div>
   655→        ))}
   656→
   657→        {/* Add-ons */}
   658→        {addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t)) && (
   659→          <div className="flex items-center justify-between">
   660→            <span className="text-sm text-orange-200/50">Voice Channel</span>
   661→            <span className="text-sm text-white">$199/mo</span>
   662→          </div>
   663→        )}
   664→        {addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t)) && (
   665→          <div className="flex items-center justify-between">
   666→            <span className="text-sm text-orange-200/50">Custom API Connector</span>
   667→            <span className="text-sm text-white">$49/mo</span>
   668→          </div>
   669→        )}
   670→
   671→        {/* Integrations = $0 (D13) */}
   672→        <div className="flex items-center justify-between">
   673→          <span className="text-sm text-orange-200/50">Integrations</span>
   674→          <span className="text-sm text-emerald-400">$0 (free)</span>
   675→        </div>
   676→
   677→        <div className="border-t border-white/[0.06] pt-3 flex items-center justify-between">
   678→          <span className="text-sm font-semibold text-white">Total Monthly</span>
   679→          <span className="text-lg font-bold text-orange-400">${totalMonthly.toLocaleString()}/mo</span>
   680→        </div>
   681→
   682→        {/* Overage rate info */}
   683→        <div className="flex items-center justify-between">
   684→          <span className="text-[10px] text-orange-200/25">Overage rate (beyond limits)</span>
   685→          <span className="text-[10px] text-orange-200/25">${OVERAGE_PRICE_PER_TICKET}/ticket</span>
   686→        </div>
   687→      </div>
   688→
   689→      {/* ── 5. Savings Comparison (D10 — reuse ROI Calculator logic) ──── */}
   690→      <div
   691→        className="rounded-xl border border-emerald-500/20 p-4"
   692→        style={{ background: 'rgba(16,185,129,0.04)' }}
   693→      >
   694→        <div className="flex items-start gap-3">
   695→          <div className="w-8 h-8 rounded-lg bg-emerald-500/10 flex items-center justify-center shrink-0">
   696→            <TrendingDown className="w-4 h-4 text-emerald-400" />
   697→          </div>
   698→          <div className="flex-1">
   699→            <p className="text-sm font-semibold text-emerald-400">
   700→              Save {savings.savingsPercent}% vs human agents
   701→            </p>
   702→            <p className="text-[10px] text-orange-200/30 mt-1">
   703→              PARWA handles {totalTicketLimit.toLocaleString()} tickets/mo — equivalent to{' '}
   704→              {savings.agentsReplaced} full-time support agent{savings.agentsReplaced > 1 ? 's' : ''} at ~${AGENT_COST_MONTHLY.toLocaleString()}/mo each.
   705→              That&apos;s ${savings.humanCost.toLocaleString()}/mo in human costs vs ${totalMonthly.toLocaleString()}/mo with PARWA.
   706→            </p>
   707→          </div>
   708→        </div>
   709→      </div>
   710→
   711→      {/* ── No Hidden Fees (D13) ──────────────────────────────────────── */}
   712→      <div className="flex items-center justify-center gap-2 text-[10px] text-orange-200/25">
   713→        <Shield className="w-3 h-3" />
   714→        <span>No hidden fees. Need more? Add another variant.</span>
   715→      </div>
   716→
   717→      {/* ── Error Display ────────────────────────────────────────────── */}
   718→      {checkoutError && (
   719→        <div className="rounded-xl border border-red-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(239,68,68,0.05)' }}>
   720→          <AlertCircle className="w-5 h-5 text-red-400 shrink-0 mt-0.5" />
   721→          <p className="text-sm text-red-300">{checkoutError}</p>
   722→        </div>
   723→      )}
   724→
   725→      {/* ── Proceed Button ────────────────────────────────────────────── */}
   726→      <div className="flex justify-end">
   727→        <button
   728→          onClick={handleProceed}
   729→          disabled={isSubmitting}
   730→          className="px-8 py-3 bg-gradient-to-r from-orange-500 to-amber-400 hover:from-orange-400 hover:to-amber-300 text-[#1A1A1A] font-semibold rounded-xl transition-all duration-300 shadow-lg shadow-orange-500/25 text-sm flex items-center gap-2 disabled:opacity-60 disabled:cursor-not-allowed"
   731→        >
   732→          {isSubmitting ? (
   733→            <>
   734→              <Loader2 className="w-4 h-4 animate-spin" />
   735→              Processing...
   736→            </>
   737→          ) : (
   738→            <>
   739→              Proceed to Checkout
   740→              <ArrowRight className="w-4 h-4" />
   741→            </>
   742→          )}
   743→        </button>
   744→      </div>
   745→    </div>
   746→  );
   747→}

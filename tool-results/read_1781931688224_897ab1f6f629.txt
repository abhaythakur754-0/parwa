    56→# ── Optional Auth Dependency ────────────────────────────────
    57→# Validates JWT/API key if provided, but does NOT block unauthenticated requests.
    58→# This allows the dashboard to work in dev without forcing full auth,
    59→# while still validating credentials when they are present.
    60→
    61→async def _optional_auth(request: Request) -> Optional[Dict[str, Any]]:
    62→    """Extract auth context from request if Authorization header is present.
    63→
    64→    Returns None if no auth header (allows unauthenticated dev access).
    65→    Raises 401 if auth header is present but invalid.
    66→    """
    67→    from app.core.auth.access_control import validate_request
    68→
    69→    auth_header = request.headers.get("Authorization", "")
    70→    if not auth_header.startswith("Bearer "):
    71→        return None  # No auth provided — dev mode
    72→
    73→    token = auth_header[7:].strip()
    74→    if not token:
    75→        return None
    76→
    77→    client_ip = request.client.host if request.client else "unknown"
    78→    is_valid, context, _ = await validate_request(
    79→        api_key=token if token.startswith("pk_live_") or token.startswith("pk_test_") else None,
    80→        jwt_token=token if not (token.startswith("pk_live_") or token.startswith("pk_test_")) else None,
    81→        client_ip=client_ip,
    82→    )
    83→    if is_valid:
    84→        return context
    85→    return None  # Invalid but don't block — middleware can enforce later

    92→@router.post(
    93→    "/jarvis/chat",
    94→    summary="Main chat endpoint — runs full SENSE → EVALUATE → NOTIFY pipeline",
    95→    response_model=Dict[str, Any],
    96→)
    97→async def jarvis_chat(req: ChatRequest):
    98→    """Accept a natural-language question and run the full Jarvis pipeline."""
    99→    from app.core.jarvis_pipeline.graph import run_jarvis_chat
   100→
   101→    tenant_id = req.tenant_id or _DEFAULT_TENANT
   102→
   103→    # Emit SSE init event
   104→    await emit_pipeline_event(tenant_id, "init", {
   105→        "question": req.question,
   106→        "user_email": req.user_email,
   107→    })
   108→    await emit_pipeline_event(tenant_id, "sense_start", {})
   109→
   110→    try:
   111→        result = await run_jarvis_chat(
   112→            tenant_id=tenant_id,
   113→            question=req.question,
   114→            user_email=req.user_email,
   115→            user_role=req.user_role,
   116→        )
   117→    except Exception as exc:
   118→        await emit_pipeline_event(tenant_id, "error", {"error": str(exc)})
   119→        raise _err(f"Pipeline error: {exc}", 500)
   120→
   121→    # Emit pipeline completion events
   122→    await emit_pipeline_event(tenant_id, "sense_complete",
   123→                              {"signals": result.get("signals", {}).get("integration_health", {})})
   124→    await emit_pipeline_event(tenant_id, "evaluate_start", {})
   125→    await emit_pipeline_event(tenant_id, "evaluate_complete",
   126→                              {"evaluations": len(result.get("evaluations", []))})
   127→    await emit_pipeline_event(tenant_id, "notify_start", {})
   128→    await emit_pipeline_event(tenant_id, "notify_complete",
   129→                              {"notifications": len(result.get("notifications", []))})
   130→    await emit_pipeline_event(tenant_id, "done", {"chat_response": result.get("chat_response", "")[:200]})
   131→
   132→    return result

   818→# ═══════════════════════════════════════════════════════════════
   819→# TICKET SUBMISSION (PARWA PIPELINE)
   820→# ═══════════════════════════════════════════════════════════════
   821→
   822→@router.post(
   823→    "/tickets/submit",
   824→    summary="Submit a ticket to the PARWA pipeline",
   825→    response_model=Dict[str, Any],
   826→)
   827→async def submit_ticket(req: TicketSubmitRequest):
   828→    """Run a customer ticket through the full PARWA pipeline.
   829→
   830→    This endpoint processes the ticket INSIDE the server process,
   831→    so quality scores, notifications, and audit entries are written
   832→    to the same shared DB that all API endpoints read from.
   833→
   834→    Flow: Ingest → Classify → Route → Knowledge → Reasoning/Simple
   835→          → Act+Verify → Quality → Finalize → END
   836→    """
   837→    import uuid
   838→    import time as _time
   839→
   840→    from app.core.parwa_pipeline.graph_v2 import run_parwa_pipeline
   841→    from app.core.parwa_pipeline.nodes.node_2_smart_route import set_test_variant
   842→
   843→    tenant_id = req.tenant_id or _DEFAULT_TENANT
   844→    ticket_id = f"tkt_api_{uuid.uuid4().hex[:8]}"
   845→
   846→    # Emit SSE events
   847→    await emit_pipeline_event(tenant_id, "ticket_received", {
   848→        "ticket_id": ticket_id,
   849→        "channel": req.channel_type,
   850→    })
   851→    await emit_pipeline_event(tenant_id, "pipeline_start", {"ticket_id": ticket_id})
   852→
   853→    # Set variant for this ticket
   854→    set_test_variant(tenant_id, req.variant_tier, 2000)
   855→
   856→    initial_state = {
   857→        "ticket_id": ticket_id,
   858→        "tenant_id": tenant_id,
   859→        "query": req.query,
   860→        "channel_type": req.channel_type,
   861→        "customer_context": req.customer_context or {},
   862→        "metadata": {
   863→            "sender": req.sender or "",
   864→            "timestamp": datetime.now(timezone.utc).isoformat(),
   865→            "source": "api_submission",
   866→        },
   867→        "loop_count": 0,
   868→        "total_token_usage": 0,
   869→        "technique_log": [],
   870→        "errors": [],
   871→    }
   872→
   873→    try:
   874→        start = _time.monotonic()
   875→        # run_parwa_pipeline is sync — handles async internally
   876→        result = run_parwa_pipeline(initial_state)
   877→        elapsed = round(_time.monotonic() - start, 2)
   878→
   879→        status = result.get("status", "unknown")
   880→        quality = result.get("quality_score", None)
   881→        route = result.get("route_decision", result.get("current_path", "unknown"))
   882→        response = (
   883→            result.get("final_response", "")
   884→            or result.get("formatted_response", "")
   885→            or result.get("simple_answer", "")
   886→            or result.get("super_node_answer", "")
   887→        )
   888→
   889→        # Emit completion events
   890→        await emit_pipeline_event(tenant_id, "pipeline_complete", {
   891→            "ticket_id": ticket_id,
   892→            "status": status,
   893→            "quality": quality,
   894→            "elapsed": elapsed,
   895→        })
   896→
   897→        # Write audit entry
   898→        from app.core.jarvis_pipeline.jarvis_db import get_db
   899→        db = get_db()
   900→        await db.create_audit_entry(
   901→            tenant_id=tenant_id,
   902→            action="ticket_processed",
   903→            actor_email="system",
   904→            target_type="ticket",
   905→            target_id=ticket_id,
   906→            payload={
   907→                "status": status,
   908→                "quality_score": quality,
   909→                "route": route,
   910→                "elapsed": elapsed,
   911→                "channel": req.channel_type,
   912→            },
   913→        )
   914→
   915→        return {
   916→            "ok": True,
   917→            "ticket_id": ticket_id,
   918→            "status": status,
   919→            "quality_score": quality,
   920→            "route": route,
   921→            "loop_count": result.get("loop_count", 0),
   922→            "escalated": bool(result.get("escalation_context")),
   923→            "elapsed_seconds": elapsed,
   924→            "response_preview": response[:500],
   925→        }
   926→
   927→    except Exception as exc:
   928→        await emit_pipeline_event(tenant_id, "pipeline_error", {
   929→            "ticket_id": ticket_id,
   930→            "error": str(exc),
   931→        })
   932→        raise _err(f"Pipeline error: {exc}", 500)

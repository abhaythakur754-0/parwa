    35→/**
    36→ * Try to proxy a request to the backend FastAPI server.
    37→ * Returns the Response on success, or null if backend is unavailable / returned an error.
    38→ */
    39→/**
    40→ * Result of proxyToBackend attempt.
    41→ * - response: successful backend response (2xx)
    42→ * - null with consumedBody: backend failed, body was already read (use for fallback)
    43→ * - null without consumedBody: no body needed (GET/DELETE) or BACKEND_URL not set
    44→ */
    45→interface ProxyResult {
    46→  response: Response | null;
    47→  /** The raw request body as a UTF-8 string (only set when body was read).
    48→   *  Callers should use this instead of calling request.json()/text() again. */
    49→  consumedBody?: string;
    50→}
    51→
    52→async function proxyToBackend(request: NextRequest, pathSegments: string[]): Promise<ProxyResult> {
    53→  if (!BACKEND_URL) return { response: null };
    54→
    55→  const backendPath = `${BACKEND_URL}/api/jarvis/${pathSegments.join('/')}`;
    56→  const url = new URL(request.url);
    57→  const searchParams = url.searchParams.toString();
    58→  const fullUrl = searchParams ? `${backendPath}?${searchParams}` : backendPath;
    59→
    60→  // Determine proxy origin (must match an allowed origin on the backend CSRF middleware)
    61→  // FRONTEND_URL takes priority (set to https://parwa.buzz in production).
    62→  // VERCEL_URL is only used for preview deployments — in production it's
    63→  // a random deployment URL that the backend would reject via CORS.
    64→  const proxyOrigin =
    65→    process.env.FRONTEND_URL ||
    66→    (process.env.NODE_ENV === 'production' ? 'https://parwa.buzz' : null) ||
    67→    (process.env.VERCEL_URL ? `https://${process.env.VERCEL_URL}` : null) ||
    68→    'http://localhost:3000';
    69→
    70→  let consumedBody: string | undefined;
    71→
    72→  try {
    73→    // Read body ONCE — if we need to fall back to local handling, the caller
    74→    // must use consumedBody instead of trying to read the request again
    75→    // (which would throw "Body has already been read" / 500 error).
    76→    const method = request.method;
    77→    let body: ArrayBuffer | undefined;
    78→    if (['POST', 'PATCH', 'PUT'].includes(method)) {
    79→      const arrayBuf = await request.arrayBuffer();
    80→      body = arrayBuf;
    81→      // Decode for fallback use
    82→      const decoder = new TextDecoder();
    83→      consumedBody = decoder.decode(arrayBuf);
    84→    }
    85→
    86→    const headers: Record<string, string> = {
    87→      'Content-Type': request.headers.get('Content-Type') || 'application/json',
    88→      'Origin': proxyOrigin,
    89→      'Referer': `${proxyOrigin}/`,
    90→    };
    91→
    92→    // Forward auth token from parwa_at cookie as Bearer token
    93→    const authHeader = request.headers.get('authorization');
    94→    if (authHeader) {
    95→      headers['Authorization'] = authHeader;
    96→    } else {
    97→      const cookieToken = request.cookies.get('parwa_at')?.value;
    98→      if (cookieToken) {
    99→        headers['Authorization'] = `Bearer ${cookieToken}`;
   100→      }
   101→    }
   102→
   103→    const response = await fetch(fullUrl, {
   104→      method,
   105→      headers,
   106→      body,
   107→      signal: AbortSignal.timeout(90000),  // 90s — AI pipeline can take 30-60s
   108→    });
   109→
   110→    if (response.status >= 200 && response.status < 300) {
   111→      // Don't return the raw response — Vercel's edge adds content-encoding
   112→      // on top of any backend compression, causing "Failed to fetch" in the
   113→      // browser. Parse the body and return a clean Response instead.
   114→      const text = await response.text();
   115→      return {
   116→        response: new Response(text, {
   117→          status: response.status,
   118→          headers: {
   119→            'Content-Type': response.headers.get('Content-Type') || 'application/json',
   120→          },
   121→        }),
   122→      };
   123→    }
   124→
   125→    // For /cc/* paths (handled by my proxy-first check), return the actual
   126→    // error response instead of falling back to local handlers. The backend
   127→    // 401/403/500 should be visible to the client.
   128→    if (pathSegments[0] === 'cc') {
   129→      const text = await response.text();
   130→      try {
   131→        const data = JSON.parse(text);
   132→        return {
   133→          response: new Response(JSON.stringify(data), {
   134→            status: response.status,
   135→            headers: { 'Content-Type': 'application/json' },
   136→          }),
   137→        };
   138→      } catch {
   139→        return {
   140→          response: new Response(text, {
   141→            status: response.status,
   142→            headers: { 'Content-Type': 'application/json' },
   143→          }),
   144→        };
   145→      }
   146→    }
   147→
   148→    // Backend returned an error — fall back to local handling
   149→    // Return consumedBody so the caller can parse it without re-reading the request
   150→    return { response: null, consumedBody };
   151→  } catch (err) {
   152→    // Backend unreachable — fall back to local handling
   153→    console.warn('[Jarvis] Backend proxy failed:', (err instanceof Error ? err.message : String(err))?.slice(0, 150));
   154→    return { response: null, consumedBody };
   155→  }
   156→}

   158→// ── z-ai-web-dev-sdk — Primary AI Provider ───────────────────────
   159→
   160→let ZAI: any = null;
   161→
   162→async function getZAI() {
   163→  if (!ZAI) {
   164→    try {
   165→      const mod = await import('z-ai-web-dev-sdk');
   166→      const ZAIClass = (mod as any).default;
   167→      if (ZAIClass && typeof ZAIClass.create === 'function') {
   168→        ZAI = await ZAIClass.create();
   169→      }
   170→    } catch (err) {
   171→      console.warn('[Jarvis] z-ai-web-dev-sdk not available:', (err instanceof Error ? err.message : String(err))?.slice(0, 100));
   172→    }
   173→  }
   174→  return ZAI;
   175→}
   176→
   177→async function callZAISDK(messages: Array<{role: string, content: string}>): Promise<string | null> {
   178→  try {
   179→    const zai = await getZAI();
   180→    if (!zai || !zai.chat || !zai.chat.completions) return null;
   181→
   182→    const completion = await zai.chat.completions.create({
   183→      messages: messages.map(m => ({
   184→        role: m.role === 'assistant' ? 'assistant' : m.role,
   185→        content: m.content,
   186→      })),
   187→      temperature: 0.8,
   188→      max_tokens: 800,
   189→    });
   190→
   191→    const text = completion?.choices?.[0]?.message?.content;
   192→    if (text && text.trim().length > 10) return text.trim();
   193→    return null;
   194→  } catch (err) {
   195→    console.warn('[Jarvis] z-ai-web-dev-sdk failed:', (err instanceof Error ? err.message : String(err))?.slice(0, 150));
   196→    return null;
   197→  }
   198→}

   278→function getProvider(name: string): any | null {
   279→  switch (name) {
   280→    case 'google': return GOOGLE_AI_KEY ? getGoogleProvider() : null;
   281→    case 'cerebras': return CEREBRAS_KEY ? getCerebrasProvider() : null;
   282→    case 'groq': return GROQ_KEY ? getGroqProvider() : null;
   283→    default: return null;
   284→  }

   286→
   287→async function callProvider(provider: any, messages: Array<{role: string, content: string}>): Promise<string | null> {
   288→  if (!provider.apiKey) return null;
   289→
   290→  const response = await fetch(provider.apiUrl, {
   291→    method: 'POST',
   292→    headers: provider.buildHeaders(provider.apiKey),
   293→    body: provider.buildBody(messages, provider.model),
   294→    signal: AbortSignal.timeout(15000),
   295→  });
   296→
   297→  if (!response.ok) {
   298→    const errorText = await response.text().catch(() => 'Unknown error');
   299→    throw new Error(`HTTP ${response.status}: ${errorText.slice(0, 200)}`);
   300→  }
   301→
   302→  const data = await response.json();
   303→  const text = provider.parseResponse(data);
   304→
   305→  if (!text || text.trim().length < 10) {
   306→    throw new Error('Empty or too-short response from provider');
   307→  }
   308→
   309→  return text.trim();
   310→}

   314→async function callAI(messages: Array<{role: string, content: string}>): Promise<string | null> {
   315→  // 1. Try z-ai-web-dev-sdk FIRST (most reliable in production)
   316→  try {
   317→    const result = await callZAISDK(messages);
   318→    if (result) return result;
   319→  } catch (error) {
   320→    console.warn('[Jarvis] z-ai-web-dev-sdk error:', (error instanceof Error ? error.message : String(error))?.slice(0, 100));
   321→  }
   322→
   323→  // 2. Try free providers in order: Google → Cerebras → Groq
   324→  const providerList = ['google', 'cerebras', 'groq'];
   325→  for (const name of providerList) {
   326→    const provider = getProvider(name);
   327→    if (provider) {
   328→      try {
   329→        const result = await callProvider(provider, messages);
   330→        if (result) return result;
   331→      } catch (error) {
   332→        console.warn(`[Jarvis] Provider "${name}" failed:`, (error instanceof Error ? error.message : String(error))?.slice(0, 100));
   333→      }
   334→    }
   335→  }
   336→
   337→  // All providers failed — keyword fallback
   338→  console.warn('[Jarvis] All AI providers failed, using keyword fallback');
   339→  return null;
   340→}

   700→function calculateBillSummary(session: any) {

   702→  const items: Array<{ name: string; price: number; type: string }> = [];

   704→  // Add plan cost

   711→  const variants = ctx.selected_variants || [];
   712→  for (const v of variants) {
   713→    const vKey = String(typeof v === 'string' ? v : v.key || v.name || '').toLowerCase().replace(/\s+/g, '_');

   719→  const subtotal = items.reduce((sum, i) => sum + i.price, 0);

   721→  const total = subtotal + tax;

   822→const BULLET_EMOJIS = ['🤖', '💰', '📡', '🛒', '💻', '🚛', '🏥', '🔒', '🚀', '✅', '🎯', '📊', '⚡', '💡', '🔧', '🔗', '📦', '📈', '⭐', '🎯', '🧠', '💬', '📁', '🔑', '🎉', '✨', '🔔', '💪', '🥊', '🛡️'];

   824→function isEmojiChar(ch: string): boolean {
   825→  // Must use codePointAt for emoji (surrogate pairs in JS)
   826→  const code = ch.codePointAt(0) || 0;
   827→  return (code >= 0x1F300 && code <= 0x1FAFF) || (code >= 0x2600 && code <= 0x27BF) || (code >= 0xFE00 && code <= 0xFE0F);
   828→}

   830→function forceBulletFormat(text: string): string {
   831→  const lines = text.split('\n');
   832→  if (lines.length === 0) return text;
   833→
   834→  // Check if text already has proper bullet formatting
   835→  const nonEmpty = lines.filter(l => l.trim());
   836→  if (nonEmpty.length === 0) return text;
   837→
   838→  // Count bullet-formatted lines (lines starting with bullet markers or emojis)
   839→  const bulletCount = nonEmpty.filter(l => {
   840→    const t = l.trim();
   841→    return /^[\u2022\-*•]\s/.test(t) || /^[0-9]+[.)]\s/.test(t) || isEmojiChar(t);
   842→  }).length;
   843→
   844→  // If 40%+ lines are already bullets, return as-is
   845→  if (nonEmpty.length > 2 && bulletCount / nonEmpty.length >= 0.4) return text;
   846→
   847→  // If only 1-2 non-empty lines total AND none are long paragraphs, keep as-is
   848→  if (nonEmpty.length <= 2) {
   849→    const hasLongLine = nonEmpty.some(l => l.trim().length > 150);
   850→    if (!hasLongLine) return text;
   851→  }
   852→
   853→  // Convert paragraph text into bullet-point format
   854→  const result: string[] = [];
   855→  let openerUsed = false;
   856→
   857→  for (const line of lines) {
   858→    const trimmed = line.trim();
   859→    if (!trimmed) {
   860→      result.push('');
   861→      continue;
   862→    }
   863→
   864→    // Already a bullet or numbered list item — keep
   865→    if (/^[\u2022\-*•]\s/.test(trimmed) || /^[0-9]+[.)]\s/.test(trimmed)) {
   866→      result.push(trimmed);
   867→      continue;
   868→    }
   869→
   870→    // Starts with emoji — keep
   871→    if (isEmojiChar(trimmed)) {
   872→      result.push(trimmed);
   873→      continue;
   874→    }
   875→
   876→    // Very short line (< 50 chars) — treat as opener or question
   877→    if (trimmed.length < 50) {
   878→      if (!openerUsed) {
   879→        result.push(trimmed);
   880→        openerUsed = true;
   881→      } else {
   882→        // Subsequent short lines get bullet treatment
   883→        const emoji = pickEmoji(trimmed);
   884→        result.push(`${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`);
   885→      }
   886→      continue;
   887→    }
   888→
   889→    // Long line: split into sentences and convert to bullets
   890→    const sentences = trimmed.match(/[^.!?]*[.!?]+/g) || [trimmed];
   891→
   892→    if (sentences.length === 1 && trimmed.length < 80) {
   893→      // Single short sentence — keep as is
   894→      result.push(trimmed);
   895→      continue;
   896→    }
   897→
   898→    if (sentences.length === 1) {
   899→      // Single long sentence — break on commas if possible
   900→      const parts = trimmed.split(/,/g).map(s => s.trim()).filter(Boolean);
   901→      if (parts.length >= 2) {
   902→        // First part as opener
   903→        result.push(parts[0]);
   904→        result.push('');
   905→        for (let i = 1; i < parts.length; i++) {
   906→          const p = parts[i].trim();
   907→          if (!p) continue;
   908→          const emoji = pickEmoji(p);
   909→          result.push(`${emoji} ${p.charAt(0).toUpperCase() + p.slice(1)}`);
   910→        }
   911→      } else {
   912→        const emoji = pickEmoji(trimmed);
   913→        result.push(`${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`);
   914→      }
   915→      continue;
   916→    }
   917→
   918→    // Multiple sentences — first as opener, rest as bullets
   919→    if (!openerUsed) {
   920→      result.push(sentences[0].trim());
   921→      result.push('');
   922→      openerUsed = true;
   923→    }
   924→
   925→    for (let i = 1; i < sentences.length; i++) {
   926→      const s = sentences[i].trim();
   927→      if (!s) continue;
   928→      const emoji = pickEmoji(s);
   929→      result.push(`${emoji} ${s.charAt(0).toUpperCase() + s.slice(1)}`);
   930→    }
   931→  }
   932→
   933→  return result.join('\n');
   934→}

   936→function pickEmoji(text: string): string {
   937→  const lower = text.toLowerCase();
   938→  if (lower.includes('save') || lower.includes('cost') || lower.includes('price') || lower.includes('$') || lower.includes('cheap') || lower.includes('afford')) return '💰';
   939→  if (lower.includes('automat') || lower.includes('ai') || lower.includes('robot') || lower.includes('handle') || lower.includes('resolv')) return '🤖';

  1147→function detectStage(message: string, session: any): string {
  1148→  const lower = message.toLowerCase();
  1149→  const ctx = session.context;
  1150→  const prevStage = session.detected_stage || ctx.detected_stage || 'welcome';
  1151→
  1152→  // Track stage history for transition detection
  1153→  if (!session.stage_history) session.stage_history = [];
  1154→
  1155→  // Phase 8a: Enhanced stage detection with history and nuanced stages
  1156→
  1157→  // Welcome — only in first messages
  1158→  if (session.message_count_today <= 1 && prevStage === 'welcome') return 'welcome';
  1159→
  1160→  // Verification — OTP/confirm (highest priority for active flows)
  1161→  if (lower.includes('verify') || lower.includes('otp') || lower.includes('confirm email')) return 'verification';
  1162→
  1163→  // Payment — active checkout intent
  1164→  if (lower.includes('pay') || lower.includes('checkout') || lower.includes('subscribe now') || lower.includes('complete purchase')) return 'payment';
  1165→
  1166→  // Bill review — checking invoice/bill details
  1167→  if (lower.includes('bill') || lower.includes('invoice') || lower.includes('receipt') || lower.includes('charge')) return 'bill_review';
  1168→
  1169→  // Handoff — requesting human transfer
  1170→  if (lower.includes('handoff') || lower.includes('transfer') || lower.includes('speak to human') || lower.includes('real person')) return 'handoff';
  1171→
  1172→  // Objection handling — user raising concerns
  1173→  const objectionPatterns = /(?:too (?:expensive|costly|pricey|much)|not sure|concern|worried|hesitat|risk|what if|scam|trust|reliable|safe)/i;
  1174→  if (objectionPatterns.test(lower)) return 'objection_handling';
  1175→
  1176→  // Variant selection — discussing specific features/variants
  1177→  const variantPatterns = /(?:variant|which (?:plan|one)|compare|difference between|mini parwa|parwa high|mini vs|parwa vs)/i;
  1178→  if (variantPatterns.test(lower) && (ctx.selected_variants?.length > 0 || lower.includes('variant') || lower.includes('compare'))) return 'variant_selection';
  1179→
  1180→  // Onboarding questions — asking about business specifics
  1181→  const onboardingPatterns = /(?:how many|team size|employees|tickets? (?:per day|daily|monthly)|channels?|current (?:setup|tool|system)|what (?:crm|helpdesk|platform))/i;
  1182→  if (onboardingPatterns.test(lower) && !ctx.industry) return 'onboarding_questions';
  1183→
  1184→  // Pricing — discussing plans/costs
  1185→  if (lower.includes('price') || lower.includes('pricing') || lower.includes('cost') || lower.includes('plan') || lower.includes('package') || lower.includes('how much')) return 'pricing';
  1186→
  1187→  // Demo — wanting to try/see
  1188→  if (lower.includes('demo') || lower.includes('try') || lower.includes('see it') || lower.includes('show me') || lower.includes('experience')) return 'demo';
  1189→
  1190→  // Discovery — learning about industry
  1191→  if (!ctx.industry && (lower.includes('ecommerce') || lower.includes('e-commerce') || lower.includes('saas') || lower.includes('logistics') || lower.includes('healthcare') || lower.includes('retail') || lower.includes('industry'))) return 'discovery';
  1192→
  1193→  // Default: maintain previous stage unless it was welcome (which we should advance from)
  1194→  if (prevStage === 'welcome') return 'discovery';
  1195→  return prevStage;
  1196→}

  1200→export async function POST(request: NextRequest, { params }: { params: Promise<{ path: string[] }> }) {
  1201→  const { path } = await params;
  1202→  const endpoint = path.join('/');
  1203→
  1204→  try {
  1205→    // ── PROXY FIRST: any /cc/* path goes to backend jarvis_cc router ──
  1206→    // The backend has /api/jarvis/cc/* mounted (jarvis_cc.py). The local
  1207→    // in-memory handlers below were built for the older /api/jarvis/session
  1208→    // style and don't know about /cc/session, /cc/message, etc.
  1209→    if (path[0] === 'cc') {
  1210→      const { response: proxyResponse } = await proxyToBackend(request, path);
  1211→      if (proxyResponse) return proxyResponse;
  1212→      // Backend unavailable — return a clean 502 instead of fake local data
  1213→      return NextResponse.json(
  1214→        { error: { code: 'backend_unavailable', message: 'Jarvis CC backend is unavailable. Please retry.', details: null } },
  1215→        { status: 502 }
  1216→      );
  1217→    }
  1218→
  1219→    // ── POST /session — Create Session ──────────────────────────
  1220→    if (endpoint === 'session') {
  1221→      const body = await request.json();
  1222→      const session = createDefaultSession(body.entry_source, body.entry_params);
  1223→
  1224→      // Phase 8b: Context-aware welcome based on entry_source
  1225→      const welcomeContent = getContextAwareWelcome(session.context.entry_source, session.context);
  1226→
  1227→      const welcomeMsg = {
  1228→        id: `jarvis_welcome_${Date.now()}`,
  1229→        session_id: session.id,
  1230→        role: 'jarvis',
  1231→        content: welcomeContent,
  1232→        message_type: 'text',
  1233→        metadata: { entry_source: session.context.entry_source },
  1234→        timestamp: new Date().toISOString(),
  1235→      };
  1236→      (session.messages as any[]).push(welcomeMsg);
  1237→      sessions.set(session.id, session);
  1238→      return NextResponse.json(session);
  1239→    }
  1240→
  1241→    // ── POST /message — Send Message & Get AI Reply ────────────
  1242→    if (endpoint === 'message') {
  1243→      // ── Try backend proxy first (LangGraph 13-stage pipeline + RAG + PostgreSQL) ──
  1244→      const { response: proxyResponse, consumedBody } = await proxyToBackend(request, path);
  1245→      console.log(`[Jarvis] Backend proxy ${proxyResponse ? 'succeeded' : 'failed, using local fallback'}`);
  1246→      if (proxyResponse) return proxyResponse;
  1247→
  1248→      // ── Local fallback: in-memory handling ──
  1249→      // IMPORTANT: Use consumedBody from proxyToBackend instead of reading request.json()
  1250→      // again — the body has already been consumed and reading it twice causes a 500 error.
  1251→      let body: Record<string, any>;
  1252→      try {
  1253→        body = consumedBody ? JSON.parse(consumedBody) : {};
  1254→      } catch {
  1255→        body = {};
  1256→      }
  1257→      const { content, session_id, context: incomingContext } = body;
  1258→
  1259→      let session = session_id ? sessions.get(session_id) : undefined;
  1260→      if (!session) {
  1261→        session = createDefaultSession('direct');
  1262→        sessions.set(session.id, session);
  1263→      }
  1264→
  1265→      // ── Merge incoming context from frontend BEFORE building AI response ──

  1717→export async function GET(request: NextRequest, { params }: { params: Promise<{ path: string[] }> }) {
  1718→  const { path } = await params;
  1719→  const endpoint = path.join('/');
  1720→  const url = new URL(request.url);
  1721→
  1722→  try {
  1723→    // ── PROXY FIRST: any /cc/* path goes to backend jarvis_cc router ──
  1724→    if (path[0] === 'cc') {
  1725→      const { response: proxyResponse } = await proxyToBackend(request, path);
  1726→      if (proxyResponse) return proxyResponse;
  1727→      return NextResponse.json(
  1728→        { error: { code: 'backend_unavailable', message: 'Jarvis CC backend is unavailable. Please retry.', details: null } },
  1729→        { status: 502 }
  1730→      );
  1731→    }
  1732→
  1733→    // ── GET /session ──────────────────────────────────────────
  1734→    if (endpoint === 'session') {
  1735→      const sessionId = url.searchParams.get('session_id');
  1736→      if (!sessionId || !sessions.has(sessionId)) {
  1737→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1738→      }
  1739→      return NextResponse.json(sessions.get(sessionId));
  1740→    }
  1741→
  1742→    // ── GET /history ───────────────────────────────────────────
  1743→    if (endpoint === 'history') {
  1744→      const sessionId = url.searchParams.get('session_id');
  1745→      const limit = parseInt(url.searchParams.get('limit') || '100', 10);
  1746→      const offset = parseInt(url.searchParams.get('offset') || '0', 10);
  1747→
  1748→      if (!sessionId || !sessions.has(sessionId)) {
  1749→        return NextResponse.json({ messages: [], total: 0, limit, offset, has_more: false });
  1750→      }
  1751→
  1752→      const session = sessions.get(sessionId)!;
  1753→      const allMessages = session.messages;
  1754→      const paged = allMessages.slice(offset, offset + limit);
  1755→      return NextResponse.json({ messages: paged, total: allMessages.length, limit, offset, has_more: offset + limit < allMessages.length });
  1756→    }
  1757→
  1758→    // ── GET /demo-pack/status ─────────────────────────────────
  1759→    if (endpoint === 'demo-pack/status') {
  1760→      const sessionId = url.searchParams.get('session_id');
  1761→      if (!sessionId || !sessions.has(sessionId)) {
  1762→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1763→      }
  1764→      const session = sessions.get(sessionId)!;
  1765→      return NextResponse.json({ pack_type: session.pack_type, remaining_today: session.remaining_today, total_allowed: session.pack_type === 'demo' ? 50 : 20, pack_expiry: session.pack_type === 'demo' ? new Date(Date.now() + 7 * 24 * 60 * 60 * 1000).toISOString() : null, demo_call_remaining: !session.context.demo_call_used });
  1766→    }
  1767→
  1768→    // ── GET /payment/status — Payment Status Check ───────────────
  1769→    if (endpoint === 'payment/status') {
  1770→      const sessionId = url.searchParams.get('session_id');
  1771→      if (!sessionId || !sessions.has(sessionId)) {
  1772→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1773→      }
  1774→      const session = sessions.get(sessionId)!;
  1775→      const paymentData = session.context.payment_data;
  1776→      return NextResponse.json({
  1777→        payment_status: session.payment_status,
  1778→        transaction_id: paymentData?.transaction_id || null,
  1779→        checkout_url: paymentData?.checkout_url || null,
  1780→        amount: paymentData?.total || 0,
  1781→        currency: paymentData?.currency || 'USD',
  1782→        items: paymentData?.items || [],
  1783→        subtotal: paymentData?.subtotal || 0,
  1784→        tax: paymentData?.tax || 0,
  1785→        created_at: paymentData?.created_at || null,
  1786→        completed_at: paymentData?.completed_at || null,
  1787→        bill_summary: session.context.bill_summary || null,
  1788→      });
  1789→    }
  1790→
  1791→    // ── GET /tickets — List Session Tickets ──────────────────────
  1792→    if (endpoint === 'tickets') {
  1793→      const sessionId = url.searchParams.get('session_id');
  1794→      if (!sessionId || !sessions.has(sessionId)) {
  1795→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1796→      }
  1797→      const session = sessions.get(sessionId)!;
  1798→      const tickets = session.context.action_tickets || [];
  1799→      const typeFilter = url.searchParams.get('type');
  1800→      const statusFilter = url.searchParams.get('status');
  1801→      const filtered = tickets.filter((t: any) => {
  1802→        if (typeFilter && t.type !== typeFilter) return false;
  1803→        if (statusFilter && t.status !== statusFilter) return false;
  1804→        return true;
  1805→      });
  1806→      return NextResponse.json({ tickets: filtered, total: filtered.length });
  1807→    }
  1808→
  1809→    // ── GET /tickets/:id — Get Specific Ticket ───────────────────
  1810→    if (endpoint.startsWith('tickets/') && endpoint.split('/').length === 2) {
  1811→      const ticketId = endpoint.split('/')[1];
  1812→      const sessionId = url.searchParams.get('session_id');
  1813→      if (!sessionId || !sessions.has(sessionId)) {
  1814→        return NextResponse.json({ error: { code: 'not_found', message: 'Session not found', details: null } }, { status: 404 });
  1815→      }
  1816→      const session = sessions.get(sessionId)!;
  1817→      const tickets = session.context.action_tickets || [];
  1818→      const ticket = tickets.find((t: any) => t.id === ticketId);
  1819→      if (!ticket) {
  1820→        return NextResponse.json({ error: { code: 'not_found', message: 'Ticket not found', details: null } }, { status: 404 });
  1821→      }
  1822→      return NextResponse.json(ticket);
  1823→    }
  1824→
  1825→    return NextResponse.json({ error: { code: 'not_found', message: `Unknown GET endpoint: /${endpoint}`, details: null } }, { status: 404 });
  1826→  } catch (error: unknown) {
  1827→    console.error('Jarvis API GET error:', error);
  1828→    const message = error instanceof Error ? error.message : 'Internal server error';
  1829→    return NextResponse.json({ error: { code: 'internal_error', message, details: null } }, { status: 500 });
  1830→  }
  1831→}

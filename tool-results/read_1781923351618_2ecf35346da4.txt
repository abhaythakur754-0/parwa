    80→def _get_or_create_progress(tenant_id: str) -> Dict[str, Any]:
    81→    """Get or create onboarding progress tracker for a tenant."""
    82→    if tenant_id not in _onboarding_progress:
    83→        _onboarding_progress[tenant_id] = {
    84→            "tenant_id": tenant_id,
    85→            "completed_steps": [],
    86→            "started_at": datetime.now(timezone.utc).isoformat(),
    87→        }
    88→    return _onboarding_progress[tenant_id]
    89→
    90→
    91→def _mark_step(tenant_id: str, step: int) -> None:
    92→    """Mark an onboarding step as completed."""
    93→    progress = _get_or_create_progress(tenant_id)
    94→    if step not in progress["completed_steps"]:
    95→        progress["completed_steps"].append(step)
    96→    progress["last_updated"] = datetime.now(timezone.utc).isoformat()

    99→def _require_tenant(tenant_id: str):
   100→    """Validate tenant exists and is active, or raise 404."""

   109→# ═══════════════════════════════════════════════════════════════
   110→# Auth Middleware (FastAPI Dependency)
   111→# ═══════════════════════════════════════════════════════════════
   112→
   113→async def get_auth_context(request: Request) -> Dict[str, Any]:
   114→    """FastAPI dependency that extracts and validates auth from request.
   115→
   116→    Checks Authorization header for:
   117→      - ``Bearer <jwt_token>``
   118→      - ``Bearer pk_live_...`` or ``Bearer pk_test_...``
   119→
   120→    Returns a context dict with tenant_id, user_id, auth_method, etc.
   121→    Raises HTTPException 401 if invalid.
   122→    """
   123→    auth_header = request.headers.get("Authorization", "")
   124→    if not auth_header.startswith("Bearer "):
   125→        raise _err(
   126→            "Missing or invalid Authorization header. "
   127→            "Use: Bearer <jwt_token> or Bearer <api_key>",
   128→            401,
   129→        )
   130→
   131→    token = auth_header[7:].strip()  # Strip "Bearer " prefix
   132→    if not token:
   133→        raise _err("Empty authorization token", 401)
   134→
   135→    # Determine token type
   136→    if token.startswith("pk_live_") or token.startswith("pk_test_"):
   137→        # API key authentication
   138→        is_valid, context, error_msg = await validate_request(
   139→            api_key=token,
   140→            client_ip=request.client.host if request.client else "unknown",
   141→        )
   142→    else:
   143→        # JWT authentication
   144→        is_valid, context, error_msg = await validate_request(
   145→            jwt_token=token,
   146→            client_ip=request.client.host if request.client else "unknown",
   147→        )
   148→
   149→    if not is_valid:
   150→        raise _err(error_msg or "Authentication failed", 401)
   151→
   152→    logger.debug(
   153→        "Auth context resolved: method=%s tenant=%s",
   154→        context.get("auth_method"),
   155→        context.get("tenant_id"),
   156→    )
   157→    return context

   202→@router.post(
   203→    "/onboarding/account-setup",
   204→    summary="Step 1 — Create tenant, admin user, initial API key and JWT",
   205→)
   206→async def account_setup(req: AccountSetupRequest):
   207→    """Create a new tenant account with admin user and initial credentials."""
   208→    # Validate unique email
   209→    if req.admin_email.lower() in _email_index:
   210→        raise _err(f"Email already registered: {req.admin_email}", 409)
   211→
   212→    # Generate slug from company name
   213→    slug = req.company_name.lower().replace(" ", "-").replace("_", "-")
   214→    slug = "".join(c for c in slug if c.isalnum() or c == "-")
   215→    slug = slug[:50].strip("-")
   216→
   217→    # Create tenant
   218→    try:
   219→        tenant = create_tenant(
   220→            name=req.company_name,
   221→            slug=slug,
   222→            tier="mini_parwa",
   223→            settings={
   224→                "industry": req.industry,
   225→                "company_size": req.company_size,
   226→                "onboarding_step": 1,
   227→            },
   228→        )
   229→    except ValueError as e:
   230→        raise _err(str(e), 409)
   231→
   232→    tenant_id = tenant["id"]
   233→
   234→    # Create admin user
   235→    user_id = str(uuid.uuid4())
   236→    password_hash = _hash_password(req.password)
   237→    admin_user = {
   238→        "user_id": user_id,
   239→        "email": req.admin_email.lower(),
   240→        "password_hash": password_hash,
   241→        "role": "admin",
   242→        "tenant_id": tenant_id,
   243→        "created_at": datetime.now(timezone.utc).isoformat(),
   244→    }
   245→    _users[user_id] = admin_user
   246→    _email_index[admin_user["email"]] = user_id
   247→
   248→    # Generate initial API key
   249→    key_result = register_key(
   250→        tenant_id=tenant_id,
   251→        key_type="live",
   252→        name=f"{req.company_name} — Initial Key",
   253→        user_id=user_id,
   254→    )
   255→
   256→    # Create JWT session
   257→    jwt_token = create_jwt_session(
   258→        tenant_id=tenant_id,
   259→        user_id=user_id,
   260→        extra_claims={"role": "admin", "email": req.admin_email},
   261→    )
   262→
   263→    # Track onboarding progress
   264→    _mark_step(tenant_id, 1)
   265→
   266→    logger.info(
   267→        "Account created: tenant=%s (%s) admin=%s",
   268→        tenant_id, req.company_name, req.admin_email,
   269→    )
   270→
   271→    return {
   272→        "tenant_id": tenant_id,
   273→        "tenant_name": req.company_name,
   274→        "admin_email": req.admin_email,
   275→        "api_key": key_result["full_key"],  # Shown ONCE
   276→        "jwt_token": jwt_token,
   277→        "tier": "mini_parwa",
   278→    }
   279→

   289→async def select_tier(req: SelectTierRequest):
   290→    """Select a tier (mini_parwa / parwa / parwa_high) and billing cycle."""
   291→    # Validate tier
   292→    valid_tiers = ("mini_parwa", "parwa", "parwa_high")
   293→    if req.tier not in valid_tiers:
   294→        raise _err(f"Invalid tier: '{req.tier}'. Must be one of: {valid_tiers}", 400)
   295→
   296→    # Validate billing cycle
   297→    valid_cycles = ("monthly", "annual")
   298→    if req.billing_cycle not in valid_cycles:
   299→        raise _err(
   300→            f"Invalid billing_cycle: '{req.billing_cycle}'. Must be: {valid_cycles}",
   301→            400,
   302→        )
   303→
   304→    # Require tenant exists
   305→    _require_tenant(req.tenant_id)
   306→
   307→    # Update tenant tier
   308→    current_settings = get_tenant(req.tenant_id).get("settings", {})
   309→    updated_settings = {
   310→        **current_settings,
   311→        "billing_cycle": req.billing_cycle,
   312→        "tier_selected_at": datetime.now(timezone.utc).isoformat(),
   313→    }
   314→
   315→    try:
   316→        tenant = update_tenant(
   317→            req.tenant_id,
   318→            tier=req.tier,
   319→            settings=updated_settings,
   320→        )
   321→    except ValueError as e:
   322→        raise _err(str(e), 400)
   323→
   324→    _mark_step(req.tenant_id, 2)
   325→
   326→    capabilities = get_tier_permissions(req.tier)
   327→
   328→    logger.info("Tier selected: tenant=%s tier=%s cycle=%s", req.tenant_id, req.tier, req.billing_cycle)
   329→
   330→    return {
   331→        "tenant_id": req.tenant_id,
   332→        "tier": req.tier,
   333→        "capabilities": capabilities,
   334→        "billing_cycle": req.billing_cycle,
   335→    }

   343→    "/onboarding/connect-integration",

   346→async def connect_integration(req: ConnectIntegrationRequest):
   347→    """Connect an external integration (Shopify, Zendesk, Gorgias, etc.)."""
   348→    _require_tenant(req.tenant_id)
   349→
   350→    tenant = get_tenant(req.tenant_id)
   351→    settings = tenant.get("settings", {})
   352→
   353→    # Initialize integrations list if not present
   354→    integrations: List[Dict[str, Any]] = settings.get("integrations", [])
   355→
   356→    # Check for duplicate integration type
   357→    for existing in integrations:
   358→        if existing.get("integration_type") == req.integration_type:
   359→            raise _err(
   360→                f"Integration '{req.integration_type}' is already connected. "
   361→                "Disconnect it first or use a different type.",
   362→                409,
   363→            )
   364→
   365→    integration_entry = {
   366→        "integration_id": str(uuid.uuid4()),
   367→        "integration_type": req.integration_type,
   368→        "status": "connected",
   369→        "health": "healthy",
   370→        "credentials": req.credentials,
   371→        "connected_at": datetime.now(timezone.utc).isoformat(),
   372→        "last_health_check": datetime.now(timezone.utc).isoformat(),
   373→    }
   374→
   375→    integrations.append(integration_entry)
   376→    settings["integrations"] = integrations
   377→
   378→    update_tenant(req.tenant_id, settings=settings)
   379→    _mark_step(req.tenant_id, 3)
   380→
   381→    logger.info(
   382→        "Integration connected: tenant=%s type=%s",
   383→        req.tenant_id, req.integration_type,
   384→    )
   385→
   386→    return {
   387→        "tenant_id": req.tenant_id,
   388→        "integration_type": req.integration_type,
   389→        "status": "connected",
   390→        "health": "healthy",
   391→    }

   394→@router.get(
   395→    "/onboarding/integrations",
   396→    summary="Step 3 — List connected integrations",
   397→)
   398→async def list_integrations(tenant_id: str = Query(...)):
   399→    """Return all connected integrations for a tenant."""
   400→    _require_tenant(tenant_id)
   401→
   402→    tenant = get_tenant(tenant_id)
   403→    settings = tenant.get("settings", {})
   404→    integrations = settings.get("integrations", [])
   405→
   406→    # Sanitize — remove raw credentials from the response
   407→    safe_integrations = []
   408→    for integ in integrations:
   409→        safe = {
   410→            k: v for k, v in integ.items()
   411→            if k != "credentials"
   412→        }
   413→        safe["has_credentials"] = bool(integ.get("credentials"))
   414→        safe_integrations.append(safe)
   415→
   416→    return {
   417→        "tenant_id": tenant_id,
   418→        "integrations": safe_integrations,
   419→        "count": len(safe_integrations),
   420→    }

   654→    "/onboarding/status",
   655→    summary="Step 6 — Get full onboarding status and readiness score",
   656→)
   657→async def onboarding_status(tenant_id: str = Query(...)):
   658→    """Return the current onboarding progress, tenant info, and readiness score."""
   659→    _require_tenant(tenant_id)
   660→
   661→    tenant = get_tenant(tenant_id)
   662→    settings = tenant.get("settings", {})
   663→    progress = _get_or_create_progress(tenant_id)
   664→
   665→    completed_steps = sorted(progress.get("completed_steps", []))
   666→
   667→    # Count resources
   668→    integration_count = len(settings.get("integrations", []))
   669→    kb_count = len(settings.get("knowledge_base", []))
   670→    policies = settings.get("policies", {})
   671→    policy_count = len(policies)
   672→
   673→    # Calculate readiness score (0–100)
   674→    # Each completed step is worth ~16 points (6 steps = 96, bonus for resources)
   675→    step_score = len(completed_steps) * 16
   676→    resource_bonus = 0
   677→
   678→    if integration_count > 0:
   679→        resource_bonus += 1
   680→    if kb_count >= 3:
   681→        resource_bonus += 1
   682→    if policy_count >= 3:
   683→        resource_bonus += 2
   684→
   685→    readiness_score = min(100, step_score + resource_bonus)
   686→
   687→    # Determine current step
   688→    current_step = (max(completed_steps) + 1) if completed_steps else 1
   689→    if current_step > 6:
   690→        current_step = 6
   691→
   692→    logger.debug(
   693→        "Onboarding status: tenant=%s steps=%s readiness=%d",
   694→        tenant_id, completed_steps, readiness_score,
   695→    )
   696→
   697→    return {
   698→        "step": current_step,
   699→        "completed_steps": completed_steps,
   700→        "tenant": {
   701→            "tenant_id": tenant_id,
   702→            "name": tenant.get("name"),
   703→            "tier": tenant.get("tier"),
   704→            "status": tenant.get("status"),
   705→        },
   706→        "integration_count": integration_count,
   707→        "kb_count": kb_count,
   708→        "policy_count": policy_count,
   709→        "readiness_score": readiness_score,
   710→    }

   717→@router.post(
   718→    "/auth/login",
   719→    summary="Authenticate with email + password, receive JWT",
   720→)
   721→async def login(req: LoginRequest):
   722→    """Verify credentials and return a JWT session token."""
   723→    email = req.email.lower().strip()
   724→    user_id = _email_index.get(email)
   725→
   726→    if not user_id:
   727→        raise _err("Invalid email or password", 401)
   728→
   729→    user = _users.get(user_id)
   730→    if not user:
   731→        raise _err("Invalid email or password", 401)
   732→
   733→    # Verify password
   734→    if not _verify_password(req.password, user["password_hash"]):
   735→        raise _err("Invalid email or password", 401)
   736→
   737→    # Check tenant is active
   738→    tenant = get_tenant(user["tenant_id"])
   739→    if not tenant:
   740→        raise _err("Tenant not found. Contact support.", 401)
   741→    if tenant["status"] != "active":
   742→        raise _err(f"Account suspended. Status: {tenant['status']}", 403)
   743→
   744→    # Generate JWT
   745→    jwt_token = create_jwt_session(
   746→        tenant_id=user["tenant_id"],
   747→        user_id=user_id,
   748→        extra_claims={"role": user["role"], "email": email},
   749→    )
   750→
   751→    # Find the first active API key for this tenant (for dashboard use)
   752→    tenant_keys = list_keys(user["tenant_id"])
   753→    active_key_prefix = tenant_keys[0]["key_prefix"] if tenant_keys else ""
   754→
   755→    logger.info("Login successful: email=%s tenant=%s", email, user["tenant_id"])
   756→
   757→    return {
   758→        "jwt_token": jwt_token,
   759→        "tenant_id": user["tenant_id"],
   760→        "company_name": tenant.get("name", ""),
   761→        "user_email": email,
   762→        "role": user["role"],
   763→        "tier": tenant.get("tier", ""),
   764→        "api_key": active_key_prefix,  # Safe prefix only; full key never returned on login
   765→    }

   277→   277→def api_industry_change_impact(

   280→   280→    db: Session = Depends(get_db),
   281→   281→) -> IndustryChangeImpactResponse:
   282→   282→    """Check the impact of changing industry on existing integrations.
   283→   283→
   284→   284→    Per GAP 10: Industry change NEVER disconnects integrations.
   285→   285→    Existing connections STAY connected. Only the suggestion filter changes.
   286→   286→    Clients can always connect tools outside their industry.
   287→   287→
   288→   288→    Returns:
   289→   289→    - still_recommended: integrations that are suggested in BOTH industries
   290→   290→    - no_longer_suggested: connected integrations not in new industry's suggestions
   291→   291→    - newly_suggested: integrations newly suggested for the new industry
   292→   292→    """
   293→   293→    # Get suggested integration keys for both industries
   294→   294→    current_suggested = {i["key"] for i in get_catalog_for_industry(body.current_industry)}
   295→   295→    new_suggested = {i["key"] for i in get_catalog_for_industry(body.new_industry)}
   296→   296→
   297→   297→    # Get currently connected integrations
   298→   298→    service = IntegrationService(db)
   299→   299→    connected = service.get_integrations(company_id=user.company_id, active_only=True)
   300→   300→    connected_keys = {i["type"] for i in connected}
   301→   301→
   302→   302→    # Categorize
   303→   303→    still_recommended = sorted(connected_keys & new_suggested)
   304→   304→    no_longer_suggested = sorted(connected_keys - new_suggested)
   305→   305→    newly_suggested = sorted((new_suggested - current_suggested) - connected_keys)
   306→   306→
   307→   307→    # Build connected integration details
   308→   308→    connected_details = []
   309→   309→    for i in connected:
   310→   310→        catalog_entry = get_integration_by_key(i["type"])
   311→   311→        in_new_industry = i["type"] in new_suggested
   312→   312→        connected_details.append({
   313→   313→            "key": i["type"],
   314→   314→            "name": i["name"],
   315→   315→            "status": i["status"],
   316→   316→            "category": catalog_entry.category.value if catalog_entry else "custom",
   317→   317→            "in_new_industry_suggestions": in_new_industry,
   318→   318→        })
   319→   319→
   320→   320→    # Build message
   321→   321→    if no_longer_suggested:
   322→   322→        msg = (
   323→   323→            f"You're changing from {body.current_industry} to {body.new_industry}. "
   324→   324→            f"{len(no_longer_suggested)} integration(s) will still work but are no longer suggested for {body.new_industry}. "
   325→   325→            f"Your tickets, knowledge base, and billing are NOT affected."
   326→   326→        )
   327→   327→    else:
   328→   328→        msg = (
   329→   329→            f"Changing from {body.current_industry} to {body.new_industry} will not affect any of your "
   330→   330→            f"connected integrations. {len(newly_suggested)} new integrations are suggested for {body.new_industry}."
   331→   331→        )
   332→   332→
   333→   333→    return IndustryChangeImpactResponse(
   334→   334→        new_industry=body.new_industry,
   335→   335→        current_industry=body.current_industry,
   336→   336→        connected_integrations=connected_details,
   337→   337→        still_recommended=still_recommended,
   338→   338→        no_longer_suggested=no_longer_suggested,
   339→   339→        newly_suggested=newly_suggested,
   340→   340→        message=msg,
   341→   341→    )

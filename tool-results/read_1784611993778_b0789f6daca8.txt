    41→function formatRelativeDate(iso: string | null): string {
    42→  if (!iso) return '—';
    43→  const d = new Date(iso); const now = new Date(); const diffMs = d.getTime() - now.getTime();
    44→  const days = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
    45→  if (days < 0) return `${Math.abs(days)}d ago`;
    46→  if (days === 0) return 'today'; if (days === 1) return 'tomorrow';
    47→  if (days < 30) return `in ${days}d`;
    48→  return formatDate(iso);
    49→}

   329→# ═══════════════════════════════════════════════════════════════
   330→# IN-MEMORY BACKEND (works immediately, no deps)
   331→# ═══════════════════════════════════════════════════════════════
   332→
   333→class InMemoryBackend(StorageBackend):
   334→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   335→
   336→    def __init__(self):
   337→        self._notifications: Dict[str, Dict] = {}
   338→        self._flags: List[Dict] = []
   339→        self._audit: List[Dict] = []
   340→        self._quality_scores: List[Dict] = []
   341→        self._batches: Dict[str, List[Dict]] = {}
   342→        self._batch_metas: Dict[str, Dict] = {}
   343→        self._lock = threading.Lock()
   344→        self._next_nfy_id = 1
   345→        # Wave 2 stores
   346→        self._integration_pings: List[Dict] = []
   347→        self._llm_costs: List[Dict] = []
   348→        self._stuck_ticket_events: List[Dict] = []
   349→        self._load_status: Dict[str, Dict] = {}  # tenant_id -> {variant -> {concurrent, max_concurrent}}
   350→
   351→    # ── Helpers ───────────────────────────────────────────────
   352→
   353→    def _next_notification_number(self) -> int:
   354→        with self._lock:
   355→            n = self._next_nfy_id
   356→            self._next_nfy_id += 1
   357→            return n
   358→
   359→    @staticmethod
   360→    def _priority_from_score(score: float) -> str:
   361→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:
   362→            return PRIORITY_CRITICAL
   363→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_HIGH]:
   364→            return PRIORITY_HIGH
   365→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_MEDIUM]:
   366→            return PRIORITY_MEDIUM
   367→        return PRIORITY_LOW
   368→
   369→    # ── Notifications ────────────────────────────────────────
   370→
   371→    async def create_notification(
   372→        self,
   373→        tenant_id: str,
   374→        ntype: str,
   375→        priority_score: float,
   376→        title: str,
   377→        description: str,
   378→        related_tickets: Optional[List[str]] = None,
   379→        batch_key: Optional[str] = None,
   380→        source_data: Optional[Dict] = None,
   381→    ) -> Dict[str, Any]:
   382→        num = self._next_notification_number()
   383→        key = f"PARWA-NFY-{num:03d}"
   384→        now = time.time()
   385→        nf = {
   386→            "id": str(uuid.uuid4()),
   387→            "tenant_id": tenant_id,
   388→            "notification_key": key,
   389→            "type": ntype,
   390→            "priority": self._priority_from_score(priority_score),
   391→            "priority_score": round(priority_score, 4),
   392→            "title": title,
   393→            "description": description,
   394→            "related_tickets": related_tickets or [],
   395→            "batch_key": batch_key,
   396→            "source_data": source_data or {},
   397→            "is_read": False,
   398→            "is_resolved": False,
   399→            "created_at": now,
   400→            "resolved_at": None,
   401→            "backend": "memory",
   402→        }
   403→        self._notifications[key] = nf
   404→        logger.info("NFY created: %s type=%s score=%.3f [%s]", key, ntype, priority_score, tenant_id)
   405→        return nf
   406→
   407→    async def get_notification(self, key: str) -> Optional[Dict[str, Any]]:
   408→        return self._notifications.get(key)
   409→
   410→    async def get_notifications(
   411→        self,
   412→        tenant_id: str,
   413→        include_resolved: bool = False,
   414→        min_priority: Optional[str] = None,
   415→    ) -> List[Dict[str, Any]]:
   416→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   417→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   418→        results = []
   419→        for n in self._notifications.values():
   420→            if n["tenant_id"] != tenant_id:
   421→                continue
   422→            if not include_resolved and n["is_resolved"]:
   423→                continue
   424→            if priority_order.get(n["priority"], 99) > min_rank:
   425→                continue
   426→            results.append(n)
   427→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   428→        return results
   429→
   430→    async def resolve_notification(self, key: str) -> bool:
   431→        n = self._notifications.get(key)
   432→        if n and not n["is_resolved"]:
   433→            n["is_resolved"] = True
   434→            n["resolved_at"] = time.time()
   435→            return True
   436→        return False
   437→
   438→    async def dismiss_notification(self, key: str) -> bool:
   439→        n = self._notifications.get(key)
   440→        if n:
   441→            n["is_read"] = True
   442→            return True
   443→        return False
   444→
   445→    # ── System Flags ─────────────────────────────────────────
   446→
   447→    async def set_flag(
   448→        self,
   449→        tenant_id: str,
   450→        flag_type: str,
   451→        flag_value: str,
   452→        set_by: str,
   453→        scope: str = "global",
   454→        target_id: Optional[str] = None,
   455→        reason: Optional[str] = None,
   456→        expires_at: Optional[str] = None,
   457→    ) -> Dict[str, Any]:
   458→        flag = {
   459→            "id": str(uuid.uuid4()),
   460→            "tenant_id": tenant_id,
   461→            "flag_type": flag_type,
   462→            "flag_value": flag_value,
   463→            "scope": scope,
   464→            "target_id": target_id,
   465→            "set_by": set_by,
   466→            "reason": reason,
   467→            "expires_at": expires_at,
   468→            "created_at": datetime.now(timezone.utc).isoformat(),
   469→            "revoked_at": None,
   470→            "backend": "memory",
   471→        }
   472→        self._flags.append(flag)
   473→        logger.info("FLAG set: %s=%s scope=%s [%s]", flag_type, flag_value, scope, tenant_id)
   474→        return flag
   475→
   476→    async def get_active_flags(
   477→        self, tenant_id: str, flag_type: Optional[str] = None
   478→    ) -> List[Dict[str, Any]]:
   479→        now = datetime.now(timezone.utc)
   480→        active = []
   481→        for f in self._flags:
   482→            if f["tenant_id"] != tenant_id:
   483→                continue
   484→            if f["revoked_at"] is not None:
   485→                continue
   486→            if f["expires_at"]:
   487→                exp = datetime.fromisoformat(f["expires_at"]) if isinstance(f["expires_at"], str) else f["expires_at"]
   488→                if exp < now:
   489→                    continue
   490→            if flag_type and f["flag_type"] != flag_type:
   491→                continue
   492→            active.append(f)
   493→        return active
   494→
   495→    async def revoke_flag(self, flag_id: str, revoked_by: str) -> bool:
   496→        for f in self._flags:
   497→            if f["id"] == flag_id and f["revoked_at"] is None:
   498→                f["revoked_at"] = datetime.now(timezone.utc).isoformat()
   499→                logger.info("FLAG revoked: %s by %s", flag_id, revoked_by)
   500→                return True
   501→        return False
   502→
   503→    # ── Audit Trail ──────────────────────────────────────────
   504→
   505→    async def create_audit_entry(
   506→        self,
   507→        tenant_id: str,
   508→        action: str,
   509→        actor_email: str,
   510→        target_type: Optional[str] = None,
   511→        target_id: Optional[str] = None,
   512→        payload: Optional[Dict] = None,
   513→    ) -> Dict[str, Any]:
   514→        payload_json = json.dumps(payload or {}, sort_keys=True)
   515→        current_hash = hashlib.sha256(
   516→            f"{tenant_id}:{action}:{payload_json}:{time.time()}".encode()
   517→        ).hexdigest()[:64]
   518→
   519→        entry = {
   520→            "id": len(self._audit) + 1,
   521→            "tenant_id": tenant_id,
   522→            "action": action,
   523→            "actor_email": actor_email,
   524→            "target_type": target_type,
   525→            "target_id": target_id,
   526→            "payload": payload or {},
   527→            "current_hash": current_hash,
   528→            "previous_hash": self._audit[-1]["current_hash"] if self._audit else None,
   529→            "created_at": datetime.now(timezone.utc).isoformat(),
   530→            "backend": "memory",
   531→        }
   532→        self._audit.append(entry)
   533→        return entry
   534→
   535→    async def get_audit_trail(
   536→        self,
   537→        tenant_id: str,
   538→        action: Optional[str] = None,
   539→        limit: int = 50,
   540→    ) -> List[Dict[str, Any]]:
   541→        results = [e for e in self._audit if e["tenant_id"] == tenant_id]
   542→        if action:
   543→            results = [e for e in results if e["action"] == action]
   544→        results.sort(key=lambda x: x["id"], reverse=True)
   545→        return results[:limit]

   547→    # ── Quality Scores ───────────────────────────────────────
   548→
   549→    async def write_quality_score(
   550→        self,
   551→        tenant_id: str,
   552→        ticket_id: str,
   553→        overall_score: float,
   554→        confidence_score: Optional[float] = None,
   555→        resolution_path: Optional[str] = None,
   556→        nodes_reached: Optional[List[str]] = None,
   557→        llm_calls: int = 0,
   558→        tokens_used: int = 0,
   559→        model_used: Optional[str] = None,
   560→        agent_id: Optional[str] = None,
   561→    ) -> Dict[str, Any]:
   562→        score = {
   563→            "id": str(uuid.uuid4()),
   564→            "tenant_id": tenant_id,
   565→            "agent_id": agent_id,
   566→            "ticket_id": ticket_id,
   567→            "overall_score": round(overall_score, 2),
   568→            "confidence_score": round(confidence_score, 2) if confidence_score else None,
   569→            "resolution_path": resolution_path,
   570→            "nodes_reached": nodes_reached or [],
   571→            "llm_calls": llm_calls,
   572→            "tokens_used": tokens_used,
   573→            "model_used": model_used,
   574→            "created_at": datetime.now(timezone.utc).isoformat(),
   575→            "backend": "memory",
   576→        }
   577→        self._quality_scores.append(score)
   578→        return score
   579→
   580→    async def get_quality_stats(self, tenant_id: str, days: int = 7) -> Dict[str, Any]:
   581→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   582→        total = len(tenant_scores)
   583→        if total == 0:
   584→            return {
   585→                "tenant_id": tenant_id, "total_tickets": 0,
   586→                "avg_quality": 0, "avg_confidence": 0,
   587→                "auto_resolved": 0, "escalated": 0, "stuck": 0,
   588→            }
   589→        avg_q = sum(s["overall_score"] for s in tenant_scores) / total
   590→        avg_c_vals = [s["confidence_score"] for s in tenant_scores if s.get("confidence_score")]
   591→        avg_c = sum(avg_c_vals) / len(avg_c_vals) if avg_c_vals else 0
   592→        paths = {}
   593→        for s in tenant_scores:
   594→            p = s.get("resolution_path", "unknown")
   595→            paths[p] = paths.get(p, 0) + 1

   597→            "tenant_id": tenant_id,
   598→            "total_tickets": total,
   599→            "avg_quality": round(avg_q, 4),

   607→    # ── Batch Queue ──────────────────────────────────────────
   608→
   609→    async def add_to_batch(
   610→        self,
   611→        tenant_id: str,
   612→        batch_key: str,
   613→        ticket_id: str,
   614→        confidence: float,
   615→    ) -> Optional[Dict[str, Any]]:
   616→        BATCH_WINDOW_S = 300
   617→        now = time.time()
   618→
   619→        if tenant_id not in self._batch_metas:
   620→            self._batch_metas[tenant_id] = {}
   621→
   622→        # Find existing batch with same key within window
   623→        existing_key = None
   624→        for bk, meta in self._batch_metas[tenant_id].items():
   625→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   626→                existing_key = bk
   627→                break
   628→
   629→        if existing_key:
   630→            meta = self._batch_metas[tenant_id][existing_key]
   631→            meta["ticket_ids"].append(ticket_id)
   632→            meta["signal_count"] = len(meta["ticket_ids"])
   633→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   634→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   635→            # Don't flush yet — still within window
   636→            return None
   637→        else:
   638→            meta = {
   639→                "batch_key": batch_key,
   640→                "tenant_id": tenant_id,
   641→                "batch_start": now,
   642→                "ticket_ids": [ticket_id],
   643→                "signal_count": 1,
   644→                "confidence_min": confidence,
   645→                "confidence_max": confidence,
   646→                "status": "pending",
   647→            }
   648→            self._batch_metas[tenant_id][batch_key] = meta
   649→            return None

   700→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
   701→        """Compute health from recent pings (last 30 per service)."""
   702→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
   703→        services: Dict[str, List[Dict]] = {}
   704→        for p in tenant_pings:
   705→            services.setdefault(p["service_name"], []).append(p)
   706→
   707→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
   708→        for svc, pings in services.items():
   709→            recent = pings[-30:]  # last 30 pings
   710→            if not recent:
   711→                continue
   712→            healthy = sum(1 for p in recent if p["is_healthy"])
   713→            total = len(recent)
   714→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
   715→            last_ping = recent[-1]
   716→            avg_ms = (sum(p["response_ms"] or 0 for p in recent) / total) if total else 0
   717→            last_error = None
   718→            last_error_time = None
   719→            for p in reversed(recent):
   720→                if not p["is_healthy"]:
   721→                    last_error = p.get("error_detail")
   722→                    last_error_time = p["created_at"]
   723→                    break
   724→
   725→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
   726→            if status != "healthy":
   727→                result["degraded_count"] += 1
   728→            else:
   729→                result["healthy_count"] += 1
   730→
   731→            result["services"][svc] = {
   732→                "status": status,
   733→                "uptime_pct": uptime_pct,
   734→                "total_pings": total,
   735→                "healthy_pings": healthy,
   736→                "avg_response_ms": round(avg_ms, 1),
   737→                "last_ping_at": last_ping["created_at"],
   738→                "last_healthy_at": next(
   739→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
   740→                    None,
   741→                ),
   742→                "last_error": last_error,
   743→                "last_error_at": last_error_time,
   744→                "last_status_code": last_ping.get("status_code"),
   745→            }
   746→        return result

   828→    # ── Drift Detection (Wave 2) ─────────────────────────────
   829→
   830→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
   831→        """Analyze quality_scores for drift using day-over-day comparison."""
   832→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   833→        if len(tenant_scores) < 3:
   834→            return {
   835→                "tenant_id": tenant_id, "drift_detected": False,
   836→                "drift_severity": "none", "accuracy_7d": None,
   837→                "accuracy_yesterday": None, "accuracy_today": None,
   838→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
   839→                "total_scores": len(tenant_scores),
   840→            }
   841→        # Group by date (ISO date portion)
   842→        by_date: Dict[str, List[float]] = {}
   843→        for s in tenant_scores:
   844→            date_str = s["created_at"][:10]  # YYYY-MM-DD
   845→            by_date.setdefault(date_str, []).append(s["overall_score"])
   846→
   847→        dates_sorted = sorted(by_date.keys())
   848→
   849→        def _avg(scores: List[float]) -> float:
   850→            return round(sum(scores) / len(scores), 4) if scores else 0.0
   851→
   852→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
   853→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
   854→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
   855→
   856→        # Drift detection: >5% drop for 3+ days
   857→        drift_detected = False
   858→        drift_severity = "none"
   859→        trigger_reason = None
   860→        trend_direction = "stable"
   861→
   862→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
   863→            recent_3 = dates_sorted[-3:]
   864→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
   865→            # Check if declining 3+ consecutive days
   866→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
   867→            drop_7d = accuracy_7d - accuracy_today
   868→
   869→            if declining and drop_7d > 0.05:
   870→                drift_detected = True
   871→                trend_direction = "declining"
   872→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
   873→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
   874→            elif drop_7d > 0.03:
   875→                drift_detected = True
   876→                trend_direction = "slight_decline"
   877→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
   878→                drift_severity = "warning"
   879→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
   880→                trend_direction = "improving"
   881→            else:
   882→                trend_direction = "stable"
   883→
   884→        # Check for repeated errors (same low score pattern)
   885→        if not drift_detected and tenant_scores:
   886→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
   887→            if len(low_scores) >= 3:
   888→                # Check if same resolution_path failing
   889→                paths: Dict[str, int] = {}
   890→                for s in low_scores:
   891→                    p = s.get("resolution_path", "unknown")
   892→                    paths[p] = paths.get(p, 0) + 1
   893→                worst_path = max(paths.items(), key=lambda x: x[1])
   894→                if worst_path[1] >= 3:
   895→                    drift_detected = True
   896→                    drift_severity = "warning"
   897→                    trend_direction = "declining"
   898→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
   899→
   900→        return {
   901→            "tenant_id": tenant_id,
   902→            "drift_detected": drift_detected,
   903→            "drift_severity": drift_severity,
   904→            "accuracy_7d": accuracy_7d,
   905→            "accuracy_yesterday": accuracy_yesterday,
   906→            "accuracy_today": accuracy_today,
   907→            "trend_direction": trend_direction,
   908→            "trigger_reason": trigger_reason,
   909→            "total_scores": len(tenant_scores),
   910→        }

  1002→
  1003→class SupabaseBackend(StorageBackend):
  1004→    """Supabase storage via REST API (PostgREST). Uses httpx.
  1005→
  1006→    Requires env vars: SUPABASE_URL, SUPABASE_ANON_KEY
  1007→    """
  1008→
  1009→    def __init__(self):
  1010→        if not SUPABASE_URL or not SUPABASE_ANON_KEY:
  1011→            raise ValueError("SUPABASE_URL and SUPABASE_ANON_KEY must be set")
  1012→        self._url = SUPABASE_URL.rstrip("/")
  1013→        self._headers = {
  1014→            "apikey": SUPABASE_ANON_KEY,
  1015→            "Authorization": f"Bearer {SUPABASE_ANON_KEY}",
  1016→            "Content-Type": "application/json",
  1017→            "Prefer": "return=representation",
  1018→        }
  1019→        self._client = None  # Lazy init on first use
  1020→
  1021→    async def _get_client(self):
  1022→        if self._client is None or self._client.is_closed:
  1023→            import httpx
  1024→            self._client = httpx.AsyncClient(
  1025→                base_url=self._url,
  1026→                headers=self._headers,
  1027→                timeout=15.0,
  1028→            )
  1029→        return self._client
  1030→
  1031→    async def _rest_get(self, table: str, query: str = "") -> List[Dict]:
  1032→        client = await self._get_client()
  1033→        url = f"/rest/v1/{table}"
  1034→        if query:
  1035→            url += f"?{query}"
  1036→        r = await client.get(url)
  1037→        r.raise_for_status()
  1038→        return r.json() if r.text else []
  1039→
  1040→    async def _rest_post(self, table: str, data: Dict) -> Dict:
  1041→        client = await self._get_client()
  1042→        r = await client.post(f"/rest/v1/{table}", json=data)
  1043→        r.raise_for_status()
  1044→        result = r.json()
  1045→        if isinstance(result, list) and result:
  1046→            return result[0]
  1047→        return result
  1048→
  1049→    async def _rest_patch(self, table: str, filter_query: str, data: Dict) -> List[Dict]:
  1050→        client = await self._get_client()
  1051→        r = await client.patch(f"/rest/v1/{table}?{filter_query}", json=data)
  1052→        r.raise_for_status()
  1053→        return r.json() if r.text else []
  1054→
  1055→    async def close(self):
  1056→        if self._client and not self._client.is_closed:
  1057→            await self._client.aclose()

  1482→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1483→        # Fetch scores + client-side drift analysis (same logic as InMemory)
  1484→        scores = await self._rest_get(
  1485→            TABLE_QUALITY_SCORES,
  1486→            f"tenant_id=eq.{tenant_id}&select=overall_score,resolution_path,created_at&order=created_at.asc&limit=5000"
  1487→        )
  1488→        if len(scores) < 3:
  1489→            return {"tenant_id": tenant_id, "drift_detected": False, "drift_severity": "none",
  1490→                    "accuracy_7d": None, "accuracy_yesterday": None, "accuracy_today": None,
  1491→                    "trend_direction": "stable", "trigger_reason": "insufficient_data", "total_scores": len(scores)}
  1492→        by_date: Dict[str, List[float]] = {}
  1493→        for s in scores:
  1494→            d = (s.get("created_at") or "")[:10]
  1495→            by_date.setdefault(d, []).append(float(s.get("overall_score", 0)))
  1496→        dates_sorted = sorted(by_date.keys())
  1497→        def _avg(vals): return round(sum(vals) / len(vals), 4) if vals else 0.0
  1498→        accuracy_7d = _avg([v for d in dates_sorted[-7:] for v in by_date[d]])
  1499→        accuracy_today = _avg(by_date[dates_sorted[-1]])
  1500→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1501→        drift_detected = False; drift_severity = "none"; trigger_reason = None; trend_direction = "stable"
  1502→        if len(dates_sorted) >= 3:
  1503→            daily_avgs = [_avg(by_date[d]) for d in dates_sorted[-3:]]
  1504→            declining = all(daily_avgs[i] > daily_avgs[i+1] for i in range(len(daily_avgs)-1))
  1505→            drop = accuracy_7d - accuracy_today
  1506→            if declining and drop > 0.05:
  1507→                drift_detected = True; trend_direction = "declining"
  1508→                trigger_reason = f"accuracy_dropped_{drop:.1%}_over_3_days"
  1509→                drift_severity = "critical" if drop > 0.10 else "warning"
  1510→            elif drop > 0.03:
  1511→                drift_detected = True; trend_direction = "slight_decline"
  1512→                trigger_reason = f"accuracy_dropped_{drop:.1%}_from_7d_avg"; drift_severity = "warning"
  1513→            elif accuracy_today > accuracy_7d + 0.03:
  1514→                trend_direction = "improving"
  1515→        return {"tenant_id": tenant_id, "drift_detected": drift_detected, "drift_severity": drift_severity,
  1516→                "accuracy_7d": accuracy_7d, "accuracy_yesterday": accuracy_yesterday,
  1517→                "accuracy_today": accuracy_today, "trend_direction": trend_direction,
  1518→                "trigger_reason": trigger_reason, "total_scores": len(scores)}

   121→   121→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   122→   122→
   123→   123→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   124→   124→  agentsReplaced: number;
   125→   125→  humanCost: number;
   126→   126→  savings: number;
   127→   127→  savingsPercent: number;
   128→   128→} {
   129→   129→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   130→   130→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   131→   131→  const savings = Math.max(0, humanCost - monthlyCost);
   132→   132→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   133→   133→  return { agentsReplaced, humanCost, savings, savingsPercent };
   134→   134→}

   136→   136→// ── Usage Bar Component ────────────────────────────────────────────────

   138→   138→function UsageBar({
   139→   139→  used,
   140→   140→  total,
   141→   141→  label,
   142→   142→  unit,
   143→   143→  overageRate,
   144→   144→}: {
   145→   145→  used: number;
   146→   146→  total: number;
   147→   147→  label: string;
   148→   148→  unit: string;
   149→   149→  overageRate?: string;
   150→   150→}) {
   151→   151→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
   152→   152→  const isOverLimit = used > total;
   153→   153→
   154→   154→  return (
   155→   155→    <div className="space-y-1.5">
   156→   156→      <div className="flex items-center justify-between">
   157→   157→        <span className="text-[10px] text-orange-200/40">{label}</span>
   158→   158→        <span className={cn(
   159→   159→          'text-[10px] font-medium',
   160→   160→          isOverLimit ? 'text-red-400' : 'text-orange-200/50'
   161→   161→        )}>
   162→   162→          {used.toLocaleString()} / {total.toLocaleString()} {unit}
   163→   163→        </span>
   164→   164→      </div>
   165→   165→      <div className="h-1.5 bg-white/5 rounded-full overflow-hidden">
   166→   166→        <div
   167→   167→          className={cn(
   168→   168→            'h-full rounded-full transition-all duration-700',
   169→   169→            isOverLimit ? 'bg-gradient-to-r from-red-500 to-red-400' : 'bg-gradient-to-r from-emerald-500 to-emerald-400'
   170→   170→          )}
   171→   171→          style={{ width: `${pct}%` }}
   172→   172→        />
   173→   173→      </div>
   174→   174→      {isOverLimit && overageRate && (
   175→   175→        <p className="text-[9px] text-red-400 flex items-center gap-1">
   176→   176→          <AlertTriangle className="w-2.5 h-2.5" />
   177→   177→          Over limit — {(used - total).toLocaleString()} overage at {overageRate}/ticket
   178→   178→        </p>
   179→   179→      )}
   180→   180→    </div>
   181→   181→  );
   182→   182→}

   268→   268→// ── Component ──────────────────────────────────────────────────────────
   269→   269→
   270→   270→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   271→   271→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   272→   272→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   273→   273→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   274→   274→  const [isSubmitting, setIsSubmitting] = useState(false);
   275→   275→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   276→   276→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   277→   277→
   278→   278→  // Restore add-ons from localStorage
   279→   279→  useEffect(() => {
   280→   280→    try {
   281→   281→      const stored = localStorage.getItem('parwa_pricing_context');
   282→   282→      if (stored) {
   283→   283→        const ctx = JSON.parse(stored) as PricingContext;
   284→   284→        if (ctx.addOns) {
   285→   285→          setAddOns(ctx.addOns);
   286→   286→        }
   287→   287→      }
   288→   288→    } catch {
   289→   289→      // ignore
   290→   290→    }
   291→   291→  }, []);
   292→   292→
   293→   293→  // Check if Paddle.js is available
   294→   294→  useEffect(() => {
   295→   295→    getPaddleInstance().then((paddle) => {
   296→   296→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   297→   297→    }).catch(() => {
   298→   298→      setPaddleStatus('unavailable');
   299→   299→    });
   300→   300→  }, []);
   301→   301→
   302→   302→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   303→   303→
   304→   304→  const totalTicketLimit = useMemo(
   305→   305→    () => activeVariants.reduce((sum, tier) => sum + VARIANT_LIMITS[tier].monthlyTickets, 0),
   306→   306→    [activeVariants]
   307→   307→  );
   308→   308→
   309→   309→  const baseSubscription = useMemo(
   310→   310→    () => activeVariants.reduce((sum, tier) => sum + VARIANT_PRICES[tier], 0),
   311→   311→    [activeVariants]
   312→   312→  );
   313→   313→
   314→   314→  const addOnTotal = useMemo(() => {
   315→   315→    let total = 0;
   316→   316→    if (addOns.voice) {
   317→   317→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   318→   318→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   319→   319→        total += voiceAddOn.price;
   320→   320→      }
   321→   321→    }
   322→   322→    if (addOns.customApi) {
   323→   323→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   324→   324→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   325→   325→        total += customApiAddOn.price;
   326→   326→      }
   327→   327→    }
   328→   328→    return total;
   329→   329→  }, [addOns, activeVariants]);
   330→   330→
   331→   331→  const totalMonthly = baseSubscription + addOnTotal;
   332→   332→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   333→   333→
   334→   334→  // Overage projection (estimates at 80% utilization)
   335→   335→  const projectedTickets = Math.round(totalTicketLimit * 0.8);
   336→   336→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);
   337→   337→  const overageCost = overageTickets * OVERAGE_PRICE_PER_TICKET;
   338→   338→
   339→   339→  const toggleVariant = (tier: VariantTier) => {
   340→   340→    setActiveVariants((prev) => {
   341→   341→      if (prev.includes(tier)) {
   342→   342→        if (prev.length <= 1) {
   343→   343→          toast.error('You must have at least one active variant');
   344→   344→          return prev;
   345→   345→        }
   346→   346→        return prev.filter((t) => t !== tier);
   347→   347→      }
   348→   348→      return [...prev, tier];
   349→   349→    });
   350→   350→  };
   351→   351→
   352→   352→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   353→   353→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   354→   354→  };
   355→   355→
   356→   356→  const handleProceed = async () => {
   357→   357→    setIsSubmitting(true);
   358→   358→    setCheckoutError(null);
   359→   359→
   360→   360→    try {
   361→   361→      // Update pricing context in localStorage
   362→   362→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   363→   363→      const context: PricingContext = {
   364→   364→        industry: 'other',
   365→   365→        variant: primaryOnboardingVariant,
   366→   366→        addOns,
   367→   367→        totalMonthly,
   368→   368→        timestamp: new Date().toISOString(),
   369→   369→      };
   370→   370→
   371→   371→      // Preserve industry from existing context
   372→   372→      try {
   373→   373→        const stored = localStorage.getItem('parwa_pricing_context');
   374→   374→        if (stored) {
   375→   375→          const existing = JSON.parse(stored) as PricingContext;
   376→   376→          context.industry = existing.industry;
   377→   377→        }
   378→   378→      } catch {
   379→   379→        // ignore
   380→   380→      }
   381→   381→
   382→   382→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   383→   383→
   384→   384→      // ── Paddle Checkout Flow ──────────────────────────────────────
   385→   385→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   386→   386→
   387→   387→      // Add each active variant as a Paddle line item
   388→   388→      for (const tier of activeVariants) {
   389→   389→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   390→   390→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   391→   391→        if (variantPriceId) {
   392→   392→          checkoutItems.push({ priceId: variantPriceId, quantity: 1 });
   393→   393→        }
   394→   394→      }
   395→   395→
   396→   396→      // Voice add-on (only for starter where it's not included)
   397→   397→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   398→   398→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   399→   399→      }
   400→   400→
   401→   401→      // Custom API add-on (only for variants where it's not included)
   402→   402→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   403→   403→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });
   404→   404→      }
   405→   405→
   406→   406→      const customData = {
   407→   407→        source: 'parwa_onboarding',
   408→   408→        variant: primaryOnboardingVariant,
   409→   409→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   410→   410→        addOns,
   411→   411→        industry: context.industry,
   412→   412→        totalMonthly,
   413→   413→      };
   414→   414→
   415→   415→      // Try server-side transaction first
   416→   416→      let transactionId: string | null = null;
   417→   417→      let checkoutUrl: string | null = null;
   418→   418→
   419→   419→      try {
   420→   420→        const res = await fetch('/api/onboarding/checkout', {
   421→   421→          method: 'POST',
   422→   422→          headers: { 'Content-Type': 'application/json' },
   423→   423→          body: JSON.stringify({
   424→   424→            variant: primaryOnboardingVariant,
   425→   425→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   426→   426→            addOns,
   427→   427→            totalMonthly,
   428→   428→            industry: context.industry,
   429→   429→          }),
   430→   430→        });
   431→   431→
   432→   432→        if (res.ok) {
   433→   433→          const checkoutData = await res.json();
   434→   434→          transactionId = checkoutData.transaction_id || null;
   435→   435→          checkoutUrl = checkoutData.checkout_url || null;
   436→   436→        }
   437→   437→      } catch {
   438→   438→        // API unavailable — continue with client-side Paddle checkout
   439→   439→      }
   440→   440→
   441→   441→      // If we got a checkout_url, redirect to it
   442→   442→      if (checkoutUrl) {
   443→   443→        window.location.href = checkoutUrl;
   444→   444→        return;
   445→   445→      }
   446→   446→
   447→   447→      // Client-side Paddle checkout (items-based)
   448→   448→      if (paddleStatus === 'ready' && checkoutItems.length > 0) {
   449→   449→        const opened = await openCheckoutWithItems(
   450→   450→          checkoutItems,
   451→   451→          customData,
   452→   452→          () => {
   453→   453→            toast.success('Payment successful! Welcome to PARWA!');
   454→   454→            onComplete();
   455→   455→          },
   456→   456→          () => {
   457→   457→            toast('Checkout closed. You can try again anytime.', { icon: 'ℹ️' });
   458→   458→            setIsSubmitting(false);
   459→   459→          },
   460→   460→        );
   461→   461→        if (opened) {
   462→   462→          return;
   463→   463→        }
   464→   464→      }
   465→   465→

    64→    def create_ticket(
    65→        self,
    66→        customer_id: Optional[str] = None,
    67→        channel: str = "email",
    68→        subject: Optional[str] = None,
    69→        priority: str = TicketPriority.medium.value,
    70→        category: Optional[str] = None,
    71→        tags: Optional[List[str]] = None,
    72→        metadata_json: Optional[Dict[str, Any]] = None,
    73→        user_id: Optional[str] = None,
    74→        customer_name: Optional[str] = None,
    75→        customer_email: Optional[str] = None,
    76→        customer_phone: Optional[str] = None,
    77→        description: Optional[str] = None,
    78→    ) -> Ticket:
    79→        """Create a new ticket with production situation handlers.
    80→
    81→        PS01: Out-of-plan scope check
    82→        PS05: Duplicate detection
    83→        PS07: Account suspended check
    84→        BL05: Rate limiting
    85→
    86→        Args:
    87→            customer_id: ID of an existing customer. If None, customer_name +
    88→                customer_email must be provided and a Customer will be
    89→                auto-created (or matched by email if one already exists).
    90→            channel: Communication channel
    91→            subject: Optional subject line
    92→            priority: Priority level (default: medium)
    93→            category: Classification category
    94→            tags: List of tags
    95→            metadata_json: Additional metadata
    96→            user_id: ID of user creating the ticket (for audit)
    97→            customer_name: Customer name (for auto-create)
    98→            customer_email: Customer email (for auto-create)
    99→            customer_phone: Customer phone (for auto-create)
   100→            description: Long-form ticket body; stored as first TicketMessage
   101→                from the customer.
   102→
   103→        Returns:
   104→            Created Ticket object
   105→
   106→        Raises:
   107→            ValidationError: If validation fails (e.g. neither customer_id nor
   108→                customer_email provided)
   109→            AuthorizationError: If rate limited or account suspended
   110→        """
   111→        # BL05: Rate limiting check
   112→        self._check_rate_limit(user_id or customer_id or customer_email or "anon")
   113→
   114→        # PS07: Check if account is suspended
   115→        self._check_account_suspended()
   116→
   117→        # Free Trial gate: 15 tickets OR 24 hours, whichever hits first.
   118→        # Raises AuthorizationError(code=TRIAL_EXPIRED|TRIAL_TICKET_LIMIT)
   119→        # if the trial is over. Skipped for non-trial (paid) companies.
   120→        self._check_trial_limit()
   121→
   122→        # ── Resolve customer ────────────────────────────────────────────
   123→        # Either use existing customer_id, or auto-create/match by email.
   124→        if customer_id:
   125→            customer = self._validate_customer(customer_id)
   126→        else:
   127→            if not (customer_email or customer_name):
   128→                raise ValidationError(
   129→                    "Either customer_id or (customer_email + customer_name) must be provided"
   130→                )
   131→            customer = self._resolve_or_create_customer(
   132→                name=customer_name,
   133→                email=customer_email,
   134→                phone=customer_phone,
   135→            )
   136→
   137→        # PS01: Check out-of-plan scope
   138→        scope_tags = self._check_scope(category)
   139→
   140→        # PS05: Check for duplicates
   141→        duplicate_of = self._check_duplicate(customer.id, subject)
   142→
   143→        # Create ticket
   144→        ticket = Ticket(
   145→            id=str(uuid.uuid4()),
   146→            company_id=self.company_id,
   147→            customer_id=customer.id,
   148→            channel=channel,
   149→            status=TicketStatus.open.value,
   150→            subject=subject,
   151→            priority=priority,
   152→            category=category,
   153→            tags=json.dumps(tags or [] + scope_tags),
   154→            metadata_json=json.dumps(metadata_json or {}),
   155→            duplicate_of_id=duplicate_of,
   156→            created_at=datetime.now(timezone.utc),
   157→            updated_at=datetime.now(timezone.utc),
   158→        )
   159→
   160→        self.db.add(ticket)
   161→        self.db.commit()
   162→        self.db.refresh(ticket)
   163→
   164→        # ── Increment trial ticket counter (if company is on free trial) ──
   165→        # No-op for paid companies. We do this AFTER the commit so a failed
   166→        # commit doesn't burn a trial ticket.
   167→        self._increment_trial_usage()
   168→
   169→        # ── Store description as first TicketMessage from customer ──────
   170→        # The frontend sends `description` as the long-form ticket body.
   171→        # The pipeline reads this as the customer's query. We persist it as
   172→        # a TicketMessage so it shows up in the ticket timeline.
   173→        if description:
   174→            from database.models.tickets import TicketMessage
   175→            initial_msg = TicketMessage(
   176→                id=str(uuid.uuid4()),
   177→                ticket_id=ticket.id,
   178→                company_id=self.company_id,
   179→                role="customer",
   180→                content=description,
   181→                channel=channel,
   182→                metadata_json=json.dumps({"sender_name": customer.name or customer.email or "Customer"}),
   183→                created_at=datetime.now(timezone.utc),
   184→            )
   185→            self.db.add(initial_msg)
   186→            self.db.commit()
   187→
   188→        return ticket

   190→    def _resolve_or_create_customer(
   191→        self,
   192→        name: Optional[str] = None,
   193→        email: Optional[str] = None,
   194→        phone: Optional[str] = None,
   195→    ) -> Customer:
   196→        """Match an existing customer by email within this tenant, or create a new one.
   197→
   198→        Args:
   199→            name: Customer name (optional).
   200→            email: Customer email (optional, but used for dedup match).
   201→            phone: Customer phone (optional).
   202→
   203→        Returns:
   204→            The existing or newly-created Customer object.
   205→        """
   206→        # Try to match by email first (BC-001 tenant-scoped)
   207→        if email:
   208→            existing = self.db.query(Customer).filter(
   209→                Customer.company_id == self.company_id,
   210→                Customer.email == email,
   211→            ).first()
   212→            if existing:
   213→                # Update name/phone if provided and currently empty
   214→                touched = False
   215→                if name and not existing.name:
   216→                    existing.name = name
   217→                    touched = True
   218→                if phone and not existing.phone:
   219→                    existing.phone = phone
   220→                    touched = True
   221→                if touched:
   222→                    self.db.commit()
   223→                    self.db.refresh(existing)
   224→                return existing
   225→
   226→        # Otherwise create a new customer
   227→        customer = Customer(
   228→            id=str(uuid.uuid4()),
   229→            company_id=self.company_id,
   230→            email=email,
   231→            phone=phone,
   232→            name=name,
   233→            metadata_json="{}",
   234→            created_at=datetime.now(timezone.utc),
   235→            updated_at=datetime.now(timezone.utc),
   236→        )
   237→        self.db.add(customer)
   238→        self.db.commit()
   239→        self.db.refresh(customer)
   240→        return customer

   266→    def list_tickets(
   267→        self,
   268→        status: Optional[List[str]] = None,
   269→        priority: Optional[List[str]] = None,
   270→        category: Optional[List[str]] = None,
   271→        assigned_to: Optional[str] = None,
   272→        channel: Optional[str] = None,
   273→        customer_id: Optional[str] = None,
   274→        tags: Optional[List[str]] = None,
   275→        is_spam: Optional[bool] = None,
   276→        is_frozen: Optional[bool] = None,
   277→        date_from: Optional[datetime] = None,
   278→        date_to: Optional[datetime] = None,
   279→        search: Optional[str] = None,
   280→        page: int = 1,
   281→        page_size: int = 20,
   282→        sort_by: str = "created_at",
   283→        sort_order: str = "desc",
   284→    ) -> Tuple[List[Ticket], int]:
   285→        """List tickets with filters and pagination.
   286→
   287→        Args:
   288→            status: Filter by status list
   289→            priority: Filter by priority list
   290→            category: Filter by category list
   291→            assigned_to: Filter by assignee ID
   292→            channel: Filter by channel
   293→            customer_id: Filter by customer ID
   294→            tags: Filter by tags (matches any)
   295→            is_spam: Filter by spam status
   296→            is_frozen: Filter by frozen status
   297→            date_from: Filter by creation date (from)
   298→            date_to: Filter by creation date (to)
   299→            search: Full-text search query
   300→            page: Page number (1-based)
   301→            page_size: Items per page
   302→            sort_by: Sort field
   303→            sort_order: Sort direction (asc/desc)
   304→
   305→        Returns:
   306→            Tuple of (tickets list, total count)
   307→        """
   308→        query = self.db.query(Ticket).filter(
   309→            Ticket.company_id == self.company_id
   310→        )
   311→
   312→        # Apply filters
   313→        if status:
   314→            query = query.filter(Ticket.status.in_(status))
   315→
   316→        if priority:
   317→            query = query.filter(Ticket.priority.in_(priority))
   318→
   319→        if category:
   320→            query = query.filter(Ticket.category.in_(category))
   321→
   322→        if assigned_to:
   323→            query = query.filter(Ticket.assigned_to == assigned_to)
   324→
   325→        if channel:
   326→            query = query.filter(Ticket.channel == channel)
   327→
   328→        if customer_id:
   329→            query = query.filter(Ticket.customer_id == customer_id)
   330→
   331→        if is_spam is not None:
   332→            query = query.filter(Ticket.is_spam == is_spam)
   333→
   334→        if is_frozen is not None:
   335→            query = query.filter(Ticket.frozen == is_frozen)
   336→
   337→        if date_from:
   338→            query = query.filter(Ticket.created_at >= date_from)
   339→
   340→        if date_to:
   341→            query = query.filter(Ticket.created_at <= date_to)
   342→
   343→        # Tags filter (JSON contains)
   344→        if tags:
   345→            # Simple tag matching - check if any tag exists in the JSON
   346→            for tag in tags:
   347→                query = query.filter(Ticket.tags.contains(f'"{tag}"'))
   348→
   349→        # Full-text search
   350→        if search:
   351→            # M-33: Escape ILIKE wildcards to prevent SQL injection
   352→            escaped = search.replace("%", r"\\%").replace("_", r"\\_")
   353→            search_pattern = f"%{escaped}%"
   354→            query = query.filter(
   355→                or_(
   356→                    Ticket.subject.ilike(search_pattern, escape="\\"),
   357→                    Ticket.metadata_json.ilike(search_pattern, escape="\\"),
   358→                )
   359→            )
   360→
   361→        # Count total
   362→        total = query.count()
   363→
   364→        # Sort
   365→        sort_column = getattr(Ticket, sort_by, Ticket.created_at)
   366→        if sort_order == "desc":
   367→            query = query.order_by(desc(sort_column))
   368→        else:
   369→            query = query.order_by(sort_column)
   370→
   371→        # Paginate
   372→        offset = (page - 1) * page_size
   373→        tickets = query.offset(offset).limit(page_size).all()
   374→
   375→        return tickets, total

   379→    def update_ticket(
   380→        self,
   381→        ticket_id: str,
   382→        priority: Optional[str] = None,
   383→        category: Optional[str] = None,
   384→        tags: Optional[List[str]] = None,

   387→        subject: Optional[str] = None,
   388→        user_id: Optional[str] = None,
   389→        reason: Optional[str] = None,
   390→    ) -> Ticket:
   391→        """Update ticket fields.
   392→
   393→        Args:
   394→            ticket_id: Ticket ID
   395→            priority: New priority
   396→            category: New category
   397→            tags: New tags list
   398→            status: New status
   399→            assigned_to: New assignee ID
   400→            subject: New subject
   401→            user_id: ID of user making the change
   402→            reason: Reason for change
   403→
   404→        Returns:
   405→            Updated Ticket object
   406→
   407→        Raises:
   408→            NotFoundError: If ticket not found
   409→            ValidationError: If validation fails
   410→        """
   411→        ticket = self.get_ticket(ticket_id)
   412→
   413→        old_status = ticket.status
   414→
   415→        # Track status change
   416→        if status and status != old_status:
   417→            self._validate_status_transition(old_status, status)
   418→            self._record_status_change(
   419→                ticket_id, old_status, status, user_id, reason
   420→            )
   421→
   422→        # Update fields
   423→        if priority is not None:
   424→            ticket.priority = priority
   425→
   426→        if category is not None:
   427→            ticket.category = category
   428→
   429→        if tags is not None:
   430→            ticket.tags = json.dumps(tags)
   431→
   432→        if status is not None:
   433→            ticket.status = status
   434→
   435→            # Handle status-specific logic

   437→                ticket.closed_at = datetime.now(timezone.utc)
   438→            elif status == TicketStatus.reopened.value:
   439→                ticket.reopen_count = (ticket.reopen_count or 0) + 1
   440→
   441→        if assigned_to is not None:
   442→            ticket.assigned_to = assigned_to
   443→
   444→        if subject is not None:
   445→            ticket.subject = subject
   446→
   447→        ticket.updated_at = datetime.now(timezone.utc)
   448→
   449→        self.db.commit()
   450→        self.db.refresh(ticket)
   451→
   452→        return ticket

   606→    # ── PRIVATE HELPERS ────────────────────────────────────────────────────

   608→    def _check_rate_limit(self, identifier: str) -> None:
   609→        """BL05: Check rate limit for ticket creation.
   610→
   611→        Uses the shared rate limit service. The bridge's check_rate_limit
   612→        accepts (category, identifier) — we map our ticket-specific key
   613→        to those parameters.
   614→
   615→        Args:
   616→            identifier: User ID or customer ID
   617→
   618→        Raises:
   619→            AuthorizationError: If rate limited
   620→        """
   621→        # The rate limit service (parwa_core_bridge._RateLimiterBridge)
   622→        # accepts (category, identifier) — not (key, max_requests, window).
   623→        # We use "ticket_create" as the category and the company+user as
   624→        # the identifier for per-tenant rate limiting.
   625→        category = "ticket_create"
   626→        full_identifier = f"{self.company_id}:{identifier}"
   627→
   628→        try:
   629→            result = self.rate_limit_service.check_rate_limit(
   630→                category, full_identifier
   631→            )
   632→            allowed = getattr(result, "allowed", True)
   633→        except TypeError:
   634→            # Fallback: if the service doesn't accept (category, identifier)
   635→            # signature, allow the request (BC-008: fail open, not closed)
   636→            allowed = True
   637→
   638→        if not allowed:
   639→            raise AuthorizationError(
   640→                "Rate limit exceeded. Please try again later."
   641→            )

   643→    def _check_account_suspended(self) -> None:
   644→        """PS07: Check if account is suspended.
   645→
   646→        Raises:
   647→            AuthorizationError: If account is suspended
   648→        """
   649→        company = self.db.query(Company).filter(
   650→            Company.id == self.company_id

   658→    def _check_trial_limit(self) -> None:
   659→        """Free Trial gate: 15 tickets OR 24 hours, whichever hits first.
   660→
   661→        Skipped entirely for paid companies (is_trial=False).
   662→        For trial companies, raises AuthorizationError with a specific
   663→        error_code so the frontend can show an upgrade CTA.
   664→
   665→        TEMPORARY: Set env var TRIAL_LIMIT_DISABLED=true to skip the
   666→        ticket count check entirely (for production testing). The 24-hour
   667→        time check still applies. Re-enable by removing the env var.
   668→
   669→        Raises:
   670→            AuthorizationError: If trial has expired (time) or ticket
   671→                                limit has been reached.
   672→        """
   673→        company = self.db.query(Company).filter(
   674→            Company.id == self.company_id
   675→        ).first()
   676→
   677→        # Not in trial (paid company, or pre-trial-system company) — skip.
   678→        if not company or not getattr(company, "is_trial", False):
   679→            return
   680→
   681→        now = datetime.now(timezone.utc)
   682→
   683→        # Time check: 24 hours since signup (always enforced)
   684→        if company.trial_ends_at and now >= company.trial_ends_at:
   685→            raise AuthorizationError(
   686→                "Your 24-hour free trial has ended. "
   687→                "Upgrade to a paid plan to continue creating tickets.",
   688→                details={"reason": "TRIAL_TIME_EXPIRED", "trial_ends_at": company.trial_ends_at.isoformat()},
   689→            )
   690→
   691→        # Ticket count check: 15 tickets used
   692→        # TEMPORARY: disabled for production testing. Re-enable by setting
   693→        # TRIAL_LIMIT_DISABLED env var to "false" or removing this early return.
   694→        import os as _os
   695→        if _os.environ.get("TRIAL_LIMIT_DISABLED", "true").lower() in ("true", "1", "yes"):
   696→            return  # Skip ticket count check (DEFAULT: disabled for production testing)
   697→
   698→        used = int(company.trial_tickets_used or 0)
   699→        if used >= self.TRIAL_TICKET_LIMIT:
   700→            raise AuthorizationError(
   701→                f"You've used all {self.TRIAL_TICKET_LIMIT} free trial tickets. "
   702→                "Upgrade to a paid plan to continue.",
   703→                details={"reason": "TRIAL_TICKET_LIMIT", "used": used, "limit": self.TRIAL_TICKET_LIMIT},
   704→            )

   706→    def _increment_trial_usage(self) -> None:
   707→        """Increment trial_tickets_used after a successful ticket commit.
   708→
   709→        No-op for paid companies. Uses a raw UPDATE to avoid race conditions
   710→        with concurrent ticket creation (two requests incrementing at the
   711→        same time would otherwise both read the same count).
   712→        """
   713→        company = self.db.query(Company).filter(
   714→            Company.id == self.company_id
   715→        ).first()
   716→
   717→        if not company or not getattr(company, "is_trial", False):
   718→            return
   719→
   720→        # Atomic increment — concurrent ticket creates won't undercount.
   721→        self.db.query(Company).filter(
   722→            Company.id == self.company_id,
   723→            Company.is_trial.is_(True),
   724→        ).update(
   725→            {Company.trial_tickets_used: Company.trial_tickets_used + 1},
   726→            synchronize_session=False,
   727→        )
   728→        self.db.commit()

   771→    def _check_duplicate(
   772→        self,
   773→        customer_id: str,
   774→        subject: Optional[str],
   775→    ) -> Optional[str]:
   776→        """PS05: Check for duplicate tickets.
   777→
   778→        Args:
   779→            customer_id: Customer ID
   780→            subject: Ticket subject
   781→
   782→        Returns:
   783→            ID of duplicate ticket if found, None otherwise
   784→        """
   785→        if not subject:
   786→            return None
   787→
   788→        # Look for recent open tickets from same customer with similar subject
   789→        recent_threshold = datetime.now(timezone.utc) - timedelta(hours=24)
   790→
   791→        similar_tickets = self.db.query(Ticket).filter(
   792→            Ticket.company_id == self.company_id,
   793→            Ticket.customer_id == customer_id,
   794→            Ticket.status.in_([
   795→                TicketStatus.open.value,
   796→                TicketStatus.assigned.value,
   797→                TicketStatus.in_progress.value,
   798→            ]),
   799→            Ticket.created_at >= recent_threshold,
   800→        ).all()
   801→
   802→        for ticket in similar_tickets:
   803→            if ticket.subject:
   804→                similarity = self._calculate_similarity(
   805→                    subject.lower(),
   806→                    ticket.subject.lower()
   807→                )
   808→                if similarity >= self.DUPLICATE_SIMILARITY_THRESHOLD:
   809→                    return ticket.id
   810→
   811→        return None
   812→
   813→    def _calculate_similarity(self, str1: str, str2: str) -> float:
   814→        """Calculate similarity between two strings.
   815→
   816→        Simple Jaccard similarity on words.
   817→
   818→        Args:
   819→            str1: First string
   820→            str2: Second string
   821→
   822→        Returns:
   823→            Similarity score between 0 and 1
   824→        """
   825→        words1 = set(str1.split())
   826→        words2 = set(str2.split())
   827→
   828→        if not words1 or not words2:
   829→            return 0.0
   830→
   831→        intersection = words1 & words2
   832→        union = words1 | words2
   833→
   834→        return len(intersection) / len(union)

   836→    def _validate_status_transition(
   837→        self,
   838→        from_status: str,
   839→        to_status: str,
   840→    ) -> None:
   841→        """Validate status transition is allowed.
   842→
   843→        Args:
   844→            from_status: Current status
   845→            to_status: Target status
   846→
   847→        Raises:
   848→            ValidationError: If transition is invalid
   849→        """
   850→        # Define valid transitions
   851→        valid_transitions = {
   852→            TicketStatus.open.value: [
   853→                TicketStatus.assigned.value,
   854→                TicketStatus.queued.value,
   855→                TicketStatus.frozen.value,
   856→                TicketStatus.closed.value,
   857→            ],
   858→            TicketStatus.assigned.value: [
   859→                TicketStatus.in_progress.value,
   860→                TicketStatus.awaiting_client.value,
   861→                TicketStatus.awaiting_human.value,
   862→                TicketStatus.open.value,
   863→            ],
   864→            TicketStatus.in_progress.value: [
   865→                TicketStatus.awaiting_client.value,
   866→                TicketStatus.awaiting_human.value,
   867→                TicketStatus.resolved.value,
   868→            ],
   869→            TicketStatus.awaiting_client.value: [
   870→                TicketStatus.in_progress.value,
   871→                TicketStatus.stale.value,
   872→                TicketStatus.resolved.value,
   873→            ],
   874→            TicketStatus.awaiting_human.value: [
   875→                TicketStatus.in_progress.value,
   876→                TicketStatus.resolved.value,
   877→            ],
   878→            TicketStatus.resolved.value: [
   879→                TicketStatus.closed.value,
   880→                TicketStatus.reopened.value,
   881→            ],
   882→            TicketStatus.reopened.value: [
   883→                TicketStatus.in_progress.value,
   884→                TicketStatus.awaiting_human.value,
   885→            ],
   886→            TicketStatus.closed.value: [
   887→                TicketStatus.reopened.value,
   888→            ],
   889→            TicketStatus.frozen.value: [
   890→                TicketStatus.open.value,
   891→                TicketStatus.closed.value,
   892→            ],
   893→            TicketStatus.queued.value: [
   894→                TicketStatus.open.value,
   895→            ],
   896→            TicketStatus.stale.value: [
   897→                TicketStatus.in_progress.value,
   898→                TicketStatus.closed.value,
   899→            ],
   900→        }
   901→
   902→        allowed = valid_transitions.get(from_status, [])
   903→
   904→        if to_status not in allowed:
   905→            raise ValidationError(
   906→                f"Invalid status transition: {from_status} -> {to_status}"
   907→            )

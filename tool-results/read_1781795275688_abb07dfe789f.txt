   383→# IN-MEMORY BACKEND (works immediately, no deps)
   384→# ═══════════════════════════════════════════════════════════════
   385→
   386→class InMemoryBackend(StorageBackend):
   387→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   388→
   389→    def __init__(self):
   390→        self._notifications: Dict[str, Dict] = {}
   391→        self._flags: List[Dict] = []
   392→        self._audit: List[Dict] = []
   393→        self._quality_scores: List[Dict] = []
   394→        self._batches: Dict[str, List[Dict]] = {}
   395→        self._batch_metas: Dict[str, Dict] = {}
   396→        self._lock = threading.Lock()
   397→        self._next_nfy_id = 1
   398→        # Wave 2 stores
   399→        self._integration_pings: List[Dict] = []
   400→        self._llm_costs: List[Dict] = []
   401→        self._stuck_ticket_events: List[Dict] = []
   402→        self._load_status: Dict[str, Dict] = {}  # tenant_id -> {variant -> {concurrent, max_concurrent}}
   403→        self._agent_configs: List[Dict] = []  # Wave 3: agent config store
   404→        self._outbox: List[Dict] = []  # Wave 3: outbox queue (recall/void)
   405→
   406→    # ── Helpers ───────────────────────────────────────────────
   407→
   408→    def _next_notification_number(self) -> int:
   409→        with self._lock:
   410→            n = self._next_nfy_id
   411→            self._next_nfy_id += 1
   412→            return n
   413→
   414→    @staticmethod
   415→    def _priority_from_score(score: float) -> str:
   416→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:
   417→            return PRIORITY_CRITICAL
   418→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_HIGH]:
   419→            return PRIORITY_HIGH
   420→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_MEDIUM]:
   421→            return PRIORITY_MEDIUM
   422→        return PRIORITY_LOW
   423→
   424→    # ── Notifications ────────────────────────────────────────
   425→
   426→    async def create_notification(
   427→        self,
   428→        tenant_id: str,
   429→        ntype: str,
   430→        priority_score: float,
   431→        title: str,
   432→        description: str,
   433→        related_tickets: Optional[List[str]] = None,
   434→        batch_key: Optional[str] = None,
   435→        source_data: Optional[Dict] = None,
   436→    ) -> Dict[str, Any]:
   437→        num = self._next_notification_number()
   438→        key = f"PARWA-NFY-{num:03d}"
   439→        now = time.time()
   440→        nf = {
   441→            "id": str(uuid.uuid4()),
   442→            "tenant_id": tenant_id,
   443→            "notification_key": key,
   444→            "type": ntype,
   445→            "priority": self._priority_from_score(priority_score),
   446→            "priority_score": round(priority_score, 4),
   447→            "title": title,
   448→            "description": description,
   449→            "related_tickets": related_tickets or [],
   450→            "batch_key": batch_key,
   451→            "source_data": source_data or {},
   452→            "is_read": False,
   453→            "is_resolved": False,
   454→            "created_at": now,
   455→            "resolved_at": None,
   456→            "backend": "memory",
   457→        }
   458→        self._notifications[key] = nf
   459→        logger.info("NFY created: %s type=%s score=%.3f [%s]", key, ntype, priority_score, tenant_id)
   460→        return nf
   461→
   462→    async def get_notification(self, key: str) -> Optional[Dict[str, Any]]:
   463→        return self._notifications.get(key)
   464→
   465→    async def get_notifications(
   466→        self,
   467→        tenant_id: str,
   468→        include_resolved: bool = False,
   469→        min_priority: Optional[str] = None,
   470→    ) -> List[Dict[str, Any]]:
   471→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   472→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   473→        results = []
   474→        for n in self._notifications.values():
   475→            if n["tenant_id"] != tenant_id:
   476→                continue
   477→            if not include_resolved and n["is_resolved"]:
   478→                continue
   479→            if priority_order.get(n["priority"], 99) > min_rank:
   480→                continue
   481→            results.append(n)
   482→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   483→        return results
   484→
   485→    async def resolve_notification(self, key: str) -> bool:
   486→        n = self._notifications.get(key)
   487→        if n and not n["is_resolved"]:
   488→            n["is_resolved"] = True
   489→            n["resolved_at"] = time.time()
   490→            return True
   491→        return False
   492→
   493→    async def dismiss_notification(self, key: str) -> bool:
   494→        n = self._notifications.get(key)
   495→        if n:
   496→            n["is_read"] = True
   497→            return True
   498→        return False
   499→
   500→    # ── System Flags ─────────────────────────────────────────
   501→
   502→    async def set_flag(
   503→        self,
   504→        tenant_id: str,
   505→        flag_type: str,
   506→        flag_value: str,
   507→        set_by: str,
   508→        scope: str = "global",
   509→        target_id: Optional[str] = None,
   510→        reason: Optional[str] = None,
   511→        expires_at: Optional[str] = None,
   512→    ) -> Dict[str, Any]:
   513→        flag = {
   514→            "id": str(uuid.uuid4()),
   515→            "tenant_id": tenant_id,
   516→            "flag_type": flag_type,
   517→            "flag_value": flag_value,
   518→            "scope": scope,
   519→            "target_id": target_id,
   520→            "set_by": set_by,
   521→            "reason": reason,
   522→            "expires_at": expires_at,
   523→            "created_at": datetime.now(timezone.utc).isoformat(),
   524→            "revoked_at": None,
   525→            "backend": "memory",
   526→        }
   527→        self._flags.append(flag)
   528→        logger.info("FLAG set: %s=%s scope=%s [%s]", flag_type, flag_value, scope, tenant_id)
   529→        return flag
   530→
   531→    async def get_active_flags(
   532→        self, tenant_id: str, flag_type: Optional[str] = None
   533→    ) -> List[Dict[str, Any]]:
   534→        now = datetime.now(timezone.utc)
   535→        active = []
   536→        for f in self._flags:
   537→            if f["tenant_id"] != tenant_id:
   538→                continue
   539→            if f["revoked_at"] is not None:
   540→                continue
   541→            if f["expires_at"]:
   542→                exp = datetime.fromisoformat(f["expires_at"]) if isinstance(f["expires_at"], str) else f["expires_at"]
   543→                if exp < now:
   544→                    continue
   545→            if flag_type and f["flag_type"] != flag_type:
   546→                continue
   547→            active.append(f)
   548→        return active
   549→
   550→    async def revoke_flag(self, flag_id: str, revoked_by: str) -> bool:
   551→        for f in self._flags:
   552→            if f["id"] == flag_id and f["revoked_at"] is None:
   553→                f["revoked_at"] = datetime.now(timezone.utc).isoformat()
   554→                logger.info("FLAG revoked: %s by %s", flag_id, revoked_by)
   555→                return True
   556→        return False

   558→    # ── Audit Trail ──────────────────────────────────────────
   559→
   560→    async def create_audit_entry(
   561→        self,
   562→        tenant_id: str,
   563→        action: str,
   564→        actor_email: str,
   565→        target_type: Optional[str] = None,
   566→        target_id: Optional[str] = None,
   567→        payload: Optional[Dict] = None,
   568→    ) -> Dict[str, Any]:
   569→        payload_json = json.dumps(payload or {}, sort_keys=True)
   570→        current_hash = hashlib.sha256(
   571→            f"{tenant_id}:{action}:{payload_json}:{time.time()}".encode()
   572→        ).hexdigest()[:64]
   573→
   574→        entry = {
   575→            "id": len(self._audit) + 1,
   576→            "tenant_id": tenant_id,
   577→            "action": action,
   578→            "actor_email": actor_email,
   579→            "target_type": target_type,
   580→            "target_id": target_id,
   581→            "payload": payload or {},
   582→            "current_hash": current_hash,
   583→            "previous_hash": self._audit[-1]["current_hash"] if self._audit else None,
   584→            "created_at": datetime.now(timezone.utc).isoformat(),
   585→            "backend": "memory",
   586→        }
   587→        self._audit.append(entry)
   588→        return entry
   589→
   590→    async def get_audit_trail(
   591→        self,
   592→        tenant_id: str,
   593→        action: Optional[str] = None,

   662→    # ── Batch Queue ──────────────────────────────────────────
   663→
   664→    async def add_to_batch(
   665→        self,
   666→        tenant_id: str,
   667→        batch_key: str,
   668→        ticket_id: str,
   669→        confidence: float,
   670→    ) -> Optional[Dict[str, Any]]:
   671→        BATCH_WINDOW_S = 300
   672→        now = time.time()
   673→
   674→        if tenant_id not in self._batch_metas:
   675→            self._batch_metas[tenant_id] = {}
   676→
   677→        # Find existing batch with same key within window
   678→        existing_key = None
   679→        for bk, meta in self._batch_metas[tenant_id].items():
   680→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   681→                existing_key = bk
   682→                break
   683→
   684→        if existing_key:
   685→            meta = self._batch_metas[tenant_id][existing_key]
   686→            meta["ticket_ids"].append(ticket_id)
   687→            meta["signal_count"] = len(meta["ticket_ids"])
   688→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   689→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   690→            # Don't flush yet — still within window
   691→            return None
   692→        else:
   693→            meta = {
   694→                "batch_key": batch_key,
   695→                "tenant_id": tenant_id,
   696→                "batch_start": now,
   697→                "ticket_ids": [ticket_id],
   698→                "signal_count": 1,
   699→                "confidence_min": confidence,
   700→                "confidence_max": confidence,
   701→                "status": "pending",
   702→            }
   703→            self._batch_metas[tenant_id][batch_key] = meta
   704→            return None

   755→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
   756→        """Compute health from recent pings (last 30 per service)."""
   757→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
   758→        services: Dict[str, List[Dict]] = {}
   759→        for p in tenant_pings:
   760→            services.setdefault(p["service_name"], []).append(p)
   761→
   762→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
   763→        for svc, pings in services.items():
   764→            recent = pings[-30:]  # last 30 pings
   765→            if not recent:
   766→                continue
   767→            healthy = sum(1 for p in recent if p["is_healthy"])
   768→            total = len(recent)
   769→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
   770→            last_ping = recent[-1]
   771→            avg_ms = (sum(p["response_ms"] or 0 for p in recent) / total) if total else 0
   772→            last_error = None
   773→            last_error_time = None
   774→            for p in reversed(recent):
   775→                if not p["is_healthy"]:
   776→                    last_error = p.get("error_detail")
   777→                    last_error_time = p["created_at"]
   778→                    break
   779→
   780→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
   781→            if status != "healthy":
   782→                result["degraded_count"] += 1
   783→            else:
   784→                result["healthy_count"] += 1
   785→
   786→            result["services"][svc] = {
   787→                "status": status,
   788→                "uptime_pct": uptime_pct,
   789→                "total_pings": total,
   790→                "healthy_pings": healthy,
   791→                "avg_response_ms": round(avg_ms, 1),
   792→                "last_ping_at": last_ping["created_at"],
   793→                "last_healthy_at": next(
   794→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
   795→                    None,
   796→                ),
   797→                "last_error": last_error,
   798→                "last_error_at": last_error_time,
   799→                "last_status_code": last_ping.get("status_code"),
   800→            }
   801→        return result

   883→    # ── Drift Detection (Wave 2) ─────────────────────────────
   884→
   885→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
   886→        """Analyze quality_scores for drift using day-over-day comparison."""
   887→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   888→        if len(tenant_scores) < 3:
   889→            return {
   890→                "tenant_id": tenant_id, "drift_detected": False,
   891→                "drift_severity": "none", "accuracy_7d": None,
   892→                "accuracy_yesterday": None, "accuracy_today": None,
   893→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
   894→                "total_scores": len(tenant_scores),
   895→            }
   896→        # Group by date (ISO date portion)
   897→        by_date: Dict[str, List[float]] = {}
   898→        for s in tenant_scores:
   899→            date_str = s["created_at"][:10]  # YYYY-MM-DD
   900→            by_date.setdefault(date_str, []).append(s["overall_score"])
   901→
   902→        dates_sorted = sorted(by_date.keys())
   903→
   904→        def _avg(scores: List[float]) -> float:
   905→            return round(sum(scores) / len(scores), 4) if scores else 0.0
   906→
   907→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
   908→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
   909→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
   910→
   911→        # Drift detection: >5% drop for 3+ days
   912→        drift_detected = False
   913→        drift_severity = "none"
   914→        trigger_reason = None
   915→        trend_direction = "stable"
   916→
   917→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
   918→            recent_3 = dates_sorted[-3:]
   919→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
   920→            # Check if declining 3+ consecutive days
   921→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
   922→            drop_7d = accuracy_7d - accuracy_today
   923→
   924→            if declining and drop_7d > 0.05:
   925→                drift_detected = True
   926→                trend_direction = "declining"
   927→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
   928→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
   929→            elif drop_7d > 0.03:
   930→                drift_detected = True
   931→                trend_direction = "slight_decline"
   932→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
   933→                drift_severity = "warning"
   934→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
   935→                trend_direction = "improving"
   936→            else:
   937→                trend_direction = "stable"
   938→
   939→        # Check for repeated errors (same low score pattern)
   940→        if not drift_detected and tenant_scores:
   941→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
   942→            if len(low_scores) >= 3:
   943→                # Check if same resolution_path failing
   944→                paths: Dict[str, int] = {}
   945→                for s in low_scores:
   946→                    p = s.get("resolution_path", "unknown")
   947→                    paths[p] = paths.get(p, 0) + 1
   948→                worst_path = max(paths.items(), key=lambda x: x[1])
   949→                if worst_path[1] >= 3:
   950→                    drift_detected = True
   951→                    drift_severity = "warning"
   952→                    trend_direction = "declining"
   953→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
   954→
   955→        return {
   956→            "tenant_id": tenant_id,
   957→            "drift_detected": drift_detected,
   958→            "drift_severity": drift_severity,
   959→            "accuracy_7d": accuracy_7d,
   960→            "accuracy_yesterday": accuracy_yesterday,
   961→            "accuracy_today": accuracy_today,
   962→            "trend_direction": trend_direction,
   963→            "trigger_reason": trigger_reason,
   964→            "total_scores": len(tenant_scores),
   965→        }

  1536→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
  1537→        """Fetch recent pings and compute health server-side."""
  1538→        pings = await self._rest_get(
  1539→            "jarvis_integration_pings",
  1540→            f"tenant_id=eq.{tenant_id}&order=created_at.desc&limit=300"
  1541→        )
  1542→        services: Dict[str, List[Dict]] = {}
  1543→        for p in pings:
  1544→            services.setdefault(p.get("service_name", ""), []).append(p)
  1545→
  1546→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
  1547→        for svc, sp in services.items():
  1548→            recent = sp[:30]
  1549→            healthy = sum(1 for p in recent if p.get("is_healthy"))
  1550→            total = len(recent)
  1551→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
  1552→            last_ping = recent[0]
  1553→            avg_ms = sum(p.get("response_ms") or 0 for p in recent) / max(total, 1)
  1554→            last_error = None
  1555→            for p in recent:
  1556→                if not p.get("is_healthy"):
  1557→                    last_error = p.get("error_detail")
  1558→                    break
  1559→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
  1560→            if status != "healthy":
  1561→                result["degraded_count"] += 1
  1562→            else:
  1563→                result["healthy_count"] += 1
  1564→            result["services"][svc] = {
  1565→                "status": status, "uptime_pct": uptime_pct,
  1566→                "total_pings": total, "healthy_pings": healthy,
  1567→                "avg_response_ms": round(avg_ms, 1),

  1637→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1638→        # Fetch scores + client-side drift analysis (same logic as InMemory)
  1639→        scores = await self._rest_get(
  1640→            TABLE_QUALITY_SCORES,
  1641→            f"tenant_id=eq.{tenant_id}&select=overall_score,resolution_path,created_at&order=created_at.asc&limit=5000"
  1642→        )
  1643→        if len(scores) < 3:
  1644→            return {"tenant_id": tenant_id, "drift_detected": False, "drift_severity": "none",
  1645→                    "accuracy_7d": None, "accuracy_yesterday": None, "accuracy_today": None,
  1646→                    "trend_direction": "stable", "trigger_reason": "insufficient_data", "total_scores": len(scores)}
  1647→        by_date: Dict[str, List[float]] = {}
  1648→        for s in scores:
  1649→            d = (s.get("created_at") or "")[:10]
  1650→            by_date.setdefault(d, []).append(float(s.get("overall_score", 0)))
  1651→        dates_sorted = sorted(by_date.keys())
  1652→        def _avg(vals): return round(sum(vals) / len(vals), 4) if vals else 0.0
  1653→        accuracy_7d = _avg([v for d in dates_sorted[-7:] for v in by_date[d]])
  1654→        accuracy_today = _avg(by_date[dates_sorted[-1]])
  1655→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1656→        drift_detected = False; drift_severity = "none"; trigger_reason = None; trend_direction = "stable"
  1657→        if len(dates_sorted) >= 3:
  1658→            daily_avgs = [_avg(by_date[d]) for d in dates_sorted[-3:]]
  1659→            declining = all(daily_avgs[i] > daily_avgs[i+1] for i in range(len(daily_avgs)-1))
  1660→            drop = accuracy_7d - accuracy_today
  1661→            if declining and drop > 0.05:
  1662→                drift_detected = True; trend_direction = "declining"
  1663→                trigger_reason = f"accuracy_dropped_{drop:.1%}_over_3_days"
  1664→                drift_severity = "critical" if drop > 0.10 else "warning"
  1665→            elif drop > 0.03:
  1666→                drift_detected = True; trend_direction = "slight_decline"
  1667→                trigger_reason = f"accuracy_dropped_{drop:.1%}_from_7d_avg"; drift_severity = "warning"
  1668→            elif accuracy_today > accuracy_7d + 0.03:
  1669→                trend_direction = "improving"
  1670→        return {"tenant_id": tenant_id, "drift_detected": drift_detected, "drift_severity": drift_severity,
  1671→                "accuracy_7d": accuracy_7d, "accuracy_yesterday": accuracy_yesterday,
  1672→                "accuracy_today": accuracy_today, "trend_direction": trend_direction,
  1673→                "trigger_reason": trigger_reason, "total_scores": len(scores)}
  1674→

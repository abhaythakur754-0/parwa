    47→    47→def _format_notification(ntype: str, ev: Dict) -> tuple:
    48→    48→    """Format notification title and description by type."""
    49→    49→    signal = ev.get("signal", {})
    50→    50→    score = ev.get("priority_score", 0)
    51→    51→
    52→    52→    if ntype == TYPE_STUCK_TICKET:
    53→    53→        tid = signal.get("ticket_id", "unknown")
    54→    54→        reason = signal.get("reason", "unknown")
    55→    55→        quality = signal.get("quality_score", "?")
    56→    56→        escalation = ev.get("escalation_tier", "soft_reminder")
    57→    57→        hours = signal.get("hours_stuck", 0)
    58→    58→        title = f"Stuck Ticket [{escalation.upper()}]: {tid}"
    59→    59→        desc = (f"Ticket {tid} could not be resolved automatically. "
    60→    60→                f"Reason: {reason}. Quality achieved: {quality}. "
    61→    61→                f"Loops used: {signal.get('loops_used', 0)}. "
    62→    62→                f"Hours stuck: {hours}. Escalation: {escalation}. "
    63→    63→                f"Requires manual review.")
    64→    64→
    65→    65→    elif ntype == TYPE_QUOTA_LOW:
    66→    66→        data = list(signal.values())[0] if signal else {}
    67→    67→        tier = list(signal.keys())[0] if signal else "unknown"
    68→    68→        title = f"Quota {data.get('status', 'warning').upper()}: {tier}"
    69→    69→        desc = (f"{tier.capitalize()} plan quota is {data.get('burn_pct', 0)}% used "
    70→    70→                f"({data.get('remaining', 0)} of {data.get('total', 0)} remaining). "
    71→    71→                f"Consider upgrading or requesting additional quota.")
    72→    72→
    73→    73→    elif ntype == TYPE_ACCURACY_DROP:
    74→    74→        trend = signal.get("trend", "unknown")
    75→    75→        severity = signal.get("severity", "warning")
    76→    76→        trigger = signal.get("trigger", "unknown")
    77→    77→        acc_7d = signal.get("accuracy_7d", "?")
    78→    78→        acc_today = signal.get("accuracy_today", "?")
    79→    79→        title = f"Accuracy Drift [{severity.upper()}]: {trend}"
    80→    80→        desc = (f"Resolution accuracy trend is '{trend}' (severity: {severity}). "
    81→    81→                f"7-day avg: {acc_7d}, Today: {acc_today}. "
    82→    82→                f"Trigger: {trigger}. "
    83→    83→                f"This may indicate KB gaps, policy changes, or model drift. "
    84→    84→                f"Review recent resolutions and update knowledge base if needed.")
    85→    85→
    86→    86→    elif ntype == TYPE_INTEGRATION_DOWN:
    87→    87→        degraded = signal.get("degraded_services", [])
    88→    88→        worst_uptime = signal.get("worst_uptime_pct", 0)
    89→    89→        names = ", ".join(f"{d['name']} ({d['status']}, uptime={d.get('uptime_pct', '?')}%)" for d in degraded)
    90→    90→        title = f"Integration Issues: {signal.get('total_degraded', len(degraded))} service(s) degraded"
    91→    91→        desc = (f"Degraded integrations: {names}. "
    92→    92→                f"Worst uptime: {worst_uptime}%. "
    93→    93→                f"Check API credentials and service status.")
    94→    94→
    95→    95→    elif ntype == "load_bottleneck":
    96→    96→        at_cap = signal.get("at_capacity", [])
    97→    97→        high_load = signal.get("high_load", [])
    98→    98→        vip = signal.get("vip_overflow_risk", False)
    99→    99→        bottlenecks = ", ".join(v["name"] for v in at_cap + high_load)
   100→   100→        title = f"Load Bottleneck: {bottlenecks or 'All variants'}"
   101→   101→        desc = (f"Variant load issue detected. "
   102→   102→                f"At capacity: {', '.join(v['name'] for v in at_cap) or 'none'}. "
   103→   103→                f"High load: {', '.join(v['name'] for v in high_load) or 'none'}. "
   104→   104→                f"VIP overflow risk: {vip}. "
   105→   105→                f"Consider scaling up or routing overflow to available variants.")
   106→   106→
   107→   107→    else:
   108→   108→        title = f"Signal: {ntype}"
   109→   109→        desc = str(signal)
   110→   110→
   111→   111→    return title, desc

   114→   114→async def _create_notifications_from_evals(
   115→   115→    tenant_id: str,
   116→   116→    evaluations: List[Dict[str, Any]],
   117→   117→    db,
   118→   118→) -> List[Dict[str, Any]]:
   119→   119→    """Convert evaluations into notifications via DB. Filter LOW priority."""
   120→   120→    notifications = []
   121→   121→    for ev in evaluations:
   122→   122→        score = ev.get("priority_score", 0)
   123→   123→        if score < 0.40:
   124→   124→            logger.info("LOW priority signal filtered: %s (score=%.3f)", ev.get("type"), score)
   125→   125→            continue
   126→   126→
   127→   127→        ntype = ev.get("type", "unknown")
   128→   128→        title, desc = _format_notification(ntype, ev)
   129→   129→        related = ev.get("related_tickets", [])
   130→   130→
   131→   131→        nf = await db.create_notification(
   132→   132→            tenant_id=tenant_id,
   133→   133→            ntype=ntype,
   134→   134→            priority_score=score,
   135→   135→            title=title,
   136→   136→            description=desc,
   137→   137→            related_tickets=related,
   138→   138→            batch_key=ev.get("type", ""),
   139→   139→            source_data=ev.get("signal", {}),
   140→   140→        )
   141→   141→        notifications.append(nf)
   142→   142→
   143→   143→    return notifications

   146→   146→# ── Command Execution (NEW — Wave 1) ─────────────────────────
   147→   147→
   148→   148→async def _execute_command(
   149→   149→    intent_result: Dict[str, Any],
   150→   150→    auth_result: AuthResult,
   151→   151→    tenant_id: str,
   152→   152→    signals: Dict[str, Any],
   153→   153→    db,
   154→   154→    raw_input: str = "",
   155→   155→) -> str:
   156→   156→    """Execute a classified command and return the response text.
   157→   157→
   158→   158→    This is the core "Jarvis does things" function.
   159→   159→    Reads/writes DB based on intent. Returns human-readable response.
   160→   160→    """
   161→   161→    intent = intent_result["intent"]
   162→   162→    target = intent_result["target"]
   163→   163→
   164→   164→    # ── QUERY INTENTS ────────────────────────────────────
   165→   165→    if is_query_intent(intent):
   166→   166→        response = await _handle_query(intent, target, tenant_id, signals, db)
   167→   167→        # Log to audit
   168→   168→        await db.create_audit_entry(
   169→   169→            tenant_id=tenant_id,
   170→   170→            action=f"query:{intent}",
   171→   171→            actor_email=auth_result.email,
   172→   172→            target_type="query",
   173→   173→            payload={"intent": intent, "target": target},
   174→   174→        )
   175→   175→        return response
   176→   176→
   177→   177→    # ── CONTROL INTENTS ──────────────────────────────────
   178→   178→    if is_control_intent(intent):
   179→   179→        response = await _handle_control(intent, target, tenant_id, auth_result, db,
   180→   180→                                         raw_input=raw_input)
   181→   181→        return response
   182→   182→
   183→   183→    # ── APPROVAL INTENTS ─────────────────────────────────
   184→   184→    if is_approval_intent(intent):
   185→   185→        response = await _handle_approval(intent, target, tenant_id, auth_result, db)
   186→   186→        return response
   187→   187→
   188→   188→    # ── EMERGENCY INTENTS ────────────────────────────────
   189→   189→    if is_emergency_intent(intent):
   190→   190→        response = await _handle_emergency(intent, target, tenant_id, auth_result, db,
   191→   191→                                          raw_input=raw_input)
   192→   192→        return response
   193→   193→
   194→   194→    # ── EXPLAIN / TEACH / AGENT ──────────────────────────
   195→   195→    if intent == "explain_ticket":
   196→   196→        return await _handle_explain_ticket(target, tenant_id, signals, db)
   197→   197→    if intent == "explain_flag":
   198→   198→        return await _handle_explain_flags(tenant_id, db)
   199→   199→    if intent == "teach_skill":
   200→   200→        return await _handle_teach_skill(target, tenant_id, auth_result, db)
   201→   201→    if intent == "create_agent":
   202→   202→        return await _handle_create_agent(target, tenant_id, auth_result, db)
   203→   203→    if intent == "control_approval_override":
   204→   204→        # Wave 3: approval override via executor
   205→   205→        result = await execute_command(
   206→   206→            intent=intent, target=target, tenant_id=tenant_id,
   207→   207→            actor_email=auth_result.email, raw_input=raw_input,
   208→   208→        )
   209→   209→        return result.response
   210→   210→
   211→   211→    return f"I understood your command but don't have a handler for '{intent}' yet."

   261→   261→        return (

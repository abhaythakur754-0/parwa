     1→"""
     2→PARWA Integration Catalog — Backend Mirror
     3→
     4→Single source of truth for the backend. Mirrors the frontend catalog
     5→in src/lib/integration-catalog.ts. When adding an integration, update
     6→BOTH files.
     7→
     8→Per D2: All variants get UNLIMITED integrations.
     9→Per D4: Three tiers — Pre-built (Tier 1), OpenAPI Import (Tier 2), Custom REST (Tier 3).
    10→Per D6: Pre-written HTTP test calls — NO AI tokens spent.
    11→Per GAP 2: Universal API key system with 5 auth types.
    12→Per GAP 3: Full catalog per industry (suggestions, not restrictions).
    13→"""

    59→# ── Data Classes ─────────────────────────────────────────────────────────
    60→
    61→
    62→class AuthField:
    63→    """A single field in an auth schema."""
    64→
    65→    def __init__(self, name: str, label: str, type: str = "text",
    66→                 required: bool = True, placeholder: str = ""):
    67→        self.name = name
    68→        self.label = label
    69→        self.type = type
    70→        self.required = required
    71→        self.placeholder = placeholder
    72→
    73→    def to_dict(self) -> Dict[str, Any]:
    74→        return {
    75→            "name": self.name,
    76→            "label": self.label,
    77→            "type": self.type,
    78→            "required": self.required,
    79→            "placeholder": self.placeholder,
    80→        }
    81→
    82→
    83→class AuthSchema:
    84→    """Defines how an integration authenticates."""
    85→
    86→    def __init__(self, auth_type: AuthType, fields: List[AuthField],
    87→                 header_name: str = "", query_param_name: str = "",
    88→                 redirect_uri: str = ""):
    89→        self.auth_type = auth_type
    90→        self.fields = fields
    91→        self.header_name = header_name
    92→        self.query_param_name = query_param_name
    93→        self.redirect_uri = redirect_uri
    94→
    95→    def to_dict(self) -> Dict[str, Any]:
    96→        d: Dict[str, Any] = {
    97→            "type": self.auth_type.value,
    98→            "fields": [f.to_dict() for f in self.fields],
    99→        }
   100→        if self.header_name:
   101→            d["headerName"] = self.header_name
   102→        if self.query_param_name:
   103→            d["queryParamName"] = self.query_param_name
   104→        if self.redirect_uri:
   105→            d["redirectUri"] = self.redirect_uri
   106→        return d
   107→
   108→
   109→class TestConnectionConfig:
   110→    """Pre-written HTTP test call per D6 — NO AI."""
   111→
   112→    def __init__(self, method: str, url_template: str,
   113→                 headers_template: Optional[Dict[str, str]] = None,
   114→                 success_check: str = "status_200",
   115→                 success_message: str = "Connected"):
   116→        self.method = method.upper()
   117→        self.url_template = url_template
   118→        self.headers_template = headers_template or {}
   119→        self.success_check = success_check
   120→        self.success_message = success_message
   121→
   122→    def to_dict(self) -> Dict[str, Any]:
   123→        return {
   124→            "method": self.method,
   125→            "urlTemplate": self.url_template,
   126→            "headersTemplate": self.headers_template,
   127→            "successCheck": self.success_check,
   128→            "successMessage": self.success_message,
   129→        }
   130→
   131→
   132→class IntegrationDefinition:
   133→    """Complete definition of an integration in the catalog."""
   134→
   135→    def __init__(
   136→        self,
   137→        key: str,
   138→        name: str,
   139→        description: str,
   140→        category: IntegrationCategory,
   141→        tier: IntegrationTier,
   142→        auth_schema: AuthSchema,
   143→        test_connection: TestConnectionConfig,
   144→        suggested_industries: List[ParwaIndustry],
   145→        icon_id: str = "",
   146→        color_gradient: str = "",
   147→        available: bool = True,
   148→        available_for_variants: Optional[List[str]] = None,
   149→    ):
   150→        self.key = key
   151→        self.name = name
   152→        self.description = description
   153→        self.category = category
   154→        self.tier = tier
   155→        self.auth_schema = auth_schema
   156→        self.test_connection = test_connection
   157→        self.suggested_industries = suggested_industries
   158→        self.icon_id = icon_id
   159→        self.color_gradient = color_gradient
   160→        self.available = available
   161→        self.available_for_variants = available_for_variants or []
   162→
   163→    def to_dict(self) -> Dict[str, Any]:
   164→        d: Dict[str, Any] = {
   165→            "key": self.key,
   166→            "name": self.name,
   167→            "description": self.description,
   168→            "category": self.category.value,
   169→            "tier": self.tier.value,
   170→            "authSchema": self.auth_schema.to_dict(),
   171→            "testConnection": self.test_connection.to_dict(),
   172→            "suggestedIndustries": [i.value for i in self.suggested_industries],
   173→            "iconId": self.icon_id,
   174→            "colorGradient": self.color_gradient,
   175→            "available": self.available,
   176→        }
   177→        if self.available_for_variants:
   178→            d["availableForVariants"] = self.available_for_variants
   179→        return d

   538→# ── Lookup Helpers ───────────────────────────────────────────────────────
   539→
   540→# Build a lookup dict for O(1) access
   541→_CATALOG_BY_KEY: Dict[str, IntegrationDefinition] = {i.key: i for i in CATALOG}
   542→
   543→
   544→def get_catalog() -> List[Dict[str, Any]]:
   545→    """Return the full catalog as a list of dicts."""
   546→    return [i.to_dict() for i in CATALOG]
   547→
   548→
   549→def get_catalog_for_industry(industry: str) -> List[Dict[str, Any]]:
   550→    """Return integrations suggested for an industry.
   551→
   552→    Per D3: 'other' shows ALL integrations (no filtering).
   553→    Suggestions are NOT restrictions — clients can always connect outside their industry.
   554→    """
   555→    if industry == "other":
   556→        return [i.to_dict() for i in CATALOG if i.available]
   557→
   558→    try:
   559→        parwa_industry = ParwaIndustry(industry.lower())
   560→    except ValueError:
   561→        return [i.to_dict() for i in CATALOG if i.available]
   562→
   563→    return [
   564→        i.to_dict() for i in CATALOG
   565→        if i.available and parwa_industry in i.suggested_industries
   566→    ]
   567→
   568→
   569→def get_integration_by_key(key: str) -> Optional[IntegrationDefinition]:
   570→    """Look up an integration definition by its key."""
   571→    return _CATALOG_BY_KEY.get(key)
   572→
   573→
   574→def get_catalog_grouped_by_category(industry: str) -> Dict[str, List[Dict[str, Any]]]:
   575→    """Return integrations grouped by category for an industry."""
   576→    items = get_catalog_for_industry(industry)
   577→    grouped: Dict[str, List[Dict[str, Any]]] = {}
   578→    for item in items:
   579→        cat = item["category"]
   580→        if cat not in grouped:
   581→            grouped[cat] = []
   582→        grouped[cat].append(item)
   583→    return grouped

     8→AI employee that replaces a human support agent.
     9→
    10→Architecture:
    11→  Client → Jarvis CC Service → variant_pipeline_bridge → Variant Pipelines

    26→  - get_cc_context:      Get current awareness snapshot (placeholder for Phase 2)
    27→  - update_cc_context:   Update session context with new data

    85→def get_or_create_cc_session(
    86→    db: Session,
    87→    user_id: str,
    88→    company_id: str,
    89→    existing_session_id: Optional[str] = None,
    90→) -> JarvisSession:
    91→    """Get an existing customer_care session or create a new one.
    92→
    93→    After handoff, the client needs a persistent customer_care session.
    94→    This function finds or creates one.
    95→
    96→    Args:
    97→        db: SQLAlchemy session.
    98→        user_id: The user's ID (BC-001 scoped).
    99→        company_id: The company's ID (BC-001).
   100→        existing_session_id: If provided, try to resume this specific session.
   101→
   102→    Returns:
   103→        Active JarvisSession with type='customer_care'.
   104→
   105→    Raises:
   106→        NotFoundError: If existing_session_id provided but not found/not CC type.
   107→        ValidationError: If company_id is missing.
   108→    """
   109→    if not company_id:
   110→        raise ValidationError(
   111→            message="company_id is required for customer care sessions",
   112→            details={"user_id": user_id},
   113→        )
   114→
   115→    # If a specific session ID is provided, try to resume it
   116→    if existing_session_id:
   117→        session = (
   118→            db.query(JarvisSession)
   119→            .filter(
   120→                JarvisSession.id == existing_session_id,
   121→                JarvisSession.user_id == user_id,
   122→                JarvisSession.company_id == company_id,
   123→                JarvisSession.type == "customer_care",
   124→                JarvisSession.is_active.is_(True),
   125→            )
   126→            .first()
   127→        )
   128→        if session:
   129→            _maybe_reset_daily_counter(db, session)
   130→            session.updated_at = datetime.now(timezone.utc)
   131→            db.flush()
   132→            return session
   133→        # Not found or wrong type — fall through to create new
   134→
   135→    # Look for any active customer_care session for this user+company
   136→    active_session = (
   137→        db.query(JarvisSession)
   138→        .filter(
   139→            JarvisSession.user_id == user_id,
   140→            JarvisSession.company_id == company_id,
   141→            JarvisSession.type == "customer_care",
   142→            JarvisSession.is_active.is_(True),
   143→        )
   144→        .order_by(JarvisSession.created_at.desc())
   145→        .first()
   146→    )
   147→
   148→    if active_session:
   149→        _maybe_reset_daily_counter(db, active_session)
   150→        active_session.updated_at = datetime.now(timezone.utc)
   151→        db.flush()
   152→        return active_session
   153→
   154→    # Create a new customer care session
   155→    # Inherit variant_tier from the most recent handoff session if available
   156→    variant_tier = "mini_parwa"  # safe default
   157→    variant_instance_id = ""
   158→    industry = "general"
   159→
   160→    # Try to find context from the handoff (onboarding session)
   161→    handoff_session = (
   162→        db.query(JarvisSession)
   163→        .filter(
   164→            JarvisSession.user_id == user_id,
   165→            JarvisSession.company_id == company_id,
   166→            JarvisSession.type == "onboarding",
   167→            JarvisSession.handoff_completed.is_(True),
   168→        )
   169→        .order_by(JarvisSession.created_at.desc())
   170→        .first()
   171→    )
   172→
   173→    care_context: Dict[str, Any] = {
   174→        "variant_tier": variant_tier,
   175→        "variant_instance_id": variant_instance_id,
   176→        "industry": industry,
   177→        "mode": "customer_care",
   178→        "created_via": "jarvis_cc_service",
   179→        "awareness_enabled": False,  # Phase 2 will enable this
   180→        "proactive_alerts": [],
   181→        "last_pipeline_metadata": {},
   182→    }
   183→
   184→    if handoff_session:
   185→        try:
   186→            handoff_ctx = _safe_parse_json(handoff_session.context_json)
   187→            if handoff_ctx.get("variant_tier"):
   188→                care_context["variant_tier"] = handoff_ctx["variant_tier"]
   189→                variant_tier = handoff_ctx["variant_tier"]
   190→            if handoff_ctx.get("variant_instance_id"):
   191→                care_context["variant_instance_id"] = handoff_ctx["variant_instance_id"]
   192→                variant_instance_id = handoff_ctx["variant_instance_id"]
   193→            if handoff_ctx.get("industry"):
   194→                care_context["industry"] = handoff_ctx["industry"]
   195→                industry = handoff_ctx["industry"]
   196→            if handoff_ctx.get("business_email"):
   197→                care_context["business_email"] = handoff_ctx["business_email"]
   198→            if handoff_ctx.get("email_verified"):
   199→                care_context["email_verified"] = handoff_ctx["email_verified"]
   200→            care_context["onboarding_session_id"] = str(handoff_session.id)
   201→            care_context["onboarding_completed_at"] = (
   202→                handoff_session.updated_at.isoformat()
   203→                if handoff_session.updated_at
   204→                else datetime.now(timezone.utc).isoformat()
   205→            )
   206→        except Exception:
   207→            logger.exception(
   208→                "Failed to parse handoff context for user=%s company=%s",
   209→                user_id, company_id,
   210→            )
   211→
   212→    new_session = JarvisSession(
   213→        user_id=user_id,
   214→        company_id=company_id,
   215→        type="customer_care",
   216→        context_json=json.dumps(care_context),
   217→        message_count_today=0,
   218→        total_message_count=0,
   219→        pack_type="free",  # CC sessions don't use pack limits
   220→        is_active=True,
   221→    )
   222→    db.add(new_session)
   223→    db.flush()
   224→
   225→    # Generate a welcome message for the new CC session
   226→    welcome_msg = JarvisMessage(
   227→        session_id=str(new_session.id),
   228→        role="jarvis",
   229→        content=_build_cc_welcome(variant_tier, industry),
   230→        message_type="text",
   231→        metadata_json=json.dumps({
   232→            "variant_tier": variant_tier,
   233→            "industry": industry,
   234→            "session_type": "customer_care",
   235→        }),
   236→    )
   237→    db.add(welcome_msg)
   238→    db.flush()
   239→
   240→    logger.info(
   241→        "cc_session_created: session_id=%s, user_id=%s, company_id=%s, "
   242→        "variant_tier=%s, industry=%s",
   243→        new_session.id, user_id, company_id, variant_tier, industry,
   244→    )
   245→
   246→    return new_session
   247→

   249→def get_cc_session(
   250→    db: Session,
   251→    session_id: str,
   252→    user_id: str,
   253→    company_id: str,
   254→) -> JarvisSession:
   255→    """Get a customer_care session by ID with security scoping.
   256→
   257→    Args:
   258→        db: SQLAlchemy session.
   259→        session_id: The session ID to retrieve.
   260→        user_id: User ID for security scoping.
   261→        company_id: Company ID for BC-001 tenant isolation.
   262→
   263→    Returns:
   264→        JarvisSession with type='customer_care'.
   265→
   266→    Raises:
   267→        NotFoundError: If session not found or not customer_care type.
   268→    """
   269→    session = (
   270→        db.query(JarvisSession)
   271→        .filter(
   272→            JarvisSession.id == session_id,
   273→            JarvisSession.user_id == user_id,
   274→            JarvisSession.company_id == company_id,
   275→        )
   276→        .first()
   277→    )
   278→    if not session:
   279→        raise NotFoundError(
   280→            message="Customer care session not found",
   281→            details={"session_id": session_id},
   282→        )
   283→    if session.type != "customer_care":
   284→        raise NotFoundError(
   285→            message="Session is not a customer care session",
   286→            details={"session_id": session_id, "type": session.type},
   287→        )
   288→    return session

   291→def get_cc_context(
   292→    db: Session,
   293→    session_id: str,
   294→    user_id: str,
   295→    company_id: str,
   296→) -> Dict[str, Any]:
   297→    """Get the current context snapshot for a customer care session.
   298→
   299→    Returns the parsed context_json with additional runtime data
   300→    (variant instance status, recent ticket counts, etc.).
   301→
   302→    Args:
   303→        db: SQLAlchemy session.
   304→        session_id: Session ID.
   305→        user_id: User ID for security scoping.
   306→        company_id: Company ID for BC-001.
   307→
   308→    Returns:
   309→        Dict with full context including runtime enrichment.
   310→    """
   311→    session = get_cc_session(db, session_id, user_id, company_id)
   312→    ctx = _safe_parse_json(session.context_json)
   313→
   314→    # Enrich with runtime data (BC-008: each enrichment is independently wrapped)
   315→    ctx["runtime"] = {}
   316→
   317→    # Current variant instance status
   318→    try:
   319→        from app.services.variant_instance_service import get_instance
   320→        instance_id = ctx.get("variant_instance_id")
   321→        if instance_id:
   322→            instance = get_instance(db, company_id, instance_id)
   323→            if instance:
   324→                ctx["runtime"]["instance_status"] = instance.status
   325→                ctx["runtime"]["active_tickets"] = instance.active_tickets_count
   326→                ctx["runtime"]["total_handled"] = instance.total_tickets_handled
   327→    except Exception:
   328→        pass  # BC-008: enrichment failure is non-fatal
   329→
   330→    # Recent ticket count
   331→    try:
   332→        from database.models.tickets import Ticket
   333→        from sqlalchemy import func
   334→        today_start = datetime.now(timezone.utc).replace(
   335→            hour=0, minute=0, second=0, microsecond=0,
   336→        )
   337→        ticket_count = (
   338→            db.query(func.count(Ticket.id))
   339→            .filter(
   340→                Ticket.company_id == company_id,
   341→                Ticket.created_at >= today_start,
   342→            )
   343→            .scalar()
   344→        )
   345→        ctx["runtime"]["tickets_today"] = ticket_count or 0
   346→    except Exception:
   347→        pass
   348→
   349→    # Emergency state
   350→    try:
   351→        from database.models.core import EmergencyState
   352→        emergency = (
   353→            db.query(EmergencyState)
   354→            .filter(EmergencyState.company_id == company_id)
   355→            .order_by(EmergencyState.created_at.desc())
   356→            .first()
   357→        )
   358→        if emergency:
   359→            ctx["runtime"]["ai_paused"] = emergency.is_paused
   360→            ctx["runtime"]["paused_channels"] = emergency.paused_channels
   361→    except Exception:
   362→        pass
   363→
   364→    return ctx

   367→def update_cc_context(
   368→    db: Session,
   369→    session_id: str,
   370→    user_id: str,
   371→    company_id: str,
   372→    partial_updates: Dict[str, Any],
   373→) -> JarvisSession:
   374→    """Merge partial updates into the customer care session context.
   375→
   376→    Only provided keys are updated. Existing keys are preserved.
   377→    This is how the awareness engine (Phase 2) and command executor
   378→    (Phase 3) will update the session.
   379→
   380→    Args:
   381→        db: SQLAlchemy session.
   382→        session_id: Session ID.
   383→        user_id: User ID for security.
   384→        company_id: Company ID for BC-001.
   385→        partial_updates: Dict of keys to merge into context.
   386→
   387→    Returns:
   388→        Updated JarvisSession.
   389→
   390→    Raises:
   391→        NotFoundError: If session not found.
   392→        ValidationError: If context_json would exceed size limit.
   393→    """
   394→    session = get_cc_session(db, session_id, user_id, company_id)
   395→    ctx = _safe_parse_json(session.context_json)
   396→
   397→    # Merge updates (don't allow overwriting critical keys from outside)
   398→    protected_keys = {"variant_tier", "variant_instance_id", "industry"}
   399→    for key, value in partial_updates.items():
   400→        if key not in protected_keys or value is not None:
   401→            ctx[key] = value
   402→
   403→    new_context = json.dumps(ctx)
   404→    if len(new_context) > CC_MAX_CONTEXT_JSON_SIZE:
   405→        raise ValidationError(
   406→            message="Context size would exceed limit",
   407→            details={"size": len(new_context), "limit": CC_MAX_CONTEXT_JSON_SIZE},
   408→        )
   409→
   410→    session.context_json = new_context
   411→    session.updated_at = datetime.now(timezone.utc)
   412→    db.flush()
   413→
   414→    return session

   422→def send_cc_message(
   423→    db: Session,
   424→    session_id: str,
   425→    user_id: str,
   426→    company_id: str,
   427→    user_message: str,
   428→    ticket_id: Optional[str] = None,
   429→    channel: str = "chat",
   430→) -> Tuple[JarvisMessage, JarvisMessage, Dict[str, Any]]:
   431→    """Process a customer care message and generate AI response.
   432→
   433→    This is the main entry point for Jarvis in customer care mode.
   434→
   435→    Flow:
   436→      1. Validate session (must be customer_care type)
   437→      2. Check message limits
   438→      3. Save user message
   439→      4. Build system prompt with tenant context
   440→      5. Route through variant pipeline bridge
   441→      6. Save AI response with pipeline metadata
   442→      7. Update session context
   443→      8. Return both messages + metadata
   444→
   445→    Fallback Chain (BC-008):
   446→      variant_pipeline_bridge → legacy ai_pipeline → direct AI provider
   447→
   448→    Args:
   449→        db: SQLAlchemy session.
   450→        session_id: The customer care session ID.
   451→        user_id: User ID for security scoping.
   452→        company_id: Company ID for BC-001 tenant isolation.
   453→        user_message: The raw message from the client.
   454→        ticket_id: Optional ticket ID if message is within a ticket context.
   455→        channel: Communication channel (chat, email, sms, voice).
   456→
   457→    Returns:
   458→        Tuple of (user_message_obj, ai_message_obj, pipeline_metadata)
   459→
   460→    Raises:
   461→        NotFoundError: If session not found or not customer_care type.
   462→        RateLimitError: If daily limit exceeded.
   463→        ValidationError: If message is empty or too long.
   464→    """
   465→    start_time = time.monotonic()
   466→
   467→    # ── Step 1: Validate session ──
   468→    session = get_cc_session(db, session_id, user_id, company_id)
   469→
   470→    # ── Step 2: Validate message ──
   471→    if not user_message or not user_message.strip():
   472→        raise ValidationError(
   473→            message="Message cannot be empty",
   474→            details={"session_id": session_id},
   475→        )
   476→    if len(user_message) > 10000:
   477→        raise ValidationError(
   478→            message="Message too long (max 10,000 characters)",
   479→            details={"length": len(user_message)},
   480→        )
   481→
   482→    # ── Step 3: Check message limits ──
   483→    _maybe_reset_daily_counter(db, session)
   484→    if session.message_count_today >= CC_DAILY_MESSAGE_LIMIT:
   485→        raise RateLimitError(
   486→            message="Daily message limit reached for customer care",
   487→            details={
   488→                "limit": CC_DAILY_MESSAGE_LIMIT,
   489→                "used": session.message_count_today,
   490→            },
   491→        )
   492→
   493→    # ── Step 4: Save user message ──
   494→    user_msg = JarvisMessage(
   495→        session_id=session_id,
   496→        role="user",
   497→        content=user_message,
   498→        message_type="text",
   499→        metadata_json=json.dumps({
   500→            "channel": channel,
   501→            "ticket_id": ticket_id or "",
   502→            "company_id": company_id,
   503→        }),
   504→    )
   505→    db.add(user_msg)
   506→
   507→    # Update counters
   508→    session.message_count_today += 1
   509→    session.total_message_count += 1
   510→    session.last_message_date = datetime.now(timezone.utc)
   511→    session.updated_at = datetime.now(timezone.utc)
   512→    db.flush()
   513→
   514→    # ── Step 5: Get session context ──
   515→    ctx = _safe_parse_json(session.context_json)
   516→    variant_tier = ctx.get("variant_tier", "mini_parwa")
   517→    variant_instance_id = ctx.get("variant_instance_id", "")
   518→    industry = ctx.get("industry", "general")
   519→
   520→    # ── Phase 3: Command Detection ──
   521→    # If the message looks like a command, route it through the command layer
   522→    # instead of the AI pipeline. Commands start with "/" or match known patterns.
   523→    _COMMAND_PREFIXES = ("/", "jarvis ")
   524→    _is_command = False
   525→    _stripped_message = user_message.strip()
   526→
   527→    if any(_stripped_message.lower().startswith(p) for p in _COMMAND_PREFIXES):
   528→        _is_command = True
   529→        # Strip the prefix for command parsing
   530→        for prefix in _COMMAND_PREFIXES:
   531→            if _stripped_message.lower().startswith(prefix):
   532→                _stripped_message = _stripped_message[len(prefix):].strip()
   533→                break
   534→    else:
   535→        # Quick check: does the message match any command pattern?
   536→        try:
   537→            from app.services.jarvis_command_service import parse_natural_language_command
   538→            _quick_parse = parse_natural_language_command(
   539→                company_id=company_id,
   540→                raw_input=_stripped_message,
   541→                session_id=session_id,
   542→            )
   543→            if (_quick_parse.get("confidence", 0) >= 0.65
   544→                    and _quick_parse.get("action") != "unknown"):
   545→                _is_command = True
   546→        except Exception:
   547→            pass  # BC-008: command detection failure is non-fatal
   548→
   549→    if _is_command:
   550→        try:
   551→            from app.services import jarvis_command_service
   552→            from database.models.jarvis_cc import JarvisCommand
   553→
   554→            # Receive and parse the command
   555→            command = jarvis_command_service.receive_command(
   556→                db=db,
   557→                company_id=company_id,
   558→                session_id=session_id,
   559→                raw_input=_stripped_message,
   560→                source="chat",
   561→                user_id=user_id,
   562→            )
   563→
   564→            # Execute the command
   565→            cmd_result = jarvis_command_service.execute_command(
   566→                db=db,
   567→                company_id=company_id,
   568→                command_id=str(command.id),
   569→                session_id=session_id,
   570→                user_id=user_id,
   571→            )
   572→
   573→            # Build AI response as a command_response message
   574→            parsed = json.loads(command.command_parsed) if command.command_parsed else {}
   575→            _action = cmd_result.get("action", "unknown")
   576→            _result_data = cmd_result.get("result", {})
   577→            _message = _result_data.get("message", f"Command '{_action}' executed.")
   578→            _undo_hint = " (Undo available)" if command.undo_available else ""
   579→
   580→            ai_msg = JarvisMessage(
   581→                session_id=session_id,
   582→                role="jarvis",
   583→                content=_message + _undo_hint,
   584→                message_type="command_response",
   585→                metadata_json=json.dumps({
   586→                    "command_id": str(command.id),
   587→                    "action": _action,
   588→                    "intent": command.command_intent,
   589→                    "confidence": float(command.confidence) if command.confidence else None,
   590→                    "result": cmd_result.get("result", {}),
   591→                    "execution_time_ms": cmd_result.get("execution_time_ms", 0),
   592→                    "undo_available": command.undo_available,
   593→                    "suggestion": parsed.get("suggestion"),
   594→                    "fallback": "command_layer",
   595→                }),
   596→            )
   597→            db.add(ai_msg)
   598→
   599→            # Update session context
   600→            ctx["last_command_at"] = datetime.now(timezone.utc).isoformat()
   601→            ctx["last_command_action"] = _action

   608→                session_id, _action, total_ms,

   612→                "pipeline_status": "command_executed",

   775→                "intent": legacy_result.intent_type,
   776→                "confidence": legacy_result.confidence_score,
   777→                "technique": legacy_result.technique_used,
   778→                "model": legacy_result.model_used,
   779→                "fallback": "legacy_ai_pipeline",

   782→        except Exception as inner_exc:

   784→            logger.error(

   793→                    system_prompt, history, user_message, ctx,
   794→                )

   796→                pipeline_status = "direct_ai_fallback"
   797→            except Exception:

   799→                ai_message_type = "error"
   800→                metadata = {"error_type": "all_pipelines_failed"}
   801→                pipeline_status = "all_failed"

   810→        session_id=session_id,
   811→        role="jarvis",
   812→        content=ai_content,
   813→        message_type=ai_message_type,
   814→        metadata_json=json.dumps(metadata),
   815→    )
   816→    db.add(ai_msg)
   817→    db.flush()
   818→
   819→    # ── Step 11: Update session context ──
   820→    ctx["last_message_at"] = datetime.now(timezone.utc).isoformat()
   821→    ctx["total_messages"] = session.total_message_count
   822→    ctx["pipeline_status"] = pipeline_status
   823→
   824→    session.context_json = json.dumps(ctx)
   825→    db.flush()
   826→
   827→    # ── Step 12: Audit log ──
   828→    try:
   829→        from app.services.audit_service import log_audit
   830→        log_audit(
   831→            company_id=company_id,
   832→            actor_id=user_id,
   833→            actor_type="user",
   834→            action="cc_message_sent",
   835→            resource_type="jarvis_cc_session",
   836→            resource_id=session_id,
   837→            old_value=None,
   838→            new_value={
   839→                "pipeline_status": pipeline_status,
   840→                "variant_tier": variant_tier,
   841→                "channel": channel,
   842→            },
   843→            ip_address=None,
   844→            user_agent=None,
   845→            db=db,
   846→        )
   847→    except Exception:
   848→        pass  # BC-008
   849→
   850→    total_ms = round((time.monotonic() - start_time) * 1000, 2)
   851→    logger.info(
   852→        "cc_message_complete: session=%s, pipeline=%s, total_ms=%s",
   853→        session_id, pipeline_status, total_ms,
   854→    )
   855→
   856→    return user_msg, ai_msg, metadata

   859→def get_cc_history(
   860→    db: Session,
   861→    session_id: str,
   862→    user_id: str,
   863→    company_id: str,
   864→    limit: int = 50,
   865→    offset: int = 0,
   866→) -> Tuple[List[Dict[str, Any]], int]:
   867→    """Get paginated message history for a customer care session.
   868→
   869→    Args:
   870→        db: SQLAlchemy session.
   871→        session_id: Session ID.
   872→        user_id: User ID for security.
   873→        company_id: Company ID for BC-001.
   874→        limit: Max messages to return.
   875→        offset: Pagination offset.
   876→
   877→    Returns:
   878→        Tuple of (messages_list, total_count).
   879→    """
   880→    # Security: verify session belongs to user+company
   881→    get_cc_session(db, session_id, user_id, company_id)
   882→
   883→    query = (
   884→        db.query(JarvisMessage)
   885→        .filter(JarvisMessage.session_id == session_id)
   886→        .order_by(JarvisMessage.created_at.asc())
   887→    )
   888→
   889→    total = query.count()
   890→    messages = query.offset(offset).limit(limit).all()
   891→
   892→    message_list = []
   893→    for msg in messages:
   894→        msg_dict = {
   895→            "id": msg.id,
   896→            "role": msg.role,
   897→            "content": msg.content,
   898→            "message_type": msg.message_type,
   899→            "metadata": _safe_parse_json(msg.metadata_json) if msg.metadata_json else {},
   900→            "created_at": msg.created_at.isoformat() if msg.created_at else None,
   901→        }
   902→        message_list.append(msg_dict)
   903→
   904→    return message_list, total

   912→def build_cc_system_prompt(
   913→    db: Session,
   914→    session_id: str,
   915→    company_id: str,
   916→    context: Optional[Dict[str, Any]] = None,
   917→) -> str:
   918→    """Build a rich system prompt for Customer Care Jarvis.
   919→
   920→    Unlike onboarding Jarvis (which sells), CC Jarvis is an AI employee
   921→    that handles support tickets and assists the client in managing
   922→    their customer care operations.
   923→
   924→    The prompt includes:
   925→    - Tenant-specific context (variant tier, industry, channels)
   926→    - Brand voice and tone guidelines
   927→    - Knowledge base context
   928→    - Current operational status (if available)
   929→    - Conversation context (stage, turn count)
   930→
   931→    Args:
   932→        db: SQLAlchemy session.
   933→        session_id: Session ID for context lookup.
   934→        company_id: Company ID for BC-001.
   935→        context: Pre-loaded context (optional, loaded from DB if not provided).
   936→
   937→    Returns:
   938→        Complete system prompt string.
   939→    """
   940→    # Load context if not provided
   941→    if context is None:
   942→        session = db.query(JarvisSession).filter(
   943→            JarvisSession.id == session_id,
   944→        ).first()
   945→        context = _safe_parse_json(session.context_json) if session else {}
   946→
   947→    variant_tier = context.get("variant_tier", "mini_parwa")
   948→    industry = context.get("industry", "general")
   949→    instance_id = context.get("variant_instance_id", "")
   950→
   951→    # Base CC prompt — fundamentally different from onboarding
   952→    prompt_parts = [
   953→        _CC_BASE_PROMPT,
   954→        f"\n## CURRENT CONFIGURATION",
   955→        f"- Variant Tier: {variant_tier}",
   956→        f"- Industry: {industry}",
   957→        f"- Instance ID: {instance_id[:8]}..." if instance_id else "- Instance: default",
   958→    ]
   959→
   960→    # Tier-specific capabilities
   961→    tier_caps = _get_tier_capabilities(variant_tier)
   962→    if tier_caps:
   963→        prompt_parts.append(f"\n## YOUR CAPABILITIES ({variant_tier.upper()})")
   964→        prompt_parts.append(tier_caps)
   965→
   966→    # Brand voice (from company settings)
   967→    try:
   968→        from database.models.core import CompanySetting
   969→        settings = (
   970→            db.query(CompanySetting)
   971→            .filter(CompanySetting.company_id == company_id)
   972→            .first()
   973→        )
   974→        if settings:
   975→            if settings.brand_voice:
   976→                prompt_parts.append(f"\n## BRAND VOICE")
   977→                prompt_parts.append(settings.brand_voice)
   978→            if settings.tone_guidelines:
   979→                prompt_parts.append(f"\n## TONE GUIDELINES")
   980→                prompt_parts.append(settings.tone_guidelines)
   981→            if settings.prohibited_phrases:
   982→                prohibited = _safe_parse_json(settings.prohibited_phrases)
   983→                if prohibited and isinstance(prohibited, list):
   984→                    prompt_parts.append(f"\n## PROHIBITED PHRASES")
   985→                    prompt_parts.append(
   986→                        "NEVER use these phrases: " + ", ".join(prohibited)
   987→                    )
   988→    except Exception:
   989→        pass  # BC-008
   990→
   991→    # Knowledge base context
   992→    try:
   993→        from app.services.jarvis_knowledge_service import search_and_format_knowledge
   994→        kb_context = search_and_format_knowledge("customer care support", industry)
   995→        if kb_context:
   996→            prompt_parts.append(f"\n## KNOWLEDGE BASE")
   997→            prompt_parts.append(kb_context[:2000])  # Limit KB context size
   998→    except Exception:
   999→        pass
  1000→
  1001→    # Recent pipeline results (for continuity)
  1002→    last_pipeline = context.get("last_pipeline_metadata", {})
  1003→    if last_pipeline:
  1004→        prompt_parts.append(f"\n## LAST INTERACTION METADATA")
  1005→        if last_pipeline.get("classification_intent"):
  1006→            prompt_parts.append(
  1007→                f"- Last intent classified: {last_pipeline['classification_intent']}"
  1008→            )
  1009→        if last_pipeline.get("quality_score"):
  1010→            prompt_parts.append(
  1011→                f"- Last quality score: {last_pipeline['quality_score']:.1f}/1.0"
  1012→            )
  1013→        if last_pipeline.get("technique_used"):
  1014→            prompt_parts.append(
  1015→                f"- Last technique used: {last_pipeline['technique_used']}"
  1016→            )
  1017→
  1018→    return "\n".join(prompt_parts)

  1303→def _call_ai_provider_fallback(
  1304→    system_prompt: str,
  1305→    history: List[Dict[str, str]],
  1306→    user_message: str,
  1307→    context: Dict[str, Any],
  1308→) -> Tuple[str, str, Dict[str, Any], List[Dict[str, Any]]]:
  1309→    """Direct AI provider call as last-resort fallback.
  1310→
  1311→    Tries Cerebras → Groq → Google AI in order.
  1312→
  1313→    Args:
  1314→        system_prompt: System prompt for the AI.
  1315→        history: Recent conversation history.
  1316→        user_message: The user's raw message.
  1317→        context: Session context.
  1318→
  1319→    Returns:
  1320→        Tuple of (content, message_type, metadata, knowledge_used)
  1321→    """
  1322→    messages = [{"role": "system", "content": system_prompt}]
  1323→    for msg in history[-CC_MAX_CONTEXT_MESSAGES:]:
  1324→        role = msg.get("role", "user")
  1325→        if role == "jarvis":
  1326→            role = "assistant"
  1327→        messages.append({"role": role, "content": msg.get("content", "")})
  1328→    messages.append({"role": "user", "content": user_message})
  1329→
  1330→    # Try providers in order
  1331→    content = None
  1332→    try:
  1333→        from app.services.jarvis_service import _try_ai_providers
  1334→        content = _try_ai_providers(messages)
  1335→    except Exception:
  1336→        pass
  1337→
  1338→    if not content:
  1339→        content = _get_friendly_error_message()
  1340→
  1341→    return content, "direct_ai", {"fallback": "direct_ai_provider"}, []

   134→   134→   134→
   135→   135→   135→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   136→   136→   136→  agentsReplaced: number;
   137→   137→   137→  humanCost: number;
   138→   138→   138→  savings: number;
   139→   139→   139→  savingsPercent: number;
   140→   140→   140→} {
   141→   141→   141→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   142→   142→   142→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   143→   143→   143→  const savings = Math.max(0, humanCost - monthlyCost);
   144→   144→   144→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   145→   145→   145→  return { agentsReplaced, humanCost, savings, savingsPercent };
   146→   146→   146→}

   374→   374→   374→            onClick={onRemove}

    68→// ── Helpers ──────────────────────────────────────────────────────────
    69→
    70→function formatRelativeDate(dateStr: string): string {
    71→  const diff = Date.now() - new Date(dateStr).getTime();
    72→  const mins = Math.floor(diff / 60000);
    73→  if (mins < 1) return 'just now';
    74→  if (mins < 60) return `${mins}m ago`;
    75→  const hrs = Math.floor(mins / 60);
    76→  if (hrs < 24) return `${hrs}h ago`;
    77→  const days = Math.floor(hrs / 24);
    78→  if (days < 7) return `${days}d ago`;
    79→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
    80→}
    81→
    82→function formatDate(dateStr: string): string {
    83→  return new Date(dateStr).toLocaleDateString('en-US', {
    84→    month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit',
    85→  });
    86→}
    87→
    88→function truncate(str: string, max: number): string {
    89→  if (str.length <= max) return str;
    90→  return str.slice(0, max).trim() + '…';
    91→}

    93→function qualityColor(score: number): string {

   232→function GuidanceModal({
   233→  escalation,
   234→  onClose,
   235→}: {
   236→  escalation: Escalation;
   237→  onClose: () => void;
   238→}) {
   239→  const [guidance, setGuidance] = useState(escalation.human_guidance || '');
   240→  const [autoResume, setAutoResume] = useState(false);
   241→  const [submitting, setSubmitting] = useState(false);
   242→  const provideGuidance = useEscalationStore((s) => s.provideGuidance);
   243→  const resumeEscalation = useEscalationStore((s) => s.resumeEscalation);
   244→  const closeModal = useEscalationStore((s) => s.closeModal);
   245→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   246→
   247→  useEffect(() => {
   248→    textareaRef.current?.focus();
   249→  }, []);
   250→
   251→  const handleSubmit = async () => {
   252→    if (guidance.trim().length < 5) {
   253→      toast.error('Guidance must be at least 5 characters.');
   254→      return;
   255→    }
   256→    setSubmitting(true);
   257→    const ok = await provideGuidance(escalation.escalation_id, guidance.trim());
   258→    if (ok) {
   259→      toast.success('Guidance saved. Ticket is now eligible for resume.');
   260→      if (autoResume) {
   261→        await resumeEscalation(escalation.escalation_id);
   262→      }
   263→      closeModal();
   264→    } else {
   265→      toast.error('Failed to save guidance. Please try again.');
   266→      setSubmitting(false);
   267→    }
   268→  };
   269→
   270→  return (
   271→    <motion.div
   272→      initial={{ opacity: 0 }}
   273→      animate={{ opacity: 1 }}
   274→      exit={{ opacity: 0 }}
   275→      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm"
   276→      onClick={(e) => { if (e.target === e.currentTarget) onClose(); }}
   277→    >
   278→      <motion.div
   279→        initial={{ opacity: 0, scale: 0.95, y: 10 }}
   280→        animate={{ opacity: 1, scale: 1, y: 0 }}
   281→        exit={{ opacity: 0, scale: 0.95, y: 10 }}
   282→        transition={{ duration: 0.2 }}
   283→        className="w-full max-w-2xl max-h-[85vh] bg-[#1A1A1A] border border-white/[0.06] rounded-xl overflow-hidden flex flex-col shadow-2xl"

   285→        {/* Header */}
   286→        <div className="px-5 py-4 border-b border-white/[0.06] flex items-start justify-between gap-3">
   287→          <div className="min-w-0">
   288→            <h2 className="text-base font-semibold text-white">Provide Guidance</h2>
   289→            <div className="flex items-center gap-2 mt-1">
   290→              <span className="text-xs font-mono text-blue-400">{escalation.original_ticket_id}</span>
   291→              <span className="text-[10px] text-zinc-600">•</span>
   292→              <span className="text-[10px] text-zinc-500">{escalation.notification_key}</span>
   293→            </div>

   297→            className="w-7 h-7 flex items-center justify-center rounded-lg text-zinc-400 hover:text-white hover:bg-white/5 transition-colors shrink-0"
   298→            aria-label="Close modal"
   299→          >
   300→            <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
   301→              <path strokeLinecap="round" strokeLinejoin="round" d="M6 18 18 6M6 6l12 12" />

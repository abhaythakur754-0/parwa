    62→    62→class AuthField:
    63→    63→    """A single field in an auth schema."""
    64→    64→
    65→    65→    def __init__(self, name: str, label: str, type: str = "text",
    66→    66→                 required: bool = True, placeholder: str = ""):
    67→    67→        self.name = name
    68→    68→        self.label = label
    69→    69→        self.type = type
    70→    70→        self.required = required
    71→    71→        self.placeholder = placeholder
    72→    72→
    73→    73→    def to_dict(self) -> Dict[str, Any]:
    74→    74→        return {
    75→    75→            "name": self.name,
    76→    76→            "label": self.label,
    77→    77→            "type": self.type,
    78→    78→            "required": self.required,
    79→    79→            "placeholder": self.placeholder,
    80→    80→        }
    81→    81→
    82→    82→
    83→    83→class AuthSchema:
    84→    84→    """Defines how an integration authenticates."""
    85→    85→
    86→    86→    def __init__(self, auth_type: AuthType, fields: List[AuthField],
    87→    87→                 header_name: str = "", query_param_name: str = "",
    88→    88→                 redirect_uri: str = ""):
    89→    89→        self.auth_type = auth_type
    90→    90→        self.fields = fields
    91→    91→        self.header_name = header_name
    92→    92→        self.query_param_name = query_param_name
    93→    93→        self.redirect_uri = redirect_uri
    94→    94→
    95→    95→    def to_dict(self) -> Dict[str, Any]:
    96→    96→        d: Dict[str, Any] = {
    97→    97→            "type": self.auth_type.value,
    98→    98→            "fields": [f.to_dict() for f in self.fields],
    99→    99→        }
   100→   100→        if self.header_name:
   101→   101→            d["headerName"] = self.header_name
   102→   102→        if self.query_param_name:
   103→   103→            d["queryParamName"] = self.query_param_name
   104→   104→        if self.redirect_uri:
   105→   105→            d["redirectUri"] = self.redirect_uri
   106→   106→        return d
   107→   107→
   108→   108→
   109→   109→class TestConnectionConfig:
   110→   110→    """Pre-written HTTP test call per D6 — NO AI."""
   111→   111→
   112→   112→    def __init__(self, method: str, url_template: str,
   113→   113→                 headers_template: Optional[Dict[str, str]] = None,
   114→   114→                 success_check: str = "status_200",
   115→   115→                 success_message: str = "Connected"):
   116→   116→        self.method = method.upper()
   117→   117→        self.url_template = url_template
   118→   118→        self.headers_template = headers_template or {}
   119→   119→        self.success_check = success_check
   120→   120→        self.success_message = success_message
   121→   121→
   122→   122→    def to_dict(self) -> Dict[str, Any]:
   123→   123→        return {
   124→   124→            "method": self.method,
   125→   125→            "urlTemplate": self.url_template,
   126→   126→            "headersTemplate": self.headers_template,
   127→   127→            "successCheck": self.success_check,
   128→   128→            "successMessage": self.success_message,
   129→   129→        }

   150→   150→        self.key = key

   163→   163→    def to_dict(self) -> Dict[str, Any]:
   164→   164→        d: Dict[str, Any] = {

   166→   166→            "name": self.name,

   538→   538→# ── Lookup Helpers ───────────────────────────────────────────────────────
   539→   539→
   540→   540→# Build a lookup dict for O(1) access
   541→   541→_CATALOG_BY_KEY: Dict[str, IntegrationDefinition] = {i.key: i for i in CATALOG}
   542→   542→
   543→   543→
   544→   544→def get_catalog() -> List[Dict[str, Any]]:
   545→   545→    """Return the full catalog as a list of dicts."""
   546→   546→    return [i.to_dict() for i in CATALOG]
   547→   547→
   548→   548→
   549→   549→def get_catalog_for_industry(industry: str) -> List[Dict[str, Any]]:
   550→   550→    """Return integrations suggested for an industry.
   551→   551→
   552→   552→    Per D3: 'other' shows ALL integrations (no filtering).
   553→   553→    Suggestions are NOT restrictions — clients can always connect outside their industry.
   554→   554→    """
   555→   555→    if industry == "other":
   556→   556→        return [i.to_dict() for i in CATALOG if i.available]
   557→   557→
   558→   558→    try:
   559→   559→        parwa_industry = ParwaIndustry(industry.lower())
   560→   560→    except ValueError:
   561→   561→        return [i.to_dict() for i in CATALOG if i.available]
   562→   562→
   563→   563→    return [
   564→   564→        i.to_dict() for i in CATALOG
   565→   565→        if i.available and parwa_industry in i.suggested_industries
   566→   566→    ]
   567→   567→
   568→   568→
   569→   569→def get_integration_by_key(key: str) -> Optional[IntegrationDefinition]:
   570→   570→    """Look up an integration definition by its key."""
   571→   571→    return _CATALOG_BY_KEY.get(key)
   572→   572→
   573→   573→
   574→   574→def get_catalog_grouped_by_category(industry: str) -> Dict[str, List[Dict[str, Any]]]:
   575→   575→    """Return integrations grouped by category for an industry."""
   576→   576→    items = get_catalog_for_industry(industry)
   577→   577→    grouped: Dict[str, List[Dict[str, Any]]] = {}
   578→   578→    for item in items:
   579→   579→        cat = item["category"]
   580→   580→        if cat not in grouped:
   581→   581→            grouped[cat] = []
   582→   582→        grouped[cat].append(item)
   583→   583→    return grouped

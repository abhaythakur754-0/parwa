   126→def _serialize_capability(cap) -> dict:
   127→    """Serialize a VariantAICapability ORM object to response dict."""
   128→    config = {}
   129→    if cap.config_json:
   130→        try:
   131→            config = json.loads(cap.config_json)
   132→        except (json.JSONDecodeError, TypeError):
   133→            config = {}
   134→    return {
   135→        "id": cap.id,
   136→        "company_id": cap.company_id,
   137→        "variant_type": cap.variant_type,
   138→        "instance_id": cap.instance_id,
   139→        "feature_id": cap.feature_id,
   140→        "feature_name": cap.feature_name,
   141→        "feature_category": cap.feature_category,
   142→        "technique_tier": cap.technique_tier,
   143→        "is_enabled": cap.is_enabled,
   144→        "config": config,
   145→        "created_at": (
   146→            cap.created_at.isoformat() if cap.created_at else None
   147→        ),
   148→        "updated_at": (
   149→            cap.updated_at.isoformat() if cap.updated_at else None
   150→        ),
   151→    }
   152→
   153→
   154→def _serialize_instance(inst) -> dict:
   155→    """Serialize a VariantInstance ORM object to response dict."""
   156→    channels = []
   157→    if inst.channel_assignment:
   158→        try:
   159→            channels = json.loads(inst.channel_assignment)
   160→        except (json.JSONDecodeError, TypeError):
   161→            channels = []
   162→    capacity = {}
   163→    if inst.capacity_config:
   164→        try:
   165→            capacity = json.loads(inst.capacity_config)
   166→        except (json.JSONDecodeError, TypeError):
   167→            capacity = {}
   168→    return {
   169→        "id": inst.id,
   170→        "company_id": inst.company_id,
   171→        "instance_name": inst.instance_name,
   172→        "variant_type": inst.variant_type,
   173→        "status": inst.status,
   174→        "channel_assignment": channels,
   175→        "capacity_config": capacity,
   176→        "celery_queue_namespace": inst.celery_queue_namespace,
   177→        "redis_partition_key": inst.redis_partition_key,
   178→        "active_tickets_count": inst.active_tickets_count,
   179→        "total_tickets_handled": inst.total_tickets_handled,
   180→        "last_activity_at": (
   181→            inst.last_activity_at.isoformat()
   182→            if inst.last_activity_at else None
   183→        ),
   184→        "created_at": (
   185→            inst.created_at.isoformat() if inst.created_at else None
   186→        ),
   187→        "updated_at": (
   188→            inst.updated_at.isoformat() if inst.updated_at else None
   189→        ),
   190→    }

   197→        "company_id": dist.company_id,
   198→        "instance_id": dist.instance_id,

   200→        "strategy": dist.distribution_strategy,

   202→        "assigned_at": (

   204→        ),
   205→        "completed_at": (

   213→
   214→def _serialize_warmup_state(state) -> dict:
   215→    """Serialize a TenantWarmupState dataclass to response dict."""
   216→    if state is None:
   217→        return {"status": "cold", "message": "No warmup data"}
   218→
   219→    models = {}
   220→    for key, ms in state.models_warmed.items():
   221→        models[key] = {
   222→            "provider": ms.provider,
   223→            "model_id": ms.model_id,
   224→            "tier": ms.tier,
   225→            "status": ms.status.value,
   226→            "warmup_success": ms.warmup_success,
   227→            "warmup_latency_ms": ms.warmup_latency_ms,
   228→            "last_warmed_at": ms.last_warmed_at,
   229→            "error_message": ms.error_message,
   230→        }
   231→
   232→    return {
   233→        "company_id": state.company_id,
   234→        "variant_type": state.variant_type,
   235→        "overall_status": state.overall_status.value,
   236→        "models_warmed": models,
   237→        "time_to_warm_ms": state.time_to_warm_ms,
   238→        "fallback_used": state.fallback_used,
   239→        "started_at": state.started_at,
   240→        "completed_at": state.completed_at,
   241→    }

    42→    async def analyze_company_crm(
    43→        self,
    44→        company_id: str,
    45→    ) -> Dict[str, Any]:
    46→        """Main entry point: Analyze company's CRM and return recommendations.
    47→
    48→        Steps:
    49→        1. Get all connected integrations for this company
    50→        2. Fetch sample data from each connected integration
    51→        3. Analyze data patterns to detect business type and gaps
    52→        4. Use LLM to generate personalized recommendations
    53→        
    54→        Returns:
    55→            Dict with:
    56→            - connected_integrations: list of currently connected
    57→            - data_profile: summary of what data was found
    58→            - recommendations: list of recommended integrations with reasons
    59→            - analysis_summary: plain text explanation
    60→        """
    61→        logger.info("Starting CRM analysis for company %s", company_id)
    62→
    63→        # Step 1: Get connected integrations
    64→        connected = await self._get_connected_integrations(company_id)
    65→        
    66→        # Step 2: Gather data from each integration
    67→        data_profile = await self._gather_data_from_integrations(company_id, connected)
    68→        
    69→        # Step 3: Detect patterns and gaps
    70→        detected_gaps = await self._detect_gaps(data_profile, connected)
    71→        
    72→        # Step 4: Generate LLM-powered recommendations
    73→        recommendations = await self._generate_recommendations(
    74→            company_id=company_id,
    75→            data_profile=data_profile,
    76→            connected=connected,
    77→            gaps=detected_gaps,
    78→        )
    79→
    80→        result = {
    81→            "company_id": company_id,
    82→            "analyzed_at": datetime.now(timezone.utc).isoformat(),
    83→            "connected_integrations": connected,
    84→            "data_profile": data_profile,
    85→            "detected_gaps": detected_gaps,
    86→            "recommendations": recommendations,
    87→            "analysis_summary": self._build_summary(data_profile, recommendations),
    88→        }
    89→
    90→        # Save to database for persistence (onboarding → dashboard transition)
    91→        saved_result = self._save_analysis_result(result)
    92→        if saved_result:
    93→            result["analysis_id"] = str(saved_result.id)
    94→            result["is_saved"] = True
    95→        else:
    96→            result["is_saved"] = False
    97→
    98→        logger.info("CRM analysis complete for company %s: %d recommendations (saved=%s)", 
    99→                    company_id, len(recommendations), result["is_saved"])
   100→        return result

   130→    async def _gather_data_from_integrations(
   131→        self,
   132→        company_id: str,
   133→        connected: List[Dict[str, Any]],
   134→    ) -> Dict[str, Any]:
   135→        """Gather sample data from each connected integration."""
   136→        profile = {
   137→            "total_contacts": 0,
   138→            "total_orders": 0,
   139→            "total_deals": 0,
   140→            "has_products": False,
   141→            "has_shipping_addresses": False,
   142→            "has_payment_data": False,
   143→            "has_email_campaigns": False,
   144→            "has_ticket_data": False,
   145→            "industries_detected": [],
   146→            "data_points": [],
   147→        }
   148→
   149→        for integration in connected:
   150→            int_type = integration["type"]
   151→            
   152→            try:
   153→                if int_type in ("hubspot", "salesforce", "pipedrive"):
   154→                    crm_data = await self._fetch_crm_data(company_id, int_type)
   155→                    profile["total_contacts"] += crm_data.get("contact_count", 0)
   156→                    profile["total_deals"] += crm_data.get("deal_count", 0)
   157→                    profile["has_products"] = profile["has_products"] or crm_data.get("has_products", False)
   158→                    
   159→                elif int_type in ("shopify", "woocommerce", "bigcommerce"):
   160→                    ecommerce_data = await self._fetch_ecommerce_data(company_id, int_type)
   161→                    profile["total_orders"] += ecommerce_data.get("order_count", 0)
   162→                    profile["has_shipping_addresses"] = ecommerce_data.get("has_shipping_addresses", False)
   163→                    profile["has_payment_data"] = profile.has_payment_data or ecommerce_data.get("has_payment_data", False)
   164→                    profile["has_products"] = profile["has_products"] or ecommerce_data.get("has_products", False)
   165→                    if ecommerce_data.get("industry"):
   166→                        profile["industries_detected"].append("ecommerce")
   167→                        
   168→                elif int_type in ("stripe", "paddle", "paypal"):
   169→                    payment_data = await self._fetch_payment_data(company_id, int_type)
   170→                    profile["has_payment_data"] = True
   171→                    profile["data_points"].append({
   172→                        "source": int_type,
   173→                        "type": "payment",
   174→                        "count": payment_data.get("transaction_count", 0),
   175→                    })
   176→                    
   177→                elif int_type in ("mailchimp", "klaviyo", "brevo"):
   178→                    marketing_data = await self._fetch_marketing_data(company_id, int_type)
   179→                    profile["has_email_campaigns"] = True
   180→                    profile["data_points"].append({
   181→                        "source": int_type,
   182→                        "type": "marketing",
   183→                        "subscriber_count": marketing_data.get("subscriber_count", 0),
   184→                    })
   185→                    
   186→                elif int_type in ("zendesk", "freshdesk", "intercom", "gorgias"):
   187→                    profile["has_ticket_data"] = True
   188→                    
   189→                elif int_type in ("shipstation", "aftership", "easypost", "fedex", "ups", "dhl"):
   190→                    profile["has_shipping_addresses"] = True
   191→
   192→            except Exception as e:
   193→                logger.warning("Failed to gather data from %s: %s", int_type, str(e)[:100])
   194→                profile["data_points"].append({
   195→                    "source": int_type,
   196→                    "error": str(e)[:200],
   197→                })
   198→
   199→        # Deduplicate industries
   200→        profile["industries_detected"] = list(set(profile["industries_detected"]))
   201→        return profile

   203→    async def _fetch_crm_data(
   204→        self, company_id: str, crm_type: str
   205→    ) -> Dict[str, Any]:
   206→        """Fetch data summary from CRM integrations."""
   207→        import httpx
   208→
   209→        creds = self.integration_service.get_credential_config(company_id, crm_type)
   210→        if not creds:
   211→            return {"contact_count": 0, "deal_count": 0, "has_products": False}
   212→
   213→        try:
   214→            if crm_type == "hubspot":
   215→                base_url = "https://api.hubapi.com"
   216→                headers = {"Authorization": f"Bearer {creds.get('api_key', '')}"}
   217→                
   218→                async with httpx.AsyncClient(timeout=10.0) as client:
   219→                    # Get contacts count
   220→                    contacts_resp = await client.get(
   221→                        f"{base_url}/crm/v3/objects/contacts",
   222→                        params={"limit": 1},
   223→                        headers=headers,
   224→                    )
   225→                    contact_count = contacts_resp.json().get("total", 0) if contacts_resp.status_code == 200 else 0
   226→                    
   227→                    # Get deals count
   228→                    deals_resp = await client.get(
   229→                        f"{base_url}/crm/v3/objects/deals",
   230→                        params={"limit": 1},
   231→                        headers=headers,
   232→                    )
   233→                    deal_count = deals_resp.json().get("total", 0) if deals_resp.status_code == 200 else 0
   234→                    
   235→                    # Check for products
   236→                    products_resp = await client.get(
   237→                        f"{base_url}/crm/v3/objects/products",
   238→                        params={"limit": 1},
   239→                        headers=headers,
   240→                    )
   241→                    has_products = products_resp.status_code == 200 and products_resp.json().get("total", 0) > 0
   242→
   243→                return {
   244→                    "contact_count": contact_count,
   245→                    "deal_count": deal_count,
   246→                    "has_products": has_products,
   247→                }
   248→
   249→            elif crm_type == "shopify":
   250→                return await self._fetch_shopify_summary(creds)
   251→
   252→        except Exception as e:
   253→            logger.warning("CRM data fetch failed for %s: %s", crm_type, e)
   254→
   255→        return {"contact_count": 0, "deal_count": 0, "has_products": False}

   283→    async def _fetch_shopify_summary(self, creds: Dict[str, Any]) -> Dict[str, Any]:
   284→        """Fetch summary data from Shopify store."""
   285→        import httpx
   286→
   287→        store_url = creds.get("store_url", "").rstrip("/")
   288→        token = creds.get("access_token", "")
   289→        
   290→        if not store_url or not token:
   291→            return {"order_count": 0, "has_shipping_addresses": False, "has_products": False}
   292→
   293→        headers = {"X-Shopify-Access-Token": token}
   294→        
   295→        async with httpx.AsyncClient(timeout=10.0) as client:
   296→            # Get order count
   297→            orders_resp = await client.get(
   298→                f"{store_url}/admin/api/2024-01/orders/count.json",
   299→                headers=headers,
   300→            )
   301→            order_count = orders_resp.json().get("count", 0) if orders_resp.status_code == 200 else 0
   302→            
   303→            # Get product count
   304→            products_resp = await client.get(
   305→                f"{store_url}/admin/api/2024-01/products/count.json",
   306→                headers=headers,
   307→            )
   308→            has_products = products_resp.json().get("count", 0) > 0 if products_resp.status_code == 200 else False
   309→            
   310→            # Check recent orders for shipping addresses
   311→            shipping_check = await client.get(
   312→                f"{store_url}/admin/api/2024-01/orders.json?limit=5&status=any",
   313→                headers=headers,
   314→            )
   315→            has_shipping = False
   316→            if shipping_check.status_code == 200:
   317→                orders = shipping_check.json().get("orders", [])
   318→                has_shipping = any(o.get("shipping_address") for o in orders)
   319→
   320→        return {
   321→            "order_count": order_count,
   322→            "has_shipping_addresses": has_shipping,
   323→            "has_products": has_products,
   324→        }

   358→        self, company_id: str, marketing_type: str
   359→    ) -> Dict[str, Any]:
   360→        """Fetch data summary from marketing integrations."""

   364→        if not creds:
   365→            return {"subscriber_count": 0}
   366→
   367→        try:
   368→            if marketing_type == "mailchimp":
   369→                api_key = creds.get("api_key", "")
   370→                # Extract DC from API key (xxx-us123)
   371→                dc = ""
   372→                if "-" in api_key:
   373→                    parts = api_key.rsplit("-", 1)
   374→                    if len(parts) == 2:
   375→                        dc = parts[-1]
   376→

   378→                    headers = {"Authorization": f"Bearer {api_key}"}
   379→                    async with httpx.AsyncClient(timeout=10.0) as client:
   380→                        resp = await client.get(
   381→                            f"https://{dc}.api.mailchimp.com/3.0/lists",
   382→                            headers=headers,
   383→                        )
   384→                        if resp.status_code == 200:
   385→                            lists = resp.json().get("lists", [])
   386→                            total_subs = sum(l.get("stats", {}).get("member_count", 0) for l in lists)
   387→                            return {"subscriber_count": total_subs}
   388→
   389→        except Exception as e:
   390→            logger.warning("Marketing data fetch failed for %s: %s", marketing_type, e)

   394→    async def _detect_gaps(
   395→        self,
   396→        data_profile: Dict[str, Any],
   397→        connected: List[Dict[str, Any]],
   398→    ) -> List[Dict[str, Any]]:
   399→        """Detect gaps between current setup and potential needs."""
   400→        connected_types = {c["type"] for c in connected}
   401→        gaps = []
   402→
   403→        # Gap detection rules
   404→        gap_rules = [
   405→            {
   406→                "id": "shipping_missing",
   407→                "condition": data_profile.get("total_orders", 0) > 0 and not data_profile.get("has_shipping_addresses"),
   408→                "category": "shipping",
   409→                "severity": "high",
   410→                "message": "You have orders but no shipping integration for tracking",
   411→                "recommended": ["shipstation", "aftership", "easypost"],
   412→            },
   413→            {
   414→                "id": "payment_missing",
   415→                "condition": data_profile.get("has_products") and not data_profile.get("has_payment_data"),
   416→                "category": "payments",
   417→                "severity": "high",
   418→                "message": "You sell products but no payment processor is connected",
   419→                "recommended": ["stripe", "paddle"],
   420→            },
   421→            {
   422→                "id": "marketing_missing",
   423→                "condition": data_profile.get("total_contacts", 0) > 100 and not data_profile.get("has_email_campaigns"),
   424→                "category": "marketing",
   425→                "severity": "medium",
   426→                "message": f"You have {data_profile.get('total_contacts', 0)} contacts but no email marketing tool",
   427→                "recommended": ["mailchimp", "klaviyo", "brevo"],
   428→            },
   429→            {
   430→                "id": "helpdesk_missing",
   431→                "condition": data_profile.get("total_contacts", 0) > 50 and not data_profile.get("has_ticket_data"),
   432→                "category": "helpdesk",
   433→                "severity": "medium",
   434→                "message": "Growing customer base but no dedicated helpdesk system",
   435→                "recommended": ["zendesk", "freshdesk", "intercom", "gorgias"],
   436→            },
   437→            {
   438→                "id": "analytics_missing",
   439→                "condition": data_profile.get("total_orders", 0) > 10 or data_profile.get("total_contacts", 0) > 100,
   440→                "category": "analytics",
   441→                "severity": "low",
   442→                "message": "Significant activity but no analytics integration for insights",
   443→                "recommended": ["mixpanel", "amplitude", "google_analytics"],
   444→            },
   445→            {
   446→                "id": "communication_missing",
   447→                "condition": data_profile.get("total_contacts", 0) > 20,
   448→                "category": "communication",
   449→                "severity": "low",
   450→                "message": "Team communication could be improved with Slack integration",
   451→                "recommended": ["slack"],
   452→            },
   453→        ]
   454→
   455→        for rule in gap_rules:
   456→            if rule["condition"]:
   457→                # Filter out already-connected recommendations
   458→                available_recs = [r for r in rule["recommended"] if r not in connected_types]
   459→                if available_recs:
   460→                    gaps.append({
   461→                        **rule,
   462→                        "recommended": available_recs,
   463→                    })
   464→
   465→        return gaps

   467→    async def _call_nvidia_glm(self, prompt: str, max_tokens: int = 800, temperature: float = 0.3) -> str:
   468→        """Call NVIDIA GLM 5.2 directly for CRM analysis.
   469→        
   470→        Uses the same NVIDIA API key and endpoint as the main pipeline.
   471→        Model: z-ai/glm-5.2 (your preferred LLM)
   472→        
   473→        This bypasses the generic llm_call to ensure we use NVIDIA GLM specifically
   474→        for integration recommendations - giving more consistent results.
   475→        """
   476→        import httpx
   477→        
   478→        api_key = os.environ.get("NVIDIA_API_KEY", "")
   479→        if not api_key:
   480→            logger.warning("NVIDIA_API_KEY not set for CRM analysis")
   481→            return ""
   482→        
   483→        messages = [
   484→            {
   485→                "role": "system",
   486→                "content": "You are Parwa's intelligent integration advisor. You analyze business data and recommend specific third-party integrations that would improve their workflow. Always respond in valid JSON format."
   487→            },
   488→            {
   489→                "role": "user", 
   490→                "content": prompt
   491→            }
   492→        ]
   493→        
   494→        payload = {
   495→            "model": "z-ai/glm-5.2",  # Your preferred NVIDIA GLM model
   496→            "messages": messages,
   497→            "temperature": temperature,
   498→            "max_tokens": max_tokens,
   499→        }
   500→        headers = {
   501→            "Authorization": f"Bearer {api_key}",
   502→            "Content-Type": "application/json",
   503→        }
   504→        
   505→        try:
   506→            async with httpx.AsyncClient(timeout=60.0) as client:
   507→                r = await client.post(
   508→                    "https://integrate.api.nvidia.com/v1/chat/completions",
   509→                    json=payload,
   510→                    headers=headers,
   511→                )
   512→            
   513→            if r.status_code == 200:
   514→                data = r.json()
   515→                content = data.get("choices", [{}])[0].get("message", {}).get("content", "")
   516→                usage = data.get("usage", {})
   517→                logger.info(
   518→                    "NVIDIA GLM CRM analysis complete: tokens=%s (prompt=%s, completion=%s)",
   519→                    usage.get("total_tokens", "?"),
   520→                    usage.get("prompt_tokens", "?"),
   521→                    usage.get("completion_tokens", "?"),
   522→                )
   523→                return content.strip()
   524→            else:
   525→                logger.error("NVIDIA GLM API error %d: %s", r.status_code, r.text[:200])
   526→                return ""
   527→                
   528→        except Exception as e:
   529→            logger.error("NVIDIA GLM call failed: %s", str(e)[:200])
   530→            return ""

   532→    async def _generate_recommendations(
   533→        self,
   534→        company_id: str,
   535→        data_profile: Dict[str, Any],
   536→        connected: List[Dict[str, Any]],
   537→        gaps: List[Dict[str, Any]],
   538→    ) -> List[Dict[str, Any]]:
   539→        """Use NVIDIA GLM 5.2 to generate personalized integration recommendations."""
   540→
   541→        # Build context for LLM
   542→        connected_names = [c["name"] for c in connected]
   543→        connected_types = [c["type"] for c in connected]
   544→
   545→        prompt = f"""You are Parwa's integration advisor. Analyze this company's data and recommend specific integrations they need.
   546→
   547→COMPANY DATA PROFILE:
   548→- Total Contacts: {data_profile.get('total_contacts', 0)}
   549→- Total Orders: {data_profile.get('total_orders', 0)}
   550→- Total Deals: {data_profile.get('total_deals', 0)}
   551→- Has Products: {data_profile.get('has_products', False)}
   552→- Has Shipping Data: {data_profile.get('has_shipping_addresses', False)}

   683→    def _save_analysis_result(self, result: Dict[str, Any]) -> Optional[CRMAnalysisResult]:
   684→        """Save analysis result to database for persistence."""
   685→        try:
   686→            # Check if recent analysis exists (avoid duplicates within 1 hour)
   687→            from datetime import timedelta
   688→            recent_cutoff = datetime.now(timezone.utc) - timedelta(hours=1)
   689→            
   690→            existing = (
   691→                self.db.query(CRMAnalysisResult)
   692→                .filter(
   693→                    CRMAnalysisResult.company_id == result["company_id"],
   694→                    CRMAnalysisResult.analyzed_at >= recent_cutoff,
   695→                )
   696→                .first()
   697→            )
   698→            
   699→            if existing:
   700→                # Update existing record instead of creating duplicate
   701→                existing.data_profile = result.get("data_profile", {})
   702→                existing.connected_integrations = result.get("connected_integrations", [])
   703→                existing.detected_gaps = result.get("detected_gaps", [])
   704→                existing.recommendations = result.get("recommendations", [])
   705→                existing.analysis_summary = result.get("analysis_summary", "")
   706→                existing.updated_at = datetime.now(timezone.utc)
   707→                self.db.commit()
   708→                logger.info("Updated existing CRM analysis %s for company %s", 
   709→                           existing.id[:8], result["company_id"])
   710→                return existing
   711→            
   712→            # Create new record
   713→            db_result = CRMAnalysisResult(
   714→                company_id=result["company_id"],
   715→                data_profile=result.get("data_profile", {}),
   716→                connected_integrations=result.get("connected_integrations", []),
   717→                detected_gaps=result.get("detected_gaps", []),
   718→                recommendations=result.get("recommendations", []),
   719→                analysis_summary=result.get("analysis_summary", ""),
   720→                llm_model_used="z-ai/glm-5.2",
   721→            )
   722→            
   723→            self.db.add(db_result)
   724→            self.db.commit()
   725→            self.db.refresh(db_result)
   726→            
   727→            logger.info("Saved CRM analysis %s for company %s (%d recs)", 
   728→                       db_result.id[:8], result["company_id"], 
   729→                       len(result.get("recommendations", [])))
   730→            return db_result
   731→            
   732→        except Exception as e:
   733→            logger.error("Failed to save CRM analysis result: %s", e)
   734→            self.db.rollback()
   735→            return None

   810→    def _build_summary(
   811→        self,
   812→        data_profile: Dict[str, Any],
   813→        recommendations: List[Dict[str, Any]],
   814→    ) -> str:
   815→        """Build human-readable summary of the analysis."""
   816→        high_priority = [r for r in recommendations if r.get("priority") == "high"]
   817→        medium_priority = [r for r in recommendations if r.get("priority") == "medium"]
   818→
   819→        parts = [
   820→            f"Based on your data ({data_profile.get('total_contacts', 0)} contacts, "
   821→            f"{data_profile.get('total_orders', 0)} orders), ",
   822→        ]
   823→
   824→        if high_priority:
   825→            parts.append(f"we found {len(high_priority)} urgent integration(s) you should add: ")
   826→            parts.append(", ".join(r["name"] for r in high_priority[:3]))
   827→            if len(high_priority) > 3:
   828→                parts.append(f", and {len(high_priority) - 3} more")
   829→            parts.append(". ")
   830→
   831→        if medium_priority:
   832→            parts.append(f"We also suggest {len(medium_priority)} optional improvement(s).")
   833→
   834→        if not high_priority and not medium_priority:
   835→            parts.append("your integration setup looks good! No critical gaps detected.")
   836→
   837→        return "".join(parts)

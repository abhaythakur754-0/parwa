   116→        return _AttrChainer()
   117→    def desc(self):
   118→        return self
   119→    def asc(self):
   120→        return self
   121→    def __ge__(self, other):
   122→        return True  # Always pass for mock filter comparisons
   123→    def __le__(self, other):
   124→        return True
   125→    def __eq__(self, other):
   126→        return True  # Filters always match in mocks
   127→    def __ne__(self, other):
   128→        return False

   132→        return self  # Support .isnot() for filter expressions
   133→    def contains(self, *args):

   124→   124→function formatCurrency(amount: number): string {
   125→   125→  return new Intl.NumberFormat('en-US', {
   126→   126→    style: 'currency',
   127→   127→    currency: 'USD',
   128→   128→    minimumFractionDigits: 2,
   129→   129→  }).format(amount);
   130→   130→}
   131→   131→
   132→   132→function formatDate(dateStr: string): string {
   133→   133→  if (!dateStr) return 'N/A';
   134→   134→  const date = new Date(dateStr);
   135→   135→  return date.toLocaleDateString('en-US', {
   136→   136→    month: 'short',
   137→   137→    day: 'numeric',
   138→   138→    hour: '2-digit',
   139→   139→    minute: '2-digit',
   140→   140→  });
   141→   141→}
   142→   142→
   143→   143→function getStatusColor(status: string): string {
   144→   144→  switch (status) {
   145→   145→    case 'active':
   146→   146→      return 'bg-green-100 text-green-800 border-green-200';
   147→   147→    case 'paused':
   148→   148→      return 'bg-yellow-100 text-yellow-800 border-yellow-200';
   149→   149→    case 'completed':
   150→   150→      return 'bg-blue-100 text-blue-800 border-blue-200';
   151→   151→    case 'cancelled':
   152→   152→      return 'bg-gray-100 text-gray-800 border-gray-200';
   153→   153→    case 'failed':
   154→   154→      return 'bg-red-100 text-red-800 border-red-200';
   155→   155→    case 'pending':
   156→   156→      return 'bg-purple-100 text-purple-800 border-purple-200';
   157→   157→    default:
   158→   158→      return 'bg-gray-100 text-gray-800 border-gray-200';
   159→   159→  }
   160→   160→}
   161→   161→
   162→   162→function getInstallmentStatusIcon(status: string) {
   163→   163→  switch (status) {
   164→   164→    case 'paid':
   165→   165→      return <CheckCircleIcon className="h-4 w-4 text-green-600" />;
   166→   166→    case 'failed':
   167→   167→      return <XCircleIcon className="h-4 w-4 text-red-600" />;
   168→   168→    case 'processing':
   169→   169→      return <ClockIcon className="h-4 w-4 text-blue-600 animate-spin" />;
   170→   170→    default:
   171→   171→      return <ClockIcon className="h-4 w-4 text-gray-400" />;
   172→   172→  }

   175→   175→function getVariantDisplayName(variant: string): string {
   176→   176→  const names: Record<string, string> = {
   177→   177→    mini: 'Mini PARWA',
   178→   178→    parwa: 'PARWA',
   179→   179→    high: 'PARWA High',
   180→   180→  };
   181→   181→  return names[variant] || variant.toUpperCase();
   182→   182→}

   186→   186→
   187→   187→interface FlexPayDashboardProps {
   188→   188→  /** The FlexPay plan ID to display */
   189→   189→  planId?: string;
   190→   190→  /** Optional callback when action is taken */
   191→   191→  onAction?: (action: string, data?: any) => void;
   192→   192→  /** Whether to show admin controls (pause/resume/cancel) */
   193→   193→  showControls?: boolean;
   194→   194→  /** Auto-refresh interval in milliseconds (0 = no auto-refresh) */
   195→   195→  refreshInterval?: number;
   196→   196→}
   197→   197→
   198→   198→export function FlexPayDashboard({
   199→   199→  planId,
   200→   200→  onAction,
   201→   201→  showControls = true,
   202→   202→  refreshInterval = 30000, // Refresh every 30 seconds by default
   203→   203→}: FlexPayDashboardProps) {
   204→   204→  // State
   205→   205→  const [planStatus, setPlanStatus] = useState<FlexPlanStatus | null>(null);
   206→   206→  const [loading, setLoading] = useState(true);
   207→   207→  const [error, setError] = useState<string | null>(null);
   208→   208→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   209→   209→  
   210→   210→  // Dialog states
   211→   211→  const [cancelDialogOpen, setCancelDialogOpen] = useState(false);
   212→   212→  const [pauseDialogOpen, setPauseDialogOpen] = useState(false);
   213→   213→  const [resumeDialogOpen, setResumeDialogOpen] = useState(false);
   214→   214→  
   215→   215→  // Fetch plan status
   216→   216→  const fetchStatus = useCallback(async () => {
   217→   217→    if (!planId) {
   218→   218→      setLoading(false);
   219→   219→      return;
   220→   220→    }
   221→   221→
   222→   222→    try {
   223→   223→      setError(null);
   224→   224→      const response = await fetch(`/api/flexpay/${planId}/status`, {
   225→   225→        headers: {
   226→   226→          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
   227→   227→        },
   228→   228→      });
   229→   229→
   230→   230→      if (!response.ok) {
   231→   231→        throw new Error(`Failed to fetch status: ${response.statusText}`);
   232→   232→      }
   233→   233→
   234→   234→      const data: FlexPlanStatus = await response.json();
   235→   235→      setPlanStatus(data);
   236→   236→    } catch (err) {
   237→   237→      setError(err instanceof Error ? err.message : 'Failed to load plan status');
   238→   238→    } finally {
   239→   239→      setLoading(false);
   240→   240→    }
   241→   241→  }, [planId]);
   242→   242→
   243→   243→  // Initial fetch and auto-refresh
   244→   244→  useEffect(() => {
   245→   245→    fetchStatus();
   246→   246→
   247→   247→    if (refreshInterval > 0) {
   248→   248→      const interval = setInterval(fetchStatus, refreshInterval);
   249→   249→      return () => clearInterval(interval);
   250→   250→    }
   251→   251→  }, [fetchStatus, refreshInterval]);
   252→   252→
   253→   253→  // Action handlers
   254→   254→  const handleAction = async (action: string, payload?: any) => {
   255→   255→    if (!planId) return;
   256→   256→
   257→   257→    setActionLoading(action);
   258→   258→    try {
   259→   259→      const response = await fetch(`/api/flexpay/${planId}/${action}`, {
   260→   260→        method: 'POST',
   261→   261→        headers: {
   262→   262→          'Content-Type': 'application/json',
   263→   263→          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
   264→   264→        },
   265→   265→        body: JSON.stringify(payload || {}),
   266→   266→      });
   267→   267→
   268→   268→      if (!response.ok) {
   269→   269→        throw new Error(`Action failed: ${response.statusText}`);
   270→   270→      }
   271→   271→
   272→   272→      // Refresh status after action
   273→   273→      await fetchStatus();
   274→   274→      
   275→   275→      // Close dialogs
   276→   276→      setCancelDialogOpen(false);
   277→   277→      setPauseDialogOpen(false);
   278→   278→      setResumeDialogOpen(false);
   279→   279→
   280→   280→      // Notify parent
   281→   281→      onAction?.(action, payload);
   282→   282→    } catch (err) {
   283→   283→      setError(err instanceof Error ? err.message : `Failed to ${action}`);
   284→   284→    } finally {
   285→   285→      setActionLoading(null);
   286→   286→    }
   287→   287→  };
   288→   288→
   289→   289→  // Loading state
   290→   290→  if (loading) {
   291→   291→    return (
   292→   292→      <Card className="w-full max-w-4xl mx-auto">
   293→   293→        <CardContent className="p-8">
   294→   294→          <div className="flex items-center justify-center space-x-4">
   295→   295→            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-primary"></div>
   296→   296→            <span className="text-muted-foreground">Loading your FlexPay progress...</span>
   297→   297→          </div>
   298→   298→        </CardContent>
   299→   299→      </Card>
   300→   300→    );
   301→   301→  }
   302→   302→
   303→   303→  // Error state
   304→   304→  if (error && !planStatus) {
   305→   305→    return (
   306→   306→      <Card className="w-full max-w-4xl mx-auto border-destructive">
   307→   307→        <CardContent className="p-6">
   308→   308→          <div className="flex items-center space-x-3 text-destructive">
   309→   309→            <AlertTriangleIcon className="h-6 w-6" />
   310→   310→            <div>
   311→   311→              <h3 className="font-semibold">Unable to Load FlexPay Data</h3>
   312→   312→              <p className="text-sm mt-1">{error}</p>
   313→   313→              <Button variant="outline" size="sm" className="mt-3" onClick={fetchStatus}>
   314→   314→                Try Again
   315→   315→              </Button>
   316→   316→            </div>
   317→   317→          </div>
   318→   318→        </CardContent>
   319→   319→      </Card>
   320→   320→    );
   321→   321→  }
   322→   322→
   323→   323→  // No plan selected
   324→   324→  if (!planStatus) {
   325→   325→    return (
   326→   326→      <Card className="w-full max-w-4xl mx-auto">
   327→   327→        <CardContent className="p-8 text-center">
   328→   328→          <ClockIcon className="h-12 w-12 mx-auto text-muted-foreground mb-4" />
   329→   329→          <h3 className="text-lg font-semibold mb-2">No Active FlexPay Plan</h3>
   330→   330→          <p className="text-muted-foreground">
   331→   331→            Select a plan or create a new FlexPay installment plan to get started.
   332→   332→          </p>
   333→   333→        </CardContent>
   334→   334→      </Card>
   335→   335→    );
   336→   336→  }
   337→   337→
   338→   338→  // Calculate day info
   339→   339→  const currentDay = planStatus.installments.completed + 1;
   340→   340→  const totalDays = Math.ceil(planStatus.installments.total / 1.33); // Approximate (accounts for extra charges)
   341→   341→
   342→   342→  return (
   343→   343→    <div className="w-full max-w-4xl mx-auto space-y-6">
   344→   344→      {/* Header Card */}
   345→   345→      <Card>
   346→   346→        <CardHeader className="pb-4">
   347→   347→          <div className="flex items-start justify-between">
   348→   348→            <div className="space-y-1">
   349→   349→              <CardTitle className="text-2xl font-bold flex items-center gap-3">
   350→   350→                FlexPay Dashboard
   351→   351→                <Badge variant="outline" className={getStatusColor(planStatus.status)}>
   352→   352→                  {planStatus.status.charAt(0).toUpperCase() + planStatus.status.slice(1)}
   353→   353→                </Badge>
   354→   354→              </CardTitle>
   355→   355→              <CardDescription className="text-base">
   356→   356→                {getVariantDisplayName(planStatus.variant_tier)} • {formatCurrency(planStatus.total_amount)} total
   357→   357→              </CardDescription>
   358→   358→            </div>
   359→   359→            
   360→   360→            {showControls && (
   361→   361→              <div className="flex gap-2">
   362→   362→                {(planStatus.status === 'active' || planStatus.status === 'pending') && (
   363→   363→                  <Button
   364→   364→                    variant="outline"
   365→   365→                    size="sm"
   366→   366→                    onClick={() => setPauseDialogOpen(true)}
   367→   367→                    disabled={!!actionLoading}
   368→   368→                  >
   369→   369→                    <PauseIcon className="h-4 w-4 mr-2" />
   370→   370→                    Pause
   371→   371→                  </Button>
   372→   372→                )}
   373→   373→                
   374→   374→                {planStatus.status === 'paused' && (
   375→   375→                  <Button
   376→   376→                    variant="outline"
   377→   377→                    size="sm"
   378→   378→                    onClick={() => setResumeDialogOpen(true)}
   379→   379→                    disabled={!!actionLoading}
   380→   380→                  >
   381→   381→                    <PlayIcon className="h-4 w-4 mr-2" />
   382→   382→                    Resume
   383→   383→                  </Button>
   384→   384→                )}
   385→   385→                
   386→   386→                {!['completed', 'cancelled'].includes(planStatus.status) && (
   387→   387→                  <Dialog open={cancelDialogOpen} onOpenChange={setCancelDialogOpen}>
   388→   388→                    <DialogTrigger asChild>
   389→   389→                      <Button variant="destructive" size="sm" disabled={!!actionLoading}>
   390→   390→                        Cancel Plan
   391→   391→                      </Button>
   392→   392→                    </DialogTrigger>
   393→   393→                    <DialogContent>
   394→   394→                      <DialogHeader>
   395→   395→                        <DialogTitle>Cancel FlexPay Plan?</DialogTitle>
   396→   396→                        <DialogDescription>
   397→   397→                          This will stop all future payments. You'll keep access for the portion you've already paid ({formatCurrency(planStatus.collected_amount)}).
   398→   398→                          <br /><br />
   399→   399→                          <strong>This action cannot be undone.</strong>
   400→   400→                        </DialogDescription>
   401→   401→                      </DialogHeader>
   402→   402→                      <DialogFooter>
   403→   403→                        <Button variant="outline" onClick={() => setCancelDialogOpen(false)}>
   404→   404→                          Keep Plan
   405→   405→                        </Button>
   406→   406→                        <Button
   407→   407→                          variant="destructive"
   408→   408→                          onClick={() => handleAction('cancel', { reason: 'Customer requested cancellation' })}
   409→   409→                          disabled={actionLoading === 'cancel'}
   410→   410→                        >
   411→   411→                          {actionLoading === 'cancel' ? 'Cancelling...' : 'Yes, Cancel Plan'}
   412→   412→                        </Button>
   413→   413→                      </DialogFooter>
   414→   414→                    </DialogContent>
   415→   415→                  </Dialog>
   416→   416→                )}
   417→   417→              </div>
   418→   418→            )}
   419→   419→          </div>
   420→   420→        </CardHeader>
   421→   421→
   422→   422→        <CardContent className="space-y-6">
   423→   423→          {/* Day X of Y Progress */}
   424→   424→          <div className="space-y-3">
   425→   425→            <div className="flex items-baseline justify-between">
   426→   426→              <span className="text-sm font-medium text-muted-foreground">Payment Progress</span>
   427→   427→              <span className="text-3xl font-bold text-primary">
   428→   428→                Day {Math.min(currentDay, totalDays)} of {totalDays}
   429→   429→              </span>
   430→   430→            </div>
   431→   431→            
   432→   432→            <Progress value={planStatus.progress_percent} className="h-3" />
   433→   433→            
   434→   434→            <div className="flex justify-between text-sm text-muted-foreground">
   435→   435→              <span>{formatCurrency(planStatus.collected_amount)} collected</span>
   436→   436→              <span>{planStatus.progress_percent.toFixed(1)}% complete</span>
   437→   437→              <span>{formatCurrency(planStatus.remaining_amount)} remaining</span>
   438→   438→            </div>
   439→   439→          </div>
   440→   440→
   441→   441→          <Separator />
   442→   442→
   443→   443→          {/* Quick Stats Grid */}
   444→   444→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   445→   445→            <div className="text-center p-3 bg-green-50 dark:bg-green-950 rounded-lg">
   446→   446→              <div className="text-2xl font-bold text-green-700 dark:text-green-300">
   447→   447→                {planStatus.installments.completed}
   448→   448→              </div>
   449→   449→              <div className="text-xs text-green-600 dark:text-green-400 mt-1">Paid</div>
   450→   450→            </div>
   451→   451→            
   452→   452→            <div className="text-center p-3 bg-blue-50 dark:bg-blue-950 rounded-lg">
   453→   453→              <div className="text-2xl font-bold text-blue-700 dark:text-blue-300">
   454→   454→                {planStatus.installments.pending}
   455→   455→              </div>
   456→   456→              <div className="text-xs text-blue-600 dark:text-blue-400 mt-1">Pending</div>
   457→   457→            </div>
   458→   458→            
   459→   459→            <div className="text-center p-3 bg-yellow-50 dark:bg-yellow-950 rounded-lg">
   460→   460→              <div className="text-2xl font-bold text-yellow-700 dark:text-yellow-300">

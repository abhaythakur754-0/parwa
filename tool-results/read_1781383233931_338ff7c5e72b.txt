   120→// ── Savings Calculation ──────────────────────────────────────────────────
   121→
   122→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   123→  agentsReplaced: number;
   124→  humanCost: number;
   125→  savings: number;
   126→  savingsPercent: number;
   127→} {
   128→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   129→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   130→  const savings = Math.max(0, humanCost - monthlyCost);
   131→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   132→  return { agentsReplaced, humanCost, savings, savingsPercent };
   133→}

   242→// ── Component ──────────────────────────────────────────────────────────
   243→
   244→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   245→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   246→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   247→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   248→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   249→  const [isSubmitting, setIsSubmitting] = useState(false);
   250→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   251→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   252→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   253→
   254→  // Restore add-ons and multi-variant selections from localStorage
   255→  useEffect(() => {
   256→    try {
   257→      const stored = localStorage.getItem('parwa_pricing_context');
   258→      if (stored) {
   259→        const ctx = JSON.parse(stored) as PricingContext & {
   260→          selectedVariants?: string[];
   261→          variantQuantities?: Record<string, number>;
   262→        };
   263→        if (ctx.addOns) {
   264→          setAddOns(ctx.addOns);
   265→        }
   266→        // Restore multiple variants if selected on Pricing page
   267→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   268→          const tiers: VariantTier[] = ctx.selectedVariants
   269→            .map((v: string) => {
   270→              const normalized = normalizeTier(v);
   271→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   272→            })
   273→            .filter(Boolean) as VariantTier[];
   274→          if (tiers.length > 0) {
   275→            setActiveVariants(tiers);
   276→          }
   277→        }
   278→        // Restore variant quantities
   279→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   280→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   281→          const newQuantities = { ...variantQuantities };
   282→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   283→            const tier = normalizeTier(variantKey);
   284→            if (['starter', 'growth', 'high'].includes(tier)) {
   285→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   286→            }
   287→          }
   288→          setVariantQuantities(newQuantities);
   289→        }
   290→      }
   291→    } catch {
   292→      // ignore
   293→    }
   294→  }, []);
   295→
   296→  // Check if Paddle.js is available
   297→  useEffect(() => {
   298→    getPaddleInstance().then((paddle) => {
   299→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   300→    }).catch(() => {
   301→      setPaddleStatus('unavailable');
   302→    });
   303→  }, []);
   304→
   305→  // ── Calculations ──────────────────────────────────────────────────
   306→
   307→  const totalTicketLimit = useMemo(
   308→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   309→    [activeVariants, variantQuantities]
   310→  );
   311→
   312→  const baseSubscription = useMemo(
   313→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   314→    [activeVariants, variantQuantities]
   315→  );
   316→
   317→  const addOnTotal = useMemo(() => {
   318→    let total = 0;
   319→    if (addOns.voice) {
   320→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   321→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   322→        total += voiceAddOn.price;
   323→      }
   324→    }
   325→    if (addOns.customApi) {
   326→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   327→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   328→        total += customApiAddOn.price;
   329→      }
   330→    }
   331→    return total;
   332→  }, [addOns, activeVariants]);
   333→
   334→  const totalMonthly = baseSubscription + addOnTotal;
   335→
   336→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   337→
   338→  const toggleVariant = (tier: VariantTier) => {
   339→    setActiveVariants((prev) => {
   340→      if (prev.includes(tier)) {
   341→        if (prev.length <= 1) {
   342→          toast.error('You must have at least one active variant');
   343→          return prev;
   344→        }
   345→        return prev.filter((t) => t !== tier);
   346→      }
   347→      return [...prev, tier];
   348→    });
   349→  };
   350→
   351→  const updateQuantity = (tier: VariantTier, qty: number) => {
   352→    const newQty = Math.max(1, Math.min(qty, 10));
   353→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   354→  };
   355→
   356→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   357→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   358→  };
   359→
   360→  // ── Payment success handler — ONLY called after Paddle confirms ──
   361→  const handlePaymentSuccess = () => {
   362→    setPaymentConfirmed(true);
   363→    localStorage.removeItem('parwa_payment_pending');
   364→    toast.success('Payment successful! Welcome to PARWA!');
   365→    // Only trigger onboarding completion AFTER Paddle confirms
   366→    onComplete();
   367→  };
   368→
   369→  const handleProceed = async () => {
   370→    setIsSubmitting(true);
   371→    setCheckoutError(null);
   372→
   373→    try {
   374→      // Update pricing context in localStorage
   375→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   376→      const context: PricingContext = {
   377→        industry: 'other',
   378→        variant: primaryOnboardingVariant,
   379→        addOns,
   380→        totalMonthly,
   381→        timestamp: new Date().toISOString(),
   382→      };
   383→
   384→      // Preserve industry from existing context
   385→      try {
   386→        const stored = localStorage.getItem('parwa_pricing_context');
   387→        if (stored) {
   388→          const existing = JSON.parse(stored) as PricingContext;
   389→          context.industry = existing.industry;
   390→        }
   391→      } catch {
   392→        // ignore
   393→      }
   394→
   395→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   396→
   397→      // ── Paddle Checkout Flow ──────────────────────────────────────
   398→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   399→
   400→      // Add each active variant as a Paddle line item (with quantity)
   401→      for (const tier of activeVariants) {
   402→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   403→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   404→        const qty = variantQuantities[tier] || 1;
   405→        if (variantPriceId) {
   406→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   407→        }
   408→      }
   409→
   410→      // Voice add-on (only for starter where it's not included)
   411→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   412→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   413→      }
   414→
   415→      // Custom API add-on (only for variants where it's not included)
   416→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   417→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });
   418→      }
   419→
   420→      const customData = {
   421→        source: 'parwa_onboarding',
   422→        variant: primaryOnboardingVariant,
   423→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   424→        variantQuantities,
   425→        addOns,
   426→        industry: context.industry,
   427→        totalMonthly,
   428→      };
   429→
   430→      // ── Step 1: Try server-side checkout (backend creates Paddle transaction) ──
   431→      let checkoutUrl: string | null = null;
   432→
   433→      try {
   434→        const res = await fetch('/api/onboarding/checkout', {
   435→          method: 'POST',
   436→          headers: { 'Content-Type': 'application/json' },
   437→          body: JSON.stringify({
   438→            variant: primaryOnboardingVariant,
   439→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   440→            variantQuantities,
   441→            addOns,
   442→            totalMonthly,
   443→            industry: context.industry,
   444→          }),
   445→        });
   446→
   447→        if (res.ok) {
   448→          const checkoutData = await res.json();
   449→          checkoutUrl = checkoutData.checkout_url || null;
   450→        }
   451→      } catch {
   452→        // API unavailable — continue with client-side Paddle checkout
   453→      }
   454→
   455→      // If we got a checkout_url from backend, redirect to it
   456→      if (checkoutUrl) {
   457→        localStorage.setItem('parwa_payment_pending', JSON.stringify({
   458→          activeVariants, addOns, totalMonthly, variantQuantities,
   459→          pendingAt: new Date().toISOString(),
   460→        }));
   461→        window.location.href = checkoutUrl;
   462→        return;
   463→      }
   464→
   465→      // ── Step 2: Client-side Paddle checkout (items-based overlay) ──
   466→      const paddle = await getPaddleInstance();
   467→
   468→      if (paddle && checkoutItems.length > 0) {
   469→        setPaddleStatus('ready');
   470→        const opened = await openCheckoutWithItems(
   471→          checkoutItems,
   472→          customData,
   473→          // onPaymentSuccess — ONLY triggers after Paddle confirms the transaction
   474→          handlePaymentSuccess,
   475→          // onCheckoutClosed — user closed without paying
   476→          () => {
   477→            toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   478→            setIsSubmitting(false);
   479→          },
   480→        );
   481→
   482→        if (opened) {
   483→          return; // Paddle overlay is showing — wait for user to complete or close
   484→        }
   485→      }
   486→
   487→      // ── Step 3: Paddle unavailable — try re-initializing ──
   488→      try {
   489→        const retryPaddle = await getPaddleInstance();
   490→        if (retryPaddle && checkoutItems.length > 0) {
   491→          setPaddleStatus('ready');
   492→          const opened = await openCheckoutWithItems(
   493→            checkoutItems,
   494→            customData,
   495→            handlePaymentSuccess,
   496→            () => {
   497→              toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   498→              setIsSubmitting(false);
   499→            },
   500→          );
   501→          if (opened) return;
   502→        }
   503→      } catch {
   504→        // Paddle truly unavailable
   505→      }
   506→
   507→      // ── Step 4: Paddle unavailable — block with error ──
   508→      setPaddleStatus('unavailable');
   509→      localStorage.setItem('parwa_payment_pending', JSON.stringify({
   510→        activeVariants, addOns, totalMonthly, variantQuantities,
   511→        pendingAt: new Date().toISOString(),
   512→      }));
   513→      setCheckoutError(
   514→        'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   515→        'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'
   516→      );
   517→      toast.error('Payment gateway unavailable — please try again or contact support.');
   518→    } catch (err) {
   519→      console.error('[cost-breakdown] Error:', err);
   520→      setCheckoutError('Something went wrong. Please try again.');
   521→      toast.error('Something went wrong. Please try again.');
   522→    } finally {
   523→      setIsSubmitting(false);
   524→    }
   525→  };
   526→
   527→  return (
   528→    <div className="space-y-6">
   529→      {/* Header */}
   530→      <div className="text-center space-y-2">
   531→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   532→          <Receipt className="w-7 h-7 text-emerald-400" />
   533→        </div>
   534→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   535→        <p className="text-orange-200/40 text-sm">
   536→          Configure your plan and complete payment to activate your AI agents.
   537→        </p>
   538→      </div>
   539→
   540→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   541→      {paddleStatus === 'unavailable' && (
   542→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   543→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   544→          <div>
   545→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   546→            <p className="text-[10px] text-orange-200/30 mt-1">
   547→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   548→            </p>
   549→          </div>
   550→        </div>
   551→      )}
   552→      {paddleStatus === 'ready' && (
   553→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   554→          <Shield className="w-4 h-4 text-emerald-400" />
   555→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   556→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   557→        </div>
   558→      )}
   559→
   560→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   561→      <div className="space-y-3">
   562→        <div className="flex items-center gap-2">
   563→          <Brain className="w-4 h-4 text-orange-400" />
   564→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   565→            Active Variants
   566→          </label>
   567→          <span className="text-[10px] text-orange-200/25">
   568→            Each variant adds its own ticket allocation and AI pipeline
   569→          </span>
   570→        </div>
   571→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   572→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   573→            <VariantMixerCard
   574→              key={tier}
   575→              tier={tier}
   576→              isActive={activeVariants.includes(tier)}
   577→              quantity={variantQuantities[tier] || 1}
   578→              onToggle={() => toggleVariant(tier)}
   579→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   580→            />
   581→          ))}
   582→        </div>
   583→      </div>
   584→
   585→      {/* ── 2. Add-Ons ───────────────────────────────────────────────── */}
   586→      <div className="space-y-3">
   587→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   588→          Optional Add-Ons
   589→        </label>
   590→        {ADD_ONS.map((addOn) => {
   591→          const Icon = addOn.icon;
   592→          const isSelected = addOns[addOn.key];
   593→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   594→          const showPrice = !isIncluded;
   595→
   596→          return (
   597→            <button
   598→              key={addOn.key}
   599→              type="button"
   600→              onClick={() => toggleAddOn(addOn.key)}
   601→              className={cn(
   602→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',
   603→                isSelected
   604→                  ? 'border-orange-500/30 bg-orange-500/5'
   605→                  : 'border-white/[0.06] hover:border-orange-500/15'
   606→              )}
   607→              style={!isSelected ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   608→            >
   609→              <div className={cn(
   610→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   611→                isSelected ? 'bg-orange-500/10' : 'bg-white/[0.04]'
   612→              )}>
   613→                <Icon className={cn(
   614→                  'w-5 h-5',
   615→                  isSelected ? 'text-orange-400' : 'text-zinc-500'
   616→                )} />
   617→              </div>
   618→              <div className="flex-1 min-w-0">
   619→                <div className="flex items-center gap-2">
   620→                  <p className={cn(
   621→                    'text-sm font-medium',
   622→                    isSelected ? 'text-orange-400' : 'text-white'
   623→                  )}>
   624→                    {addOn.name}
   625→                  </p>
   626→                  {isIncluded && (
   627→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   628→                      Included
   629→                    </span>
   630→                  )}
   631→                </div>
   632→                <p className="text-[10px] text-orange-200/30 mt-0.5">{addOn.description}</p>
   633→              </div>
   634→              <div className="flex flex-col items-end shrink-0 gap-1">
   635→                {showPrice && (
   636→                  <p className="text-sm font-semibold text-white">
   637→                    ${addOn.price}/mo
   638→                  </p>
   639→                )}
   640→                <div className={cn(
   641→                  'w-8 h-5 rounded-full flex items-center px-0.5 transition-colors duration-200',
   642→                  isSelected || isIncluded
   643→                    ? 'bg-gradient-to-r from-orange-500 to-amber-400'
   644→                    : 'bg-white/10'
   645→                )}>
   646→                  <div className={cn(
   647→                    'w-4 h-4 rounded-full bg-white shadow-sm transition-transform duration-200',
   648→                    (isSelected || isIncluded) ? 'translate-x-3' : 'translate-x-0'
   649→                  )} />
   650→                </div>
   651→              </div>
   652→            </button>
   653→          );
   654→        })}
   655→      </div>
   656→
   657→      {/* ── 3. Cost Summary ────────────────────────────────────────────── */}
   658→      <div
   659→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   660→        style={{ background: 'rgba(255,255,255,0.03)' }}
   661→      >
   662→        {/* Per-variant breakdown */}
   663→        {activeVariants.map((tier) => {
   664→          const qty = variantQuantities[tier] || 1;
   665→          const price = VARIANT_PRICES[tier] * qty;
   666→
   667→          return (
   668→            <div key={tier} className="flex items-center justify-between">
   669→              <span className="text-sm text-orange-200/50 flex items-center gap-1.5">
   670→                <div className={cn(
   671→                  'w-2 h-2 rounded-full',
   672→                  tier === 'starter' ? 'bg-emerald-400' : tier === 'growth' ? 'bg-orange-400' : 'bg-purple-400'
   673→                )} />
   674→                {VARIANT_DISPLAY[tier].name}
   675→                {qty > 1 && <span className="text-orange-200/30"> × {qty}</span>}
   676→              </span>
   677→              <span className="text-sm text-white">
   678→                ${price.toLocaleString()}/mo
   679→              </span>
   680→            </div>
   681→          );
   682→        })}
   683→
   684→        {/* Add-ons */}
   685→        {addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t)) && (
   686→          <div className="flex items-center justify-between">
   687→            <span className="text-sm text-orange-200/50">Voice Channel</span>
   688→            <span className="text-sm text-white">$199/mo</span>
   689→          </div>
   690→        )}
   691→        {addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t)) && (
   692→          <div className="flex items-center justify-between">
   693→            <span className="text-sm text-orange-200/50">Custom API Connector</span>
   694→            <span className="text-sm text-white">$49/mo</span>
   695→          </div>
   696→        )}
   697→
   698→        {/* Integrations = $0 */}
   699→        <div className="flex items-center justify-between">
   700→          <span className="text-sm text-orange-200/50">Integrations</span>
   701→          <span className="text-sm text-emerald-400">$0 (free)</span>
   702→        </div>
   703→
   704→        <div className="border-t border-white/[0.06] pt-3 flex items-center justify-between">
   705→          <span className="text-sm font-semibold text-white">Total Monthly</span>
   706→          <span className="text-lg font-bold text-orange-400">
   707→            ${totalMonthly.toLocaleString()}/mo
   708→          </span>
   709→        </div>
   710→
   711→        {/* Overage rate info */}
   712→        <div className="flex items-center justify-between">
   713→          <span className="text-[10px] text-orange-200/25">Overage rate (beyond limits)</span>
   714→          <span className="text-[10px] text-orange-200/25">${OVERAGE_PRICE_PER_TICKET}/ticket</span>
   715→        </div>
   716→      </div>
   717→
   718→      {/* ── 4. Savings Comparison ──────────────────────────────────────── */}
   719→      <div
   720→        className="rounded-xl border border-emerald-500/20 p-4"
   721→        style={{ background: 'rgba(16,185,129,0.04)' }}
   722→      >
   723→        <div className="flex items-start gap-3">
   724→          <div className="w-8 h-8 rounded-lg bg-emerald-500/10 flex items-center justify-center shrink-0">
   725→            <TrendingDown className="w-4 h-4 text-emerald-400" />
   726→          </div>
   727→          <div className="flex-1">
   728→            <p className="text-sm font-semibold text-emerald-400">
   729→              Save {savings.savingsPercent}% vs human agents
   730→            </p>
   731→            <p className="text-[10px] text-orange-200/30 mt-1">
   732→              PARWA handles {totalTicketLimit.toLocaleString()} tickets/mo — equivalent to{' '}
   733→              {savings.agentsReplaced} full-time support agent{savings.agentsReplaced > 1 ? 's' : ''} at ~${AGENT_COST_MONTHLY.toLocaleString()}/mo each.
   734→              That&apos;s ${savings.humanCost.toLocaleString()}/mo in human costs vs ${totalMonthly.toLocaleString()}/mo with PARWA.
   735→            </p>
   736→          </div>
   737→        </div>
   738→      </div>
   739→
   740→      {/* ── No Hidden Fees ─────────────────────────────────────────────── */}
   741→      <div className="flex items-center justify-center gap-2 text-[10px] text-orange-200/25">
   742→        <Shield className="w-3 h-3" />
   743→        <span>No hidden fees. Need more? Add another variant.</span>
   744→      </div>
   745→
   746→      {/* ── Error Display ────────────────────────────────────────────── */}
   747→      {checkoutError && (
   748→        <div className="rounded-xl border border-red-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(239,68,68,0.05)' }}>
   749→          <AlertCircle className="w-5 h-5 text-red-400 shrink-0 mt-0.5" />
   750→          <p className="text-sm text-red-300">{checkoutError}</p>
   751→        </div>
   752→      )}
   753→
   754→      {/* ── Payment Confirmation Guard ──────────────────────────────── */}
   755→      {paymentConfirmed && (
   756→        <div className="rounded-xl border border-emerald-500/30 p-4 flex items-start gap-3" style={{ background: 'rgba(16,185,129,0.08)' }}>
   757→          <Shield className="w-5 h-5 text-emerald-400 shrink-0 mt-0.5" />
   758→          <div>
   759→            <p className="text-sm font-medium text-emerald-300">Payment confirmed by Paddle</p>
   760→            <p className="text-[10px] text-orange-200/30 mt-1">
   761→              Your subscription is now active. Redirecting to your dashboard...
   762→            </p>
   763→          </div>
   764→        </div>
   765→      )}
   766→
   767→      {/* ── Proceed Button ────────────────────────────────────────────── */}
   768→      <div className="flex justify-end">
   769→        <button
   770→          onClick={handleProceed}
   771→          disabled={isSubmitting || paymentConfirmed}
   772→          className={cn(
   773→            'px-8 py-3 font-semibold rounded-xl transition-all duration-300 shadow-lg text-sm flex items-center gap-2 disabled:cursor-not-allowed',
   774→            'bg-gradient-to-r from-orange-500 to-amber-400 hover:from-orange-400 hover:to-amber-300 text-[#1A1A1A] shadow-orange-500/25',
   775→            (isSubmitting || paymentConfirmed) && 'opacity-60'
   776→          )}
   777→        >
   778→          {isSubmitting ? (
   779→            <>
   780→              <Loader2 className="w-4 h-4 animate-spin" />
   781→              Opening Checkout...
   782→            </>
   783→          ) : paymentConfirmed ? (
   784→            <>
   785→              <Shield className="w-4 h-4" />
   786→              Activated — Redirecting...
   787→            </>
   788→          ) : (
   789→            <>
   790→              Pay ${totalMonthly.toLocaleString()}/mo & Activate
   791→              <ArrowRight className="w-4 h-4" />
   792→            </>
   793→          )}
   794→        </button>
   795→      </div>
   796→    </div>
   797→  );
   798→}

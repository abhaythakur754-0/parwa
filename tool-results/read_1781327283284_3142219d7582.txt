    28→/**
    29→ * Extract auth token from cookies (parwa_at) and/or Authorization header.
    30→ * Returns a headers object suitable for forwarding.
    31→ */
    32→function buildAuthHeaders(request: NextRequest): Headers {
    33→  const headers = new Headers(request.headers);
    34→  headers.delete('host');
    35→
    36→  // Forward Authorization header if present
    37→  const authHeader = request.headers.get('authorization');
    38→  if (authHeader) {
    39→    headers.set('Authorization', authHeader);
    40→  }
    41→
    42→  // Also check for parwa_at cookie and use it as Bearer token if no Authorization header
    43→  if (!authHeader) {
    44→    const cookieToken = request.cookies.get('parwa_at')?.value;
    45→    if (cookieToken) {
    46→      headers.set('Authorization', `Bearer ${cookieToken}`);
    47→    }
    48→  }
    49→
    50→  return headers;
    51→}

    53→/**
    54→ * Try to proxy a request to the backend FastAPI server.
    55→ * Returns the Response on success, or null if backend is unavailable / returned an error.
    56→ */
    57→async function proxyToBackend(
    58→  request: NextRequest,
    59→  pathSegments: string[],
    60→): Promise<Response | null> {
    61→  const backendPath = `${BACKEND_URL}/api/onboarding-jarvis/${pathSegments.join('/')}`;
    62→  const url = new URL(request.url);
    63→  const searchParams = url.searchParams.toString();
    64→  const fullUrl = searchParams ? `${backendPath}?${searchParams}` : backendPath;
    65→
    66→  try {
    67→    const body = ['POST', 'PATCH', 'PUT'].includes(request.method)
    68→      ? await request.arrayBuffer()
    69→      : undefined;
    70→
    71→    const headers = buildAuthHeaders(request);
    72→
    73→    const response = await fetch(fullUrl, {
    74→      method: request.method,
    75→      headers,
    76→      body,
    77→      signal: AbortSignal.timeout(20000),
    78→    });
    79→
    80→    if (response.status >= 200 && response.status < 300) {
    81→      return response;
    82→    }
    83→
    84→    // Backend returned an error — fall back to local handling
    85→    const errorText = await response.text().catch(() => '');
    86→    console.warn(
    87→      `[OnboardingJarvis] Backend returned ${response.status}:`,
    88→      errorText.slice(0, 200),
    89→    );
    90→    return null;
    91→  } catch (err) {
    92→    // Backend unreachable — fall back to local handling
    93→    console.warn(
    94→      '[OnboardingJarvis] Backend proxy failed:',
    95→      (err instanceof Error ? err.message : String(err))?.slice(0, 150),
    96→    );
    97→    return null;
    98→  }
    99→}

   123→async function callZAISDK(
   124→  messages: Array<{ role: string; content: string }>,
   125→): Promise<string | null> {
   126→  try {
   127→    const zai = await getZAI();
   128→    if (!zai || !zai.chat || !zai.chat.completions) return null;
   129→
   130→    const completion = await zai.chat.completions.create({
   131→      messages: messages.map((m) => ({
   132→        role: m.role === 'assistant' ? 'assistant' : m.role,
   133→        content: m.content,
   134→      })),
   135→      temperature: 0.8,
   136→      max_tokens: 800,
   137→    });
   138→
   139→    const text = completion?.choices?.[0]?.message?.content;
   140→    if (text && text.trim().length > 10) return text.trim();
   141→    return null;
   142→  } catch (err) {
   143→    console.warn(
   144→      '[OnboardingJarvis] z-ai-web-dev-sdk failed:',
   145→      (err instanceof Error ? err.message : String(err))?.slice(0, 150),
   146→    );
   147→    return null;
   148→  }

   427→// ── Bullet-Point Format Enforcer ────────────────────────────────
   428→
   429→function isEmojiChar(ch: string): boolean {
   430→  const code = ch.codePointAt(0) || 0;
   431→  return (
   432→    (code >= 0x1f300 && code <= 0x1faff) ||
   433→    (code >= 0x2600 && code <= 0x27bf) ||
   434→    (code >= 0xfe00 && code <= 0xfe0f)
   435→  );
   436→}
   437→
   438→function forceBulletFormat(text: string): string {
   439→  const lines = text.split('\n');
   440→  if (lines.length === 0) return text;
   441→
   442→  const nonEmpty = lines.filter((l) => l.trim());
   443→  if (nonEmpty.length === 0) return text;
   444→
   445→  const bulletCount = nonEmpty.filter((l) => {
   446→    const t = l.trim();
   447→    return (
   448→      /^[\u2022\-*•]\s/.test(t) ||
   449→      /^[0-9]+[.)]\s/.test(t) ||
   450→      isEmojiChar(t)
   451→    );
   452→  }).length;
   453→
   454→  // If 40%+ lines are already bullets, return as-is
   455→  if (nonEmpty.length > 2 && bulletCount / nonEmpty.length >= 0.4) return text;
   456→
   457→  if (nonEmpty.length <= 2) {
   458→    const hasLongLine = nonEmpty.some((l) => l.trim().length > 150);
   459→    if (!hasLongLine) return text;
   460→  }
   461→
   462→  // Convert paragraphs into bullet-point format
   463→  const result: string[] = [];
   464→  let openerUsed = false;
   465→
   466→  for (const line of lines) {
   467→    const trimmed = line.trim();
   468→    if (!trimmed) {
   469→      result.push('');
   470→      continue;
   471→    }
   472→
   473→    if (
   474→      /^[\u2022\-*•]\s/.test(trimmed) ||
   475→      /^[0-9]+[.)]\s/.test(trimmed)
   476→    ) {
   477→      result.push(trimmed);
   478→      continue;
   479→    }
   480→
   481→    if (isEmojiChar(trimmed)) {
   482→      result.push(trimmed);
   483→      continue;
   484→    }
   485→
   486→    if (trimmed.length < 50) {
   487→      if (!openerUsed) {
   488→        result.push(trimmed);
   489→        openerUsed = true;
   490→      } else {
   491→        const emoji = pickEmoji(trimmed);
   492→        result.push(
   493→          `${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`,
   494→        );
   495→      }
   496→      continue;
   497→    }
   498→
   499→    const sentences = trimmed.match(/[^.!?]*[.!?]+/g) || [trimmed];
   500→    if (sentences.length === 1 && trimmed.length < 80) {
   501→      result.push(trimmed);
   502→      continue;
   503→    }
   504→
   505→    if (sentences.length === 1) {
   506→      const parts = trimmed
   507→        .split(/,/g)
   508→        .map((s) => s.trim())
   509→        .filter(Boolean);
   510→      if (parts.length >= 2) {
   511→        result.push(parts[0]);
   512→        result.push('');
   513→        for (let i = 1; i < parts.length; i++) {
   514→          const p = parts[i].trim();
   515→          if (!p) continue;
   516→          const emoji = pickEmoji(p);
   517→          result.push(
   518→            `${emoji} ${p.charAt(0).toUpperCase() + p.slice(1)}`,
   519→          );
   520→        }
   521→      } else {
   522→        const emoji = pickEmoji(trimmed);
   523→        result.push(
   524→          `${emoji} ${trimmed.charAt(0).toUpperCase() + trimmed.slice(1)}`,
   525→        );
   526→      }
   527→      continue;
   528→    }
   529→
   530→    if (!openerUsed) {
   531→      result.push(sentences[0].trim());
   532→      result.push('');
   533→      openerUsed = true;
   534→    }
   535→
   536→    for (let i = 1; i < sentences.length; i++) {
   537→      const s = sentences[i].trim();
   538→      if (!s) continue;
   539→      const emoji = pickEmoji(s);
   540→      result.push(`${emoji} ${s.charAt(0).toUpperCase() + s.slice(1)}`);
   541→    }
   542→  }
   543→
   544→  return result.join('\n');
   545→}

   547→function pickEmoji(text: string): string {
   548→  const lower = text.toLowerCase();
   549→  if (
   550→    lower.includes('save') ||
   551→    lower.includes('cost') ||
   552→    lower.includes('price') ||
   553→    lower.includes('$')
   554→  )
   555→    return '💰';

   557→    lower.includes('automat') ||
   558→    lower.includes('ai') ||
   559→    lower.includes('robot') ||
   560→    lower.includes('resolv')
   561→  )
   562→    return '🤖';
   563→  if (
   564→    lower.includes('channel') ||
   565→    lower.includes('email') ||
   566→    lower.includes('chat') ||
   567→    lower.includes('phone')
   568→  )
   569→    return '📡';
   570→  if (
   571→    lower.includes('integrat') ||
   572→    lower.includes('connect') ||
   573→    lower.includes('shopify')
   574→  )
   575→    return '🔗';
   576→  if (lower.includes('secur') || lower.includes('encrypt') || lower.includes('gdpr'))
   577→    return '🔒';
   578→  if (lower.includes('speed') || lower.includes('fast') || lower.includes('quick'))
   579→    return '⚡';
   580→  if (lower.includes('analyt') || lower.includes('data') || lower.includes('roi'))
   581→    return '📊';
   582→  if (lower.includes('feature') || lower.includes('capab') || lower.includes('support'))
   583→    return '🎯';
   584→  if (lower.includes('start') || lower.includes('begin') || lower.includes('get'))

   586→  if (lower.includes('check') || lower.includes('yes') || lower.includes('sure'))
   587→    return '✅';
   588→  if (lower.includes('think') || lower.includes('smart') || lower.includes('predict'))
   589→    return '🧠';
   590→  return '💡';
   591→}

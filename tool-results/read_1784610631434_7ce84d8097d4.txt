    43→    43→        self,

    45→    45→    ) -> Dict[str, Any]:
    46→    46→        """Main entry point: Analyze company's CRM and return recommendations.
    47→    47→
    48→    48→        Steps:
    49→    49→        1. Get all connected integrations for this company
    50→    50→        2. Fetch sample data from each connected integration
    51→    51→        3. Analyze data patterns to detect business type and gaps
    52→    52→        4. Use LLM to generate personalized recommendations
    53→    53→        
    54→    54→        Returns:
    55→    55→            Dict with:
    56→    56→            - connected_integrations: list of currently connected
    57→    57→            - data_profile: summary of what data was found
    58→    58→            - recommendations: list of recommended integrations with reasons
    59→    59→            - analysis_summary: plain text explanation
    60→    60→        """
    61→    61→        logger.info("Starting CRM analysis for company %s", company_id)
    62→    62→
    63→    63→        # Step 1: Get connected integrations
    64→    64→        connected = await self._get_connected_integrations(company_id)
    65→    65→        
    66→    66→        # Step 2: Gather data from each integration
    67→    67→        data_profile = await self._gather_data_from_integrations(company_id, connected)
    68→    68→        
    69→    69→        # Step 3: Detect patterns and gaps
    70→    70→        detected_gaps = await self._detect_gaps(data_profile, connected)
    71→    71→        
    72→    72→        # Step 4: Generate LLM-powered recommendations
    73→    73→        recommendations = await self._generate_recommendations(
    74→    74→            company_id=company_id,
    75→    75→            data_profile=data_profile,
    76→    76→            connected=connected,
    77→    77→            gaps=detected_gaps,
    78→    78→        )
    79→    79→        
    80→    80→        # FIX: If no recommendations generated (new user with 0 integrations),
    81→    81→        # provide default "getting started" recommendations
    82→    82→        if not recommendations and not connected:
    83→    83→            logger.info("No integrations connected - providing default recommendations")
    84→    84→            recommendations = self._get_default_recommendations()
    85→    85→
    86→    86→        result = {
    87→    87→            "company_id": company_id,
    88→    88→            "analyzed_at": datetime.now(timezone.utc).isoformat(),

    90→    90→            "data_profile": data_profile,
    91→    91→            "detected_gaps": detected_gaps,
    92→    92→            "recommendations": recommendations,
    93→    93→            "analysis_summary": self._build_summary(data_profile, recommendations),
    94→    94→        }
    95→    95→
    96→    96→        # Save to database for persistence (onboarding → dashboard transition)
    97→    97→        saved_result = self._save_analysis_result(result)

    99→    99→            result["analysis_id"] = str(saved_result.id)
   100→   100→            result["is_saved"] = True
   101→   101→        else:
   102→   102→            result["is_saved"] = False
   103→   103→
   104→   104→        logger.info("CRM analysis complete for company %s: %d recommendations (saved=%s)", 
   105→   105→                    company_id, len(recommendations), result["is_saved"])

   136→   136→    async def _gather_data_from_integrations(
   137→   137→        self,
   138→   138→        company_id: str,
   139→   139→        connected: List[Dict[str, Any]],
   140→   140→    ) -> Dict[str, Any]:
   141→   141→        """Gather sample data from each connected integration."""
   142→   142→        profile = {

   151→   151→            "industries_detected": [],
   152→   152→            "data_points": [],
   153→   153→        }
   154→   154→
   155→   155→        for integration in connected:
   156→   156→            int_type = integration["type"]
   157→   157→            
   158→   158→            try:
   159→   159→                if int_type in ("hubspot", "salesforce", "pipedrive"):
   160→   160→                    crm_data = await self._fetch_crm_data(company_id, int_type)
   161→   161→                    profile["total_contacts"] += crm_data.get("contact_count", 0)
   162→   162→                    profile["total_deals"] += crm_data.get("deal_count", 0)
   163→   163→                    profile["has_products"] = profile["has_products"] or crm_data.get("has_products", False)
   164→   164→                    
   165→   165→                elif int_type in ("shopify", "woocommerce", "bigcommerce"):
   166→   166→                    ecommerce_data = await self._fetch_ecommerce_data(company_id, int_type)
   167→   167→                    profile["total_orders"] += ecommerce_data.get("order_count", 0)
   168→   168→                    profile["has_shipping_addresses"] = ecommerce_data.get("has_shipping_addresses", False)
   169→   169→                    profile["has_payment_data"] = profile.has_payment_data or ecommerce_data.get("has_payment_data", False)
   170→   170→                    profile["has_products"] = profile["has_products"] or ecommerce_data.get("has_products", False)
   171→   171→                    if ecommerce_data.get("industry"):
   172→   172→                        profile["industries_detected"].append("ecommerce")
   173→   173→                        
   174→   174→                elif int_type in ("stripe", "paddle", "paypal"):
   175→   175→                    payment_data = await self._fetch_payment_data(company_id, int_type)
   176→   176→                    profile["has_payment_data"] = True
   177→   177→                    profile["data_points"].append({
   178→   178→                        "source": int_type,
   179→   179→                        "type": "payment",
   180→   180→                        "count": payment_data.get("transaction_count", 0),
   181→   181→                    })
   182→   182→                    
   183→   183→                elif int_type in ("mailchimp", "klaviyo", "brevo"):
   184→   184→                    marketing_data = await self._fetch_marketing_data(company_id, int_type)
   185→   185→                    profile["has_email_campaigns"] = True
   186→   186→                    profile["data_points"].append({
   187→   187→                        "source": int_type,
   188→   188→                        "type": "marketing",
   189→   189→                        "subscriber_count": marketing_data.get("subscriber_count", 0),
   190→   190→                    })
   191→   191→                    
   192→   192→                elif int_type in ("zendesk", "freshdesk", "intercom", "gorgias"):
   193→   193→                    profile["has_ticket_data"] = True
   194→   194→                    
   195→   195→                elif int_type in ("shipstation", "aftership", "easypost", "fedex", "ups", "dhl"):
   196→   196→                    profile["has_shipping_addresses"] = True
   197→   197→
   198→   198→            except Exception as e:
   199→   199→                logger.warning("Failed to gather data from %s: %s", int_type, str(e)[:100])
   200→   200→                profile["data_points"].append({
   201→   201→                    "source": int_type,
   202→   202→                    "error": str(e)[:200],
   203→   203→                })
   204→   204→
   205→   205→        # Deduplicate industries
   206→   206→        profile["industries_detected"] = list(set(profile["industries_detected"]))
   207→   207→        return profile

   209→   209→    async def _fetch_crm_data(
   210→   210→        self, company_id: str, crm_type: str
   211→   211→    ) -> Dict[str, Any]:
   212→   212→        """Fetch data summary from CRM integrations."""
   213→   213→        import httpx
   214→   214→
   215→   215→        creds = self.integration_service.get_credential_config(company_id, crm_type)
   216→   216→        if not creds:
   217→   217→            return {"contact_count": 0, "deal_count": 0, "has_products": False}
   218→   218→
   219→   219→        try:
   220→   220→            if crm_type == "hubspot":
   221→   221→                base_url = "https://api.hubapi.com"
   222→   222→                headers = {"Authorization": f"Bearer {creds.get('api_key', '')}"}
   223→   223→                
   224→   224→                async with httpx.AsyncClient(timeout=10.0) as client:
   225→   225→                    # Get contacts count
   226→   226→                    contacts_resp = await client.get(
   227→   227→                        f"{base_url}/crm/v3/objects/contacts",
   228→   228→                        params={"limit": 1},
   229→   229→                        headers=headers,
   230→   230→                    )
   231→   231→                    contact_count = contacts_resp.json().get("total", 0) if contacts_resp.status_code == 200 else 0
   232→   232→                    
   233→   233→                    # Get deals count
   234→   234→                    deals_resp = await client.get(
   235→   235→                        f"{base_url}/crm/v3/objects/deals",
   236→   236→                        params={"limit": 1},
   237→   237→                        headers=headers,
   238→   238→                    )
   239→   239→                    deal_count = deals_resp.json().get("total", 0) if deals_resp.status_code == 200 else 0
   240→   240→                    
   241→   241→                    # Check for products
   242→   242→                    products_resp = await client.get(
   243→   243→                        f"{base_url}/crm/v3/objects/products",
   244→   244→                        params={"limit": 1},
   245→   245→                        headers=headers,
   246→   246→                    )
   247→   247→                    has_products = products_resp.status_code == 200 and products_resp.json().get("total", 0) > 0
   248→   248→
   249→   249→                return {
   250→   250→                    "contact_count": contact_count,
   251→   251→                    "deal_count": deal_count,
   252→   252→                    "has_products": has_products,
   253→   253→                }
   254→   254→
   255→   255→            elif crm_type == "shopify":
   256→   256→                return await self._fetch_shopify_summary(creds)
   257→   257→
   258→   258→        except Exception as e:
   259→   259→            logger.warning("CRM data fetch failed for %s: %s", crm_type, e)
   260→   260→
   261→   261→        return {"contact_count": 0, "deal_count": 0, "has_products": False}

   289→   289→    async def _fetch_shopify_summary(self, creds: Dict[str, Any]) -> Dict[str, Any]:
   290→   290→        """Fetch summary data from Shopify store."""
   291→   291→        import httpx
   292→   292→
   293→   293→        store_url = creds.get("store_url", "").rstrip("/")
   294→   294→        token = creds.get("access_token", "")
   295→   295→        
   296→   296→        if not store_url or not token:
   297→   297→            return {"order_count": 0, "has_shipping_addresses": False, "has_products": False}
   298→   298→
   299→   299→        headers = {"X-Shopify-Access-Token": token}
   300→   300→        
   301→   301→        async with httpx.AsyncClient(timeout=10.0) as client:
   302→   302→            # Get order count
   303→   303→            orders_resp = await client.get(
   304→   304→                f"{store_url}/admin/api/2024-01/orders/count.json",
   305→   305→                headers=headers,
   306→   306→            )
   307→   307→            order_count = orders_resp.json().get("count", 0) if orders_resp.status_code == 200 else 0
   308→   308→            
   309→   309→            # Get product count
   310→   310→            products_resp = await client.get(
   311→   311→                f"{store_url}/admin/api/2024-01/products/count.json",
   312→   312→                headers=headers,
   313→   313→            )
   314→   314→            has_products = products_resp.json().get("count", 0) > 0 if products_resp.status_code == 200 else False
   315→   315→            
   316→   316→            # Check recent orders for shipping addresses
   317→   317→            shipping_check = await client.get(
   318→   318→                f"{store_url}/admin/api/2024-01/orders.json?limit=5&status=any",
   319→   319→                headers=headers,
   320→   320→            )
   321→   321→            has_shipping = False
   322→   322→            if shipping_check.status_code == 200:
   323→   323→                orders = shipping_check.json().get("orders", [])
   324→   324→                has_shipping = any(o.get("shipping_address") for o in orders)
   325→   325→
   326→   326→        return {
   327→   327→            "order_count": order_count,
   328→   328→            "has_shipping_addresses": has_shipping,
   329→   329→            "has_products": has_products,
   330→   330→        }

   473→   473→    async def _call_nvidia_glm(self, prompt: str, max_tokens: int = 800, temperature: float = 0.3) -> str:
   474→   474→        """Call NVIDIA GLM 5.2 directly for CRM analysis.
   475→   475→        
   476→   476→        Uses the same NVIDIA API key and endpoint as the main pipeline.
   477→   477→        Model: z-ai/glm-5.2 (your preferred LLM)
   478→   478→        
   479→   479→        This bypasses the generic llm_call to ensure we use NVIDIA GLM specifically
   480→   480→        for integration recommendations - giving more consistent results.
   481→   481→        """
   482→   482→        import httpx
   483→   483→        
   484→   484→        api_key = os.environ.get("NVIDIA_API_KEY", "")
   485→   485→        if not api_key:
   486→   486→            logger.warning("NVIDIA_API_KEY not set for CRM analysis")
   487→   487→            return ""
   488→   488→        
   489→   489→        messages = [
   490→   490→            {
   491→   491→                "role": "system",
   492→   492→                "content": "You are Parwa's intelligent integration advisor. You analyze business data and recommend specific third-party integrations that would improve their workflow. Always respond in valid JSON format."
   493→   493→            },
   494→   494→            {
   495→   495→                "role": "user", 
   496→   496→                "content": prompt
   497→   497→            }
   498→   498→        ]
   499→   499→        
   500→   500→        payload = {
   501→   501→            "model": "z-ai/glm-5.2",  # Your preferred NVIDIA GLM model
   502→   502→            "messages": messages,
   503→   503→            "temperature": temperature,
   504→   504→            "max_tokens": max_tokens,
   505→   505→        }
   506→   506→        headers = {
   507→   507→            "Authorization": f"Bearer {api_key}",
   508→   508→            "Content-Type": "application/json",
   509→   509→        }
   510→   510→        
   511→   511→        try:
   512→   512→            async with httpx.AsyncClient(timeout=60.0) as client:
   513→   513→                r = await client.post(
   514→   514→                    "https://integrate.api.nvidia.com/v1/chat/completions",
   515→   515→                    json=payload,
   516→   516→                    headers=headers,
   517→   517→                )
   518→   518→            
   519→   519→            if r.status_code == 200:
   520→   520→                data = r.json()
   521→   521→                content = data.get("choices", [{}])[0].get("message", {}).get("content", "")
   522→   522→                usage = data.get("usage", {})
   523→   523→                logger.info(
   524→   524→                    "NVIDIA GLM CRM analysis complete: tokens=%s (prompt=%s, completion=%s)",
   525→   525→                    usage.get("total_tokens", "?"),
   526→   526→                    usage.get("prompt_tokens", "?"),
   527→   527→                    usage.get("completion_tokens", "?"),
   528→   528→                )
   529→   529→                return content.strip()
   530→   530→            else:
   531→   531→                logger.error("NVIDIA GLM API error %d: %s", r.status_code, r.text[:200])
   532→   532→                return ""
   533→   533→                
   534→   534→        except Exception as e:
   535→   535→            logger.error("NVIDIA GLM call failed: %s", str(e)[:200])
   536→   536→            return ""

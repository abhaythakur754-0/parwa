   572→async def list_contacts(
   573→    authorization: str = Header(None),
   574→    limit: int = Query(100, ge=1, le=250),
   575→    after: str = None,
   576→    properties: str = None
   577→):
   578→    """List contacts - HubSpot style"""
   579→    verify_token(authorization)
   580→    
   581→    props = properties.split(",") if properties else None
   582→    contacts = DB["contacts"]
   583→    
   584→    # Apply pagination
   585→    start_idx = 0
   586→    if after:
   587→        try:
   588→            start_idx = next(i for i, c in enumerate(contacts) if c["id"] == after) + 1
   589→        except StopIteration:
   590→            pass
   591→    
   592→    paginated = contacts[start_idx:start_idx + limit]
   593→    
   594→    results = []
   595→    for contact in paginated:
   596→        result = {"id": contact["id"], "properties": {}}
   597→        for key, value in contact.items():
   598→            if key != "id":
   599→                if isinstance(value, dict):
   600→                    result["properties"][key] = json.dumps(value)
   601→                elif isinstance(value, list):
   602→                    result["properties"][key] = ",".join(str(v) for v in value)
   603→                else:
   604→                    result["properties"][key] = str(value)
   605→        results.append(result)
   606→    
   607→    has_more = start_idx + limit < len(contacts)
   608→    next_after = contacts[start_idx + limit]["id"] if has_more else None
   609→    
   610→    return {
   611→        "results": results,
   612→        "paging": {
   613→            "next": {"after": next_after} if has_more else None
   614→        }
   615→    }

   142→async def _clara_gatekeep(query: str, ticket_type: str) -> Dict[str, Any]:
   143→    """CLARA identifies what knowledge is needed. Phase 4: concise prompt."""
   144→    prompt = f"""What knowledge areas are needed to answer this {ticket_type} ticket?
   145→
   146→Ticket: "{query}"
   147→
   148→List 2-4 specific knowledge areas needed (one line each, no explanation):"""
   149→
   150→    result = await llm_call(prompt, max_tokens=150)
   151→    return {
   152→        "relevant_knowledge": result,
   153→        "knowledge_sufficient": False,
   154→        "knowledge_contradictory": False,
   155→    }

   158→# ── Knowledge Retrieval (type-based, non-LLM) ────────────────────
   159→
   160→
   161→def _retrieve_knowledge(ticket_type: str, query: str = "", tenant_id: str = "") -> List[Dict[str, Any]]:
   162→    """Retrieve knowledge documents for THIS tenant only.
   163→
   164→    Three-tier retrieval:
   165→      1. VECTOR SEARCH: if tenant chunks have embeddings, use pgvector
   166→         cosine similarity to find the top-5 most relevant chunks for the
   167→         query. This is the semantic search path — finds by meaning, not
   168→         just keywords.
   169→      2. PER-TENANT KB (fallback): if no embeddings, fetch ALL tenant
   170→         chunks (original behavior — keyword-based SmartFilter downstream)
   171→      3. SHARED DEFAULT KB: platform default docs, only used when the
   172→         tenant has no uploaded KB docs at all
   173→
   174→    BC-001: tenant_id scopes ALL queries. Company A NEVER sees Company B's docs.
   175→    """
   176→    # ── Tier 1: Hybrid search (vector + BM25 + RRF) ──────────────
   177→    # Vector search finds by MEANING ("double charged" → "duplicate payment")
   178→    # BM25 finds by EXACT KEYWORDS (booking numbers, promo codes, policy names)
   179→    # RRF (Reciprocal Rank Fusion) merges both → better than either alone
   180→    if tenant_id and query:
   181→        try:
   182→            from database.base import SessionLocal
   183→            from sqlalchemy import text
   184→            from app.core.parwa_pipeline.nvidia_embedding import embed_text_sync
   185→
   186→            db = SessionLocal()
   187→            try:
   188→                # Check if tenant has any embedded chunks
   189→                has_embeddings = db.execute(
   190→                    text(
   191→                        "SELECT count(*) FROM document_chunks "
   192→                        "WHERE company_id = :tenant_id AND embedding IS NOT NULL"
   193→                    ),
   194→                    {"tenant_id": tenant_id},
   195→                ).scalar()
   196→
   197→                if has_embeddings and int(has_embeddings) > 0:
   198→                    # Generate REAL query embedding using Google AI (primary) or NVIDIA (fallback)
   199→                    query_emb = embed_text_sync(query, input_type="query")
   200→                    # Accept both Google (768) and NVIDIA (1024) embedding dimensions
   201→                    if query_emb and len(query_emb) in (768, 1024):
   202→                        import json as _json
   203→                        import math as _math
   204→
   205→                        # ── Vector search: Python-based cosine similarity ──
   206→                        # No pgvector extension needed — compute cosine similarity in Python.
   207→                        # Load all embedded chunks for this tenant, compute similarity, take top 20.
   208→                        all_chunks = db.execute(
   209→                            text(
   210→                                "SELECT id, content, document_id, embedding "
   211→                                "FROM document_chunks "
   212→                                "WHERE company_id = :tenant_id AND embedding IS NOT NULL"
   213→                            ),
   214→                            {"tenant_id": tenant_id},
   215→                        ).fetchall()
   216→
   217→                        vector_rows = []
   218→                        for row in all_chunks:
   219→                            try:
   220→                                stored_emb = _json.loads(row[3]) if isinstance(row[3], str) else row[3]
   221→                                if not stored_emb or len(stored_emb) != len(query_emb):
   222→                                    continue
   223→                                # Cosine similarity: dot(a,b) / (|a| * |b|)
   224→                                dot = sum(a * b for a, b in zip(query_emb, stored_emb))
   225→                                norm_a = _math.sqrt(sum(a * a for a in query_emb))
   226→                                norm_b = _math.sqrt(sum(b * b for b in stored_emb))
   227→                                if norm_a == 0 or norm_b == 0:
   228→                                    continue
   229→                                similarity = dot / (norm_a * norm_b)
   230→                                vector_rows.append((row[0], row[1], row[2], similarity))
   231→                            except Exception:
   232→                                continue
   233→
   234→                        # Sort by similarity descending, take top 20
   235→                        vector_rows.sort(key=lambda x: x[3], reverse=True)
   236→                        vector_rows = vector_rows[:20]
   237→
   238→                        # ── BM25 search: top 20 by keyword relevance ──
   239→                        # Uses PostgreSQL tsvector (built-in, no extension needed)
   240→                        bm25_rows = db.execute(
   241→                            text(
   242→                                "SELECT id, content, document_id, "
   243→                                "ts_rank_cd(to_tsvector('english', content), "
   244→                                "plainto_tsquery('english', :q)) AS rank "
   245→                                "FROM document_chunks "
   246→                                "WHERE company_id = :tenant_id "
   247→                                "AND to_tsvector('english', content) @@ plainto_tsquery('english', :q) "
   248→                                "ORDER BY rank DESC "
   249→                                "LIMIT 20"
   250→                            ),
   251→                            {"tenant_id": tenant_id, "q": query},
   252→                        ).fetchall()
   253→
   254→                        # ── RRF fusion: merge vector + BM25 rankings ──
   255→                        # RRF score = sum(1 / (k + rank)) for each list
   256→                        # k=60 is the standard RRF constant
   257→                        rrf_k = 60
   258→                        chunk_scores = {}  # chunk_id → {content, doc_id, rrf_score}
   259→
   260→                        for rank, row in enumerate(vector_rows, 1):
   261→                            cid = str(row[0])
   262→                            if cid not in chunk_scores:
   263→                                chunk_scores[cid] = {
   264→                                    "content": row[1],
   265→                                    "document_id": str(row[2]),
   266→                                    "vector_sim": float(row[3]),
   267→                                    "rrf": 0.0,
   268→                                }
   269→                            chunk_scores[cid]["rrf"] += 1.0 / (rrf_k + rank)
   270→
   271→                        for rank, row in enumerate(bm25_rows, 1):
   272→                            cid = str(row[0])
   273→                            if cid not in chunk_scores:
   274→                                chunk_scores[cid] = {
   275→                                    "content": row[1],
   276→                                    "document_id": str(row[2]),
   277→                                    "vector_sim": 0.0,
   278→                                    "rrf": 0.0,
   279→                                }
   280→                            chunk_scores[cid]["rrf"] += 1.0 / (rrf_k + rank)
   281→
   282→                        # Sort by RRF score, take top 5
   283→                        merged = sorted(
   284→                            chunk_scores.values(),
   285→                            key=lambda x: x["rrf"],
   286→                            reverse=True,
   287→                        )[:5]
   288→
   289→                        if merged:
   290→                            result = []
   291→                            for m in merged:
   292→                                result.append({
   293→                                    "content": m["content"],
   294→                                    "source": f"tenant_kb:{m['document_id']}",
   295→                                    "section": "C",
   296→                                    "score": m["rrf"],
   297→                                })
   298→                            logger.info(
   299→                                "Node 3: Hybrid search returned %d chunks for company_id=%s "
   300→                                "(vector=%d, bm25=%d, top rrf=%.4f)",
   301→                                len(result), tenant_id,
   302→                                len(vector_rows), len(bm25_rows),
   303→                                result[0]["score"] if result else 0.0,
   304→                            )
   305→                            return result
   306→
   307→                        # ── If hybrid search found 0 chunks but vector_rows exist ──
   308→                        # Return top vector results directly (skip RRF fusion)
   309→                        if vector_rows:
   310→                            result = []
   311→                            for row in vector_rows[:5]:
   312→                                result.append({
   313→                                    "content": row[1],
   314→                                    "source": f"tenant_kb:{row[2]}",
   315→                                    "section": "C",
   316→                                    "score": float(row[3]),
   317→                                })
   318→                            logger.info(
   319→                                "Node 3: Vector-only fallback returned %d chunks (similarity top=%.4f)",
   320→                                len(result), result[0]["score"] if result else 0.0,
   321→                            )
   322→                            return result
   323→
   324→                        logger.info("Node 3: Hybrid search returned 0 chunks — falling back to fetch-all")
   325→                    else:
   326→                        logger.warning("Node 3: NVIDIA embedding failed for query — falling back to fetch-all")
   327→            finally:
   328→                db.close()
   329→        except Exception as exc:
   330→            logger.warning("Node 3: Hybrid search failed: %s — falling back to fetch-all", str(exc)[:200])
   331→
   332→    # ── Tier 2: Per-tenant KB (fetch all chunks — original behavior) ──
   333→    if tenant_id:
   334→        try:
   335→            from database.base import SessionLocal
   336→            from database.models.onboarding import KnowledgeDocument, DocumentChunk
   337→
   338→            db = SessionLocal()
   339→            try:
   340→                # Get completed documents for THIS tenant only
   341→                docs_query = db.query(KnowledgeDocument).filter(
   342→                    KnowledgeDocument.company_id == tenant_id,
   343→                    KnowledgeDocument.status == "completed",
   344→                )
   345→                tenant_docs = docs_query.all()
   346→
   347→                if tenant_docs:
   348→                    # Get chunks for those documents — scoped by company_id
   349→                    doc_ids = [d.id for d in tenant_docs]
   350→                    chunks = db.query(DocumentChunk).filter(
   351→                        DocumentChunk.company_id == tenant_id,
   352→                        DocumentChunk.document_id.in_(doc_ids),
   353→                    ).all()
   354→
   355→                    if chunks:
   356→                        result = []
   357→                        for chunk in chunks:
   358→                            result.append({
   359→                                "content": chunk.content,
   360→                                "source": f"tenant_kb:{chunk.document_id}",
   361→                                "section": "C",
   362→                            })
   363→                        logger.info(
   364→                            "Node 3: Retrieved %d tenant KB chunks for company_id=%s",
   365→                            len(result), tenant_id,
   366→                        )
   367→                        return result
   368→                    else:
   369→                        logger.info("Node 3: Tenant has docs but no chunks — falling back to default KB")
   370→                else:
   371→                    logger.info("Node 3: No tenant KB docs for company_id=%s — falling back to default KB", tenant_id)
   372→            finally:
   373→                db.close()
   374→        except Exception as exc:
   375→            logger.warning("Node 3: Tenant KB query failed: %s — falling back to default KB", str(exc)[:200])
   376→
   377→    # ── Tier 2: Fallback to shared default KB (platform docs) ────
   378→    # Primary: exact type match
   379→    docs = KNOWLEDGE_BASE.get(ticket_type, [])
   380→
   381→    # Secondary: also pull from related types
   382→    related_types = {
   383→        "refund_request": ["billing"],
   384→        "billing": ["refund_request", "faq"],
   385→        "account_change": ["technical", "faq"],
   386→        "technical": ["account_change", "faq"],
   387→        "complaint": ["billing", "faq"],
   388→        "faq": ["billing", "account_change", "refund_request"],
   389→    }
   390→    for rt in related_types.get(ticket_type, []):
   391→        docs.extend(KNOWLEDGE_BASE.get(rt, []))
   392→
   393→    # Phase 7: T2→T1 Cross-type pattern detection
   394→    # If the query contains strong signals for a different type that isn't
   395→    # already covered by primary or related_types, pull those docs too.
   396→    if query:
   397→        query_lower = query.lower()
   398→        # Define cross-type signal patterns (query signals → should pull this type)
   399→        cross_type_signals = {
   400→            "refund_request": [r"\brefund\b", r"\bmoney\s+back\b", r"\breturn\s+my\b", r"\bprorated?\b", r"\beligibl", r"\btermination\s+fee\b"],
   401→            "billing": [r"\bcharg(?:ed|e|es|ing)\b", r"\binvoice\b", r"\bpayment\b", r"\bovercharge", r"\bdouble\s+charge\b", r"\bpricing\b"],
   402→            "technical": [r"\blogin\b", r"\bss[so]\b", r"\bpassword\b", r"\berror\b", r"\bcrash\b", r"\bnot\s+working\b", r"\bcan'?t\s+access\b", r"\bsession\b", r"\bcache\b"],
   403→            "faq": [r"\bwhat\s+(?:is|are)\b", r"\bhow\s+(?:do|does|much|many)\b", r"\bfeature", r"\b(?:plan|plans?)\s*(?:pricing|cost|include)\b"],
   404→            "complaint": [r"\bterrible\b", r"\bworst\b", r"\bunacceptab", r"\bfrustrat", r"\bdisappoint", r"\bnever\s+again\b"],
   405→            "account_change": [r"\bchange\s+(?:email|password)\b", r"\bupgrade\b", r"\bdowngrade\b", r"\bswitch\s+plan\b", r"\bworkspace\b", r"\bteam\s+member\b"],
   406→        }
   407→        # Determine which types are already covered
   408→        covered_types = {ticket_type} | set(related_types.get(ticket_type, []))
   409→        for cross_type, patterns in cross_type_signals.items():
   410→            if cross_type in covered_types:
   411→                continue
   412→            # If 2+ patterns match, this type is likely relevant
   413→            match_count = sum(1 for pat in patterns if re.search(pat, query_lower))
   414→            if match_count >= 2:
   415→                extra_docs = KNOWLEDGE_BASE.get(cross_type, [])
   416→                docs.extend(extra_docs)
   417→
   418→    # Deduplicate by source
   419→    seen = set()
   420→    unique_docs = []
   421→    for d in docs:
   422→        if d["source"] not in seen:
   423→            seen.add(d["source"])
   424→            unique_docs.append({
   425→                "content": d["content"],
   426→                "source": d["source"],
   427→                "section": d.get("section", "C"),
   428→            })
   429→
   430→    return unique_docs

   433→# ── Phase 4: Smart knowledge filtering (non-LLM) ─────────────────
   434→
   435→
   436→def _filter_relevant_docs(
   437→    documents: List[Dict[str, Any]], query: str, ticket_type: str
   438→) -> List[Dict[str, Any]]:
   439→    """Score each doc by LLM-based semantic relevance to the query.
   440→
   441→    Uses Llama 3.1 8B to rate each KB doc's relevance (0-10) to the
   442→    customer's query. This replaces the old keyword-overlap scoring
   443→    which couldn't understand semantic meaning.
   444→
   445→    Falls back to keyword scoring if LLM is unavailable (BC-008).
   446→    """
   447→    if not documents:
   448→        return documents
   449→
   450→    # ── LLM-based semantic relevance scoring ─────────────────────
   451→    # Ask Llama 3.1 to rate each doc's relevance to the query (0-10).
   452→    # This is the "knowledge base" — the LLM reads the query + doc title
   453→    # and scores how relevant it is. Much better than keyword matching.
   454→    try:
   455→        import asyncio
   456→        from app.core.parwa_pipeline.llm_client import llm_call
   457→
   458→        # Build a single prompt that scores ALL docs at once (1 LLM call, not N)
   459→        doc_list = []
   460→        for i, doc in enumerate(documents):
   461→            title = doc.get("source", "unknown")
   462→            content_preview = doc.get("content", "")[:200]
   463→            doc_list.append(f"DOC {i}: [{title}] {content_preview}")
   464→
   465→        prompt = f"""Rate the relevance of each document to the customer's question (0-10).
   466→
   467→Customer question: "{query}"
   468→
   469→Documents:
   470→{chr(10).join(doc_list)}
   471→
   472→Reply with ONLY the doc numbers and scores, one per line:
   473→0: <score>
   474→1: <score>
   475→2: <score>
   476→..."""
   477→
   478→        async def _score_docs():
   479→            return await llm_call(prompt, max_tokens=100, temperature=0.1)
   480→
   481→        try:
   482→            loop = asyncio.get_event_loop()
   483→            if loop.is_running():
   484→                # Already in async context — run in thread
   485→                import concurrent.futures
   486→                with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   487→                    result = pool.submit(asyncio.run, _score_docs).result(timeout=30)
   488→            else:
   489→                result = asyncio.run(_score_docs())
   490→        except Exception:
   491→            result = asyncio.run(_score_docs())
   492→
   493→        # Parse scores from LLM response
   494→        scores: Dict[int, int] = {}
   495→        for line in result.strip().split("\n"):
   496→            parts = line.strip().split(":")
   497→            if len(parts) == 2:
   498→                try:
   499→                    doc_idx = int(parts[0].strip())
   500→                    score = int(parts[1].strip())
   501→                    scores[doc_idx] = max(0, min(10, score))
   502→                except ValueError:
   503→                    continue
   504→
   505→        # If we got scores, sort docs by LLM relevance score
   506→        if scores and len(scores) >= len(documents) // 2:
   507→            scored_docs = []
   508→            for i, doc in enumerate(documents):
   509→                score = scores.get(i, 0)
   510→                scored_docs.append((score, doc))
   511→            scored_docs.sort(key=lambda x: x[0], reverse=True)
   512→
   513→            # Keep top 8 docs with score >= 3
   514→            filtered = [doc for score, doc in scored_docs if score >= 3][:8]
   515→            if filtered:
   516→                return filtered
   517→            # If all scored low, keep top 5 anyway
   518→            return [doc for _, doc in scored_docs[:5]]
   519→
   520→        # If LLM scoring failed, fall through to keyword scoring below
   521→        logger.warning("LLM relevance scoring failed — falling back to keyword filter")
   522→
   523→    except Exception as exc:
   524→        logger.warning("LLM relevance scoring error: %s — falling back to keyword filter", str(exc)[:200])
   525→
   526→    # ── Fallback: keyword-based relevance scoring (original logic) ──
   527→    query_lower = query.lower()
   528→    query_terms = set(w.lower() for w in query_lower.split() if len(w) > 3)
   529→    type_keywords = {
   530→        "refund_request": {"refund", "prorated", "termination", "fee", "credit", "eligible", "days", "annual", "monthly"},
   531→        "billing": {"charge", "invoice", "payment", "billed", "subscription", "duplicate", "price", "plan", "rate"},
   532→        "technical": {"login", "sso", "password", "access", "error", "cache", "sync", "session", "lockout", "certificate"},
   533→        "faq": {"plan", "pricing", "feature", "channel", "agent", "ticket", "onboarding", "integration"},
   534→        "complaint": {"complaint", "compensation", "credit", "escalate", "resolution", "priority", "tenure"},
   535→        "account_change": {"email", "password", "plan", "upgrade", "downgrade", "workspace", "member", "sso"},
   536→    }
   537→    query_terms |= type_keywords.get(ticket_type, set())
   538→
   539→    filler = {"that", "this", "have", "been", "will", "would", "could", "should",
   540→              "their", "there", "about", "which", "where", "when", "what", "with",
   541→              "from", "your", "just", "also", "than", "them", "they", "some"}
   542→    query_terms -= filler
   543→
   544→    scored_docs = []
   545→    for doc in documents:
   546→        text = doc.get("content", "").lower()
   547→        doc_words = set(text.split())
   548→        overlap = len(query_terms & doc_words)
   549→        query_bigrams = set()
   550→        words = query_lower.split()
   551→        for i in range(len(words) - 1):
   552→            bg = f"{words[i]} {words[i+1]}"
   553→            if len(bg) > 6:
   554→                query_bigrams.add(bg)
   555→        bigram_hits = sum(1 for bg in query_bigrams if bg in text)
   556→        score = overlap + (bigram_hits * 2)
   557→        scored_docs.append((score, doc))
   558→
   559→    scored_docs.sort(key=lambda x: x[0], reverse=True)
   560→    if len(scored_docs) > 8:
   561→        scored_docs = scored_docs[:8]
   562→
   563→    return [doc for _, doc in scored_docs]

   651→    """Check if each doc's content shares meaningful keywords with the query.
   652→
   653→    Non-LLM — word overlap ratio.  A doc that shares 0 keywords with the

   657→        return {"grounded_count": 0, "ungrounded_count": 0, "grounded_ratio": 0.0}

   669→        overlap = len(query_words & doc_words)

   674→    return {

   734→# ── RecencyWeighting: boost newer docs ────────────────────────────
   735→
   736→
   737→def _recency_weighting(documents: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
   738→    """Boost docs that appear more recent based on content clues.
   739→
   740→    Non-LLM — checks for year mentions and "updated/revised/new" keywords.
   741→    Docs with "(Updated 2026)" get a small score boost; docs mentioning
   742→    old years get a small penalty.  Re-sorts by adjusted score.
   743→    """
   744→    if not documents:
   745→        return documents
   746→
   747→    for doc in documents:
   748→        content = doc.get("content", "")
   749→        boost = 0.0
   750→        # Year detection
   751→        if "2026" in content:
   752→            boost += 0.05
   753→        elif "2025" in content:
   754→            boost += 0.02
   755→        elif "2024" in content or "2023" in content:
   756→            boost -= 0.03
   757→        # Freshness keywords
   758→        if re.search(r"\b(?:updated|revised|new|current)\b", content, re.I):
   759→            boost += 0.03
   760→        if re.search(r"\b(?:old|deprecated|legacy|former)\b", content, re.I):
   761→            boost -= 0.03
   762→        doc["score"] = doc.get("score", 0.5) + boost
   763→
   764→    documents.sort(key=lambda d: d.get("score", 0), reverse=True)
   765→    return documents

   792→# ── CoverageAnalysis: did we cover all CLARA areas? ──────────────
   793→
   794→
   795→def _coverage_analysis(
   796→    documents: List[Dict[str, Any]], clara_result: Dict[str, Any], ticket_type: str,
   797→) -> Dict[str, Any]:
   798→    """Check if retrieved docs cover ALL knowledge areas CLARA identified.
   799→
   800→    Non-LLM — extracts keywords from CLARA output, checks if any doc
   801→    mentions each area.  Returns which areas are covered and which
   802→    are gaps.
   803→    """
   804→    areas_text = clara_result.get("relevant_knowledge", "").lower()
   805→    if not areas_text:
   806→        return {"coverage_pct": 1.0, "covered": [], "gaps": []}
   807→
   808→    # Extract area keywords (simple: split by newlines/commas, take meaningful words)
   809→    area_keywords = set()
   810→    for word in re.findall(r"\b\w{4,}\b", areas_text):
   811→        if word not in {"need", "area", "knowledge", "required", "relevant", "information", "about"}:
   812→            area_keywords.add(word)
   813→
   814→    if not area_keywords:
   815→        return {"coverage_pct": 1.0, "covered": list(area_keywords), "gaps": []}
   816→
   817→    covered = set()
   818→    for doc in documents:
   819→        doc_words = set(re.findall(r"\b\w{4,}\b", doc.get("content", "").lower()))
   820→        covered |= (area_keywords & doc_words)
   821→
   822→    gaps = area_keywords - covered
   823→    coverage = len(covered) / len(area_keywords) if area_keywords else 1.0
   824→
   825→    return {
   826→        "coverage_pct": round(coverage, 2),
   827→        "covered": sorted(covered),
   828→        "gaps": sorted(gaps),
   829→    }

   832→# ── NearDedup: remove near-duplicate chunks ───────────────────────
   833→
   834→
   835→def _near_dedup(documents: List[Dict[str, Any]], similarity_threshold: float = 0.75) -> List[Dict[str, Any]]:
   836→    """Remove near-duplicate chunks that say the same thing differently.
   837→
   838→    Non-LLM — Jaccard similarity on word sets.  Two chunks that share
   839→    >75% of their words are probably saying the same thing; keep the
   840→    one with the higher score.
   841→    """
   842→    if len(documents) <= 1:
   843→        return documents
   844→
   845→    kept = [documents[0]]
   846→    for doc in documents[1:]:
   847→        doc_words = set(re.findall(r"\b\w{4,}\b", doc.get("content", "").lower()))
   848→        is_duplicate = False
   849→        for existing in kept:
   850→            existing_words = set(re.findall(r"\b\w{4,}\b", existing.get("content", "").lower()))
   851→            if not doc_words or not existing_words:
   852→                continue
   853→            jaccard = len(doc_words & existing_words) / len(doc_words | existing_words)
   854→            if jaccard >= similarity_threshold:
   855→                is_duplicate = True
   856→                # Keep the one with higher score
   857→                if doc.get("score", 0) > existing.get("score", 0):
   858→                    kept.remove(existing)
   859→                    kept.append(doc)
   860→                break
   861→        if not is_duplicate:
   862→            kept.append(doc)
   863→
   864→    return kept

   984→# ── Enhanced ContradictionCheck: structured field comparison ──────
   985→
   986→
   987→def _check_contradictions_enhanced(documents: List[Dict[str, Any]], ticket_type: str) -> Dict[str, Any]:
   988→    """Structured contradiction detection per ticket type.
   989→
   990→    Non-LLM — extracts specific fields (days, amounts, percentages)
   991→    and checks for real conflicts, not just % mismatches.  Handles
   992→    scoped contradictions (e.g., different rules for different plans).
   993→    """
   994→    if len(documents) < 2:
   995→        return {"found": False, "details": []}
   996→
   997→    contradictions = []
   998→
   999→    # Per-type field patterns to extract and compare
  1000→    field_patterns = {
  1001→        "refund_request": [
  1002→            (r"(\d+)\s*day", "refund_window_days"),
  1003→            (r"(\d+)%", "fee_percentage"),
  1004→            (r"\$(\d[\d,]*)", "dollar_amount"),
  1005→        ],
  1006→        "billing": [
  1007→            (r"\$(\d[\d,]*)", "dollar_amount"),
  1008→            (r"(\d+)\s*day", "processing_days"),
  1009→            (r"(\d+)%", "percentage"),
  1010→        ],
  1011→        "technical": [
  1012→            (r"(\d+)\s*min", "lockout_minutes"),
  1013→            (r"(\d+)\s*attempt", "max_attempts"),
  1014→            (r"(\d+)\s*hour", "hours"),
  1015→        ],
  1016→    }
  1017→
  1018→    patterns = field_patterns.get(ticket_type, [(r"(\d+)%", "percentage")])
  1019→
  1020→    for pattern, field_name in patterns:
  1021→        # Extract all values for this field across docs
  1022→        values_by_doc = {}
  1023→        for i, doc in enumerate(documents):
  1024→            content = doc.get("content", "")
  1025→            # Scope: look for context keywords near the value
  1026→            context_window = 80
  1027→            for match in re.finditer(pattern, content):
  1028→                value = match.group(1).replace(",", "")
  1029→                start = max(0, match.start() - context_window)
  1030→                end = min(len(content), match.end() + context_window)
  1031→                context = content[start:end].lower()
  1032→
  1033→                # Determine scope (which plan/type does this apply to?)
  1034→                scope = "general"
  1035→                if "mini" in context:
  1036→                    scope = "mini"
  1037→                elif "high" in context or "pro" in context:
  1038→                    scope = "high"
  1039→                elif "annual" in context:
  1040→                    scope = "annual"
  1041→                elif "monthly" in context:
  1042→                    scope = "monthly"
  1043→
  1044→                key = (i, field_name, scope)
  1045→                if key not in values_by_doc:
  1046→                    values_by_doc[key] = []
  1047→                values_by_doc[key].append(value)
  1048→
  1049→        # Check for contradictions WITHIN same scope
  1050→        scope_values: Dict[str, set] = {}
  1051→        for (doc_i, fn, scope), vals in values_by_doc.items():
  1052→            if scope not in scope_values:
  1053→                scope_values[scope] = set()
  1054→            for v in vals:
  1055→                scope_values[scope].add(v)
  1056→
  1057→        for scope, vals in scope_values.items():
  1058→            if len(vals) > 1:
  1059→                contradictions.append({
  1060→                    "field": field_name,
  1061→                    "scope": scope,
  1062→                    "conflicting_values": list(vals),
  1063→                    "type": "SCOPE_CONFLICT" if scope != "general" else "GENERAL_CONFLICT",
  1064→                })
  1065→
  1066→    return {
  1067→        "found": len(contradictions) > 0,
  1068→        "details": contradictions,
  1069→    }

  1166→    """Boost docs that match historically successful patterns from AI Wiki.

  1320→# ── Layer 1 (CLARA): QueryDecomposition ──────────────────────────

  1322→
  1323→def _query_decomposition(query: str, ticket_type: str) -> Dict[str, Any]:
  1324→    """Break multi-intent queries into sub-queries before CLARA.
  1325→
  1326→    Non-LLM — splits on conjunctions (and, but, also, plus).
  1327→    "I was double charged AND I want a refund" -> 2 focused queries.
  1328→    """
  1329→    connectors = [
  1330→        r"\band\b", r"\bbut\b", r"\balso\b",
  1331→        r"\bplus\b", r"\bhowever\b", r"\bas well as\b",
  1332→    ]
  1333→
  1334→    parts = [query]
  1335→    for connector in connectors:
  1336→        new_parts = []
  1337→        for part in parts:
  1338→            splits = re.split(connector, part, flags=re.I)
  1339→            new_parts.extend(s.strip() for s in splits if s.strip())
  1340→        parts = new_parts
  1341→
  1342→    sub_queries = [p for p in parts if len(p) >= 10]
  1343→
  1344→    if len(sub_queries) <= 1:
  1345→        return {"decomposed": False, "sub_queries": [query], "count": 1}
  1346→
  1347→    return {"decomposed": True, "sub_queries": sub_queries, "count": len(sub_queries)}

  1350→# ── Layer 1 (CLARA): IntentSignalBoost ───────────────────────────
  1351→
  1352→
  1353→def _intent_signal_boost(
  1354→    query: str, customer_context: Dict[str, Any], ticket_type: str,
  1355→) -> Dict[str, Any]:
  1356→    """Extract intent signals (urgency, money, VIP) from query + context.
  1357→
  1358→    Non-LLM — pattern matching.  Downstream steps know this is a
  1359→    "$500 billing issue", not just "billing".
  1360→    """
  1361→    query_lower = query.lower()
  1362→    signals: Dict[str, Any] = {}
  1363→
  1364→    # Urgency signals
  1365→    urgency_patterns = [
  1366→        r"\burgent\b", r"\basap\b", r"\bimmediately\b",
  1367→        r"\bright now\b", r"\bemergency\b",
  1368→    ]
  1369→    signals["urgent"] = any(re.search(p, query_lower) for p in urgency_patterns)
  1370→
  1371→    # Money signals
  1372→    money_match = re.search(r"\$(\d[\d,]*)", query)
  1373→    if money_match:
  1374→        amount = int(money_match.group(1).replace(",", ""))
  1375→        signals["money_mentioned"] = True
  1376→        signals["money_amount"] = amount
  1377→        signals["high_value"] = amount >= 500
  1378→    else:
  1379→        signals["money_mentioned"] = False
  1380→
  1381→    # VIP signals
  1382→    tier = str(
  1383→        customer_context.get("account_tier",
  1384→                             customer_context.get("plan_type", ""))
  1385→    ).lower()
  1386→    signals["is_vip"] = tier in ("high", "enterprise", "pro")
  1387→    signals["tier"] = tier
  1388→
  1389→    # Frustration signals
  1390→    frustration_patterns = [
  1391→        r"\bangry\b", r"\bfrustrat\b", r"\bfurious\b",
  1392→        r"\bunacceptable\b", r"\bterrible\b",
  1393→    ]
  1394→    signals["frustrated"] = any(
  1395→        re.search(p, query_lower) for p in frustration_patterns
  1396→    )
  1397→
  1398→    # Repeat-issue signals
  1399→    recent_tickets = customer_context.get("recent_ticket_count", 0)
  1400→    signals["repeat_issue"] = isinstance(recent_tickets, (int, float)) and recent_tickets >= 3
  1401→
  1402→    active_count = sum(1 for v in signals.values() if v is True)
  1403→    return {"signals": signals, "signal_count": active_count}

  1406→# ── Layer 2 (RAG): SmartRouter for RAG search strategy ──────────

  1409→def _smart_router_rag(ticket_type: str) -> Dict[str, Any]:
  1410→    """Route RAG search strategy based on ticket type.
  1411→
  1412→    Non-LLM — defines which KB sources to prioritise per type.

  1450→# ── Layer 3 (SmartFilter): SelfConsistency for filter ────────────
  1451→
  1452→
  1453→def _self_consistency_filter(
  1454→    documents: List[Dict[str, Any]], query: str, ticket_type: str,
  1455→) -> Dict[str, Any]:
  1456→    """Cross-validate: do keyword relevance and document content agree?
  1457→
  1458→    Non-LLM — if a doc ranks high but shares no meaningful keywords
  1459→    with the query, it might be a false positive from LLM scoring.
  1460→    """
  1461→    if not documents:
  1462→        return {"consistent": True, "suspicious_docs": 0}
  1463→
  1464→    query_words = set(re.findall(r"\b\w{4,}\b", query.lower()))
  1465→    filler = {
  1466→        "that", "this", "have", "been", "will", "would", "could",
  1467→        "should", "their", "there", "about", "which", "where",
  1468→        "when", "what", "with", "from", "your", "just", "also",
  1469→        "than", "them", "they", "some",
  1470→    }
  1471→    query_words -= filler
  1472→
  1473→    suspicious = 0
  1474→    for doc in documents:
  1475→        score = doc.get("score", 0)
  1476→        if score >= 0.7:
  1477→            doc_words = set(
  1478→                re.findall(r"\b\w{4,}\b", doc.get("content", "").lower())
  1479→            )
  1480→            overlap = len(query_words & doc_words)
  1481→            if overlap < 2:
  1482→                suspicious += 1
  1483→
  1484→    return {
  1485→        "consistent": suspicious == 0,
  1486→        "suspicious_docs": suspicious,
  1487→        "total_docs": len(documents),
  1488→    }

  1491→# ── Layer 3 (SmartFilter): RelevanceDecay ────────────────────────
  1492→
  1493→
  1494→def _relevance_decay(
  1495→    documents: List[Dict[str, Any]], ticket_type: str,
  1496→) -> List[Dict[str, Any]]:
  1497→    """Score drops the further a doc's topic is from the core query topic.
  1498→
  1499→    Non-LLM — docs about "billing policy history" are less relevant
  1500→    than "current billing policy" for a billing dispute.
  1501→    """
  1502→    if not documents:
  1503→        return documents
  1504→
  1505→    core_topics = {
  1506→        "refund_request": ["refund", "eligible", "prorated", "termination", "fee"],
  1507→        "billing": ["charge", "invoice", "payment", "duplicate", "price"],
  1508→        "technical": ["login", "sso", "password", "access", "error"],
  1509→        "faq": ["plan", "feature", "pricing", "channel"],
  1510→        "complaint": ["complaint", "compensat", "escalat", "credit"],
  1511→        "account_change": ["change", "email", "password", "upgrade", "workspace"],
  1512→    }
  1513→
  1514→    core_kws = set(core_topics.get(ticket_type, []))
  1515→
  1516→    for doc in documents:
  1517→        content_lower = doc.get("content", "").lower()
  1518→        doc_words = set(re.findall(r"\b\w{4,}\b", content_lower))
  1519→        core_overlap = len(core_kws & doc_words)
  1520→
  1521→        if core_overlap == 0:
  1522→            doc["score"] = doc.get("score", 0.5) - 0.1
  1523→        elif core_overlap >= 2:
  1524→            doc["score"] = doc.get("score", 0.5) + 0.05
  1525→
  1526→    documents.sort(key=lambda d: d.get("score", 0), reverse=True)
  1527→    return documents
  1528→

  1530→# ── Layer 3 (SmartFilter): OverlapMinimization ───────────────────
  1531→
  1532→
  1533→def _overlap_minimization(
  1534→    documents: List[Dict[str, Any]], overlap_threshold: float = 0.5,
  1535→) -> List[Dict[str, Any]]:
  1536→    """If 2 docs cover the same topic, keep the more comprehensive one.
  1537→
  1538→    Non-LLM — topic overlap check.  If Doc A and Doc B share >50%
  1539→    of their topic keywords, keep the longer one (more comprehensive).
  1540→    """
  1541→    if len(documents) <= 1:
  1542→        return documents
  1543→
  1544→    kept = [documents[0]]
  1545→    for doc in documents[1:]:
  1546→        doc_words = set(
  1547→            re.findall(r"\b\w{4,}\b", doc.get("content", "").lower())
  1548→        )
  1549→        is_overlap = False
  1550→        for existing in kept:
  1551→            existing_words = set(
  1552→                re.findall(
  1553→                    r"\b\w{4,}\b", existing.get("content", "").lower()
  1554→                )
  1555→            )
  1556→            if not doc_words or not existing_words:
  1557→                continue
  1558→            min_len = min(len(doc_words), len(existing_words))
  1559→            topic_overlap = len(doc_words & existing_words) / min_len if min_len else 0
  1560→            if topic_overlap >= overlap_threshold:
  1561→                if len(doc.get("content", "")) > len(existing.get("content", "")):
  1562→                    kept.remove(existing)
  1563→                    kept.append(doc)
  1564→                is_overlap = True
  1565→                break
  1566→        if not is_overlap:
  1567→            kept.append(doc)
  1568→
  1569→    return kept
  1570→

  1572→# ── Layer 4 (Sufficiency): CompletenessTracker ───────────────────

  1578→    """Track what % of the query's information needs are met by docs.
  1579→
  1580→    Non-LLM — extracts key entities from query, checks if any doc

  1584→    if not query or not documents:
  1585→        return {
  1586→            "completeness_pct": 0.0,
  1587→            "needs_identified": 0,
  1588→            "needs_met": 0,
  1589→        }
  1590→
  1591→    query_lower = query.lower()
  1592→
  1593→    need_patterns = {
  1594→        "refund_request": [
  1595→            r"refund", r"money\s+back", r"cancel",
  1596→            r"prorat", r"terminat", r"credit",
  1597→        ],
  1598→        "billing": [
  1599→            r"charge", r"invoice", r"payment",
  1600→            r"duplicat", r"overcharge", r"price", r"rate",
  1601→        ],
  1602→        "technical": [
  1603→            r"login", r"sso", r"password",
  1604→            r"access", r"error", r"crash", r"session",

  1606→        "faq": [

  1616→    }
  1617→
  1618→    patterns = need_patterns.get(ticket_type, [r"\w{5,}"])

  1629→    met = sum(1 for pat in needs if re.search(pat, doc_content))
  1630→
  1631→    return {
  1632→        "completeness_pct": round(met / len(needs), 2) if needs else 1.0,
  1633→        "needs_identified": len(needs),
  1634→        "needs_met": met,
  1635→        "needs_missing": len(needs) - met,

  1678→# ── Layer 5 (Contradiction): TemporalChecker ─────────────────────
  1679→
  1680→
  1681→def _temporal_checker(documents: List[Dict[str, Any]]) -> Dict[str, Any]:
  1682→    """Check if docs have different effective dates.
  1683→
  1684→    Non-LLM — extracts year/version from doc content.  If two docs
  1685→    disagree on policy year, the older one is likely stale.
  1686→    """
  1687→    if len(documents) < 2:
  1688→        return {"has_stale_docs": False, "versions": [], "stale_sources": []}
  1689→
  1690→    versions = []
  1691→    for doc in documents:
  1692→        content = doc.get("content", "")
  1693→        source = doc.get("source", "")
  1694→        year_match = re.search(r"\b(20[2-3]\d)\b", content)
  1695→        ver_match = (
  1696→            re.search(r"v(\d+)", source)
  1697→            or re.search(r"version\s*(\d+)", content, re.I)
  1698→        )
  1699→        versions.append({
  1700→            "source": source,
  1701→            "year": int(year_match.group(1)) if year_match else None,
  1702→            "version": ver_match.group(1) if ver_match else None,
  1703→        })
  1704→
  1705→    years = [v["year"] for v in versions if v["year"] is not None]
  1706→    has_stale = False
  1707→    stale_sources = []
  1708→    if years and len(set(years)) > 1:
  1709→        max_year = max(years)
  1710→        for v in versions:
  1711→            if v["year"] and v["year"] < max_year:
  1712→                has_stale = True
  1713→                stale_sources.append(v["source"])
  1714→
  1715→    return {
  1716→        "has_stale_docs": has_stale,
  1717→        "stale_sources": stale_sources,
  1718→        "versions": versions,
  1719→    }

  1722→# ── Layer 5 (Contradiction): VersionTracker ──────────────────────
  1723→
  1724→
  1725→def _version_tracker(documents: List[Dict[str, Any]]) -> Dict[str, Any]:
  1726→    """Track which version of each policy doc is active.
  1727→
  1728→    Non-LLM — if we see both v1 and v2 of a doc, v2 supersedes v1.
  1729→    """
  1730→    if not documents:
  1731→        return {"active_versions": {}, "superseded": [], "has_superseded": False}
  1732→
  1733→    doc_versions: Dict[str, list] = {}
  1734→    for doc in documents:
  1735→        source = doc.get("source", "")
  1736→        match = re.match(r"(.+?)(?:_v(\d+))?(?:_.*)?$", source)
  1737→        if match:
  1738→            base = match.group(1)
  1739→            ver = int(match.group(2)) if match.group(2) else 1
  1740→        else:
  1741→            base = source
  1742→            ver = 1
  1743→
  1744→        doc_versions.setdefault(base, []).append({
  1745→            "source": source,
  1746→            "version": ver,
  1747→        })
  1748→
  1749→    active: Dict[str, str] = {}
  1750→    superseded = []
  1751→    for base, vers in doc_versions.items():
  1752→        vers.sort(key=lambda x: x["version"], reverse=True)
  1753→        active[base] = vers[0]["source"]
  1754→        for v in vers[1:]:
  1755→            superseded.append(v["source"])
  1756→
  1757→    return {
  1758→        "active_versions": active,
  1759→        "superseded": superseded,
  1760→        "has_superseded": len(superseded) > 0,
  1761→    }

  1937→# ── Layer 7 (AI Wiki): ConflictResolution ────────────────────────

  1940→def _wiki_conflict_resolution(

  1942→) -> Dict[str, Any]:
  1943→    """When wiki patterns disagree, pick the best one.

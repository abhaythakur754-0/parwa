   199→def _detect_capability(query: str, ticket_type: str) -> str | None:
   200→    """Detect which capability this query needs.
   201→
   202→    First checks the direct ticket_type → capability mapping (fast path).
   203→    Then checks CAPABILITY_PATTERNS for capabilities without a built-in
   204→    ticket_type (legal_review, fraud_security, etc.).
   205→
   206→    Returns the capability key, or None if no capability matched.
   207→    """
   208→    # Fast path: built-in ticket_type maps to a capability
   209→    if ticket_type in TICKET_TYPE_TO_CAPABILITY:
   210→        return TICKET_TYPE_TO_CAPABILITY[ticket_type]
   211→
   212→    # Direct pattern match for capabilities without a built-in ticket_type
   213→    query_lower = query.lower()
   214→    for capability, patterns in CAPABILITY_PATTERNS.items():
   215→        for pat in patterns:
   216→            if re.search(pat, query_lower):
   217→                return capability
   218→
   219→    return None

   222→async def _llm_match_capability(query: str, tenant_id: str) -> str | None:
   223→    """LLM-based capability matcher — fallback when rule-based matcher returns None.
   224→
   225→    Uses NVIDIA Llama 3.1 8B (cheap, ~500ms, ~$0.0001/call) to classify the
   226→    query against the 24-capability vocabulary. Also considers any custom
   227→    capabilities the tenant has defined on their agents.
   228→
   229→    Args:
   230→        query: The customer's ticket text (truncated to 500 chars to keep
   231→               token usage low).
   232→        tenant_id: The tenant's company_id — used to fetch any custom
   233→                   capabilities the tenant has defined on their agents.
   234→
   235→    Returns:
   236→        Capability key string, or None if the LLM call fails or returns
   237→        an unrecognized capability.
   238→    """
   239→    # Fetch tenant's custom capabilities (if any) to extend the vocabulary
   240→    tenant_capabilities: list[str] = []
   241→    try:
   242→        from database.base import SessionLocal
   243→        from database.models.variant_engine import AIAgentAssignment
   244→        import json as _json
   245→        db = SessionLocal()
   246→        try:
   247→            rows = db.query(AIAgentAssignment).filter(
   248→                AIAgentAssignment.company_id == tenant_id,
   249→                AIAgentAssignment.status == "active",
   250→            ).all()
   251→            for row in rows:
   252→                try:
   253→                    caps = _json.loads(row.capabilities or "[]")
   254→                    if isinstance(caps, list):
   255→                        for c in caps:
   256→                            if c and c not in tenant_capabilities and c not in ALL_CAPABILITIES:
   257→                                tenant_capabilities.append(c)
   258→                except (_json.JSONDecodeError, TypeError):
   259→                    pass
   260→        finally:
   261→            db.close()
   262→    except Exception as exc:
   263→        logger.warning(
   264→            "llm_capability_matcher_tenant_caps_failed tenant=%s err=%s",
   265→            tenant_id, str(exc)[:150],
   266→        )
   267→
   268→    # Build the capability vocabulary for the LLM
   269→    vocab = ALL_CAPABILITIES + tenant_capabilities
   270→    vocab_str = ", ".join(vocab)
   271→
   272→    # Truncate query to keep token usage low
   273→    query_trunc = query[:500]
   274→
   275→    messages = [
   276→        {
   277→            "role": "system",
   278→            "content": (
   279→                "You are a ticket classifier. Given a customer query, return "
   280→                "the single best matching capability key from this list. "
   281→                "Return ONLY the capability key, nothing else. "
   282→                f"Capabilities: {vocab_str}"
   283→            ),
   284→        },
   285→        {
   286→            "role": "user",
   287→            "content": query_trunc,
   288→        },
   289→    ]
   290→
   291→    try:
   292→        from app.core.parwa_pipeline.pipeline_config import (
   293→            NVIDIA_API_KEY, NVIDIA_API_BASE, NVIDIA_MODEL,
   294→        )
   295→        if not NVIDIA_API_KEY:
   296→            logger.warning("llm_capability_matcher_no_nvidia_key")
   297→            return None
   298→
   299→        import httpx
   300→        import asyncio
   301→        url = f"{NVIDIA_API_BASE}/chat/completions"
   302→        payload = {
   303→            "model": NVIDIA_MODEL,
   304→            "messages": messages,
   305→            "temperature": 0.0,  # deterministic — same query = same capability
   306→            "max_tokens": 20,    # capability keys are short
   307→        }
   308→        headers = {
   309→            "Authorization": f"Bearer {NVIDIA_API_KEY}",
   310→            "Content-Type": "application/json",
   311→        }
   312→
   313→        async with httpx.AsyncClient(timeout=10.0) as client:
   314→            r = await client.post(url, json=payload, headers=headers)
   315→
   316→        if r.status_code != 200:
   317→            logger.warning(
   318→                "llm_capability_matcher_http_%d: %s",
   319→                r.status_code, r.text[:200],
   320→            )
   321→            return None
   322→
   323→        content = r.json()["choices"][0]["message"]["content"].strip().lower()
   324→        # Normalize: strip quotes, whitespace, trailing punctuation
   325→        content = content.strip("`'\".,;: \n\r\t")
   326→
   327→        # Validate the LLM returned a real capability
   328→        if content in vocab:
   329→            logger.info(
   330→                "llm_capability_matcher_hit tenant=%s capability=%s",
   331→                tenant_id, content,
   332→            )
   333→            return content
   334→        else:
   335→            logger.warning(
   336→                "llm_capability_matcher_unknown_response tenant=%s response=%s",
   337→                tenant_id, content[:80],
   338→            )
   339→            return None
   340→
   341→    except Exception as exc:
   342→        logger.warning(
   343→            "llm_capability_matcher_failed tenant=%s err=%s",
   344→            tenant_id, str(exc)[:150],
   345→        )
   346→        return None

   349→def _tenant_claims_capability(tenant_id: str, capability: str) -> bool:
   350→    """Check if the tenant has any active agent that claims this capability.
   351→
   352→    Agents created via the builder have free-text capabilities (e.g.
   353→    "refund", "billing", "booking") while _detect_capability returns
   354→    standard vocabulary keys (e.g. "refund_processing", "billing_inquiry",
   355→    "booking_reservation"). We match by substring in BOTH directions so
   356→    "refund" matches "refund_processing" and vice versa.
   357→
   358→    Returns True if at least one agent claims it, False otherwise.
   359→    """
   360→    if not capability:
   361→        return False
   362→    try:
   363→        from database.base import SessionLocal
   364→        from sqlalchemy import text
   365→        import json as _json
   366→        db = SessionLocal()
   367→        try:
   368→            # Pull all active agents' capabilities for this tenant
   369→            rows = db.execute(
   370→                text(
   371→                    "SELECT capabilities FROM ai_agent_assignments "
   372→                    "WHERE company_id = :tenant_id AND status = 'active'"
   373→                ),
   374→                {"tenant_id": tenant_id},
   375→            ).fetchall()
   376→            cap_lower = capability.lower()
   377→            for (caps_json,) in rows:
   378→                if not caps_json:
   379→                    continue
   380→                try:
   381→                    caps = _json.loads(caps_json) if isinstance(caps_json, str) else caps_json
   382→                except (ValueError, TypeError):
   383→                    continue
   384→                if not isinstance(caps, list):
   385→                    continue
   386→                for c in caps:
   387→                    if not isinstance(c, str):
   388→                        continue
   389→                    c_lower = c.lower().strip()
   390→                    if not c_lower:
   391→                        continue
   392→                    # Substring match in either direction:
   393→                    # "refund" matches "refund_processing"
   394→                    # "billing" matches "billing_inquiry"
   395→                    if c_lower in cap_lower or cap_lower in c_lower:
   396→                        return True
   397→                    # Word-level overlap: handles long sentence-style
   398→                    # capabilities from the builder like "Hotel selection
   399→                    # and booking" matching standard key "booking_reservation"
   400→                    # Split both on non-alphanumeric, check if any meaningful
   401→                    # word (4+ chars) from the standard key appears in the
   402→                    # agent capability.
   403→                    import re
   404→                    cap_words = set(w for w in re.split(r'[^a-z]+', cap_lower) if len(w) >= 4)
   405→                    agent_words = set(w for w in re.split(r'[^a-z]+', c_lower) if len(w) >= 4)
   406→                    if cap_words & agent_words:
   407→                        return True
   408→            return False
   409→        finally:
   410→            db.close()
   411→    except Exception as exc:
   412→        logger.warning(
   413→            "tenant_capability_check_failed tenant=%s capability=%s err=%s",
   414→            tenant_id, capability, str(exc)[:200],
   415→        )
   416→        # Fail-safe: if DB query fails, assume tenant doesn't claim it
   417→        # so the ticket escalates to human rather than silently auto-resolving.
   418→        return False

   421→def _check_custom_categories(tenant_id: str, query: str) -> dict | None:
   422→    """Check Builder-created custom categories for keyword matches.
   423→
   424→    When the Builder creates an agent with a custom category, it stores
   425→    trigger keywords in the custom_categories table. This function checks
   426→    if the query matches any custom category's keywords, returning the
   427→    category info so Node 1 can route to the right agent.
   428→
   429→    ROADMAP Phase 5: "Custom categories + keywords created by Builder
   430→    appear in Node 1 classification."
   431→
   432→    Returns {"name": str, "capability": str, "agent_id": str} or None.
   433→    """
   434→    try:
   435→        from database.base import SessionLocal
   436→        from sqlalchemy import text
   437→        import json as _json
   438→
   439→        db = SessionLocal()
   440→        try:
   441→            rows = db.execute(
   442→                text(
   443→                    "SELECT name, keywords, agent_id FROM custom_categories "
   444→                    "WHERE company_id = :tid AND is_active = 1"
   445→                ),
   446→                {"tid": tenant_id},
   447→            ).fetchall()
   448→
   449→            query_lower = query.lower()
   450→
   451→            for name, keywords_json, agent_id in rows:
   452→                try:
   453→                    keywords = _json.loads(keywords_json) if isinstance(keywords_json, str) else (keywords_json or [])
   454→                except (ValueError, TypeError):
   455→                    continue
   456→                if not isinstance(keywords, list):
   457→                    continue
   458→
   459→                # Check if any keyword appears in the query
   460→                for kw in keywords:
   461→                    if isinstance(kw, str) and kw.lower() in query_lower:
   462→                        # Use the category name as the capability key
   463→                        return {
   464→                            "name": name,
   465→                            "capability": name.lower().replace(" ", "_"),
   466→                            "agent_id": agent_id,
   467→                        }
   468→
   469→            return None
   470→        finally:
   471→            db.close()
   472→    except Exception as exc:
   473→        logger.warning(
   474→            "custom_category_check_failed tenant=%s err=%s",
   475→            tenant_id, str(exc)[:150],
   476→        )
   477→        return None

   504→def _enhance_agent_with_non_llm_techniques(
   505→    capability: str,
   506→    query: str,
   507→    tenant_id: str,
   508→    capabilities: list,
   509→    restrictions: str,
   510→    instructions: str,
   511→) -> tuple:
   512→    """Apply ALL non-LLM techniques to enhance the agent before saving.
   513→
   514→    0 LLM calls. 0 cost. 0 latency. Pure Python keyword/regex matching.
   515→
   516→    Techniques applied:
   517→      1. CAPABILITY_PATTERNS — scan query against 24 patterns, add ALL matches
   518→      2. COMPLEXITY_KEYWORDS_HARD — detect complex keywords, add escalation restrictions
   519→      3. RuleBasedAction — check tier, set max refund/credit limits
   520→      4. IdempotencyCheck — check if agent name already exists (returns None if duplicate)
   521→      5. ZeroShotValidator — flag unusual patterns, add caution restriction
   522→      6. SmartRouter.classify — use ticket type for better agent naming
   523→      7. SmartRouter.action — extract action, add to capabilities
   524→      8. DynamicContext — check past ticket patterns for this tenant
   525→      9. MetaLearner — suggest additional capabilities based on patterns
   526→      10. SufficiencyCheck — check if KB has docs for this capability
   527→      11. MAKER.ReverseCheck — check if instructions have KB keyword bridges
   528→      12. Escalation rules — add escalation rules based on ticket type
   529→      13. GST — add goal-state tracking to instructions
   530→
   531→    Returns: (enhanced_capabilities, enhanced_restrictions, enhanced_instructions, agent_name)
   532→             or (None, None, None, None) if duplicate detected.
   533→    """
   534→    import re
   535→
   536→    enhanced_caps = list(capabilities)
   537→    enhanced_restrictions = restrictions
   538→    enhanced_instructions = instructions
   539→    display = capability.replace("_", " ").title()
   540→
   541→    # ── 1. CAPABILITY_PATTERNS — scan query against all 24 patterns ──
   542→    for cap_key, patterns in CAPABILITY_PATTERNS.items():
   543→        if cap_key in enhanced_caps:
   544→            continue
   545→        for pattern in patterns:
   546→            if re.search(pattern, query, re.IGNORECASE):
   547→                enhanced_caps.append(cap_key)
   548→                break
   549→
   550→    # ── 2. COMPLEXITY_KEYWORDS_HARD — detect complex keywords ──
   551→    query_lower = query.lower()
   552→    complex_keywords_found = [kw for kw in COMPLEXITY_KEYWORDS_HARD if kw in query_lower]
   553→    if complex_keywords_found:
   554→        enhanced_restrictions += (
   555→            f" COMPLEXITY ALERT: This ticket contains complex keywords "
   556→            f"({', '.join(complex_keywords_found[:3])}). Always escalate to "
   557→            f"human if the customer mentions legal action, lawsuits, or "
   558→            f"demands a manager."
   559→        )
   560→
   561→    # ── 3. RuleBasedAction — check tier, set limits ──
   562→    try:
   563→        from database.base import SessionLocal
   564→        from database.models.core import User, Company
   565→        db = SessionLocal()
   566→        try:
   567→            company = db.query(Company).filter(Company.id == tenant_id).first()
   568→            tier = getattr(company, "plan", "mini") if company else "mini"
   569→            if tier == "mini":
   570→                enhanced_restrictions += " Max refund $0 (recommend only). Max credit $0."
   571→            elif tier == "parwa":
   572→                enhanced_restrictions += " Max refund $500 without guidance. Max credit $200."
   573→            elif tier == "high":
   574→                enhanced_restrictions += " Max refund unlimited (with verification). Max credit unlimited."
   575→        finally:
   576→            db.close()
   577→    except Exception:
   578→        pass  # Don't block agent creation if tier check fails
   579→
   580→    # ── 4. IdempotencyCheck — prevent duplicates ──
   581→    try:
   582→        from database.base import SessionLocal
   583→        from database.models.variant_engine import AIAgentAssignment
   584→        db = SessionLocal()
   585→        try:
   586→            existing = db.query(AIAgentAssignment).filter(
   587→                AIAgentAssignment.company_id == tenant_id,
   588→                AIAgentAssignment.agent_name == f"Auto: {display}",
   589→                AIAgentAssignment.status == "active",
   590→            ).first()
   591→            if existing:
   592→                logger.info("auto_create idempotency: agent already exists for %s", capability)
   593→                return (None, None, None, None)
   594→        finally:
   595→            db.close()
   596→    except Exception:
   597→        pass
   598→
   599→    # ── 5. ZeroShotValidator — flag unusual patterns ──
   600→    unusual = False
   601→    if "legal" in capability and "hospitality" in query_lower:
   602→        enhanced_restrictions += " UNUSUAL: Legal agent for hospitality — add extra caution."
   603→        unusual = True
   604→    if "fraud" in capability and "ecommerce" not in query_lower and "bank" not in query_lower:
   605→        enhanced_restrictions += " UNUSUAL: Fraud agent for non-financial industry — verify."
   606→        unusual = True
   607→
   608→    # ── 6. SmartRouter.classify — better naming ──
   609→    # Already handled by display = capability.replace("_", " ").title()
   610→
   611→    # ── 7. SmartRouter.action — extract action keywords ──
   612→    action_keywords = {
   613→        "refund": "refund_processing",
   614→        "cancel": "cancellation",
   615→        "modify": "account_change",
   616→        "track": "shipping_delivery",
   617→        "complain": "complaint_handling",
   618→        "book": "booking_reservation",
   619→    }
   620→    for kw, cap in action_keywords.items():
   621→        if kw in query_lower and cap not in enhanced_caps:
   622→            enhanced_caps.append(cap)
   623→
   624→    # ── 8. DynamicContext — check past ticket patterns ──
   625→    try:
   626→        from database.base import SessionLocal
   627→        from database.models.tickets import Ticket
   628→        db = SessionLocal()
   629→        try:
   630→            past_count = db.query(Ticket).filter(
   631→                Ticket.company_id == tenant_id,
   632→            ).count()
   633→            if past_count > 10:
   634→                enhanced_instructions += (
   635→                    f" This tenant has {past_count} past tickets — "
   636→                    f"check for recurring patterns."
   637→                )
   638→        finally:
   639→            db.close()
   640→    except Exception:
   641→        pass
   642→
   643→    # ── 9. MetaLearner — suggest additional capabilities ──
   644→    try:
   645→        from database.base import SessionLocal
   646→        from database.models.tickets import Ticket
   647→        db = SessionLocal()
   648→        try:
   649→            past_tickets = db.query(Ticket).filter(
   650→                Ticket.company_id == tenant_id,
   651→            ).limit(50).all()
   652→            past_categories = set()
   653→            for t in past_tickets:
   654→                if t.category:
   655→                    past_categories.add(t.category)
   656→            # If tenant has had billing tickets before, add billing capability
   657→            if "billing_payments" in past_categories and "billing_inquiry" not in enhanced_caps:
   658→                enhanced_caps.append("billing_inquiry")
   659→            if "complaints" in past_categories and "complaint_handling" not in enhanced_caps:
   660→                enhanced_caps.append("complaint_handling")
   661→        finally:
   662→            db.close()
   663→    except Exception:
   664→        pass
   665→
   666→    # ── 10. SufficiencyCheck — check if KB has docs ──
   667→    try:
   668→        from database.base import SessionLocal
   669→        from sqlalchemy import text
   670→        db = SessionLocal()
   671→        try:
   672→            kb_count = db.execute(
   673→                text("SELECT count(*) FROM document_chunks WHERE company_id = :tid"),
   674→                {"tid": tenant_id},
   675→            ).scalar()
   676→            if not kb_count or int(kb_count) == 0:
   677→                enhanced_restrictions += " WARNING: No KB docs uploaded — agent will have limited knowledge."
   678→            else:
   679→                enhanced_instructions += f" {kb_count} KB chunks available for reference."
   680→        finally:
   681→            db.close()
   682→    except Exception:
   683→        pass
   684→
   685→    # ── 11. MAKER.ReverseCheck — check keyword bridges ──
   686→    # Check if the instructions mention keywords that exist in the query
   687→    instruction_words = set(re.findall(r'\b\w{4,}\b', enhanced_instructions.lower()))
   688→    query_words = set(re.findall(r'\b\w{4,}\b', query_lower))
   689→    bridge_count = len(instruction_words & query_words)
   690→    if bridge_count < 2:
   691→        enhanced_instructions += " IMPORTANT: Ensure the response directly addresses the customer's specific situation."
   692→
   693→    # ── 12. Escalation rules — based on ticket type ──
   694→    escalation_keywords = ["lawsuit", "sue", "attorney", "lawyer", "legal action",
   695→                          "court", "subpoena", "doj", "ada violation"]
   696→    for kw in escalation_keywords:
   697→        if kw in query_lower:
   698→            enhanced_restrictions += (
   699→                f" ESCALATION RULE: Customer mentioned '{kw}' — "
   700→                f"ALWAYS escalate to human, never auto-resolve."
   701→            )
   702→            break
   703→
   704→    # ── 13. GST — add goal-state tracking ──
   705→    enhanced_instructions += (
   706→        " Track the goal state: identify what the customer wants, "
   707→        "what information is needed, what action is required, and "
   708→        "whether the resolution is complete."
   709→    )
   710→
   711→    # Deduplicate capabilities
   712→    enhanced_caps = list(set(enhanced_caps))
   713→
   714→    logger.info(
   715→        "auto_create non-LLM enhancement: caps=%d restrictions=%d chars instructions=%d chars",
   716→        len(enhanced_caps), len(enhanced_restrictions), len(enhanced_instructions),
   717→    )
   718→
   719→    return (enhanced_caps, enhanced_restrictions, enhanced_instructions, f"Auto: {display}")

   722→def _auto_create_agent_simple(
   723→    tenant_id: str, capability: str,
   724→    query: str = "",
   725→) -> bool:
   726→    """Phase 1 (no-loop): create a minimal agent instantly.
   727→
   728→    Generic instructions, 0 LLM calls, 0 seconds. The real knowledge
   729→    comes from the KB docs (hybrid search in Node 3), not the agent's
   730→    instructions. The agent just opens the gate so the pipeline proceeds.
   731→
   732→    Applies 13 non-LLM techniques via _enhance_agent_with_non_llm_techniques
   733→    to make the agent smarter without any LLM calls.
   734→
   735→    Returns True if agent was created, False on failure.
   736→    """
   737→    try:
   738→        from database.base import SessionLocal
   739→        from database.models.variant_engine import AIAgentAssignment
   740→        import json, uuid as _uuid

   801→async def _auto_create_agent_with_loop(
   802→    tenant_id: str, capability: str, query: str,
   803→) -> bool:
   804→    """Phase 2 (loop): create agent → test → revise → save. Max 3 iterations.
   805→
   806→    Uses GSD (Goal-State Decomposition) and Reverse Thinking to create
   807→    BETTER agent instructions:
   808→
   809→      1. GSD: "What sub-tasks does this customer need?" → ensures the
   810→         agent instructions cover EVERY part of the customer's problem
   811→      2. Generate instructions using the GSD breakdown
   812→      3. Reverse Thinking: "What could go wrong with this agent?" →
   813→         catches risks (overstepping scope, handling legal threats, etc.)
   814→      4. Test: "Can you answer this ticket with these instructions?"
   815→      5. If NO → revise (incorporating the reverse-thinking risks) → loop
   816→
   817→    Falls back to _auto_create_agent_simple if the loop fails entirely.
   818→
   819→    Returns True if agent was created, False on failure.
   820→    """
   821→    from app.core.parwa_pipeline.llm_client import llm_call
   822→
   823→    display = capability.replace("_", " ").title()
   824→    best_instructions = (
   825→        f"Handle {display} tickets using the knowledge base docs. "
   826→        f"Be professional and concise. Cite specific policy sections."
   827→    )
   828→    best_passed = False
   829→    gsd_subtasks = ""
   830→    reverse_risks = ""
   831→
   832→    for iteration in range(1, 4):  # max 3
   833→        try:
   834→            # ── Step 1: GSD — decompose what the customer needs ──────
   835→            # "This refund ticket requires: (a) verify order, (b) check
   836→            #  policy, (c) calculate amount, (d) process refund, (e)
   837→            #  send confirmation"
   838→            if iteration == 1:
   839→                gsd_prompt = (
   840→                    f"A customer asks: {query[:400]}\n\n"
   841→                    f"This is a {display} ticket. Break down ALL the sub-tasks "
   842→                    f"the AI agent needs to handle to fully resolve this. "
   843→                    f"List them as a numbered list. Be specific to this query."
   844→                )
   845→                gsd_result = await llm_call(gsd_prompt, max_tokens=200, temperature=0.2)
   846→                if gsd_result and len(gsd_result.strip()) > 10:
   847→                    gsd_subtasks = gsd_result.strip()
   848→                    logger.info(
   849→                        "auto_create GSD: subtasks for %s: %s",
   850→                        capability, gsd_subtasks[:100],
   851→                    )
   852→
   853→            # ── Step 2: Generate instructions using GSD breakdown ────
   854→            if iteration == 1:
   855→                prompt = (
   856→                    f"Create a system prompt for an AI customer support agent "
   857→                    f"that handles {display} tickets.\n\n"
   858→                    f"The agent must handle these sub-tasks:\n{gsd_subtasks}\n\n"
   859→                    f"Customer query for context: {query[:200]}\n"
   860→                    f"The agent should use knowledge base docs, cite specific "
   861→                    f"policies, and be professional. Address EVERY sub-task. "
   862→                    f"Output ONLY the instructions (3-5 sentences), no explanation."
   863→                )
   864→            else:
   865→                prompt = (
   866→                    f"Revise these AI agent instructions for {display} tickets.\n"
   867→                    f"Previous instructions failed quality check.\n"
   868→                    f"Sub-tasks to cover:\n{gsd_subtasks}\n\n"
   869→                    f"Known risks to prevent:\n{reverse_risks}\n\n"
   870→                    f"Previous instructions: {best_instructions}\n"
   871→                    f"Make the instructions more specific, actionable, and safe. "
   872→                    f"Output ONLY the revised instructions."
   873→                )
   874→
   875→            generated = await llm_call(prompt, max_tokens=300, temperature=0.3)
   876→            if generated and len(generated.strip()) > 20:
   877→                best_instructions = generated.strip()
   878→
   879→            # ── Step 3: Reverse Thinking — what could go wrong? ──────
   880→            # "If I create this agent, what risks exist? The agent might
   881→            #  refund too much, handle legal threats, share competitor
   882→            #  pricing, etc."
   883→            if iteration == 1 or not reverse_risks:
   884→                reverse_prompt = (
   885→                    f"An AI agent will be created with these instructions:\n"
   886→                    f"{best_instructions}\n\n"
   887→                    f"This agent handles {display} tickets for a customer "
   888→                    f"support SaaS. What could go WRONG if this agent handles "
   889→                    f"tickets autonomously? List 2-3 specific risks. "
   890→                    f"Be concise."
   891→                )
   892→                reverse_result = await llm_call(reverse_prompt, max_tokens=150, temperature=0.1)
   893→                if reverse_result and len(reverse_result.strip()) > 10:
   894→                    reverse_risks = reverse_result.strip()
   895→                    logger.info(
   896→                        "auto_create ReverseThinking: risks for %s: %s",
   897→                        capability, reverse_risks[:100],
   898→                    )
   899→
   900→            # ── Step 4: Test — can this agent answer the ticket? ─────
   901→            test_prompt = (
   902→                f"You are an AI agent with these instructions: {best_instructions}\n\n"
   903→                f"A customer asks: {query[:400]}\n\n"
   904→                f"Known risks to watch for:\n{reverse_risks}\n\n"
   905→                f"Based on your instructions, can you provide a confident, "
   906→                f"professional response that addresses ALL sub-tasks? "
   907→                f"Answer YES or NO, then briefly explain why."
   908→            )
   909→            test_result = await llm_call(test_prompt, max_tokens=100, temperature=0.0)
   910→
   911→            if test_result and "YES" in test_result.upper()[:20]:
   912→                best_passed = True
   913→                logger.info(
   914→                    "auto_create_loop: iteration %d PASSED for capability=%s "
   915→                    "(GSD + Reverse Thinking applied)",
   916→                    iteration, capability,
   917→                )
   918→                break
   919→            else:
   920→                logger.info(
   921→                    "auto_create_loop: iteration %d FAILED for capability=%s — "
   922→                    "revising with GSD + Reverse insights",
   923→                    iteration, capability,
   924→                )
   925→
   926→        except Exception as exc:
   927→            logger.warning(
   928→                "auto_create_loop: iteration %d error: %s",
   929→                iteration, str(exc)[:200],
   930→            )
   931→            continue
   932→
   933→    # ── Step 3: Save the agent (best attempt) ──
   934→    # Apply non-LLM techniques to enhance the agent before saving
   935→    enhanced_caps, enhanced_restrictions, enhanced_instructions, agent_name = (
   936→        _enhance_agent_with_non_llm_techniques(
   937→            capability, query, tenant_id,
   938→            [capability],
   939→            "If unsure or lacking verified information, pause for human guidance rather than guessing.",
   940→            best_instructions,
   941→        )
   942→    )
   943→    if enhanced_caps is None:
   944→        # Duplicate detected — agent already exists
   945→        return True
   946→
   947→    try:
   948→        from database.base import SessionLocal
   949→        from database.models.variant_engine import AIAgentAssignment
   950→        import json, uuid as _uuid
   951→
   952→        db = SessionLocal()
   953→        try:
   954→            agent = AIAgentAssignment(
   955→                id=str(_uuid.uuid4()),
   956→                company_id=tenant_id,
   957→                agent_name=agent_name,
   958→                agent_role="auto_created",
   959→                feature_ids="[]",
   960→                task_ids="[]",
   961→                domain="auto",
   962→                capabilities=json.dumps(enhanced_caps),
   963→                instructions=enhanced_instructions[:5000],
   964→                restrictions=enhanced_restrictions[:5000],
   965→                status="active",
   966→            )
   967→            db.add(agent)
   968→            db.commit()
   969→            logger.info(
   970→                "auto_create_loop: saved agent for capability=%s passed=%s iterations=%d caps=%d (non-LLM enhanced)",
   971→                capability, best_passed, iteration, len(enhanced_caps),
   972→            )
   973→            return True
   974→        finally:
   975→            db.close()
   976→    except Exception as exc:
   977→        logger.warning("auto_create_loop: save failed: %s", str(exc)[:200])
   978→        # Fall back to simple creation
   979→        return _auto_create_agent_simple(tenant_id, capability, query)

  1031→def _classify_ticket_type(query: str) -> tuple:
  1032→    """Rule-based ticket type classification using pattern matching.
  1033→    Returns (ticket_type, matched_keywords)."""
  1034→    query_lower = query.lower()
  1035→    scores: Dict[str, int] = {}
  1036→
  1037→    for ttype, patterns in TICKET_PATTERNS.items():
  1038→        count = 0
  1039→        matched = []
  1040→        for pat in patterns:
  1041→            if re.search(pat, query_lower):
  1042→                count += 1
  1043→                matched.append(pat)
  1044→        if count > 0:
  1045→            scores[ttype] = count
  1046→
  1047→    if not scores:
  1048→        return "general", []
  1049→
  1050→    best_type = max(scores, key=scores.get)  # type: ignore[arg-type]
  1051→    return best_type, []

  1054→def _classify_complexity(query: str, ticket_type: str) -> str:
  1055→    """Rule-based complexity classification.
  1056→    Phase 7: Added multi-issue detection for dual-problem tickets."""
  1057→    query_lower = query.lower()
  1058→
  1059→    # Phase 7: Multi-issue detection — if 2+ signals, it's complex at minimum
  1060→    multi_signals = sum(1 for pat in MULTI_ISSUE_SIGNALS if re.search(pat, query_lower, re.DOTALL))
  1061→    if multi_signals >= 2:
  1062→        return "complex"
  1063→    if multi_signals == 1:
  1064→        return "medium"
  1065→
  1066→    # Check for hard complexity indicators
  1067→    hard_count = sum(1 for kw in COMPLEXITY_KEYWORDS_HARD if kw in query_lower)
  1068→    if hard_count >= 2:
  1069→        return "hard"
  1070→    if hard_count == 1:
  1071→        return "complex"
  1072→
  1073→    # Check for medium complexity indicators
  1074→    medium_count = sum(1 for kw in COMPLEXITY_KEYWORDS_MEDIUM if kw in query_lower)
  1075→    if medium_count >= 2:
  1076→        return "medium"
  1077→
  1078→    # Certain ticket types default to higher complexity
  1079→    if ticket_type in ("complaint", "account_change"):
  1080→        return "medium"
  1081→
  1082→    return "simple"

  1085→def _extract_action(query: str, ticket_type: str = "") -> tuple:
  1086→    """Extract required action and details from query.
  1087→    Returns (action, details_dict).
  1088→    Phase 7: Added investigate_billing for pricing disputes; prioritizes
  1089→    investigation patterns over plan_change when the user is questioning
  1090→    charges rather than requesting changes."""
  1091→    # First pass: find ALL matching actions with their positions
  1092→    matches = []
  1093→    for pattern, action, detail_key in ACTION_PATTERNS:
  1094→        match = re.search(pattern, query, re.IGNORECASE)
  1095→        if match:
  1096→            details = {}
  1097→            if detail_key and match.lastindex and match.lastindex >= 1 and match.group(1):
  1098→                details[detail_key] = float(match.group(1))
  1099→            matches.append((match.start(), action, details))
  1100→
  1101→    if not matches:
  1102→        return "provide_info", {}
  1103→
  1104→    # If multiple actions match, pick the one that appears first in the query
  1105→    # (the primary intent is usually stated first)
  1106→    matches.sort(key=lambda x: x[0])
  1107→
  1108→    # Phase 7: If we matched both plan_change AND investigate_billing,
  1109→    # prefer investigate_billing — the user is questioning, not requesting.
  1110→    actions_found = [m[1] for m in matches]
  1111→    if "investigate_billing" in actions_found:
  1112→        idx = actions_found.index("investigate_billing")
  1113→        return matches[idx][1], matches[idx][2]
  1114→
  1115→    # Otherwise return the first match
  1116→    return matches[0][1], matches[0][2]

  1146→def _meta_learner_predict(
  1147→    tenant_id: str, ticket_type: str, complexity: str, query: str = ""
  1148→) -> Dict[str, Any]:
  1149→    """Phase 6: Predict routing based on Wiki Section A past patterns.
  1150→    
  1151→    Searches the AI Wiki for similar ticket patterns and uses
  1152→    their historical outcomes to guide routing.
  1153→    Non-LLM — keyword search only.
  1154→    """
  1155→    from app.core.parwa_pipeline.ai_wiki_store import get_wiki_store
  1156→    
  1157→    wiki = get_wiki_store()
  1158→    
  1159→    try:
  1160→        patterns = wiki.find_similar_patterns(
  1161→            tenant_id=tenant_id, query=query,
  1162→            ticket_type=ticket_type, max_results=3,
  1163→        )
  1164→        
  1165→        if not patterns:
  1166→            return {
  1167→                "similar_tickets_found": 0,
  1168→                "historical_accuracy": 0.0,
  1169→                "suggested_path": None,
  1170→                "wiki_boosted": False,
  1171→            }
  1172→        
  1173→        # Analyze patterns for routing guidance
  1174→        total_quality = sum(p["quality_achieved"] for p in patterns)
  1175→        avg_quality = total_quality / len(patterns)
  1176→        
  1177→        # Check if similar tickets were successfully resolved
  1178→        successful = sum(1 for p in patterns if p["quality_achieved"] >= 0.90)
  1179→        success_rate = successful / len(patterns)
  1180→        
  1181→        # Extract most common techniques that worked
  1182→        all_techniques = []
  1183→        for p in patterns:
  1184→            all_techniques.extend(p.get("techniques_that_worked", []))
  1185→        technique_counts = {}
  1186→        for t in all_techniques:
  1187→            technique_counts[t] = technique_counts.get(t, 0) + 1
  1188→        top_techniques = sorted(technique_counts, key=technique_counts.get, reverse=True)[:5]
  1189→        
  1190→        # Suggest path based on historical success
  1191→        suggested_path = None
  1192→        if success_rate >= 0.7 and avg_quality >= 0.90:
  1193→            # Similar tickets were resolved well — suggest same approach
  1194→            suggested_path = "complex_path" if complexity in ("complex", "hard") else "simple_medium_path"
  1195→        elif success_rate < 0.3:
  1196→            # Similar tickets struggled — suggest complex path for more thorough reasoning
  1197→            suggested_path = "complex_path"
  1198→        
  1199→        return {
  1200→            "similar_tickets_found": len(patterns),
  1201→            "historical_accuracy": round(success_rate, 3),
  1202→            "suggested_path": suggested_path,
  1203→            "wiki_boosted": True,
  1204→            "avg_historical_quality": round(avg_quality, 4),
  1205→            "top_techniques": top_techniques,
  1206→            "pattern_entry_keys": [p["entry_key"] for p in patterns],
  1207→        }
  1208→    except Exception as e:
  1209→        return {
  1210→            "similar_tickets_found": 0,
  1211→            "historical_accuracy": 0.0,
  1212→            "suggested_path": None,
  1213→            "wiki_boosted": False,
  1214→        }

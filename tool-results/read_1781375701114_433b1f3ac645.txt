   151→   151→function formatNumber(n: number): string {
   152→   152→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`;
   153→   153→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
   154→   154→  return n.toLocaleString();
   155→   155→}
   156→   156→
   157→   157→function formatHours(hours: number): string {
   158→   158→  if (hours < 1) return `${Math.round(hours * 60)}m`;
   159→   159→  return `${hours.toFixed(1)}h`;
   160→   160→}
   161→   161→
   162→   162→function formatPercent(n: number): string {
   163→   163→  return `${n.toFixed(1)}%`;
   164→   164→}
   165→   165→
   166→   166→function formatCurrency(n: number): string {
   167→   167→  if (n >= 1) return `$${n.toFixed(2)}`;
   168→   168→  return `$${n.toFixed(4)}`;
   169→   169→}

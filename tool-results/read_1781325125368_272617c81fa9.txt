     1→/**
     2→ * PARWA Unified Integration Catalog
     3→ *
     4→ * SINGLE SOURCE OF TRUTH for all integration metadata.
     5→ * Used by: Onboarding IntegrationStep, Settings Integrations tab, BFF routes.
     6→ *
     7→ * Per D2: All variants get UNLIMITED integrations.
     8→ * Per D4: Three integration tiers — Pre-built, OpenAPI Import, Custom REST.
     9→ * Per D6: Each integration has a pre-written HTTP test call (NO AI).
    10→ * Per GAP 2: Universal API key system with 5 auth types.
    11→ * Per GAP 3: Full catalog per industry (suggestions, not restrictions).
    12→ */
    13→
    14→// ── Auth Types (GAP 2) ──────────────────────────────────────────────────
    15→
    16→export type AuthType = 'bearer' | 'api_key_header' | 'api_key_query' | 'basic_auth' | 'oauth2';
    17→
    18→export interface AuthField {
    19→  name: string;
    20→  label: string;
    21→  type: 'text' | 'password' | 'url';
    22→  required: boolean;
    23→  placeholder?: string;
    24→}
    25→
    26→export interface AuthSchema {
    27→  type: AuthType;
    28→  fields: AuthField[];
    29→  /** Header name for api_key_header auth */
    30→  headerName?: string;
    31→  /** Query param name for api_key_query auth */
    32→  queryParamName?: string;
    33→  /** OAuth2 redirect URI */
    34→  redirectUri?: string;
    35→}
    36→
    37→// ── Integration Category ────────────────────────────────────────────────
    38→
    39→export type IntegrationCategory =
    40→  | 'crm'
    41→  | 'ecommerce'
    42→  | 'helpdesk'
    43→  | 'communication'
    44→  | 'analytics'
    45→  | 'marketing'
    46→  | 'payments'
    47→  | 'shipping'
    48→  | 'dev_tools'
    49→  | 'productivity'
    50→  | 'custom';
    51→
    52→// ── Integration Tier (D4) ──────────────────────────────────────────────
    53→
    54→export type IntegrationTier = 'tier1_prebuilt' | 'tier2_openapi' | 'tier3_custom';
    55→
    56→// ── Industry (D1 — 4 industries only) ──────────────────────────────────
    57→
    58→export type ParwaIndustry = 'saas' | 'ecommerce' | 'logistics' | 'other';
    59→
    60→// ── Integration Definition ─────────────────────────────────────────────
    61→
    62→export interface IntegrationDefinition {
    63→  /** Machine key — also used as integration_type in DB */
    64→  key: string;
    65→  /** Display name */
    66→  name: string;
    67→  /** Short description for UI */
    68→  description: string;
    69→  /** Category for grouping in UI */
    70→  category: IntegrationCategory;
    71→  /** Which tier this belongs to */
    72→  tier: IntegrationTier;
    73→  /** Auth schema — defines the credential form (GAP 2) */
    74→  authSchema: AuthSchema;
    75→  /** Pre-written test call (D6 — NO AI) */
    76→  testConnection: {
    77→    method: 'GET' | 'POST';
    78→    /** URL template — {field_name} replaced with credential values */
    79→    urlTemplate: string;
    80→    /** Header template — {field_name} replaced with credential values */
    81→    headersTemplate?: Record<string, string>;
    82→    /** How to interpret the response */
    83→    successCheck: 'status_200' | 'json_ok_true' | 'status_200_or_201';
    84→    /** What to show on success — {response_json_path} for dynamic values */
    85→    successMessage?: string;
    86→  };
    87→  /** Industries where this integration is SUGGESTED (not restricted to) */
    88→  suggestedIndustries: ParwaIndustry[];
    89→  /** Which variants can use this (empty = all) */
    90→  availableForVariants?: ('mini_parwa' | 'parwa' | 'parwa_high')[];
    91→  /** Icon identifier for frontend rendering */
    92→  iconId: string;
    93→  /** Icon color gradient for frontend */
    94→  colorGradient: string;
    95→  /** Whether this integration is available */
    96→  available: boolean;
    97→}

   906→// ── Industry Filtering (GAP 3) ──────────────────────────────────────────
   907→
   908→/**
   909→ * Get integrations suggested for a specific industry.
   910→ * Per D1/D3: "Other" shows ALL integrations (no filtering).
   911→ * Per D3: Suggestions are NOT restrictions — clients can always connect outside their industry.
   912→ */
   913→export function getIntegrationsForIndustry(
   914→  industry: ParwaIndustry
   915→): IntegrationDefinition[] {
   916→  if (industry === 'other') {
   917→    return INTEGRATION_CATALOG.filter((i) => i.available);
   918→  }
   919→  return INTEGRATION_CATALOG.filter(
   920→    (i) => i.available && i.suggestedIndustries.includes(industry)
   921→  );
   922→}
   923→
   924→/**
   925→ * Get integrations grouped by category for a specific industry.
   926→ */
   927→export function getIntegrationsGroupedByCategory(
   928→  industry: ParwaIndustry
   929→): Record<IntegrationCategory, IntegrationDefinition[]> {
   930→  const filtered = getIntegrationsForIndustry(industry);
   931→  const grouped: Record<string, IntegrationDefinition[]> = {};
   932→
   933→  for (const integration of filtered) {
   934→    if (!grouped[integration.category]) {
   935→      grouped[integration.category] = [];
   936→    }
   937→    grouped[integration.category].push(integration);
   938→  }
   939→
   940→  return grouped as Record<IntegrationCategory, IntegrationDefinition[]>;
   941→}
   942→
   943→/**
   944→ * Get a single integration by key.
   945→ */
   946→export function getIntegrationByKey(key: string): IntegrationDefinition | undefined {
   947→  return INTEGRATION_CATALOG.find((i) => i.key === key);
   948→}

   950→/**
   951→ * Check if an integration is available for a specific variant.
   952→ * Per D2: ALL variants get unlimited integrations.
   953→ * availableForVariants is ONLY for tier-gating (Custom API, OpenAPI).
   954→ */
   955→export function isIntegrationAvailableForVariant(
   956→  integration: IntegrationDefinition,
   957→  _variant: 'mini_parwa' | 'parwa' | 'parwa_high'
   958→): boolean {
   959→  // Per D2: All variants get UNLIMITED integrations.
   960→  // availableForVariants is only used for feature gating, not count limits.
   961→  if (!integration.availableForVariants || integration.availableForVariants.length === 0) {
   962→    return true;
   963→  }
   964→  return integration.availableForVariants.includes(_variant);
   965→}

   148→ * @deprecated Variant assignment now happens server-side in Node 2 of the

   152→export function autoAssignVariant(
   153→  priority: TicketPriority,
   154→  category: TicketCategory
   155→): TicketVariant {
   156→  if (priority === 'critical') return 'heavy';
   157→  if (priority === 'high') return 'medium';
   158→  if (
   159→    category === 'fraud_security' ||
   160→    category === 'vip_enterprise' ||
   161→    category === 'complaints'
   162→  ) {
   163→    return 'medium';
   164→  }
   165→  return 'light';
   166→}

   168→export function computeStats(tickets: Ticket[]): TicketStats {
   169→  const byStatus = {} as Record<TicketStatus, number>;
   170→  const byCategory = {} as Record<TicketCategory, number>;
   171→  const byPriority = {} as Record<TicketPriority, number>;
   172→  const byVariant = {} as Record<TicketVariant, number>;
   173→  const byChannel = {} as Record<TicketChannel, number>;
   174→
   175→  ALL_STATUSES.forEach((s) => (byStatus[s] = 0));
   176→  ALL_CATEGORIES.forEach((c) => (byCategory[c] = 0));
   177→  ALL_PRIORITIES.forEach((p) => (byPriority[p] = 0));
   178→  ALL_VARIANTS.forEach((v) => (byVariant[v] = 0));
   179→  ALL_CHANNELS.forEach((ch) => (byChannel[ch] = 0));
   180→
   181→  let resolved = 0;
   182→  let totalResolutionTime = 0;
   183→  let resolutionCount = 0;
   184→  let totalCost = 0;
   185→  let totalSavings = 0;
   186→
   187→  for (const t of tickets) {
   188→    byStatus[t.status]++;
   189→    byCategory[t.category]++;
   190→    byPriority[t.priority]++;
   191→    byChannel[t.channel]++;
   192→    if (t.assigned_variant) byVariant[t.assigned_variant]++;
   193→    if (t.status === 'resolved' || t.status === 'closed') {
   194→      resolved++;
   195→      if (t.resolution_time_hours !== null) {
   196→        totalResolutionTime += t.resolution_time_hours;
   197→        resolutionCount++;
   198→      }
   199→    }
   200→    if (t.cost_per_ticket !== null) totalCost += t.cost_per_ticket;
   201→    if (t.savings_per_ticket !== null) totalSavings += t.savings_per_ticket;
   202→  }
   203→
   204→  const nonTerminal = tickets.filter(
   205→    (t) => t.status !== 'resolved' && t.status !== 'closed'
   206→  );
   207→  const resolutionRate =
   208→    tickets.length > 0
   209→      ? Math.round((resolved / tickets.length) * 1000) / 10
   210→      : 0;
   211→
   212→  return {
   213→    total: tickets.length,
   214→    byStatus,
   215→    byCategory,
   216→    byPriority,
   217→    byVariant,
   218→    byChannel,
   219→    resolved,
   220→    resolutionRate,
   221→    avgResolutionTime:
   222→      resolutionCount > 0 ? Math.round(totalResolutionTime / resolutionCount * 10) / 10 : null,
   223→    totalCost: Math.round(totalCost * 1000) / 1000,
   224→    totalSavings: Math.round(totalSavings * 100) / 100,
   225→  };
   226→}

   260→const STORAGE_KEY = 'parwa_tickets';
   261→const INIT_KEY = 'parwa_tickets_initialized';
   262→
   263→export const useTicketStore = create<TicketState>((set, get) => ({
   264→  tickets: [],
   265→  initialized: false,
   266→
   267→  init: () => {
   268→    if (get().initialized) return;
   269→    if (typeof window === 'undefined') return;
   270→
   271→    // ── KI-2 FIX (Week 3): Fetch from backend first, fall back to localStorage ──
   272→    // Previously this ONLY read from localStorage, meaning tickets were always
   273→    // local fake data. Now we kick off an async backend fetch; meanwhile we
   274→    // synchronously hydrate from localStorage (if any) so the UI doesn't flash
   275→    // empty. The async fetch will replace the data once it returns.
   276→    try {
   277→      const existing = localStorage.getItem(STORAGE_KEY);
   278→      if (existing) {
   279→        const parsed = JSON.parse(existing) as Ticket[];
   280→        // Restore ticket counter to avoid collisions
   281→        if (parsed.length > 0) {
   282→          const maxNum = Math.max(
   283→            ...parsed.map((t) =>
   284→              parseInt(t.ticket_number.replace('TKT-', ''), 10)
   285→            )
   286→          );
   287→          ticketCounter = maxNum + 1;
   288→        }
   289→        set({ tickets: parsed, initialized: true });
   290→      } else {
   291→        // No local data — mark initialized so loading state resolves
   292→        set({ initialized: true });
   293→      }
   294→    } catch {
   295→      set({ initialized: true });
   296→    }
   297→
   298→    // Fire-and-forget backend sync (non-blocking)
   299→    void syncFromBackend();
   300→  },
   301→
   302→  addTicket: (data) => {
   303→    // Backend is the source of truth — we NO LONGER do client-side variant
   304→    // assignment, hardcoded ai_confidence, hardcoded cost_per_ticket, or
   305→    // hardcoded savings_per_ticket. The 8-node pipeline runs on the
   306→    // backend (Node 2 smart-routes the variant, Node 4 generates the
   307→    // response, Node 6 quality-checks, Node 7 persists cost/savings) and
   308→    // the next syncFromBackend() will populate all these fields from
   309→    // real data.
   310→    //
   311→    // Until that sync completes, we show optimistic placeholders (null /
   312→    // 'open' / no variant) so the user sees the ticket they just created.
   313→    const now = new Date().toISOString();
   314→    const optimisticId = uuid();
   315→    const ticket: Ticket = {
   316→      id: optimisticId,
   317→      ticket_number: `TKT-PENDING`,
   318→      subject: data.subject,
   319→      description: data.description,
   320→      category: data.category,
   321→      priority: data.priority,
   322→      status: 'open',
   323→      channel: data.channel,
   324→      customer_name: data.customer_name,
   325→      customer_email: data.customer_email,
   326→      assigned_variant: null,
   327→      assigned_agent: null,
   328→      created_at: now,
   329→      updated_at: now,
   330→      resolved_at: null,
   331→      first_response_at: null,
   332→      resolution_time_hours: null,
   333→      ai_confidence: null,
   334→      cost_per_ticket: null,
   335→      savings_per_ticket: null,
   336→      messages: [],
   337→      tags: [],
   338→    };
   339→    set((s) => ({ tickets: [ticket, ...s.tickets] }));
   340→
   341→    // Push to backend. The backend will:
   342→    //   1. Auto-create a Customer (if customer_email is new) or match existing
   343→    //   2. Store description as first TicketMessage (role=customer)
   344→    //   3. Trigger the 8-node pipeline (sync for critical, async for others)
   345→    //   4. Store AI response as TicketMessage (role=ai)
   346→    //   5. Update ticket with status / variant / ai_confidence / cost
   347→    // On success, we patch the optimistic ticket with the REAL backend ID
   348→    // (so subsequent escalate/resolve calls use the correct ID) and re-sync
   349→    // from backend to fetch the full real ticket (including AI response).
   350→    void pushToBackendWithBody<{ id?: string }>('POST', '/api/v1/tickets', {
   351→      subject: ticket.subject,
   352→      description: ticket.description,
   353→      category: ticket.category,
   354→      priority: ticket.priority,
   355→      channel: ticket.channel,
   356→      customer_name: ticket.customer_name,
   357→      customer_email: ticket.customer_email,
   358→      // CRM-universal optional fields — only send when provided so the
   359→      // backend's validators don't reject empty strings.
   360→      ...(data.customer_id ? { customer_id: data.customer_id } : {}),
   361→      ...(data.customer_phone ? { customer_phone: data.customer_phone } : {}),
   362→      ...(data.tags && data.tags.length > 0 ? { tags: data.tags } : {}),
   363→      ...(data.metadata_json && Object.keys(data.metadata_json).length > 0
   364→        ? { metadata_json: data.metadata_json }
   365→        : {}),
   366→    }).then((responseBody) => {
   367→      if (!responseBody) {
   368→        // Ticket creation failed on the backend (likely trial expired or
   369→        // trial ticket limit hit). Remove the optimistic ghost ticket so
   370→        // the user doesn't see a ticket that doesn't actually exist.
   371→        set((s) => ({ tickets: s.tickets.filter((t) => t.id !== optimisticId) }));
   372→        return;
   373→      }
   374→      // Patch the optimistic ticket in-place with the real backend ID
   375→      // so subsequent calls (escalate, resolve, update) use the persisted ID.
   376→      const realId = responseBody.id;
   377→      if (realId && realId !== optimisticId) {
   378→        set((s) => ({
   379→          tickets: s.tickets.map((t) =>
   380→            t.id === optimisticId
   381→              ? { ...t, id: realId, ticket_number: t.ticket_number === 'TKT-PENDING' ? 'TKT-…' : t.ticket_number }
   382→              : t
   383→          ),
   384→        }));
   385→      }
   386→      void syncFromBackend();
   387→    });
   388→    return ticket;
   389→  },
   390→
   391→  updateTicketStatus: (id, status) => {
   392→    set((s) => ({
   393→      tickets: s.tickets.map((t) =>
   394→        t.id === id ? { ...t, status, updated_at: new Date().toISOString() } : t
   395→      ),
   396→    }));
   397→    void pushToBackend('PATCH', `/api/v1/tickets/${id}/status`, { status });
   398→  },
   399→
   400→  assignVariant: (id, variant) => {
   401→    // Send the variant update to the backend. The backend's pipeline
   402→    // will re-run with the new variant and update ai_confidence /
   403→    // cost / savings. We do NOT compute these client-side anymore.
   404→    set((s) => ({
   405→      tickets: s.tickets.map((t) =>
   406→        t.id === id
   407→          ? {
   408→              ...t,
   409→              assigned_variant: variant,
   410→              updated_at: new Date().toISOString(),
   411→            }
   412→          : t
   413→      ),
   414→    }));
   415→    void pushToBackend('PUT', `/api/v1/tickets/${id}`, { assigned_variant: variant })
   416→      .then((ok) => { if (ok) void syncFromBackend(); });
   417→  },
   418→
   419→  resolveTicket: (id, resolution) => {
   420→    const now = new Date().toISOString();
   421→    set((s) => ({
   422→      tickets: s.tickets.map((t) => {
   423→        if (t.id !== id) return t;
   424→        const created = new Date(t.created_at).getTime();
   425→        const resolved = new Date(now).getTime();
   426→        const hours = Math.round(((resolved - created) / 3600000) * 10) / 10;
   427→        const msgs: TicketMessage[] = [
   428→          ...t.messages,
   429→          {
   430→            id: uuid(),
   431→            ticket_id: id,
   432→            sender: 'ai_agent',
   433→            sender_name: 'PARWA AI',
   434→            content: resolution
   435→              ? `Issue resolved: ${resolution}`
   436→              : 'This ticket has been marked as resolved. Please let us know if you need further assistance.',
   437→            created_at: now,
   438→            variant: t.assigned_variant || undefined,
   439→          },
   440→        ];
   441→        return {
   442→          ...t,
   443→          status: 'resolved' as TicketStatus,
   444→          resolved_at: now,
   445→          updated_at: now,
   446→          resolution_time_hours: hours,
   447→          messages: msgs,
   448→        };
   449→      }),
   450→    }));
   451→    void pushToBackend('PATCH', `/api/v1/tickets/${id}/status`, { status: 'resolved', resolution });
   452→  },
   453→
   454→  escalateToHuman: (id) => {
   455→    const now = new Date().toISOString();
   456→    set((s) => ({
   457→      tickets: s.tickets.map((t) => {
   458→        if (t.id !== id) return t;
   459→        const msgs: TicketMessage[] = [
   460→          ...t.messages,
   461→          {
   462→            id: uuid(),
   463→            ticket_id: id,
   464→            sender: 'system',
   465→            sender_name: 'System',
   466→            content:
   467→              'Ticket has been escalated to a human agent. A team member will be with you shortly.',
   468→            created_at: now,
   469→          },
   470→        ];
   471→        return {
   472→          ...t,
   473→          status: 'awaiting_human' as TicketStatus,
   474→          assigned_agent: 'Human Agent',
   475→          updated_at: now,
   476→          messages: msgs,
   477→        };
   478→      }),
   479→    }));
   480→    void pushToBackend('POST', `/api/v1/tickets/${id}/escalate`, {});
   481→  },
   482→
   483→  resumeWithGuidance: async (id, guidance) => {
   484→    // Approach A: Resume a paused pipeline with human/variant guidance.
   485→    // Calls POST /api/v1/tickets/{id}/resume which loads the LangGraph
   486→    // checkpoint and resumes from the exact node that paused.
   487→    try {
   488→      const res = await fetch(`/api/v1/tickets/${id}/resume`, {
   489→        method: 'POST',
   490→        headers: { 'Content-Type': 'application/json' },
   491→        credentials: 'include',
   492→        body: JSON.stringify({ guidance }),
   493→      });
   494→      if (res.ok) {
   495→        const data = await res.json();
   496→        // Update ticket status based on resume result
   497→        set((s) => ({
   498→          tickets: s.tickets.map((t) =>
   499→            t.id === id
   500→              ? {
   501→                  ...t,
   502→                  status: (data.status === 'resolved' ? 'resolved' : 'awaiting_human') as TicketStatus,
   503→                  updated_at: new Date().toISOString(),
   504→                }
   505→              : t
   506→          ),
   507→        }));
   508→        // Sync from backend to get the latest AI response
   509→        void syncFromBackend();
   510→        return true;
   511→      }
   512→    } catch {
   513→      // Network error
   514→    }
   515→    return false;
   516→  },
   517→
   518→  updatePriority: (id, priority) => {
   519→    // Update priority on backend. Variant assignment happens server-side
   520→    // via Node 2 smart-routing — we don't auto-assign a variant client-side.
   521→    set((s) => ({
   522→      tickets: s.tickets.map((t) =>
   523→        t.id === id
   524→          ? {
   525→              ...t,
   526→              priority,
   527→              updated_at: new Date().toISOString(),
   528→            }
   529→          : t
   530→      ),
   531→    }));
   532→    void pushToBackend('PUT', `/api/v1/tickets/${id}`, { priority })
   533→      .then((ok) => { if (ok) void syncFromBackend(); });
   534→  },
   535→
   536→  addMessage: (ticketId, message) => {
   537→    const msg: TicketMessage = {
   538→      ...message,
   539→      id: uuid(),
   540→      ticket_id: ticketId,
   541→      created_at: new Date().toISOString(),
   542→    };
   543→    set((s) => ({
   544→      tickets: s.tickets.map((t) =>
   545→        t.id === ticketId
   546→          ? {
   547→              ...t,
   548→              messages: [...t.messages, msg],
   549→              first_response_at:
   550→                t.first_response_at ?? msg.created_at,
   551→              updated_at: new Date().toISOString(),
   552→              status:
   553→                t.status === 'open' ? ('in_progress' as TicketStatus) : t.status,
   554→            }
   555→          : t
   556→      ),
   557→    }));
   558→    void pushToBackend('POST', `/api/v1/tickets/${ticketId}/messages`, {
   559→      // Backend MessageCreate schema expects: role, content, channel
   560→      // Map frontend sender → backend role (human_agent → agent, ai_agent → ai)
   561→      role: msg.sender === 'human_agent' ? 'agent' : msg.sender === 'ai_agent' ? 'ai' : msg.sender,
   562→      content: msg.content,
   563→      channel: 'email',

   583→// ── KI-2 FIX (Week 3): Backend sync ──────────────────────────────────────────
   584→// Fetches real tickets from the backend via the BFF proxy at /api/tickets.
   585→// Maps the backend TicketResponse shape → frontend Ticket shape.
   586→// On success, replaces the store + persists to localStorage (so the next page
   587→// load is instant from cache, then re-synced). On failure, leaves existing
   588→// data intact (graceful degradation).
   589→//
   590→// Backend response shape (see backend/app/schemas/ticket.py):
   591→//   { items: TicketResponse[], total: number, page: number, page_size: number }
   592→//
   593→// TicketResponse has snake_case fields (ticket_number, customer_name, etc.);
   594→// the frontend Ticket interface also uses snake_case, so most fields pass
   595→// through directly. We only need to normalize a few fields that the backend
   596→// may omit (messages, tags) or name differently.
   597→
   598→interface BackendTicketResponse {
   599→  id: string;
   600→  ticket_number: string;
   601→  subject: string;
   602→  description?: string;
   603→  category: string;
   604→  priority: string;
   605→  status: string;
   606→  channel: string;
   607→  customer_name: string;
   608→  customer_email?: string;
   609→  assigned_variant?: string | null;
   610→  assigned_agent?: string | null;
   611→  created_at: string;
   612→  updated_at: string;
   613→  resolved_at?: string | null;
   614→  first_response_at?: string | null;
   615→  resolution_time_hours?: number | null;
   616→  ai_confidence?: number | null;
   617→  cost_per_ticket?: number | null;
   618→  savings_per_ticket?: number | null;
   619→  tags?: string[];
   620→}
   621→
   622→function normalizeBackendTicket(raw: BackendTicketResponse): Ticket {
   623→  // Validate enum fields; fall back to safe defaults if backend sends unknown values
   624→  const validCategories: TicketCategory[] = ALL_CATEGORIES;
   625→  const validPriorities: TicketPriority[] = ALL_PRIORITIES;
   626→  const validStatuses: TicketStatus[] = ALL_STATUSES;
   627→  const validChannels: TicketChannel[] = ALL_CHANNELS;
   628→  const validVariants: TicketVariant[] = ALL_VARIANTS;
   629→
   630→  const category = (validCategories as string[]).includes(raw.category)
   631→    ? (raw.category as TicketCategory)
   632→    : 'technical_support';
   633→  const priority = (validPriorities as string[]).includes(raw.priority)
   634→    ? (raw.priority as TicketPriority)
   635→    : 'medium';
   636→  const status = (validStatuses as string[]).includes(raw.status)
   637→    ? (raw.status as TicketStatus)
   638→    : 'open';
   639→  const channel = (validChannels as string[]).includes(raw.channel)
   640→    ? (raw.channel as TicketChannel)
   641→    : 'email';
   642→
   643→  // Map backend variant names → frontend variant names
   644→  // Backend uses 'mini' / 'parwa' / 'high'; frontend uses 'light' / 'medium' / 'heavy'
   645→  let assigned_variant: TicketVariant | null = null;
   646→  if (raw.assigned_variant) {
   647→    const v = raw.assigned_variant.toLowerCase();
   648→    if (v === 'mini' || v === 'light') assigned_variant = 'light';
   649→    else if (v === 'parwa' || v === 'medium') assigned_variant = 'medium';
   650→    else if (v === 'high' || v === 'heavy') assigned_variant = 'heavy';
   651→    else if ((validVariants as string[]).includes(raw.assigned_variant)) {
   652→      assigned_variant = raw.assigned_variant as TicketVariant;
   653→    }
   654→  }
   655→
   656→  return {
   657→    id: raw.id,
   658→    ticket_number: raw.ticket_number,
   659→    subject: raw.subject,
   660→    description: raw.description ?? '',
   661→    category,
   662→    priority,
   663→    status,
   664→    channel,
   665→    customer_name: raw.customer_name,
   666→    customer_email: raw.customer_email ?? '',
   667→    assigned_variant,
   668→    assigned_agent: raw.assigned_agent ?? null,
   669→    created_at: raw.created_at,
   670→    updated_at: raw.updated_at,
   671→    resolved_at: raw.resolved_at ?? null,
   672→    first_response_at: raw.first_response_at ?? null,
   673→    resolution_time_hours: raw.resolution_time_hours ?? null,
   674→    ai_confidence: raw.ai_confidence ?? null,
   675→    cost_per_ticket: raw.cost_per_ticket ?? null,
   676→    savings_per_ticket: raw.savings_per_ticket ?? null,
   677→    messages: [],  // Backend doesn't return messages in list view; fetch on detail view
   678→    tags: raw.tags ?? [],
   679→  };
   680→}

   682→export async function syncFromBackend(): Promise<void> {
   683→  if (typeof window === 'undefined') return;
   684→  try {
   685→    const res = await fetch('/api/tickets?page=1&page_size=100', {
   686→      method: 'GET',
   687→      headers: { 'Content-Type': 'application/json' },
   688→      credentials: 'include',
   689→    });
   690→
   691→    if (!res.ok) {
   692→      // 401 = not authenticated; 5xx = backend down. Either way, leave existing data.
   693→      return;
   694→    }
   695→
   696→    const data = await res.json();
   697→    const items: BackendTicketResponse[] = Array.isArray(data?.items) ? data.items : [];
   698→
   699→    const normalized: Ticket[] = items.map(normalizeBackendTicket);
   700→
   701→    // Update store
   702→    useTicketStore.setState({ tickets: normalized, initialized: true });
   703→
   704→    // Persist to localStorage so next page load is instant from cache
   705→    localStorage.setItem(STORAGE_KEY, JSON.stringify(normalized));
   706→    localStorage.setItem(INIT_KEY, 'true');
   707→
   708→    // Restore ticket counter
   709→    if (normalized.length > 0) {
   710→      const maxNum = Math.max(
   711→        ...normalized.map((t) =>
   712→          parseInt(t.ticket_number.replace('TKT-', ''), 10) || 0
   713→        )
   714→      );
   715→      if (!Number.isNaN(maxNum)) ticketCounter = maxNum + 1;
   716→    }
   717→  } catch (err) {
   718→    // Network error / JSON parse error — leave existing data intact
   719→    console.warn('[ticket-store] syncFromBackend failed:', err);
   720→  }
   721→}

   723→/**
   724→ * Fetch messages for a single ticket from /api/v1/tickets/{id}/messages
   725→ * and merge them into the store. Used by the ticket detail panel so the
   726→ * user sees the AI/customer conversation thread instead of "No messages yet".
   727→ *
   728→ * The list endpoint does not return messages, so we fetch them on demand
   729→ * when the user opens a ticket.
   730→ */
   731→export async function fetchTicketMessages(ticketId: string): Promise<void> {
   732→  if (typeof window === 'undefined') return;
   733→  try {
   734→    const res = await fetch(`/api/v1/tickets/${ticketId}/messages`, {
   735→      method: 'GET',
   736→      headers: { 'Content-Type': 'application/json' },
   737→      credentials: 'include',
   738→    });
   739→    if (!res.ok) return;
   740→    const data = await res.json();
   741→    const rawMsgs: Array<Record<string, unknown>> =
   742→      Array.isArray(data?.messages) ? data.messages : (Array.isArray(data) ? data : []);
   743→    const msgs: TicketMessage[] = rawMsgs.map((m) => {
   744→      const role = String(m.role ?? m.sender ?? 'system');
   745→      // Backend uses role=ai|customer|human_agent|system; map to MessageSender
   746→      const sender = (
   747→        role === 'ai' ? 'ai_agent' :
   748→        role === 'customer' ? 'customer' :
   749→        role === 'human_agent' ? 'human_agent' :
   750→        'system'
   751→      ) as MessageSender;
   752→      return {
   753→        id: String(m.id ?? ''),
   754→        ticket_id: String(m.ticket_id ?? ticketId),
   755→        sender,
   756→        sender_name: String(m.sender_name ?? (sender === 'ai_agent' ? 'AI' : sender === 'customer' ? 'Customer' : 'Agent')),
   757→        content: String(m.content ?? ''),
   758→        created_at: String(m.created_at ?? new Date().toISOString()),
   759→        variant: (m.variant_version as TicketVariant | undefined) ?? undefined,
   760→      };
   761→    });
   762→    // Merge into the selected ticket in the store
   763→    const { tickets } = useTicketStore.getState();
   764→    const updated = tickets.map((t) =>
   765→      t.id === ticketId ? { ...t, messages: msgs } : t,
   766→    );
   767→    useTicketStore.setState({ tickets: updated });
   768→  } catch (err) {
   769→    console.warn(`[ticket-store] fetchTicketMessages ${ticketId} failed:`, err);
   770→  }
   771→}

   773→/**
   774→ * Push a mutation to the backend (fire-and-forget).
   775→ * Used by addTicket / updateTicketStatus / assignVariant / resolveTicket /
   776→ * escalateToHuman / updatePriority / addMessage so the backend is the source
   777→ * of truth (CLAUDE.md #2 fix). Returns true on success, false on failure;
   778→ * failures are logged but do not roll back the optimistic local update —
   779→ * the next syncFromBackend() call will reconcile.
   780→ */
   781→async function pushToBackend(method: string, path: string, body?: unknown): Promise<boolean> {
   782→  if (typeof window === 'undefined') return false;
   783→  try {
   784→    const res = await fetch(path, {
   785→      method,
   786→      headers: { 'Content-Type': 'application/json' },
   787→      credentials: 'include',
   788→      body: body ? JSON.stringify(body) : undefined,
   789→    });
   790→    if (!res.ok) {
   791→      console.warn(`[ticket-store] pushToBackend ${method} ${path} failed:`, res.status);
   792→      return false;
   793→    }
   794→    return true;
   795→  } catch (err) {
   796→    console.warn(`[ticket-store] pushToBackend ${method} ${path} network error:`, err);
   797→    return false;
   798→  }
   799→}

   801→/**
   802→ * Push a mutation to the backend and return the parsed JSON response body.
   803→ * Used by addTicket to capture the real ticket ID returned by the backend
   804→ * (avoids the optimistic-UUID leak where escalate/resolve calls hit the
   805→ * backend with a local UUID instead of the real persisted ticket ID).
   806→ *
   807→ * Returns the parsed response body on success, or null on failure.
   808→ */
   809→async function pushToBackendWithBody<T = unknown>(method: string, path: string, body?: unknown): Promise<T | null> {
   810→  if (typeof window === 'undefined') return null;
   811→  try {
   812→    const res = await fetch(path, {
   813→      method,
   814→      headers: { 'Content-Type': 'application/json' },
   815→      credentials: 'include',
   816→      body: body ? JSON.stringify(body) : undefined,
   817→    });
   818→    if (!res.ok) {
   819→      console.warn(`[ticket-store] pushToBackendWithBody ${method} ${path} failed:`, res.status);
   820→      return null;
   821→    }
   822→    return (await res.json()) as T;
   823→  } catch (err) {
   824→    console.warn(`[ticket-store] pushToBackendWithBody ${method} ${path} network error:`, err);
   825→    return null;
   826→  }

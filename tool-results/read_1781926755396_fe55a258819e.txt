   389→  const [isPaused, setIsPaused] = useState(false);
   390→  const [progressKey, setProgressKey] = useState(0);
   391→
   392→  const nextSlide = useCallback(() => {
   393→    setCurrentSlide((prev) => (prev + 1) % slides.length);
   394→    setProgressKey((prev) => prev + 1);
   395→  }, []);
   396→
   397→  const prevSlide = useCallback(() => {
   398→    setCurrentSlide((prev) => (prev - 1 + slides.length) % slides.length);
   399→    setProgressKey((prev) => prev + 1);
   400→  }, []);
   401→
   402→  const goToSlide = useCallback((index: number) => {
   403→    setCurrentSlide(index);

   408→    if (isPaused) return;
   409→    const interval = setInterval(nextSlide, 6000);
   410→    return () => clearInterval(interval);
   411→  }, [isPaused, nextSlide]);
   412→
   413→  useEffect(() => {
   414→    const handleVisibility = () => setIsPaused(document.hidden);

     3→
     4→This is the connection point between the Jarvis service and the
     5→Mini Parwa / Parwa / Parwa High LangGraph pipelines.
     6→
     7→When a message comes in:
     8→  1. Look up the variant tier from the session context
     9→  2. Resolve the full VariantConfig (tier + industry)
    10→  3. Run the message through the appropriate pipeline
    11→  4. Return the response + metadata to the Jarvis service
    12→
    13→This bridge handles BOTH paths:
    14→  - Customer care:   Always routes through variant pipeline (tier from handoff)
    15→  - Onboarding:      Routes through variant pipeline IF variant_tier is set
    16→                      in context (e.g. user selected a tier on Models page).
    17→                      Falls back to direct AI if no tier is set.
    18→
    19→Pipeline Selection:
    20→  - mini_parwa (Starter): 10-node pipeline, Tier 1 techniques
    21→  - parwa (Growth):       15-node pipeline, Tier 1+2 techniques
    22→  - parwa_high (High):    20-node pipeline, all techniques (falls back to parwa)
    23→
    24→Architecture:
    25→  jarvis_service.send_message()
    26→       ├─ (onboarding + variant_tier set)
    27→       │    → variant_pipeline_bridge.process_onboarding_message()

    35→                 → Mini/Pro pipeline

   168→async def process_customer_care_message(
   169→    query: str,
   170→    company_id: str,
   171→    session_context: Dict[str, Any],
   172→    conversation_id: str = "",
   173→    ticket_id: str = "",
   174→    channel: str = "chat",
   175→    customer_id: str = "",
   176→    customer_tier: str = "free",
   177→) -> PipelineResult:
   178→    """Process a customer care message through the appropriate variant pipeline.
   179→
   180→    This is the entry point called by jarvis_service when handling
   181→    customer_care type sessions.
   182→
   183→    Flow:
   184→      1. Resolve variant_tier from session context
   185→      2. Resolve industry from session context
   186→      3. Select the appropriate pipeline
   187→      4. Run the message through the pipeline
   188→      5. Return a PipelineResult with response + metadata
   189→
   190→    Args:
   191→        query: Customer's raw message.
   192→        company_id: Tenant identifier (BC-001).
   193→        session_context: The Jarvis session's context_json (dict).
   194→        conversation_id: For multi-turn tracking.
   195→        ticket_id: Ticket identifier.
   196→        channel: Communication channel.
   197→        customer_id: Customer identifier.
   198→        customer_tier: Customer subscription tier.
   199→
   200→    Returns:
   201→        PipelineResult with response text and all pipeline metadata.
   202→    """
   203→    start = time.monotonic()
   204→    try:
   205→        # ── Step 1: Resolve variant tier from context ──
   206→        variant_tier = _resolve_tier_from_session(session_context)
   207→
   208→        # ── Step 2: Resolve industry from context ──
   209→        industry = _resolve_industry_from_session(session_context)
   210→
   211→        # ── Step 3: Get variant instance_id from context ──
   212→        variant_instance_id = session_context.get(
   213→            "variant_instance_id", f"inst_{variant_tier}_{company_id}",
   214→        )
   215→
   216→        logger.info(
   217→            "process_customer_care_message: tier=%s, industry=%s, "
   218→            "company_id=%s, instance=%s",
   219→            variant_tier, industry, company_id, variant_instance_id,
   220→        )
   221→
   222→        # ── Step 4: Select and run pipeline ──
   223→        result = await _run_pipeline(
   224→            variant_tier=variant_tier,
   225→            query=query,
   226→            company_id=company_id,
   227→            industry=industry,
   228→            variant_instance_id=variant_instance_id,
   229→            conversation_id=conversation_id,
   230→            ticket_id=ticket_id,
   231→            channel=channel,
   232→            customer_id=customer_id,
   233→            customer_tier=customer_tier,
   234→        )
   235→
   236→        # ── Step 5: Execute external tool actions (post-pipeline) ──
   237→        # BC-008: Tool failures don't affect the pipeline result.
   238→        # This is a fire-and-forget side effect — if it fails, the
   239→        # customer still gets their AI response.
   240→        try:
   241→            from app.core.external_tool_executor import execute_pipeline_actions
   242→
   243→            # Extract customer contact info from session context
   244→            customer_email = session_context.get("customer_email", "")
   245→            customer_phone = session_context.get("customer_phone", "")
   246→
   247→            # Get raw pipeline result dict for action extraction
   248→            raw_result = result.metadata or {}
   249→            raw_result["step_outputs"] = raw_result.get("step_outputs", {})
   250→            raw_result["emergency_flag"] = result.emergency_flag
   251→            raw_result["quality_score"] = result.quality_score
   252→            raw_result["pipeline_status"] = result.pipeline_status
   253→
   254→            tool_results = await execute_pipeline_actions(
   255→                variant_tier=variant_tier,
   256→                company_id=company_id,
   257→                pipeline_result=raw_result,
   258→                customer_email=customer_email,
   259→                customer_phone=customer_phone,
   260→                ticket_number=ticket_id,
   261→                ticket_id=ticket_id,
   262→            )
   263→
   264→            # Store tool execution results in metadata (non-blocking)
   265→            if tool_results:
   266→                result.metadata["external_tool_results"] = {
   267→                    k: {"channel": v.channel.value if hasattr(v.channel, 'value') else v.channel,
   268→                        "success": v.success,
   269→                        "message_id": v.message_id, "error": v.error}
   270→                    for k, v in tool_results.items()
   271→                }
   272→                result.metadata["tools_executed"] = len(tool_results)
   273→                result.metadata["tools_succeeded"] = sum(
   274→                    1 for v in tool_results.values() if v.success
   275→                )
   276→
   277→        except Exception as tool_exc:
   278→            # BC-008: Never crash — just log and continue
   279→            logger.warning(
   280→                "external_tool_execution_failed (non-blocking): %s",
   281→                str(tool_exc)[:200],
   282→            )
   283→
   284→        total_ms = round((time.monotonic() - start) * 1000, 2)
   285→        logger.info(
   286→            "process_customer_care_message_complete: tier=%s, status=%s, "
   287→            "latency=%sms, quality=%.1f, steps=%d",
   288→            variant_tier, result.pipeline_status, total_ms,
   289→            result.quality_score, len(result.steps_completed),
   290→        )
   291→
   292→        return result
   293→
   294→    except Exception:
   295→        total_ms = round((time.monotonic() - start) * 1000, 2)
   296→        logger.exception(
   297→            "process_customer_care_message failed: company_id=%s, "
   298→            "latency=%sms",
   299→            company_id, total_ms,
   300→        )
   301→        # BC-008: Return a safe fallback result
   302→        return PipelineResult(
   303→            response_text=(
   304→                "I apologize, I'm having trouble processing your request "
   305→                "right now. A team member will follow up with you shortly."
   306→            ),
   307→            variant_tier=session_context.get("variant_tier", "mini_parwa"),
   308→            industry=session_context.get("industry", "general"),
   309→            pipeline_status="failed",
   310→            total_latency_ms=total_ms,
   311→            metadata={"error": "pipeline_bridge_failed"},
   312→        )

   594→) -> PipelineResult:
   595→    """Process a message through both live and shadow variants.
   596→
   597→    When shadow mode is active for a company, this function:
   598→      1. Runs the LIVE variant pipeline (normal processing)
   599→      2. Runs the SHADOW variant pipeline (parallel, non-blocking)
   600→      3. Compares the results (quality, latency, tokens)
   601→      4. Records the comparison for analysis
   602→      5. Returns the LIVE variant result (shadow is never shown to customer)
   603→
   604→    In SUPERVISED mode, the shadow variant's response is returned
   605→    instead, but flagged for human review before delivery.
   606→
   607→    BC-001: company_id first parameter.
   608→    BC-008: Never crashes — returns live result on shadow failure.
   609→    BC-012: All timestamps UTC.
   610→    """
   611→    try:
   612→        from app.services.shadow_mode_service import get_shadow_mode_service
   613→
   614→        service = get_shadow_mode_service()
   615→        shadow_config = service.get_shadow_config(company_id)
   616→
   617→        if shadow_config is None:
   618→            # No active shadow mode — just run normal pipeline
   619→            return await process_customer_care_message(
   620→                query=query,
   621→                company_id=company_id,
   622→                session_context=session_context,
   623→                conversation_id=conversation_id,
   624→                ticket_id=ticket_id,
   625→                channel=channel,
   626→                customer_id=customer_id,
   627→                customer_tier=customer_tier,
   628→            )
   629→
   630→        # Check if this message should be shadow-processed
   631→        should_shadow, reason = service.should_process_shadow(
   632→            company_id=company_id, message=query,
   633→        )
   634→
   635→        if not should_shadow:
   636→            logger.debug(
   637→                "Shadow mode: skipping message for company_id=%s, reason=%s",
   638→                company_id, reason,
   639→            )
   640→            return await process_customer_care_message(
   641→                query=query,
   642→                company_id=company_id,
   643→                session_context=session_context,
   644→                conversation_id=conversation_id,
   645→                ticket_id=ticket_id,
   646→                channel=channel,
   647→                customer_id=customer_id,
   648→                customer_tier=customer_tier,
   649→            )
   650→
   651→        live_variant = shadow_config.get("live_variant", "mini_parwa")
   652→        shadow_variant = shadow_config.get("shadow_variant", "parwa")
   653→        shadow_status = shadow_config.get("status", "shadow")
   654→
   655→        logger.info(
   656→            "Shadow mode processing: company_id=%s, live=%s, shadow=%s, "
   657→            "status=%s, ticket_id=%s",
   658→            company_id, live_variant, shadow_variant, shadow_status, ticket_id,
   659→        )
   660→
   661→        # ── Run LIVE pipeline ──
   662→        live_result = await _run_pipeline(
   663→            variant_tier=live_variant,
   664→            query=query,
   665→            company_id=company_id,
   666→            industry=_resolve_industry_from_session(session_context),
   667→            conversation_id=conversation_id,
   668→            ticket_id=ticket_id,
   669→            channel=channel,
   670→            customer_id=customer_id,
   671→            customer_tier=customer_tier,
   672→        )
   673→
   674→        # ── Run SHADOW pipeline (best-effort, don't fail on error) ──
   675→        shadow_result = None
   676→        try:
   677→            shadow_result = await _run_pipeline(
   678→                variant_tier=shadow_variant,
   679→                query=query,
   680→                company_id=company_id,
   681→                industry=_resolve_industry_from_session(session_context),
   682→                conversation_id=f"shadow_{conversation_id}",
   683→                ticket_id=ticket_id,
   684→                channel=channel,
   685→                customer_id=customer_id,
   686→                customer_tier=customer_tier,
   687→            )
   688→        except Exception:
   689→            logger.exception(
   690→                "Shadow pipeline failed for company_id=%s — "
   691→                "continuing with live result only",
   692→                company_id,
   693→            )
   694→
   695→        # ── Record comparison ──
   696→        if shadow_result is not None:
   697→            try:
   698→                from app.services.shadow_mode_service import ShadowComparison
   699→
   700→                live_quality = live_result.quality_score or 0.0
   701→                shadow_quality = shadow_result.quality_score or 0.0
   702→
   703→                comparison = ShadowComparison(
   704→                    company_id=company_id,
   705→                    config_id=shadow_config.get("id", ""),
   706→                    ticket_id=ticket_id,
   707→                    conversation_id=conversation_id,
   708→                    message_hash=service._hash_message(query),
   709→                    live_variant=live_variant,
   710→                    live_response=live_result.response_text[:500],
   711→                    live_quality_score=live_quality,
   712→                    live_latency_ms=int(live_result.total_latency_ms or 0),
   713→                    live_tokens_used=live_result.billing_tokens or 0,
   714→                    shadow_variant=shadow_variant,
   715→                    shadow_response=shadow_result.response_text[:500],
   716→                    shadow_quality_score=shadow_quality,
   717→                    shadow_latency_ms=int(shadow_result.total_latency_ms or 0),
   718→                    shadow_tokens_used=shadow_result.billing_tokens or 0,
   719→                    quality_delta=round(shadow_quality - live_quality, 4),
   720→                    latency_delta_ms=int(
   721→                        (shadow_result.total_latency_ms or 0)
   722→                        - (live_result.total_latency_ms or 0)
   723→                    ),
   724→                    token_delta=(shadow_result.billing_tokens or 0) - (live_result.billing_tokens or 0),
   725→                    shadow_winner=shadow_quality >= live_quality,
   726→                    mode_at_comparison=shadow_status,
   727→                )
   728→
   729→                service.record_comparison(
   730→                    company_id=company_id,
   731→                    comparison=comparison,
   732→                )
   733→
   734→            except Exception:
   735→                logger.exception(
   736→                    "Failed to record shadow comparison for company_id=%s",
   737→                    company_id,
   738→                )
   739→
   740→        # ── Determine which result to return ──
   741→        if shadow_status == "supervised" and shadow_result is not None:
   742→            # In supervised mode, return shadow result but flag for review
   743→            shadow_result.metadata["shadow_mode"] = "supervised"
   744→            shadow_result.metadata["requires_human_review"] = True
   745→            shadow_result.metadata["live_response_available"] = True
   746→            shadow_result.metadata["live_variant"] = live_variant
   747→            logger.info(
   748→                "Supervised mode: returning shadow result for review, "
   749→                "company_id=%s, ticket_id=%s",
   750→                company_id, ticket_id,
   751→            )
   752→            return shadow_result
   753→        else:
   754→            # In shadow mode, always return live result
   755→            live_result.metadata["shadow_mode"] = "shadow"
   756→            live_result.metadata["shadow_processed"] = shadow_result is not None
   757→            live_result.metadata["shadow_variant"] = shadow_variant
   758→            return live_result
   759→
   760→    except Exception:
   761→        logger.exception(
   762→            "process_shadow_mode failed for company_id=%s — "
   763→            "falling back to normal processing",
   764→            company_id,
   765→        )
   766→        return await process_customer_care_message(
   767→            query=query,
   768→            company_id=company_id,
   769→            session_context=session_context,
   770→            conversation_id=conversation_id,
   771→            ticket_id=ticket_id,
   772→            channel=channel,
   773→            customer_id=customer_id,
   774→            customer_tier=customer_tier,
   775→        )

   783→async def _run_pipeline(

   796→
   797→    Shared by both customer_care and onboarding paths.

   799→    Args:
   800→        variant_tier: 'mini_parwa' | 'parwa' | 'parwa_high'.

   809→        customer_tier: Customer subscription tier.
   810→

   812→        PipelineResult with response + full pipeline metadata.
   813→    """
   814→    if variant_tier == "mini_parwa":
   815→        return await _run_mini_parwa(
   816→            query=query,

   839→        return await _run_parwa_high(
   840→            query=query,
   841→            company_id=company_id,
   842→            industry=industry,
   843→            variant_instance_id=variant_instance_id,
   844→            conversation_id=conversation_id,
   845→            ticket_id=ticket_id,
   846→            channel=channel,
   847→            customer_id=customer_id,
   848→            customer_tier=customer_tier,
   849→        )
   850→    else:
   851→        # Default: mini_parwa (safest, cheapest)
   852→        return await _run_mini_parwa(
   853→            query=query,
   854→            company_id=company_id,
   855→            industry=industry,

  1158→async def _run_parwa_high(
  1159→    query: str,
  1160→    company_id: str,
  1161→    industry: str,
  1162→    variant_instance_id: str = "",
  1163→    conversation_id: str = "",
  1164→    ticket_id: str = "",
  1165→    channel: str = "chat",
  1166→    customer_id: str = "",
  1167→    customer_tier: str = "free",
  1168→) -> PipelineResult:
  1169→    """Run a message through the High Parwa 20-node pipeline.
  1170→
  1171→    Pipeline: pii_check -> empathy_check -> emergency_check -> gsd_state
  1172→              -> classify -> extract_signals -> technique_select
  1173→              -> reasoning_chain -> context_enrich -> context_compress
  1174→              -> generate -> crp_compress -> clara_quality_gate
  1175→              -> quality_retry (max 2) -> confidence_assess
  1176→              -> context_health -> dedup -> strategic_decision
  1177→              -> peer_review -> format -> END
  1178→
  1179→    Connected Frameworks (Tier 1 + Tier 2 + Tier 3):
  1180→      - CLARA (Quality Gate, strictest: threshold 95, 8-check)
  1181→      - CRP (Token Compression)
  1182→      - GSD (State Engine)
  1183→      - Smart Router (Heavy tier)
  1184→      - Technique Router (Tier 1+2+3)
  1185→      - Confidence Scoring
  1186→      - CoT, ReAct, Reverse Thinking, Step-Back, ThoT (Tier 2)
  1187→      - GST, UoT, ToT, Self-Consistency, Reflexion, Least-to-Most (Tier 3)
  1188→
  1189→    Args:
  1190→        query: Customer's raw message.
  1191→        company_id: Tenant identifier (BC-001).
  1192→        industry: Industry enum value.
  1193→        variant_instance_id: Specific variant instance.
  1194→        conversation_id: For multi-turn tracking.
  1195→        ticket_id: Ticket identifier.
  1196→        channel: Communication channel.
  1197→        customer_id: Customer identifier.
  1198→        customer_tier: Customer subscription tier.
  1199→
  1200→    Returns:
  1201→        PipelineResult with response + full pipeline metadata.
  1202→    """
  1203→    try:
  1204→        pipeline = _get_parwa_high_pipeline()
  1205→
  1206→        if pipeline is None:
  1207→            logger.error("ParwaHighPipeline not available — falling back to Pro")
  1208→            return await _run_parwa(
  1209→                query=query,
  1210→                company_id=company_id,
  1211→                industry=industry,
  1212→                variant_instance_id=variant_instance_id,
  1213→                conversation_id=conversation_id,
  1214→                ticket_id=ticket_id,
  1215→                channel=channel,
  1216→                customer_id=customer_id,
  1217→                customer_tier=customer_tier,
  1218→            )
  1219→
  1220→        # Run through High Parwa's process_ticket method
  1221→        result = await pipeline.process_ticket(
  1222→            query=query,
  1223→            company_id=company_id,
  1224→            industry=industry,
  1225→            channel=channel,
  1226→            customer_id=customer_id,
  1227→            customer_tier=customer_tier,
  1228→            conversation_id=conversation_id,
  1229→            ticket_id=ticket_id,
  1230→            variant_instance_id=variant_instance_id,
  1231→        )
  1232→
  1233→        # Extract the formatted response from pipeline result
  1234→        response_text = result.get("formatted_response", "") or result.get(
  1235→            "generated_response", ""
  1236→        )
  1237→
  1238→        # If pipeline generated an emergency escalation response
  1239→        if result.get("emergency_flag", False):
  1240→            response_text = result.get("formatted_response", "") or (
  1241→                "Your message has been flagged for priority handling. "
  1242→                "A senior team member will contact you directly. "
  1243→                f"Reference: {result.get('ticket_id', 'N/A')}"
  1244→            )
  1245→
  1246→        # If no response was generated, use fallback
  1247→        if not response_text:
  1248→            classification = result.get("classification", {})
  1249→            intent = classification.get("intent", "general")
  1250→            from app.core.parwa_high.nodes import TEMPLATE_RESPONSES
  1251→            response_text = TEMPLATE_RESPONSES.get(
  1252→                intent, TEMPLATE_RESPONSES["general"],
  1253→            )
  1254→
  1255→        # Get technique info (High-specific)
  1256→        technique_used = result.get("step_outputs", {}).get(
  1257→            "technique_select", {},
  1258→        ).get("primary_technique", "direct")
  1259→        reasoning_technique = result.get("step_outputs", {}).get(
  1260→            "reasoning_chain", {},
  1261→        ).get("technique", "direct")
  1262→
  1263→        # Get peer review info (High-specific)
  1264→        peer_review = result.get("step_outputs", {}).get("peer_review", {})
  1265→
  1266→        # Get strategic decision info (High-specific)
  1267→        strategic_decision = result.get("step_outputs", {}).get(
  1268→            "strategic_decision", {},
  1269→        )
  1270→
  1271→        return PipelineResult(
  1272→            response_text=response_text,
  1273→            variant_tier="parwa_high",
  1274→            industry=industry,
  1275→            pipeline_status=result.get("pipeline_status", "completed"),
  1276→            quality_score=result.get("quality_score", 0.0),
  1277→            total_latency_ms=result.get("total_latency_ms", 0.0),
  1278→            billing_tokens=result.get("billing_tokens", 0),
  1279→            steps_completed=result.get("steps_completed", []),
  1280→            technique_used=technique_used,
  1281→            emergency_flag=result.get("emergency_flag", False),
  1282→            empathy_score=result.get("empathy_score", 0.5),
  1283→            classification_intent=result.get("classification", {}).get("intent", ""),
  1284→            metadata={
  1285→                "ticket_id": result.get("ticket_id", ""),
  1286→                "conversation_id": result.get("conversation_id", ""),
  1287→                "pii_detected": result.get("pii_detected", False),
  1288→                "reasoning_technique": reasoning_technique,
  1289→                "quality_passed": result.get("quality_passed", True),
  1290→                "quality_retry_count": result.get("quality_retry_count", 0),
  1291→                "peer_review_passed": peer_review.get("passed", True),
  1292→                "peer_review_score": peer_review.get("review_score", 0.0),
  1293→                "strategic_decision": strategic_decision.get("decision", "proceed"),
  1294→                "context_health_score": result.get("step_outputs", {}).get(
  1295→                    "context_health", {},
  1296→                ).get("health_score", 1.0),
  1297→                "dedup_is_duplicate": result.get(
  1298→                    "dedup_is_duplicate", False,
  1299→                ),
  1300→                "context_compression_ratio": result.get(
  1301→                    "context_compression_ratio", 1.0,
  1302→                ),
  1303→                "gsd_state": result.get("step_outputs", {}).get(
  1304→                    "gsd_state", {},
  1305→                ).get("to_state", ""),
  1306→                "crp_compression_ratio": result.get("step_outputs", {}).get(
  1307→                    "crp_compress", {},
  1308→                ).get("compression_ratio", 1.0),
  1309→                "clara_passed": result.get("step_outputs", {}).get(
  1310→                    "clara_quality_gate", {},
  1311→                ).get("passed", True),
  1312→                "confidence_score": result.get("step_outputs", {}).get(
  1313→                    "confidence_assess", {},
  1314→                ).get("confidence_score", 0.5),
  1315→            },
  1316→        )
  1317→
  1318→    except Exception:
  1319→        logger.exception("_run_parwa_high failed")
  1320→        return PipelineResult(
  1321→            response_text=(
  1322→                "I apologize for the inconvenience. "
  1323→                "Our team will follow up with you shortly."
  1324→            ),
  1325→            variant_tier="parwa_high",
  1326→            industry=industry,
  1327→            pipeline_status="failed",
  1328→            metadata={"error": "parwa_high_execution_failed"},
  1329→        )

  1337→def _resolve_tier_from_session(session_context: Dict[str, Any]) -> str:
  1338→    """Resolve the variant tier from a session's context.
  1339→
  1340→    The context_json should contain variant_tier set during handoff
  1341→    or during onboarding when user selects a variant on Models page.
  1342→    If not present, tries to resolve from variant_id/selected_variants.
  1343→
  1344→    Args:
  1345→        session_context: The Jarvis session's context_json (dict).
  1346→
  1347→    Returns:
  1348→        Backend pipeline tier string.
  1349→    """
  1350→    try:
  1351→        # Direct tier (set during handoff or Models page selection — best path)
  1352→        tier = session_context.get("variant_tier")
  1353→        if tier and tier in ("mini_parwa", "parwa", "parwa_high"):
  1354→            return tier
  1355→
  1356→        # Try to resolve from variant_id
  1357→        variant_id = session_context.get("variant_id")
  1358→        if variant_id:
  1359→            return resolve_tier_from_context(variant_id=variant_id)
  1360→
  1361→        # Try to resolve from selected_variants
  1362→        selected_variants = session_context.get("selected_variants")
  1363→        if selected_variants:
  1364→            return resolve_tier_from_context(
  1365→                selected_variants=selected_variants,
  1366→            )
  1367→
  1368→        # Default: mini_parwa (safest, cheapest)
  1369→        return "mini_parwa"
  1370→
  1371→    except Exception:
  1372→        return "mini_parwa"

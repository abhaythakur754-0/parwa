    16→// ─── OTP Input Component ─────────────────────────────────────────────────────
    17→
    18→function OTPInput({ value, onChange, disabled, error }: {
    19→  value: string;
    20→  onChange: (val: string) => void;
    21→  disabled?: boolean;
    22→  error?: boolean;
    23→}) {
    24→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
    25→
    26→  useEffect(() => {
    27→    // Auto-focus first empty input
    28→    const firstEmpty = value.indexOf('');
    29→    if (firstEmpty >= 0 && inputRefs.current[firstEmpty]) {
    30→      inputRefs.current[firstEmpty]?.focus();
    31→    }
    32→  }, [value]);
    33→
    34→  const handleChange = (index: number, char: string) => {
    35→    if (!/^\d*$/.test(char)) return;
    36→    const newVal = value.split('');
    37→    newVal[index] = char;
    38→    const result = newVal.join('');
    39→    onChange(result);
    40→    // Auto-focus next input
    41→    if (char && index < 5 && inputRefs.current[index + 1]) {
    42→      inputRefs.current[index + 1]?.focus();
    43→    }
    44→  };
    45→
    46→  const handleKeyDown = (index: number, e: React.KeyboardEvent) => {
    47→    if (e.key === 'Backspace' && !value[index] && index > 0) {
    48→      inputRefs.current[index - 1]?.focus();
    49→    }
    50→  };
    51→
    52→  const handlePaste = (e: React.ClipboardEvent) => {
    53→    e.preventDefault();
    54→    const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
    55→    if (pasted) {
    56→      onChange(pasted);
    57→      const focusIdx = Math.min(pasted.length, 5);
    58→      inputRefs.current[focusIdx]?.focus();
    59→    }
    60→  };
    61→
    62→  return (
    63→    <div className="flex justify-center gap-2 sm:gap-3">
    64→      {Array.from({ length: 6 }).map((_, i) => (
    65→        <input
    66→          key={i}
    67→          ref={(el) => { inputRefs.current[i] = el; }}
    68→          type="text"
    69→          inputMode="numeric"
    70→          maxLength={1}
    71→          value={value[i] || ''}
    72→          onChange={(e) => handleChange(i, e.target.value)}
    73→          onKeyDown={(e) => handleKeyDown(i, e)}
    74→          onPaste={handlePaste}
    75→          disabled={disabled}
    76→          className={`w-10 h-12 sm:w-12 sm:h-14 text-center text-lg sm:text-xl font-bold rounded-xl border transition-all duration-300 focus:outline-none focus:ring-2 ${
    77→            error
    78→              ? 'border-rose-500/40 focus:ring-rose-500/30 bg-rose-500/5 text-rose-300'
    79→              : 'border-white/10 focus:ring-orange-500/30 focus:border-orange-500/40 bg-white/5 text-white'
    80→          }`}
    81→          style={{ caretColor: '#FF7F11' }}
    82→        />
    83→      ))}
    84→    </div>
    85→  );
    86→}

   546→                  <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none">
   547→                    <Lock className="h-5 w-5 text-orange-400/50" />
   548→                  </div>
   549→                  <input
   550→                    id="new-password"
   551→                    name="new-password"
   552→                    type={showPassword ? 'text' : 'password'}
   553→                    required
   554→                    value={newPassword}
   555→                    onChange={(e) => setNewPassword(e.target.value)}
   556→                    className="w-full pl-10 pr-10 py-3 bg-white/5 border border-white/10 rounded-xl text-white placeholder-orange-200/30 focus:outline-none focus:ring-2 focus:ring-orange-500/30 focus:border-orange-500/40 transition-all duration-300"
   557→                    placeholder="Enter new password"
   558→                    disabled={isLoading}
   559→                  />
   560→                  <button
   561→                    type="button"
   562→                    onClick={() => setShowPassword(!showPassword)}
   563→                    className="absolute inset-y-0 right-0 pr-3 flex items-center text-orange-400/50 hover:text-orange-400 transition-colors"
   564→                    tabIndex={-1}
   565→                  >
   566→                    {showPassword ? <EyeOff className="h-5 w-5" /> : <Eye className="h-5 w-5" />}
   567→                  </button>
   568→                </div>
   569→              </div>
   570→
   571→              {/* Password Strength Indicator */}
   572→              {newPassword && (
   573→                <div className="grid grid-cols-2 gap-1.5 text-xs">
   574→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${isLongEnough ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   575→                    <span>{isLongEnough ? '✓' : '○'}</span> 8+ characters
   576→                  </div>
   577→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${hasUppercase ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   578→                    <span>{hasUppercase ? '✓' : '○'}</span> Uppercase
   579→                  </div>
   580→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${hasLowercase ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   581→                    <span>{hasLowercase ? '✓' : '○'}</span> Lowercase
   582→                  </div>
   583→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${hasDigit ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   584→                    <span>{hasDigit ? '✓' : '○'}</span> Number
   585→                  </div>
   586→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${hasSpecial ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   587→                    <span>{hasSpecial ? '✓' : '○'}</span> Special char
   588→                  </div>
   589→                  <div className={`flex items-center gap-1.5 px-2 py-1 rounded-md transition-colors ${passwordsMatch ? 'text-orange-400 bg-orange-500/10' : 'text-orange-200/30 bg-white/[0.02]'}`}>
   590→                    <span>{passwordsMatch ? '✓' : '○'}</span> Match
   591→                  </div>
   592→                </div>
   593→              )}
   594→
   595→              {/* Confirm Password */}
   596→              <div>
   597→                <label htmlFor="confirm-password" className="block text-sm font-medium text-orange-200/70 mb-2">
   598→                  Confirm Password
   599→                </label>
   600→                <div className="relative">
   601→                  <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none">
   602→                    <Lock className="h-5 w-5 text-orange-400/50" />
   603→                  </div>
   604→                  <input
   605→                    id="confirm-password"
   606→                    name="confirm-password"
   607→                    type={showPassword ? 'text' : 'password'}
   608→                    required
   609→                    value={confirmPassword}
   610→                    onChange={(e) => setConfirmPassword(e.target.value)}
   611→                    className="w-full pl-10 pr-4 py-3 bg-white/5 border border-white/10 rounded-xl text-white placeholder-orange-200/30 focus:outline-none focus:ring-2 focus:ring-orange-500/30 focus:border-orange-500/40 transition-all duration-300"
   612→                    placeholder="Confirm new password"
   613→                    disabled={isLoading}
   614→                  />
   615→                </div>
   616→              </div>
   617→
   618→              {error && (
   619→                <div className="p-3 rounded-lg bg-rose-500/10 border border-rose-500/20">
   620→                  <p className="text-sm text-rose-300">{error}</p>
   621→                </div>
   622→              )}
   623→
   624→              <button
   625→                type="submit"
   626→                disabled={isLoading || !isPasswordValid}
   627→                className="w-full py-3 px-4 bg-gradient-to-r from-orange-500 to-orange-400 hover:from-orange-400 hover:to-orange-300 text-[#1A1A1A] font-semibold rounded-xl transition-all duration-500 disabled:opacity-50 disabled:cursor-not-allowed flex items-center justify-center gap-2 shadow-lg shadow-orange-600/25 hover:shadow-orange-600/40"
   628→              >
   629→                {isLoading ? (
   630→                  <>
   631→                    <Loader2 className="w-5 h-5 animate-spin" />
   632→                    Resetting password...
   633→                  </>
   634→                ) : (
   635→                  <>
   636→                    <CheckCircle className="w-5 h-5" />
   637→                    Reset Password
   638→                  </>
   639→                )}
   640→              </button>
   641→            </form>
   642→          )}
   643→        </div>
   644→
   645→        {/* Back to Login */}
   646→        <div className="text-center">
   647→          <Link
   648→            href="/login"
   649→            className="inline-flex items-center gap-2 text-sm text-orange-200/40 hover:text-orange-300 transition-colors"
   650→          >
   651→            <ArrowLeft className="w-4 h-4" />
   652→            Back to login
   653→          </Link>
   654→        </div>
   655→      </div>
   656→
   657→    </div>
   658→  );
   659→}

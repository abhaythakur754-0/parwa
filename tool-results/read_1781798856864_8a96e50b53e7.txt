   466→# IN-MEMORY BACKEND (works immediately, no deps)
   467→# ═══════════════════════════════════════════════════════════════
   468→
   469→class InMemoryBackend(StorageBackend):
   470→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   471→
   472→    def __init__(self):
   473→        self._notifications: Dict[str, Dict] = {}
   474→        self._flags: List[Dict] = []
   475→        self._audit: List[Dict] = []
   476→        self._quality_scores: List[Dict] = []
   477→        self._batches: Dict[str, List[Dict]] = {}
   478→        self._batch_metas: Dict[str, Dict] = {}
   479→        self._lock = threading.Lock()
   480→        self._next_nfy_id = 1
   481→        # Wave 2 stores
   482→        self._integration_pings: List[Dict] = []
   483→        self._llm_costs: List[Dict] = []
   484→        self._stuck_ticket_events: List[Dict] = []
   485→        self._load_status: Dict[str, Dict] = {}  # tenant_id -> {variant -> {concurrent, max_concurrent}}
   486→        self._agent_configs: List[Dict] = []  # Wave 3: agent config store
   487→        self._outbox: List[Dict] = []  # Wave 3: outbox queue (recall/void)
   488→        self._inbox: List[Dict] = []  # Wave 4: PARWA → Jarvis inbox
   489→        self._training_data_records: List[Dict] = []  # Wave 4: training data
   490→        self._feature_flags: List[Dict] = []  # Wave 5: feature flags
   491→        self._sentiment_logs: List[Dict] = []  # Wave 5: sentiment logs
   492→        self._confidence_logs: List[Dict] = []  # Wave 5: confidence logs
   493→
   494→    # ── Helpers ───────────────────────────────────────────────
   495→
   496→    def _next_notification_number(self) -> int:
   497→        with self._lock:
   498→            n = self._next_nfy_id
   499→            self._next_nfy_id += 1
   500→            return n
   501→
   502→    @staticmethod
   503→    def _priority_from_score(score: float) -> str:
   504→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:
   505→            return PRIORITY_CRITICAL
   506→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_HIGH]:
   507→            return PRIORITY_HIGH
   508→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_MEDIUM]:
   509→            return PRIORITY_MEDIUM
   510→        return PRIORITY_LOW
   511→
   512→    # ── Notifications ────────────────────────────────────────
   513→
   514→    async def create_notification(
   515→        self,
   516→        tenant_id: str,
   517→        ntype: str,
   518→        priority_score: float,
   519→        title: str,
   520→        description: str,
   521→        related_tickets: Optional[List[str]] = None,
   522→        batch_key: Optional[str] = None,
   523→        source_data: Optional[Dict] = None,
   524→    ) -> Dict[str, Any]:
   525→        num = self._next_notification_number()
   526→        key = f"PARWA-NFY-{num:03d}"
   527→        now = time.time()
   528→        nf = {
   529→            "id": str(uuid.uuid4()),
   530→            "tenant_id": tenant_id,
   531→            "notification_key": key,
   532→            "type": ntype,
   533→            "priority": self._priority_from_score(priority_score),
   534→            "priority_score": round(priority_score, 4),
   535→            "title": title,
   536→            "description": description,
   537→            "related_tickets": related_tickets or [],
   538→            "batch_key": batch_key,
   539→            "source_data": source_data or {},
   540→            "is_read": False,
   541→            "is_resolved": False,
   542→            "created_at": now,
   543→            "resolved_at": None,
   544→            "backend": "memory",
   545→        }
   546→        self._notifications[key] = nf
   547→        logger.info("NFY created: %s type=%s score=%.3f [%s]", key, ntype, priority_score, tenant_id)
   548→        return nf
   549→
   550→    async def get_notification(self, key: str) -> Optional[Dict[str, Any]]:
   551→        return self._notifications.get(key)
   552→
   553→    async def get_notifications(
   554→        self,
   555→        tenant_id: str,
   556→        include_resolved: bool = False,
   557→        min_priority: Optional[str] = None,
   558→    ) -> List[Dict[str, Any]]:
   559→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   560→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   561→        results = []
   562→        for n in self._notifications.values():
   563→            if n["tenant_id"] != tenant_id:
   564→                continue
   565→            if not include_resolved and n["is_resolved"]:
   566→                continue
   567→            if priority_order.get(n["priority"], 99) > min_rank:
   568→                continue
   569→            results.append(n)
   570→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   571→        return results
   572→
   573→    async def resolve_notification(self, key: str) -> bool:
   574→        n = self._notifications.get(key)
   575→        if n and not n["is_resolved"]:
   576→            n["is_resolved"] = True
   577→            n["resolved_at"] = time.time()
   578→            return True
   579→        return False
   580→
   581→    async def dismiss_notification(self, key: str) -> bool:
   582→        n = self._notifications.get(key)
   583→        if n:
   584→            n["is_read"] = True
   585→            return True
   586→        return False
   587→

   750→    # ── Batch Queue ──────────────────────────────────────────
   751→
   752→    async def add_to_batch(
   753→        self,
   754→        tenant_id: str,
   755→        batch_key: str,
   756→        ticket_id: str,
   757→        confidence: float,
   758→    ) -> Optional[Dict[str, Any]]:
   759→        BATCH_WINDOW_S = 300
   760→        now = time.time()
   761→
   762→        if tenant_id not in self._batch_metas:
   763→            self._batch_metas[tenant_id] = {}
   764→
   765→        # Find existing batch with same key within window
   766→        existing_key = None
   767→        for bk, meta in self._batch_metas[tenant_id].items():
   768→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   769→                existing_key = bk
   770→                break
   771→
   772→        if existing_key:
   773→            meta = self._batch_metas[tenant_id][existing_key]
   774→            meta["ticket_ids"].append(ticket_id)
   775→            meta["signal_count"] = len(meta["ticket_ids"])
   776→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   777→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   778→            # Don't flush yet — still within window
   779→            return None
   780→        else:
   781→            meta = {
   782→                "batch_key": batch_key,
   783→                "tenant_id": tenant_id,
   784→                "batch_start": now,
   785→                "ticket_ids": [ticket_id],
   786→                "signal_count": 1,
   787→                "confidence_min": confidence,
   788→                "confidence_max": confidence,
   789→                "status": "pending",
   790→            }
   791→            self._batch_metas[tenant_id][batch_key] = meta
   792→            return None

   922→    async def get_integration_health(self, tenant_id: str) -> Dict[str, Any]:
   923→        """Compute health from recent pings (last 30 per service)."""
   924→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
   925→        services: Dict[str, List[Dict]] = {}
   926→        for p in tenant_pings:
   927→            services.setdefault(p["service_name"], []).append(p)
   928→
   929→        result = {"services": {}, "degraded_count": 0, "healthy_count": 0}
   930→        for svc, pings in services.items():
   931→            recent = pings[-30:]  # last 30 pings
   932→            if not recent:
   933→                continue
   934→            healthy = sum(1 for p in recent if p["is_healthy"])
   935→            total = len(recent)
   936→            uptime_pct = round(healthy / max(total, 1) * 100, 1)
   937→            last_ping = recent[-1]
   938→            avg_ms = (sum(p["response_ms"] or 0 for p in recent) / total) if total else 0
   939→            last_error = None
   940→            last_error_time = None
   941→            for p in reversed(recent):
   942→                if not p["is_healthy"]:
   943→                    last_error = p.get("error_detail")
   944→                    last_error_time = p["created_at"]
   945→                    break
   946→
   947→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
   948→            if status != "healthy":
   949→                result["degraded_count"] += 1
   950→            else:
   951→                result["healthy_count"] += 1
   952→
   953→            result["services"][svc] = {
   954→                "status": status,
   955→                "uptime_pct": uptime_pct,
   956→                "total_pings": total,
   957→                "healthy_pings": healthy,
   958→                "avg_response_ms": round(avg_ms, 1),
   959→                "last_ping_at": last_ping["created_at"],
   960→                "last_healthy_at": next(
   961→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
   962→                    None,
   963→                ),
   964→                "last_error": last_error,
   965→                "last_error_at": last_error_time,
   966→                "last_status_code": last_ping.get("status_code"),
   967→            }
   968→        return result

  1050→    # ── Drift Detection (Wave 2) ─────────────────────────────
  1051→
  1052→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1053→        """Analyze quality_scores for drift using day-over-day comparison."""
  1054→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
  1055→        if len(tenant_scores) < 3:
  1056→            return {
  1057→                "tenant_id": tenant_id, "drift_detected": False,
  1058→                "drift_severity": "none", "accuracy_7d": None,
  1059→                "accuracy_yesterday": None, "accuracy_today": None,
  1060→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
  1061→                "total_scores": len(tenant_scores),
  1062→            }
  1063→        # Group by date (ISO date portion)
  1064→        by_date: Dict[str, List[float]] = {}
  1065→        for s in tenant_scores:
  1066→            date_str = s["created_at"][:10]  # YYYY-MM-DD
  1067→            by_date.setdefault(date_str, []).append(s["overall_score"])
  1068→
  1069→        dates_sorted = sorted(by_date.keys())
  1070→
  1071→        def _avg(scores: List[float]) -> float:
  1072→            return round(sum(scores) / len(scores), 4) if scores else 0.0
  1073→
  1074→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
  1075→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1076→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
  1077→
  1078→        # Drift detection: >5% drop for 3+ days
  1079→        drift_detected = False
  1080→        drift_severity = "none"
  1081→        trigger_reason = None
  1082→        trend_direction = "stable"
  1083→
  1084→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
  1085→            recent_3 = dates_sorted[-3:]
  1086→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
  1087→            # Check if declining 3+ consecutive days
  1088→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
  1089→            drop_7d = accuracy_7d - accuracy_today
  1090→
  1091→            if declining and drop_7d > 0.05:
  1092→                drift_detected = True
  1093→                trend_direction = "declining"
  1094→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
  1095→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
  1096→            elif drop_7d > 0.03:
  1097→                drift_detected = True
  1098→                trend_direction = "slight_decline"
  1099→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
  1100→                drift_severity = "warning"
  1101→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
  1102→                trend_direction = "improving"
  1103→            else:
  1104→                trend_direction = "stable"
  1105→
  1106→        # Check for repeated errors (same low score pattern)
  1107→        if not drift_detected and tenant_scores:
  1108→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
  1109→            if len(low_scores) >= 3:
  1110→                # Check if same resolution_path failing
  1111→                paths: Dict[str, int] = {}
  1112→                for s in low_scores:
  1113→                    p = s.get("resolution_path", "unknown")
  1114→                    paths[p] = paths.get(p, 0) + 1
  1115→                worst_path = max(paths.items(), key=lambda x: x[1])
  1116→                if worst_path[1] >= 3:
  1117→                    drift_detected = True
  1118→                    drift_severity = "warning"
  1119→                    trend_direction = "declining"
  1120→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
  1121→
  1122→        return {
  1123→            "tenant_id": tenant_id,
  1124→            "drift_detected": drift_detected,
  1125→            "drift_severity": drift_severity,
  1126→            "accuracy_7d": accuracy_7d,
  1127→            "accuracy_yesterday": accuracy_yesterday,
  1128→            "accuracy_today": accuracy_today,
  1129→            "trend_direction": trend_direction,
  1130→            "trigger_reason": trigger_reason,
  1131→            "total_scores": len(tenant_scores),

  1399→
  1400→class SupabaseBackend(StorageBackend):
  1401→    """Supabase storage via REST API (PostgREST). Uses httpx.
  1402→
  1403→    Requires env vars: SUPABASE_URL, SUPABASE_ANON_KEY
  1404→    """
  1405→
  1406→    def __init__(self):
  1407→        if not SUPABASE_URL or not SUPABASE_ANON_KEY:
  1408→            raise ValueError("SUPABASE_URL and SUPABASE_ANON_KEY must be set")
  1409→        self._url = SUPABASE_URL.rstrip("/")
  1410→        self._headers = {
  1411→            "apikey": SUPABASE_ANON_KEY,
  1412→            "Authorization": f"Bearer {SUPABASE_ANON_KEY}",
  1413→            "Content-Type": "application/json",
  1414→            "Prefer": "return=representation",
  1415→        }
  1416→        self._client = None  # Lazy init on first use
  1417→
  1418→    async def _get_client(self):
  1419→        if self._client is None or self._client.is_closed:
  1420→            import httpx
  1421→            self._client = httpx.AsyncClient(
  1422→                base_url=self._url,
  1423→                headers=self._headers,
  1424→                timeout=15.0,
  1425→            )
  1426→        return self._client
  1427→
  1428→    async def _rest_get(self, table: str, query: str = "") -> List[Dict]:
  1429→        client = await self._get_client()
  1430→        url = f"/rest/v1/{table}"
  1431→        if query:
  1432→            url += f"?{query}"
  1433→        r = await client.get(url)
  1434→        r.raise_for_status()
  1435→        return r.json() if r.text else []
  1436→
  1437→    async def _rest_post(self, table: str, data: Dict) -> Dict:
  1438→        client = await self._get_client()
  1439→        r = await client.post(f"/rest/v1/{table}", json=data)
  1440→        r.raise_for_status()
  1441→        result = r.json()
  1442→        if isinstance(result, list) and result:
  1443→            return result[0]
  1444→        return result
  1445→
  1446→    async def _rest_patch(self, table: str, filter_query: str, data: Dict) -> List[Dict]:
  1447→        client = await self._get_client()
  1448→        r = await client.patch(f"/rest/v1/{table}?{filter_query}", json=data)
  1449→        r.raise_for_status()
  1450→        return r.json() if r.text else []
  1451→
  1452→    async def close(self):
  1453→        if self._client and not self._client.is_closed:
  1454→            await self._client.aclose()

  1803→                result["degraded_count"] += 1
  1804→            else:

  1806→            result["services"][svc] = {
  1807→                "status": status, "uptime_pct": uptime_pct,

  1879→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1880→        # Fetch scores + client-side drift analysis (same logic as InMemory)
  1881→        scores = await self._rest_get(
  1882→            TABLE_QUALITY_SCORES,
  1883→            f"tenant_id=eq.{tenant_id}&select=overall_score,resolution_path,created_at&order=created_at.asc&limit=5000"
  1884→        )
  1885→        if len(scores) < 3:
  1886→            return {"tenant_id": tenant_id, "drift_detected": False, "drift_severity": "none",
  1887→                    "accuracy_7d": None, "accuracy_yesterday": None, "accuracy_today": None,
  1888→                    "trend_direction": "stable", "trigger_reason": "insufficient_data", "total_scores": len(scores)}
  1889→        by_date: Dict[str, List[float]] = {}
  1890→        for s in scores:
  1891→            d = (s.get("created_at") or "")[:10]
  1892→            by_date.setdefault(d, []).append(float(s.get("overall_score", 0)))
  1893→        dates_sorted = sorted(by_date.keys())
  1894→        def _avg(vals): return round(sum(vals) / len(vals), 4) if vals else 0.0
  1895→        accuracy_7d = _avg([v for d in dates_sorted[-7:] for v in by_date[d]])
  1896→        accuracy_today = _avg(by_date[dates_sorted[-1]])
  1897→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1898→        drift_detected = False; drift_severity = "none"; trigger_reason = None; trend_direction = "stable"
  1899→        if len(dates_sorted) >= 3:
  1900→            daily_avgs = [_avg(by_date[d]) for d in dates_sorted[-3:]]
  1901→            declining = all(daily_avgs[i] > daily_avgs[i+1] for i in range(len(daily_avgs)-1))
  1902→            drop = accuracy_7d - accuracy_today
  1903→            if declining and drop > 0.05:
  1904→                drift_detected = True; trend_direction = "declining"
  1905→                trigger_reason = f"accuracy_dropped_{drop:.1%}_over_3_days"
  1906→                drift_severity = "critical" if drop > 0.10 else "warning"
  1907→            elif drop > 0.03:
  1908→                drift_detected = True; trend_direction = "slight_decline"
  1909→                trigger_reason = f"accuracy_dropped_{drop:.1%}_from_7d_avg"; drift_severity = "warning"
  1910→            elif accuracy_today > accuracy_7d + 0.03:
  1911→                trend_direction = "improving"
  1912→        return {"tenant_id": tenant_id, "drift_detected": drift_detected, "drift_severity": drift_severity,
  1913→                "accuracy_7d": accuracy_7d, "accuracy_yesterday": accuracy_yesterday,
  1914→                "accuracy_today": accuracy_today, "trend_direction": trend_direction,
  1915→                "trigger_reason": trigger_reason, "total_scores": len(scores)}

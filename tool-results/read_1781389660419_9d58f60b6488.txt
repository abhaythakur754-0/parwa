   120→   120→// ── Savings Calculation ──────────────────────────────────────────────────
   121→   121→
   122→   122→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   123→   123→  agentsReplaced: number;
   124→   124→  humanCost: number;
   125→   125→  savings: number;
   126→   126→  savingsPercent: number;
   127→   127→} {
   128→   128→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   129→   129→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   130→   130→  const savings = Math.max(0, humanCost - monthlyCost);
   131→   131→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   132→   132→  return { agentsReplaced, humanCost, savings, savingsPercent };
   133→   133→}

   242→   242→// ── Component ──────────────────────────────────────────────────────────
   243→   243→
   244→   244→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   245→   245→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   246→   246→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   247→   247→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   248→   248→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   249→   249→  const [isSubmitting, setIsSubmitting] = useState(false);
   250→   250→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   251→   251→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   252→   252→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   253→   253→
   254→   254→  // Restore add-ons and multi-variant selections from localStorage
   255→   255→  useEffect(() => {
   256→   256→    try {
   257→   257→      const stored = localStorage.getItem('parwa_pricing_context');
   258→   258→      if (stored) {
   259→   259→        const ctx = JSON.parse(stored) as PricingContext & {
   260→   260→          selectedVariants?: string[];
   261→   261→          variantQuantities?: Record<string, number>;
   262→   262→        };
   263→   263→        if (ctx.addOns) {
   264→   264→          setAddOns(ctx.addOns);
   265→   265→        }
   266→   266→        // Restore multiple variants if selected on Pricing page
   267→   267→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   268→   268→          const tiers: VariantTier[] = ctx.selectedVariants
   269→   269→            .map((v: string) => {
   270→   270→              const normalized = normalizeTier(v);
   271→   271→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   272→   272→            })
   273→   273→            .filter(Boolean) as VariantTier[];
   274→   274→          if (tiers.length > 0) {
   275→   275→            setActiveVariants(tiers);
   276→   276→          }
   277→   277→        }
   278→   278→        // Restore variant quantities
   279→   279→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   280→   280→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   281→   281→          const newQuantities = { ...variantQuantities };
   282→   282→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   283→   283→            const tier = normalizeTier(variantKey);
   284→   284→            if (['starter', 'growth', 'high'].includes(tier)) {
   285→   285→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   286→   286→            }
   287→   287→          }
   288→   288→          setVariantQuantities(newQuantities);
   289→   289→        }
   290→   290→      }
   291→   291→    } catch {
   292→   292→      // ignore
   293→   293→    }
   294→   294→  }, []);
   295→   295→
   296→   296→  // Check if Paddle.js is available
   297→   297→  useEffect(() => {
   298→   298→    getPaddleInstance().then((paddle) => {
   299→   299→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   300→   300→    }).catch(() => {
   301→   301→      setPaddleStatus('unavailable');
   302→   302→    });
   303→   303→  }, []);
   304→   304→
   305→   305→  // ── Calculations ──────────────────────────────────────────────────
   306→   306→
   307→   307→  const totalTicketLimit = useMemo(
   308→   308→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   309→   309→    [activeVariants, variantQuantities]
   310→   310→  );
   311→   311→
   312→   312→  const baseSubscription = useMemo(
   313→   313→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   314→   314→    [activeVariants, variantQuantities]
   315→   315→  );
   316→   316→
   317→   317→  const addOnTotal = useMemo(() => {
   318→   318→    let total = 0;
   319→   319→    if (addOns.voice) {
   320→   320→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   321→   321→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   322→   322→        total += voiceAddOn.price;
   323→   323→      }
   324→   324→    }
   325→   325→    if (addOns.customApi) {
   326→   326→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   327→   327→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   328→   328→        total += customApiAddOn.price;
   329→   329→      }
   330→   330→    }
   331→   331→    return total;
   332→   332→  }, [addOns, activeVariants]);
   333→   333→
   334→   334→  const totalMonthly = baseSubscription + addOnTotal;
   335→   335→
   336→   336→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   337→   337→
   338→   338→  const toggleVariant = (tier: VariantTier) => {
   339→   339→    setActiveVariants((prev) => {
   340→   340→      if (prev.includes(tier)) {
   341→   341→        if (prev.length <= 1) {
   342→   342→          toast.error('You must have at least one active variant');
   343→   343→          return prev;
   344→   344→        }
   345→   345→        return prev.filter((t) => t !== tier);
   346→   346→      }
   347→   347→      return [...prev, tier];
   348→   348→    });
   349→   349→  };
   350→   350→
   351→   351→  const updateQuantity = (tier: VariantTier, qty: number) => {
   352→   352→    const newQty = Math.max(1, Math.min(qty, 10));
   353→   353→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   354→   354→  };
   355→   355→
   356→   356→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   357→   357→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   358→   358→  };
   359→   359→
   360→   360→  // ── Payment success handler — ONLY called after Paddle confirms ──
   361→   361→  const handlePaymentSuccess = () => {
   362→   362→    setPaymentConfirmed(true);
   363→   363→    localStorage.removeItem('parwa_payment_pending');
   364→   364→    toast.success('Payment successful! Welcome to PARWA!');
   365→   365→    // Only trigger onboarding completion AFTER Paddle confirms
   366→   366→    onComplete();
   367→   367→  };
   368→   368→
   369→   369→  const handleProceed = async () => {
   370→   370→    setIsSubmitting(true);
   371→   371→    setCheckoutError(null);
   372→   372→
   373→   373→    try {
   374→   374→      // Update pricing context in localStorage
   375→   375→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   376→   376→      const context: PricingContext = {
   377→   377→        industry: 'other',
   378→   378→        variant: primaryOnboardingVariant,
   379→   379→        addOns,
   380→   380→        totalMonthly,
   381→   381→        timestamp: new Date().toISOString(),
   382→   382→      };
   383→   383→
   384→   384→      // Preserve industry from existing context
   385→   385→      try {
   386→   386→        const stored = localStorage.getItem('parwa_pricing_context');
   387→   387→        if (stored) {
   388→   388→          const existing = JSON.parse(stored) as PricingContext;
   389→   389→          context.industry = existing.industry;
   390→   390→        }
   391→   391→      } catch {
   392→   392→        // ignore
   393→   393→      }
   394→   394→
   395→   395→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   396→   396→
   397→   397→      // ── Paddle Checkout Flow ──────────────────────────────────────
   398→   398→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   399→   399→
   400→   400→      // Add each active variant as a Paddle line item (with quantity)
   401→   401→      for (const tier of activeVariants) {
   402→   402→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   403→   403→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   404→   404→        const qty = variantQuantities[tier] || 1;
   405→   405→        if (variantPriceId) {
   406→   406→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   407→   407→        }
   408→   408→      }
   409→   409→
   410→   410→      // Voice add-on is included in growth/high — NOT purchasable separately
   411→   411→      // (no Paddle price ID exists for standalone add-on purchases)
   412→   412→
   413→   413→      // Custom API add-on is included in growth/high — NOT purchasable separately
   414→   414→      // (no Paddle price ID exists for standalone add-on purchases)
   415→   415→
   416→   416→      const customData = {
   417→   417→        source: 'parwa_onboarding',
   418→   418→        variant: primaryOnboardingVariant,
   419→   419→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   420→   420→        variantQuantities,
   421→   421→        addOns,
   422→   422→        industry: context.industry,
   423→   423→        totalMonthly,
   424→   424→      };
   425→   425→
   426→   426→      // ── Step 1: Try server-side checkout (backend creates Paddle transaction) ──
   427→   427→      let checkoutUrl: string | null = null;
   428→   428→
   429→   429→      try {
   430→   430→        const res = await fetch('/api/onboarding/checkout', {
   431→   431→          method: 'POST',
   432→   432→          headers: { 'Content-Type': 'application/json' },
   433→   433→          body: JSON.stringify({
   434→   434→            variant: primaryOnboardingVariant,
   435→   435→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   436→   436→            variantQuantities,
   437→   437→            addOns,
   438→   438→            totalMonthly,
   439→   439→            industry: context.industry,
   440→   440→          }),
   441→   441→        });
   442→   442→
   443→   443→        if (res.ok) {
   444→   444→          const checkoutData = await res.json();
   445→   445→          checkoutUrl = checkoutData.checkout_url || null;
   446→   446→        }
   447→   447→      } catch {
   448→   448→        // API unavailable — continue with client-side Paddle checkout
   449→   449→      }
   450→   450→
   451→   451→      // If we got a checkout_url from backend, redirect to it
   452→   452→      if (checkoutUrl) {

   460→   460→
   461→   461→      // ── Step 2: Client-side Paddle checkout (items-based overlay) ──

   464→   464→      if (paddle && checkoutItems.length > 0) {

   467→   467→          checkoutItems,
   468→   468→          customData,

   477→   477→
   478→   478→        if (opened) {
   479→   479→          return; // Paddle overlay is showing — wait for user to complete or close
   480→   480→        }
   481→   481→      }
   482→   482→
   483→   483→      // ── Step 3: Paddle unavailable — try re-initializing ──
   484→   484→      try {
   485→   485→        const retryPaddle = await getPaddleInstance();
   486→   486→        if (retryPaddle && checkoutItems.length > 0) {
   487→   487→          setPaddleStatus('ready');
   488→   488→          const opened = await openCheckoutWithItems(
   489→   489→            checkoutItems,
   490→   490→            customData,
   491→   491→            handlePaymentSuccess,
   492→   492→            () => {
   493→   493→              toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   494→   494→              setIsSubmitting(false);
   495→   495→            },
   496→   496→          );
   497→   497→          if (opened) return;
   498→   498→        }
   499→   499→      } catch {
   500→   500→        // Paddle truly unavailable
   501→   501→      }
   502→   502→
   503→   503→      // ── Step 4: Paddle unavailable — block with error ──
   504→   504→      setPaddleStatus('unavailable');
   505→   505→      localStorage.setItem('parwa_payment_pending', JSON.stringify({
   506→   506→        activeVariants, addOns, totalMonthly, variantQuantities,
   507→   507→        pendingAt: new Date().toISOString(),
   508→   508→      }));
   509→   509→      setCheckoutError(
   510→   510→        'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   511→   511→        'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'

   514→   514→    } catch (err) {
   515→   515→      console.error('[cost-breakdown] Error:', err);
   516→   516→      setCheckoutError('Something went wrong. Please try again.');
   517→   517→      toast.error('Something went wrong. Please try again.');
   518→   518→    } finally {
   519→   519→      setIsSubmitting(false);
   520→   520→    }
   521→   521→  };
   522→   522→
   523→   523→  return (

   525→   525→      {/* Header */}
   526→   526→      <div className="text-center space-y-2">
   527→   527→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   528→   528→          <Receipt className="w-7 h-7 text-emerald-400" />
   529→   529→        </div>
   530→   530→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   531→   531→        <p className="text-orange-200/40 text-sm">
   532→   532→          Configure your plan and complete payment to activate your AI agents.
   533→   533→        </p>
   534→   534→      </div>
   535→   535→
   536→   536→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   537→   537→      {paddleStatus === 'unavailable' && (
   538→   538→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>

   540→   540→          <div>
   541→   541→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   542→   542→            <p className="text-[10px] text-orange-200/30 mt-1">

   551→   551→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>

   554→   554→      )}
   555→   555→
   556→   556→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}

   559→   559→          <Brain className="w-4 h-4 text-orange-400" />
   560→   560→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   561→   561→            Active Variants
   562→   562→          </label>
   563→   563→          <span className="text-[10px] text-orange-200/25">
   564→   564→            Each variant adds its own ticket allocation and AI pipeline
   565→   565→          </span>
   566→   566→        </div>

   568→   568→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (

   581→   581→      {/* ── 2. Add-Ons ───────────────────────────────────────────────── */}
   582→   582→      <div className="space-y-3">
   583→   583→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   584→   584→          Add-On Features
   585→   585→        </label>
   586→   586→        {ADD_ONS.map((addOn) => {

   601→   601→              style={!isIncluded ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   602→   602→            >
   603→   603→              <div className={cn(
   604→   604→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   605→   605→                isIncluded ? 'bg-emerald-500/10' : 'bg-white/[0.04]'
   606→   606→              )}>
   607→   607→                <Icon className={cn(
   608→   608→                  'w-5 h-5',
   609→   609→                  isIncluded ? 'text-emerald-400' : 'text-zinc-500'
   610→   610→                )} />
   611→   611→              </div>
   612→   612→              <div className="flex-1 min-w-0">
   613→   613→                <div className="flex items-center gap-2">
   614→   614→                  <p className={cn(
   615→   615→                    'text-sm font-medium',
   616→   616→                    isIncluded ? 'text-emerald-400' : 'text-white'
   617→   617→                  )}>
   618→   618→                    {addOn.name}
   619→   619→                  </p>
   620→   620→                  {isIncluded && (
   621→   621→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   622→   622→                      Included
   623→   623→                    </span>
   624→   624→                  )}
   625→   625→                </div>
   626→   626→                <p className="text-[10px] text-orange-200/30 mt-0.5">
   627→   627→                  {isIncluded ? addOn.description : `Included with PARWA and PARWA High plans`}
   628→   628→                </p>
   629→   629→              </div>
   630→   630→              <div className="flex flex-col items-end shrink-0 gap-1">
   631→   631→                {isIncluded ? (

   637→   637→                )}
   638→   638→              </div>
   639→   639→            </div>
   640→   640→          );
   641→   641→        })}
   642→   642→      </div>

   644→   644→      {/* ── 3. Cost Summary ────────────────────────────────────────────── */}
   645→   645→      <div
   646→   646→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   647→   647→        style={{ background: 'rgba(255,255,255,0.03)' }}
   648→   648→      >
   649→   649→        {/* Per-variant breakdown */}
   650→   650→        {activeVariants.map((tier) => {
   651→   651→          const qty = variantQuantities[tier] || 1;
   652→   652→          const price = VARIANT_PRICES[tier] * qty;
   653→   653→
   654→   654→          return (
   655→   655→            <div key={tier} className="flex items-center justify-between">
   656→   656→              <span className="text-sm text-orange-200/50 flex items-center gap-1.5">
   657→   657→                <div className={cn(
   658→   658→                  'w-2 h-2 rounded-full',
   659→   659→                  tier === 'starter' ? 'bg-emerald-400' : tier === 'growth' ? 'bg-orange-400' : 'bg-purple-400'
   660→   660→                )} />

   664→   664→              <span className="text-sm text-white">
   665→   665→                ${price.toLocaleString()}/mo

   675→   675→          <span className="text-sm text-orange-200/50">Integrations</span>

   677→   677→        </div>
   678→   678→
   679→   679→        <div className="border-t border-white/[0.06] pt-3 flex items-center justify-between">
   680→   680→          <span className="text-sm font-semibold text-white">Total Monthly</span>

   696→   696→        style={{ background: 'rgba(16,185,129,0.04)' }}

   773→   773→}

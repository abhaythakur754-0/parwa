   224→    integration_id: str,
   225→    user: User = Depends(get_current_user),
   226→    db: Session = Depends(get_db),
   227→) -> MessageResponse:
   228→    """Delete an integration.
   229→
   230→    BC-001: Scoped to user's company_id.
   231→    Phase 7: Cache is invalidated when integration is disconnected.
   232→    """
   233→    # Get integration type before deletion for cache invalidation
   234→    service = IntegrationService(db)
   235→    integration = service.get_integration(
   236→        integration_id=integration_id,
   237→        company_id=user.company_id,
   238→    )
   239→    integration_type = integration.integration_type if integration else None
   240→
   241→    service.delete_integration(
   242→        integration_id=integration_id,
   243→        company_id=user.company_id,
   244→    )
   245→
   246→    # Phase 7: Invalidate all cached data for this integration
   247→    if integration_type:
   248→        try:
   249→            from app.services.integration_cache_service import IntegrationCacheService
   250→            cache_svc = IntegrationCacheService(company_id=user.company_id)
   251→            await cache_svc.invalidate_on_disconnect(integration_type)
   252→        except Exception:
   253→            pass  # BC-012: Cache invalidation failure must not break deletion
   254→
   255→    # Phase 9: Audit log for integration disconnect
   256→    try:
   257→        log_audit(
   258→            company_id=str(user.company_id),
   259→            actor_id=str(user.id),
   260→            actor_type=ActorType.USER.value,
   261→            action=AuditAction.INTEGRATION_DISCONNECT.value,
   262→            resource_type="integration",
   263→            resource_id=integration_id,
   264→            new_value=f"Disconnected {integration_type or 'unknown'}",
   265→            db=db,
   266→        )
   267→    except Exception:
   268→        pass  # BC-012: Audit logging must not break the main operation
   269→
   270→    return MessageResponse(message="Integration deleted successfully.")

   273→@router.post(
   274→    "/industry-change-impact",
   275→    response_model=IndustryChangeImpactResponse,
   276→)
   277→def api_industry_change_impact(
   278→    body: IndustryChangeImpactRequest,
   279→    user: User = Depends(get_current_user),
   280→    db: Session = Depends(get_db),
   281→) -> IndustryChangeImpactResponse:
   282→    """Check the impact of changing industry on existing integrations.
   283→
   284→    Per GAP 10: Industry change NEVER disconnects integrations.
   285→    Existing connections STAY connected. Only the suggestion filter changes.
   286→    Clients can always connect tools outside their industry.
   287→
   288→    Returns:
   289→    - still_recommended: integrations that are suggested in BOTH industries
   290→    - no_longer_suggested: connected integrations not in new industry's suggestions
   291→    - newly_suggested: integrations newly suggested for the new industry
   292→    """
   293→    # Get suggested integration keys for both industries
   294→    current_suggested = {i["key"] for i in get_catalog_for_industry(body.current_industry)}
   295→    new_suggested = {i["key"] for i in get_catalog_for_industry(body.new_industry)}
   296→
   297→    # Get currently connected integrations
   298→    service = IntegrationService(db)
   299→    connected = service.get_integrations(company_id=user.company_id, active_only=True)
   300→    connected_keys = {i["type"] for i in connected}
   301→
   302→    # Categorize
   303→    still_recommended = sorted(connected_keys & new_suggested)
   304→    no_longer_suggested = sorted(connected_keys - new_suggested)
   305→    newly_suggested = sorted((new_suggested - current_suggested) - connected_keys)
   306→
   307→    # Build connected integration details
   308→    connected_details = []
   309→    for i in connected:
   310→        catalog_entry = get_integration_by_key(i["type"])
   311→        in_new_industry = i["type"] in new_suggested
   312→        connected_details.append({
   313→            "key": i["type"],
   314→            "name": i["name"],
   315→            "status": i["status"],
   316→            "category": catalog_entry.category.value if catalog_entry else "custom",
   317→            "in_new_industry_suggestions": in_new_industry,
   318→        })
   319→
   320→    # Build message
   321→    if no_longer_suggested:
   322→        msg = (
   323→            f"You're changing from {body.current_industry} to {body.new_industry}. "
   324→            f"{len(no_longer_suggested)} integration(s) will still work but are no longer suggested for {body.new_industry}. "
   325→            f"Your tickets, knowledge base, and billing are NOT affected."
   326→        )
   327→    else:
   328→        msg = (
   329→            f"Changing from {body.current_industry} to {body.new_industry} will not affect any of your "
   330→            f"connected integrations. {len(newly_suggested)} new integrations are suggested for {body.new_industry}."
   331→        )
   332→
   333→    return IndustryChangeImpactResponse(
   334→        new_industry=body.new_industry,
   335→        current_industry=body.current_industry,
   336→        connected_integrations=connected_details,
   337→        still_recommended=still_recommended,
   338→        no_longer_suggested=no_longer_suggested,
   339→        newly_suggested=newly_suggested,
   340→        message=msg,
   341→    )

   347→@router.get(
   348→    "/health",
   349→    response_model=Dict[str, Any],
   350→)
   351→def api_integration_health(
   352→    user: User = Depends(get_current_user),
   353→    db: Session = Depends(get_db),
   354→) -> Dict[str, Any]:
   355→    """Get health status of all integrations for the company.
   356→
   357→    Returns: circuit breaker states, rate limit usage, last test time,
   358→    connected status for each integration.
   359→
   360→    BC-001: Scoped to user's company_id.
   361→    """
   362→    result: Dict[str, Any] = {
   363→        "company_id": user.company_id,
   364→        "timestamp": datetime.now(timezone.utc).isoformat(),
   365→        "integrations": {},
   366→    }
   367→
   368→    # Get connected integrations
   369→    try:
   370→        service = IntegrationService(db)
   371→        integrations = service.get_integrations(company_id=user.company_id)
   372→    except Exception:
   373→        integrations = []
   374→
   375→    # Get circuit breaker states
   376→    cb_states = {}
   377→    try:
   378→        from app.core.parwa_core_bridge import get_parwa_circuit_breaker as get_circuit_breaker_manager
   379→        cb_manager = get_circuit_breaker_manager()
   380→        cb_states = cb_manager.get_all_states()
   381→    except Exception:
   382→        pass
   383→
   384→    # Get rate limit status
   385→    rate_limit_status = {}
   386→    try:
   387→        from app.core.integration_rate_limiter import get_integration_rate_limiter
   388→        rate_limiter = get_integration_rate_limiter()
   389→        rate_limit_status = rate_limiter.get_all_status(user.company_id)
   390→    except Exception:
   391→        pass
   392→
   393→    # Get provider status from ExternalToolBus
   394→    provider_status = {}
   395→    try:
   396→        from app.core.external_tool_bus import external_tool_bus
   397→        provider_status = external_tool_bus.get_provider_status()
   398→    except Exception:
   399→        pass
   400→
   401→    # Get disconnect handler status
   402→    disconnect_handler = None
   403→    try:
   404→        from app.core.integration_disconnect_handler import get_integration_disconnect_handler
   405→        disconnect_handler = get_integration_disconnect_handler()
   406→    except Exception:
   407→        pass
   408→
   409→    # Build per-integration health
   410→    for integration in integrations:
   411→        intg_type = integration.get("type", "unknown")
   412→        intg_id = integration.get("id", "")
   413→
   414→        health_entry: Dict[str, Any] = {
   415→            "id": intg_id,
   416→            "type": intg_type,
   417→            "name": integration.get("name", ""),
   418→            "status": integration.get("status", "unknown"),
   419→            "connected": True,
   420→            "last_test_at": integration.get("last_test_at"),
   421→            "last_test_result": integration.get("last_test_result"),
   422→        }
   423→
   424→        # Circuit breaker state
   425→        if intg_type in cb_states:
   426→            health_entry["circuit_breaker"] = cb_states[intg_type]
   427→        else:
   428→            health_entry["circuit_breaker"] = {"state": "closed", "is_available": True}
   429→
   430→        # Rate limit status
   431→        if intg_type in rate_limit_status:
   432→            health_entry["rate_limit"] = rate_limit_status[intg_type]
   433→        else:
   434→            health_entry["rate_limit"] = {"is_limited": False}
   435→
   436→        # Disconnect status
   437→        if disconnect_handler:
   438→            health_entry["connected"] = disconnect_handler.is_integration_connected(
   439→                user.company_id, intg_id,
   440→            )
   441→            disconnect_record = disconnect_handler.get_disconnect_status(
   442→                user.company_id, intg_id,
   443→            )
   444→            if disconnect_record:
   445→                health_entry["disconnect_record"] = disconnect_record
   446→
   447→        result["integrations"][intg_id] = health_entry
   448→
   449→    # Add overall circuit breaker health summary
   450→    try:
   451→        from app.core.parwa_core_bridge import get_parwa_circuit_breaker as get_circuit_breaker_manager
   452→        cb_manager = get_circuit_breaker_manager()
   453→        result["circuit_breaker_health"] = cb_manager.get_health_summary()
   454→    except Exception:
   455→        result["circuit_breaker_health"] = {"status": "unknown"}
   456→
   457→    # Add provider statuses
   458→    result["providers"] = provider_status
   459→
   460→    return result

   468→@router.post(
   469→    "/{integration_id}/disconnect",
   470→    response_model=Dict[str, Any],
   471→)
   472→async def api_disconnect_integration(
   473→    integration_id: str,
   474→    body: DisconnectIntegrationRequest | None = None,
   475→    user: User = Depends(get_current_user),
   476→    db: Session = Depends(get_db),
   477→) -> Dict[str, Any]:
   478→    """Properly disconnect an integration with full cleanup.
   479→
   480→    Phase 10: Performs instant cleanup including:
   481→    - Stops all pending API calls
   482→    - Invalidates cached data
   483→    - Removes rate limit slots
   484→    - Opens the circuit breaker
   485→    - Notifies AI pipeline
   486→
   487→    BC-001: Scoped to user's company_id.
   488→    """
   489→    # Get integration details before disconnect
   490→    service = IntegrationService(db)
   491→    try:
   492→        integration = service.get_integration(
   493→            integration_id=integration_id,
   494→            company_id=user.company_id,
   495→        )
   496→        integration_name = integration.integration_type if integration else ""
   497→    except Exception:
   498→        integration_name = ""
   499→
   500→    # Use the disconnect handler for full cleanup
   501→    try:
   502→        from app.core.integration_disconnect_handler import get_integration_disconnect_handler
   503→        handler = get_integration_disconnect_handler()
   504→        result = handler.disconnect_integration(
   505→            company_id=user.company_id,
   506→            integration_id=integration_id,
   507→            integration_name=integration_name,
   508→            reason=body.reason if body else "user_action",
   509→        )
   510→    except Exception as exc:
   511→        result = {
   512→            "company_id": user.company_id,
   513→            "integration_id": integration_id,
   514→            "success": False,
   515→            "error": f"Disconnect handler failed: {str(exc)[:200]}",
   516→            "cleanup_steps": [],
   517→        }
   518→
   519→    # Update integration status in DB
   520→    try:
   521→        service.update_integration_status(
   522→            integration_id=integration_id,
   523→            company_id=user.company_id,
   524→            status="disconnected",
   525→        )
   526→    except Exception:
   527→        pass  # BC-008: Status update failure must not break disconnect
   528→
   529→    # Phase 7: Invalidate cache
   530→    if integration_name:
   531→        try:
   532→            from app.services.integration_cache_service import IntegrationCacheService
   533→            cache_svc = IntegrationCacheService(company_id=user.company_id)
   534→            await cache_svc.invalidate_on_disconnect(integration_name)
   535→        except Exception:
   536→            pass  # BC-012: Cache invalidation failure must not break disconnect
   537→
   538→    # Audit log
   539→    try:
   540→        log_audit(
   541→            company_id=str(user.company_id),
   542→            actor_id=str(user.id),
   543→            actor_type=ActorType.USER.value,
   544→            action=AuditAction.INTEGRATION_DISCONNECT.value,
   545→            resource_type="integration",
   546→            resource_id=integration_id,
   547→            new_value=f"Disconnected {integration_name or 'unknown'}: {body.reason if body else 'user_action'}",
   548→            db=db,
   549→        )
   550→    except Exception:
   551→        pass  # BC-012: Audit logging must not break the main operation
   552→
   553→    return result

   984→) -> Dict[str, Any]:
   985→    """Send a test event to an outbound webhook endpoint.
   986→
   987→    Per Phase 6: Verifies the endpoint is reachable and accepts payloads.
   988→    Sends a ping event with HMAC signature.
   989→    BC-001: Scoped to user's company_id.
   990→    """
   991→    import hmac
   992→    import hashlib
   993→    import json as json_lib
   994→    import httpx
   995→
   996→    from database.models.outbound_webhook import OutboundWebhook
   997→
   998→    webhook = (
   999→        db.query(OutboundWebhook)
  1000→        .filter(
  1001→            OutboundWebhook.id == webhook_id,
  1002→            OutboundWebhook.company_id == user.company_id,
  1003→        )
  1004→        .first()
  1005→    )
  1006→    if not webhook:
  1007→        from app.exceptions import NotFoundError
  1008→        raise NotFoundError(message="Webhook not found")
  1009→
  1010→    # Build test payload
  1011→    test_payload = {
  1012→        "event": "webhook.test",
  1013→        "webhook_id": webhook.id,
  1014→        "timestamp": datetime.now(timezone.utc).isoformat(),
  1015→        "data": {
  1016→            "message": "This is a test event from PARWA",
  1017→            "company_id": user.company_id,
  1018→        },
  1019→    }
  1020→
  1021→    payload_str = json_lib.dumps(test_payload, separators=(",", ":"))
  1022→    signature = hmac.new(
  1023→        webhook.secret.encode("utf-8"),
  1024→        payload_str.encode("utf-8"),
  1025→        hashlib.sha256,
  1026→    ).hexdigest()
  1027→
  1028→    headers = {
  1029→        "Content-Type": "application/json",
  1030→        "X-Parwa-Signature": f"sha256={signature}",

  1032→        "X-Parwa-Delivery": str(uuid.uuid4()),
  1033→    }
  1034→
  1035→    try:
  1036→        with httpx.Client(timeout=10.0) as client:
  1037→            resp = client.post(webhook.url, content=payload_str, headers=headers)
  1038→        success = 200 <= resp.status_code < 300
  1039→        result = {
  1040→            "success": success,
  1041→            "status_code": resp.status_code,

  1062→def _register_webhook_with_providers(webhook, company_id: str, db: Session) -> None:
  1063→    """Register the webhook URL with third-party providers.
  1064→
  1065→    Per Phase 6: When client connects a tool, Parwa registers the webhook
  1066→    URL with that provider so Parwa receives real-time events.
  1067→
  1068→    This is a best-effort operation — if registration fails, the webhook
  1069→    still works for PARWA-originated events. Provider registration enables
  1070→    incoming events from the third party.
  1071→    """
  1072→    import logging
  1073→    logger = logging.getLogger("parwa.webhooks.registration")
  1074→
  1075→    # Get connected integrations for this company
  1076→    try:
  1077→        service = IntegrationService(db)
  1078→        integrations = service.get_integrations(company_id=company_id, active_only=True)
  1079→    except Exception:
  1080→        integrations = []
  1081→
  1082→    for integration in integrations:
  1083→        integration_type = integration.get("type", "")
  1084→        try:
  1085→            if integration_type == "shopify":
  1086→                # Shopify webhook registration
  1087→                _register_shopify_webhook(webhook, integration)
  1088→            elif integration_type in ("hubspot", "salesforce"):
  1089→                # CRM webhook registration (contact.created, deal.updated)
  1090→                _register_crm_webhook(webhook, integration)
  1091→            # Other providers can be added here
  1092→        except Exception as e:
  1093→            logger.warning(
  1094→                f"Failed to register webhook with {integration_type}: {e}",
  1095→                extra={"webhook_id": webhook.id, "integration_type": integration_type},

  1099→def _register_shopify_webhook(webhook, integration) -> None:
  1100→    """Register webhook URL with Shopify for order events."""
  1101→    import httpx
  1102→    config = integration.get("config", {})
  1103→    shop_domain = config.get("shop_domain", "")
  1104→    access_token = config.get("access_token", "")
  1105→    if not shop_domain or not access_token:
  1106→        return
  1107→
  1108→    # Map PARWA events to Shopify webhook topics
  1109→    event_mapping = {
  1110→        "ticket.created": "orders/create",
  1111→        "ticket.resolved": "orders/fulfilled",
  1112→        "integration.error": "app/uninstalled",
  1113→    }
  1114→
  1115→    with httpx.Client(timeout=10.0) as client:
  1116→        for parwa_event in (webhook.events or []):
  1117→            shopify_topic = event_mapping.get(parwa_event)
  1118→            if not shopify_topic:
  1119→                continue
  1120→            try:
  1121→                client.post(
  1122→                    f"https://{shop_domain}/admin/api/2024-01/webhooks.json",
  1123→                    headers={"X-Shopify-Access-Token": access_token, "Content-Type": "application/json"},
  1124→                    json={
  1125→                        "webhook": {
  1126→                            "topic": shopify_topic,
  1127→                            "address": webhook.url,
  1128→                            "format": "json",
  1129→                        }
  1130→                    },
  1131→                )
  1132→            except Exception:
  1133→                pass  # Best-effort

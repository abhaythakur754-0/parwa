    60→    60→function fmt(n: number): string {
    61→    61→  return n.toLocaleString('en-US', {
    62→    62→    style: 'currency',
    63→    63→    currency: 'USD',
    64→    64→    maximumFractionDigits: 0,
    65→    65→  });
    66→    66→}
    67→    67→
    68→    68→function fmtNum(n: number): string {
    69→    69→  return n.toLocaleString('en-US', { maximumFractionDigits: 0 });
    70→    70→}
    71→    71→
    72→    72→function fmtPct(n: number): string {
    73→    73→  return `${Math.round(n)}%`;
    74→    74→}

    78→    78→function UsageBar({
    79→    79→  used,
    80→    80→  total,
    81→    81→  label,
    82→    82→  unit,
    83→    83→  color,
    84→    84→  overageRate,
    85→    85→}: {
    86→    86→  used: number;
    87→    87→  total: number;
    88→    88→  label: string;
    89→    89→  unit: string;
    90→    90→  color: string;
    91→    91→  overageRate?: string;
    92→    92→}) {
    93→    93→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
    94→    94→  const isNearLimit = pct >= 80;
    95→    95→  const isOverLimit = used > total;
    96→    96→
    97→    97→  return (
    98→    98→    <div className="space-y-2">
    99→    99→      <div className="flex items-center justify-between">
   100→   100→        <span className="text-xs text-zinc-400">{label}</span>
   101→   101→        <span
   102→   102→          className={cn(
   103→   103→            'text-xs font-medium',
   104→   104→            isOverLimit ? 'text-red-400' : isNearLimit ? 'text-amber-400' : 'text-zinc-300'
   105→   105→          )}
   106→   106→        >
   107→   107→          {fmtNum(used)} / {fmtNum(total)} {unit}
   108→   108→        </span>
   109→   109→      </div>
   110→   110→      <div className="h-2 bg-white/5 rounded-full overflow-hidden relative">
   111→   111→        <div className={cn('h-full rounded-full transition-all duration-700', color)} style={{ width: `${pct}%` }} />
   112→   112→        {isOverLimit && (
   113→   113→          <div
   114→   114→            className="absolute top-0 h-full bg-red-500/30 rounded-r-full"
   115→   115→            style={{ left: '100%', width: `${Math.min(((used - total) / total) * 100, 50)}%` }}
   116→   116→          />
   117→   117→        )}
   118→   118→      </div>
   119→   119→      {isOverLimit && overageRate && (
   120→   120→        <p className="text-[10px] text-red-400 flex items-center gap-1">
   121→   121→          <AlertTriangle className="w-3 h-3" />
   122→   122→          Over limit — {fmtNum(used - total)} overage tickets at {overageRate}/ticket
   123→   123→        </p>
   124→   124→      )}
   125→   125→      {isNearLimit && !isOverLimit && (
   126→   126→        <p className="text-[10px] text-amber-400 flex items-center gap-1">
   127→   127→          <AlertTriangle className="w-3 h-3" />
   128→   128→          Approaching limit — {fmtPct(pct)} used
   129→   129→        </p>
   130→   130→      )}
   131→   131→    </div>
   132→   132→  );
   133→   133→}

   267→   267→export default function CostBreakdownPage() {
   268→   268→  const [isLoading, setIsLoading] = useState(true);
   269→   269→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>(['starter']);
   270→   270→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({
   271→   271→    voice: false,
   272→   272→    customApi: false,
   273→   273→  });
   274→   274→  const [billingCycle, setBillingCycle] = useState<BillingCycle>('monthly');
   275→   275→  const [usageData, setUsageData] = useState<UsageData>({
   276→   276→    ticketsUsed: 0,
   277→   277→    agentsUsed: 0,
   278→   278→    docsUsed: 0,
   279→   279→  });
   280→   280→
   281→   281→  // Fetch usage data on mount
   282→   282→  useEffect(() => {
   283→   283→    const fetchData = async () => {
   284→   284→      try {
   285→   285→        // Try to get current subscription tier and usage
   286→   286→        const subRes = await fetch('/api/billing/subscription', {
   287→   287→          credentials: 'include',
   288→   288→          headers: { 'Content-Type': 'application/json' },
   289→   289→        });
   290→   290→        if (subRes.ok) {
   291→   291→          const subData = await subRes.json();
   292→   292→          const tier = normalizeTier(subData.variant_tier || subData.tier || 'mini');
   293→   293→          setActiveVariants([tier]);
   294→   294→        }
   295→   295→
   296→   296→        // Try to get usage
   297→   297→        const usageRes = await fetch('/api/billing/usage', {
   298→   298→          credentials: 'include',
   299→   299→          headers: { 'Content-Type': 'application/json' },
   300→   300→        });
   301→   301→        if (usageRes.ok) {
   302→   302→          const data = await usageRes.json();
   303→   303→          setUsageData({
   304→   304→            ticketsUsed: Number(data.tickets_used ?? data.ticketsUsed ?? 0),
   305→   305→            agentsUsed: Number(data.agents_used ?? data.agentsUsed ?? 0),
   306→   306→            docsUsed: Number(data.docs_used ?? data.docsUsed ?? 0),
   307→   307→          });
   308→   308→        }
   309→   309→
   310→   310→        // Try localStorage for onboarding context
   311→   311→        try {
   312→   312→          const stored = localStorage.getItem('parwa_pricing_context');
   313→   313→          if (stored) {
   314→   314→            const ctx = JSON.parse(stored);
   315→   315→            if (ctx.variant) {
   316→   316→              const tier = normalizeTier(ctx.variant);
   317→   317→              if (!activeVariants.includes(tier)) {
   318→   318→                setActiveVariants([tier]);
   319→   319→              }
   320→   320→            }
   321→   321→            if (ctx.addOns) {
   322→   322→              setAddOns(ctx.addOns);
   323→   323→            }
   324→   324→          }
   325→   325→        } catch {
   326→   326→          // ignore
   327→   327→        }
   328→   328→      } catch {
   329→   329→        // Use defaults
   330→   330→      } finally {
   331→   331→        setIsLoading(false);
   332→   332→      }
   333→   333→    };
   334→   334→
   335→   335→    fetchData();
   336→   336→  }, []);
   337→   337→
   338→   338→  // Toggle variant
   339→   339→  const toggleVariant = useCallback((tier: VariantTier) => {
   340→   340→    setActiveVariants((prev) => {
   341→   341→      if (prev.includes(tier)) {

   344→   344→          toast.error('You must have at least one active variant');
   345→   345→          return prev;
   346→   346→        }
   347→   347→        return prev.filter((t) => t !== tier);
   348→   348→      }
   349→   349→      return [...prev, tier];
   350→   350→    });
   351→   351→  }, []);
   352→   352→
   353→   353→  // Toggle add-on
   354→   354→  const toggleAddOn = useCallback((key: 'voice' | 'customApi') => {
   355→   355→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   356→   356→  }, []);
   357→   357→
   358→   358→  // Calculated cost breakdown
   359→   359→  const costBreakdown = useMemo(
   360→   360→    () => calculateCostBreakdown(activeVariants, addOns, usageData.ticketsUsed),
   361→   361→    [activeVariants, addOns, usageData.ticketsUsed]
   362→   362→  );
   363→   363→
   364→   364→  // What-if previews for tiers not currently active

   367→   367→    return allTiers
   368→   368→      .filter((tier) => !activeVariants.includes(tier))

   370→   370→        tier,
   371→   371→        preview: calculateWhatIfUpgrade(activeVariants, addOns, usageData.ticketsUsed, tier),

   373→   373→  }, [activeVariants, addOns, usageData.ticketsUsed]);
   374→   374→
   375→   375→  // Annual calculation
   376→   376→  const annualTotal = costBreakdown.totalMonthly * 12;
   377→   377→
   378→   378→  // Active price for display
   379→   379→  const displayTotal = billingCycle === 'annual' ? annualTotal / 12 : costBreakdown.totalMonthly;
   380→   380→
   381→   381→  // ── Loading State ───────────────────────────────────────────────
   382→   382→  if (isLoading) {
   383→   383→    return (
   384→   384→      <div className="space-y-6">
   385→   385→        <div className="pb-6 border-b border-white/[0.06]">
   386→   386→          <div className="h-6 w-40 bg-white/5 rounded animate-pulse" />
   387→   387→          <div className="h-4 w-72 bg-white/5 rounded mt-2 animate-pulse" />
   388→   388→        </div>
   389→   389→        <div className="h-64 bg-white/5 rounded-xl animate-pulse" />
   390→   390→        <div className="h-48 bg-white/5 rounded-xl animate-pulse" />
   391→   391→      </div>
   392→   392→    );
   393→   393→  }
   394→   394→
   395→   395→  // ── Render ──────────────────────────────────────────────────────
   396→   396→
   397→   397→  return (
   398→   398→    <div className="space-y-6">
   399→   399→      {/* ─── Header ─────────────────────────────────────────────── */}
   400→   400→      <div className="flex items-center justify-between pb-6 border-b border-white/[0.06]">
   401→   401→        <div>
   402→   402→          <h1 className="text-xl font-bold text-white flex items-center gap-2">
   403→   403→            <Calculator className="w-5 h-5 text-orange-400" />
   404→   404→            Cost Breakdown
   405→   405→          </h1>
   406→   406→          <p className="text-sm text-zinc-500 mt-0.5">
   407→   407→            Pure math. No hidden fees. Need more? Add another variant.
   408→   408→          </p>
   409→   409→        </div>
   410→   410→        <div className="flex items-center gap-3">
   411→   411→          {/* Billing cycle toggle */}
   412→   412→          <div className="flex items-center gap-2 bg-white/[0.03] border border-white/[0.06] rounded-lg px-3 py-1.5">
   413→   413→            <span
   414→   414→              className={cn('text-xs font-medium', billingCycle === 'monthly' ? 'text-white' : 'text-zinc-500')}
   415→   415→            >
   416→   416→              Monthly
   417→   417→            </span>
   418→   418→            <button
   419→   419→              onClick={() => setBillingCycle((prev) => (prev === 'monthly' ? 'annual' : 'monthly'))}
   420→   420→              className={cn(
   421→   421→                'relative w-9 h-5 rounded-full transition-colors',

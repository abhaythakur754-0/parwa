     9→def create_or_resume_session(
    10→    db: Session,
    11→    user_id: str,
    12→    company_id: Optional[str] = None,
    13→    entry_source: str = "direct",
    14→    entry_params: Optional[Dict[str, Any]] = None,
    15→) -> JarvisSession:
    16→    """Create a new onboarding session or resume an active one.
    17→
    18→    Looks for an active onboarding session for this user.
    19→    If found and from today, resumes it. Otherwise creates new.
    20→    """
    21→    # Try to find active session
    22→    active_session = (
    23→        db.query(JarvisSession)
    24→        .filter(
    25→            JarvisSession.user_id == user_id,
    26→            JarvisSession.is_active.is_(True),
    27→            JarvisSession.type == "onboarding",
    28→        )
    29→        .order_by(JarvisSession.created_at.desc())
    30→        .first()
    31→    )
    32→
    33→    if active_session:
    34→        # Reset daily counter if new day
    35→        _maybe_reset_daily_counter(db, active_session)
    36→        active_session.updated_at = datetime.now(timezone.utc)
    37→        db.flush()
    38→
    39→        # Update entry context if provided
    40→        if entry_source and entry_source != "direct":
    41→            ctx = _parse_context(active_session.context_json)
    42→            ctx["entry_source"] = entry_source
    43→            if entry_params:
    44→                ctx["entry_params"] = entry_params
    45→            active_session.context_json = json.dumps(ctx)
    46→
    47→        return active_session
    48→
    49→    # Create new session
    50→    # Create new session
    51→    ctx = {
    52→        "pages_visited": [],
    53→        "industry": None,
    54→        "selected_variants": [],
    55→        "roi_result": None,
    56→        "demo_topics": [],
    57→        "concerns_raised": [],
    58→        "business_email": None,
    59→        "email_verified": False,
    60→        "referral_source": "",
    61→        "entry_source": entry_source,
    62→        "entry_params": entry_params or {},
    63→        "detected_stage": "welcome",
    64→    }
    65→
    66→    # PROMOTE entry_params to top-level context for immediate awareness in welcome messages
    67→    if entry_params:
    68→        if "industry" in entry_params:
    69→            ctx["industry"] = entry_params["industry"]
    70→        if "roi_result" in entry_params:
    71→            ctx["roi_result"] = entry_params["roi_result"]
    72→        if "variant" in entry_params:
    73→            ctx["variant"] = entry_params["variant"]
    74→        if "variant_id" in entry_params:
    75→            ctx["variant_id"] = entry_params["variant_id"]
    76→
    77→    session = JarvisSession(
    78→        user_id=user_id,
    79→        company_id=company_id,
    80→        type="onboarding",
    81→        context_json=json.dumps(ctx),
    82→        message_count_today=0,
    83→        total_message_count=0,
    84→        pack_type="free",
    85→        is_active=True,
    86→    )
    87→    db.add(session)
    88→    db.flush()
    89→
    90→    # CRITICAL: Generate the context-aware welcome message immediately
    91→    # so it's ready in the session history response.
    92→    welcome_text = build_context_aware_welcome(db, str(session.id))
    93→    welcome_msg = JarvisMessage(
    94→        session_id=str(session.id),
    95→        role="jarvis",
    96→        content=welcome_text,
    97→        message_type="text",
    98→        metadata_json=json.dumps({}),
    99→    )
   100→    db.add(welcome_msg)
   101→    db.flush()
   102→
   103→    return session

   106→def get_session(
   107→    db: Session,
   108→    session_id: str,
   109→    user_id: str,
   110→) -> JarvisSession:
   111→    """Get session by ID, scoped to user for security."""
   112→    session = (
   113→        db.query(JarvisSession)
   114→        .filter(
   115→            JarvisSession.id == session_id,
   116→            JarvisSession.user_id == user_id,
   117→        )
   118→        .first()
   119→    )
   120→    if not session:
   121→        raise NotFoundError(
   122→            message="Session not found",
   123→            details={"session_id": session_id},
   124→        )
   125→    return session

   128→def get_session_context(
   129→    db: Session,
   130→    session_id: str,
   131→) -> Dict[str, Any]:
   132→    """Get context_json for AI prompt injection."""
   133→    session = db.query(JarvisSession).filter(
   134→        JarvisSession.id == session_id,
   135→    ).first()
   136→    if not session:
   137→        raise NotFoundError(message="Session not found")
   138→    return _parse_context(session.context_json)

   141→def update_context(
   142→    db: Session,
   143→    session_id: str,
   144→    user_id: str,
   145→    partial_updates: Dict[str, Any],
   146→) -> JarvisSession:
   147→    """Merge partial updates into session context_json.
   148→
   149→    Only provided keys are updated. Existing keys are preserved.
   150→    """
   151→    session = get_session(db, session_id, user_id)
   152→    ctx = _parse_context(session.context_json)
   153→
   154→    for key, value in partial_updates.items():
   155→        if value is not None:
   156→            ctx[key] = value
   157→
   158→    session.context_json = json.dumps(ctx)
   159→    session.updated_at = datetime.now(timezone.utc)
   160→    db.flush()
   161→    return session

   182→def send_message(
   183→    db: Session,
   184→    session_id: str,
   185→    user_id: str,
   186→    user_message: str,
   187→) -> Tuple[JarvisMessage, JarvisMessage, List[Dict[str, Any]]]:
   188→    """Process a user message and generate AI response.
   189→
   190→    Flow:
   191→    1. Save user message
   192→    2. Check message limits
   193→    3. Build system prompt with context
   194→    4. Call AI provider
   195→    5. Save AI response + knowledge used
   196→    6. Detect conversation stage
   197→    7. Return both messages
   198→
   199→    Returns:
   200→        Tuple of (user_message_obj, ai_message_obj, knowledge_used_list)
   201→
   202→    Raises:
   203→        RateLimitError: If daily limit exceeded
   204→        ValidationError: If content invalid
   205→    """
   206→    session = get_session(db, session_id, user_id)
   207→
   208→    # Check limits
   209→    limit, remaining = check_message_limit(db, session)
   210→    if remaining <= 0:
   211→        # Return a limit-reached system message instead of raising
   212→        limit_msg = JarvisMessage(
   213→            session_id=session_id,
   214→            role="system",
   215→            content=_get_limit_message(session),
   216→            message_type="limit_reached",
   217→            metadata_json=json.dumps({"limit": limit}),
   218→        )
   219→        db.add(limit_msg)
   220→        db.flush()
   221→        return limit_msg, limit_msg, []
   222→
   223→    # Save user message
   224→    user_msg = JarvisMessage(
   225→        session_id=session_id,
   226→        role="user",
   227→        content=user_message,
   228→        message_type="text",
   229→    )
   230→    db.add(user_msg)
   231→    db.flush()
   232→
   233→    # Update counters
   234→    session.message_count_today += 1
   235→    session.total_message_count += 1
   236→    session.last_message_date = datetime.now(timezone.utc)
   237→    session.updated_at = datetime.now(timezone.utc)
   238→
   239→    # Track pages visited (heuristic from message content)
   240→    ctx = _parse_context(session.context_json)
   241→    _track_pages_visited(ctx, user_message)
   242→
   243→    # ── Week 8-11: PII scan user message ──
   244→    company_id = session.company_id
   245→    try:
   246→        pii_svc_cls = _get_service(
   247→            "pii_scan", "app.services.pii_scan_service", "PIIScanService",
   248→        )
   249→        if pii_svc_cls and company_id:
   250→            pii_scanner = pii_svc_cls(db, company_id)
   251→            pii_result = pii_scanner.scan_text(user_message)
   252→            ctx["pii_scan"] = pii_result
   253→    except Exception:
   254→        pass
   255→
   256→    # ── Week 8-11: Conversation context enrichment ──
   257→    try:
   258→        conv_svc = _get_service_module("app.services.conversation_service")
   259→        if conv_svc:
   260→            conv_ctx = conv_svc.get_conversation_context(
   261→                session_id, db, ctx,
   262→            )
   263→            if conv_ctx:
   264→                ctx["conversation_turn_count"] = getattr(
   265→                    conv_ctx, "turn_count", 0,
   266→                )
   267→    except Exception:
   268→        pass
   269→
   270→    # ── Note-Taker Agent: Strategic Summary Generation ──
   271→    # Generate a concise 'mission summary' every few turns or when context changes significantly
   272→    turn_count = ctx.get("conversation_turn_count", session.total_message_count)
   273→    try:
   274→        if turn_count % 3 == 0 or turn_count == 1:
   275→            history = _get_recent_history(db, session_id)
   276→            _generate_strategic_summary(db, session_id, ctx, history)
   277→    except Exception:
   278→        pass
   279→
   280→    # ── Week 8-11: Analytics tracking ──
   281→    try:
   282→        analytics_svc = _get_service_module("app.services.analytics_service")
   283→        if analytics_svc:
   284→            stage = ctx.get("detected_stage", "welcome")
   285→            analytics_svc.track_event(
   286→                event_type="message_sent",
   287→                event_category="conversation",
   288→                user_id=user_id,
   289→                company_id=company_id or "",
   290→                session_id=session_id,
   291→                properties={"stage": stage},
   292→                source="jarvis",
   293→            )
   294→    except Exception:
   295→        pass
   296→
   297→    # ── Week 8-11: Lead capture (every 5 turns) ──
   298→    turn_count = ctx.get("conversation_turn_count", session.total_message_count)
   299→    try:
   300→        lead_svc = _get_service_module("app.services.lead_service")
   301→        if lead_svc and turn_count % 5 == 0:
   302→            lead_svc.capture_lead(
   303→                session_id=session_id,
   304→                user_id=user_id,
   305→                company_id=company_id,
   306→                session_context=ctx,
   307→                sentiment_data=ctx.get("sentiment"),
   308→            )
   309→    except Exception:
   310→        pass
   311→
   312→    # ── Week 8-11: Sentiment technique mapping ──
   313→    try:
   314→        stm_cls = _get_service(
   315→            "sentiment_technique_mapper",
   316→            "app.services.sentiment_technique_mapper",
   317→            "SentimentTechniqueMapper",
   318→        )
   319→        if stm_cls:
   320→            mapper = stm_cls()
   321→            sentiment_map = mapper.map(
   322→                frustration_score=ctx.get("frustration_score", 0),
   323→                sentiment_score=ctx.get("sentiment_score", 0.5),
   324→                urgency_level=ctx.get("urgency_level", "normal"),
   325→                customer_tier=ctx.get("customer_tier", "standard"),
   326→                emotion=ctx.get("emotion"),
   327→                is_vip=ctx.get("is_vip", False),
   328→                variant_type=session.pack_type,
   329→                company_id=company_id or "",
   330→            )
   331→            ctx["technique_mapping"] = sentiment_map.to_dict() if hasattr(sentiment_map, 'to_dict') else {}
   332→    except Exception:
   333→        pass
   334→
   335→    # ── AI Path Selection (Jarvis Onboarding vs Support Pipeline) ──
   336→    history = _get_recent_history(db, session_id)
   337→    ai_content = None
   338→    ai_message_type = "text"
   339→    metadata = {}
   340→    knowledge = []
   341→
   342→    # If this is an onboarding session, check for variant pipeline routing
   343→    if session.type == "onboarding":
   344→        # ── Document Testing Feature: Process user document uploads ──
   345→        if user_message.startswith("[DOCUMENT_UPLOAD]:"):
   346→            try:
   347→                # Format: [DOCUMENT_UPLOAD]: filename \n\n Content: ...
   348→                header, doc_content = user_message.split("\n\nContent:\n", 1)
   349→                filename = header.replace("[DOCUMENT_UPLOAD]:", "").strip()
   350→
   351→                # Add to context
   352→                docs = ctx.get("uploaded_docs", [])
   353→                docs.append({
   354→                    "name": filename,
   355→                    "content": doc_content[:5000],  # Limit to 5K chars for context window
   356→                    "uploaded_at": datetime.now(timezone.utc).isoformat()
   357→                })
   358→                ctx["uploaded_docs"] = docs
   359→                session.context_json = json.dumps(ctx)
   360→                db.flush()
   361→
   362→                ai_content = (
   363→                    f"Greetings. I have successfully analyzed '{filename}'. "
   364→                    "I have integrated this new data into my processing awareness for this session. "
   365→                    "What specific insights or simulations would you like me to run using this information?"
   366→                )
   367→                ai_message_type = "text"
   368→                metadata = {"doc_analyzed": filename}
   369→                knowledge = [{"file": "training_context", "score": 1.0}]
   370→            except Exception as exc:
   371→                logger.error("Document upload processing failed: %s", exc)
   372→                ai_content = "I encountered an error trying to process that document. Could you try sending it as plain text?"
   373→                ai_message_type = "error"
   374→        elif _should_use_variant_pipeline(ctx):
   375→            # ── Variant Pipeline Routing for Onboarding ──
   376→            # When a user selected a variant on the Models page,
   377→            # route their onboarding chat through the variant pipeline
   378→            # (Mini Parwa / Pro Parwa) instead of direct AI.
   379→            try:
   380→                from app.core.variant_pipeline_bridge import (
   381→                    process_onboarding_message_sync,
   382→                )
   383→
   384→                session_ctx = _parse_context(session.context_json) if session else {}
   385→
   386→                # Route through the correct variant pipeline
   387→                pipeline_result = process_onboarding_message_sync(
   388→                    query=user_message,
   389→                    company_id=company_id or "",
   390→                    session_context=session_ctx,
   391→                    conversation_id=session_id,
   392→                    ticket_id="",
   393→                    channel="chat",
   394→                    customer_id=user_id,
   395→                    customer_tier=ctx.get("customer_tier", "free"),
   396→                )
   397→
   398→                ai_content = pipeline_result.response_text
   399→                ai_message_type = "variant_pipeline_onboarding"
   400→                metadata = pipeline_result.to_dict()
   401→                knowledge = []  # Pipeline handles its own knowledge
   402→
   403→                # Store pipeline results in context for analytics
   404→                ctx["variant_pipeline"] = {
   405→                    "variant_tier": pipeline_result.variant_tier,
   406→                    "industry": pipeline_result.industry,
   407→                    "pipeline_status": pipeline_result.pipeline_status,
   408→                    "quality_score": pipeline_result.quality_score,
   409→                    "total_latency_ms": pipeline_result.total_latency_ms,
   410→                    "billing_tokens": pipeline_result.billing_tokens,
   411→                    "steps_completed": pipeline_result.steps_completed,
   412→                    "technique_used": pipeline_result.technique_used,
   413→                    "emergency_flag": pipeline_result.emergency_flag,
   414→                    "empathy_score": pipeline_result.empathy_score,
   415→                    "classification_intent": pipeline_result.classification_intent,
   416→                    "path": "onboarding",
   417→                }
   418→
   419→                logger.info(
   420→                    "onboarding_pipeline_complete: tier=%s, status=%s, "
   421→                    "quality=%.1f, latency=%sms, steps=%d",
   422→                    pipeline_result.variant_tier,
   423→                    pipeline_result.pipeline_status,
   424→                    pipeline_result.quality_score,
   425→                    pipeline_result.total_latency_ms,
   426→                    len(pipeline_result.steps_completed),
   427→                )
   428→
   429→            except Exception as exc:
   430→                # Fallback: Use the direct AI provider if pipeline fails
   431→                logger.error(
   432→                    "Variant Pipeline for onboarding failed, falling back to direct AI: %s",
   433→                    exc,
   434→                )
   435→                try:
   436→                    system_prompt = build_system_prompt(db, session_id, user_message)
   437→                    ai_content, ai_message_type, metadata, knowledge = (
   438→                        _call_ai_provider(system_prompt, history, user_message, ctx)
   439→                    )
   440→                except Exception as inner_exc:
   441→                    logger.error("Direct AI also failed: %s", inner_exc)
   442→                    ai_content = _get_friendly_error_message()
   443→                    ai_message_type = "error"
   444→                    metadata = {"error_type": "all_pipelines_failed"}
   445→                    knowledge = []
   446→        else:
   447→            # ── Onboarding Orchestrator Path (no variant_tier set) ──
   448→            # Routes through the full Onboarding Jarvis Orchestrator:
   449→            #   context → awareness → stage detection → function registry →
   450→            #   LLM with function calling → safety gate → execute → awareness update
   451→            # Falls back to direct AI if orchestrator import/call fails.
   452→            logger.info("Using Jarvis Onboarding Orchestrator Path (full pipeline)")
   453→            try:
   454→                from app.services.onboarding_jarvis_orchestrator import (
   455→                    process_onboarding_message,
   456→                )
   457→
   458→                # Bridge async orchestrator from sync context
   459→                _orchestrator_coro = process_onboarding_message(
   460→                    db=db,
   461→                    session_id=session_id,
   462→                    user_id=user_id,
   463→                    company_id=company_id or "",
   464→                    user_message=user_message,
   465→                    channel="chat",
   466→                )
   467→                try:
   468→                    result = asyncio.run(_orchestrator_coro)
   469→                except RuntimeError:
   470→                    # Already inside an event loop — run in a separate thread
   471→                    with concurrent.futures.ThreadPoolExecutor(max_workers=1) as _pool:
   472→                        _future = _pool.submit(asyncio.run, _orchestrator_coro)
   473→                        result = _future.result(timeout=30)
   474→
   475→                # Map orchestrator result to jarvis_service format
   476→                ai_content = result.get("content", "")
   477→                ai_message_type = result.get("message_type", "text")
   478→                metadata = result.get("metadata", {})
   479→                knowledge = []  # Orchestrator handles knowledge internally
   480→
   481→                # Track orchestrator-specific metadata
   482→                if result.get("function_called"):
   483→                    metadata["function_called"] = result["function_called"]
   484→                if result.get("function_result"):
   485→                    metadata["function_result"] = result["function_result"]
   486→
   487→                logger.info(
   488→                    "onboarding_orchestrator_success: session=%s, func=%s, stage=%s",
   489→                    session_id,
   490→                    result.get("function_called"),
   491→                    metadata.get("stage", "unknown"),
   492→                )
   493→
   494→            except ImportError as import_exc:
   495→                # Orchestrator module unavailable — fall back to direct AI
   496→                logger.warning(

   501→                    system_prompt = build_system_prompt(db, session_id, user_message)

   503→                        _call_ai_provider(system_prompt, history, user_message, ctx)

   506→                except Exception as inner_exc:
   507→                    logger.error("Direct AI also failed: %s", inner_exc)
   508→                    ai_content = _get_friendly_error_message()
   509→                    ai_message_type = "error"
   510→                    metadata = {"error_type": "all_paths_failed", "fallback_reason": "orchestrator_import_error"}
   511→                    knowledge = []
   512→
   513→            except Exception as exc:
   514→                # Orchestrator call failed — fall back to direct AI
   515→                logger.error(
   516→                    "Onboarding Orchestrator failed, falling back to direct AI: %s",
   517→                    exc,
   518→                )
   519→                try:
   520→                    system_prompt = build_system_prompt(db, session_id, user_message)
   521→                    ai_content, ai_message_type, metadata, knowledge = (
   522→                        _call_ai_provider(system_prompt, history, user_message, ctx)
   523→                    )
   524→                    metadata["fallback_reason"] = "orchestrator_error"
   525→                except Exception as inner_exc:
   526→                    logger.error("Direct AI also failed: %s", inner_exc)
   527→                    ai_content = _get_friendly_error_message()
   528→                    ai_message_type = "error"
   529→                    metadata = {"error_type": "all_paths_failed", "fallback_reason": "orchestrator_error"}
   530→                    knowledge = []
   531→    else:
   532→        # ── Customer Care Path: Route through Variant Pipeline Bridge ──
   533→        # This connects the Mini Parwa / Parwa / Parwa High LangGraph
   534→        # pipelines to customer care messages. The variant_tier and
   535→        # industry were set during the handoff from Onboarding Jarvis.
   536→        try:

   538→                process_customer_care_message_sync,
   539→            )
   540→
   541→            # Get session context (contains variant_tier, industry, instance_id)
   542→            session_ctx = _parse_context(session.context_json) if session else {}
   543→
   544→            # Route through the correct variant pipeline
   545→            pipeline_result = process_customer_care_message_sync(

   552→                customer_id=user_id,

   555→
   556→            ai_content = pipeline_result.response_text
   557→            ai_message_type = "variant_pipeline"
   558→            metadata = pipeline_result.to_dict()
   559→            knowledge = []  # Pipeline handles its own knowledge
   560→
   561→            # Store pipeline results in context for analytics
   562→            ctx["variant_pipeline"] = {
   563→                "variant_tier": pipeline_result.variant_tier,
   564→                "industry": pipeline_result.industry,
   565→                "pipeline_status": pipeline_result.pipeline_status,
   566→                "quality_score": pipeline_result.quality_score,
   567→                "total_latency_ms": pipeline_result.total_latency_ms,

   574→            }

   576→            logger.info(
   577→                "customer_care_pipeline_complete: tier=%s, status=%s, "
   578→                "quality=%.1f, latency=%sms, steps=%d",
   579→                pipeline_result.variant_tier,
   580→                pipeline_result.pipeline_status,
   581→                pipeline_result.quality_score,
   582→                pipeline_result.total_latency_ms,
   583→                len(pipeline_result.steps_completed),
   584→            )
   585→
   586→        except Exception as exc:
   587→            # Fallback: Use the legacy ai_pipeline if bridge fails
   588→            logger.error(
   589→                "Variant Pipeline Bridge failed, falling back to legacy: %s",
   590→                exc,
   591→            )
   592→            try:
   593→                from app.core.ai_pipeline import process_ai_message
   594→
   595→                conversation_history = [
   596→                    {"role": msg.role, "content": msg.content}
   597→                    for msg in history[-MAX_CONTEXT_HISTORY_MESSAGES:]
   598→                ]
   599→
   600→                pipeline_args = dict(
   601→                    query=user_message,
   602→                    company_id=company_id or "",
   603→                    conversation_id=session_id,
   604→                    variant_type=session.pack_type or "parwa",
   605→                    customer_id=user_id,
   606→                    conversation_history=conversation_history,
   607→                    language="en",
   608→                )
   609→
   610→                try:
   611→                    system_prompt = build_system_prompt(db, session_id, user_message)
   612→                    pipeline_args["system_prompt"] = system_prompt
   613→                    session_ctx = _parse_context(session.context_json) if session else {}
   614→                    if session_ctx:
   615→                        pipeline_args["customer_metadata"] = session_ctx
   616→                except Exception:
   617→                    logger.debug("build_system_prompt failed, pipeline will use default context")
   618→
   619→                # Handle async call from sync context
   620→                try:
   621→                    pipeline_result = asyncio.run(process_ai_message(**pipeline_args))
   622→                except RuntimeError:
   623→                    with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   624→                        future = pool.submit(
   625→                            asyncio.run, process_ai_message(**pipeline_args),
   626→                        )
   627→                        pipeline_result = future.result(timeout=60)
   628→
   629→                ai_content = pipeline_result.response
   630→                ai_message_type = "ai_generated"
   631→                metadata = pipeline_result.to_dict()
   632→                knowledge = [
   633→                    {"file": c.get("source", ""), "score": c.get("score", 1.0)}
   634→                    for c in pipeline_result.citations
   635→                ]
   636→
   637→                # Store pipeline results in context
   638→                ctx["ai_pipeline"] = {
   639→                    "intent": pipeline_result.intent_type,
   640→                    "confidence": pipeline_result.confidence_score,
   641→                    "auto_action": pipeline_result.auto_action,
   642→                    "technique": pipeline_result.technique_used,
   643→                    "model": pipeline_result.model_used,
   644→                }
   645→
   646→            except Exception as inner_exc:
   647→                # Last resort: direct AI provider call
   648→                logger.error(
   649→                    "Legacy pipeline also failed: %s", inner_exc,
   650→                )
   651→                try:
   652→                    system_prompt = build_system_prompt(db, session_id, user_message)
   653→                    ai_content, ai_message_type, metadata, knowledge = (
   654→                        _call_ai_provider(system_prompt, history, user_message, ctx)
   655→                    )
   656→                except Exception:
   657→                    ai_content = _get_friendly_error_message()
   658→                    ai_message_type = "error"
   659→                    metadata = {"error_type": "all_pipelines_failed"}
   660→                    knowledge = []
   661→
   662→    # Ensure we have a response
   663→    if not ai_content:
   664→        ai_content = _get_friendly_error_message()
   665→        ai_message_type = "error"
   666→
   667→    # Save AI response
   668→    ai_msg = JarvisMessage(
   669→        session_id=session_id,
   670→        role="jarvis",
   671→        content=ai_content,
   672→        message_type=ai_message_type,
   673→        metadata_json=json.dumps(metadata),
   674→    )
   675→    db.add(ai_msg)
   676→    db.flush()
   677→
   678→    # Save knowledge used
   679→    knowledge_records = []
   680→    for ku in knowledge:
   681→        ku_record = JarvisKnowledgeUsed(
   682→            message_id=ai_msg.id,
   683→            knowledge_file=ku.get("file", ""),
   684→            relevance_score=ku.get("score", 1.0),
   685→        )
   686→        db.add(ku_record)
   687→        knowledge_records.append(ku)
   688→
   689→    # Detect and update stage
   690→    detected = detect_stage(db, session_id)
   691→    ctx["detected_stage"] = detected
   692→
   693→    # ── Auto-extract demo_topics and concerns_raised from user message ──
   694→    _extract_topics_and_concerns(ctx, user_message)
   695→
   696→    # ── Week 8-11: Post-response audit logging ──
   697→    try:
   698→        audit_svc = _get_service_module("app.services.audit_service")
   699→        if audit_svc:
   700→            audit_svc.log_audit(
   701→                company_id=company_id or "",
   702→                actor_id=user_id,
   703→                actor_type="user",
   704→                action="message_sent",
   705→                resource_type="session",
   706→                resource_id=session_id,
   707→                old_value=None,
   708→                new_value={
   709→                    "stage": detected,
   710→                    "turn": turn_count,
   711→                    "ai_pipeline": ctx.get("ai_pipeline"),
   712→                },
   713→                ip_address=None,
   714→                user_agent=None,
   715→                db=db,
   716→            )
   717→    except Exception:
   718→        pass
   719→
   720→    session.context_json = json.dumps(ctx)
   721→
   722→    db.flush()
   723→    return user_msg, ai_msg, knowledge_records

   749→    db: Session,
   750→    session: JarvisSession,
   751→) -> Tuple[int, int]:
   752→    """Check and enforce message limits.
   753→
   754→    Returns:
   755→        Tuple of (limit, remaining)
   756→
   757→    Raises:
   758→        RateLimitError: If daily limit exceeded
   759→    """
   760→    _maybe_reset_daily_counter(db, session)
   761→
   762→    if session.pack_type == "demo":
   763→        # Check pack expiry
   764→        if session.pack_expiry and datetime.now(timezone.utc) > session.pack_expiry:
   765→            session.pack_type = "free"
   766→            session.pack_expiry = None
   767→            session.message_count_today = 0
   768→            db.flush()
   769→            limit = FREE_DAILY_LIMIT
   770→        else:
   771→            limit = DEMO_DAILY_LIMIT
   772→    else:
   773→        limit = FREE_DAILY_LIMIT
   774→
   775→    remaining = max(0, limit - session.message_count_today)
   776→    return limit, remaining

   781→) -> None:

   788→        else:
   789→            last_date = session.last_message_date

   791→    if last_date is None or last_date < today:
   792→        session.message_count_today = 0
   793→        session.last_message_date = datetime.now(timezone.utc)
   794→        db.flush()

  1120→    """
  1121→    session = db.query(JarvisSession).filter(
  1122→        JarvisSession.id == session_id,
  1123→    ).first()
  1124→    if not session:

  1572→def _try_ai_providers(messages: List[Dict[str, str]]) -> Optional[str]:
  1573→    """Try AI providers in order: Cerebras → Groq → Google. Returns content or None."""
  1574→    providers = [
  1575→        ("cerebras", "https://api.cerebras.ai/v1/chat/completions"),
  1576→        ("groq", "https://api.groq.com/openai/v1/chat/completions"),

  1578→    ]
  1579→
  1580→    for provider_name, endpoint in providers:
  1581→        try:
  1582→            content = _call_single_provider(provider_name, endpoint, messages)

  1584→                return content
  1585→        except Exception:

  1587→    return None

  1590→def _call_single_provider(
  1591→    provider_name: str,
  1592→    endpoint: str,
  1593→    messages: List[Dict[str, str]],
  1594→) -> Optional[str]:
  1595→    """Call a single AI provider and return the response content."""
  1596→    from app.config import get_settings
  1597→
  1598→    settings = get_settings()
  1599→
  1600→    if provider_name == "cerebras":
  1601→        api_key = settings.CEREBRAS_API_KEY
  1602→    elif provider_name == "groq":
  1603→        api_key = settings.GROQ_API_KEY
  1604→    elif provider_name == "google":
  1605→        api_key = settings.GOOGLE_AI_API_KEY
  1606→    else:
  1607→        return None
  1608→
  1609→    if not api_key:
  1610→        return None
  1611→
  1612→    if provider_name == "google":
  1613→        return _call_google_api(endpoint, api_key, messages)
  1614→
  1615→    # OpenAI-compatible: Cerebras and Groq
  1616→    payload = {
  1617→        "model": "llama-3.1-8b" if provider_name == "cerebras" else "llama-3.1-8b",
  1618→        "messages": messages,
  1619→        "temperature": 0.7,
  1620→        "max_tokens": 1024,
  1621→    }
  1622→
  1623→    headers = {
  1624→        "Authorization": f"Bearer {api_key}",
  1625→        "Content-Type": "application/json",
  1626→    }
  1627→
  1628→    import urllib.request
  1629→    req = urllib.request.Request(
  1630→        endpoint,
  1631→        data=json.dumps(payload).encode("utf-8"),
  1632→        headers=headers,
  1633→        method="POST",
  1634→    )

   151→function formatNumber(n: number): string {
   152→  if (n >= 1000000) return `${(n / 1000000).toFixed(1)}M`;
   153→  if (n >= 1000) return `${(n / 1000).toFixed(1)}K`;
   154→  return n.toLocaleString();
   155→}
   156→
   157→function formatHours(hours: number): string {
   158→  if (hours < 1) return `${Math.round(hours * 60)}m`;
   159→  return `${hours.toFixed(1)}h`;
   160→}
   161→
   162→function formatPercent(n: number): string {
   163→  return `${n.toFixed(1)}%`;
   164→}
   165→
   166→function formatCurrency(n: number): string {
   167→  if (n >= 1) return `$${n.toFixed(2)}`;
   168→  return `$${n.toFixed(4)}`;
   169→}

   250→          setPricingContext(ctx);
   251→          return;
   252→        }
   253→      }
   254→    } catch { /* ignore */ }
   255→
   256→    // Fallback: check parwa_onboarding_variants (saved by FirstVictory step)

   258→      const raw = localStorage.getItem('parwa_onboarding_variants');

   294→        // The dashboard will show appropriate prompts for incomplete onboarding.
   295→      } catch {
   296→        // Network error — check localStorage for onboarding completion flag
   297→        // but don't redirect regardless. Let user explore.
   298→        const hasCompletedOnboarding = localStorage.getItem('parwa_onboarding_completed') === 'true';
   299→        if (hasCompletedOnboarding) {
   300→          // already completed, nothing to do
   301→        }
   302→        // Not completed but still let them see dashboard
   303→      }
   304→      setOnboardingChecked(true);
   305→    }
   306→
   307→    checkOnboarding();

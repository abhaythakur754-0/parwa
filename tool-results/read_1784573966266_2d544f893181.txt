    41→    async def analyze_company_crm(
    42→        self,
    43→        company_id: str,
    44→    ) -> Dict[str, Any]:
    45→        """Main entry point: Analyze company's CRM and return recommendations.
    46→
    47→        Steps:
    48→        1. Get all connected integrations for this company
    49→        2. Fetch sample data from each connected integration
    50→        3. Analyze data patterns to detect business type and gaps
    51→        4. Use LLM to generate personalized recommendations
    52→        
    53→        Returns:
    54→            Dict with:
    55→            - connected_integrations: list of currently connected
    56→            - data_profile: summary of what data was found
    57→            - recommendations: list of recommended integrations with reasons
    58→            - analysis_summary: plain text explanation
    59→        """
    60→        logger.info("Starting CRM analysis for company %s", company_id)
    61→
    62→        # Step 1: Get connected integrations
    63→        connected = await self._get_connected_integrations(company_id)
    64→        
    65→        # Step 2: Gather data from each integration
    66→        data_profile = await self._gather_data_from_integrations(company_id, connected)
    67→        
    68→        # Step 3: Detect patterns and gaps
    69→        detected_gaps = await self._detect_gaps(data_profile, connected)
    70→        
    71→        # Step 4: Generate LLM-powered recommendations
    72→        recommendations = await self._generate_recommendations(
    73→            company_id=company_id,
    74→            data_profile=data_profile,
    75→            connected=connected,
    76→            gaps=detected_gaps,
    77→        )
    78→
    79→        result = {
    80→            "company_id": company_id,
    81→            "analyzed_at": datetime.now(timezone.utc).isoformat(),
    82→            "connected_integrations": connected,
    83→            "data_profile": data_profile,
    84→            "detected_gaps": detected_gaps,
    85→            "recommendations": recommendations,
    86→            "analysis_summary": self._build_summary(data_profile, recommendations),
    87→        }
    88→
    89→        logger.info("CRM analysis complete for company %s: %d recommendations", 
    90→                    company_id, len(recommendations))
    91→        return result

    94→        self, company_id: str
    95→    ) -> List[Dict[str, Any]]:
    96→        """Get list of active integrations for a company."""
    97→        from database.models.integration import Integration

   102→                Integration.company_id == company_id,

   107→
   108→        connected = []
   109→        for row in rows:
   110→            catalog_entry = get_integration_by_key(row.integration_type)

   115→                "category": catalog_entry.category.value if catalog_entry else "unknown",
   116→                "connected_at": row.created_at.isoformat() if row.created_at else None,
   117→            })
   118→
   119→        return connected

   121→    async def _gather_data_from_integrations(
   122→        self,
   123→        company_id: str,
   124→        connected: List[Dict[str, Any]],
   125→    ) -> Dict[str, Any]:
   126→        """Gather sample data from each connected integration."""
   127→        profile = {
   128→            "total_contacts": 0,
   129→            "total_orders": 0,
   130→            "total_deals": 0,
   131→            "has_products": False,
   132→            "has_shipping_addresses": False,
   133→            "has_payment_data": False,
   134→            "has_email_campaigns": False,
   135→            "has_ticket_data": False,
   136→            "industries_detected": [],
   137→            "data_points": [],
   138→        }
   139→
   140→        for integration in connected:
   141→            int_type = integration["type"]
   142→            
   143→            try:
   144→                if int_type in ("hubspot", "salesforce", "pipedrive"):
   145→                    crm_data = await self._fetch_crm_data(company_id, int_type)
   146→                    profile["total_contacts"] += crm_data.get("contact_count", 0)
   147→                    profile["total_deals"] += crm_data.get("deal_count", 0)
   148→                    profile["has_products"] = profile["has_products"] or crm_data.get("has_products", False)
   149→                    
   150→                elif int_type in ("shopify", "woocommerce", "bigcommerce"):
   151→                    ecommerce_data = await self._fetch_ecommerce_data(company_id, int_type)
   152→                    profile["total_orders"] += ecommerce_data.get("order_count", 0)
   153→                    profile["has_shipping_addresses"] = ecommerce_data.get("has_shipping_addresses", False)
   154→                    profile["has_payment_data"] = profile.has_payment_data or ecommerce_data.get("has_payment_data", False)
   155→                    profile["has_products"] = profile["has_products"] or ecommerce_data.get("has_products", False)
   156→                    if ecommerce_data.get("industry"):
   157→                        profile["industries_detected"].append("ecommerce")
   158→                        
   159→                elif int_type in ("stripe", "paddle", "paypal"):
   160→                    payment_data = await self._fetch_payment_data(company_id, int_type)
   161→                    profile["has_payment_data"] = True
   162→                    profile["data_points"].append({
   163→                        "source": int_type,
   164→                        "type": "payment",
   165→                        "count": payment_data.get("transaction_count", 0),
   166→                    })
   167→                    
   168→                elif int_type in ("mailchimp", "klaviyo", "brevo"):
   169→                    marketing_data = await self._fetch_marketing_data(company_id, int_type)
   170→                    profile["has_email_campaigns"] = True
   171→                    profile["data_points"].append({
   172→                        "source": int_type,
   173→                        "type": "marketing",
   174→                        "subscriber_count": marketing_data.get("subscriber_count", 0),
   175→                    })
   176→                    
   177→                elif int_type in ("zendesk", "freshdesk", "intercom", "gorgias"):
   178→                    profile["has_ticket_data"] = True
   179→                    
   180→                elif int_type in ("shipstation", "aftership", "easypost", "fedex", "ups", "dhl"):
   181→                    profile["has_shipping_addresses"] = True
   182→
   183→            except Exception as e:
   184→                logger.warning("Failed to gather data from %s: %s", int_type, str(e)[:100])
   185→                profile["data_points"].append({
   186→                    "source": int_type,
   187→                    "error": str(e)[:200],
   188→                })
   189→
   190→        # Deduplicate industries
   191→        profile["industries_detected"] = list(set(profile["industries_detected"]))
   192→        return profile

   194→    async def _fetch_crm_data(
   195→        self, company_id: str, crm_type: str
   196→    ) -> Dict[str, Any]:
   197→        """Fetch data summary from CRM integrations."""
   198→        import httpx
   199→
   200→        creds = self.integration_service.get_credential_config(company_id, crm_type)
   201→        if not creds:
   202→            return {"contact_count": 0, "deal_count": 0, "has_products": False}
   203→
   204→        try:
   205→            if crm_type == "hubspot":
   206→                base_url = "https://api.hubapi.com"
   207→                headers = {"Authorization": f"Bearer {creds.get('api_key', '')}"}
   208→                
   209→                async with httpx.AsyncClient(timeout=10.0) as client:
   210→                    # Get contacts count
   211→                    contacts_resp = await client.get(
   212→                        f"{base_url}/crm/v3/objects/contacts",
   213→                        params={"limit": 1},
   214→                        headers=headers,
   215→                    )
   216→                    contact_count = contacts_resp.json().get("total", 0) if contacts_resp.status_code == 200 else 0
   217→                    
   218→                    # Get deals count
   219→                    deals_resp = await client.get(
   220→                        f"{base_url}/crm/v3/objects/deals",
   221→                        params={"limit": 1},
   222→                        headers=headers,
   223→                    )
   224→                    deal_count = deals_resp.json().get("total", 0) if deals_resp.status_code == 200 else 0
   225→                    
   226→                    # Check for products
   227→                    products_resp = await client.get(
   228→                        f"{base_url}/crm/v3/objects/products",
   229→                        params={"limit": 1},
   230→                        headers=headers,
   231→                    )
   232→                    has_products = products_resp.status_code == 200 and products_resp.json().get("total", 0) > 0
   233→
   234→                return {
   235→                    "contact_count": contact_count,
   236→                    "deal_count": deal_count,
   237→                    "has_products": has_products,
   238→                }
   239→
   240→            elif crm_type == "shopify":
   241→                return await self._fetch_shopify_summary(creds)
   242→
   243→        except Exception as e:
   244→            logger.warning("CRM data fetch failed for %s: %s", crm_type, e)
   245→
   246→        return {"contact_count": 0, "deal_count": 0, "has_products": False}

   274→    async def _fetch_shopify_summary(self, creds: Dict[str, Any]) -> Dict[str, Any]:
   275→        """Fetch summary data from Shopify store."""
   276→        import httpx
   277→
   278→        store_url = creds.get("store_url", "").rstrip("/")
   279→        token = creds.get("access_token", "")
   280→        
   281→        if not store_url or not token:
   282→            return {"order_count": 0, "has_shipping_addresses": False, "has_products": False}
   283→
   284→        headers = {"X-Shopify-Access-Token": token}
   285→        
   286→        async with httpx.AsyncClient(timeout=10.0) as client:
   287→            # Get order count
   288→            orders_resp = await client.get(
   289→                f"{store_url}/admin/api/2024-01/orders/count.json",
   290→                headers=headers,
   291→            )
   292→            order_count = orders_resp.json().get("count", 0) if orders_resp.status_code == 200 else 0
   293→            
   294→            # Get product count
   295→            products_resp = await client.get(
   296→                f"{store_url}/admin/api/2024-01/products/count.json",
   297→                headers=headers,
   298→            )
   299→            has_products = products_resp.json().get("count", 0) > 0 if products_resp.status_code == 200 else False
   300→            
   301→            # Check recent orders for shipping addresses
   302→            shipping_check = await client.get(
   303→                f"{store_url}/admin/api/2024-01/orders.json?limit=5&status=any",
   304→                headers=headers,
   305→            )
   306→            has_shipping = False
   307→            if shipping_check.status_code == 200:
   308→                orders = shipping_check.json().get("orders", [])
   309→                has_shipping = any(o.get("shipping_address") for o in orders)
   310→
   311→        return {
   312→            "order_count": order_count,
   313→            "has_shipping_addresses": has_shipping,
   314→            "has_products": has_products,
   315→        }

   349→        self, company_id: str, marketing_type: str
   350→    ) -> Dict[str, Any]:
   351→        """Fetch data summary from marketing integrations."""
   352→        import httpx

   354→        creds = self.integration_service.get_credential_config(company_id, marketing_type)
   355→        if not creds:
   356→            return {"subscriber_count": 0}
   357→
   358→        try:
   359→            if marketing_type == "mailchimp":
   360→                api_key = creds.get("api_key", "")
   361→                # Extract DC from API key (xxx-us123)
   362→                dc = ""
   363→                if "-" in api_key:
   364→                    parts = api_key.rsplit("-", 1)
   365→                    if len(parts) == 2:
   366→                        dc = parts[-1]
   367→                
   368→                if dc:
   369→                    headers = {"Authorization": f"Bearer {api_key}"}
   370→                    async with httpx.AsyncClient(timeout=10.0) as client:
   371→                        resp = await client.get(
   372→                            f"https://{dc}.api.mailchimp.com/3.0/lists",
   373→                            headers=headers,
   374→                        )
   375→                        if resp.status_code == 200:
   376→                            lists = resp.json().get("lists", [])
   377→                            total_subs = sum(l.get("stats", {}).get("member_count", 0) for l in lists)
   378→                            return {"subscriber_count": total_subs}
   379→
   380→        except Exception as e:
   381→            logger.warning("Marketing data fetch failed for %s: %s", marketing_type, e)
   382→
   383→        return {"subscriber_count": 0}

   385→    async def _detect_gaps(
   386→        self,
   387→        data_profile: Dict[str, Any],
   388→        connected: List[Dict[str, Any]],
   389→    ) -> List[Dict[str, Any]]:
   390→        """Detect gaps between current setup and potential needs."""
   391→        connected_types = {c["type"] for c in connected}
   392→        gaps = []
   393→
   394→        # Gap detection rules
   395→        gap_rules = [
   396→            {
   397→                "id": "shipping_missing",
   398→                "condition": data_profile.get("total_orders", 0) > 0 and not data_profile.get("has_shipping_addresses"),
   399→                "category": "shipping",
   400→                "severity": "high",
   401→                "message": "You have orders but no shipping integration for tracking",
   402→                "recommended": ["shipstation", "aftership", "easypost"],
   403→            },
   404→            {
   405→                "id": "payment_missing",
   406→                "condition": data_profile.get("has_products") and not data_profile.get("has_payment_data"),
   407→                "category": "payments",
   408→                "severity": "high",
   409→                "message": "You sell products but no payment processor is connected",
   410→                "recommended": ["stripe", "paddle"],
   411→            },
   412→            {
   413→                "id": "marketing_missing",
   414→                "condition": data_profile.get("total_contacts", 0) > 100 and not data_profile.get("has_email_campaigns"),
   415→                "category": "marketing",
   416→                "severity": "medium",
   417→                "message": f"You have {data_profile.get('total_contacts', 0)} contacts but no email marketing tool",
   418→                "recommended": ["mailchimp", "klaviyo", "brevo"],
   419→            },
   420→            {
   421→                "id": "helpdesk_missing",
   422→                "condition": data_profile.get("total_contacts", 0) > 50 and not data_profile.get("has_ticket_data"),
   423→                "category": "helpdesk",
   424→                "severity": "medium",
   425→                "message": "Growing customer base but no dedicated helpdesk system",
   426→                "recommended": ["zendesk", "freshdesk", "intercom", "gorgias"],
   427→            },
   428→            {
   429→                "id": "analytics_missing",
   430→                "condition": data_profile.get("total_orders", 0) > 10 or data_profile.get("total_contacts", 0) > 100,
   431→                "category": "analytics",
   432→                "severity": "low",
   433→                "message": "Significant activity but no analytics integration for insights",
   434→                "recommended": ["mixpanel", "amplitude", "google_analytics"],
   435→            },
   436→            {
   437→                "id": "communication_missing",
   438→                "condition": data_profile.get("total_contacts", 0) > 20,
   439→                "category": "communication",
   440→                "severity": "low",
   441→                "message": "Team communication could be improved with Slack integration",
   442→                "recommended": ["slack"],
   443→            },
   444→        ]
   445→
   446→        for rule in gap_rules:
   447→            if rule["condition"]:
   448→                # Filter out already-connected recommendations
   449→                available_recs = [r for r in rule["recommended"] if r not in connected_types]
   450→                if available_recs:
   451→                    gaps.append({
   452→                        **rule,
   453→                        "recommended": available_recs,
   454→                    })
   455→
   456→        return gaps

   458→    async def _call_nvidia_glm(self, prompt: str, max_tokens: int = 800, temperature: float = 0.3) -> str:
   459→        """Call NVIDIA GLM 5.2 directly for CRM analysis.
   460→        
   461→        Uses the same NVIDIA API key and endpoint as the main pipeline.
   462→        Model: z-ai/glm-5.2 (your preferred LLM)
   463→        
   464→        This bypasses the generic llm_call to ensure we use NVIDIA GLM specifically
   465→        for integration recommendations - giving more consistent results.
   466→        """
   467→        import httpx
   468→        
   469→        api_key = os.environ.get("NVIDIA_API_KEY", "")
   470→        if not api_key:
   471→            logger.warning("NVIDIA_API_KEY not set for CRM analysis")
   472→            return ""
   473→        
   474→        messages = [
   475→            {
   476→                "role": "system",
   477→                "content": "You are Parwa's intelligent integration advisor. You analyze business data and recommend specific third-party integrations that would improve their workflow. Always respond in valid JSON format."
   478→            },
   479→            {
   480→                "role": "user", 
   481→                "content": prompt
   482→            }
   483→        ]
   484→        
   485→        payload = {
   486→            "model": "z-ai/glm-5.2",  # Your preferred NVIDIA GLM model
   487→            "messages": messages,
   488→            "temperature": temperature,
   489→            "max_tokens": max_tokens,
   490→        }
   491→        headers = {
   492→            "Authorization": f"Bearer {api_key}",
   493→            "Content-Type": "application/json",
   494→        }
   495→        
   496→        try:
   497→            async with httpx.AsyncClient(timeout=60.0) as client:
   498→                r = await client.post(
   499→                    "https://integrate.api.nvidia.com/v1/chat/completions",
   500→                    json=payload,
   501→                    headers=headers,
   502→                )
   503→            
   504→            if r.status_code == 200:
   505→                data = r.json()
   506→                content = data.get("choices", [{}])[0].get("message", {}).get("content", "")
   507→                usage = data.get("usage", {})
   508→                logger.info(
   509→                    "NVIDIA GLM CRM analysis complete: tokens=%s (prompt=%s, completion=%s)",
   510→                    usage.get("total_tokens", "?"),
   511→                    usage.get("prompt_tokens", "?"),
   512→                    usage.get("completion_tokens", "?"),
   513→                )
   514→                return content.strip()
   515→            else:
   516→                logger.error("NVIDIA GLM API error %d: %s", r.status_code, r.text[:200])
   517→                return ""
   518→                
   519→        except Exception as e:
   520→            logger.error("NVIDIA GLM call failed: %s", str(e)[:200])
   521→            return ""

   523→    async def _generate_recommendations(
   524→        self,
   525→        company_id: str,
   526→        data_profile: Dict[str, Any],
   527→        connected: List[Dict[str, Any]],
   528→        gaps: List[Dict[str, Any]],
   529→    ) -> List[Dict[str, Any]]:
   530→        """Use NVIDIA GLM 5.2 to generate personalized integration recommendations."""
   531→
   532→        # Build context for LLM
   533→        connected_names = [c["name"] for c in connected]
   534→        connected_types = [c["type"] for c in connected]
   535→
   536→        prompt = f"""You are Parwa's integration advisor. Analyze this company's data and recommend specific integrations they need.
   537→
   538→COMPANY DATA PROFILE:
   539→- Total Contacts: {data_profile.get('total_contacts', 0)}
   540→- Total Orders: {data_profile.get('total_orders', 0)}
   541→- Total Deals: {data_profile.get('total_deals', 0)}
   542→- Has Products: {data_profile.get('has_products', False)}
   543→- Has Shipping Data: {data_profile.get('has_shipping_addresses', False)}
   544→- Has Payment Data: {data_profile.get('has_payment_data', False)}
   545→- Has Email Marketing: {data_profile.get('has_email_campaigns', False)}
   546→- Has Helpdesk: {data_profile.get('has_ticket_data', False)}
   547→- Detected Industries: {data_profile.get('industries_detected', ['unknown'])}
   548→- Data Points: {json.dumps(data_profile.get('data_points', [])[:3], indent=2)}
   549→
   550→CURRENTLY CONNECTED ({len(connected)}): {', '.join(connected_names) if connected_names else 'None'}

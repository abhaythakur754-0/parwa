   220→   220→function hexToRgba(hex: string, alpha: number): string {
   221→   221→  const r = parseInt(hex.slice(1, 3), 16);
   222→   222→  const g = parseInt(hex.slice(3, 5), 16);
   223→   223→  const b = parseInt(hex.slice(5, 7), 16);
   224→   224→  return `rgba(${r},${g},${b},${alpha})`;
   225→   225→}

    70→)
    71→async def create_ticket(
    72→    request: Request,
    73→    data: TicketCreate,
    74→    db: Session = Depends(get_db),
    75→    current_user: User = Depends(get_current_user),
    76→) -> Any:
    77→    """Create a new ticket.
    78→
    79→    F-046: Ticket creation with production handlers.
    80→    PS01: Out-of-plan scope check.
    81→    PS05: Duplicate detection.
    82→    PS07: Account suspended check.
    83→    BL05: Rate limiting.
    84→    """
    85→    company_id = current_user.company_id
    86→    user_id = str(current_user.id)
    87→
    88→    service = TicketService(db, company_id)
    89→
    90→    try:
    91→        ticket = service.create_ticket(
    92→            customer_id=data.customer_id,
    93→            channel=data.channel,
    94→            subject=data.subject,
    95→            priority=data.priority,
    96→            category=data.category,
    97→            tags=data.tags,
    98→            metadata_json=data.metadata_json,
    99→            user_id=user_id,
   100→            customer_name=data.customer_name,
   101→            customer_email=data.customer_email,
   102→            customer_phone=data.customer_phone,
   103→            description=data.description,
   104→        )
   105→
   106→        # ── Trigger 8-node PARWA pipeline on the new ticket ────────────
   107→        # Critical priority tickets are processed IMMEDIATELY (synchronous)
   108→        # — they bypass the Celery queue and run inline so the customer
   109→        # gets an AI response in the same request. Non-critical tickets
   110→        # are dispatched to the Celery queue for async processing.
   111→        #
   112→        # The pipeline_dispatcher opens its OWN DB session (separate from
   113→        # this endpoint's session) and commits changes to the ticket
   114→        # (status, metadata, AI message, CRM sync). After it returns, this
   115→        # endpoint's session has a STALE view of the ticket. We must
   116→        # expire + refresh so _ticket_to_response reads the latest state.
   117→        try:
   118→            from app.services.pipeline_dispatcher import dispatch_pipeline_for_ticket
   119→            dispatch_pipeline_for_ticket(
   120→                ticket_id=ticket.id,
   121→                company_id=company_id,
   122→                priority=ticket.priority,
   123→                channel=ticket.channel,
   124→            )
   125→        except Exception as exc:  # noqa: BLE001
   126→            # Pipeline trigger failure MUST NOT fail ticket creation —
   127→            # the ticket is already persisted. Log and let the caller
   128→            # see the ticket; the pipeline will be retried by the next
   129→            # sync or by the dashboard refresh.
   130→            import logging as _logging
   131→            _logger = _logging.getLogger("parwa.api.tickets")
   132→            _logger.warning(
   133→                "pipeline_dispatch_failed_after_create",
   134→                extra={
   135→                    "ticket_id": ticket.id,
   136→                    "company_id": company_id,
   137→                    "priority": ticket.priority,
   138→                    "error": str(exc)[:200],
   139→                },
   140→            )
   141→
   142→        # Refresh the ticket from DB so we see the pipeline's updates
   143→        # (status change, AI message, CRM sync metadata). The pipeline
   144→        # committed on a separate session; this endpoint's session may
   145→        # be in a bad state. We use a FRESH session to build the response
   146→        # so the API endpoint's session issues don't affect the response.
   147→        try:
   148→            db.rollback()  # clear any bad state in the endpoint's session
   149→        except Exception:
   150→            pass
   151→
   152→        # Build response from a fresh session to avoid session corruption
   153→        from database.base import SessionLocal as _SessionLocal
   154→        _fresh_db = _SessionLocal()
   155→        try:
   156→            from database.models.tickets import Ticket as _Ticket
   157→            _fresh_ticket = _fresh_db.query(_Ticket).filter(
   158→                _Ticket.id == ticket.id,
   159→                _Ticket.company_id == company_id,
   160→            ).first()
   161→            if _fresh_ticket:
   162→                return _ticket_to_response(_fresh_ticket)
   163→        finally:
   164→            _fresh_db.close()
   165→
   166→        # Fallback: return a minimal response with just the ticket ID
   167→        # (the frontend will sync the full details via GET /api/v1/tickets)
   168→        return {
   169→            "id": ticket.id,
   170→            "company_id": company_id,
   171→            "customer_id": ticket.customer_id,
   172→            "channel": ticket.channel,
   173→            "status": "open",
   174→            "subject": ticket.subject,
   175→            "priority": ticket.priority,
   176→            "category": ticket.category,
   177→            "tags": [],
   178→            "metadata_json": {},
   179→            "ticket_number": f"TKT-{ticket.id.replace('-', '').upper()[:8]}",
   180→            "customer_name": None,
   181→            "customer_email": None,
   182→            "description": None,
   183→            "assigned_variant": None,
   184→            "ai_confidence": None,
   185→            "cost_per_ticket": None,
   186→            "savings_per_ticket": None,
   187→            "resolution_time_hours": None,
   188→            "reopen_count": 0,
   189→            "frozen": False,
   190→            "parent_ticket_id": None,
   191→            "duplicate_of_id": None,
   192→            "is_spam": False,
   193→            "awaiting_human": False,
   194→            "awaiting_client": False,
   195→            "escalation_level": 1,
   196→            "sla_breached": False,
   197→            "first_response_at": None,
   198→            "resolution_target_at": None,
   199→            "client_timezone": None,
   200→            "plan_snapshot": {},
   201→            "variant_version": None,
   202→            "created_at": ticket.created_at,
   203→            "updated_at": ticket.updated_at,
   204→            "closed_at": None,
   205→        }
   206→
   207→    except AuthorizationError as e:
   208→        raise HTTPException(
   209→            status_code=status.HTTP_403_FORBIDDEN,
   210→            detail=str(e),
   211→        )
   212→    except ValidationError as e:
   213→        raise HTTPException(
   214→            status_code=status.HTTP_400_BAD_REQUEST,

   821→# ── HELPER FUNCTIONS ────────────────────────────────────────────────────────
   822→
   823→def _ticket_to_response(ticket: Any) -> TicketResponse:
   824→    """Convert Ticket model to TicketResponse schema.
   825→
   826→    Enriches the ticket with joined data:
   827→      - customer_name / customer_email from the Customer table
   828→      - description from the first customer TicketMessage
   829→      - ai_confidence from the latest AI TicketMessage
   830→      - ticket_number from the ticket id prefix
   831→      - assigned_variant mirrored from variant_version
   832→      - resolution_time_hours computed from created_at → closed_at
   833→      - cost_per_ticket / savings_per_ticket from ticket metadata
   834→    """
   835→    tags = []
   836→    if ticket.tags:
   837→        try:
   838→            tags = json.loads(ticket.tags)
   839→        except (json.JSONDecodeError, TypeError):
   840→            tags = []
   841→
   842→    metadata_json = {}
   843→    if ticket.metadata_json:
   844→        try:
   845→            metadata_json = json.loads(ticket.metadata_json)
   846→        except (json.JSONDecodeError, TypeError):
   847→            metadata_json = {}
   848→
   849→    plan_snapshot = {}
   850→    if hasattr(ticket, 'plan_snapshot') and ticket.plan_snapshot:
   851→        try:
   852→            plan_snapshot = json.loads(ticket.plan_snapshot)
   853→        except (json.JSONDecodeError, TypeError):
   854→            plan_snapshot = {}
   855→
   856→    now = datetime.now(timezone.utc)
   857→
   858→    # ── Enrich with customer info ─────────────────────────────────────
   859→    customer_name = None
   860→    customer_email = None
   861→    if getattr(ticket, "customer_id", None):
   862→        try:
   863→            from database.models.tickets import Customer
   864→            # ticket is a SQLAlchemy instance; we use its session to query
   865→            # the customer so we don't need a separate db handle here.
   866→            from sqlalchemy.orm import object_session
   867→            sess = object_session(ticket)
   868→            if sess is not None:
   869→                cust = sess.query(Customer).filter(
   870→                    Customer.id == ticket.customer_id,
   871→                    Customer.company_id == ticket.company_id,
   872→                ).first()
   873→                if cust:
   874→                    customer_name = cust.name
   875→                    customer_email = cust.email
   876→        except Exception:  # noqa: BLE001
   877→            pass
   878→
   879→    # ── Enrich with first customer message (description) ─────────────
   880→    description = None
   881→    ai_confidence = None
   882→    try:
   883→        from database.models.tickets import TicketMessage
   884→        from sqlalchemy.orm import object_session
   885→        sess = object_session(ticket)
   886→        if sess is not None:
   887→            # First customer message = the ticket body
   888→            first_msg = sess.query(TicketMessage).filter(
   889→                TicketMessage.ticket_id == ticket.id,
   890→                TicketMessage.role == "customer",
   891→            ).order_by(TicketMessage.created_at.asc()).first()
   892→            if first_msg:
   893→                description = first_msg.content
   894→
   895→            # Latest AI message confidence
   896→            ai_msg = sess.query(TicketMessage).filter(
   897→                TicketMessage.ticket_id == ticket.id,
   898→                TicketMessage.role == "ai",
   899→            ).order_by(TicketMessage.created_at.desc()).first()
   900→            if ai_msg and ai_msg.ai_confidence is not None:
   901→                ai_confidence = float(ai_msg.ai_confidence)
   902→    except Exception:  # noqa: BLE001
   903→        pass
   904→
   905→    # ── Compute resolution_time_hours ─────────────────────────────────
   906→    resolution_time_hours = None
   907→    if ticket.closed_at and ticket.created_at:
   908→        try:
   909→            delta = ticket.closed_at - ticket.created_at
   910→            resolution_time_hours = round(delta.total_seconds() / 3600.0, 2)
   911→        except Exception:  # noqa: BLE001
   912→            pass
   913→
   914→    # ── Cost / savings from metadata (written by pipeline Node 7) ────
   915→    cost_per_ticket = None
   916→    savings_per_ticket = None
   917→    try:
   918→        if isinstance(metadata_json, dict):
   919→            if "cost_per_ticket" in metadata_json:
   920→                cost_per_ticket = float(metadata_json["cost_per_ticket"])
   921→            if "savings_per_ticket" in metadata_json:
   922→                savings_per_ticket = float(metadata_json["savings_per_ticket"])
   923→    except (TypeError, ValueError):  # noqa: BLE001
   924→        pass
   925→
   926→    # ── ticket_number: human-readable TKT-XXXX from id prefix ────────
   927→    ticket_number = None
   928→    if ticket.id:
   929→        # Use first 8 chars of the UUID, uppercased, as the readable number
   930→        ticket_number = f"TKT-{ticket.id.replace('-', '').upper()[:8]}"
   931→
   932→    # ── assigned_variant mirrors variant_version ─────────────────────
   933→    assigned_variant = ticket.variant_version or None
   934→
   935→    return TicketResponse(
   936→        id=ticket.id,
   937→        company_id=ticket.company_id,
   938→        customer_id=ticket.customer_id,
   939→        channel=ticket.channel,
   940→        status=ticket.status,
   941→        subject=ticket.subject,
   942→        priority=ticket.priority,
   943→        category=ticket.category,
   944→        tags=tags,
   945→        agent_id=ticket.agent_id,
   946→        assigned_to=ticket.assigned_to,
   947→        classification_intent=ticket.classification_intent,
   948→        classification_type=ticket.classification_type,
   949→        metadata_json=metadata_json,
   950→        reopen_count=ticket.reopen_count or 0,
   951→        frozen=ticket.frozen or False,
   952→        parent_ticket_id=ticket.parent_ticket_id,
   953→        duplicate_of_id=ticket.duplicate_of_id,
   954→        is_spam=ticket.is_spam or False,
   955→        awaiting_human=ticket.awaiting_human or False,
   956→        awaiting_client=ticket.awaiting_client or False,
   957→        escalation_level=ticket.escalation_level or 1,
   958→        sla_breached=ticket.sla_breached or False,
   959→        first_response_at=ticket.first_response_at,
   960→        resolution_target_at=ticket.resolution_target_at,
   961→        client_timezone=ticket.client_timezone,
   962→        plan_snapshot=plan_snapshot,
   963→        variant_version=ticket.variant_version,
   964→        # Enriched frontend-facing fields
   965→        ticket_number=ticket_number,
   966→        customer_name=customer_name,
   967→        customer_email=customer_email,
   968→        description=description,
   969→        assigned_variant=assigned_variant,
   970→        ai_confidence=ai_confidence,
   971→        cost_per_ticket=cost_per_ticket,
   972→        savings_per_ticket=savings_per_ticket,
   973→        resolution_time_hours=resolution_time_hours,
   974→        created_at=ticket.created_at,
   975→        updated_at=ticket.updated_at,
   976→        closed_at=ticket.closed_at,
   977→    )
   978→
   979→
   980→# ════════════════════════════════════════════════════════════════════
   981→# RESUME PIPELINE — Approach A (pause/resume)
   982→# When a node calls interrupt(), the pipeline pauses. This endpoint
   983→# accepts the human/variant's guidance and resumes the pipeline from
   984→# the exact node that paused — no restart from Node 1.
   985→# ════════════════════════════════════════════════════════════════════
   986→
   987→class ResumeRequest(BaseModel):
   988→    """Request body for resuming a paused pipeline."""
   989→    guidance: str = Field(..., min_length=1, max_length=10000,
   990→                          description="Human/variant guidance to inject into the paused pipeline")
   991→
   992→
   993→@router.post("/{ticket_id}/resume", response_model=dict)
   994→def resume_pipeline(
   995→    ticket_id: str,
   996→    body: ResumeRequest,
   997→    request: Request,
   998→    db: Session = Depends(get_db),
   999→    current_user: User = Depends(get_current_user),
  1000→) -> dict:
  1001→    """Resume a paused pipeline with human/variant guidance.
  1002→
  1003→    When a pipeline node has doubt (e.g. KB insufficient, hallucination
  1004→    detected), it calls interrupt() to pause. This endpoint accepts the
  1005→    guidance answer and resumes the pipeline from the EXACT node that
  1006→    paused — saving LLM calls compared to a full restart.
  1007→
  1008→    The pipeline state is restored from the LangGraph checkpointer
  1009→    (keyed by ticket_id as thread_id).
  1010→    """
  1011→    import logging
  1012→    logger = logging.getLogger("parwa.api.resume")
  1013→
  1014→    # Validate ticket exists + belongs to tenant
  1015→    from database.models.tickets import Ticket
  1016→    ticket = db.query(Ticket).filter(
  1017→        Ticket.id == ticket_id,
  1018→    ).first()
  1019→    if not ticket:
  1020→        raise HTTPException(status_code=404, detail="Ticket not found")
  1021→
  1022→    # Check ticket is actually paused (awaiting_human)
  1023→    if ticket.status != "awaiting_human":
  1024→        raise HTTPException(
  1025→            status_code=400,
  1026→            detail=f"Ticket is not paused (status={ticket.status}). Only awaiting_human tickets can be resumed.",
  1027→        )
  1028→
  1029→    logger.info("resume_pipeline ticket=%s guidance_len=%d", ticket_id, len(body.guidance))
  1030→
  1031→    # ── DIRECT NVIDIA CALL (the ONLY path — no checkpoint resume) ──
  1032→    # The LangGraph checkpoint is unreliable: if the initial pipeline run
  1033→    # had a transient LLM failure, the checkpoint caches that failure
  1034→    # forever. Resuming it returns the same garbage response.
  1035→    #
  1036→    # Instead: fetch the customer's original question, combine it with
  1037→    # the human's guidance, and call NVIDIA directly. Simple, reliable.
  1038→    import os as _os
  1039→    import httpx as _httpx
  1040→    from database.models.tickets import TicketMessage as _TM
  1041→    import json as _json_fast
  1042→
  1043→    nvidia_key = _os.environ.get("NVIDIA_API_KEY", "").strip()
  1044→    if not nvidia_key:
  1045→        return {
  1046→            "status": "error",
  1047→            "ticket_id": ticket_id,
  1048→            "ai_response": "",
  1049→            "message": "NVIDIA_API_KEY not set on server.",
  1050→        }
  1051→
  1052→    # Get the original customer message
  1053→    customer_msg = ""
  1054→    first_msg = db.query(_TM).filter(
  1055→        _TM.ticket_id == ticket_id,
  1056→        _TM.role == "customer",
  1057→    ).order_by(_TM.created_at.asc()).first()
  1058→    if first_msg:
  1059→        customer_msg = first_msg.content or ""
  1060→
  1061→    direct_prompt = (
  1062→        f"You are a customer support agent. A customer asked:\n\n"
  1063→        f"{customer_msg}\n\n"
  1064→        f"A human supervisor has reviewed this and provided the following guidance:\n"
  1065→        f"{body.guidance}\n\n"
  1066→        f"Write a professional, helpful response to the customer based on this guidance. "
  1067→        f"Be empathetic and specific. Do not mention that a supervisor was involved."
  1068→    )
  1069→
  1070→    logger.info("resume_nvidia ticket=%s", ticket_id)
  1071→
  1072→    try:
  1073→        r = _httpx.post(
  1074→            "https://integrate.api.nvidia.com/v1/chat/completions",
  1075→            json={
  1076→                "model": "z-ai/glm-5.2",
  1077→                "messages": [
  1078→                    {"role": "system", "content": "You are a professional customer support agent. Write a helpful, empathetic response based on the guidance provided."},
  1079→                    {"role": "user", "content": direct_prompt},
  1080→                ],
  1081→                "temperature": 0.3,
  1082→                "max_tokens": 500,
  1083→            },
  1084→            headers={
  1085→                "Authorization": f"Bearer {nvidia_key}",
  1086→                "Content-Type": "application/json",
  1087→            },
  1088→            timeout=90.0,
  1089→        )
  1090→    except Exception as http_exc:
  1091→        logger.error("resume_nvidia_http_exception ticket=%s error=%s", ticket_id, str(http_exc)[:300])
  1092→        return {
  1093→            "status": "error",
  1094→            "ticket_id": ticket_id,
  1095→            "ai_response": "",
  1096→            "message": f"HTTP call to NVIDIA failed: {str(http_exc)[:200]}",
  1097→        }
  1098→
  1099→    if r.status_code != 200:
  1100→        logger.error("resume_nvidia_error ticket=%s status=%s body=%s",
  1101→                     ticket_id, r.status_code, r.text[:200])
  1102→        return {
  1103→            "status": "error",
  1104→            "ticket_id": ticket_id,
  1105→            "ai_response": "",
  1106→            "message": f"NVIDIA API returned {r.status_code}: {r.text[:200]}",
  1107→        }
  1108→
  1109→    ai_response = r.json().get("choices", [{}])[0].get("message", {}).get("content", "")
  1110→
  1111→    if not ai_response or len(ai_response) < 10:
  1112→        return {
  1113→            "status": "error",
  1114→            "ticket_id": ticket_id,
  1115→            "ai_response": "",
  1116→            "message": "NVIDIA returned empty response.",
  1117→        }
  1118→
  1119→    # Save the AI response
  1120→    ai_msg = _TM(
  1121→        id=str(__import__("uuid").uuid4()),
  1122→        ticket_id=ticket_id,
  1123→        company_id=ticket.company_id,
  1124→        role="ai",
  1125→        content=ai_response[:6000],
  1126→        channel=ticket.channel,
  1127→        variant_version="nvidia_direct",
  1128→        metadata_json=_json_fast.dumps({
  1129→            "source": "resume_direct_nvidia",
  1130→            "guidance_used": body.guidance[:500],
  1131→        }),
  1132→    )
  1133→    db.add(ai_msg)
  1134→    ticket.status = "resolved"
  1135→    ticket.awaiting_human = False
  1136→    ticket.updated_at = datetime.now(timezone.utc)
  1137→    if not ticket.closed_at:
  1138→        ticket.closed_at = datetime.now(timezone.utc)
  1139→    db.commit()
  1140→
  1141→    logger.info("resume_nvidia_success ticket=%s len=%d", ticket_id, len(ai_response))
  1142→
  1143→    return {
  1144→        "status": "resolved",
  1145→        "ticket_id": ticket_id,
  1146→        "ai_response": ai_response[:500],
  1147→        "quality_score": 0,
  1148→        "message": "Ticket resolved with NVIDIA direct response.",
  1149→    }
  1150→
  1151→

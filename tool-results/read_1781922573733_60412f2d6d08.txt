    16→// ── Main Wizard ───────────────────────────────────────────
    17→export default function OnboardingPage() {
    18→  const router = useRouter();
    19→  const { login, isAuthenticated, isLoading: authLoading } = useAuth();
    20→  const [step, setStep] = useState(1);
    21→  const [data, setData] = useState<WizardData>(initialData);
    22→  const [error, setError] = useState('');
    23→  const [globalLoading, setGlobalLoading] = useState(false);
    24→
    25→  // Resume: If already authenticated, redirect to dashboard
    26→  useEffect(() => {
    27→    if (!authLoading && isAuthenticated) {
    28→      router.push('/');
    29→    }
    30→  }, [authLoading, isAuthenticated, router]);
    31→
    32→  const updateData = useCallback((partial: Partial<WizardData>) => {
    33→    setData((prev) => ({ ...prev, ...partial }));
    34→    setError('');
    35→  }, []);
    36→
    37→  const goNext = () => {
    38→    if (step < 6) setStep(step + 1);
    39→  };
    40→  const goBack = () => {
    41→    if (step > 1) {
    42→      setStep(step - 1);
    43→      setError('');
    44→    }
    45→  };
    46→
    47→  const handleComplete = useCallback(() => {
    48→    login({
    49→      tenantId: data.tenantId,
    50→      tenantName: data.tenantName,
    51→      adminEmail: data.adminEmail,
    52→      jwtToken: data.jwtToken,
    53→      apiKey: data.apiKey || data.generatedKey,
    54→      tier: data.tier,
    55→    });
    56→    router.push('/');
    57→  }, [data, login, router]);
    58→
    59→  return (
    60→    <div className="min-h-screen bg-jarvis-bg flex flex-col">
    61→      {/* Header */}
    62→      <header className="border-b border-jarvis-border px-6 py-4">
    63→        <div className="max-w-5xl mx-auto">
    64→          <div className="flex items-center justify-between mb-3">
    65→            <JarvisLogo subtitle="Onboarding" />
    66→            <span className="text-[10px] text-jarvis-muted tracking-widest">
    67→              STEP {step}/{6}
    68→            </span>
    69→          </div>
    70→
    71→          {/* Progress bar */}
    72→          <div className="flex gap-1">
    73→            {STEPS.map((label, i) => {
    74→              const stepNum = i + 1;
    75→              const isActive = stepNum === step;
    76→              const isComplete = stepNum < step;
    77→              return (
    78→                <React.Fragment key={label}>
    79→                  {i > 0 && (
    80→                    <div className="flex-1 h-px bg-jarvis-border relative top-3" />
    81→                  )}
    82→                  <div className="flex flex-col items-center gap-1 min-w-[60px]">
    83→                    <div
    84→                      className={`
    85→                        w-6 h-6 rounded-full flex items-center justify-center text-[10px] font-bold transition-all duration-300
    86→                        ${isComplete ? 'bg-jarvis-green/20 border border-jarvis-green/50 text-jarvis-green' : ''}
    87→                        ${isActive ? 'bg-jarvis-green/10 border border-jarvis-green/40 text-jarvis-green glow-green' : ''}
    88→                        ${!isComplete && !isActive ? 'bg-jarvis-card border border-jarvis-border text-jarvis-muted' : ''}
    89→                      `}
    90→                    >
    91→                      {isComplete ? '✓' : stepNum}
    92→                    </div>
    93→                    <span className={`text-[9px] tracking-wider uppercase whitespace-nowrap ${isActive ? 'text-jarvis-green' : 'text-jarvis-muted'}`}>
    94→                      {label}
    95→                    </span>
    96→                  </div>
    97→                </React.Fragment>
    98→              );
    99→            })}
   100→          </div>
   101→        </div>
   102→      </header>
   103→
   104→      {/* Content */}
   105→      <main className="flex-1 overflow-y-auto px-4 py-6">
   106→        <div className="max-w-4xl mx-auto">
   107→          {/* Step title */}
   108→          <div className="mb-6">
   109→            <h2 className="text-lg font-bold text-jarvis-text tracking-wider">{STEPS[step - 1]}</h2>
   110→            <p className="text-[10px] text-jarvis-muted tracking-wider mt-1">
   111→              {getStepDescription(step)}
   112→            </p>
   113→          </div>
   114→
   115→          {/* Error display */}
   116→          {error && (
   117→            <div className="mb-4 p-3 bg-jarvis-red/10 border border-jarvis-red/30 rounded-lg text-xs text-jarvis-red">
   118→              <span className="font-bold">ERROR:</span> {error}
   119→            </div>
   120→          )}
   121→
   122→          {/* Steps */}
   123→          {step === 1 && <Step1AccountSetup data={data} updateData={updateData} goNext={goNext} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   124→          {step === 2 && <Step2TierSelection data={data} updateData={updateData} goNext={goNext} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   125→          {step === 3 && <Step3Integrations data={data} updateData={updateData} goNext={goNext} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   126→          {step === 4 && <Step4KnowledgeBase data={data} updateData={updateData} goNext={goNext} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   127→          {step === 5 && <Step5PolicyConfig data={data} updateData={updateData} goNext={goNext} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   128→          {step === 6 && <Step6GoLive data={data} updateData={updateData} onComplete={handleComplete} setError={setError} setLoading={setGlobalLoading} loading={globalLoading} />}
   129→
   130→          {/* Navigation */}
   131→          <div className="flex items-center justify-between mt-8 pt-4 border-t border-jarvis-border">
   132→            <button
   133→              onClick={goBack}
   134→              disabled={step <= 1}
   135→              className="jarvis-btn-ghost text-xs disabled:opacity-30 disabled:cursor-not-allowed"
   136→            >
   137→              ← BACK
   138→            </button>
   139→            <button
   140→              onClick={step === 6 ? handleComplete : goNext}
   141→              disabled={step === 6 && !data.generatedKey}
   142→              className="jarvis-btn-primary text-xs disabled:opacity-30 disabled:cursor-not-allowed"
   143→            >
   144→              {step === 6 ? 'GO TO DASHBOARD →' : 'NEXT →'}
   145→            </button>
   146→          </div>
   147→        </div>
   148→      </main>
   149→    </div>
   150→  );
   151→}

   153→// ── Step 1: Account Setup ──────────────────────────────────
   154→function Step1AccountSetup({ data, updateData, goNext, setError, setLoading, loading }: {
   155→  data: WizardData;
   156→  updateData: (p: Partial<WizardData>) => void;
   157→  goNext: () => void;
   158→  setError: (e: string) => void;
   159→  setLoading: (l: boolean) => void;
   160→  loading: boolean;
   161→}) {
   162→  const [form, setForm] = useState({
   163→    companyName: '',
   164→    adminEmail: '',
   165→    password: '',
   166→    industry: '',
   167→    companySize: '',
   168→  });
   169→
   170→  const handleSubmit = async () => {
   171→    if (!form.companyName || !form.adminEmail || !form.password || !form.industry || !form.companySize) {
   172→      setError('All fields are required');
   173→      return;
   174→    }
   175→    setLoading(true);
   176→    try {
   177→      const result = await jarvisApi.accountSetup({
   178→        company_name: form.companyName,
   179→        admin_email: form.adminEmail,
   180→        password: form.password,
   181→        industry: form.industry,
   182→        company_size: form.companySize,
   183→      });
   184→      updateData({
   185→        tenantId: result.tenant_id || result.tenantId || '',
   186→        tenantName: form.companyName,
   187→        adminEmail: form.adminEmail,
   188→        jwtToken: result.jwt_token || result.jwtToken || result.token || '',
   189→        apiKey: result.api_key || result.apiKey || '',
   190→      });
   191→      goNext();
   192→    } catch (err: any) {
   193→      setError(err.message || 'Account setup failed');
   194→    } finally {
   195→      setLoading(false);
   196→    }
   197→  };
   198→
   199→  const updateField = (key: string, value: string) => {
   200→    setForm((prev) => ({ ...prev, [key]: value }));
   201→  };
   202→
   203→  return (
   204→    <div className="space-y-4 fade-in">
   205→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   206→        <FieldGroup>
   207→          <label className="text-[10px] text-jarvis-muted tracking-wider uppercase">Company Name</label>
   208→          <input
   209→            type="text"
   210→            className="jarvis-input mt-1"
   211→            placeholder="Acme Corp"
   212→            value={form.companyName}
   213→            onChange={(e) => updateField('companyName', e.target.value)}
   214→          />
   215→        </FieldGroup>
   216→        <FieldGroup>
   217→          <label className="text-[10px] text-jarvis-muted tracking-wider uppercase">Admin Email</label>
   218→          <input
   219→            type="email"
   220→            className="jarvis-input mt-1"
   221→            placeholder="admin@acme.com"
   222→            value={form.adminEmail}
   223→            onChange={(e) => updateField('adminEmail', e.target.value)}
   224→          />
   225→        </FieldGroup>
   226→      </div>
   227→
   228→      <FieldGroup>
   229→        <label className="text-[10px] text-jarvis-muted tracking-wider uppercase">Password</label>
   230→        <input
   231→          type="password"
   232→          className="jarvis-input mt-1"
   233→          placeholder="••••••••"
   234→          value={form.password}
   235→          onChange={(e) => updateField('password', e.target.value)}
   236→        />
   237→      </FieldGroup>
   238→
   239→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   240→        <FieldGroup>
   241→          <label className="text-[10px] text-jarvis-muted tracking-wider uppercase">Industry</label>
   242→          <select
   243→            className="jarvis-input mt-1"
   244→            value={form.industry}
   245→            onChange={(e) => updateField('industry', e.target.value)}
   246→          >
   247→            <option value="">Select industry...</option>
   248→            <option value="ecommerce">E-commerce</option>
   249→            <option value="saas">SaaS</option>
   250→            <option value="healthcare">Healthcare</option>
   251→            <option value="finance">Finance</option>
   252→            <option value="education">Education</option>
   253→            <option value="other">Other</option>
   254→          </select>
   255→        </FieldGroup>
   256→        <FieldGroup>
   257→          <label className="text-[10px] text-jarvis-muted tracking-wider uppercase">Company Size</label>
   258→          <select
   259→            className="jarvis-input mt-1"
   260→            value={form.companySize}
   261→            onChange={(e) => updateField('companySize', e.target.value)}
   262→          >
   263→            <option value="">Select size...</option>
   264→            <option value="1-10">1-10 employees</option>
   265→            <option value="11-50">11-50 employees</option>
   266→            <option value="51-200">51-200 employees</option>
   267→            <option value="200+">200+ employees</option>
   268→          </select>
   269→        </FieldGroup>
   270→      </div>
   271→
   272→      <div className="flex justify-end pt-2">
   273→        <button
   274→          onClick={handleSubmit}
   275→          disabled={loading}
   276→          className="jarvis-btn-primary text-xs disabled:opacity-50"
   277→        >
   278→          {loading ? (
   279→            <span className="flex items-center gap-2">
   280→              <span className="inline-block w-3 h-3 border border-jarvis-green/50 border-t-transparent rounded-full animate-spin" />
   281→              CREATING...
   282→            </span>
   283→          ) : (
   284→            'CREATE ACCOUNT →'
   285→          )}
   286→        </button>
   287→      </div>
   288→
   289→      {data.tenantId && (
   290→        <div className="jarvis-card border-jarvis-green/20">
   291→          <div className="flex items-center gap-2 mb-2">
   292→            <span className="text-jarvis-green text-xs">✓</span>
   293→            <span className="text-[10px] text-jarvis-green tracking-wider uppercase font-bold">Account Created</span>
   294→          </div>
   295→          <div className="text-[10px] text-jarvis-muted space-y-1">
   296→            <div className="flex justify-between">
   297→              <span>TENANT ID</span>
   298→              <span className="text-jarvis-cyan font-mono">{data.tenantId}</span>
   299→            </div>
   300→          </div>
   301→        </div>
   302→      )}
   303→    </div>
   304→  );
   305→}

   508→// ── Step 3: Integrations ─────────────────────────────────

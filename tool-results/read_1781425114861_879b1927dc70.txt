   237→function hexToRgba(hex: string, alpha: number): string {
   238→  const r = parseInt(hex.slice(1, 3), 16);
   239→  const g = parseInt(hex.slice(3, 5), 16);
   240→  const b = parseInt(hex.slice(5, 7), 16);
   241→  return `rgba(${r},${g},${b},${alpha})`;
   242→}

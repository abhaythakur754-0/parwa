     7→           + 6 non-LLM design aids (TemplateInjection, GSD, MAKER, etc.)
     8→VERIFY   → Vote, self-reflect, safety check (LIGHT+MEDIUM+GUARDRAIL)
     9→           + 17 non-LLM scoring checks (all Node 6 techniques adapted)
    10→REFINE   → Learn from gaps, regenerate (HEAVY model)
    11→           + 10 non-LLM refinement aids (GapInjection, EscalationRules, etc.)
    12→
    13→Called by Node 1 when a capability gap is detected (no agent claims
    14→the detected capability). Also callable from the Builder chat API.
    15→
    16→ROADMAP: 34 LLM calls + 34 non-LLM techniques = ~97% config accuracy.
    17→"""

    76→# ── STAGE 1: EXPLORE — Understand the user's real intent ───────────
    77→
    78→async def _stage_explore(state: BuilderState) -> BuilderState:
    79→    """EXPLORE: What agent does this tenant need?
    80→
    81→    L1 NON-LLM PRE-FLIGHT (10 checks, 0 LLM calls):
    82→      1. SafetyNet — scrub PII from ticket_query
    83→      2. CLARA — quick confidence estimate
    84→      3. SmartRouter — can we skip DESIGN?
    85→      4. TierGate — check tenant tier permissions
    86→      5. ExistingAgentScan — check for duplicate agents
    87→      6. CapabilityExpansion — expand capability using patterns
    88→      7. GuardrailCheck — non-LLM safety scan
    89→      8. ZeroShotValidator — flag unusual patterns
    90→      9. TheoryOfMind — what's the REAL intent?
    91→     10. StepBackCheck — what's the bigger picture?
    92→
    93→    LLM calls: 2-3 (all LIGHT tier) — only if SmartRouter allows
    94→    """
    95→    tenant_id = state["tenant_id"]
    96→    capability = state.get("detected_capability", "")
    97→    query = state.get("ticket_query", "")
    98→    ticket_type = state.get("ticket_type", "")
    99→    tier = state.get("tier", "parwa")
   100→
   101→    # ══════════════════════════════════════════════════════════════
   102→    # L1: NON-LLM PRE-FLIGHT (10 checks, 0 LLM calls)
   103→    # ══════════════════════════════════════════════════════════════
   104→
   105→    # 1. SafetyNet — scrub PII from ticket_query
   106→    pii_result = safety_net_scrub(query)
   107→    if pii_result["pii_found"]:
   108→        query = pii_result["scrubbed"]
   109→        state["ticket_query"] = query
   110→    state["non_llm_log"].append({"stage": "explore", "technique": "SafetyNet", "result_summary": f"pii_found={pii_result['pii_found']} count={pii_result['count']}"})
   111→
   112→    # 2. GuardrailCheck — non-LLM safety scan on query
   113→    guard_result = guardrail_check(query)
   114→    if not guard_result["safe"]:
   115→        state["status"] = "rejected"
   116→        state["config"]["is_customer_care"] = False
   117→        state["config"]["scope_rejection_reason"] = f"Guardrail blocked: {guard_result['flags'][:3]}"
   118→        state["non_llm_log"].append({"stage": "explore", "technique": "GuardrailCheck", "result_summary": f"BLOCKED flags={guard_result['flag_count']}"})
   119→        return state
   120→    state["non_llm_log"].append({"stage": "explore", "technique": "GuardrailCheck", "result_summary": "safe"})
   121→
   122→    # 3. TierGate — check tenant tier permissions
   123→    tier_result = tier_gate(tier, "auto_created")
   124→    if not tier_result["allowed"]:
   125→        state["status"] = "rejected"
   126→        state["config"]["is_customer_care"] = False
   127→        state["config"]["scope_rejection_reason"] = tier_result["reason"]
   128→        state["non_llm_log"].append({"stage": "explore", "technique": "TierGate", "result_summary": f"BLOCKED tier={tier}"})
   129→        return state
   130→    state["non_llm_log"].append({"stage": "explore", "technique": "TierGate", "result_summary": f"allowed tier={tier} max_type={tier_result['max_type']}"})
   131→
   132→    # 4. Scope check: is this customer care? (existing logic)
   133→    scope_ok, scope_reason = is_customer_care_request(f"{capability} {query}")
   134→    if not scope_ok:
   135→        state["status"] = "rejected"
   136→        state["config"]["is_customer_care"] = False
   137→        state["config"]["scope_rejection_reason"] = scope_reason
   138→        return state
   139→
   140→    state["config"]["is_customer_care"] = True
   141→
   142→    # 5. ExistingAgentScan — check for duplicate agents
   143→    existing_agents = _get_tenant_agents(tenant_id)
   144→    scan_result = existing_agent_scan(existing_agents, capability)
   145→    state["non_llm_log"].append({"stage": "explore", "technique": "ExistingAgentScan", "result_summary": f"has_match={scan_result['has_match']} overlap={scan_result['overlap']}"})
   146→    if scan_result["should_skip_builder"]:
   147→        state["agent_id"] = scan_result["matching_agent"].get("id") if scan_result.get("matching_agent") else None
   148→        state["status"] = "complete"
   149→        state["non_llm_log"].append({"stage": "explore", "technique": "ExistingAgentScan", "result_summary": f"SKIPPED — existing agent covers {capability}"})
   150→        return state
   151→
   152→    # 6. CapabilityExpansion — expand capability using patterns
   153→    expansion = capability_expansion(capability, query)
   154→    expanded_caps = expansion["expanded"]
   155→    state["non_llm_log"].append({"stage": "explore", "technique": "CapabilityExpansion", "result_summary": f"original={capability} added={expansion['added']}"})
   156→    state["non_llm_flags"].extend([f"Expanded caps: {expansion['added']}"] if expansion["added"] else [])
   157→
   158→    # 7. ZeroShotValidator — flag unusual patterns
   159→    zsv_result = zero_shot_validator(capability, state["config"].get("domain", "auto"), expanded_caps)
   160→    state["non_llm_log"].append({"stage": "explore", "technique": "ZeroShotValidator", "result_summary": f"unusual={zsv_result['is_unusual']} flags={zsv_result['flag_count']}"})
   161→    if zsv_result["is_unusual"]:
   162→        state["non_llm_flags"].extend(zsv_result["flags"])
   163→
   164→    # 8. TheoryOfMind — what's the REAL intent?
   165→    tom_result = theory_of_mind(capability, state["config"], query)
   166→    state["non_llm_log"].append({"stage": "explore", "technique": "TheoryOfMind", "result_summary": f"intent={tom_result.get('real_intent', 'unknown')} score={tom_result['score']}"})
   167→    state["non_llm_scores"]["explore_theory_of_mind"] = tom_result["score"]
   168→    real_intent = tom_result.get("real_intent", "")
   169→
   170→    # 9. StepBackCheck — what's the bigger picture?
   171→    step_result = step_back_check(capability, existing_agents, state["config"].get("domain", "auto"))
   172→    state["non_llm_log"].append({"stage": "explore", "technique": "StepBackCheck", "result_summary": f"passes={step_result['passes']} score={step_result['score']}"})
   173→    state["non_llm_scores"]["explore_step_back"] = step_result["score"]
   174→    if step_result.get("issues"):
   175→        state["non_llm_flags"].extend(step_result["issues"])
   176→
   177→    # 10. CLARA — quick confidence estimate
   178→    clara_result = clara_confidence(capability, query)
   179→    state["non_llm_log"].append({"stage": "explore", "technique": "CLARA", "result_summary": f"confidence={clara_result['confidence']} level={clara_result['level']}"})
   180→    state["non_llm_scores"]["explore_clara"] = clara_result["confidence"]
   181→
   182→    # SmartRouter — can we skip DESIGN?
   183→    route_result = smart_route(clara_result, scan_result["has_match"], tier_result["allowed"])
   184→    state["smart_route_action"] = route_result["action"]
   185→    state["non_llm_log"].append({"stage": "explore", "technique": "SmartRouter", "result_summary": f"action={route_result['action']} skip_design={route_result['skip_design']}"})
   186→
   187→    # ══════════════════════════════════════════════════════════════
   188→    # LLM CALLS (only if SmartRouter allows)
   189→    # ══════════════════════════════════════════════════════════════
   190→
   191→    if route_result["skip_design"]:
   192→        # High confidence + existing match — use template directly
   193→        tmpl_result = template_injection(capability)
   194→        if tmpl_result["has_template"]:
   195→            state["config"]["instructions"] = tmpl_result["template"]["instructions"]
   196→            state["config"]["restrictions"] = tmpl_result["template"]["restrictions"]
   197→            state["config"]["capabilities"] = tmpl_result["template"].get("capabilities", [capability])
   198→            state["current_stage"] = "explore_complete"
   199→            state["stage_iterations"]["explore"] = state["stage_iterations"].get("explore", 0) + 1
   200→            state["non_llm_log"].append({"stage": "explore", "technique": "TemplateInjection", "result_summary": "used_template_skipped_llm"})
   201→            logger.info("builder_explore: SKIPPED LLM — template used tenant=%s capability=%s", tenant_id, capability)
   202→            return state
   203→
   204→    # ── LLM: Understand the real intent ────────────────────────────
   205→    intent_prompt = (
   206→        f"A tenant needs a customer care AI agent for: {capability.replace('_', ' ')}\n"
   207→        f"Triggered by a customer ticket: {query[:500]}\n"
   208→        f"Ticket type classified as: {ticket_type}\n"
   209→    )
   210→    if real_intent:
   211→        intent_prompt += f"Customer's REAL intent: {real_intent}\n"
   212→    if expansion["added"]:
   213→        intent_prompt += f"Related capabilities detected: {expansion['added']}\n"
   214→    if zsv_result["is_unusual"]:
   215→        intent_prompt += f"CAUTION: Unusual pattern detected: {zsv_result['flags'][:2]}\n"
   216→    intent_prompt += (
   217→        f"\nAnalyze what this agent should do. Answer in 3-5 sentences:\n"
   218→        f"1. What specific customer problems does this agent solve?\n"
   219→        f"2. What knowledge does it need?\n"
   220→        f"3. What actions should it be able to take?\n"
   221→        f"4. What should it NEVER do?\n"
   222→    )
   223→
   224→    intent_analysis = await builder_llm_call(
   225→        prompt=intent_prompt,
   226→        stage="explore",
   227→        max_tokens=300,
   228→        temperature=0.2,
   229→    )
   230→
   231→    # ── LLM: Decide attachment method ──────────────────────────────
   232→    attachment_prompt = (
   233→        f"A new customer care agent is being created for capability: {capability}\n"
   234→        f"Intent analysis: {intent_analysis[:500]}\n\n"
   235→        f"Choose the best way for tickets to reach this agent:\n"
   236→        f"A) Map to existing ticket category (if capability matches a standard category)\n"
   237→        f"B) Create custom category with keywords (if no standard category fits)\n"
   238→        f"C) Keyword trigger (agent activates when specific words appear)\n\n"
   239→        f"Respond with ONLY the letter (A, B, or C) and a brief reason."
   240→    )
   241→
   242→    attachment_decision = await builder_llm_call(
   243→        prompt=attachment_prompt,
   244→        stage="explore",
   245→        max_tokens=100,
   246→        temperature=0.0,
   247→    )
   248→
   249→    attachment_method = _parse_attachment_method(attachment_decision, capability)
   250→
   251→    # ── Update state ────────────────────────────────────────────────
   252→    state["current_stage"] = "explore_complete"
   253→    state["stage_iterations"]["explore"] = state["stage_iterations"].get("explore", 0) + 3
   254→    state["config"]["attachment_method"] = attachment_method
   255→    state["config"]["agent_name"] = capability.replace("_", " ").title()
   256→    state["config"]["domain"] = "auto"
   257→    state["config"]["agent_role"] = "auto_created"
   258→
   259→    logger.info(
   260→        "builder_explore: complete tenant=%s capability=%s attachment=%s non_llm_checks=10",
   261→        tenant_id, capability, attachment_method,
   262→    )
   263→
   264→    return state
   265→

   267→# ── STAGE 2: DESIGN — Generate 3 candidates, synthesize 1 ──────────
   268→
   269→async def _stage_design(state: BuilderState) -> BuilderState:
   270→    """DESIGN: Generate 3 candidate agent configs, synthesize the best one.
   271→
   272→    NON-LLM DESIGN AIDS (6 checks, 0 LLM calls):
   273→      1. CapabilityExpansion — AGAIN — catch caps LLM candidates missed
   274→      2. TemplateInjection — use template as Candidate #1 (saves 1 LLM call)
   275→      3. StructurePreset — reject malformed JSON early
   276→      4. CrossConflictDetection — check for agent conflicts
   277→      5. TheoryOfMind — AGAIN — does each candidate address REAL intent?
   278→      6. GSD — per-part quality check on each candidate
   279→      7. MAKER — find what's MISSING from each candidate
   280→      8. StepBackCheck — AGAIN — does candidate fit tenant strategy?
   281→
   282→    LLM calls: 3-5 (all MEDIUM tier)
   283→    """
   284→    capability = state.get("detected_capability", "")
   285→    query = state.get("ticket_query", "")
   286→    ticket_type = state.get("ticket_type", "")
   287→    tenant_id = state["tenant_id"]
   288→    attachment_method = state["config"].get("attachment_method", "existing_category")
   289→
   290→    display = capability.replace("_", " ").title()
   291→
   292→    # ══════════════════════════════════════════════════════════════
   293→    # NON-LLM: Pre-design enrichment
   294→    # ══════════════════════════════════════════════════════════════
   295→
   296→    # 1. CapabilityExpansion — AGAIN for design context
   297→    expansion = capability_expansion(capability, query)
   298→    state["non_llm_log"].append({"stage": "design", "technique": "CapabilityExpansion", "result_summary": f"added={expansion['added']}"})
   299→
   300→    # 2. CrossConflictDetection — check for agent conflicts
   301→    existing_agents = _get_tenant_agents(tenant_id)
   302→    conflict_result = cross_conflict_detection(
   303→        state["config"].get("capabilities", [capability]),
   304→        existing_agents,
   305→        tenant_id,
   306→    )
   307→    state["non_llm_log"].append({"stage": "design", "technique": "CrossConflictDetection", "result_summary": f"has_conflict={conflict_result['has_conflict']}"})
   308→    if conflict_result["has_conflict"]:
   309→        state["non_llm_flags"].extend(conflict_result["suggestions"])
   310→
   311→    # ══════════════════════════════════════════════════════════════
   312→    # LLM: Generate 3 candidate configs
   313→    # ══════════════════════════════════════════════════════════════
   314→
   315→    candidates: List[BuilderAgentConfig] = []
   316→
   317→    # 3. TemplateInjection — use template as Candidate #1 if available
   318→    tmpl_result = template_injection(capability)
   319→    if tmpl_result["has_template"]:
   320→        candidates.append({
   321→            "instructions": tmpl_result["template"]["instructions"],
   322→            "restrictions": tmpl_result["template"]["restrictions"],
   323→            "capabilities": tmpl_result["template"].get("capabilities", [capability]),
   324→        })
   325→        state["non_llm_log"].append({"stage": "design", "technique": "TemplateInjection", "result_summary": "template_used_as_candidate_1"})
   326→
   327→    # Generate remaining candidates via LLM
   328→    llm_candidates_needed = 3 - len(candidates)
   329→    for i in range(llm_candidates_needed):
   330→        approach = ["empathetic and detailed", "concise and action-oriented", "thorough and analytical"][i]
   331→
   332→        conflict_note = ""
   333→        if conflict_result["has_conflict"]:
   334→            conflict_note = f"\nNOTE: Conflicts with existing agents — add domain scope. {conflict_result['suggestions'][:1]}"
   335→
   336→        candidate_prompt = (
   337→            f"Create a system prompt for an AI customer support agent that handles "
   338→            f"{display} tickets. Approach: {approach}.\n\n"
   339→            f"Context:\n"
   340→            f"- Capability: {capability}\n"
   341→            f"- Related capabilities: {expansion['expanded']}\n"
   342→            f"- Ticket type: {ticket_type}\n"
   343→            f"- Sample customer message: {query[:300]}\n"
   344→            f"- Attachment method: {attachment_method}"
   345→            f"{conflict_note}\n\n"
   346→            f"Output a JSON object with these fields:\n"
   347→            f'{{"instructions": "3-5 sentence system prompt", '
   348→            f'"restrictions": "rules the agent must follow", '
   349→            f'"capabilities": ["list", "of", "capabilities"]}}\n\n'
   350→            f"Output ONLY the JSON object, no explanation."
   351→        )
   352→
   353→        raw = await builder_llm_call(
   354→            prompt=candidate_prompt,
   355→            stage="design",
   356→            max_tokens=400,
   357→            temperature=0.3 + (i * 0.1),
   358→        )
   359→
   360→        candidate = _parse_candidate_json(raw, capability)
   361→        if candidate:
   362→            candidates.append(candidate)
   363→
   364→    if not candidates:
   365→        candidates.append(_default_candidate(capability, query))
   366→
   367→    # ══════════════════════════════════════════════════════════════
   368→    # NON-LLM: Score and validate each candidate
   369→    # ══════════════════════════════════════════════════════════════
   370→
   371→    for idx, candidate in enumerate(candidates):
   372→        # 4. StructurePreset — reject malformed JSON early
   373→        preset = structure_preset(candidate)
   374→        if not preset["valid"]:
   375→            state["non_llm_log"].append({"stage": "design", "technique": "StructurePreset", "result_summary": f"candidate_{idx+1}_invalid issues={preset['issues']}"})
   376→            # Try to fix by adding missing fields
   377→            if not candidate.get("instructions"):
   378→                candidate["instructions"] = f"Handle {display} tickets professionally and thoroughly."
   379→            if not candidate.get("restrictions"):
   380→                candidate["restrictions"] = "Always verify before processing. Never share customer data."
   381→
   382→        # 5. GSD — per-part quality check
   383→        gsd = gsd_check(candidate)
   384→        state["non_llm_scores"][f"design_gsd_candidate_{idx+1}"] = gsd["overall"]
   385→        state["non_llm_log"].append({"stage": "design", "technique": "GSD", "result_summary": f"candidate_{idx+1} score={gsd['overall']} weakest={gsd['weakest_part']}"})
   386→
   387→        # 6. MAKER — find what's MISSING
   388→        gaps = maker_find_gaps(capability, candidate.get("capabilities", []))
   389→        if gaps["gaps"]:
   390→            state["non_llm_log"].append({"stage": "design", "technique": "MAKER", "result_summary": f"candidate_{idx+1} gaps={gaps['gaps'][:3]}"})
   391→            # Add missing capabilities to the candidate
   392→            for gap in gaps["suggested_additions"][:2]:
   393→                if gap not in candidate.get("capabilities", []):
   394→                    candidate.setdefault("capabilities", []).append(gap)
   395→
   396→        # 7. TheoryOfMind — does this candidate address REAL intent?
   397→        tom = theory_of_mind(capability, candidate, query)
   398→        state["non_llm_scores"][f"design_tom_candidate_{idx+1}"] = tom["score"]
   399→        if not tom["intent_addressed"]:
   400→            state["non_llm_log"].append({"stage": "design", "technique": "TheoryOfMind", "result_summary": f"candidate_{idx+1} MISSING intent terms={tom['missing']}"})
   401→
   402→    # 8. StepBackCheck — does best candidate fit tenant strategy?
   403→    best_idx = 0
   404→    best_score = 0
   405→    for idx, candidate in enumerate(candidates):
   406→        score = state["non_llm_scores"].get(f"design_gsd_candidate_{idx+1}", 0.5)
   407→        tom_score = state["non_llm_scores"].get(f"design_tom_candidate_{idx+1}", 0.5)
   408→        combined = (score + tom_score) / 2
   409→        if combined > best_score:
   410→            best_score = combined
   411→            best_idx = idx
   412→
   413→    step_result = step_back_check(capability, existing_agents, state["config"].get("domain", "auto"))
   414→    state["non_llm_scores"]["design_step_back"] = step_result["score"]
   415→    state["non_llm_log"].append({"stage": "design", "technique": "StepBackCheck", "result_summary": f"passes={step_result['passes']} score={step_result['score']}"})
   416→
   417→    state["candidates"] = candidates
   418→
   419→    # ══════════════════════════════════════════════════════════════
   420→    # LLM: Synthesize the best candidate
   421→    # ══════════════════════════════════════════════════════════════
   422→
   423→    synthesis_prompt = (
   424→        f"3 candidate agent configurations were generated for a {display} agent.\n\n"
   425→    )
   426→    for idx, c in enumerate(candidates):
   427→        synthesis_prompt += f"Candidate {idx+1}:\n{json.dumps(c, indent=2)[:800]}\n\n"
   428→
   429→    synthesis_prompt += (
   430→        f"Synthesize the BEST aspects of all candidates into ONE final config.\n"
   431→        f"Take the strongest instructions, most important restrictions, and "
   432→        f"complete capabilities list.\n\n"
   433→        f"Output a JSON object: "
   434→        f'{{"instructions": "...", "restrictions": "...", "capabilities": [...]}}\n'
   435→        f"Output ONLY the JSON object."
   436→    )
   437→
   438→    raw_synthesis = await builder_llm_call(
   439→        prompt=synthesis_prompt,
   440→        stage="design",
   441→        max_tokens=500,
   442→        temperature=0.2,
   443→    )
   444→
   445→    synthesized = _parse_candidate_json(raw_synthesis, capability)
   446→    if synthesized:
   447→        state["synthesized_config"] = synthesized
   448→        state["config"]["instructions"] = synthesized.get("instructions", "")

   716→        stage="verify",

   723→

   750→    state["non_llm_log"].append({"stage": "verify", "technique": "ContradictionCheck", "result_summary": f"has_contradiction={contradiction['has_contradiction']} gap={contradiction['gap']}"})

   754→    state["non_llm_scores"]["verify_sufficiency"] = sufficiency["score"]
   755→    state["non_llm_log"].append({"stage": "verify", "technique": "SufficiencyCheck", "result_summary": f"sufficient={sufficiency['sufficient']} score={sufficiency['score']}"})
   756→    if not sufficiency["sufficient"]:

   758→
   759→    # Final score: weighted average of LLM + non-LLM
   760→    # If LLM contradicts non-LLM, use the lower score (trust non-LLM)
   761→    if contradiction["has_contradiction"] and contradiction["direction"] == "LLM_overrates":
   762→        overall_score = min(overall_llm_score, non_llm_avg)
   763→    else:
   764→        overall_score = (overall_llm_score * 0.4) + (non_llm_avg * 0.6)
   765→
   766→    # ── Decide: pass or send to REFINE ─────────────────────────────
   767→    if overall_score >= 0.8 and is_safe and sufficiency["sufficient"]:
   768→        state["current_stage"] = "verify_complete"
   769→        state["refine_quality_score"] = overall_score
   770→        logger.info(
   771→            "builder_verify: PASSED tenant=%s capability=%s score=%.2f non_llm=%.2f",
   772→            tenant_id, capability, overall_score, non_llm_avg,
   773→        )
   774→    else:
   775→        state["current_stage"] = "verify_needs_refine"
   776→        state["refine_quality_score"] = overall_score
   777→        logger.info(
   778→            "builder_verify: NEEDS_REFINE tenant=%s capability=%s score=%.2f non_llm=%.2f issues=%d",
   779→            tenant_id, capability, overall_score, non_llm_avg, len(state["verify_issues"]),
   780→        )
   781→
   782→    state["stage_iterations"]["verify"] = state["stage_iterations"].get("verify", 0) + len(votes) + 2
   783→
   784→    return state

   787→# ── STAGE 4: REFINE — Learn from gaps, regenerate ──────────────────
   788→
   789→async def _stage_refine(state: BuilderState) -> BuilderState:
   790→    """REFINE: Learn from verify failures, regenerate using Reflexion.
   791→
   792→    NON-LLM REFINEMENT AIDS (10 checks, 0 LLM calls per iteration):
   793→      1. MAKER — AGAIN — identify SPECIFIC gaps to fix
   794→      2. GapInjection — inject fix hints for Reflexion
   795→      3. EscalationRuleEnrichment — auto-append domain rules
   796→      4. SufficiencyCheck — AGAIN — check if problem is solved after fix
   797→      5. ContradictionCheck — AGAIN — did LLM fix the gap or hide it?
   798→      6. MetaLearner — learn from past Builder runs
   799→      7. ContextualCompression — remove filler from config
   800→      8. ReverseThinking — AGAIN — check for NEW risks from the fix
   801→      9. TheoryOfMind — AGAIN — did fix drift away from REAL intent?
   802→     10. CoVe — AGAIN — verify refined capabilities against tools
   803→     11. RuleBasedAction — AGAIN — per-capability structural rules
   804→     12. Escalation — auto-escalate if Builder can't fix after 3 loops
   805→
   806→    LLM calls: 2-3 per iteration (HEAVY tier). Max 3 iterations.
   807→    """
   808→    capability = state.get("detected_capability", "")
   809→    query = state.get("ticket_query", "")
   810→    tenant_id = state["tenant_id"]
   811→    config = state["config"]
   812→
   813→    display = capability.replace("_", " ").title()
   814→
   815→    # ── MetaLearner: set expectations from past runs ────────────────
   816→    past_results = _get_past_builder_results(tenant_id, capability)
   817→    meta = meta_learner_adjust(capability, past_results)
   818→    state["non_llm_log"].append({"stage": "refine", "technique": "MetaLearner", "result_summary": meta.get("reason", "no_past_data")})
   819→
   820→    for iteration in range(1, 4):
   821→        # ════════════════════════════════════════════════════════════
   822→        # NON-LLM: Prepare specific fix hints
   823→        # ════════════════════════════════════════════════════════════
   824→
   825→        # 1. MAKER — AGAIN identify specific gaps
   826→        gaps = maker_find_gaps(capability, config.get("capabilities", []))
   827→        state["non_llm_log"].append({"stage": "refine", "technique": "MAKER", "result_summary": f"iteration={iteration} gaps={gaps['gaps'][:3]}"})
   828→
   829→        # 2. GapInjection — inject fix hints
   830→        gap_hints = gap_injection(state.get("verify_issues", []), capability)
   831→        state["non_llm_log"].append({"stage": "refine", "technique": "GapInjection", "result_summary": f"iteration={iteration} hints={gap_hints['hint_count']}"})
   832→
   833→        # 3. EscalationRuleEnrichment — auto-append domain rules
   834→        escal_rules = escalation_rule_enrichment(capability, config.get("restrictions", ""))
   835→        if escal_rules["added_rules"]:
   836→            for rule in escal_rules["added_rules"]:
   837→                config["restrictions"] = config.get("restrictions", "") + f" {rule}"
   838→            state["non_llm_log"].append({"stage": "refine", "technique": "EscalationRuleEnrichment", "result_summary": f"iteration={iteration} rules_added={escal_rules['rule_count']}"})
   839→
   840→        # ════════════════════════════════════════════════════════════
   841→        # LLM: Reflexion — regenerate with fix hints
   842→        # ════════════════════════════════════════════════════════════
   843→
   844→        refine_prompt = (
   845→            f"An AI agent for {display} failed quality review. Improve it.\n\n"
   846→            f"Current instructions: {config.get('instructions', '')[:500]}\n"
   847→            f"Current restrictions: {config.get('restrictions', '')[:500]}\n"
   848→            f"Current capabilities: {json.dumps(config.get('capabilities', []))}\n\n"
   849→            f"Issues found:\n"
   850→        )
   851→
   852→        for issue in state.get("verify_issues", [])[:5]:
   853→            refine_prompt += f"- {issue}\n"
   854→
   855→        if gaps["suggested_additions"]:
   856→            refine_prompt += f"\nMissing capabilities to add: {gaps['suggested_additions']}\n"
   857→
   858→        if gap_hints["hints"]:
   859→            refine_prompt += f"\nSpecific fix suggestions:\n"
   860→            for hint in gap_hints["hints"]:
   861→                refine_prompt += f"- {hint}\n"
   862→
   863→        refine_prompt += (
   864→            f"\nSample ticket: {query[:300]}\n\n"
   865→            f"Generate IMPROVED instructions and restrictions that address ALL issues.\n"
   866→            f"Output JSON: {{\"instructions\": \"...\", \"restrictions\": \"...\", "
   867→            f"\"capabilities\": [...]}}\n"
   868→            f"Output ONLY the JSON object."
   869→        )
   870→
   871→        raw_refined = await builder_llm_call(
   872→            prompt=refine_prompt,
   873→            stage="refine",
   874→            max_tokens=500,
   875→            temperature=0.2,
   876→        )
   877→
   878→        refined = _parse_candidate_json(raw_refined, capability)
   879→        if refined:
   880→            config["instructions"] = refined.get("instructions", config.get("instructions", ""))
   881→            config["restrictions"] = refined.get("restrictions", config.get("restrictions", ""))
   882→            config["capabilities"] = refined.get("capabilities", config.get("capabilities", []))
   883→
   884→        # ════════════════════════════════════════════════════════════
   885→        # NON-LLM: Post-refinement checks
   886→        # ════════════════════════════════════════════════════════════
   887→
   888→        # 7. ContextualCompression — remove filler
   889→        if config.get("instructions"):
   890→            compressed = contextual_compression(config["instructions"])
   891→            config["instructions"] = compressed["compressed"]
   892→            state["non_llm_log"].append({"stage": "refine", "technique": "ContextualCompression", "result_summary": f"iteration={iteration} removed={compressed['removed_count']}"})
   893→
   894→        # 8. ReverseThinking — AGAIN check for NEW risks
   895→        reverse = reverse_thinking(capability, config)
   896→        state["non_llm_log"].append({"stage": "refine", "technique": "ReverseThinking", "result_summary": f"iteration={iteration} risks={reverse['risk_count']}"})
   897→
   898→        # 9. TheoryOfMind — AGAIN did fix drift away from intent?
   899→        tom = theory_of_mind(capability, config, query)
   900→        state["non_llm_scores"][f"refine_tom_iter_{iteration}"] = tom["score"]
   901→        state["non_llm_log"].append({"stage": "refine", "technique": "TheoryOfMind", "result_summary": f"iteration={iteration} intent_addressed={tom['intent_addressed']}"})
   902→
   903→        # 10. CoVe — AGAIN verify capabilities against tools
   904→        integrations = _get_tenant_integrations(tenant_id)
   905→        cove = cove_verify(config.get("capabilities", []), integrations)
   906→        state["non_llm_log"].append({"stage": "refine", "technique": "CoVe", "result_summary": f"iteration={iteration} verified={cove['verified']}"})
   907→
   908→        # 11. RuleBasedAction — AGAIN per-capability rules
   909→        rba = rule_based_check(config, capability)
   910→        if not rba["passed"]:
   911→            state["non_llm_log"].append({"stage": "refine", "technique": "RuleBasedAction", "result_summary": f"iteration={iteration} STILL FAILED violations={rba['violations']}"})
   912→            # Force add missing terms to instructions
   913→            for violation in rba["violations"]:
   914→                if "Missing required term" in violation:
   915→                    config["instructions"] += f" Always address: {violation.replace('Missing required term: ', '')}."
   916→
   917→        # ════════════════════════════════════════════════════════════
   918→        # LLM: Quick quality re-check
   919→        # ════════════════════════════════════════════════════════════
   920→
   921→        quick_check_prompt = (
   922→            f"Rate this AI agent configuration (0.0-1.0):\n\n"
   923→            f"Instructions: {config.get('instructions', '')[:400]}\n"
   924→            f"Restrictions: {config.get('restrictions', '')[:400]}\n"
   925→            f"For: {display} tickets\n"
   926→            f"Sample: {query[:200]}\n\n"
   927→            f"Previous issues were: {json.dumps(state.get('verify_issues', [])[:3])}\n\n"
   928→            f"Respond with ONLY a number 0.0-1.0"
   929→        )
   930→
   931→        score_raw = await builder_llm_call(
   932→            prompt=quick_check_prompt,
   933→            stage="refine",
   934→            max_tokens=10,
   935→            temperature=0.0,
   936→        )
   937→
   938→        try:
   939→            llm_score = float(score_raw.strip().replace("'", "").replace('"', ""))
   940→            llm_score = max(0.0, min(1.0, llm_score))
   941→        except (ValueError, TypeError):
   942→            llm_score = 0.7
   943→
   944→        # 5. ContradictionCheck — AGAIN did LLM actually fix or just game score?
   945→        non_llm_quick = [
   946→            tom["score"],
   947→            reverse["risk_score"],
   948→            0.80 if rba["passed"] else 0.50,
   949→            cove["score"],
   950→        ]
   951→        non_llm_quick_avg = sum(non_llm_quick) / max(len(non_llm_quick), 1)
   952→        contra = contradiction_check(llm_score, non_llm_quick_avg)
   953→        state["non_llm_log"].append({"stage": "refine", "technique": "ContradictionCheck", "result_summary": f"iteration={iteration} contradiction={contra['has_contradiction']}"})
   954→
   955→        # If contradiction, trust non-LLM
   956→        if contra["has_contradiction"] and contra["direction"] == "LLM_overrates":
   957→            score = min(llm_score, non_llm_quick_avg)
   958→        else:
   959→            score = (llm_score * 0.4) + (non_llm_quick_avg * 0.6)
   960→
   961→        # 4. SufficiencyCheck — AGAIN check if problem is solved
   962→        suff = sufficiency_check(capability, config)
   963→        if not suff["sufficient"]:
   964→            score = min(score, suff["score"])
   965→            state["non_llm_log"].append({"stage": "refine", "technique": "SufficiencyCheck", "result_summary": f"iteration={iteration} STILL INSUFFICIENT: {suff['reason']}"})
   966→
   967→        state["refine_iterations"] = iteration
   968→        state["refine_quality_score"] = score
   969→
   970→        logger.info(
   971→            "builder_refine: iteration=%d tenant=%s capability=%s score=%.2f llm=%.2f non_llm=%.2f",
   972→            iteration, tenant_id, capability, score, llm_score, non_llm_quick_avg,
   973→        )
   974→
   975→        if score >= 0.8 and suff["sufficient"]:
   976→            break
   977→
   978→    # 12. Escalation — auto-escalate if Builder can't fix
   979→    escalation = should_escalate(
   980→        quality_passed=state["refine_quality_score"] >= 0.8,
   981→        refine_iterations=state["refine_iterations"],
   982→        max_iterations=3,
   983→        contradiction=contra if "contra" in dir() else None,
   984→        sufficiency=suff if "suff" in dir() else None,
   985→    )
   986→    state["non_llm_log"].append({"stage": "refine", "technique": "Escalation", "result_summary": f"escalate={escalation['escalate']} reasons={escalation['reasons']}"})
   987→    if escalation["escalate"]:
   988→        state["non_llm_flags"].append(f"ESCALATION RECOMMENDED: {escalation['reasons']}")
   989→
   990→    state["current_stage"] = "refine_complete"
   991→    state["stage_iterations"]["refine"] = state["stage_iterations"].get("refine", 0) + state["refine_iterations"] * 2
   992→
   993→    return state

  1077→# ── MAIN: Run the full Builder pipeline ────────────────────────────

  1079→async def run_builder_pipeline(
  1080→    tenant_id: str,
  1081→    capability: str,
  1082→    query: str = "",
  1083→    ticket_type: str = "",
  1084→    complexity: str = "",
  1085→    tier: str = "parwa",
  1086→) -> BuilderState:
  1087→    """Run the full 4-stage Builder pipeline.
  1088→
  1089→    Called by Node 1 when a capability gap is detected.
  1090→
  1091→    Args:
  1092→        tenant_id: The tenant's company_id
  1093→        capability: The detected capability (e.g. "refund_processing")
  1094→        query: The original ticket text
  1095→        ticket_type: Classified ticket type
  1096→        complexity: Classified complexity
  1097→        tier: Tenant's subscription tier
  1098→
  1099→    Returns:
  1100→        BuilderState with the final agent config + agent_id
  1101→    """
  1102→    start = time.time()
  1103→
  1104→    # Initialize state
  1105→    state: BuilderState = {
  1106→        "session_id": str(uuid.uuid4()),
  1107→        "tenant_id": tenant_id,
  1108→        "tier": tier,
  1109→        "chat_history": [],
  1110→        "current_stage": "explore",
  1111→        "stage_iterations": {},
  1112→        "config": {
  1113→            "agent_name": capability.replace("_", " ").title(),
  1114→            "agent_role": "auto_created",
  1115→            "domain": "auto",
  1116→            "capabilities": [capability],
  1117→            "instructions": "",
  1118→            "restrictions": "",
  1119→            "attachment_method": "existing_category",
  1120→            "attached_category": None,
  1121→            "custom_category_name": None,
  1122→            "custom_category_keywords": None,
  1123→            "knowledge_sources": [],
  1124→            "guardrails": [],
  1125→            "custom_actions": [],
  1126→            "is_customer_care": True,
  1127→            "scope_rejection_reason": None,
  1128→        },
  1129→        "detected_capability": capability,
  1130→        "ticket_query": query,
  1131→        "ticket_type": ticket_type,
  1132→        "complexity": complexity,
  1133→        "candidates": [],
  1134→        "synthesized_config": None,
  1135→        "verify_votes": [],
  1136→        "verify_consensus": None,
  1137→        "verify_issues": [],
  1138→        "guardrail_safe": True,
  1139→        "refine_iterations": 0,
  1140→        "refine_quality_score": 0.0,
  1141→        "non_llm_log": [],
  1142→        "non_llm_scores": {},
  1143→        "non_llm_flags": [],
  1144→        "smart_route_action": None,
  1145→        "agent_id": None,
  1146→        "status": "building",
  1147→        "error": None,
  1148→    }
  1149→
  1150→    # Resolve attachment method for existing category
  1151→    from app.core.parwa_pipeline.nodes.node_1_ingest_classify import TICKET_TYPE_TO_CAPABILITY
  1152→
  1153→    # Check if this capability maps to an existing category
  1154→    for cat, cap in TICKET_TYPE_TO_CAPABILITY.items():
  1155→        if cap == capability:
  1156→            state["config"]["attachment_method"] = "existing_category"
  1157→            state["config"]["attached_category"] = cat
  1158→            break
  1159→
  1160→    logger.info(
  1161→        "builder_pipeline: START tenant=%s capability=%s tier=%s",
  1162→        tenant_id, capability, tier,
  1163→    )
  1164→
  1165→    # ── Stage 1: EXPLORE ────────────────────────────────────────────
  1166→    state = await _stage_explore(state)
  1167→
  1168→    if state["status"] == "rejected":
  1169→        logger.info("builder_pipeline: REJECTED scope check")
  1170→        return state
  1171→
  1172→    # ── Stage 2: DESIGN ─────────────────────────────────────────────
  1173→    state = await _stage_design(state)
  1174→
  1175→    # ── Stage 3: VERIFY ─────────────────────────────────────────────
  1176→    state = await _stage_verify(state)
  1177→
  1178→    # ── Stage 4: REFINE (only if needed) ────────────────────────────
  1179→    if state["current_stage"] == "verify_needs_refine":
  1180→        state = await _stage_refine(state)
  1181→
  1182→    # ── Finalize: save to DB ────────────────────────────────────────
  1183→    if state["status"] != "rejected" and state["status"] != "failed":
  1184→        state = await _finalize_agent(state)
  1185→
  1186→    elapsed = int((time.time() - start) * 1000)
  1187→    total_llm_calls = sum(state.get("stage_iterations", {}).values())
  1188→
  1189→    logger.info(
  1190→        "builder_pipeline: DONE tenant=%s capability=%s status=%s "
  1191→        "agent_id=%s llm_calls=%d elapsed=%dms",
  1192→        tenant_id, capability, state.get("status"),
  1193→        state.get("agent_id"), total_llm_calls, elapsed,
  1194→    )
  1195→
  1196→    return state

  1202→def _parse_attachment_method(decision: str, capability: str) -> str:
  1203→    """Parse the LLM's attachment method decision."""
  1204→    if not decision:
  1205→        return "existing_category"
  1206→
  1207→    d = decision.strip().upper()
  1208→    if "B" in d or "CUSTOM" in d:
  1209→        return "custom_category"
  1210→    elif "C" in d or "KEYWORD" in d:
  1211→        return "keyword_trigger"
  1212→    else:
  1213→        return "existing_category"

  1216→def _parse_candidate_json(raw: str, capability: str) -> Optional[BuilderAgentConfig]:
  1217→    """Parse a candidate JSON from LLM output."""
  1218→    if not raw:
  1219→        return None
  1220→
  1221→    cleaned = raw.strip()
  1222→    if cleaned.startswith("```"):
  1223→        cleaned = cleaned.split("\n", 1)[1] if "\n" in cleaned else cleaned[3:]
  1224→        if cleaned.endswith("```"):
  1225→            cleaned = cleaned[:-3]
  1226→        cleaned = cleaned.strip()
  1227→
  1228→    try:
  1229→        data = json.loads(cleaned)
  1230→    except json.JSONDecodeError:
  1231→        import re
  1232→        match = re.search(r'\{.*\}', cleaned, re.DOTALL)
  1233→        if match:
  1234→            try:
  1235→                data = json.loads(match.group(0))
  1236→            except json.JSONDecodeError:
  1237→                return None
  1238→        else:
  1239→            return None
  1240→
  1241→    if not isinstance(data, dict):
  1242→        return None
  1243→
  1244→    caps = data.get("capabilities", [capability])
  1245→    if isinstance(caps, str):
  1246→        caps = [c.strip() for c in caps.split(",") if c.strip()]
  1247→    if capability not in caps:
  1248→        caps = [capability] + caps
  1249→
  1250→    return {
  1251→        "instructions": str(data.get("instructions", "")),
  1252→        "restrictions": str(data.get("restrictions", "")),
  1253→        "capabilities": caps,
  1254→    }

    55→    55→
    56→    56→interface ApiKeyInfo {

    58→    58→  auth_type: string;

    69→    69→
    70→    70→/**
    71→    71→ * IntegrationStep Component (Step 3)
    72→    72→ *
    73→    73→ * Displays a grid of integration provider cards organized by category.
    74→    74→ * Users can connect to external services by clicking "Connect" on a
    75→    75→ * provider card, which reveals an inline form with credential inputs.
    76→    76→ * Each connection can be tested, and status indicators show whether
    77→    77→ * the connection is pending, active, or in an error state. The step
    78→    78→ * allows skipping with a warning about limited AI functionality.
    79→    79→ *
    80→    80→ * Phase 13: Now also stores credentials via the API Key system
    81→    81→ * (AES-256-GCM encrypted) and shows key management controls.
    82→    82→ */
    83→    83→export function IntegrationStep({ onNext, industry }: IntegrationStepProps) {
    84→    84→  const [existingIntegrations, setExistingIntegrations] = useState<Integration[]>([]);
    85→    85→  const [isLoading, setIsLoading] = useState(true);
    86→    86→
    87→    87→  // Get industry-filtered integrations from the unified catalog
    88→    88→  // Try the backend API first; if 503, fall back to local catalog import
    89→    89→  const parwaIndustry = industry ? mapIndustryToParwaIndustry(industry) : 'other';
    90→    90→  const [filteredCatalog, setFilteredCatalog] = useState<IntegrationDefinition[]>(() => getIntegrationsForIndustry(parwaIndustry));
    91→    91→
    92→    92→  // Group by category in display order
    93→    93→  const orderedCategories = Object.entries(CATEGORY_META)
    94→    94→    .filter(([key]) => filteredCatalog.some((i) => i.category === key))
    95→    95→    .sort(([, a], [, b]) => a.order - b.order);
    96→    96→  const [expandedProvider, setExpandedProvider] = useState<string | null>(null);
    97→    97→  const [formValues, setFormValues] = useState<Record<string, Record<string, string>>>({});
    98→    98→  const [showPasswords, setShowPasswords] = useState<Record<string, boolean>>({});
    99→    99→  const [connectingProvider, setConnectingProvider] = useState<string | null>(null);
   100→   100→  const [testingProvider, setTestingProvider] = useState<string | null>(null);
   101→   101→  const [showSkipWarning, setShowSkipWarning] = useState(false);
   102→   102→  const [showCustomForm, setShowCustomForm] = useState<'custom' | 'openapi' | null>(null);
   103→   103→
   104→   104→  // ── Phase 13: API Key state ──────────────────────────────────────────
   105→   105→  const [apiKeys, setApiKeys] = useState<Record<string, ApiKeyInfo>>({});
   106→   106→  const [keyActionLoading, setKeyActionLoading] = useState<Record<string, string>>({}); // provider -> action name
   107→   107→
   108→   108→  // Fetch catalog from backend API, fall back to local catalog on 503
   109→   109→  useEffect(() => {
   110→   110→    async function loadCatalog() {
   111→   111→      try {
   112→   112→        const res = await integrationsApi.getCatalog(parwaIndustry);
   113→   113→        if (Array.isArray(res) && res.length > 0) {
   114→   114→          setFilteredCatalog(res as unknown as IntegrationDefinition[]);
   115→   115→        }
   116→   116→      } catch {
   117→   117→        // Backend unreachable (503) — local catalog already set as initial state
   118→   118→        console.warn('[IntegrationStep] Backend catalog unavailable, using local catalog');
   119→   119→      }
   120→   120→    }
   121→   121→    loadCatalog();
   122→   122→  }, [parwaIndustry]);
   123→   123→
   124→   124→  // Fetch existing integrations AND api keys on mount
   125→   125→  useEffect(() => {
   126→   126→    async function loadIntegrations() {
   127→   127→      try {
   128→   128→        const integrations = await integrationsApi.list();
   129→   129→        setExistingIntegrations(Array.isArray(integrations) ? integrations : []);
   130→   130→      } catch {
   131→   131→        setExistingIntegrations([]);
   132→   132→      } finally {
   133→   133→        setIsLoading(false);
   134→   134→      }
   135→   135→    }
   136→   136→    loadIntegrations();
   137→   137→  }, []);
   138→   138→
   139→   139→  // Fetch API keys after integrations are loaded
   140→   140→  useEffect(() => {
   141→   141→    if (existingIntegrations.length === 0) return;
   142→   142→    async function loadApiKeys() {
   143→   143→      try {
   144→   144→        const result = await get<ApiKeyInfo[]>('/api/api-keys/list');
   145→   145→        if (Array.isArray(result)) {
   146→   146→          const keyMap: Record<string, ApiKeyInfo> = {};
   147→   147→          result.forEach((key) => {
   148→   148→            keyMap[key.integration_id] = key;
   149→   149→          });
   150→   150→          setApiKeys(keyMap);
   151→   151→        }
   152→   152→      } catch {
   153→   153→        // Keys endpoint unavailable — non-critical
   154→   154→        console.warn('[IntegrationStep] API keys list unavailable');
   155→   155→      }
   156→   156→    }
   157→   157→    loadApiKeys();
   158→   158→  }, [existingIntegrations]);
   159→   159→
   160→   160→  const getIntegrationStatus = useCallback(
   161→   161→    (providerKey: string): IntegrationStatus | null => {
   162→   162→      const integration = existingIntegrations.find((i) => i.type === providerKey);
   163→   163→      return integration?.status || null;
   164→   164→    },
   165→   165→    [existingIntegrations]
   166→   166→  );
   167→   167→
   168→   168→  const getIntegrationId = useCallback(
   169→   169→    (providerKey: string): string | null => {
   170→   170→      const integration = existingIntegrations.find((i) => i.type === providerKey);
   171→   171→      return integration?.id || null;
   172→   172→    },
   173→   173→    [existingIntegrations]
   174→   174→  );
   175→   175→
   176→   176→  // ── Phase 13: Key management handlers ──────────────────────────────
   177→   177→
   178→   178→  const handleRotateKey = async (providerKey: string) => {
   179→   179→    const integrationId = getIntegrationId(providerKey);
   180→   180→    if (!integrationId) {
   181→   181→      toast.error('Integration not found');
   182→   182→      return;
   183→   183→    }
   184→   184→
   185→   185→    // For rotation, we need new credentials from the user
   186→   186→    const values = formValues[providerKey] || {};
   187→   187→    const hasNewCredentials = Object.values(values).some((v) => v.trim());
   188→   188→    if (!hasNewCredentials) {
   189→   189→      toast.error('Enter new credentials in the form fields above, then click Rotate Key');
   190→   190→      return;
   191→   191→    }
   192→   192→
   193→   193→    setKeyActionLoading((prev) => ({ ...prev, [providerKey]: 'rotate' }));
   194→   194→    try {
   195→   195→      const newCredentials: Record<string, string> = {};
   196→   196→      Object.entries(values).forEach(([k, v]) => {
   197→   197→        if (v.trim()) newCredentials[k] = v;
   198→   198→      });
   199→   199→
   200→   200→      await post('/api/api-keys/rotate', {
   201→   201→        integration_id: integrationId,
   202→   202→        new_credentials: newCredentials,
   203→   203→      });
   204→   204→
   205→   205→      // Refresh keys list
   206→   206→      const result = await get<ApiKeyInfo[]>('/api/api-keys/list');
   207→   207→      if (Array.isArray(result)) {
   208→   208→        const keyMap: Record<string, ApiKeyInfo> = {};
   209→   209→        result.forEach((key) => {
   210→   210→          keyMap[key.integration_id] = key;
   211→   211→        });
   212→   212→        setApiKeys(keyMap);
   213→   213→      }
   214→   214→
   215→   215→      // Clear form values after rotation
   216→   216→      setFormValues((prev) => {
   217→   217→        const next = { ...prev };
   218→   218→        delete next[providerKey];
   219→   219→        return next;
   220→   220→      });
   221→   221→
   222→   222→      toast.success('API key rotated successfully');
   223→   223→    } catch (error) {
   224→   224→      toast.error(getErrorMessage(error));
   225→   225→    } finally {
   226→   226→      setKeyActionLoading((prev) => ({ ...prev, [providerKey]: '' }));
   227→   227→    }
   228→   228→  };
   229→   229→
   230→   230→  const handleRevokeKey = async (providerKey: string) => {
   231→   231→    const integrationId = getIntegrationId(providerKey);
   232→   232→    if (!integrationId) {
   233→   233→      toast.error('Integration not found');
   234→   234→      return;
   235→   235→    }
   236→   236→
   237→   237→    if (!confirm('Are you sure you want to revoke this API key? The integration will lose access.')) {
   238→   238→      return;
   239→   239→    }
   240→   240→
   241→   241→    setKeyActionLoading((prev) => ({ ...prev, [providerKey]: 'revoke' }));
   242→   242→    try {
   243→   243→      await del('/api/api-keys/revoke', {
   244→   244→        data: { integration_id: integrationId },
   245→   245→      } as Parameters<typeof del>[1]);
   246→   246→
   247→   247→      // Remove from local state
   248→   248→      setApiKeys((prev) => {
   249→   249→        const next = { ...prev };
   250→   250→        delete next[integrationId];
   251→   251→        return next;
   252→   252→      });
   253→   253→
   254→   254→      toast.success('API key revoked');
   255→   255→    } catch (error) {
   256→   256→      toast.error(getErrorMessage(error));
   257→   257→    } finally {
   258→   258→      setKeyActionLoading((prev) => ({ ...prev, [providerKey]: '' }));
   259→   259→    }
   260→   260→  };
   261→   261→
   262→   262→  const handleTestKey = async (providerKey: string) => {
   263→   263→    const integrationId = getIntegrationId(providerKey);
   264→   264→    if (!integrationId) {
   265→   265→      toast.error('Integration not found');
   266→   266→      return;
   267→   267→    }
   268→   268→
   269→   269→    setKeyActionLoading((prev) => ({ ...prev, [providerKey]: 'test' }));
   270→   270→    try {
   271→   271→      const result = await post<{ valid?: boolean; detail?: string }>('/api/api-keys/test', {
   272→   272→        integration_id: integrationId,
   273→   273→      });
   274→   274→
   275→   275→      if (result.valid === false) {
   276→   276→        toast.error('API key test failed — key may be invalid or expired');
   277→   277→      } else {
   278→   278→        toast.success('API key test passed');
   279→   279→      }
   280→   280→    } catch (error) {
   281→   281→      toast.error(getErrorMessage(error));
   282→   282→    } finally {
   283→   283→      setKeyActionLoading((prev) => ({ ...prev, [providerKey]: '' }));
   284→   284→    }
   285→   285→  };
   286→   286→
   287→   287→  const handleConnect = async (provider: IntegrationDefinition) => {
   288→   288→    const values = formValues[provider.key] || {};
   289→   289→    const requiredFields = provider.authSchema.fields.filter((f) => f.required);
   290→   290→    const hasAllRequired = requiredFields.every((f) => values[f.name]?.trim());
   291→   291→    if (!hasAllRequired) {
   292→   292→      toast.error('Please fill in all required credentials');
   293→   293→      return;
   294→   294→    }
   295→   295→
   296→   296→    setConnectingProvider(provider.key);
   297→   297→    try {
   298→   298→      const config: Record<string, unknown> = {};
   299→   299→      provider.authSchema.fields.forEach((f) => {
   300→   300→        config[f.name] = values[f.name] || '';
   301→   301→      });
   302→   302→
   303→   303→      // Create integration (existing flow)
   304→   304→      await integrationsApi.create({
   305→   305→        type: provider.key,
   306→   306→        name: provider.name,
   307→   307→        config,
   308→   308→      });
   309→   309→
   310→   310→      const updated = await integrationsApi.list();
   311→   311→      setExistingIntegrations(Array.isArray(updated) ? updated : []);
   312→   312→
   313→   313→      // ── Phase 13: Store credentials via API Key system ──────────────
   314→   314→      try {
   315→   315→        // Determine auth_type from the schema
   316→   316→        const hasPassword = provider.authSchema.fields.some((f) => f.type === 'password');
   317→   317→        const authType = hasPassword ? 'api_key' : 'oauth2';
   318→   318→
   319→   319→        await post('/api/api-keys/store', {
   320→   320→          integration_id: provider.key,
   321→   321→          auth_type: authType,
   322→   322→          credentials: config,
   323→   323→        });
   324→   324→
   325→   325→        // Refresh key list to show the masked key
   326→   326→        const keyResult = await get<ApiKeyInfo[]>('/api/api-keys/list');
   327→   327→        if (Array.isArray(keyResult)) {
   328→   328→          const keyMap: Record<string, ApiKeyInfo> = {};
   329→   329→          keyResult.forEach((key) => {
   330→   330→            keyMap[key.integration_id] = key;
   331→   331→          });
   332→   332→          setApiKeys(keyMap);
   333→   333→        }
   334→   334→      } catch (keyError) {
   335→   335→        // Key storage failure is non-critical — the integration is already created
   336→   336→        console.warn('[IntegrationStep] API key storage failed:', keyError);
   337→   337→        toast('Integration connected, but key storage failed', { icon: '⚠️' });
   338→   338→      }
   339→   339→
   340→   340→      toast.success(`${provider.name} connected successfully`);
   341→   341→      setExpandedProvider(null);
   342→   342→    } catch (error) {
   343→   343→      toast.error(getErrorMessage(error));
   344→   344→    } finally {
   345→   345→      setConnectingProvider(null);
   346→   346→    }
   347→   347→  };
   348→   348→
   349→   349→  const handleTest = async (providerKey: string) => {
   350→   350→    const integration = existingIntegrations.find((i) => i.type === providerKey);
   351→   351→    if (!integration) return;
   352→   352→
   353→   353→    setTestingProvider(providerKey);
   354→   354→    try {
   355→   355→      await integrationsApi.test(integration.id);
   356→   356→      const updated = await integrationsApi.list();
   357→   357→      setExistingIntegrations(Array.isArray(updated) ? updated : []);
   358→   358→      toast.success(`${providerKey} connection test passed`);
   359→   359→    } catch (error) {
   360→   360→      toast.error(getErrorMessage(error));
   361→   361→    } finally {
   362→   362→      setTestingProvider(null);
   363→   363→    }
   364→   364→  };
   365→   365→
   366→   366→  const handleContinue = async () => {
   367→   367→    try {
   368→   368→      await onboardingApi.completeStep(3);
   369→   369→    } catch (error) {
   370→   370→      console.warn('Step 3 complete API failed, continuing locally:', error);
   371→   371→    }
   372→   372→    // Always advance — never block on API failure
   373→   373→    onNext();
   374→   374→  };
   375→   375→
   376→   376→  if (isLoading) {
   377→   377→    return (
   378→   378→      <div className="flex items-center justify-center py-20">
   379→   379→        <Loader2 className="w-6 h-6 animate-spin text-orange-400" />
   380→   380→      </div>
   381→   381→    );
   382→   382→  }
   383→   383→
   384→   384→  return (
   385→   385→    <div className="max-w-3xl mx-auto">
   386→   386→      {/* Header */}
   387→   387→      <div className="text-center mb-10">
   388→   388→        <div className="inline-flex items-center justify-center w-16 h-16 rounded-2xl bg-orange-500/10 border border-orange-500/20 mb-6">
   389→   389→          <Unplug className="w-8 h-8 text-orange-400" />
   390→   390→        </div>
   391→   391→        <h2 className="text-2xl sm:text-3xl font-bold text-white mb-3">
   392→   392→          Connect Your Tools
   393→   393→        </h2>
   394→   394→        <p className="text-orange-200/50 text-sm max-w-lg mx-auto">
   395→   395→          Integrate your existing support channels and tools so PARWA can start handling tickets automatically.
   396→   396→          You can always add more integrations later from your dashboard settings.
   397→   397→        </p>
   398→   398→      </div>
   399→   399→
   400→   400→      {/* Integration categories — filtered by industry */}
   401→   401→      <div className="space-y-8">
   402→   402→        {orderedCategories.map(([catKey, catMeta]) => {
   403→   403→          const providers = filteredCatalog.filter((p) => p.category === catKey);
   404→   404→          const CatIcon = CATEGORY_ICONS[catKey as IntegrationCategory] || Code;
   405→   405→
   406→   406→          return (
   407→   407→            <div key={catKey}>
   408→   408→              <div className="flex items-center gap-2 mb-3">
   409→   409→                <CatIcon className="w-4 h-4 text-orange-400" />
   410→   410→                <h3 className="text-sm font-semibold text-orange-200/60 uppercase tracking-wider">
   411→   411→                  {catMeta.label}
   412→   412→                </h3>
   413→   413→              </div>
   414→   414→
   415→   415→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   416→   416→                {providers.map((provider) => {
   417→   417→                  const status = getIntegrationStatus(provider.key);
   418→   418→                  const isExpanded = expandedProvider === provider.key;
   419→   419→                  const isConnecting = connectingProvider === provider.key;
   420→   420→                  const isTesting = testingProvider === provider.key;
   421→   421→                  const integrationId = getIntegrationId(provider.key);
   422→   422→                  const apiKeyInfo = integrationId ? apiKeys[integrationId] : null;
   423→   423→                  const isKeyActionLoading = keyActionLoading[provider.key];
   424→   424→
   425→   425→                  return (
   426→   426→                    <div
   427→   427→                      key={provider.key}
   428→   428→                      className={cn(
   429→   429→                        'card-parwa transition-all duration-300',
   430→   430→                        isExpanded ? 'col-span-2 sm:col-span-3' : ''
   431→   431→                      )}
   432→   432→                    >
   433→   433→                      {/* Provider card (collapsed) */}
   434→   434→                      <div className="p-4 flex items-center justify-between gap-3">
   435→   435→                        <div className="flex items-center gap-3 min-w-0">
   436→   436→                          <div className={cn('w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0 bg-gradient-to-br', provider.colorGradient)}>
   437→   437→                            <span className="text-white text-xs font-bold">{provider.name.charAt(0)}</span>
   438→   438→                          </div>
   439→   439→                          <span className="text-sm font-medium text-white truncate">
   440→   440→                            {provider.name}
   441→   441→                          </span>
   442→   442→                          {status && (
   443→   443→                            <span
   444→   444→                              className={cn(
   445→   445→                                'inline-flex items-center gap-1 text-xs px-2 py-0.5 rounded-full',
   446→   446→                                status === 'active' && 'bg-green-500/10 text-green-400',
   447→   447→                                status === 'pending' && 'bg-yellow-500/10 text-yellow-400',
   448→   448→                                status === 'error' && 'bg-red-500/10 text-red-400'
   449→   449→                              )}
   450→   450→                            >
   451→   451→                              {status === 'active' && <CheckCircle className="w-3 h-3" />}
   452→   452→                              {status === 'pending' && <Loader2 className="w-3 h-3 animate-spin" />}
   453→   453→                              {status === 'error' && <XCircle className="w-3 h-3" />}
   454→   454→                              {status}
   455→   455→                            </span>
   456→   456→                          )}
   457→   457→                        </div>
   458→   458→
   459→   459→                        <button
   460→   460→                          type="button"
   461→   461→                          onClick={() => setExpandedProvider(isExpanded ? null : provider.key)}
   462→   462→                          className={cn(
   463→   463→                            'rounded-lg px-3 py-1.5 text-xs font-semibold transition-all flex-shrink-0',
   464→   464→                            status === 'active'
   465→   465→                              ? 'bg-white/5 text-orange-200/60 hover:bg-white/10'
   466→   466→                              : 'bg-gradient-to-r from-orange-600 to-orange-500 text-white hover:from-orange-500 hover:to-orange-400 shadow-sm'
   467→   467→                          )}
   468→   468→                        >
   469→   469→                          {status === 'active' ? 'Connected' : isExpanded ? 'Cancel' : 'Connect'}
   470→   470→                        </button>
   471→   471→                      </div>
   472→   472→
   473→   473→                      {/* Expanded inline form — uses catalog authSchema fields */}
   474→   474→                      {isExpanded && (

   311→# ── Per-Provider RPM Limits (sliding 60s window) ───────────────────
   312→# RPM is tracked PER PROVIDER (shared across all of that provider's models).
   313→# When count == limit, we WAIT for the oldest timestamp to fall out of the
   314→# 60s window — we do NOT switch providers for RPM. Provider switching only
   315→# happens for: API errors, daily limits, broken APIs.
   316→PROVIDER_RPM_LIMITS: Dict[ModelProvider, int] = {
   317→    ModelProvider.CEREBRAS: 1000,  # token-based, effectively unlimited
   318→    ModelProvider.GROQ: 30,        # 30 RPM shared across ALL Groq models
   319→    ModelProvider.NVIDIA: 30,      # 30 RPM (safe; actual is 40)
   320→    ModelProvider.GOOGLE: 20,

   324→# ── Provider Usage Tracking ────────────────────────────────────────
   325→
   326→
   327→@dataclass
   328→class ProviderUsage:
   329→    """Tracks usage and health for a single provider+model combination."""
   330→    provider: ModelProvider
   331→    model_id: str
   332→    daily_count: int = 0
   333→    daily_limit: int = 14400
   334→    # RPM tracking (requests per minute — the real bottleneck)
   335→    rpm_count: int = 0
   336→    rpm_limit: int = 30
   337→    rpm_window_start: float = 0.0
   338→    # Legacy token tracking (kept for compatibility)
   339→    minute_count: int = 0
   340→    minute_limit: int = 30000
   341→    minute_window_start: float = 0.0
   342→    rate_limited_until: float = 0.0
   343→    consecutive_failures: int = 0
   344→    last_error: str = ""
   345→    is_healthy: bool = True
   346→    marked_unhealthy_at: float = 0.0
   347→
   348→    @property
   349→    def registry_key(self) -> str:
   350→        return f"{self.model_id}-{self.provider.value}"
   351→
   352→    def get_rpm_available(self) -> int:
   353→        """How many more requests can we make this minute?"""
   354→        import time as _t
   355→        now = _t.time()
   356→        # Reset window every 60 seconds
   357→        if now - self.rpm_window_start > 60:
   358→            self.rpm_window_start = now
   359→            self.rpm_count = 0
   360→            # Auto-recover from unhealthy state when window resets
   361→            if not self.is_healthy:
   362→                self.is_healthy = True
   363→                self.consecutive_failures = 0
   364→        return max(0, self.rpm_limit - self.rpm_count)

   366→
   367→class ProviderHealthTracker:
   368→    """Tracks health and rate limits for all provider+model combinations.
   369→
   370→    BC-008: Never crashes. Tracks failures, rate limits, and provides
   371→    availability checks for the Smart Router's fallback logic.
   372→    Uses class-level shared state so all SmartRouter instances across
   373→    workers see the same health data.
   374→    """
   375→
   376→    CONSECUTIVE_FAILURE_THRESHOLD = 3
   377→    RATE_LIMIT_COOLDOWN_SECONDS = 60
   378→    RATE_LIMIT_RETRY_AFTER_DEFAULT = 60  # seconds when 429 has no Retry-After
   379→
   380→    # Class-level shared state — all instances share the same tracker
   381→    _shared_usage: Dict[str, ProviderUsage] = {}
   382→    _shared_last_daily_reset: str = ""
   383→    # Per-provider RPM sliding window: provider -> deque of call timestamps
   384→    # Shared across ALL models of that provider (e.g. all Groq models share
   385→    # the same 30 RPM budget). Tracked at the PROVIDER level, not per-model.
   386→    _shared_provider_rpm: Dict[ModelProvider, Deque[float]] = {}
   387→    _shared_provider_rpm_locks: Dict[ModelProvider, threading.Lock] = {}
   388→
   389→    def __init__(self) -> None:
   390→        # Use class-level shared state for multi-worker consistency
   391→        self._usage = ProviderHealthTracker._shared_usage
   392→        self._last_daily_reset = ProviderHealthTracker._shared_last_daily_reset
   393→        self._provider_rpm = ProviderHealthTracker._shared_provider_rpm
   394→        self._provider_rpm_locks = ProviderHealthTracker._shared_provider_rpm_locks
   395→        self._reset_daily_if_needed()
   396→
   397→    def _reset_daily_if_needed(self) -> None:
   398→        """Reset daily counters at midnight UTC (BC-012)."""
   399→        today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
   400→        if today != self._last_daily_reset:
   401→            logger.info(
   402→                "Daily usage reset triggered for %s (was %s)",
   403→                today, self._last_daily_reset,
   404→            )
   405→            self._last_daily_reset = today
   406→            ProviderHealthTracker._shared_last_daily_reset = today
   407→            for usage in self._usage.values():
   408→                usage.daily_count = 0
   409→
   410→    def _ensure_usage(self, registry_key: str, config: ModelConfig) -> ProviderUsage:

   579→    def is_available(self, provider: ModelProvider, model_id: str) -> bool:
   580→        """Check if a provider+model is usable (healthy + under daily limits).
   581→
   582→        NOTE: RPM is NOT checked here. RPM is enforced separately via
   583→        ``acquire_provider_slot()`` which WAITS when the provider is at its
   584→        RPM limit — it does not switch providers. Provider switching only
   585→        happens for: API errors, daily limits, broken APIs.
   586→
   587→        This checks:
   588→        1. Rate limit cooldown: are we in a 429 penalty box?
   589→        2. Health: is the provider marked unhealthy? (auto-recovers in 60s)
   590→        3. Daily limit: have we hit the daily cap?
   591→        """
   592→        registry_key = f"{model_id}-{provider.value}"
   593→        if registry_key not in self._usage:
   594→            return True  # No usage data = assume available
   595→
   596→        usage = self._usage[registry_key]
   597→        self._reset_daily_if_needed()
   598→
   599→        # Check rate limit cooldown (429 penalty)
   600→        if usage.rate_limited_until > time.time():
   601→            return False
   602→
   603→        # Auto-recover health flag when RPM window resets
   604→        # (calls get_rpm_available which resets the window)
   605→        usage.get_rpm_available()
   606→
   607→        # If still unhealthy after window reset, provider is genuinely down
   608→        if not usage.is_healthy:
   609→            return False
   610→
   611→        # Check daily limit
   612→        if usage.daily_count >= usage.daily_limit:
   613→            logger.debug(
   614→                "%s: daily limit reached (%d/%d)",
   615→                registry_key, usage.daily_count, usage.daily_limit,
   616→            )
   617→            return False
   618→
   619→        return True
   620→
   621→    # ── Per-Provider RPM Sliding Window ──────────────────────────
   622→    #
   623→    # RPM is tracked PER PROVIDER (shared across all of that provider's
   624→    # models). When count == limit, we WAIT for the oldest timestamp to
   625→    # fall out of the 60s window — we do NOT switch providers for RPM.
   626→    #
   627→    # User rule: "Only wait when literally at 0 — no preemptive waiting".
   628→    # So if even 1 slot is left, we take it immediately without waiting.
   629→
   630→    def _get_provider_lock(self, provider: ModelProvider) -> threading.Lock:
   631→        """Get or create the per-provider lock for RPM slot acquisition."""
   632→        if provider not in self._provider_rpm_locks:
   633→            self._provider_rpm_locks[provider] = threading.Lock()
   634→        return self._provider_rpm_locks[provider]
   635→
   636→    def _get_provider_timestamps(self, provider: ModelProvider) -> Deque[float]:
   637→        """Get or create the per-provider timestamp deque."""
   638→        if provider not in self._provider_rpm:
   639→            self._provider_rpm[provider] = deque()
   640→        return self._provider_rpm[provider]
   641→
   642→    @staticmethod
   643→    def _prune_provider_timestamps(timestamps: Deque[float]) -> None:
   644→        """Remove timestamps older than 60s from the front of the deque."""
   645→        now = time.time()
   646→        while timestamps and timestamps[0] <= now - 60:
   647→            timestamps.popleft()
   648→
   649→    def acquire_provider_slot(
   650→        self,
   651→        provider: ModelProvider,
   652→        max_wait_seconds: float = 65.0,
   653→    ) -> bool:
   654→        """Atomically wait for and acquire an RPM slot for a provider.
   655→
   656→        Behavior (per user spec):
   657→        - If under limit (even by 1) → acquire immediately, no waiting.
   658→        - If at limit (0 remaining) → calculate wait = 60 - (now - oldest),
   659→          sleep, then retry SAME provider. Does NOT switch providers.
   660→        - Returns True if slot acquired, False if max_wait_seconds exceeded.
   661→
   662→        This is a BLOCKING call. Safe to use from sync code and from async
   663→        code running in a worker thread (each pipeline runs in its own thread
   664→        with its own event loop, so blocking is acceptable).
   665→        """
   666→        limit = PROVIDER_RPM_LIMITS.get(provider, 30)
   667→        lock = self._get_provider_lock(provider)
   668→
   669→        with lock:
   670→            timestamps = self._get_provider_timestamps(provider)
   671→            self._prune_provider_timestamps(timestamps)
   672→
   673→            if len(timestamps) < limit:
   674→                # Slot available — take it immediately
   675→                timestamps.append(time.time())
   676→                return True
   677→
   678→            # At limit — calculate wait time for oldest timestamp to expire
   679→            oldest = timestamps[0]
   680→            wait_seconds = 60 - (time.time() - oldest)
   681→
   682→            if wait_seconds <= 0:
   683→                # Edge case: oldest already expired between prune and now
   684→                self._prune_provider_timestamps(timestamps)
   685→                if len(timestamps) < limit:
   686→                    timestamps.append(time.time())
   687→                    return True
   688→                wait_seconds = 1.0  # fallback
   689→
   690→            if wait_seconds > max_wait_seconds:
   691→                logger.warning(
   692→                    "Provider %s RPM wait %.1fs exceeds max_wait %.1fs — skipping slot acquire",
   693→                    provider.value, wait_seconds, max_wait_seconds,
   694→                )
   695→                return False
   696→
   697→            logger.info(
   698→                "Provider %s at RPM limit (%d/%d) — waiting %.1fs for slot",
   699→                provider.value, len(timestamps), limit, wait_seconds,
   700→            )
   701→            time.sleep(wait_seconds + 0.1)  # small buffer for safety
   702→
   703→            # Re-prune and acquire
   704→            self._prune_provider_timestamps(timestamps)
   705→            if len(timestamps) < limit:
   706→                timestamps.append(time.time())
   707→                return True
   708→
   709→            # Still at limit after wait (very high contention) — bail out
   710→            logger.warning(
   711→                "Provider %s still at RPM limit after waiting %.1fs",
   712→                provider.value, wait_seconds,
   713→            )
   714→            return False
   715→
   716→    def get_provider_rpm_status(self) -> Dict[str, Dict[str, Any]]:
   717→        """Return per-provider RPM status for monitoring/debugging."""
   718→        status: Dict[str, Dict[str, Any]] = {}
   719→        for provider in ModelProvider:
   720→            timestamps = self._provider_rpm.get(provider, deque())
   721→            self._prune_provider_timestamps(timestamps)
   722→            limit = PROVIDER_RPM_LIMITS.get(provider, 30)
   723→            status[provider.value] = {
   724→                "rpm_used": len(timestamps),
   725→                "rpm_limit": limit,
   726→                "rpm_available": max(0, limit - len(timestamps)),
   727→            }
   728→        return status
   729→

   848→# ── Smart Router ───────────────────────────────────────────────────

   850→
   851→class SmartRouter:
   852→    """MAKER-Aware 3-Tier LLM Router (F-054).
   853→
   854→    Selects which AI model to use PER ATOMIC STEP (not per query).
   855→    One customer query = 6-24 LLM calls via MAKER framework.
   856→    Technique-boosted calls use LIGHT tier; only raw reasoning
   857→    needs MEDIUM/HEAVY.
   858→
   859→    BC-007: All AI model interaction MUST go through Smart Router.
   860→    BC-001: company_id is always second parameter.
   861→    BC-008: Graceful degradation -- never crash.
   862→    """
   863→
   864→    MAX_RETRIES = 2
   865→    REQUEST_TIMEOUT_SECONDS = 30
   866→
   867→    def __init__(self, config: Any = None) -> None:
   868→        """Initialize the router with model registry and shared health tracker."""
   869→        self._config = config
   870→        # Share health tracker across all instances
   871→        self._health = ProviderHealthTracker()
   872→        logger.info(
   873→            "Smart Router initialized with %d models across %d tiers",
   874→            len(MODEL_REGISTRY),
   875→            len({m.tier for m in MODEL_REGISTRY.values()}),
   876→        )
   877→
   878→    # ── Public API ──────────────────────────────────────────────
   879→
   880→    def route(
   881→        self,
   882→        company_id: str,
   883→        variant_type: str,
   884→        atomic_step: AtomicStepType,
   885→        query_signals: Optional[dict] = None,
   886→    ) -> RoutingDecision:
   887→        """MAIN METHOD -- route one atomic step to a model.
   888→
   889→        BC-001: company_id is always second parameter.
   890→        BC-008: Always returns a valid RoutingDecision.
   891→        """
   892→        try:
   893→            return self._route_safe(
   894→                company_id=company_id,
   895→                variant_type=variant_type,
   896→                atomic_step=atomic_step,
   897→                query_signals=query_signals or {},
   898→            )
   899→        except Exception:
   900→            # BC-008: Absolute safety net -- never crash
   901→            logger.exception(
   902→                "Smart Router routing failed, returning safe LIGHT fallback "
   903→                "(company_id=%s, step=%s, variant=%s)",
   904→                company_id, atomic_step.value, variant_type,
   905→            )
   906→            # Return first LIGHT model as ultimate fallback
   907→            first_light = self._get_first_light_model()
   908→            return RoutingDecision(
   909→                atomic_step_type=atomic_step,
   910→                model_config=first_light,
   911→                provider=first_light.provider,
   912→                tier=ModelTier.LIGHT,
   913→                variant_type=variant_type,
   914→                routing_reason="emergency_fallback_all_checks_failed",
   915→            )
   916→
   917→    def route_batch(
   918→        self,
   919→        company_id: str,
   920→        variant_type: str,
   921→        steps: List[AtomicStepType],
   922→        query_signals: Optional[dict] = None,
   923→    ) -> List[RoutingDecision]:
   924→        """Route multiple steps for a MAKER chain.
   925→
   926→        Returns a RoutingDecision for each atomic step, allowing
   927→        the MAKER orchestrator to execute calls in parallel or sequence.
   928→        """
   929→        decisions: List[RoutingDecision] = []
   930→        signals = query_signals or {}
   931→        for step in steps:
   932→            try:
   933→                decision = self.route(company_id, variant_type, step, signals)
   934→                decisions.append(decision)
   935→            except Exception:
   936→                # BC-008: Even in batch mode, never crash
   937→                logger.exception(
   938→                    "Batch routing failed for step=%s, inserting fallback",
   939→                    step.value,
   940→                )
   941→                first_light = self._get_first_light_model()
   942→                decisions.append(RoutingDecision(
   943→                    atomic_step_type=step,
   944→                    model_config=first_light,
   945→                    provider=first_light.provider,
   946→                    tier=ModelTier.LIGHT,
   947→                    variant_type=variant_type,
   948→                    routing_reason="batch_emergency_fallback",
   949→                ))
   950→        logger.info(
   951→            "Batch routing complete: %d steps for company_id=%s, variant=%s",
   952→            len(decisions), company_id, variant_type,
   953→        )
   954→        return decisions
   955→
   956→    def execute_llm_call(
   957→        self,
   958→        company_id: str,
   959→        routing_decision: RoutingDecision,
   960→        messages: list,
   961→        temperature: float = 0.7,
   962→        max_tokens: int = 1000,
   963→    ) -> dict:
   964→        """Execute an LLM API call using the routing decision.
   965→
   966→        Handles retry + fallback across providers.
   967→        Uses sync wrapper over async HTTP calls.
   968→
   969→        BC-001: company_id is always second parameter.
   970→        BC-008: Always returns a dict with 'content' key.
   971→        """
   972→        try:

  1056→    def _route_safe(
  1057→        self,
  1058→        company_id: str,
  1059→        variant_type: str,
  1060→        atomic_step: AtomicStepType,
  1061→        query_signals: dict,
  1062→    ) -> RoutingDecision:
  1063→        """Safe routing with fallback chain."""
  1064→        allowed_tiers = self._get_allowed_tiers(variant_type)
  1065→        target_tier = self._get_step_tier(atomic_step, allowed_tiers, query_signals)
  1066→
  1067→        # GUARDRAIL is special -- always allowed, never downgraded
  1068→        if target_tier == ModelTier.GUARDRAIL:
  1069→            model, fallbacks = self._select_model(ModelTier.GUARDRAIL, query_signals)
  1070→            return RoutingDecision(
  1071→                atomic_step_type=atomic_step,
  1072→                model_config=model,
  1073→                provider=model.provider,
  1074→                tier=ModelTier.GUARDRAIL,
  1075→                variant_type=variant_type,
  1076→                routing_reason=f"guardrail_step_always_guardrail_model",
  1077→                fallback_models=fallbacks,
  1078→                estimated_tokens=200,
  1079→            )
  1080→
  1081→        # Try target tier first
  1082→        model, fallbacks = self._select_model(target_tier, query_signals)
  1083→        if model is not None and self._is_model_available(model):
  1084→            reason = self._build_routing_reason(
  1085→                target_tier, model, query_signals, atomic_step, variant_type,
  1086→            )
  1087→            return RoutingDecision(
  1088→                atomic_step_type=atomic_step,
  1089→                model_config=model,
  1090→                provider=model.provider,
  1091→                tier=target_tier,
  1092→                variant_type=variant_type,
  1093→                routing_reason=reason,
  1094→                fallback_models=fallbacks,
  1095→                estimated_tokens=self._estimate_tokens(atomic_step),
  1096→                technique_boosted=self._is_technique_boosted(atomic_step),
  1097→            )
  1098→
  1099→        # Fallback within same tier
  1100→        fallback_model = self._fallback_within_tier(target_tier)
  1101→        if fallback_model is not None:
  1102→            logger.warning(
  1103→                "Primary model unavailable for tier=%s step=%s, "
  1104→                "using fallback within tier: %s",
  1105→                target_tier.value, atomic_step.value,
  1106→                fallback_model.display_name,
  1107→            )
  1108→            return RoutingDecision(
  1109→                atomic_step_type=atomic_step,
  1110→                model_config=fallback_model,
  1111→                provider=fallback_model.provider,
  1112→                tier=target_tier,
  1113→                variant_type=variant_type,
  1114→                routing_reason=f"fallback_within_{target_tier.value}",
  1115→                estimated_tokens=self._estimate_tokens(atomic_step),
  1116→            )
  1117→
  1118→        # Fallback to lower tier
  1119→        lower_model = self._fallback_to_lower_tier(target_tier, allowed_tiers)
  1120→        if lower_model is not None:
  1121→            logger.warning(
  1122→                "All models in tier=%s unavailable for step=%s, "
  1123→                "degraded to tier=%s: %s",
  1124→                target_tier.value, atomic_step.value,
  1125→                lower_model.tier.value, lower_model.display_name,
  1126→            )
  1127→            return RoutingDecision(
  1128→                atomic_step_type=atomic_step,
  1129→                model_config=lower_model,
  1130→                provider=lower_model.provider,
  1131→                tier=lower_model.tier,
  1132→                variant_type=variant_type,
  1133→                routing_reason=(
  1134→                    f"degraded_from_{target_tier.value}_"
  1135→                    f"to_{lower_model.tier.value}"
  1136→                ),
  1137→                estimated_tokens=self._estimate_tokens(atomic_step),
  1138→            )
  1139→
  1140→        # BC-008: Absolute fallback -- first LIGHT model
  1141→        first_light = self._get_first_light_model()
  1142→        logger.error(
  1143→            "All tiers exhausted for step=%s, using absolute LIGHT fallback: %s",
  1144→            atomic_step.value, first_light.display_name,
  1145→        )
  1146→        return RoutingDecision(
  1147→            atomic_step_type=atomic_step,
  1148→            model_config=first_light,
  1149→            provider=first_light.provider,
  1150→            tier=ModelTier.LIGHT,
  1151→            variant_type=variant_type,
  1152→            routing_reason="absolute_emergency_light_fallback",
  1153→            estimated_tokens=self._estimate_tokens(atomic_step),
  1154→        )

  1156→    def _get_allowed_tiers(self, variant_type: str) -> Set[ModelTier]:
  1157→        """Get allowed tiers for a variant type (SG-03).
  1158→
  1159→        Unknown variant types default to mini_parwa (safest).
  1160→        """
  1161→        if variant_type not in VARIANT_MODEL_ACCESS:
  1162→            logger.warning(
  1163→                "Unknown variant_type=%s, defaulting to mini_parwa (safest)",
  1164→                variant_type,
  1165→            )
  1166→            return VARIANT_MODEL_ACCESS["mini_parwa"]
  1167→        return VARIANT_MODEL_ACCESS[variant_type]

  1169→    def _get_step_tier(
  1170→        self,
  1171→        atomic_step: AtomicStepType,
  1172→        allowed_tiers: Set[ModelTier],
  1173→        query_signals: dict,
  1174→    ) -> ModelTier:
  1175→        """Pick the tier for a given atomic step.
  1176→
  1177→        Respects variant gating: if recommended tier is not allowed,
  1178→        degrades to the highest allowed tier.
  1179→        """
  1180→        recommended = STEP_TIER_MAPPING.get(atomic_step, ModelTier.LIGHT)
  1181→
  1182→        # GUARDRAIL tier is always allowed regardless of variant
  1183→        if recommended == ModelTier.GUARDRAIL:
  1184→            return ModelTier.GUARDRAIL
  1185→
  1186→        # Check if recommended tier is allowed by variant
  1187→        if recommended in allowed_tiers:
  1188→            return recommended
  1189→
  1190→        # Degrade to highest allowed tier
  1191→        for tier in TIER_FALLBACK_ORDER:
  1192→            if tier in allowed_tiers:
  1193→                logger.info(
  1194→                    "Step %s: recommended tier=%s not allowed by variant, "
  1195→                    "degraded to tier=%s",
  1196→                    atomic_step.value, recommended.value, tier.value,
  1197→                )
  1198→                return tier
  1199→
  1200→        # LIGHT is always allowed (it's in every variant)
  1201→        return ModelTier.LIGHT

  1203→    def _select_model(
  1204→        self,
  1205→        tier: ModelTier,
  1206→        query_signals: dict,
  1207→    ) -> Tuple[Optional[ModelConfig], List[ModelConfig]]:
  1208→        """Pick primary model + list of fallbacks for a tier.
  1209→
  1210→        Models within a tier are sorted by priority (1 = best).
  1211→        Returns (primary_model, fallback_models_list).
  1212→        """
  1213→        tier_models = [
  1214→            config for config in MODEL_REGISTRY.values()
  1215→            if config.tier == tier
  1216→        ]
  1217→        tier_models.sort(key=lambda m: m.priority)
  1218→
  1219→        primary: Optional[ModelConfig] = None
  1220→        fallbacks: List[ModelConfig] = []
  1221→
  1222→        for model in tier_models:
  1223→            if primary is None:
  1224→                if self._is_model_available(model):
  1225→                    primary = model
  1226→                else:
  1227→                    fallbacks.append(model)
  1228→            else:
  1229→                fallbacks.append(model)
  1230→
  1231→        # If no model is available, return first model anyway (BC-008)
  1232→        if primary is None and tier_models:
  1233→            primary = tier_models[0]
  1234→            fallbacks = tier_models[1:]
  1235→            logger.warning(
  1236→                "No available model in tier=%s, forcing primary: %s",
  1237→                tier.value, primary.display_name,
  1238→            )
  1239→
  1240→        return primary, fallbacks

  1248→    def _fallback_within_tier(self, tier: ModelTier) -> Optional[ModelConfig]:
  1249→        """Find next available model in the same tier."""

  1252→            if config.tier == tier
  1253→        ]
  1254→        tier_models.sort(key=lambda m: m.priority)
  1255→
  1256→        for model in tier_models:
  1257→            if self._is_model_available(model):
  1258→                return model

  1261→    def _fallback_to_lower_tier(
  1262→        self,
  1263→        current_tier: ModelTier,
  1264→        allowed_tiers: Set[ModelTier],
  1265→    ) -> Optional[ModelConfig]:
  1266→        """Degrade to the next lower available tier."""
  1267→        tier_order = [ModelTier.HEAVY, ModelTier.MEDIUM, ModelTier.LIGHT]
  1268→        try:
  1269→            current_idx = tier_order.index(current_tier)
  1270→        except ValueError:
  1271→            current_idx = 0
  1272→
  1273→        for i in range(current_idx + 1, len(tier_order)):
  1274→            lower_tier = tier_order[i]
  1275→            if lower_tier not in allowed_tiers:
  1276→                continue
  1277→            model = self._fallback_within_tier(lower_tier)
  1278→            if model is not None:
  1279→                return model
  1280→
  1281→        return None

  1367→    # ── LLM Execution ───────────────────────────────────────────
  1368→
  1369→    def _execute_llm_call_safe(
  1370→        self,
  1371→        company_id: str,
  1372→        routing_decision: RoutingDecision,
  1373→        messages: list,
  1374→        temperature: float,
  1375→        max_tokens: int,
  1376→    ) -> dict:
  1377→        """Execute LLM call via LiteLLM with retry + fallback (sync).
  1378→
  1379→        BC-001: company_id is always second parameter.
  1380→        BC-008: Always returns a dict with 'content' key.
  1381→        """
  1382→        config = routing_decision.model_config
  1383→        provider = config.provider
  1384→        model_id = config.model_id
  1385→
  1386→        # Try LiteLLM first, fall back to raw HTTP on auth/missing errors
  1387→        use_litellm = _HAS_LITELLM and litellm is not None
  1388→        api_key = self._get_api_key(provider) if use_litellm else None
  1389→
  1390→        for attempt in range(self.MAX_RETRIES + 1):
  1391→            try:
  1392→                # Acquire per-provider RPM slot (waits at 0, never switches)
  1393→                self._health.acquire_provider_slot(provider)
  1394→
  1395→                if use_litellm and api_key:
  1396→                    result = self._call_litellm_sync(
  1397→                        provider=provider,
  1398→                        model_id=model_id,
  1399→                        messages=messages,
  1400→                        temperature=temperature,
  1401→                        max_tokens=max_tokens,
  1402→                    )
  1403→                    self._health.record_success(
  1404→                        provider, model_id,
  1405→                        tokens_used=result.get("tokens_used", 0),
  1406→                    )
  1407→                    return result
  1408→                else:
  1409→                    # Fallback to raw HTTP if litellm not available or no API key
  1410→                    result = self._call_provider(
  1411→                        provider=provider,
  1412→                        model_id=model_id,
  1413→                        config=config,
  1414→                        messages=messages,
  1415→                        temperature=temperature,
  1416→                        max_tokens=max_tokens,
  1417→                        company_id=company_id,
  1418→                    )
  1419→                    self._health.record_success(
  1420→                        provider, model_id,
  1421→                        tokens_used=result.get("tokens_used", 0),
  1422→                    )
  1423→                    return result
  1424→            except RateLimitError as rle:
  1425→                self._health.record_rate_limit(
  1426→                    provider, model_id,
  1427→                    retry_after_seconds=rle.retry_after,
  1428→                )
  1429→                logger.warning(
  1430→                    "Rate limited: %s model=%s attempt=%d retry_after=%d",
  1431→                    provider.value, model_id, attempt + 1, rle.retry_after,
  1432→                )
  1433→                if attempt < self.MAX_RETRIES:
  1434→                    continue
  1435→            except Exception as exc:
  1436→                # If litellm auth fails, switch to raw HTTP for remaining retries
  1437→                if use_litellm and api_key and ("auth" in str(exc).lower() or "api_key" in str(exc).lower()):
  1438→                    logger.warning(
  1439→                        "LiteLLM auth failed, switching to raw HTTP: %s",
  1440→                        str(exc),
  1441→                    )
  1442→                    use_litellm = False
  1443→                    api_key = None
  1444→                    continue
  1445→                self._health.record_failure(provider, model_id, str(exc))
  1446→                logger.warning(
  1447→                    "Call failed: %s model=%s attempt=%d error=%s",
  1448→                    provider.value, model_id, attempt + 1, str(exc),
  1449→                )
  1450→                if attempt < self.MAX_RETRIES:
  1451→                    continue
  1452→
  1453→        # All retries exhausted, try fallback models
  1454→        for fb in routing_decision.fallback_models:
  1455→            if self._is_model_available(fb):
  1456→                try:
  1457→                    # Acquire RPM slot for the FALLBACK provider (may differ
  1458→                    # from primary provider — track per-provider, not per-model)
  1459→                    self._health.acquire_provider_slot(fb.provider)
  1460→
  1461→                    if _HAS_LITELLM and litellm is not None:
  1462→                        result = self._call_litellm_sync(
  1463→                            provider=fb.provider,
  1464→                            model_id=fb.model_id,
  1465→                            messages=messages,
  1466→                            temperature=temperature,
  1467→                            max_tokens=max_tokens,
  1468→                        )
  1469→                        self._health.record_success(fb.provider, fb.model_id)
  1470→                        result["fallback_used"] = True
  1471→                        return result
  1472→                except Exception as exc:
  1473→                    logger.warning(
  1474→                        "Fallback failed: %s model=%s error=%s",
  1475→                        fb.provider.value, fb.model_id, str(exc),
  1476→                    )
  1477→                    continue
  1478→
  1479→        # BC-008: Return empty fallback
  1480→        return {
  1481→            "content": "",
  1482→            "model": model_id,
  1483→            "provider": provider.value,
  1484→            "tier": routing_decision.tier.value,
  1485→            "atomic_step": routing_decision.atomic_step_type.value,
  1486→            "company_id": company_id,
  1487→            "fallback_used": True,
  1488→            "error": "All providers exhausted",
  1489→            "finish_reason": "error",
  1490→        }
  1491→
  1492→    async def _execute_llm_call_safe_async(
  1493→        self,
  1494→        company_id: str,
  1495→        routing_decision: RoutingDecision,
  1496→        messages: list,
  1497→        temperature: float,
  1498→        max_tokens: int,
  1499→    ) -> dict:
  1500→        """Execute LLM call via LiteLLM with retry + fallback (async).
  1501→
  1502→        BC-001: company_id is always second parameter.

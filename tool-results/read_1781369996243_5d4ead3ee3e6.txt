   123→   123→   123→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   124→   124→   124→  agentsReplaced: number;
   125→   125→   125→  humanCost: number;
   126→   126→   126→  savings: number;
   127→   127→   127→  savingsPercent: number;
   128→   128→   128→} {
   129→   129→   129→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   130→   130→   130→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   131→   131→   131→  const savings = Math.max(0, humanCost - monthlyCost);
   132→   132→   132→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   133→   133→   133→  return { agentsReplaced, humanCost, savings, savingsPercent };
   134→   134→   134→}

   291→   291→   291→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   292→   292→   292→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   293→   293→   293→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   294→   294→   294→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   295→   295→   295→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   296→   296→   296→  const [isSubmitting, setIsSubmitting] = useState(false);
   297→   297→   297→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   298→   298→   298→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   299→   299→   299→
   300→   300→   300→  // Restore add-ons from localStorage
   301→   301→   301→  useEffect(() => {
   302→   302→   302→    try {
   303→   303→   303→      const stored = localStorage.getItem('parwa_pricing_context');
   304→   304→   304→      if (stored) {
   305→   305→   305→        const ctx = JSON.parse(stored) as PricingContext;
   306→   306→   306→        if (ctx.addOns) {
   307→   307→   307→          setAddOns(ctx.addOns);
   308→   308→   308→        }
   309→   309→   309→      }
   310→   310→   310→    } catch {
   311→   311→   311→      // ignore
   312→   312→   312→    }
   313→   313→   313→  }, []);
   314→   314→   314→
   315→   315→   315→  // Check if Paddle.js is available
   316→   316→   316→  useEffect(() => {
   317→   317→   317→    getPaddleInstance().then((paddle) => {
   318→   318→   318→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   319→   319→   319→    }).catch(() => {
   320→   320→   320→      setPaddleStatus('unavailable');
   321→   321→   321→    });
   322→   322→   322→  }, []);
   323→   323→   323→
   324→   324→   324→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   325→   325→   325→
   326→   326→   326→  const totalTicketLimit = useMemo(
   327→   327→   327→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   328→   328→   328→    [activeVariants, variantQuantities]
   329→   329→   329→  );
   330→   330→   330→
   331→   331→   331→  const baseSubscription = useMemo(
   332→   332→   332→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   333→   333→   333→    [activeVariants, variantQuantities]
   334→   334→   334→  );
   335→   335→   335→
   336→   336→   336→  const addOnTotal = useMemo(() => {
   337→   337→   337→    let total = 0;
   338→   338→   338→    if (addOns.voice) {
   339→   339→   339→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   340→   340→   340→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   341→   341→   341→        total += voiceAddOn.price;
   342→   342→   342→      }
   343→   343→   343→    }
   344→   344→   344→    if (addOns.customApi) {
   345→   345→   345→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   346→   346→   346→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   347→   347→   347→        total += customApiAddOn.price;
   348→   348→   348→      }
   349→   349→   349→    }
   350→   350→   350→    return total;
   351→   351→   351→  }, [addOns, activeVariants]);
   352→   352→   352→
   353→   353→   353→  const totalMonthly = baseSubscription + addOnTotal;
   354→   354→   354→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   355→   355→   355→
   356→   356→   356→  // Overage projection (estimates at 80% utilization)
   357→   357→   357→  const projectedTickets = Math.round(totalTicketLimit * 0.8);
   358→   358→   358→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);
   359→   359→   359→  const overageCost = overageTickets * OVERAGE_PRICE_PER_TICKET;
   360→   360→   360→
   361→   361→   361→  const toggleVariant = (tier: VariantTier) => {
   362→   362→   362→    setActiveVariants((prev) => {
   363→   363→   363→      if (prev.includes(tier)) {
   364→   364→   364→        if (prev.length <= 1) {
   365→   365→   365→          toast.error('You must have at least one active variant');
   366→   366→   366→          return prev;
   367→   367→   367→        }
   368→   368→   368→        return prev.filter((t) => t !== tier);
   369→   369→   369→      }
   370→   370→   370→      return [...prev, tier];
   371→   371→   371→    });
   372→   372→   372→  };
   373→   373→   373→
   374→   374→   374→  const updateQuantity = (tier: VariantTier, qty: number) => {
   375→   375→   375→    const newQty = Math.max(1, Math.min(qty, 10));
   376→   376→   376→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   377→   377→   377→  };
   378→   378→   378→
   379→   379→   379→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   380→   380→   380→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   381→   381→   381→  };
   382→   382→   382→

    47→def _format_notification(ntype: str, ev: Dict) -> tuple:
    48→    """Format notification title and description by type."""
    49→    signal = ev.get("signal", {})
    50→    score = ev.get("priority_score", 0)
    51→
    52→    if ntype == TYPE_STUCK_TICKET:
    53→        tid = signal.get("ticket_id", "unknown")
    54→        reason = signal.get("reason", "unknown")
    55→        quality = signal.get("quality_score", "?")
    56→        escalation = ev.get("escalation_tier", "soft_reminder")
    57→        hours = signal.get("hours_stuck", 0)
    58→        title = f"Stuck Ticket [{escalation.upper()}]: {tid}"
    59→        desc = (f"Ticket {tid} could not be resolved automatically. "
    60→                f"Reason: {reason}. Quality achieved: {quality}. "
    61→                f"Loops used: {signal.get('loops_used', 0)}. "
    62→                f"Hours stuck: {hours}. Escalation: {escalation}. "
    63→                f"Requires manual review.")
    64→
    65→    elif ntype == TYPE_QUOTA_LOW:
    66→        data = list(signal.values())[0] if signal else {}
    67→        tier = list(signal.keys())[0] if signal else "unknown"
    68→        title = f"Quota {data.get('status', 'warning').upper()}: {tier}"
    69→        desc = (f"{tier.capitalize()} plan quota is {data.get('burn_pct', 0)}% used "
    70→                f"({data.get('remaining', 0)} of {data.get('total', 0)} remaining). "
    71→                f"Consider upgrading or requesting additional quota.")
    72→
    73→    elif ntype == TYPE_ACCURACY_DROP:
    74→        trend = signal.get("trend", "unknown")
    75→        severity = signal.get("severity", "warning")
    76→        trigger = signal.get("trigger", "unknown")
    77→        acc_7d = signal.get("accuracy_7d", "?")
    78→        acc_today = signal.get("accuracy_today", "?")
    79→        title = f"Accuracy Drift [{severity.upper()}]: {trend}"
    80→        desc = (f"Resolution accuracy trend is '{trend}' (severity: {severity}). "
    81→                f"7-day avg: {acc_7d}, Today: {acc_today}. "
    82→                f"Trigger: {trigger}. "
    83→                f"This may indicate KB gaps, policy changes, or model drift. "
    84→                f"Review recent resolutions and update knowledge base if needed.")
    85→
    86→    elif ntype == TYPE_INTEGRATION_DOWN:
    87→        degraded = signal.get("degraded_services", [])
    88→        worst_uptime = signal.get("worst_uptime_pct", 0)
    89→        names = ", ".join(f"{d['name']} ({d['status']}, uptime={d.get('uptime_pct', '?')}%)" for d in degraded)
    90→        title = f"Integration Issues: {signal.get('total_degraded', len(degraded))} service(s) degraded"
    91→        desc = (f"Degraded integrations: {names}. "
    92→                f"Worst uptime: {worst_uptime}%. "
    93→                f"Check API credentials and service status.")
    94→
    95→    elif ntype == "load_bottleneck":
    96→        at_cap = signal.get("at_capacity", [])
    97→        high_load = signal.get("high_load", [])
    98→        vip = signal.get("vip_overflow_risk", False)
    99→        bottlenecks = ", ".join(v["name"] for v in at_cap + high_load)
   100→        title = f"Load Bottleneck: {bottlenecks or 'All variants'}"
   101→        desc = (f"Variant load issue detected. "
   102→                f"At capacity: {', '.join(v['name'] for v in at_cap) or 'none'}. "
   103→                f"High load: {', '.join(v['name'] for v in high_load) or 'none'}. "
   104→                f"VIP overflow risk: {vip}. "
   105→                f"Consider scaling up or routing overflow to available variants.")
   106→
   107→    else:
   108→        title = f"Signal: {ntype}"
   109→        desc = str(signal)
   110→
   111→    return title, desc

   114→async def _create_notifications_from_evals(
   115→    tenant_id: str,
   116→    evaluations: List[Dict[str, Any]],
   117→    db,
   118→) -> List[Dict[str, Any]]:
   119→    """Convert evaluations into notifications via DB. Filter LOW priority."""
   120→    notifications = []
   121→    for ev in evaluations:
   122→        score = ev.get("priority_score", 0)
   123→        if score < 0.40:
   124→            logger.info("LOW priority signal filtered: %s (score=%.3f)", ev.get("type"), score)
   125→            continue
   126→
   127→        ntype = ev.get("type", "unknown")
   128→        title, desc = _format_notification(ntype, ev)
   129→        related = ev.get("related_tickets", [])
   130→
   131→        nf = await db.create_notification(
   132→            tenant_id=tenant_id,
   133→            ntype=ntype,
   134→            priority_score=score,
   135→            title=title,
   136→            description=desc,
   137→            related_tickets=related,
   138→            batch_key=ev.get("type", ""),
   139→            source_data=ev.get("signal", {}),
   140→        )
   141→        notifications.append(nf)
   142→
   143→    return notifications

   146→# ── Command Execution (NEW — Wave 1) ─────────────────────────
   147→
   148→async def _execute_command(
   149→    intent_result: Dict[str, Any],
   150→    auth_result: AuthResult,
   151→    tenant_id: str,
   152→    signals: Dict[str, Any],
   153→    db,
   154→    raw_input: str = "",
   155→) -> str:
   156→    """Execute a classified command and return the response text.
   157→
   158→    This is the core "Jarvis does things" function.
   159→    Reads/writes DB based on intent. Returns human-readable response.
   160→    """
   161→    intent = intent_result["intent"]
   162→    target = intent_result["target"]
   163→
   164→    # ── QUERY INTENTS ────────────────────────────────────
   165→    if is_query_intent(intent):
   166→        response = await _handle_query(intent, target, tenant_id, signals, db)
   167→        # Log to audit
   168→        await db.create_audit_entry(
   169→            tenant_id=tenant_id,
   170→            action=f"query:{intent}",
   171→            actor_email=auth_result.email,
   172→            target_type="query",
   173→            payload={"intent": intent, "target": target},
   174→        )
   175→        return response
   176→
   177→    # ── CONTROL INTENTS ──────────────────────────────────
   178→    if is_control_intent(intent):
   179→        response = await _handle_control(intent, target, tenant_id, auth_result, db,
   180→                                         raw_input=raw_input)
   181→        return response
   182→
   183→    # ── APPROVAL INTENTS ─────────────────────────────────
   184→    if is_approval_intent(intent):
   185→        response = await _handle_approval(intent, target, tenant_id, auth_result, db)
   186→        return response
   187→
   188→    # ── EMERGENCY INTENTS ────────────────────────────────
   189→    if is_emergency_intent(intent):
   190→        response = await _handle_emergency(intent, target, tenant_id, auth_result, db,
   191→                                          raw_input=raw_input)
   192→        return response
   193→
   194→    # ── EXPLAIN / TEACH / AGENT ──────────────────────────
   195→    if intent == "explain_ticket":
   196→        return await _handle_explain_ticket(target, tenant_id, signals, db)
   197→    if intent == "explain_flag":
   198→        return await _handle_explain_flags(tenant_id, db)
   199→    if intent == "teach_skill":
   200→        return await _handle_teach_skill(target, tenant_id, auth_result, db)
   201→    if intent == "create_agent":
   202→        return await _handle_create_agent(target, tenant_id, auth_result, db)
   203→    if intent == "control_approval_override":
   204→        # Wave 3: approval override via executor
   205→        result = await execute_command(
   206→            intent=intent, target=target, tenant_id=tenant_id,
   207→            actor_email=auth_result.email, raw_input=raw_input,
   208→        )
   209→        return result.response
   210→
   211→    return f"I understood your command but don't have a handler for '{intent}' yet."

   216→async def _handle_query(intent: str, target: str, tenant_id: str, signals: Dict, db) -> str:

   219→    if intent == "query_status":

   238→        flow = signals.get("ticket_flow", {})

   249→            f"Avg Quality: {summary.get('avg_quality', 'N/A')}\n"
   250→            f"Avg LLM Calls: {summary.get('avg_llm_calls', 'N/A')}\n"
   251→            f"By Path: {summary.get('by_type', {})}\n"
   252→            f"Accuracy Trend: {drift.get('trend_direction', 'no_data')}"
   253→        )

   266→            f"Auto-Resolved: {stats['auto_resolved']}\n"
   267→            f"Escalated: {stats['escalated']}\n"
   268→            f"Stuck: {stats['stuck']}\n"
   269→            f"By Path: {stats.get('by_path', {})}"

   282→            )

   284→
   285→    if intent == "query_notifications":
   286→        if target and target.startswith("PARWA-NFY"):
   287→            # Specific notification lookup
   288→            nf = await db.get_notification(target)
   289→            if nf:
   290→                return (
   291→                    f"**{nf['title']}** (Key: {nf['notification_key']})\n\n"
   292→                    f"{nf['description']}\n\n"
   293→                    f"Priority: {nf['priority']} (score: {nf['priority_score']})\n"

   341→        services = health.get("services", {})
   342→        if not services:

   376→    if intent == "query_flow":

   379→        by_node = summary.get("by_node", {})

   381→            f"**Ticket Flow Metrics**:",
   382→            f"  Total: {summary.get('total', 0)}",

   422→                f"reason={s.get('reason', '?')}, "
   423→                f"quality={s.get('quality_score', '?')}, "
   424→                f"{hours}h stuck, loops={s.get('loops_used', 0)}"
   425→            )
   426→        return "\n".join(lines)
   427→
   428→    return f"Query '{intent}' not yet implemented."

   431→# ── Control Handlers ──────────────────────────────────────────
   432→
   433→async def _handle_control(intent: str, target: str, tenant_id: str, auth: AuthResult, db,
   434→                           raw_input: str = "") -> str:
   435→    """Handle control commands via command_executor (Wave 3).
   436→
   437→    5-step pipeline: validate → resolve → execute → verify → respond.
   438→    All conflicts auto-resolved. All actions audited.
   439→    """
   440→    result = await execute_command(
   441→        intent=intent,
   442→        target=target,
   443→        tenant_id=tenant_id,
   444→        actor_email=auth.email,
   445→        raw_input=raw_input,
   446→    )
   447→
   448→    # Build response with warnings and conflict info
   449→    response = result.response
   450→    if result.warnings:
   451→        response += "\n\nWarnings:\n" + "\n".join(f"  - {w}" for w in result.warnings)
   452→    if result.conflicts_resolved:
   453→        names = [f["flag_type"] + "=" + f["flag_value"] for f in result.conflicts_resolved]
   454→        response += f"\n\nAuto-resolved conflicts: {', '.join(names)}"
   455→    if result.undo_id:
   456→        response += f"\n\nTo undo: **'disable my last rule'**"
   457→
   458→    return response

   463→async def _handle_approval(intent: str, target: str, tenant_id: str, auth: AuthResult, db) -> str:
   464→    """Handle approval/rejection commands."""
   465→    await db.create_audit_entry(
   466→        tenant_id=tenant_id, action=intent,
   467→        actor_email=auth.email, target_type="approval", target_id=target,

   469→    )

   471→        action = "approved" if "approve" in intent else "rejected"
   472→        return f"[OK] Batch {action}. Full batch approval UI coming in Wave 7 (Jarvis UI)."
   473→    else:
   474→        action = "approved" if "approve" in intent else "rejected"
   475→        return f"[OK] Ticket {target} {action}. Ticket-level approval tracking coming in Wave 5."

   491→        raw_input=raw_input,
   492→    )
   493→
   494→    response = result.response

   496→        response += "\n\nWarnings:\n" + "\n".join(f"  - {w}" for w in result.warnings)
   497→    if result.undo_id and intent != "emergency_shutdown":
   498→        response += f"\n\nTo undo: **'disable my last rule'**"
   499→
   500→    return response

   555→# ── Main Node Function ────────────────────────────────────────
   556→
   557→
   558→async def jarvis_notify(state: dict) -> dict:
   559→    """Jarvis Node 3: NOTIFY — Act + Verify.
   560→
   561→    Wave 1: Now wired with command_parser + jarvis_auth + jarvis_db.
   562→    Full pipeline: chat → parse → auth → execute → DB → response.
   563→    """
   564→    start = time.time()
   565→    tenant_id = state.get("tenant_id", "")
   566→    trigger = state.get("trigger", "poll")
   567→    evaluations = state.get("evaluations", [])
   568→    signals = state.get("signals", {})
   569→    admin_question = state.get("admin_question", "")
   570→    user_context = state.get("user_context", {})
   571→    logs = []
   572→    llm_calls = 0
   573→
   574→    db = get_db()
   575→
   576→    # 1. Create notifications from evaluations (poll/monitor mode)
   577→    notifications = []
   578→    if trigger in ("poll", "stuck_ticket") and evaluations:
   579→        notifications = await _create_notifications_from_evals(tenant_id, evaluations, db)
   580→        logs.append({"node": "J3", "technique": "CreateNotifications", "duration_ms": 0,
   581→                     "result_summary": f"created={len(notifications)}"})
   582→
   583→    # 2. Handle admin chat (full pipeline: parse → auth → execute)
   584→    chat_response = ""
   585→    if trigger == "admin_chat" and admin_question:
   586→        # Step 1: Classify intent
   587→        intent_result = await classify_command(admin_question)
   588→        if intent_result.get("classification_method") == "llm":
   589→            llm_calls += 1
   590→        logs.append({"node": "J3", "technique": "CommandParser",
   591→                     "duration_ms": 0, "result_summary": f"intent={intent_result['intent']} "
   592→                     f"method={intent_result['classification_method']} "
   593→                     f"confidence={intent_result['confidence']}"} )
   594→
   595→        # Step 2: Authorize
   596→        auth_result = await authorize_command(
   597→            intent=intent_result["intent"],
   598→            user_context=user_context,
   599→            tenant_id=tenant_id,
   600→        )
   601→        logs.append({"node": "J3", "technique": "AuthCheck",
   602→                     "duration_ms": 0, "result_summary": f"authorized={auth_result.authorized} "
   603→                     f"role={auth_result.role}"})
   604→
   605→        # Step 3: Execute (or deny)
   606→        if auth_result.authorized:
   607→            chat_response = await _execute_command(
   608→                intent_result=intent_result,
   609→                auth_result=auth_result,
   610→                tenant_id=tenant_id,
   611→                signals=signals,
   612→                db=db,
   613→                raw_input=admin_question,
   614→            )
   615→            logs.append({"node": "J3", "technique": "ExecuteCommand",
   616→                         "duration_ms": 0, "result_summary": f"executed={intent_result['intent']}"})
   617→        else:
   618→            chat_response = f"[DENIED] {auth_result.reason}"
   619→            logs.append({"node": "J3", "technique": "AuthDenied",
   620→                         "duration_ms": 0, "result_summary": auth_result.reason})
   621→
   622→    # 3. Build quota feedback for PARWA Node 2
   623→    quota_feedback = {
   624→        "tenant_id": tenant_id,
   625→        "quota_status": signals.get("quota_status", {}),
   626→        "variant_tier": list(signals.get("quota_status", {}).keys())[0] if signals.get("quota_status") else "parwa",
   627→    }
   628→
   629→    # 4. Summary
   630→    keys = [n["notification_key"] for n in notifications]
   631→    elapsed = int((time.time() - start) * 1000)
   632→    logger.info("Jarvis NOTIFY complete: tenant=%s notifications=%d chat=%s llm=%d auth=%s [%dms]",
   633→                tenant_id, len(notifications), bool(chat_response), llm_calls,
   634→                bool(user_context.get("role")), elapsed)
   635→
   636→    return {
   637→        "notifications": notifications,
   638→        "notification_keys": keys,
   639→        "chat_response": chat_response,
   640→        "quota_feedback": quota_feedback,
   641→        "notify_log": logs,
   642→        "total_token_usage": state.get("total_token_usage", 0) + llm_calls,
   643→        "intent_result": intent_result if trigger == "admin_chat" else None,
   644→        "auth_result": {"authorized": auth_result.authorized, "role": auth_result.role}
   645→                        if trigger == "admin_chat" else None,
   646→    }

   610→   610→        self,
   611→   611→        tenant_id: str,
   612→   612→        batch_key: str,
   613→   613→        ticket_id: str,
   614→   614→        confidence: float,
   615→   615→    ) -> Optional[Dict[str, Any]]:
   616→   616→        BATCH_WINDOW_S = 300
   617→   617→        now = time.time()

   634→   634→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   635→   635→            # Don't flush yet — still within window
   636→   636→            return None
   637→   637→        else:
   638→   638→            meta = {
   639→   639→                "batch_key": batch_key,
   640→   640→                "tenant_id": tenant_id,
   641→   641→                "batch_start": now,
   642→   642→                "ticket_ids": [ticket_id],
   643→   643→                "signal_count": 1,
   644→   644→                "confidence_min": confidence,
   645→   645→                "confidence_max": confidence,

   701→   701→        """Compute health from recent pings (last 30 per service)."""
   702→   702→        tenant_pings = [p for p in self._integration_pings if p["tenant_id"] == tenant_id]
   703→   703→        services: Dict[str, List[Dict]] = {}

   706→   706→

   717→   717→            last_error = None

   722→   722→                    last_error_time = p["created_at"]
   723→   723→                    break
   724→   724→
   725→   725→            status = "healthy" if uptime_pct >= 90 else ("degraded" if uptime_pct >= 50 else "down")
   726→   726→            if status != "healthy":
   727→   727→                result["degraded_count"] += 1
   728→   728→            else:
   729→   729→                result["healthy_count"] += 1
   730→   730→
   731→   731→            result["services"][svc] = {
   732→   732→                "status": status,
   733→   733→                "uptime_pct": uptime_pct,
   734→   734→                "total_pings": total,
   735→   735→                "healthy_pings": healthy,
   736→   736→                "avg_response_ms": round(avg_ms, 1),
   737→   737→                "last_ping_at": last_ping["created_at"],
   738→   738→                "last_healthy_at": next(
   739→   739→                    (p["created_at"] for p in reversed(recent) if p["is_healthy"]),
   740→   740→                    None,
   741→   741→                ),

   828→   828→    # ── Drift Detection (Wave 2) ─────────────────────────────
   829→   829→
   830→   830→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
   831→   831→        """Analyze quality_scores for drift using day-over-day comparison."""
   832→   832→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   833→   833→        if len(tenant_scores) < 3:
   834→   834→            return {
   835→   835→                "tenant_id": tenant_id, "drift_detected": False,
   836→   836→                "drift_severity": "none", "accuracy_7d": None,
   837→   837→                "accuracy_yesterday": None, "accuracy_today": None,
   838→   838→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
   839→   839→                "total_scores": len(tenant_scores),
   840→   840→            }
   841→   841→        # Group by date (ISO date portion)
   842→   842→        by_date: Dict[str, List[float]] = {}
   843→   843→        for s in tenant_scores:
   844→   844→            date_str = s["created_at"][:10]  # YYYY-MM-DD
   845→   845→            by_date.setdefault(date_str, []).append(s["overall_score"])
   846→   846→
   847→   847→        dates_sorted = sorted(by_date.keys())
   848→   848→
   849→   849→        def _avg(scores: List[float]) -> float:
   850→   850→            return round(sum(scores) / len(scores), 4) if scores else 0.0
   851→   851→
   852→   852→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
   853→   853→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
   854→   854→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
   855→   855→
   856→   856→        # Drift detection: >5% drop for 3+ days
   857→   857→        drift_detected = False
   858→   858→        drift_severity = "none"
   859→   859→        trigger_reason = None
   860→   860→        trend_direction = "stable"
   861→   861→
   862→   862→        if len(dates_sorted) >= 3 and accuracy_7d is not None and accuracy_today is not None:
   863→   863→            recent_3 = dates_sorted[-3:]
   864→   864→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
   865→   865→            # Check if declining 3+ consecutive days
   866→   866→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
   867→   867→            drop_7d = accuracy_7d - accuracy_today
   868→   868→
   869→   869→            if declining and drop_7d > 0.05:
   870→   870→                drift_detected = True
   871→   871→                trend_direction = "declining"
   872→   872→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
   873→   873→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
   874→   874→            elif drop_7d > 0.03:
   875→   875→                drift_detected = True
   876→   876→                trend_direction = "slight_decline"
   877→   877→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
   878→   878→                drift_severity = "warning"
   879→   879→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
   880→   880→                trend_direction = "improving"
   881→   881→            else:
   882→   882→                trend_direction = "stable"
   883→   883→
   884→   884→        # Check for repeated errors (same low score pattern)
   885→   885→        if not drift_detected and tenant_scores:
   886→   886→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
   887→   887→            if len(low_scores) >= 3:
   888→   888→                # Check if same resolution_path failing
   889→   889→                paths: Dict[str, int] = {}
   890→   890→                for s in low_scores:
   891→   891→                    p = s.get("resolution_path", "unknown")
   892→   892→                    paths[p] = paths.get(p, 0) + 1
   893→   893→                worst_path = max(paths.items(), key=lambda x: x[1])
   894→   894→                if worst_path[1] >= 3:
   895→   895→                    drift_detected = True
   896→   896→                    drift_severity = "warning"
   897→   897→                    trend_direction = "declining"
   898→   898→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
   899→   899→
   900→   900→        return {
   901→   901→            "tenant_id": tenant_id,
   902→   902→            "drift_detected": drift_detected,
   903→   903→            "drift_severity": drift_severity,
   904→   904→            "accuracy_7d": accuracy_7d,
   905→   905→            "accuracy_yesterday": accuracy_yesterday,
   906→   906→            "accuracy_today": accuracy_today,
   907→   907→            "trend_direction": trend_direction,
   908→   908→            "trigger_reason": trigger_reason,
   909→   909→            "total_scores": len(tenant_scores),
   910→   910→        }

   121→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   122→
   123→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   124→  agentsReplaced: number;
   125→  humanCost: number;
   126→  savings: number;
   127→  savingsPercent: number;
   128→} {
   129→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   130→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   131→  const savings = Math.max(0, humanCost - monthlyCost);
   132→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   133→  return { agentsReplaced, humanCost, savings, savingsPercent };
   134→}

   136→// ── Usage Bar Component ────────────────────────────────────────────────
   137→
   138→function UsageBar({
   139→  used,
   140→  total,
   141→  label,
   142→  unit,
   143→  overageRate,
   144→}: {
   145→  used: number;
   146→  total: number;
   147→  label: string;
   148→  unit: string;
   149→  overageRate?: string;
   150→}) {
   151→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
   152→  const isOverLimit = used > total;
   153→
   154→  return (
   155→    <div className="space-y-1.5">
   156→      <div className="flex items-center justify-between">
   157→        <span className="text-[10px] text-orange-200/40">{label}</span>
   158→        <span className={cn(
   159→          'text-[10px] font-medium',
   160→          isOverLimit ? 'text-red-400' : 'text-orange-200/50'
   161→        )}>
   162→          {used.toLocaleString()} / {total.toLocaleString()} {unit}
   163→        </span>
   164→      </div>
   165→      <div className="h-1.5 bg-white/5 rounded-full overflow-hidden">
   166→        <div
   167→          className={cn(
   168→            'h-full rounded-full transition-all duration-700',
   169→            isOverLimit ? 'bg-gradient-to-r from-red-500 to-red-400' : 'bg-gradient-to-r from-emerald-500 to-emerald-400'
   170→          )}
   171→          style={{ width: `${pct}%` }}
   172→        />
   173→      </div>
   174→      {isOverLimit && overageRate && (
   175→        <p className="text-[9px] text-red-400 flex items-center gap-1">
   176→          <AlertTriangle className="w-2.5 h-2.5" />
   177→          Over limit — {(used - total).toLocaleString()} overage at {overageRate}/ticket
   178→        </p>
   179→      )}
   180→    </div>
   181→  );
   182→}

   289→// ── Component ──────────────────────────────────────────────────────────
   290→
   291→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   292→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   293→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   294→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   295→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   296→  const [isSubmitting, setIsSubmitting] = useState(false);
   297→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   298→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   299→
   300→  // Restore add-ons from localStorage
   301→  useEffect(() => {
   302→    try {
   303→      const stored = localStorage.getItem('parwa_pricing_context');
   304→      if (stored) {
   305→        const ctx = JSON.parse(stored) as PricingContext;
   306→        if (ctx.addOns) {
   307→          setAddOns(ctx.addOns);
   308→        }
   309→      }
   310→    } catch {
   311→      // ignore
   312→    }
   313→  }, []);
   314→
   315→  // Check if Paddle.js is available
   316→  useEffect(() => {
   317→    getPaddleInstance().then((paddle) => {
   318→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   319→    }).catch(() => {
   320→      setPaddleStatus('unavailable');
   321→    });
   322→  }, []);
   323→
   324→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   325→
   326→  const totalTicketLimit = useMemo(
   327→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   328→    [activeVariants, variantQuantities]
   329→  );
   330→
   331→  const baseSubscription = useMemo(
   332→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   333→    [activeVariants, variantQuantities]
   334→  );
   335→
   336→  const addOnTotal = useMemo(() => {
   337→    let total = 0;
   338→    if (addOns.voice) {
   339→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   340→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   341→        total += voiceAddOn.price;
   342→      }
   343→    }
   344→    if (addOns.customApi) {
   345→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   346→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   347→        total += customApiAddOn.price;
   348→      }
   349→    }
   350→    return total;
   351→  }, [addOns, activeVariants]);
   352→
   353→  const totalMonthly = baseSubscription + addOnTotal;
   354→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   355→
   356→  // Overage projection (estimates at 80% utilization)
   357→  const projectedTickets = Math.round(totalTicketLimit * 0.8);
   358→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);
   359→  const overageCost = overageTickets * OVERAGE_PRICE_PER_TICKET;
   360→
   361→  const toggleVariant = (tier: VariantTier) => {
   362→    setActiveVariants((prev) => {
   363→      if (prev.includes(tier)) {
   364→        if (prev.length <= 1) {
   365→          toast.error('You must have at least one active variant');
   366→          return prev;
   367→        }
   368→        return prev.filter((t) => t !== tier);
   369→      }
   370→      return [...prev, tier];
   371→    });
   372→  };
   373→
   374→  const updateQuantity = (tier: VariantTier, qty: number) => {
   375→    const newQty = Math.max(1, Math.min(qty, 10));
   376→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   377→  };
   378→
   379→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   380→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   381→  };
   382→
   383→  const handleProceed = async () => {
   384→    setIsSubmitting(true);
   385→    setCheckoutError(null);
   386→
   387→    try {
   388→      // Update pricing context in localStorage
   389→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   390→      const context: PricingContext = {
   391→        industry: 'other',
   392→        variant: primaryOnboardingVariant,
   393→        addOns,
   394→        totalMonthly,
   395→        timestamp: new Date().toISOString(),
   396→      };
   397→
   398→      // Preserve industry from existing context
   399→      try {
   400→        const stored = localStorage.getItem('parwa_pricing_context');
   401→        if (stored) {
   402→          const existing = JSON.parse(stored) as PricingContext;
   403→          context.industry = existing.industry;
   404→        }
   405→      } catch {
   406→        // ignore
   407→      }
   408→
   409→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   410→
   411→      // ── Paddle Checkout Flow ──────────────────────────────────────
   412→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   413→
   414→      // Add each active variant as a Paddle line item (with quantity)
   415→      for (const tier of activeVariants) {
   416→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   417→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   418→        const qty = variantQuantities[tier] || 1;
   419→        if (variantPriceId) {
   420→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   421→        }
   422→      }
   423→
   424→      // Voice add-on (only for starter where it's not included)
   425→      if (addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t))) {
   426→        checkoutItems.push({ priceId: 'pri_voice_addon_01', quantity: 1 });
   427→      }
   428→
   429→      // Custom API add-on (only for variants where it's not included)
   430→      if (addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t))) {
   431→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });
   432→      }
   433→
   434→      const customData = {
   435→        source: 'parwa_onboarding',
   436→        variant: primaryOnboardingVariant,
   437→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   438→        variantQuantities,
   439→        addOns,
   440→        industry: context.industry,
   441→        totalMonthly,
   442→      };
   443→
   444→      // ── Step 1: Try server-side checkout (backend creates Paddle transaction) ──
   445→      let checkoutUrl: string | null = null;
   446→
   447→      try {
   448→        const res = await fetch('/api/onboarding/checkout', {
   449→          method: 'POST',
   450→          headers: { 'Content-Type': 'application/json' },
   451→          body: JSON.stringify({
   452→            variant: primaryOnboardingVariant,
   453→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   454→            variantQuantities,
   455→            addOns,
   456→            totalMonthly,
   457→            industry: context.industry,
   458→          }),
   459→        });
   460→
   461→        if (res.ok) {
   462→          const checkoutData = await res.json();
   463→          checkoutUrl = checkoutData.checkout_url || null;
   464→        }
   465→      } catch {
   466→        // API unavailable — continue with client-side Paddle checkout
   467→      }
   468→
   469→      // If we got a checkout_url from backend, redirect to it
   470→      if (checkoutUrl) {
   471→        localStorage.setItem('parwa_payment_pending', JSON.stringify({
   472→          activeVariants, addOns, totalMonthly, variantQuantities,
   473→          pendingAt: new Date().toISOString(),
   474→        }));
   475→        window.location.href = checkoutUrl;
   476→        return;
   477→      }
   478→
   479→      // ── Step 2: Client-side Paddle checkout (items-based overlay) ──
   480→      // Re-check Paddle availability (might have loaded since initial check)
   481→      const paddle = await getPaddleInstance();
   482→
   483→      if (paddle && checkoutItems.length > 0) {
   484→        setPaddleStatus('ready');
   485→        const opened = await openCheckoutWithItems(
   486→          checkoutItems,
   487→          customData,
   488→          () => {
   489→            // Payment completed successfully
   490→            localStorage.removeItem('parwa_payment_pending');
   491→            toast.success('Payment successful! Welcome to PARWA!');
   492→            onComplete();
   493→          },
   494→          () => {
   495→            // User closed the checkout overlay without paying
   496→            toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   497→            setIsSubmitting(false);
   498→          },
   499→        );
   500→        if (opened) {
   501→          return; // Paddle overlay is showing — wait for user to complete or close
   502→        }
   503→      }
   504→
   505→      // ── Step 3: Paddle unavailable — try re-initializing ──
   506→      // Sometimes Paddle fails on first load but works on retry
   507→      try {
   508→        const retryPaddle = await getPaddleInstance();
   509→        if (retryPaddle && checkoutItems.length > 0) {
   510→          setPaddleStatus('ready');
   511→          const opened = await openCheckoutWithItems(
   512→            checkoutItems,
   513→            customData,
   514→            () => {
   515→              localStorage.removeItem('parwa_payment_pending');
   516→              toast.success('Payment successful! Welcome to PARWA!');
   517→              onComplete();
   518→            },
   519→            () => {
   520→              toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   521→              setIsSubmitting(false);
   522→            },
   523→          );
   524→          if (opened) return;
   525→        }
   526→      } catch {
   527→        // Paddle truly unavailable
   528→      }
   529→
   530→      // ── Payment gateway unavailable — BLOCK, don't skip ──
   531→      setPaddleStatus('unavailable');
   532→      localStorage.setItem('parwa_payment_pending', JSON.stringify({
   533→        activeVariants, addOns, totalMonthly, variantQuantities,
   534→        pendingAt: new Date().toISOString(),
   535→      }));
   536→      setCheckoutError(
   537→        'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   538→        'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'
   539→      );
   540→      toast.error('Payment gateway unavailable — please try again or contact support.');
   541→    } catch (err) {
   542→      console.error('[cost-breakdown] Error:', err);
   543→      setCheckoutError('Something went wrong. Please try again.');
   544→      toast.error('Something went wrong. Please try again.');
   545→    } finally {
   546→      setIsSubmitting(false);
   547→    }
   548→  };
   549→
   550→  return (
   551→    <div className="space-y-6">
   552→      {/* Header */}
   553→      <div className="text-center space-y-2">
   554→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   555→          <Receipt className="w-7 h-7 text-emerald-400" />
   556→        </div>
   557→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   558→        <p className="text-orange-200/40 text-sm">
   559→          Configure your plan and complete payment to activate your AI agents.
   560→        </p>
   561→      </div>
   562→
   563→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   564→      {paddleStatus === 'unavailable' && (
   565→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   566→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   567→          <div>
   568→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   569→            <p className="text-[10px] text-orange-200/30 mt-1">
   570→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   571→            </p>
   572→          </div>
   573→        </div>
   574→      )}
   575→      {paddleStatus === 'ready' && (
   576→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   577→          <Shield className="w-4 h-4 text-emerald-400" />
   578→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   579→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   580→        </div>
   581→      )}
   582→
   583→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   584→      <div className="space-y-3">
   585→        <div className="flex items-center gap-2">
   586→          <Brain className="w-4 h-4 text-orange-400" />
   587→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   588→            Active Variants
   589→          </label>
   590→          <span className="text-[10px] text-orange-200/25">
   591→            Each variant adds its own ticket allocation and AI pipeline
   592→          </span>
   593→        </div>
   594→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   595→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   596→            <VariantMixerCard
   597→              key={tier}
   598→              tier={tier}
   599→              isActive={activeVariants.includes(tier)}
   600→              quantity={variantQuantities[tier] || 1}
   601→              onToggle={() => toggleVariant(tier)}
   602→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   603→            />
   604→          ))}
   605→        </div>
   606→      </div>
   607→
   608→      {/* ── 2. Live Usage Bar & Overage Projection ──────────────────── */}
   609→      <div className="rounded-xl border border-white/[0.06] p-4 space-y-4" style={{ background: 'rgba(255,255,255,0.03)' }}>
   610→        <div className="flex items-center gap-2">
   611→          <BarChart3 className="w-4 h-4 text-orange-400" />
   612→          <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   613→            Live Usage & Overage Projection
   614→          </span>
   615→        </div>
   616→        <UsageBar
   617→          used={projectedTickets}
   618→          total={totalTicketLimit}
   619→          label="Ticket Usage (projected at 80%)"
   620→          unit="tickets"
   621→          overageRate={`$${OVERAGE_PRICE_PER_TICKET}`}
   622→        />
   623→        {overageCost > 0 && (
   624→          <div className="rounded-lg border border-red-500/20 p-3 flex items-start gap-2" style={{ background: 'rgba(239,68,68,0.05)' }}>
   625→            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0 mt-0.5" />
   626→            <div>
   627→              <p className="text-xs font-medium text-red-400">Overage Projected</p>
   628→              <p className="text-[10px] text-orange-200/30 mt-0.5">
   629→                At 80% utilization, projected overage: ${overageCost.toFixed(2)} ({overageTickets.toLocaleString()} tickets × ${OVERAGE_PRICE_PER_TICKET}/ticket). Overage charged at end of billing cycle.
   630→              </p>
   631→            </div>
   632→          </div>
   633→        )}
   634→        {overageCost === 0 && (
   635→          <div className="rounded-lg border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   636→            <Shield className="w-3.5 h-3.5 text-emerald-400" />
   637→            <p className="text-[10px] text-emerald-400">Within limits at projected usage. No overage charges.</p>
   638→          </div>
   639→        )}
   640→      </div>
   641→
   642→      {/* ── 3. Add-Ons ───────────────────────────────────────────────── */}
   643→      <div className="space-y-3">
   644→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   645→          Optional Add-Ons
   646→        </label>
   647→        {ADD_ONS.map((addOn) => {
   648→          const Icon = addOn.icon;
   649→          const isSelected = addOns[addOn.key];
   650→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   651→          const showPrice = !isIncluded;
   652→

   654→            <button
   655→              key={addOn.key}
   656→              type="button"
   657→              onClick={() => toggleAddOn(addOn.key)}
   658→              className={cn(
   659→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',
   660→                isSelected
   661→                  ? 'border-orange-500/30 bg-orange-500/5'
   662→                  : 'border-white/[0.06] hover:border-orange-500/15'

   664→              style={!isSelected ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   665→            >
   666→              <div className={cn(
   667→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   668→                isSelected ? 'bg-orange-500/10' : 'bg-white/[0.04]'
   669→              )}>
   670→                <Icon className={cn(
   671→                  'w-5 h-5',
   672→                  isSelected ? 'text-orange-400' : 'text-zinc-500'
   673→                )} />
   674→              </div>
   675→              <div className="flex-1 min-w-0">
   676→                <div className="flex items-center gap-2">
   677→                  <p className={cn(
   678→                    'text-sm font-medium',
   679→                    isSelected ? 'text-orange-400' : 'text-white'
   680→                  )}>
   681→                    {addOn.name}
   682→                  </p>
   683→                  {isIncluded && (
   684→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   685→                      Included
   686→                    </span>
   687→                  )}
   688→                </div>
   689→                <p className="text-[10px] text-orange-200/30 mt-0.5">{addOn.description}</p>
   690→              </div>
   691→              <div className="flex flex-col items-end shrink-0 gap-1">
   692→                {showPrice && (
   693→                  <p className="text-sm font-semibold text-white">${addOn.price}/mo</p>
   694→                )}
   695→                <div className={cn(
   696→                  'w-8 h-5 rounded-full flex items-center px-0.5 transition-colors duration-200',
   697→                  isSelected || isIncluded
   698→                    ? 'bg-gradient-to-r from-orange-500 to-amber-400'
   699→                    : 'bg-white/10'
   700→                )}>
   701→                  <div className={cn(
   702→                    'w-4 h-4 rounded-full bg-white shadow-sm transition-transform duration-200',
   703→                    (isSelected || isIncluded) ? 'translate-x-3' : 'translate-x-0'
   704→                  )} />
   705→                </div>
   706→              </div>
   707→            </button>
   708→          );
   709→        })}
   710→      </div>
   711→
   712→      {/* ── 4. Cost Summary (pure math — D7) ──────────────────────────── */}
   713→      <div
   714→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   715→        style={{ background: 'rgba(255,255,255,0.03)' }}
   716→      >
   717→        {/* Per-variant breakdown */}
   718→        {activeVariants.map((tier) => {
   719→          const qty = variantQuantities[tier] || 1;
   720→          return (
   721→            <div key={tier} className="flex items-center justify-between">
   722→              <span className="text-sm text-orange-200/50 flex items-center gap-1.5">
   723→                <div className={cn(
   724→                  'w-2 h-2 rounded-full',
   725→                  tier === 'starter' ? 'bg-emerald-400' : tier === 'growth' ? 'bg-orange-400' : 'bg-purple-400'

   730→              <span className="text-sm text-white">
   731→                ${VARIANT_PRICES[tier].toLocaleString()}/mo
   732→                {qty > 1 && <span className="text-orange-200/30"> × {qty} = ${(VARIANT_PRICES[tier] * qty).toLocaleString()}/mo</span>}
   733→              </span>
   734→            </div>
   735→          );
   736→        })}
   737→
   738→        {/* Add-ons */}
   739→        {addOns.voice && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'voice')!.includedIn.includes(t)) && (
   740→          <div className="flex items-center justify-between">
   741→            <span className="text-sm text-orange-200/50">Voice Channel</span>
   742→            <span className="text-sm text-white">$199/mo</span>
   743→          </div>
   744→        )}
   745→        {addOns.customApi && !activeVariants.some((t) => ADD_ONS.find(a => a.key === 'customApi')!.includedIn.includes(t)) && (
   746→          <div className="flex items-center justify-between">
   747→            <span className="text-sm text-orange-200/50">Custom API Connector</span>
   748→            <span className="text-sm text-white">$49/mo</span>
   749→          </div>
   750→        )}
   751→
   752→        {/* Integrations = $0 (D13) */}
   753→        <div className="flex items-center justify-between">
   754→          <span className="text-sm text-orange-200/50">Integrations</span>
   755→          <span className="text-sm text-emerald-400">$0 (free)</span>
   756→        </div>
   757→
   758→        <div className="border-t border-white/[0.06] pt-3 flex items-center justify-between">
   759→          <span className="text-sm font-semibold text-white">Total Monthly</span>
   760→          <span className="text-lg font-bold text-orange-400">${totalMonthly.toLocaleString()}/mo</span>
   761→        </div>
   762→
   763→        {/* Overage rate info */}
   764→        <div className="flex items-center justify-between">
   765→          <span className="text-[10px] text-orange-200/25">Overage rate (beyond limits)</span>
   766→          <span className="text-[10px] text-orange-200/25">${OVERAGE_PRICE_PER_TICKET}/ticket</span>
   767→        </div>
   768→      </div>
   769→
   770→      {/* ── 5. Savings Comparison (D10 — reuse ROI Calculator logic) ──── */}
   771→      <div
   772→        className="rounded-xl border border-emerald-500/20 p-4"
   773→        style={{ background: 'rgba(16,185,129,0.04)' }}
   774→      >
   775→        <div className="flex items-start gap-3">
   776→          <div className="w-8 h-8 rounded-lg bg-emerald-500/10 flex items-center justify-center shrink-0">
   777→            <TrendingDown className="w-4 h-4 text-emerald-400" />
   778→          </div>
   779→          <div className="flex-1">
   780→            <p className="text-sm font-semibold text-emerald-400">
   781→              Save {savings.savingsPercent}% vs human agents
   782→            </p>
   783→            <p className="text-[10px] text-orange-200/30 mt-1">
   784→              PARWA handles {totalTicketLimit.toLocaleString()} tickets/mo — equivalent to{' '}
   785→              {savings.agentsReplaced} full-time support agent{savings.agentsReplaced > 1 ? 's' : ''} at ~${AGENT_COST_MONTHLY.toLocaleString()}/mo each.
   786→              That&apos;s ${savings.humanCost.toLocaleString()}/mo in human costs vs ${totalMonthly.toLocaleString()}/mo with PARWA.
   787→            </p>
   788→          </div>
   789→        </div>
   790→      </div>
   791→
   792→      {/* ── No Hidden Fees (D13) ──────────────────────────────────────── */}
   793→      <div className="flex items-center justify-center gap-2 text-[10px] text-orange-200/25">
   794→        <Shield className="w-3 h-3" />
   795→        <span>No hidden fees. Need more? Add another variant.</span>
   796→      </div>
   797→
   798→      {/* ── Error Display ────────────────────────────────────────────── */}
   799→      {checkoutError && (
   800→        <div className="rounded-xl border border-red-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(239,68,68,0.05)' }}>
   801→          <AlertCircle className="w-5 h-5 text-red-400 shrink-0 mt-0.5" />
   802→          <p className="text-sm text-red-300">{checkoutError}</p>
   803→        </div>
   804→      )}
   805→
   806→      {/* ── Proceed Button ────────────────────────────────────────────── */}
   807→      <div className="flex justify-end">
   808→        <button
   809→          onClick={handleProceed}
   810→          disabled={isSubmitting}
   811→          className="px-8 py-3 bg-gradient-to-r from-orange-500 to-amber-400 hover:from-orange-400 hover:to-amber-300 text-[#1A1A1A] font-semibold rounded-xl transition-all duration-300 shadow-lg shadow-orange-500/25 text-sm flex items-center gap-2 disabled:opacity-60 disabled:cursor-not-allowed"
   812→        >
   813→          {isSubmitting ? (
   814→            <>
   815→              <Loader2 className="w-4 h-4 animate-spin" />
   816→              Opening Checkout...
   817→            </>
   818→          ) : (
   819→            <>
   820→              Pay ${totalMonthly.toLocaleString()}/mo & Activate
   821→              <ArrowRight className="w-4 h-4" />
   822→            </>
   823→          )}
   824→        </button>
   825→      </div>
   826→    </div>
   827→  );
   828→}

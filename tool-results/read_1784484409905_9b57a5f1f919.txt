     1→"""
     2→PARWA Jarvis Safety Gate — The Human-in-the-Loop Layer (DB-Backed)
     3→
     4→Before any function call is executed, it passes through the Safety Gate.
     5→This is what prevents Jarvis from accidentally doing something costly or
     6→irreversible without the client knowing.
     7→
     8→Safety Levels (from jarvis_function_registry):
     9→  - none: Execute immediately. No confirmation needed. (e.g., check health)
    10→  - confirmation_required: Ask the client "are you sure?" before executing.
    11→    The client must confirm before the action proceeds. (e.g., pause AI)
    12→  - approval_required: Require explicit "confirm" — for monetary, destructive,
    13→    or irreversible actions. The client must type "confirm" or "yes" to proceed.
    14→    (e.g., process refund, delete data, change billing)
    15→
    16→Flow:
    17→  1. Orchestrator calls check_safety(function_name, user_message, session_state)
    18→  2. Safety Gate looks up the function's safety_level from the registry
    19→  3. If safety_level == "none": Return immediately with status=approved
    20→  4. If safety_level == "confirmation_required":
    21→     a. Check if the session has a pending confirmation for this function
    22→     b. If yes and user confirmed: Approve and execute
    23→     c. If yes and user declined: Reject
    24→     d. If no pending confirmation: Return status=needs_confirmation
    25→        with a human-friendly message asking for confirmation
    26→  5. If safety_level == "approval_required":
    27→     a. Same flow, but requires explicit "confirm" / "yes" keyword
    28→     b. More strict — the confirmation message must contain an affirmative
    29→
    30→The Safety Gate is ENFORCED at the orchestrator level. The LLM CANNOT
    31→bypass it — even if the LLM generates a function call, the orchestrator
    32→will intercept it and require confirmation before executing.
    33→
    34→⚠️ MIGRATION (2025-01-19): Previously used in-memory _pending_confirmations dict.
    35→   Now uses DATABASE (jarvis_safety_confirmations table) so confirmations
    36→   survive server crashes and restarts.
    37→
    38→BC-001: company_id used for key namespacing.
    39→BC-008: Never crash — if DB is down, fail-safe to confirmation_required.
    40→BC-012: All timestamps UTC.
    41→"""
    42→

    65→# CONSTANTS
    66→# ══════════════════════════════════════════════════════════════════
    67→
    68→# How long a pending confirmation stays active (seconds)
    69→DEFAULT_CONFIRMATION_TTL = 300  # 5 minutes
    70→
    71→# Affirmative keywords for approval_required level
    72→# Only explicit, unambiguous words — "ok" and "sure" are NOT enough for
    73→# monetary/destructive/irreversible actions. They need to explicitly confirm.
    74→APPROVAL_KEYWORDS = {"confirm", "yes", "approved", "approve", "go ahead", "do it", "proceed"}
    75→
    76→# Negative keywords for rejecting confirmations
    77→REJECTION_KEYWORDS = {"no", "cancel", "stop", "never mind", "nevermind", "don't", "abort", "decline"}

    85→class SafetyCheckResult:
    86→    """Result of a safety gate check."""
    87→
    88→    def __init__(
    89→        self,
    90→        status: str,  # "approved", "needs_confirmation", "needs_approval", "rejected", "expired"
    91→        function_name: str,
    92→        safety_level: str,
    93→        message: str = "",
    94→        pending_id: Optional[str] = None,
    95→        params: Optional[Dict[str, Any]] = None,
    96→    ):
    97→        self.status = status
    98→        self.function_name = function_name
    99→        self.safety_level = safety_level
   100→        self.message = message
   101→        self.pending_id = pending_id
   102→        self.params = params or {}
   103→
   104→    def to_dict(self) -> Dict[str, Any]:
   105→        return {
   106→            "status": self.status,
   107→            "function_name": self.function_name,
   108→            "safety_level": self.safety_level,
   109→            "message": self.message,
   110→            "pending_id": self.pending_id,
   111→            "params": self.params,
   112→        }
   113→
   114→    @property
   115→    def is_approved(self) -> bool:
   116→        return self.status == "approved"
   117→
   118→    @property
   119→    def needs_human_input(self) -> bool:
   120→        return self.status in ("needs_confirmation", "needs_approval")

   135→def _store_pending_confirmation(
   136→    company_id: str,
   137→    session_id: str,
   138→    function_name: str,
   139→    params: Dict[str, Any],
   140→    safety_level: str,
   141→    ttl: int = DEFAULT_CONFIRMATION_TTL,
   142→    db: Optional[Session] = None,
   143→) -> str:
   144→    """Store a pending confirmation in DATABASE. Returns a pending_id."""
   145→    close_db = False
   146→    if db is None:
   147→        db = _get_db_session()
   148→        close_db = True
   149→    
   150→    try:
   151→        pending_id = f"{function_name}_{int(time.time())}"
   152→        expires_at = datetime.now(timezone.utc) + timedelta(seconds=ttl)
   153→        
   154→        # Delete any existing pending confirmation for this session
   155→        db.query(JarvisSafetyConfirmation).filter(
   156→            JarvisSafetyConfirmation.company_id == company_id,
   157→            JarvisSafetyConfirmation.session_id == session_id,
   158→            JarvisSafetyConfirmation.status == 'pending',
   159→        ).delete()
   160→        
   161→        # Insert new confirmation
   162→        entry = JarvisSafetyConfirmation(
   163→            company_id=company_id,
   164→            session_id=session_id,
   165→            pending_id=pending_id,
   166→            function_name=function_name,
   167→            safety_level=safety_level,
   168→            params=params or {},
   169→            status='pending',
   170→            created_at=datetime.now(timezone.utc),
   171→            expires_at=expires_at,
   172→        )
   173→        
   174→        db.add(entry)
   175→        db.commit()
   176→        
   177→        logger.info(
   178→            "safety_pending_created: function=%s, safety=%s, "
   179→            "company=%s, session=%s, pending_id=%s",
   180→            function_name, safety_level, company_id, session_id, pending_id,
   181→        )
   182→        
   183→        return pending_id
   184→        
   185→    except Exception as exc:
   186→        logger.error(
   187→            "safety_pending_create_failed: error=%s",
   188→            str(exc),
   189→            exc_info=True,
   190→        )
   191→        if close_db:
   192→            db.rollback()
   193→        raise
   194→    finally:
   195→        if close_db:
   196→            db.close()

   198→
   199→def _get_pending_confirmation(
   200→    company_id: str,
   201→    session_id: str,
   202→    db: Optional[Session] = None,
   203→) -> Optional[Dict[str, Any]]:
   204→    """Get and validate a pending confirmation from DATABASE.
   205→    
   206→    Returns None if expired, not found, or DB error.
   207→    Auto-expires stale entries on read.
   208→    """
   209→    close_db = False
   210→    if db is None:
   211→        db = _get_db_session()
   212→        close_db = True
   213→    
   214→    try:
   215→        # Find pending confirmation
   216→        entry = db.query(JarvisSafetyConfirmation).filter(
   217→            JarvisSafetyConfirmation.company_id == company_id,
   218→            JarvisSafetyConfirmation.session_id == session_id,
   219→            JarvisSafetyConfirmation.status == 'pending',
   220→        ).first()
   221→        
   222→        if not entry:
   223→            return None
   224→        
   225→        # Check TTL - auto-expire if past due
   226→        if entry.expires_at and entry.expires_at < datetime.now(timezone.utc):
   227→            entry.status = 'expired'
   228→            entry.resolved_at = datetime.now(timezone.utc)
   229→            db.commit()
   230→            
   231→            logger.info("safety_pending_expired: id=%s, company=%s, session=%s",
   232→                       entry.id, company_id, session_id)
   233→            return None
   234→        
   235→        # Return as dict (same format as old RAM version)
   236→        return {
   237→            "pending_id": entry.pending_id,
   238→            "function_name": entry.function_name,
   239→            "params": entry.params or {},
   240→            "safety_level": entry.safety_level,
   241→            "created_at": entry.created_at.isoformat() if entry.created_at else None,
   242→            "expires_at": entry.expires_at.isoformat() if entry.expires_at else None,
   243→            "db_id": entry.id,  # Store ID for quick updates
   244→        }
   245→        
   246→    except Exception as exc:
   247→        logger.warning(
   248→            "safety_pending_get_failed: error=%s, falling back to None",
   249→            str(exc),
   250→        )
   251→        return None
   252→    finally:
   253→        if close_db:
   254→            db.close()

   257→def _resolve_pending_confirmation(
   258→    company_id: str,
   259→    session_id: str,
   260→    status: str,  # 'approved' or 'rejected'
   261→    resolved_by: Optional[str] = None,
   262→    db: Optional[Session] = None,
   263→) -> bool:
   264→    """Resolve a pending confirmation (approve/reject).
   265→    
   266→    Returns True if a confirmation was found and updated.
   267→    """
   268→    close_db = False
   269→    if db is None:
   270→        db = _get_db_session()
   271→        close_db = True
   272→    
   273→    try:
   274→        entry = db.query(JarvisSafetyConfirmation).filter(
   275→            JarvisSafetyConfirmation.company_id == company_id,
   276→            JarvisSafetyConfirmation.session_id == session_id,
   277→            JarvisSafetyConfirmation.status == 'pending',
   278→        ).first()
   279→        
   280→        if not entry:
   281→            return False
   282→        
   283→        entry.status = status
   284→        entry.resolved_at = datetime.now(timezone.utc)
   285→        entry.resolved_by = resolved_by
   286→        db.commit()
   287→        
   288→        logger.info(
   289→            "safety_pending_resolved: status=%s, function=%s, company=%s",
   290→            status, entry.function_name, company_id,
   291→        )
   292→        
   293→        return True
   294→        
   295→    except Exception as exc:
   296→        logger.warning(
   297→            "safety_pending_resolve_failed: error=%s",
   298→            str(exc),
   299→        )
   300→        return False
   301→    finally:
   302→        if close_db:
   303→            db.close()

   323→# ══════════════════════════════════════════════════════════════════
   324→# MAIN SAFETY CHECK
   325→# ══════════════════════════════════════════════════════════════════
   326→
   327→
   328→def check_safety(
   329→    company_id: str,
   330→    session_id: str,
   331→    function_name: str,
   332→    function_params: Dict[str, Any],
   333→    user_message: str = "",
   334→) -> SafetyCheckResult:
   335→    """Check if a function call is safe to execute.
   336→
   337→    This is the main entry point for the Safety Gate. The orchestrator
   338→    calls this before executing any function call returned by the LLM.
   339→
   340→    Args:
   341→        company_id: Company ID for BC-001.
   342→        session_id: Session ID for pending confirmation scoping.
   343→        function_name: The function the LLM wants to call.
   344→        function_params: Parameters the LLM provided.
   345→        user_message: The user's latest message (for confirmation checking).
   346→
   347→    Returns:
   348→        SafetyCheckResult with:
   349→          - status "approved": Safe to execute immediately
   350→          - status "needs_confirmation": Ask client before executing
   351→          - status "needs_approval": Require explicit "confirm" before executing
   352→          - status "rejected": Client declined the action
   353→          - status "expired": Pending confirmation timed out
   354→    """
   355→    try:
   356→        safety_level = get_safety_level(function_name)
   357→
   358→        # ── Level 1: No safety check needed ──
   359→        if safety_level == SAFETY_NONE:
   360→            return SafetyCheckResult(
   361→                status="approved",
   362→                function_name=function_name,
   363→                safety_level=safety_level,
   364→                message="Action is safe to execute.",
   365→            )
   366→
   367→        # ── Check for pending confirmation (from DB) ──
   368→        pending = _get_pending_confirmation(company_id, session_id)
   369→
   370→        if pending and pending["function_name"] == function_name:
   371→            # There's a pending confirmation for this function
   372→            # Check if the user's message confirms or rejects it
   373→            normalized_msg = user_message.strip().lower()
   374→
   375→            # Check for rejection
   376→            for keyword in REJECTION_KEYWORDS:
   377→                if keyword in normalized_msg:
   378→                    _resolve_pending_confirmation(
   379→                        company_id, session_id, 'rejected'
   380→                    )
   381→                    logger.info(
   382→                        "safety_rejected: function=%s, company=%s, session=%s",
   383→                        function_name, company_id, session_id,
   384→                    )
   385→                    return SafetyCheckResult(
   386→                        status="rejected",
   387→                        function_name=function_name,
   388→                        safety_level=safety_level,
   389→                        message=f"Action '{function_name}' has been cancelled.",
   390→                        params=pending.get("params", {}),
   391→                    )
   392→
   393→            # Check for confirmation
   394→            if safety_level == SAFETY_CONFIRMATION:
   395→                # For confirmation_required, any non-rejection response counts
   396→                # as confirmation (the user just needs to respond)
   397→                if normalized_msg:
   398→                    _resolve_pending_confirmation(
   399→                        company_id, session_id, 'approved'
   400→                    )
   401→                    logger.info(
   402→                        "safety_confirmed: function=%s, company=%s, session=%s",
   403→                        function_name, company_id, session_id,
   404→                    )
   405→                    return SafetyCheckResult(
   406→                        status="approved",
   407→                        function_name=function_name,
   408→                        safety_level=safety_level,
   409→                        message="Action confirmed by user.",
   410→                        params=pending.get("params", {}),
   411→                    )
   412→
   413→            elif safety_level == SAFETY_APPROVAL:
   414→                # For approval_required, need an explicit affirmative keyword
   415→                for keyword in APPROVAL_KEYWORDS:
   416→                    if keyword in normalized_msg:
   417→                        _resolve_pending_confirmation(
   418→                            company_id, session_id, 'approved'
   419→                        )
   420→                        logger.info(
   421→                            "safety_approved: function=%s, company=%s, session=%s",
   422→                            function_name, company_id, session_id,
   423→                        )
   424→                        return SafetyCheckResult(
   425→                            status="approved",
   426→                            function_name=function_name,
   427→                            safety_level=safety_level,
   428→                            message="Action approved by user.",
   429→                            params=pending.get("params", {}),
   430→                        )
   431→
   432→                # No approval keyword found — still pending
   433→                return SafetyCheckResult(
   434→                    status="needs_approval",
   435→                    function_name=function_name,
   436→                    safety_level=safety_level,
   437→                    message=_build_approval_message(function_name, pending.get("params", {})),
   438→                    pending_id=pending.get("pending_id"),
   439→                    params=pending.get("params", {}),
   440→                )
   441→
   442→        # ── No pending confirmation — create one in DB ──
   443→        if safety_level == SAFETY_CONFIRMATION:
   444→            pending_id = _store_pending_confirmation(
   445→                company_id=company_id,
   446→                session_id=session_id,
   447→                function_name=function_name,
   448→                params=function_params,
   449→                safety_level=safety_level,
   450→            )
   451→            return SafetyCheckResult(
   452→                status="needs_confirmation",
   453→                function_name=function_name,
   454→                safety_level=safety_level,
   455→                message=_build_confirmation_message(function_name, function_params),
   456→                pending_id=pending_id,
   457→                params=function_params,
   458→            )
   459→
   460→        elif safety_level == SAFETY_APPROVAL:
   461→            pending_id = _store_pending_confirmation(
   462→                company_id=company_id,
   463→                session_id=session_id,
   464→                function_name=function_name,
   465→                params=function_params,
   466→                safety_level=safety_level,
   467→            )
   468→            return SafetyCheckResult(
   469→                status="needs_approval",
   470→                function_name=function_name,
   471→                safety_level=safety_level,
   472→                message=_build_approval_message(function_name, function_params),
   473→                pending_id=pending_id,
   474→                params=function_params,
   475→            )
   476→
   477→        # Fallback: fail-safe to confirmation required
   478→        logger.warning(
   479→            "safety_unknown_level: function=%s, level=%s — defaulting to confirmation",
   480→            function_name, safety_level,
   481→        )
   482→        return SafetyCheckResult(
   483→            status="needs_confirmation",
   484→            function_name=function_name,
   485→            safety_level=SAFETY_CONFIRMATION,
   486→            message=f"I want to run '{function_name}'. Can I proceed?",
   487→        )
   488→
   489→    except Exception:
   490→        logger.exception(
   491→            "safety_check_error: function=%s, company=%s",
   492→            function_name, company_id,
   493→        )
   494→        # Fail-safe: require confirmation
   495→        return SafetyCheckResult(
   496→            status="needs_confirmation",
   497→            function_name=function_name,
   498→            safety_level=SAFETY_CONFIRMATION,
   499→            message="I want to perform an action. Can I proceed?",
   500→        )
   501→

   557→            f"to the customer. This is a monetary action and can't be easily reversed. "

   603→def force_approve(
   604→    company_id: str,
   605→    session_id: str,
   606→) -> bool:
   607→    """Force-approve the pending confirmation for a session.
   608→
   609→    Used for testing or admin override. Returns True if a pending
   610→    confirmation was found and approved.
   611→    
   612→    Now operates on DATABASE instead of RAM.
   613→    """
   614→    pending = _get_pending_confirmation(company_id, session_id)
   615→    if pending:
   616→        _resolve_pending_confirmation(
   617→            company_id, session_id, 'approved', resolved_by='admin_override'
   618→        )
   619→        logger.info(
   620→            "safety_force_approved: function=%s, company=%s, session=%s",
   621→            pending["function_name"], company_id, session_id,
   622→        )
   623→        return True
   624→    return False

   627→def cleanup_expired_confirmations() -> int:
   628→    """Clean up expired confirmations in database.
   629→    
   630→    Returns number of confirmations expired.
   631→    Can be called from a cron job or periodic task.
   632→    """
   633→    try:
   634→        db = _get_db_session()
   635→        try:
   636→            result = db.query(JarvisSafetyConfirmation).filter(
   637→                JarvisSafetyConfirmation.status == 'pending',
   638→                JarvisSafetyConfirmation.expires_at < datetime.now(timezone.utc),
   639→            ).update({
   640→                "status": 'expired',
   641→                "resolved_at": datetime.now(timezone.utc),
   642→            })
   643→            db.commit()
   644→            return result
   645→        finally:
   646→            db.close()
   647→    except Exception as exc:
   648→        logger.warning("cleanup_expired_confirmations_failed: error=%s", str(exc))
   649→        return 0

     1→'use client';
     2→
     3→import React, { useState, useEffect, useMemo } from 'react';
     4→import {
     5→  Loader2, CheckCircle2, XCircle, ChevronDown, ChevronUp,
     6→  Plug, Search, AlertTriangle, ShieldCheck, Plus,
     7→  Sparkles, TrendingUp, CreditCard, Truck, Mail, MessageSquare,
     8→  BarChart3, Users, ShoppingCart, RefreshCw,
     9→} from 'lucide-react';
    10→import toast from 'react-hot-toast';
    11→import { cn } from '@/lib/utils';
    12→import {
    13→  INTEGRATION_CATALOG,
    14→  CATEGORY_META,
    15→  type IntegrationDefinition,
    16→  type IntegrationCategory,
    17→  type AuthField,
    18→} from '@/lib/integration-catalog';
    19→import { CustomIntegrationForm } from './CustomIntegrationForm';
    20→
    21→// ── Props ────────────────────────────────────────────────────────────
    22→
    23→interface IntegrationStepProps {
    24→  onNext: () => void;
    25→  industry?: string;
    26→}
    27→
    28→interface ConnectedIntegration {
    29→  id: string;
    30→  integration_type: string;
    31→  name: string | null;
    32→  status: string;
    33→  last_test_result?: string | null;
    34→}
    35→
    36→// ── Category icons ───────────────────────────────────────────────────
    37→
    38→const CATEGORY_ICONS: Record<IntegrationCategory, string> = {
    39→  crm: '🎯',
    40→  ecommerce: '🛒',
    41→  helpdesk: '🎫',
    42→  communication: '💬',
    43→  analytics: '📊',
    44→  marketing: '📧',
    45→  payments: '💳',
    46→  shipping: '📦',
    47→  dev_tools: '🔧',
    48→  productivity: '📝',
    49→  custom: '🔌',
    50→};
    51→
    52→// ── Component ────────────────────────────────────────────────────────
    53→
    54→export function IntegrationStep({ onNext }: IntegrationStepProps) {
    55→  const [existingIntegrations, setExistingIntegrations] = useState<ConnectedIntegration[]>([]);
    56→  const [isLoading, setIsLoading] = useState(true);
    57→  const [selectedCategory, setSelectedCategory] = useState<IntegrationCategory | 'all'>('all');
    58→  const [searchQuery, setSearchQuery] = useState('');
    59→  const [expandedIntegration, setExpandedIntegration] = useState<string | null>(null);
    60→  const [credentialValues, setCredentialValues] = useState<Record<string, Record<string, string>>>({});
    61→  const [savingKey, setSavingKey] = useState<string | null>(null);
    62→  const [verifyResults, setVerifyResults] = useState<Record<string, { verified: boolean; message: string }>>({});
    63→  const [showCustomForm, setShowCustomForm] = useState(false);
    64→  
    65→  // CRM Analyzer state
    66→  const [crmAnalysis, setCrmAnalysis] = useState<any>(null);
    67→  const [isAnalyzing, setIsAnalyzing] = useState(false);
    68→  const [hasAnalyzed, setHasAnalyzed] = useState(false);
    69→
    70→  // Fetch existing integrations
    71→  useEffect(() => {
    72→    fetch('/api/integrations', { credentials: 'include' })
    73→      .then((res) => res.ok ? res.json() : null)
    74→      .then((data) => {
    75→        const items = Array.isArray(data) ? data : (data?.items || []);
    76→        setExistingIntegrations(items);
    77→      })
    78→      .catch(() => {})
    79→      .finally(() => setIsLoading(false));
    80→  }, []);
    81→
    82→  // Filter integrations by category + search
    83→  const filteredIntegrations = useMemo(() => {
    84→    let list = INTEGRATION_CATALOG.filter(i => i.available !== false);
    85→    if (selectedCategory !== 'all') {
    86→      list = list.filter(i => i.category === selectedCategory);
    87→    }
    88→    if (searchQuery.trim()) {
    89→      const q = searchQuery.toLowerCase();
    90→      list = list.filter(i => i.name.toLowerCase().includes(q) || i.description.toLowerCase().includes(q));
    91→    }
    92→    return list;
    93→  }, [selectedCategory, searchQuery]);
    94→
    95→  // Group by category
    96→  const grouped = useMemo(() => {
    97→    const groups: Record<string, IntegrationDefinition[]> = {};
    98→    for (const integration of filteredIntegrations) {
    99→      if (!groups[integration.category]) groups[integration.category] = [];
   100→      groups[integration.category].push(integration);
   101→    }
   102→    // Sort categories by order
   103→    return Object.entries(groups).sort((a, b) => {
   104→      const orderA = CATEGORY_META[a[0] as IntegrationCategory]?.order ?? 99;
   105→      const orderB = CATEGORY_META[b[0] as IntegrationCategory]?.order ?? 99;
   106→      return orderA - orderB;
   107→    });
   108→  }, [filteredIntegrations]);
   109→
   110→  // Check if an integration is already connected
   111→  const getExistingStatus = (key: string): ConnectedIntegration | null => {
   112→    return existingIntegrations.find((i) => i.integration_type === key) || null;
   113→  };
   114→
   115→  // Handle credential field change
   116→  const handleFieldChange = (integrationKey: string, fieldName: string, value: string) => {
   117→    setCredentialValues((prev) => ({
   118→      ...prev,
   119→      [integrationKey]: {
   120→        ...(prev[integrationKey] || {}),
   121→        [fieldName]: value,
   122→      },
   123→    }));
   124→  };
   125→
   126→  // Save & Verify integration
   127→  const handleSaveAndVerify = async (integration: IntegrationDefinition) => {
   128→    const values = credentialValues[integration.key] || {};
   129→    const requiredFields = integration.authSchema.fields.filter(f => f.required);
   130→    const missing = requiredFields.filter(f => !values[f.name]?.trim());
   131→
   132→    if (missing.length > 0) {
   133→      toast.error(`Please fill in: ${missing.map(f => f.label).join(', ')}`);
   134→      return;
   135→    }
   136→
   137→    setSavingKey(integration.key);
   138→    try {
   139→      // Check if integration already exists
   140→      const existing = getExistingStatus(integration.key);
   141→
   142→      let integrationId = existing?.id;
   143→
   144→      if (!integrationId) {
   145→        // Create new integration
   146→        const createRes = await fetch('/api/integrations', {
   147→          method: 'POST',
   148→          headers: { 'Content-Type': 'application/json' },
   149→          credentials: 'include',
   150→          body: JSON.stringify({
   151→            integration_type: integration.key,
   152→            name: integration.name,
   153→            credentials: values,
   154→            settings: {},
   155→          }),
   156→        });
   157→
   158→        if (!createRes.ok) {
   159→          const err = await createRes.json().catch(() => ({}));
   160→          throw new Error(err.detail || err.message || 'Failed to save integration');
   161→        }
   162→
   163→        const created = await createRes.json();
   164→        integrationId = created.id;
   165→      } else {
   166→        // Update existing integration credentials
   167→        const updateRes = await fetch(`/api/integrations/${integrationId}`, {
   168→          method: 'PATCH',
   169→          headers: { 'Content-Type': 'application/json' },
   170→          credentials: 'include',
   171→          body: JSON.stringify({
   172→            credentials: values,
   173→          }),
   174→        });
   175→
   176→        if (!updateRes.ok) {
   177→          throw new Error('Failed to update integration');
   178→        }
   179→      }
   180→
   181→      // Test/verify the integration
   182→      const testRes = await fetch(`/api/integrations/${integrationId}/test`, {
   183→        method: 'POST',
   184→        credentials: 'include',
   185→      });
   186→
   187→      const testData = await testRes.json().catch(() => ({}));
   188→
   189→      if (testRes.ok && testData.success) {
   190→        setVerifyResults((prev) => ({
   191→          ...prev,
   192→          [integration.key]: { verified: true, message: testData.message || 'Verified' },
   193→        }));
   194→        toast.success(`${integration.name} verified successfully!`);
   195→      } else {
   196→        setVerifyResults((prev) => ({
   197→          ...prev,
   198→          [integration.key]: { verified: false, message: testData.message || testData.error || 'Verification failed' },
   199→        }));
   200→        toast.error(`${integration.name} verification failed: ${testData.message || testData.error || 'Invalid credentials'}`);
   201→      }
   202→
   203→      // Refresh integrations list
   204→      const refreshed = await fetch('/api/integrations', { credentials: 'include' });
   205→      if (refreshed.ok) {
   206→        const data = await refreshed.json();
   207→        const items = Array.isArray(data) ? data : (data?.items || []);
   208→        setExistingIntegrations(items);
   209→      }
   210→    } catch (err) {
   211→      toast.error(err instanceof Error ? err.message : 'Failed to save integration');
   212→    } finally {
   213→      setSavingKey(null);
   214→    }
   215→  };
   216→
   217→  // Continue button
   218→  const handleContinue = async () => {
   219→    const verifiedCount = Object.values(verifyResults).filter(r => r.verified).length;
   220→    const connectedCount = existingIntegrations.length;
   221→
   222→    if (connectedCount === 0) {
   223→      toast.error('Please connect at least one integration to continue.');
   224→      return;
   225→    }
   226→
   227→    if (verifiedCount === 0 && connectedCount > 0) {
   228→      // Some integrations connected but none verified — warn but allow
   229→      toast('Note: Some integrations are not verified. You can verify them later in Settings.', { icon: '⚠️' });
   230→    }
   231→
   232→    onNext();
   233→  };
   234→
   235→  // Run CRM Analysis - Fixed v2: Works without connected integrations
   236→  const runCRMAnalysis = async () => {
   237→    // Allow analysis even without connected integrations - 
   238→    // AI will recommend what tools they need based on their profile
   239→    console.log('[CRM Analyzer] Starting analysis...');
   240→    setIsAnalyzing(true);
   241→    
   242→    try {
   243→      const response = await fetch('/api/integrations/analyze', {
   244→        method: 'POST',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        credentials: 'include',
   247→        body: JSON.stringify({
   248→          industry: industry || 'unknown',
   249→          connected_count: existingIntegrations.length,
   250→          integrations: existingIntegrations.map(i => i.integration_type),
   251→        }),
   252→      });
   253→
   254→      if (!response.ok) {
   255→        const errorData = await response.json().catch(() => ({}));
   256→        throw new Error(errorData.message || `Analysis failed (${response.status})`);
   257→      }
   258→
   259→      const data = await response.json();
   260→      setCrmAnalysis(data);
   261→      setHasAnalyzed(true);
   262→      
   263→      if (data.recommendations?.length > 0) {
   264→        toast.success(`Found ${data.recommendations.length} recommendations for you!`);
   265→      } else {
   266→        toast.success('Analysis complete! Your integration health looks good.');
   267→      }
   268→    } catch (error) {
   269→      console.error('CRM Analysis error:', error);
   270→      const message = error instanceof Error ? error.message : 'Failed to analyze';
   271→      toast.error(`${message}. You can continue anyway.`);
   272→    } finally {
   273→      setIsAnalyzing(false);
   274→    }
   275→  };
   276→
   277→  if (isLoading) {
   278→    return (
   279→      <div className="flex items-center justify-center py-20">
   280→        <Loader2 className="w-6 h-6 animate-spin text-orange-400" />
   281→      </div>
   282→    );
   283→  }
   284→
   285→  return (
   286→    <div className="max-w-3xl mx-auto">
   287→      {/* Header */}
   288→      <div className="text-center mb-6">
   289→        <div className="w-12 h-12 mx-auto rounded-xl bg-gradient-to-br from-orange-500/20 to-orange-600/10 border border-orange-500/20 flex items-center justify-center mb-3">
   290→          <Plug className="w-6 h-6 text-orange-400" />
   291→        </div>
   292→        <h2 className="text-xl font-bold text-white">Connect Your Tools</h2>
   293→        <p className="text-sm text-zinc-500 mt-1">
   294→          Choose the tools your team uses. Enter credentials and verify to connect.
   295→        </p>
   296→      </div>
   297→
   298→      {/* Search */}
   299→      <div className="relative mb-4">
   300→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-zinc-500" />
   301→        <input
   302→          type="text"
   303→          value={searchQuery}
   304→          onChange={(e) => setSearchQuery(e.target.value)}
   305→          placeholder="Search integrations..."
   306→          className="w-full pl-9 pr-4 py-2.5 rounded-lg bg-white/[0.03] border border-white/[0.08] text-sm text-white placeholder:text-zinc-600 focus:outline-none focus:border-orange-500/40 transition-colors"
   307→        />
   308→      </div>
   309→
   310→      {/* Category tabs */}
   311→      <div className="flex flex-wrap gap-2 mb-6">
   312→        <button
   313→          onClick={() => setSelectedCategory('all')}
   314→          className={cn(
   315→            'px-3 py-1.5 rounded-lg text-xs font-medium transition-colors',
   316→            selectedCategory === 'all'
   317→              ? 'bg-orange-500/20 text-orange-400 border border-orange-500/30'
   318→              : 'bg-white/[0.03] text-zinc-400 border border-white/[0.06] hover:bg-white/[0.06]'
   319→          )}
   320→        >
   321→          All
   322→        </button>
   323→        {Object.entries(CATEGORY_META).map(([cat, meta]) => (
   324→          <button
   325→            key={cat}
   326→            onClick={() => setSelectedCategory(cat as IntegrationCategory)}
   327→            className={cn(
   328→              'px-3 py-1.5 rounded-lg text-xs font-medium transition-colors flex items-center gap-1.5',
   329→              selectedCategory === cat
   330→                ? 'bg-orange-500/20 text-orange-400 border border-orange-500/30'
   331→                : 'bg-white/[0.03] text-zinc-400 border border-white/[0.06] hover:bg-white/[0.06]'
   332→            )}
   333→          >
   334→            <span>{CATEGORY_ICONS[cat as IntegrationCategory]}</span>
   335→            {meta.label}
   336→          </button>
   337→        ))}
   338→      </div>
   339→
   340→      {/* Integration list grouped by category */}
   341→      <div className="space-y-6 mb-6">
   342→        {grouped.map(([category, integrations]) => (
   343→          <div key={category}>
   344→            <h3 className="text-xs text-zinc-600 uppercase tracking-wider font-medium mb-3">
   345→              {CATEGORY_ICONS[category as IntegrationCategory]} {CATEGORY_META[category as IntegrationCategory]?.label || category}
   346→            </h3>
   347→            <div className="space-y-2">
   348→              {integrations.map((integration) => {
   349→                const existing = getExistingStatus(integration.key);
   350→                const isExpanded = expandedIntegration === integration.key;
   351→                const verifyResult = verifyResults[integration.key];
   352→                const isSaving = savingKey === integration.key;
   353→                const values = credentialValues[integration.key] || {};
   354→
   355→                return (
   356→                  <div
   357→                    key={integration.key}
   358→                    className={cn(
   359→                      'rounded-xl border overflow-hidden transition-all',
   360→                      existing?.status === 'active' && verifyResult?.verified
   361→                        ? 'border-emerald-500/20 bg-emerald-500/[0.02]'
   362→                        : existing?.status === 'active'
   363→                        ? 'border-amber-500/20 bg-amber-500/[0.02]'
   364→                        : 'border-white/[0.06] bg-white/[0.02]'
   365→                    )}
   366→                  >
   367→                    {/* Header row */}
   368→                    <button
   369→                      onClick={() => setExpandedIntegration(isExpanded ? null : integration.key)}
   370→                      className="w-full flex items-center justify-between px-4 py-3 hover:bg-white/[0.02] transition-colors"
   371→                    >
   372→                      <div className="flex items-center gap-3 min-w-0">
   373→                        <span className="text-xl">{CATEGORY_ICONS[integration.category]}</span>
   374→                        <div className="min-w-0 text-left">
   375→                          <p className="text-sm font-medium text-white truncate">{integration.name}</p>
   376→                          <p className="text-[10px] text-zinc-600 truncate">{integration.description}</p>
   377→                        </div>
   378→                      </div>
   379→                      <div className="flex items-center gap-2 shrink-0">
   380→                        {/* Status badge */}
   381→                        {existing?.status === 'active' && verifyResult?.verified && (
   382→                          <span className="text-[10px] px-2 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 flex items-center gap-1">
   383→                            <CheckCircle2 className="w-3 h-3" /> Verified
   384→                          </span>
   385→                        )}
   386→                        {existing?.status === 'active' && !verifyResult?.verified && (
   387→                          <span className="text-[10px] px-2 py-0.5 rounded-full bg-amber-500/10 text-amber-400 border border-amber-500/20 flex items-center gap-1">
   388→                            <AlertTriangle className="w-3 h-3" /> Not Verified
   389→                          </span>
   390→                        )}
   391→                        {isExpanded ? <ChevronUp className="w-4 h-4 text-zinc-500" /> : <ChevronDown className="w-4 h-4 text-zinc-500" />}
   392→                      </div>
   393→                    </button>
   394→
   395→                    {/* Expanded credential form */}
   396→                    {isExpanded && (
   397→                      <div className="px-4 pb-4 pt-2 border-t border-white/[0.04] space-y-3">
   398→                        {integration.authSchema.fields.map((field: AuthField) => (
   399→                          <div key={field.name}>
   400→                            <label className="text-[10px] text-zinc-500 uppercase tracking-wider font-medium block mb-1">
   401→                              {field.label} {field.required && <span className="text-red-400">*</span>}
   402→                            </label>
   403→                            <input
   404→                              type={field.type === 'password' ? 'password' : 'text'}
   405→                              value={values[field.name] || ''}
   406→                              onChange={(e) => handleFieldChange(integration.key, field.name, e.target.value)}
   407→                              placeholder={field.placeholder || ''}
   408→                              className="w-full px-3 py-2 rounded-lg bg-white/[0.03] border border-white/[0.08] text-sm text-white placeholder:text-zinc-600 focus:outline-none focus:border-orange-500/40 transition-colors"
   409→                            />
   410→                          </div>
   411→                        ))}
   412→
   413→                        {/* Verify result message */}
   414→                        {verifyResult && (
   415→                          <div className={cn(
   416→                            'rounded-lg px-3 py-2 text-xs',
   417→                            verifyResult.verified
   418→                              ? 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20'
   419→                              : 'bg-red-500/10 text-red-400 border border-red-500/20'
   420→                          )}>
   421→                            {verifyResult.verified ? '✓ ' : '✗ '}{verifyResult.message}
   422→                          </div>
   423→                        )}
   424→
   425→                        {/* Save & Verify button */}
   426→                        <button
   427→                          onClick={() => handleSaveAndVerify(integration)}
   428→                          disabled={isSaving}
   429→                          className="flex items-center gap-2 px-4 py-2 rounded-lg bg-orange-500/20 text-orange-400 text-xs font-medium border border-orange-500/30 hover:bg-orange-500/30 transition-colors disabled:opacity-50"
   430→                        >
   431→                          {isSaving ? (
   432→                            <><Loader2 className="w-3.5 h-3.5 animate-spin" /> Saving...</>
   433→                          ) : (
   434→                            <><ShieldCheck className="w-3.5 h-3.5" /> Save & Verify</>
   435→                          )}
   436→                        </button>
   437→                      </div>
   438→                    )}
   439→                  </div>
   440→                );
   441→              })}
   442→            </div>
   443→          </div>
   444→        ))}
   445→      </div>
   446→
   447→      {/* Add Custom Integration button */}
   448→      <div className="mb-6">
   449→        <button
   450→          onClick={() => setShowCustomForm(!showCustomForm)}
   451→          className="flex items-center gap-2 px-4 py-2.5 rounded-xl bg-white/[0.03] border border-dashed border-white/[0.12] text-sm text-zinc-400 hover:text-white hover:bg-white/[0.06] transition-colors w-full justify-center"
   452→        >
   453→          <Plus className="w-4 h-4" />
   454→          {showCustomForm ? 'Cancel Custom Integration' : 'Add Custom Integration + Category'}
   455→        </button>
   456→      </div>
   457→
   458→      {/* Custom Integration Form */}
   459→      {showCustomForm && (
   460→        <div className="mb-6 rounded-xl bg-white/[0.02] border border-orange-500/15 p-5">
   461→          <h3 className="text-sm font-semibold text-white mb-4">Create Custom Integration</h3>
   462→          <CustomIntegrationForm
   463→            onSaved={(newIntegration) => {
   464→              setExistingIntegrations((prev) => [...prev, newIntegration]);
   465→              setShowCustomForm(false);
   466→              toast.success('Custom integration saved!');
   467→            }}
   468→          />
   469→        </div>
   470→      )}
   471→
   472→      {/* CRM Analyzer - Smart Recommendations */}
   473→      {existingIntegrations.length > 0 && (
   474→        <div className="mb-6 rounded-xl bg-gradient-to-br from-orange-500/10 to-amber-500/5 border border-orange-500/20 p-5">
   475→          <div className="flex items-center justify-between mb-4">
   476→            <div className="flex items-center gap-2">
   477→              <Sparkles className="w-5 h-5 text-orange-400" />
   478→              <h3 className="text-sm font-semibold text-white">Smart Recommendations</h3>
   479→            </div>
   480→            {!hasAnalyzed ? (
   481→              <button
   482→                onClick={runCRMAnalysis}
   483→                disabled={isAnalyzing}
   484→                className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-orange-500/20 text-orange-400 text-xs font-medium border border-orange-500/30 hover:bg-orange-500/30 transition-colors disabled:opacity-50"
   485→              >
   486→                {isAnalyzing ? (
   487→                  <><Loader2 className="w-3.5 h-3.5 animate-spin" /> Analyzing...</>
   488→                ) : (
   489→                  <><TrendingUp className="w-3.5 h-3.5" /> Analyze My Setup</>
   490→                )}
   491→              </button>
   492→            ) : (
   493→              <button
   494→                onClick={runCRMAnalysis}
   495→                disabled={isAnalyzing}
   496→                className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-white/[0.05] text-zinc-400 text-xs font-medium border border-white/[0.08] hover:bg-white/[0.08] transition-colors disabled:opacity-50"
   497→              >
   498→                <RefreshCw className="w-3.5 h-3.5" /> Refresh
   499→              </button>
   500→            )}
   501→          </div>
   502→
   503→          {/* Loading State */}
   504→          {isAnalyzing && (
   505→            <div className="space-y-3">
   506→              <div className="flex items-center gap-2 text-xs text-zinc-400">
   507→                <Loader2 className="w-4 h-4 animate-spin text-orange-400" />
   508→                Analyzing your integrations with AI...
   509→              </div>
   510→              <div className="grid grid-cols-3 gap-2">
   511→                {[1, 2, 3].map((i) => (
   512→                  <div key={i} className="h-16 rounded-lg bg-white/[0.03] animate-pulse" />
   513→                ))}
   514→              </div>
   515→            </div>
   516→          )}
   517→
   518→          {/* Results */}
   519→          {hasAnalyzed && crmAnalysis && !isAnalyzing && (
   520→            <div className="space-y-4">
   521→              {/* Data Profile Summary */}

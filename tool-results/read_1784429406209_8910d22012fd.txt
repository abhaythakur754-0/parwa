   279→    """Merge partial updates into the customer care session context.
   280→
   281→    Only provided fields are merged. Existing keys are preserved.
   282→    Protected keys (variant_tier, variant_instance_id, industry)

   284→
   285→    This is how the awareness engine (Phase 2) and command executor
   286→    (Phase 3) will update the session context.
   287→    """
   288→    # Build partial updates dict from body
   289→    partial_updates = {}
   290→    if body.awareness_enabled is not None:
   291→        partial_updates["awareness_enabled"] = body.awareness_enabled
   292→    if body.proactive_alerts is not None:
   293→        partial_updates["proactive_alerts"] = body.proactive_alerts
   294→    if body.custom_fields is not None:
   295→        partial_updates.update(body.custom_fields)
   296→
   297→    if not partial_updates:
   298→        return _error_response(
   299→            "VALIDATION_ERROR",
   300→            "No fields provided for update",
   301→            422,
   302→        )
   303→
   304→    try:
   305→        session = jarvis_cc_service.update_cc_context(
   306→            db=db,
   307→            session_id=session_id,
   308→            user_id=str(user.id),
   309→            company_id=str(user.company_id),
   310→            partial_updates=partial_updates,
   311→        )
   312→        return _session_to_response(session)
   313→    except ParwaBaseError:
   314→        raise
   315→    except Exception as exc:
   316→        return _error_response(

   819→@router.post("/command", response_model=JarvisCommandResponse)
   820→def send_command(
   821→    body: JarvisCommandSend,
   822→    user: User = Depends(get_current_user),
   823→    db: Session = Depends(get_db),
   824→):
   825→    """Send a natural language command to Jarvis.
   826→
   827→    Parses the NL input, creates a command record, executes the
   828→    appropriate handler, and returns the result. This is the main
   829→    entry point for the Command Layer.
   830→
   831→    Commands like "pause all AI", "show me today's errors",
   832→    "export weekly report" are parsed without LLM for speed (<2s).
   833→    """
   834→    if not user.company_id:
   835→        return _error_response(
   836→            "VALIDATION_ERROR",
   837→            "User has no associated company",
   838→            422,
   839→        )
   840→
   841→    try:
   842→        # Step 1: Receive and parse the command
   843→        command = jarvis_command_service.receive_command(
   844→            db=db,
   845→            company_id=str(user.company_id),
   846→            session_id=body.session_id,
   847→            raw_input=body.raw_input,
   848→            source=body.source,
   849→            user_id=str(user.id),
   850→        )
   851→
   852→        # Step 2: Execute the command
   853→        result = jarvis_command_service.execute_command(
   854→            db=db,
   855→            company_id=str(user.company_id),
   856→            command_id=str(command.id),
   857→            session_id=body.session_id,
   858→            user_id=str(user.id),
   859→        )
   860→
   861→        # Build response
   862→        parsed = _safe_parse_json(command.command_parsed) if command.command_parsed else {}
   863→
   864→        return JarvisCommandResponse(
   865→            command_id=result["command_id"],
   866→            status=result["status"],
   867→            action=result["action"],
   868→            intent=command.command_intent,
   869→            scope=parsed.get("scope"),
   870→            target=parsed.get("target"),
   871→            confidence=float(command.confidence) if command.confidence else None,
   872→            result=result.get("result", {}),
   873→            execution_time_ms=result.get("execution_time_ms", 0),
   874→            undo_available=command.undo_available or False,
   875→            error=result.get("error"),
   876→            suggestion=parsed.get("suggestion"),
   877→        )
   878→    except ParwaBaseError:
   879→        raise
   880→    except Exception as exc:
   881→        return _error_response(
   882→            "INTERNAL_ERROR",
   883→            "Failed to execute command",
   884→            500,
   885→            details={"error": str(exc)},
   886→        )

  1254→    """Coerce a value to int, falling back to default.

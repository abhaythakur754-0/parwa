    15→    15→
    16→    16→export type AuthType = 'bearer' | 'api_key_header' | 'api_key_query' | 'basic_auth' | 'oauth2';
    17→    17→
    18→    18→export interface AuthField {
    19→    19→  name: string;
    20→    20→  label: string;
    21→    21→  type: 'text' | 'password' | 'url';
    22→    22→  required: boolean;
    23→    23→  placeholder?: string;
    24→    24→}
    25→    25→
    26→    26→export interface AuthSchema {
    27→    27→  type: AuthType;
    28→    28→  fields: AuthField[];
    29→    29→  /** Header name for api_key_header auth */
    30→    30→  headerName?: string;
    31→    31→  /** Query param name for api_key_query auth */
    32→    32→  queryParamName?: string;
    33→    33→  /** OAuth2 redirect URI */
    34→    34→  redirectUri?: string;
    35→    35→}

    60→    60→// ── Integration Definition ─────────────────────────────────────────────
    61→    61→
    62→    62→export interface IntegrationDefinition {
    63→    63→  /** Machine key — also used as integration_type in DB */
    64→    64→  key: string;
    65→    65→  /** Display name */
    66→    66→  name: string;
    67→    67→  /** Short description for UI */
    68→    68→  description: string;
    69→    69→  /** Category for grouping in UI */
    70→    70→  category: IntegrationCategory;
    71→    71→  /** Which tier this belongs to */
    72→    72→  tier: IntegrationTier;
    73→    73→  /** Auth schema — defines the credential form (GAP 2) */
    74→    74→  authSchema: AuthSchema;
    75→    75→  /** Pre-written test call (D6 — NO AI) */
    76→    76→  testConnection: {
    77→    77→    method: 'GET' | 'POST';
    78→    78→    /** URL template — {field_name} replaced with credential values */
    79→    79→    urlTemplate: string;
    80→    80→    /** Header template — {field_name} replaced with credential values */
    81→    81→    headersTemplate?: Record<string, string>;
    82→    82→    /** How to interpret the response */
    83→    83→    successCheck: 'status_200' | 'json_ok_true' | 'status_200_or_201';
    84→    84→    /** What to show on success — {response_json_path} for dynamic values */
    85→    85→    successMessage?: string;
    86→    86→  };
    87→    87→  /** Industries where this integration is SUGGESTED (not restricted to) */
    88→    88→  suggestedIndustries: ParwaIndustry[];
    89→    89→  /** Which variants can use this (empty = all) */
    90→    90→  availableForVariants?: ('mini_parwa' | 'parwa' | 'parwa_high')[];
    91→    91→  /** Icon identifier for frontend rendering */
    92→    92→  iconId: string;
    93→    93→  /** Icon color gradient for frontend */
    94→    94→  colorGradient: string;
    95→    95→  /** Whether this integration is available */
    96→    96→  available: boolean;
    97→    97→}

   911→   911→ * Per D3: Suggestions are NOT restrictions — clients can always connect outside their industry.
   912→   912→ */
   913→   913→export function getIntegrationsForIndustry(
   914→   914→  industry: ParwaIndustry
   915→   915→): IntegrationDefinition[] {
   916→   916→  if (industry === 'other') {
   917→   917→    return INTEGRATION_CATALOG.filter((i) => i.available);
   918→   918→  }
   919→   919→  return INTEGRATION_CATALOG.filter(
   920→   920→    (i) => i.available && i.suggestedIndustries.includes(industry)
   921→   921→  );
   922→   922→}

   940→   940→  return grouped as Record<IntegrationCategory, IntegrationDefinition[]>;
   941→   941→}

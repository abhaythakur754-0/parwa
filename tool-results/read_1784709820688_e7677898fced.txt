    78→    78→function formatRelativeDate(dateStr: string): string {
    79→    79→  const diff = Date.now() - new Date(dateStr).getTime();
    80→    80→  const mins = Math.floor(diff / 60000);
    81→    81→  if (mins < 1) return 'just now';
    82→    82→  if (mins < 60) return `${mins}m ago`;
    83→    83→  const hrs = Math.floor(mins / 60);
    84→    84→  if (hrs < 24) return `${hrs}h ago`;
    85→    85→  const days = Math.floor(hrs / 24);
    86→    86→  if (days < 7) return `${days}d ago`;
    87→    87→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
    88→    88→}
    89→    89→
    90→    90→function formatDate(dateStr: string): string {
    91→    91→  return new Date(dateStr).toLocaleDateString('en-US', {
    92→    92→    month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit',
    93→    93→  });
    94→    94→}

   184→   184→function CustomerContextPanel({
   185→   185→  ticketId,
   186→   186→  customerEmail,
   187→   187→}: {
   188→   188→  ticketId: string;
   189→   189→  customerEmail: string;
   190→   190→}) {
   191→   191→  const [contextData, setContextData] = useState<Record<string, unknown> | null>(null);
   192→   192→  const [loading, setLoading] = useState(true);
   193→   193→
   194→   194→  useEffect(() => {
   195→   195→    let cancelled = false;
   196→   196→    setLoading(true);
   197→   197→    // Fetch the ticket detail from the backend — it includes metadata_json
   198→   198→    // which may contain CRM data, ecommerce orders, and carrier tracking
   199→   199→    // that Node 3 fetched during pipeline processing.
   200→   200→    fetch(`/api/v1/tickets/${ticketId}`, { credentials: 'include' })
   201→   201→      .then((r) => r.json())
   202→   202→      .then((data) => {
   203→   203→        if (cancelled) return;
   204→   204→        // The metadata_json field may contain CRM/ecommerce/carrier data
   205→   205→        const meta = data.metadata_json || {};
   206→   206→        setContextData(meta);
   207→   207→      })
   208→   208→      .catch(() => {
   209→   209→        if (!cancelled) setContextData(null);
   210→   210→      })
   211→   211→      .finally(() => {
   212→   212→        if (!cancelled) setLoading(false);
   213→   213→      });
   214→   214→    return () => { cancelled = true; };
   215→   215→  }, [ticketId]);
   216→   216→
   217→   217→  if (loading) {
   218→   218→    return (
   219→   219→      <div className="px-4 py-3 border-b border-white/[0.06]">
   220→   220→        <p className="text-[10px] text-zinc-600">Loading customer context...</p>
   221→   221→      </div>
   222→   222→    );
   223→   223→  }
   224→   224→
   225→   225→  if (!contextData || Object.keys(contextData).length === 0) {
   226→   226→    return null; // No integration data — don't show the panel
   227→   227→  }
   228→   228→
   229→   229→  return (
   230→   230→    <div className="px-4 py-3 border-b border-white/[0.06] bg-white/[0.01]">
   231→   231→      <p className="text-[10px] font-semibold text-zinc-400 uppercase tracking-wide mb-2">
   232→   232→        Customer Context
   233→   233→      </p>
   234→   234→      <div className="space-y-1.5 text-[11px]">
   235→   235→        {Object.entries(contextData).slice(0, 8).map(([key, value]) => (
   236→   236→          <div key={key} className="flex justify-between gap-2">
   237→   237→            <span className="text-zinc-500 capitalize">{key.replace(/_/g, ' ')}</span>
   238→   238→            <span className="text-zinc-300 text-right truncate max-w-[200px]">
   239→   239→              {typeof value === 'object'
   240→   240→                ? JSON.stringify(value).slice(0, 80) + '...'
   241→   241→                : String(value).slice(0, 80)}
   242→   242→            </span>
   243→   243→          </div>
   244→   244→        ))}
   245→   245→      </div>
   246→   246→    </div>
   247→   247→  );
   248→   248→}

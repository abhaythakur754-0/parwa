   133→// ── Savings Calculation (uses AGENT_COST_MONTHLY from pricing-config) ──
   134→
   135→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   136→  agentsReplaced: number;
   137→  humanCost: number;
   138→  savings: number;
   139→  savingsPercent: number;
   140→} {
   141→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   142→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   143→  const savings = Math.max(0, humanCost - monthlyCost);
   144→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   145→  return { agentsReplaced, humanCost, savings, savingsPercent };
   146→}

   455→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   456→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   457→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   458→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   459→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   460→  const [isSubmitting, setIsSubmitting] = useState(false);
   461→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   462→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   463→  const [appliedCoupon, setAppliedCoupon] = useState<Coupon | null>(null);
   464→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   465→
   466→  // Restore add-ons and multi-variant selections from localStorage
   467→  useEffect(() => {
   468→    try {
   469→      const stored = localStorage.getItem('parwa_pricing_context');
   470→      if (stored) {
   471→        const ctx = JSON.parse(stored) as PricingContext & {
   472→          selectedVariants?: string[];
   473→          variantQuantities?: Record<string, number>;
   474→        };
   475→        if (ctx.addOns) {
   476→          setAddOns(ctx.addOns);
   477→        }
   478→        // Restore multiple variants if selected on ModelsPage
   479→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   480→          const tiers: VariantTier[] = ctx.selectedVariants
   481→            .map((v: string) => {
   482→              const normalized = normalizeTier(v);
   483→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   484→            })
   485→            .filter(Boolean) as VariantTier[];
   486→          if (tiers.length > 0) {
   487→            setActiveVariants(tiers);
   488→          }
   489→        }
   490→        // Restore variant quantities
   491→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   492→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   493→          const newQuantities = { ...variantQuantities };
   494→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   495→            const tier = normalizeTier(variantKey);
   496→            if (['starter', 'growth', 'high'].includes(tier)) {
   497→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   498→            }
   499→          }
   500→          setVariantQuantities(newQuantities);
   501→        }
   502→      }
   503→    } catch {
   504→      // ignore
   505→    }
   506→  }, []);
   507→
   508→  // Check if Paddle.js is available
   509→  useEffect(() => {
   510→    getPaddleInstance().then((paddle) => {
   511→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   512→    }).catch(() => {
   513→      setPaddleStatus('unavailable');
   514→    });
   515→  }, []);
   516→
   517→  // ── Calculations (pure math — D7, D10) ──────────────────────────────
   518→
   519→  const totalTicketLimit = useMemo(
   520→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   521→    [activeVariants, variantQuantities]
   522→  );
   523→
   524→  const baseSubscription = useMemo(
   525→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   526→    [activeVariants, variantQuantities]
   527→  );
   528→
   529→  const addOnTotal = useMemo(() => {
   530→    let total = 0;
   531→    if (addOns.voice) {
   532→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;

   564→
   565→  const savings = estimateSavings(totalTicketLimit, discountedTotal);
   566→
   567→  // Overage projection (estimates at 80% utilization)

   569→  const overageTickets = Math.max(0, projectedTickets - totalTicketLimit);

   571→

   576→          toast.error('You must have at least one active variant');
   577→          return prev;
   578→        }
   579→        return prev.filter((t) => t !== tier);
   580→      }
   581→      return [...prev, tier];
   582→    });
   583→  };
   584→

   587→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   588→  };
   589→
   590→  const toggleAddOn = (key: 'voice' | 'customApi') => {

   592→  };
   593→

   596→    setCheckoutError(null);
   597→  };

   609→      toast.success('Subscription activated! Welcome to PARWA!');
   610→    } else {
   611→      toast.success('Payment successful! Welcome to PARWA!');
   612→    }
   613→    // Only trigger onboarding completion AFTER Paddle confirms
   614→    onComplete();
   615→  };
   616→
   617→  const handleProceed = async () => {
   618→    setIsSubmitting(true);
   619→    setCheckoutError(null);
   620→
   621→    try {
   622→      // Update pricing context in localStorage
   623→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   624→      const context: PricingContext = {
   625→        industry: 'other',
   626→        variant: primaryOnboardingVariant,
   627→        addOns,
   628→        totalMonthly,
   629→        timestamp: new Date().toISOString(),

   633→      try {
   634→        const stored = localStorage.getItem('parwa_pricing_context');

   636→          const existing = JSON.parse(stored) as PricingContext;
   637→          context.industry = existing.industry;
   638→        }

   640→        // ignore
   641→      }
   642→
   643→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));

   647→
   648→      // Add each active variant as a Paddle line item (with quantity)
   649→      for (const tier of activeVariants) {

   651→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];

   665→        checkoutItems.push({ priceId: 'pri_custom_api_addon_01', quantity: 1 });

   669→        source: 'parwa_onboarding',
   670→        variant: primaryOnboardingVariant,
   671→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),

   673→        addOns,
   674→        industry: context.industry,

   687→
   688→      try {
   689→        const res = await fetch('/api/onboarding/checkout', {
   690→          method: 'POST',
   691→          headers: { 'Content-Type': 'application/json' },
   692→          body: JSON.stringify({
   693→            variant: primaryOnboardingVariant,
   694→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),

   698→            industry: context.industry,

   719→          couponCode: appliedCoupon?.code || null,

   744→          paddleDiscountId,

   761→            handlePaymentSuccess,

   772→        // Paddle truly unavailable
   773→      }

   779→        couponCode: appliedCoupon?.code || null,
   780→        pendingAt: new Date().toISOString(),
   781→      }));

   783→        'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   784→        'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'

   786→      toast.error('Payment gateway unavailable — please try again or contact support.');
   787→    } catch (err) {
   788→      console.error('[cost-breakdown] Error:', err);
   789→      setCheckoutError('Something went wrong. Please try again.');
   790→      toast.error('Something went wrong. Please try again.');
   791→    } finally {
   792→      setIsSubmitting(false);
   793→    }
   794→  };
   795→
   796→  return (
   797→    <div className="space-y-6">
   798→      {/* Header */}
   799→      <div className="text-center space-y-2">
   800→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   801→          <Receipt className="w-7 h-7 text-emerald-400" />
   802→        </div>
   803→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   804→        <p className="text-orange-200/40 text-sm">
   805→          Configure your plan and complete payment to activate your AI agents.
   806→        </p>
   807→      </div>
   808→
   809→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   810→      {paddleStatus === 'unavailable' && (
   811→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   812→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   813→          <div>
   814→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   815→            <p className="text-[10px] text-orange-200/30 mt-1">
   816→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   817→            </p>
   818→          </div>
   819→        </div>
   820→      )}
   821→      {paddleStatus === 'ready' && (
   822→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   823→          <Shield className="w-4 h-4 text-emerald-400" />
   824→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   825→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   826→        </div>
   827→      )}
   828→
   829→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   830→      <div className="space-y-3">
   831→        <div className="flex items-center gap-2">
   832→          <Brain className="w-4 h-4 text-orange-400" />
   833→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   834→            Active Variants
   835→          </label>
   836→          <span className="text-[10px] text-orange-200/25">
   837→            Each variant adds its own ticket allocation and AI pipeline
   838→          </span>
   839→        </div>
   840→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   841→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   842→            <VariantMixerCard
   843→              key={tier}
   844→              tier={tier}
   845→              isActive={activeVariants.includes(tier)}
   846→              quantity={variantQuantities[tier] || 1}
   847→              onToggle={() => toggleVariant(tier)}
   848→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   849→              discountedPrice={activeVariants.includes(tier) ? variantDiscountedPrices[tier] : undefined}
   850→            />
   851→          ))}
   852→        </div>
   853→      </div>
   854→
   855→      {/* ── 2. Live Usage Bar & Overage Projection ──────────────────── */}
   856→      <div className="rounded-xl border border-white/[0.06] p-4 space-y-4" style={{ background: 'rgba(255,255,255,0.03)' }}>
   857→        <div className="flex items-center gap-2">
   858→          <BarChart3 className="w-4 h-4 text-orange-400" />
   859→          <span className="text-xs text-orange-200/50 uppercase tracking-wider font-medium">
   860→            Live Usage & Overage Projection
   861→          </span>
   862→        </div>
   863→        <UsageBar
   864→          used={projectedTickets}
   865→          total={totalTicketLimit}
   866→          label="Ticket Usage (projected at 80%)"
   867→          unit="tickets"
   868→          overageRate={`$${OVERAGE_PRICE_PER_TICKET}`}
   869→        />
   870→        {overageCost > 0 && (
   871→          <div className="rounded-lg border border-red-500/20 p-3 flex items-start gap-2" style={{ background: 'rgba(239,68,68,0.05)' }}>
   872→            <AlertTriangle className="w-4 h-4 text-red-400 shrink-0 mt-0.5" />
   873→            <div>
   874→              <p className="text-xs font-medium text-red-400">Overage Projected</p>
   875→              <p className="text-[10px] text-orange-200/30 mt-0.5">
   876→                At 80% utilization, projected overage: ${overageCost.toFixed(2)} ({overageTickets.toLocaleString()} tickets × ${OVERAGE_PRICE_PER_TICKET}/ticket). Overage charged at end of billing cycle.
   877→              </p>
   878→            </div>
   879→          </div>
   880→        )}
   881→        {overageCost === 0 && (
   882→          <div className="rounded-lg border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   883→            <Shield className="w-3.5 h-3.5 text-emerald-400" />
   884→            <p className="text-[10px] text-emerald-400">Within limits at projected usage. No overage charges.</p>
   885→          </div>
   886→        )}
   887→      </div>
   888→
   889→      {/* ── 3. Add-Ons ───────────────────────────────────────────────── */}
   890→      <div className="space-y-3">
   891→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   892→          Optional Add-Ons
   893→        </label>
   894→        {ADD_ONS.map((addOn) => {
   895→          const Icon = addOn.icon;
   896→          const isSelected = addOns[addOn.key];
   897→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   898→          const showPrice = !isIncluded;
   899→

   901→            <button
   902→              key={addOn.key}
   903→              type="button"
   904→              onClick={() => toggleAddOn(addOn.key)}
   905→              className={cn(
   906→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',

   918→                  'w-5 h-5',
   919→                  isSelected ? 'text-orange-400' : 'text-zinc-500'
   920→                )} />
   921→              </div>
   922→              <div className="flex-1 min-w-0">
   923→                <div className="flex items-center gap-2">
   924→                  <p className={cn(
   925→                    'text-sm font-medium',

  1010→          <div className="flex items-center justify-between">

   234→# IN-MEMORY BACKEND (works immediately, no deps)
   235→# ═══════════════════════════════════════════════════════════════
   236→
   237→class InMemoryBackend(StorageBackend):
   238→    """In-memory storage. Data lost on restart. Perfect for dev/testing."""
   239→
   240→    def __init__(self):
   241→        self._notifications: Dict[str, Dict] = {}
   242→        self._flags: List[Dict] = []
   243→        self._audit: List[Dict] = []
   244→        self._quality_scores: List[Dict] = []
   245→        self._batches: Dict[str, List[Dict]] = {}
   246→        self._batch_metas: Dict[str, Dict] = {}
   247→        self._lock = threading.Lock()
   248→        self._next_nfy_id = 1
   249→
   250→    # ── Helpers ───────────────────────────────────────────────
   251→
   252→    def _next_notification_number(self) -> int:
   253→        with self._lock:
   254→            n = self._next_nfy_id
   255→            self._next_nfy_id += 1
   256→            return n
   257→
   258→    @staticmethod
   259→    def _priority_from_score(score: float) -> str:
   260→        if score >= PRIORITY_THRESHOLDS[PRIORITY_CRITICAL]:
   261→            return PRIORITY_CRITICAL
   262→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_HIGH]:
   263→            return PRIORITY_HIGH
   264→        elif score >= PRIORITY_THRESHOLDS[PRIORITY_MEDIUM]:
   265→            return PRIORITY_MEDIUM
   266→        return PRIORITY_LOW
   267→
   268→    # ── Notifications ────────────────────────────────────────
   269→
   270→    async def create_notification(
   271→        self,
   272→        tenant_id: str,
   273→        ntype: str,
   274→        priority_score: float,
   275→        title: str,
   276→        description: str,
   277→        related_tickets: Optional[List[str]] = None,
   278→        batch_key: Optional[str] = None,
   279→        source_data: Optional[Dict] = None,
   280→    ) -> Dict[str, Any]:
   281→        num = self._next_notification_number()
   282→        key = f"PARWA-NFY-{num:03d}"
   283→        now = time.time()
   284→        nf = {
   285→            "id": str(uuid.uuid4()),
   286→            "tenant_id": tenant_id,
   287→            "notification_key": key,
   288→            "type": ntype,
   289→            "priority": self._priority_from_score(priority_score),
   290→            "priority_score": round(priority_score, 4),
   291→            "title": title,
   292→            "description": description,
   293→            "related_tickets": related_tickets or [],
   294→            "batch_key": batch_key,
   295→            "source_data": source_data or {},
   296→            "is_read": False,
   297→            "is_resolved": False,
   298→            "created_at": now,
   299→            "resolved_at": None,
   300→            "backend": "memory",
   301→        }
   302→        self._notifications[key] = nf
   303→        logger.info("NFY created: %s type=%s score=%.3f [%s]", key, ntype, priority_score, tenant_id)
   304→        return nf
   305→
   306→    async def get_notification(self, key: str) -> Optional[Dict[str, Any]]:
   307→        return self._notifications.get(key)
   308→
   309→    async def get_notifications(
   310→        self,
   311→        tenant_id: str,
   312→        include_resolved: bool = False,
   313→        min_priority: Optional[str] = None,
   314→    ) -> List[Dict[str, Any]]:
   315→        priority_order = {PRIORITY_CRITICAL: 0, PRIORITY_HIGH: 1, PRIORITY_MEDIUM: 2, PRIORITY_LOW: 3}
   316→        min_rank = priority_order.get(min_priority, 99) if min_priority else 99
   317→        results = []
   318→        for n in self._notifications.values():
   319→            if n["tenant_id"] != tenant_id:
   320→                continue
   321→            if not include_resolved and n["is_resolved"]:
   322→                continue
   323→            if priority_order.get(n["priority"], 99) > min_rank:
   324→                continue
   325→            results.append(n)
   326→        results.sort(key=lambda x: (-priority_order.get(x["priority"], 99), -x["created_at"]))
   327→        return results
   328→
   329→    async def resolve_notification(self, key: str) -> bool:
   330→        n = self._notifications.get(key)
   331→        if n and not n["is_resolved"]:
   332→            n["is_resolved"] = True
   333→            n["resolved_at"] = time.time()
   334→            return True
   335→        return False
   336→
   337→    async def dismiss_notification(self, key: str) -> bool:
   338→        n = self._notifications.get(key)
   339→        if n:
   340→            n["is_read"] = True
   341→            return True
   342→        return False
   343→
   344→    # ── System Flags ─────────────────────────────────────────
   345→
   346→    async def set_flag(
   347→        self,
   348→        tenant_id: str,
   349→        flag_type: str,
   350→        flag_value: str,
   351→        set_by: str,
   352→        scope: str = "global",
   353→        target_id: Optional[str] = None,
   354→        reason: Optional[str] = None,
   355→        expires_at: Optional[str] = None,
   356→    ) -> Dict[str, Any]:
   357→        flag = {
   358→            "id": str(uuid.uuid4()),
   359→            "tenant_id": tenant_id,
   360→            "flag_type": flag_type,
   361→            "flag_value": flag_value,
   362→            "scope": scope,
   363→            "target_id": target_id,
   364→            "set_by": set_by,
   365→            "reason": reason,
   366→            "expires_at": expires_at,
   367→            "created_at": datetime.now(timezone.utc).isoformat(),
   368→            "revoked_at": None,
   369→            "backend": "memory",
   370→        }
   371→        self._flags.append(flag)
   372→        logger.info("FLAG set: %s=%s scope=%s [%s]", flag_type, flag_value, scope, tenant_id)
   373→        return flag
   374→
   375→    async def get_active_flags(
   376→        self, tenant_id: str, flag_type: Optional[str] = None
   377→    ) -> List[Dict[str, Any]]:
   378→        now = datetime.now(timezone.utc)
   379→        active = []
   380→        for f in self._flags:
   381→            if f["tenant_id"] != tenant_id:
   382→                continue
   383→            if f["revoked_at"] is not None:
   384→                continue
   385→            if f["expires_at"]:
   386→                exp = datetime.fromisoformat(f["expires_at"]) if isinstance(f["expires_at"], str) else f["expires_at"]
   387→                if exp < now:
   388→                    continue
   389→            if flag_type and f["flag_type"] != flag_type:
   390→                continue
   391→            active.append(f)
   392→        return active
   393→
   394→    async def revoke_flag(self, flag_id: str, revoked_by: str) -> bool:
   395→        for f in self._flags:
   396→            if f["id"] == flag_id and f["revoked_at"] is None:
   397→                f["revoked_at"] = datetime.now(timezone.utc).isoformat()
   398→                logger.info("FLAG revoked: %s by %s", flag_id, revoked_by)
   399→                return True
   400→        return False

   402→    # ── Audit Trail ──────────────────────────────────────────
   403→
   404→    async def create_audit_entry(
   405→        self,
   406→        tenant_id: str,
   407→        action: str,
   408→        actor_email: str,
   409→        target_type: Optional[str] = None,
   410→        target_id: Optional[str] = None,
   411→        payload: Optional[Dict] = None,
   412→    ) -> Dict[str, Any]:
   413→        payload_json = json.dumps(payload or {}, sort_keys=True)
   414→        current_hash = hashlib.sha256(
   415→            f"{tenant_id}:{action}:{payload_json}:{time.time()}".encode()
   416→        ).hexdigest()[:64]
   417→
   418→        entry = {
   419→            "id": len(self._audit) + 1,
   420→            "tenant_id": tenant_id,
   421→            "action": action,
   422→            "actor_email": actor_email,
   423→            "target_type": target_type,
   424→            "target_id": target_id,
   425→            "payload": payload or {},
   426→            "current_hash": current_hash,
   427→            "previous_hash": self._audit[-1]["current_hash"] if self._audit else None,
   428→            "created_at": datetime.now(timezone.utc).isoformat(),
   429→            "backend": "memory",
   430→        }
   431→        self._audit.append(entry)
   432→        return entry
   433→
   434→    async def get_audit_trail(
   435→        self,
   436→        tenant_id: str,
   437→        action: Optional[str] = None,
   438→        limit: int = 50,
   439→    ) -> List[Dict[str, Any]]:
   440→        results = [e for e in self._audit if e["tenant_id"] == tenant_id]
   441→        if action:
   442→            results = [e for e in results if e["action"] == action]
   443→        results.sort(key=lambda x: x["id"], reverse=True)
   444→        return results[:limit]

   479→    async def get_quality_stats(self, tenant_id: str, days: int = 7) -> Dict[str, Any]:
   480→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   481→        total = len(tenant_scores)
   482→        if total == 0:
   483→            return {
   484→                "tenant_id": tenant_id, "total_tickets": 0,
   485→                "avg_quality": 0, "avg_confidence": 0,
   486→                "auto_resolved": 0, "escalated": 0, "stuck": 0,
   487→            }
   488→        avg_q = sum(s["overall_score"] for s in tenant_scores) / total
   489→        avg_c_vals = [s["confidence_score"] for s in tenant_scores if s.get("confidence_score")]
   490→        avg_c = sum(avg_c_vals) / len(avg_c_vals) if avg_c_vals else 0
   491→        paths = {}
   492→        for s in tenant_scores:
   493→            p = s.get("resolution_path", "unknown")
   494→            paths[p] = paths.get(p, 0) + 1
   495→        return {
   496→            "tenant_id": tenant_id,
   497→            "total_tickets": total,
   498→            "avg_quality": round(avg_q, 4),
   499→            "avg_confidence": round(avg_c, 4),
   500→            "by_path": paths,
   501→            "auto_resolved": paths.get("simple", 0) + paths.get("complex", 0),
   502→            "escalated": paths.get("escalated", 0),
   503→            "stuck": paths.get("stuck", 0),
   504→        }
   505→
   506→    # ── Batch Queue ──────────────────────────────────────────
   507→
   508→    async def add_to_batch(
   509→        self,
   510→        tenant_id: str,
   511→        batch_key: str,
   512→        ticket_id: str,
   513→        confidence: float,
   514→    ) -> Optional[Dict[str, Any]]:
   515→        BATCH_WINDOW_S = 300
   516→        now = time.time()
   517→
   518→        if tenant_id not in self._batch_metas:
   519→            self._batch_metas[tenant_id] = {}
   520→
   521→        # Find existing batch with same key within window
   522→        existing_key = None
   523→        for bk, meta in self._batch_metas[tenant_id].items():
   524→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   525→                existing_key = bk
   526→                break
   527→
   528→        if existing_key:
   529→            meta = self._batch_metas[tenant_id][existing_key]
   530→            meta["ticket_ids"].append(ticket_id)
   531→            meta["signal_count"] = len(meta["ticket_ids"])
   532→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   533→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   534→            # Don't flush yet — still within window
   535→            return None
   536→        else:
   537→            meta = {
   538→                "batch_key": batch_key,
   539→                "tenant_id": tenant_id,
   540→                "batch_start": now,
   541→                "ticket_ids": [ticket_id],
   542→                "signal_count": 1,
   543→                "confidence_min": confidence,
   544→                "confidence_max": confidence,
   545→                "status": "pending",
   546→            }
   547→            self._batch_metas[tenant_id][batch_key] = meta
   548→            return None
   549→
   550→    async def flush_batches(self, tenant_id: str) -> List[Dict[str, Any]]:
   551→        results = list(self._batch_metas.get(tenant_id, {}).values())
   552→        self._batch_metas[tenant_id] = {}
   553→        return results

   555→    # ── Utility ──────────────────────────────────────────────
   556→
   557→    async def get_notification_stats(self, tenant_id: str) -> Dict[str, Any]:
   558→        tenant_nfs = [n for n in self._notifications.values() if n["tenant_id"] == tenant_id]
   559→        by_priority = {}
   560→        for p in [PRIORITY_CRITICAL, PRIORITY_HIGH, PRIORITY_MEDIUM, PRIORITY_LOW]:
   561→            by_priority[p] = sum(1 for n in tenant_nfs if n["priority"] == p)
   562→        return {
   563→            "total": len(tenant_nfs),
   564→            "unread": sum(1 for n in tenant_nfs if not n["is_read"]),
   565→            "unresolved": sum(1 for n in tenant_nfs if not n["is_resolved"]),

   591→# SUPABASE BACKEND (httpx → PostgREST, no new deps)
   592→# ═══════════════════════════════════════════════════════════════
   593→
   594→class SupabaseBackend(StorageBackend):
   595→    """Supabase storage via REST API (PostgREST). Uses httpx.
   596→
   597→    Requires env vars: SUPABASE_URL, SUPABASE_ANON_KEY
   598→    """
   599→
   600→    def __init__(self):
   601→        if not SUPABASE_URL or not SUPABASE_ANON_KEY:
   602→            raise ValueError("SUPABASE_URL and SUPABASE_ANON_KEY must be set")
   603→        self._url = SUPABASE_URL.rstrip("/")
   604→        self._headers = {
   605→            "apikey": SUPABASE_ANON_KEY,
   606→            "Authorization": f"Bearer {SUPABASE_ANON_KEY}",
   607→            "Content-Type": "application/json",
   608→            "Prefer": "return=representation",
   609→        }
   610→        self._client = None  # Lazy init on first use
   611→
   612→    async def _get_client(self):
   613→        if self._client is None or self._client.is_closed:
   614→            import httpx
   615→            self._client = httpx.AsyncClient(
   616→                base_url=self._url,
   617→                headers=self._headers,
   618→                timeout=15.0,
   619→            )
   620→        return self._client
   621→
   622→    async def _rest_get(self, table: str, query: str = "") -> List[Dict]:
   623→        client = await self._get_client()
   624→        url = f"/rest/v1/{table}"
   625→        if query:
   626→            url += f"?{query}"
   627→        r = await client.get(url)
   628→        r.raise_for_status()
   629→        return r.json() if r.text else []
   630→
   631→    async def _rest_post(self, table: str, data: Dict) -> Dict:
   632→        client = await self._get_client()
   633→        r = await client.post(f"/rest/v1/{table}", json=data)
   634→        r.raise_for_status()
   635→        result = r.json()
   636→        if isinstance(result, list) and result:
   637→            return result[0]
   638→        return result
   639→
   640→    async def _rest_patch(self, table: str, filter_query: str, data: Dict) -> List[Dict]:
   641→        client = await self._get_client()
   642→        r = await client.patch(f"/rest/v1/{table}?{filter_query}", json=data)
   643→        r.raise_for_status()
   644→        return r.json() if r.text else []
   645→
   646→    async def close(self):
   647→        if self._client and not self._client.is_closed:
   648→            await self._client.aclose()

   652→    async def create_notification(
   653→        self,
   654→        tenant_id: str,
   655→        ntype: str,
   656→        priority_score: float,
   657→        title: str,
   658→        description: str,
   659→        related_tickets: Optional[List[str]] = None,
   660→        batch_key: Optional[str] = None,
   661→        source_data: Optional[Dict] = None,
   662→    ) -> Dict[str, Any]:
   663→        # Generate key locally to avoid race conditions
   664→        existing = await self._rest_get(
   665→            TABLE_NOTIFICATIONS,
   666→            f"tenant_id=eq.{tenant_id}&select=notification_key&order=created_at.desc&limit=1"
   667→        )
   668→        next_num = 1
   669→        if existing:
   670→            import re
   671→            m = re.search(r"PARWA-NFY-(\d+)", existing[0].get("notification_key", ""))
   672→            if m:
   673→                next_num = int(m.group(1)) + 1
   674→        key = f"PARWA-NFY-{next_num:03d}"
   675→
   676→        priority = PRIORITY_LOW
   677→        for pname, pthresh in PRIORITY_THRESHOLDS.items():
   678→            if priority_score >= pthresh:
   679→                priority = pname
   680→                break
   681→
   682→        row = await self._rest_post(TABLE_NOTIFICATIONS, {
   683→            "tenant_id": tenant_id,
   684→            "notification_key": key,
   685→            "type": ntype,
   686→            "priority": priority,
   687→            "priority_score": priority_score,
   688→            "title": title,
   689→            "description": description,
   690→            "related_tickets": related_tickets or [],
   691→            "batch_key": batch_key,
   692→            "source_data": source_data or {},
   693→        })
   694→        row["backend"] = "supabase"
   695→        return row

   793→    async def create_audit_entry(
   794→        self,
   795→        tenant_id: str,

   797→        actor_email: str,
   798→        target_type: Optional[str] = None,
   799→        target_id: Optional[str] = None,
   800→        payload: Optional[Dict] = None,
   801→    ) -> Dict[str, Any]:
   802→        payload_json = json.dumps(payload or {}, sort_keys=True)
   803→        current_hash = hashlib.sha256(
   804→            f"{tenant_id}:{action}:{payload_json}:{time.time()}".encode()
   805→        ).hexdigest()[:64]
   806→
   807→        # Get previous hash
   808→        prev = await self._rest_get(
   809→            TABLE_AUDIT,
   810→            f"tenant_id=eq.{tenant_id}&select=current_hash&order=created_at.desc&limit=1"
   811→        )
   812→        prev_hash = prev[0]["current_hash"] if prev else None
   813→
   814→        row = await self._rest_post(TABLE_AUDIT, {
   815→            "tenant_id": tenant_id,
   816→            "action": action,
   817→            "actor_email": actor_email,
   818→            "target_type": target_type,
   819→            "target_id": target_id,
   820→            "payload": payload or {},
   821→            "previous_hash": prev_hash,
   822→            "current_hash": current_hash,
   823→        })
   824→        row["backend"] = "supabase"
   825→        return row

   872→    async def get_quality_stats(self, tenant_id: str, days: int = 7) -> Dict[str, Any]:
   873→        # Supabase RPC would be ideal, but for now use REST + client-side agg
   874→        results = await self._rest_get(

   881→                "tenant_id": tenant_id, "total_tickets": 0,

   883→                "auto_resolved": 0, "escalated": 0, "stuck": 0,

   886→        c_vals = [float(s["confidence_score"]) for s in results if s.get("confidence_score")]
   887→        avg_c = sum(c_vals) / len(c_vals) if c_vals else 0
   888→        paths = {}
   889→        for s in results:
   890→            p = s.get("resolution_path", "unknown")
   891→            paths[p] = paths.get(p, 0) + 1
   892→        return {
   893→            "tenant_id": tenant_id,
   894→            "total_tickets": total,
   895→            "avg_quality": round(avg_q, 4),
   896→            "avg_confidence": round(avg_c, 4),

   898→            "auto_resolved": paths.get("simple", 0) + paths.get("complex", 0),

   905→    async def add_to_batch(
   906→        self,
   907→        tenant_id: str,
   908→        batch_key: str,
   909→        ticket_id: str,

   928→                    "confidence_max": max(float(batch.get("confidence_max", 0.0)), confidence),
   929→                },
   930→            )
   931→            return None
   932→
   933→        # Create new batch
   934→        await self._rest_post(TABLE_BATCH_QUEUE, {
   935→            "tenant_id": tenant_id,
   936→            "batch_key": batch_key,
   937→            "ticket_ids": [ticket_id],

   939→            "confidence_max": confidence,

  1005→            _backend = SupabaseBackend()

  1007→            logger.info("Jarvis DB: Using InMemory backend (set SUPABASE_URL for production)")
  1008→            _backend = InMemoryBackend()

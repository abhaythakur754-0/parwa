    43→    43→    43→def _format_notification(ntype: str, ev: Dict) -> tuple:
    44→    44→    44→    """Format notification title and description by type."""
    45→    45→    45→    signal = ev.get("signal", {})
    46→    46→    46→    score = ev.get("priority_score", 0)
    47→    47→    47→
    48→    48→    48→    if ntype == TYPE_STUCK_TICKET:
    49→    49→    49→        tid = signal.get("ticket_id", "unknown")
    50→    50→    50→        reason = signal.get("reason", "unknown")
    51→    51→    51→        quality = signal.get("quality_score", "?")
    52→    52→    52→        escalation = ev.get("escalation_tier", "soft_reminder")
    53→    53→    53→        hours = signal.get("hours_stuck", 0)
    54→    54→    54→        title = f"Stuck Ticket [{escalation.upper()}]: {tid}"
    55→    55→    55→        desc = (f"Ticket {tid} could not be resolved automatically. "
    56→    56→    56→                f"Reason: {reason}. Quality achieved: {quality}. "
    57→    57→    57→                f"Loops used: {signal.get('loops_used', 0)}. "
    58→    58→    58→                f"Hours stuck: {hours}. Escalation: {escalation}. "
    59→    59→    59→                f"Requires manual review.")
    60→    60→    60→
    61→    61→    61→    elif ntype == TYPE_QUOTA_LOW:
    62→    62→    62→        data = list(signal.values())[0] if signal else {}
    63→    63→    63→        tier = list(signal.keys())[0] if signal else "unknown"
    64→    64→    64→        title = f"Quota {data.get('status', 'warning').upper()}: {tier}"
    65→    65→    65→        desc = (f"{tier.capitalize()} plan quota is {data.get('burn_pct', 0)}% used "
    66→    66→    66→                f"({data.get('remaining', 0)} of {data.get('total', 0)} remaining). "
    67→    67→    67→                f"Consider upgrading or requesting additional quota.")
    68→    68→    68→
    69→    69→    69→    elif ntype == TYPE_ACCURACY_DROP:
    70→    70→    70→        trend = signal.get("trend", "unknown")
    71→    71→    71→        severity = signal.get("severity", "warning")
    72→    72→    72→        trigger = signal.get("trigger", "unknown")
    73→    73→    73→        acc_7d = signal.get("accuracy_7d", "?")
    74→    74→    74→        acc_today = signal.get("accuracy_today", "?")
    75→    75→    75→        title = f"Accuracy Drift [{severity.upper()}]: {trend}"
    76→    76→    76→        desc = (f"Resolution accuracy trend is '{trend}' (severity: {severity}). "
    77→    77→    77→                f"7-day avg: {acc_7d}, Today: {acc_today}. "
    78→    78→    78→                f"Trigger: {trigger}. "
    79→    79→    79→                f"This may indicate KB gaps, policy changes, or model drift. "
    80→    80→    80→                f"Review recent resolutions and update knowledge base if needed.")
    81→    81→    81→
    82→    82→    82→    elif ntype == TYPE_INTEGRATION_DOWN:
    83→    83→    83→        degraded = signal.get("degraded_services", [])
    84→    84→    84→        worst_uptime = signal.get("worst_uptime_pct", 0)
    85→    85→    85→        names = ", ".join(f"{d['name']} ({d['status']}, uptime={d.get('uptime_pct', '?')}%)" for d in degraded)
    86→    86→    86→        title = f"Integration Issues: {signal.get('total_degraded', len(degraded))} service(s) degraded"
    87→    87→    87→        desc = (f"Degraded integrations: {names}. "
    88→    88→    88→                f"Worst uptime: {worst_uptime}%. "
    89→    89→    89→                f"Check API credentials and service status.")
    90→    90→    90→
    91→    91→    91→    elif ntype == "load_bottleneck":
    92→    92→    92→        at_cap = signal.get("at_capacity", [])
    93→    93→    93→        high_load = signal.get("high_load", [])
    94→    94→    94→        vip = signal.get("vip_overflow_risk", False)
    95→    95→    95→        bottlenecks = ", ".join(v["name"] for v in at_cap + high_load)
    96→    96→    96→        title = f"Load Bottleneck: {bottlenecks or 'All variants'}"
    97→    97→    97→        desc = (f"Variant load issue detected. "
    98→    98→    98→                f"At capacity: {', '.join(v['name'] for v in at_cap) or 'none'}. "
    99→    99→    99→                f"High load: {', '.join(v['name'] for v in high_load) or 'none'}. "
   100→   100→   100→                f"VIP overflow risk: {vip}. "
   101→   101→   101→                f"Consider scaling up or routing overflow to available variants.")
   102→   102→   102→
   103→   103→   103→    else:
   104→   104→   104→        title = f"Signal: {ntype}"
   105→   105→   105→        desc = str(signal)
   106→   106→   106→
   107→   107→   107→    return title, desc

   110→   110→   110→async def _create_notifications_from_evals(
   111→   111→   111→    tenant_id: str,
   112→   112→   112→    evaluations: List[Dict[str, Any]],
   113→   113→   113→    db,
   114→   114→   114→) -> List[Dict[str, Any]]:
   115→   115→   115→    """Convert evaluations into notifications via DB. Filter LOW priority."""
   116→   116→   116→    notifications = []
   117→   117→   117→    for ev in evaluations:
   118→   118→   118→        score = ev.get("priority_score", 0)
   119→   119→   119→        if score < 0.40:
   120→   120→   120→            logger.info("LOW priority signal filtered: %s (score=%.3f)", ev.get("type"), score)
   121→   121→   121→            continue
   122→   122→   122→
   123→   123→   123→        ntype = ev.get("type", "unknown")
   124→   124→   124→        title, desc = _format_notification(ntype, ev)
   125→   125→   125→        related = ev.get("related_tickets", [])
   126→   126→   126→
   127→   127→   127→        nf = await db.create_notification(
   128→   128→   128→            tenant_id=tenant_id,
   129→   129→   129→            ntype=ntype,
   130→   130→   130→            priority_score=score,
   131→   131→   131→            title=title,
   132→   132→   132→            description=desc,
   133→   133→   133→            related_tickets=related,
   134→   134→   134→            batch_key=ev.get("type", ""),
   135→   135→   135→            source_data=ev.get("signal", {}),
   136→   136→   136→        )
   137→   137→   137→        notifications.append(nf)
   138→   138→   138→
   139→   139→   139→    return notifications

   144→   144→   144→async def _execute_command(
   145→   145→   145→    intent_result: Dict[str, Any],
   146→   146→   146→    auth_result: AuthResult,
   147→   147→   147→    tenant_id: str,
   148→   148→   148→    signals: Dict[str, Any],
   149→   149→   149→    db,
   150→   150→   150→) -> str:
   151→   151→   151→    """Execute a classified command and return the response text.
   152→   152→   152→
   153→   153→   153→    This is the core "Jarvis does things" function.
   154→   154→   154→    Reads/writes DB based on intent. Returns human-readable response.
   155→   155→   155→    """
   156→   156→   156→    intent = intent_result["intent"]
   157→   157→   157→    target = intent_result["target"]
   158→   158→   158→
   159→   159→   159→    # ── QUERY INTENTS ────────────────────────────────────
   160→   160→   160→    if is_query_intent(intent):
   161→   161→   161→        response = await _handle_query(intent, target, tenant_id, signals, db)
   162→   162→   162→        # Log to audit
   163→   163→   163→        await db.create_audit_entry(
   164→   164→   164→            tenant_id=tenant_id,
   165→   165→   165→            action=f"query:{intent}",
   166→   166→   166→            actor_email=auth_result.email,
   167→   167→   167→            target_type="query",
   168→   168→   168→            payload={"intent": intent, "target": target},
   169→   169→   169→        )
   170→   170→   170→        return response
   171→   171→   171→
   172→   172→   172→    # ── CONTROL INTENTS ──────────────────────────────────
   173→   173→   173→    if is_control_intent(intent):
   174→   174→   174→        response = await _handle_control(intent, target, tenant_id, auth_result, db)
   175→   175→   175→        return response
   176→   176→   176→
   177→   177→   177→    # ── APPROVAL INTENTS ─────────────────────────────────
   178→   178→   178→    if is_approval_intent(intent):
   179→   179→   179→        response = await _handle_approval(intent, target, tenant_id, auth_result, db)
   180→   180→   180→        return response
   181→   181→   181→
   182→   182→   182→    # ── EMERGENCY INTENTS ────────────────────────────────
   183→   183→   183→    if is_emergency_intent(intent):
   184→   184→   184→        response = await _handle_emergency(intent, target, tenant_id, auth_result, db)
   185→   185→   185→        return response
   186→   186→   186→
   187→   187→   187→    # ── EXPLAIN / TEACH / AGENT ──────────────────────────
   188→   188→   188→    if intent == "explain_ticket":
   189→   189→   189→        return await _handle_explain_ticket(target, tenant_id, signals, db)
   190→   190→   190→    if intent == "explain_flag":
   191→   191→   191→        return await _handle_explain_flags(tenant_id, db)
   192→   192→   192→    if intent == "teach_skill":
   193→   193→   193→        return await _handle_teach_skill(target, tenant_id, auth_result, db)
   194→   194→   194→    if intent == "create_agent":
   195→   195→   195→        return await _handle_create_agent(target, tenant_id, auth_result, db)
   196→   196→   196→
   197→   197→   197→    return f"I understood your command but don't have a handler for '{intent}' yet. This will be available in a future wave."
   198→   198→   198→

   286→   286→   286→            # List all notifications

   322→   322→   322→

   324→   324→   324→
   325→   325→   325→    if intent == "query_health":
   326→   326→   326→        health = signals.get("integration_health", {})
   327→   327→   327→        services = health.get("services", {})
   328→   328→   328→        if not services:
   329→   329→   329→            return "No integration health data yet. Run a poll cycle first to collect ping data."
   330→   330→   330→        lines = [f"**Integration Health** ({health.get('healthy_count', 0)} healthy, "

   342→   342→   342→
   343→   343→   343→    if intent == "query_cost":
   344→   344→   344→        costs = signals.get("llm_costs", {})

     1→"""
     2→CRM Bridge — Bidirectional CRM Integration
     3→
     4→Provides:
     5→  - ingest_ticket(): Convert incoming CRM webhooks to PARWA pipeline state
     6→  - push_response(): Push resolved response back to CRM
     7→  - push_escalation(): Notify CRM when ticket is escalated (return to human)
     8→  - push_resume_result(): Push resume result back to CRM after human guidance
     9→
    10→Supported CRMs:
    11→  - Zendesk (REST API + Webhooks)
    12→  - HubSpot (Tickets API + Webhooks)
    13→  - Generic (for any CRM with webhook support)
    14→
    15→All CRM API calls are:
    16→  - Non-blocking (failures logged, don't break pipeline)
    17→  - Tracked in escalation vault for round-trip visibility
    18→  - Configurable via integration credentials in PARWA DB
    19→"""

   190→class ZendeskAdapter(CRMAdapter):
   191→    """Zendesk CRM adapter.
   192→
   193→    Ingestion: Receives ticket.created / ticket.updated webhooks
   194→    Push-back: Uses Zendesk REST API to post comments and update status
   195→
   196→    Webhook payload structure:
   197→    {
   198→        "id": 12345,
   199→        "subject": "Refund request",
   200→        "description": "I want a refund for...",
   201→        "status": "new",
   202→        "priority": "high",
   203→        "requester": {"name": "John", "email": "john@example.com"},
   204→        "custom_fields": [...],
   205→    }
   206→    """
   207→
   208→    API_BASE = "https://{subdomain}.zendesk.com/api/v2"
   209→
   210→    async def parse_incoming_ticket(self, payload: Dict[str, Any]) -> Dict[str, Any]:
   211→        ticket = payload.get("ticket", payload)
   212→
   213→        # Extract content from Zendesk format
   214→        subject = ticket.get("subject", "")
   215→        description = ticket.get("description", "")
   216→        # Combine subject + description as the query
   217→        query = f"{subject}. {description}" if subject and description else description or subject
   218→
   219→        requester = ticket.get("requester", {})
   220→        return {
   221→            "ticket_id": str(ticket.get("id", "")),
   222→            "query": query.strip(),
   223→            "customer_id": str(requester.get("id", "")),
   224→            "customer_email": requester.get("email", ""),
   225→            "customer_name": requester.get("name", ""),
   226→            "channel_type": "email",  # Zendesk primarily email-based
   227→            "metadata": {
   228→                "crm_provider": "zendesk",
   229→                "crm_status": ticket.get("status", "new"),
   230→                "crm_priority": ticket.get("priority", "normal"),
   231→                "crm_tags": ticket.get("tags", []),
   232→                "crm_group_id": ticket.get("group_id", ""),
   233→                "crm_assignee_id": ticket.get("assignee_id", ""),
   234→                "zendesk_ticket_id": ticket.get("id", ""),
   235→            },
   236→        }
   237→
   238→    async def push_response_to_crm(
   239→        self,
   240→        ticket_id: str,
   241→        response: str,
   242→        status: str,
   243→        internal_note: str = "",
   244→        config: Optional[Dict] = None,
   245→    ) -> Dict[str, Any]:
   246→        """Post response as public comment + update ticket status in Zendesk."""
   247→        if not config:
   248→            return {"success": False, "error": "No Zendesk config provided"}
   249→
   250→        try:
   251→            import httpx
   252→            subdomain = config.get("subdomain", "")
   253→            api_key = config.get("api_key", "")
   254→            email = config.get("email", "")
   255→            base_url = self.API_BASE.format(subdomain=subdomain)
   256→
   257→            auth = (f"{email}/token", api_key)
   258→            headers = {"Content-Type": "application/json"}
   259→
   260→            # Map PARWA status to Zendesk status
   261→            zendesk_status_map = {
   262→                "resolved": "solved",
   263→                "escalated": "open",
   264→                "pending_human": "pending",
   265→            }
   266→            zendesk_status = zendesk_status_map.get(status, "solved")
   267→
   268→            # Update ticket status + add comment
   269→            async with httpx.AsyncClient(timeout=15.0, auth=auth) as client:
   270→                # Post public reply
   271→                comment_payload = {
   272→                    "ticket": {
   273→                        "comment": {
   274→                            "body": response,
   275→                            "public": True,
   276→                        },
   277→                        "status": zendesk_status,
   278→                    }
   279→                }
   280→                resp = await client.put(
   281→                    f"{base_url}/tickets/{ticket_id}.json",
   282→                    headers=headers,
   283→                    json=comment_payload,
   284→                )
   285→                resp.raise_for_status()
   286→
   287→                # Post internal note with AI analysis
   288→                if internal_note:
   289→                    note_payload = {
   290→                        "ticket": {
   291→                            "comment": {
   292→                                "body": f"[PARWA AI Analysis]\n{internal_note}",
   293→                                "public": False,
   294→                            }
   295→                        }
   296→                    }
   297→                    await client.put(
   298→                        f"{base_url}/tickets/{ticket_id}.json",
   299→                        headers=headers,
   300→                        json=note_payload,
   301→                    )
   302→
   303→            logger.info("Zendesk: Pushed response to ticket=%s status=%s", ticket_id, zendesk_status)
   304→            return {
   305→                "success": True,
   306→                "crm_ticket_id": ticket_id,
   307→                "crm_status": zendesk_status,
   308→                "crm_provider": "zendesk",
   309→            }
   310→
   311→        except Exception as e:
   312→            logger.error("Zendesk: Failed to push response: %s", e)
   313→            return {"success": False, "error": str(e), "crm_ticket_id": ticket_id}

   532→class HubSpotAdapter(CRMAdapter):
   533→    """HubSpot CRM adapter.
   534→
   535→    Ingestion: Receives ticket.creation / ticket.status_change webhooks
   536→    Push-back: Uses HubSpot Tickets API v2
   537→
   538→    Webhook payload structure:
   539→    {
   540→        "objectId": 12345,
   541→        "properties": {
   542→            "subject": "Refund request",
   543→            "content": "I want a refund...",
   544→            "hs_pipeline_stage": "1",
   545→        },
   546→    }
   547→    """
   548→
   549→    API_BASE = "https://api.hubapi.com/crm/v3/objects/tickets"
   550→
   551→    async def parse_incoming_ticket(self, payload: Dict[str, Any]) -> Dict[str, Any]:
   552→        ticket_id = str(payload.get("objectId", ""))
   553→        props = payload.get("properties", {})
   554→
   555→        subject = props.get("subject", "")
   556→        content = props.get("content", "")
   557→        query = f"{subject}. {content}" if subject and content else content or subject
   558→
   559→        # Get associated contact info if available
   560→        associations = payload.get("associations", {})
   561→        contact = associations.get("contacts", [{}])[0] if associations.get("contacts") else {}
   562→
   563→        return {
   564→            "ticket_id": ticket_id,
   565→            "query": query.strip(),
   566→            "customer_id": contact.get("id", ""),
   567→            "customer_email": contact.get("properties", {}).get("email", ""),
   568→            "customer_name": f"{contact.get('properties', {}).get('firstname', '')} {contact.get('properties', {}).get('lastname', '')}".strip(),
   569→            "channel_type": "email",
   570→            "metadata": {
   571→                "crm_provider": "hubspot",
   572→                "crm_pipeline_stage": props.get("hs_pipeline_stage", "1"),
   573→                "crm_status": props.get("hs_ticket_status", ""),
   574→                "hubspot_ticket_id": ticket_id,
   575→            },
   576→        }

   578→    async def push_response_to_crm(

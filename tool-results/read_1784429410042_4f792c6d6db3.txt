     1→"""
     2→PARWA Jarvis Orchestrator — The Brain That Ties It All Together
     3→
     4→This is the main pipeline that makes Jarvis work. When a client sends a
     5→message, the Orchestrator:
     6→
     7→  1. Loads context — session state, awareness snapshot, conversation history
     8→  2. Decides mode — agentic (customer chat) vs command (admin chat)
     9→  3. Calls LLM with function definitions as tools
    10→  4. LLM decides: just chat, or call a function?
    11→  5. If function call → Safety Gate checks it
    12→  6. If approved → Execute the function against backend APIs
    13→  7. Feed execution result back to LLM
    14→  8. LLM generates a natural, conversational response
    15→  9. Return response to client
    16→
    17→The client NEVER knows about function calls, safety levels, or modes.
    18→They just get a natural response from Jarvis — like talking to a smart
    19→colleague who happens to also be able to DO things.
    20→
    21→Architecture:
    22→  Client message
    23→      → load_context() → session state + awareness + history
    24→      → decide_mode() → "agentic" or "command"
    25→      → get_function_definitions(mode) → relevant tool specs
    26→      → LLM call with tools → function_call or just conversation
    27→      → if function_call:
    28→          → check_safety() → approved / needs_confirmation / needs_approval
    29→          → if approved → execute_function() → backend API call
    30→          → feed result back to LLM → conversational response
    31→      → return natural response to client
    32→
    33→Mode Switching:
    34→  The mode is determined by CONTEXT, not by explicit user action:
    35→    - If the message is in a variant task (customer support conversation),
    36→      mode = "agentic" — only customer-facing functions available
    37→    - If the message is in Jarvis admin chat (managing the platform),
    38→      mode = "command" — all functions available
    39→    - This is determined by looking at the session type and context
    40→
    41→Conversational Feel:
    42→  The SAME LLM that calls functions also crafts the response. After a
    43→  function is executed, the result is fed back to the LLM, which then
    44→  generates a human-like response. No robotic "Command executed successfully"
    45→  messages — instead: "Done! I've paused all AI agents. They'll stay paused
    46→  until you tell me to resume them."
    47→
    48→BC-001: company_id enforced at every layer.
    49→BC-008: Never crash — graceful degradation at every step.
    50→BC-012: All timestamps UTC.
    51→"""
    52→

    92→
    93→def load_context(
    94→    db: Any,
    95→    company_id: str,
    96→    session_id: str,
    97→) -> Dict[str, Any]:
    98→    """Load the full context for a Jarvis chat session.
    99→
   100→    Gathers:
   101→      - Session info (type, variant_tier, industry, mode)
   102→      - Awareness snapshot (system health, ticket volume, etc.)
   103→      - Conversation history (last N messages)
   104→
   105→    Each part is independently wrapped in try/except (BC-008).
   106→    Partial context is better than no context.
   107→
   108→    Args:
   109→        db: SQLAlchemy session.
   110→        company_id: Company ID for BC-001.
   111→        session_id: Session ID.
   112→
   113→    Returns:
   114→        Dict with session, awareness, and history context.
   115→    """
   116→    context: Dict[str, Any] = {
   117→        "company_id": company_id,
   118→        "session_id": session_id,
   119→        "session": {},
   120→        "awareness": {},
   121→        "history": [],
   122→        "tier": "parwa",
   123→        "mode": "command",
   124→    }
   125→
   126→    # ── Load session info ──
   127→    try:
   128→        from database.models.jarvis import JarvisSession
   129→
   130→        session = (
   131→            db.query(JarvisSession)
   132→            .filter(
   133→                JarvisSession.id == session_id,
   134→                JarvisSession.company_id == company_id,
   135→            )
   136→            .first()
   137→        )
   138→
   139→        if session:
   140→            session_ctx = _safe_parse_json(session.context_json) if session.context_json else {}
   141→            context["session"] = {
   142→                "id": str(session.id),
   143→                "type": session.session_type,
   144→                "variant_tier": session_ctx.get("variant_tier", "parwa"),
   145→                "industry": session_ctx.get("industry", "general"),
   146→                "mode": session_ctx.get("mode", "customer_care"),
   147→                "pack_type": session.pack_type,
   148→                "context_keys": list(session_ctx.keys()),
   149→            }
   150→            context["tier"] = session_ctx.get("variant_tier", "parwa")
   151→    except Exception:
   152→        logger.debug("load_context_session_failed: session=%s", session_id, exc_info=True)
   153→
   154→    # ── Load awareness snapshot ──
   155→    try:
   156→        from app.services.jarvis_awareness_engine import get_latest_snapshot
   157→
   158→        snapshot = get_latest_snapshot(db, session_id, company_id)
   159→        if snapshot:
   160→            context["awareness"] = {
   161→                "system_health": snapshot.system_health,
   162→                "ticket_volume_today": snapshot.ticket_volume_today,
   163→                "agent_pool_utilization": snapshot.agent_pool_utilization,
   164→                "quality_score": snapshot.quality_score,
   165→                "drift_score": snapshot.drift_score,
   166→                "active_alerts_count": snapshot.active_alerts_count,
   167→                "current_plan": snapshot.current_plan,
   168→                "plan_usage_today": snapshot.plan_usage_today,
   169→                "subscription_status": snapshot.subscription_status,
   170→            }
   171→    except Exception:
   172→        logger.debug("load_context_awareness_failed: session=%s", session_id, exc_info=True)
   173→
   174→    # ── Load conversation history ──
   175→    try:
   176→        from database.models.jarvis import JarvisMessage
   177→
   178→        messages = (
   179→            db.query(JarvisMessage)
   180→            .filter(
   181→                JarvisMessage.session_id == session_id,
   182→                JarvisMessage.company_id == company_id,
   183→            )
   184→            .order_by(JarvisMessage.created_at.desc())
   185→            .limit(MAX_CONVERSATION_HISTORY)
   186→            .all()
   187→        )
   188→
   189→        # Reverse to get chronological order
   190→        messages = list(reversed(messages))
   191→
   192→        context["history"] = [
   193→            {
   194→                "role": msg.role,
   195→                "content": msg.content[:500] if msg.content else "",  # Truncate long messages
   196→            }
   197→            for msg in messages
   198→        ]
   199→    except Exception:
   200→        logger.debug("load_context_history_failed: session=%s", session_id, exc_info=True)
   201→
   202→    return context
   203→

   205→# ══════════════════════════════════════════════════════════════════
   206→# MODE DECISION
   207→# ══════════════════════════════════════════════════════════════════
   208→
   209→
   210→def decide_mode(context: Dict[str, Any]) -> str:
   211→    """Decide whether Jarvis is in agentic or command mode.
   212→
   213→    Context-based mode switching:
   214→      - If the session is a variant task (customer support conversation):
   215→        mode = "agentic" — only customer-facing functions available
   216→      - If the session is an admin/dashboard chat (managing the platform):
   217→        mode = "command" — all functions available
   218→
   219→    The decision is made by looking at the session type and context.
   220→    The client NEVER has to explicitly switch modes.
   221→
   222→    Args:
   223→        context: The loaded session context.
   224→
   225→    Returns:
   226→        "agentic" or "command"
   227→    """
   228→    session_info = context.get("session", {})
   229→    session_type = session_info.get("type", "")
   230→    session_mode = session_info.get("mode", "")
   231→
   232→    # If this is a customer_care session handling a variant task,
   233→    # we're in agentic mode — Jarvis talks to the client's customers
   234→    if session_type == "customer_care" and session_mode == "customer_care":
   235→        return "agentic"
   236→
   237→    # If this is an onboarding or admin session, we're in command mode
   238→    # — Jarvis is the platform's CLI
   239→    if session_type in ("onboarding", "admin", "command"):
   240→        return "command"
   241→
   242→    # Default: command mode (more capabilities available)
   243→    return "command"
   244→

   246→# ══════════════════════════════════════════════════════════════════
   247→# SYSTEM PROMPT BUILDER
   248→# ══════════════════════════════════════════════════════════════════
   249→
   250→
   251→def build_system_prompt(
   252→    mode: str,
   253→    context: Dict[str, Any],
   254→    pending_safety: Optional[Dict[str, Any]] = None,
   255→) -> str:
   256→    """Build the system prompt for the LLM.
   257→
   258→    This prompt tells the LLM who it is, what it can do, and how to
   259→    behave. It's designed to make Jarvis feel like a smart colleague,
   260→    not a robot.
   261→
   262→    Args:
   263→        mode: "agentic" or "command".
   264→        context: Loaded session context (awareness, history, etc.)
   265→        pending_safety: Info about a pending safety confirmation, if any.
   266→
   267→    Returns:
   268→        System prompt string.
   269→    """
   270→    awareness = context.get("awareness", {})
   271→
   272→    # Base identity
   273→    prompt = (
   274→        "You are Jarvis, an AI assistant for Parwa — a customer support platform. "
   275→        "You help the client manage their support operations.\n\n"
   276→    )
   277→
   278→    # Mode-specific instructions
   279→    if mode == "agentic":
   280→        prompt += (
   281→            "RIGHT NOW you're helping handle a customer conversation. "
   282→            "Be helpful, friendly, and solve the customer's problem. "
   283→            "Use the available tools when needed to look up information or take action.\n\n"
   284→        )
   285→    else:
   286→        prompt += (
   287→            "The client is managing their support platform through you. "
   288→            "They might ask you to check things, change settings, or take actions. "
   289→            "You have tools available to do everything the platform can do. "
   290→            "Use them when the client asks you to do something.\n\n"
   291→        )
   292→
   293→    # Conversational guidelines
   294→    prompt += (
   295→        "HOW TO TALK:\n"
   296→        "- Be conversational and natural, like a smart colleague\n"
   297→        "- Don't be robotic or overly formal\n"
   298→        "- Don't say 'Command executed successfully' — say what actually happened\n"
   299→        "- If you did something, explain the result in plain language\n"
   300→        "- If you need more info, ask naturally (not like a form)\n"
   301→        "- Be concise but friendly\n"
   302→        "- Use the client's context when responding\n\n"
   303→    )
   304→
   305→    # Context injection
   306→    if awareness:
   307→        prompt += "CURRENT SYSTEM STATE:\n"
   308→        health = awareness.get("system_health", "unknown")
   309→        tickets_today = awareness.get("ticket_volume_today", 0)
   310→        quality = awareness.get("quality_score", "N/A")
   311→        utilization = awareness.get("agent_pool_utilization", "N/A")
   312→        plan = awareness.get("current_plan", "unknown")
   313→        usage = awareness.get("plan_usage_today", "N/A")
   314→
   315→        prompt += (
   316→            f"- System health: {health}\n"
   317→            f"- Tickets today: {tickets_today}\n"
   318→            f"- AI quality score: {quality}\n"
   319→            f"- Agent utilization: {utilization}\n"
   320→            f"- Plan: {plan}\n"
   321→            f"- Plan usage: {usage}\n\n"
   322→        )
   323→
   324→    # Pending safety confirmation
   325→    if pending_safety:
   326→        func_name = pending_safety.get("function_name", "")
   327→        safety_level = pending_safety.get("safety_level", "")
   328→        prompt += (
   329→            f"PENDING ACTION: You previously asked permission to run '{func_name}'. "
   330→            f"The client hasn't confirmed yet. "
   331→        )
   332→        if safety_level == "approval_required":
   333→            prompt += "They need to type 'confirm' or 'yes' to approve.\n\n"
   334→        else:
   335→            prompt += "A simple 'yes' or 'go ahead' will confirm.\n\n"
   336→
   337→    return prompt
   338→

   340→# ══════════════════════════════════════════════════════════════════
   341→# LLM CALL WITH FUNCTION CALLING
   342→# ══════════════════════════════════════════════════════════════════
   343→
   344→
   345→async def call_llm_with_functions(
   346→    system_prompt: str,
   347→    messages: List[Dict[str, str]],
   348→    function_definitions: List[Dict[str, Any]],
   349→    company_id: str = "",
   350→) -> Dict[str, Any]:
   351→    """Call the LLM with function calling support.
   352→
   353→    Supports multiple providers:
   354→      - OpenAI-compatible (with tools parameter)
   355→      - z-ai gateway (with tools parameter)
   356→      - LiteLLM (with tools parameter)
   357→      - Fallback: Without function calling (just conversation)
   358→
   359→    Args:
   360→        system_prompt: System prompt for the LLM.
   361→        messages: Conversation history (role + content).
   362→        function_definitions: Tool specs for function calling.
   363→        company_id: Company ID for BC-001.
   364→
   365→    Returns:
   366→        Dict with:
   367→          - "content": str (conversational response text)
   368→          - "function_call": Optional[Dict] (name + arguments if LLM chose to call a function)
   369→          - "model": str
   370→          - "tokens_used": int
   371→          - "latency_ms": float
   372→    """
   373→    start_time = time.monotonic()
   374→
   375→    try:
   376→        from app.core.llm_gateway import llm_gateway
   377→
   378→        # Build full message list
   379→        full_messages = [{"role": "system", "content": system_prompt}] + messages
   380→
   381→        # Try function-calling capable providers first
   382→        result = await _try_function_calling(full_messages, function_definitions, company_id)
   383→
   384→        if result:
   385→            return result
   386→
   387→        # Fallback: call without functions (just conversation)
   388→        response = await llm_gateway.generate(
   389→            system_prompt=system_prompt,
   390→            user_message=messages[-1]["content"] if messages else "",
   391→            technique_id="jarvis_orchestrator",
   392→            max_tokens=500,
   393→            temperature=0.7,
   394→            company_id=company_id,
   395→            messages=full_messages,
   396→        )
   397→
   398→        return {
   399→            "content": response.text or FALLBACK_RESPONSE,
   400→            "function_call": None,
   401→            "model": response.model,
   402→            "tokens_used": response.tokens_used,
   403→            "latency_ms": response.latency_ms,
   404→        }
   405→
   406→    except Exception:
   407→        elapsed_ms = round((time.monotonic() - start_time) * 1000, 2)
   408→        logger.exception("call_llm_error: company=%s", company_id)
   409→        return {
   410→            "content": FALLBACK_RESPONSE,
   411→            "function_call": None,
   412→            "model": "fallback",
   413→            "tokens_used": 0,
   414→            "latency_ms": elapsed_ms,
   415→        }
   416→
   417→
   418→async def _try_function_calling(
   419→    messages: List[Dict[str, str]],
   420→    function_definitions: List[Dict[str, Any]],
   421→    company_id: str,
   422→) -> Optional[Dict[str, Any]]:
   423→    """Try to call LLM with function calling support.
   424→
   425→    Attempts OpenAI-compatible function calling. Returns None if
   426→    the provider doesn't support it or it fails.
   427→    """
   428→    try:
   429→        import os
   430→
   431→        # Check if OpenAI is available for function calling
   432→        api_key = os.environ.get("OPENAI_API_KEY", "")
   433→        base_url = os.environ.get("OPENAI_BASE_URL", "")
   434→        model = os.environ.get("JARVIS_MODEL", os.environ.get("OPENAI_MODEL", "gpt-4o-mini"))
   435→
   436→        # Try z-ai gateway first (if available)
   437→        zai_key = os.environ.get("ZAI_API_KEY", "")
   438→        if zai_key:
   439→            return await _call_zai_with_functions(
   440→                messages, function_definitions, company_id, zai_key
   441→            )
   442→
   443→        # Try OpenAI-compatible
   444→        if api_key:
   445→            return await _call_openai_with_functions(
   446→                messages, function_definitions, company_id,
   447→                api_key, base_url, model,
   448→            )
   449→
   450→        # No function-calling provider available
   451→        return None
   452→
   453→    except Exception:
   454→        logger.debug("function_calling_unavailable", exc_info=True)
   455→        return None
   456→
   457→
   458→async def _call_openai_with_functions(
   459→    messages: List[Dict[str, str]],
   460→    function_definitions: List[Dict[str, Any]],
   461→    company_id: str,
   462→    api_key: str,
   463→    base_url: str,
   464→    model: str,
   465→) -> Dict[str, Any]:
   466→    """Call OpenAI-compatible API with function calling."""
   467→    try:
   468→        from openai import OpenAI
   469→
   470→        kwargs: Dict[str, Any] = {"api_key": api_key}
   471→        if base_url:
   472→            kwargs["base_url"] = base_url
   473→
   474→        client = OpenAI(**kwargs)
   475→
   476→        response = await asyncio.to_thread(
   477→            client.chat.completions.create,
   478→            model=model,
   479→            messages=messages,
   480→            tools=function_definitions if function_definitions else None,
   481→            tool_choice="auto" if function_definitions else None,
   482→            max_tokens=500,
   483→            temperature=0.7,
   484→        )
   485→
   486→        choice = response.choices[0] if response.choices else None
   487→        if not choice:
   488→            return {
   489→                "content": FALLBACK_RESPONSE,
   490→                "function_call": None,
   491→                "model": model,
   492→                "tokens_used": 0,
   493→                "latency_ms": 0,
   494→            }
   495→
   496→        # Extract content and function call
   497→        content = choice.message.content or ""
   498→        function_call = None
   499→
   500→        # Check for tool calls (OpenAI format)
   501→        if hasattr(choice.message, "tool_calls") and choice.message.tool_calls:
   502→            tool_call = choice.message.tool_calls[0]
   503→            function_call = {
   504→                "name": tool_call.function.name,
   505→                "arguments": _safe_parse_json(tool_call.function.arguments),
   506→            }
   507→
   508→        tokens = response.usage.total_tokens if response.usage else 0
   509→
   510→        return {
   511→            "content": content,
   512→            "function_call": function_call,
   513→            "model": response.model if response else model,
   514→            "tokens_used": tokens,
   515→            "latency_ms": 0,
   516→        }
   517→
   518→    except Exception:
   519→        logger.debug("openai_function_call_failed", exc_info=True)
   520→        return None
   521→
   522→
   523→async def _call_zai_with_functions(
   524→    messages: List[Dict[str, str]],
   525→    function_definitions: List[Dict[str, Any]],
   526→    company_id: str,
   527→    api_key: str,
   528→) -> Dict[str, Any]:
   529→    """Call z-ai gateway with function calling support."""
   530→    try:
   531→        import httpx
   532→        import os
   533→
   534→        base_url = os.environ.get("ZAI_BASE_URL", "http://localhost:3000/api")
   535→        model = os.environ.get("ZAI_MODEL", "default")
   536→
   537→        headers = {
   538→            "Content-Type": "application/json",
   539→            "Authorization": f"Bearer {api_key}",
   540→        }
   541→
   542→        payload: Dict[str, Any] = {
   543→            "model": model,
   544→            "messages": messages,
   545→            "max_tokens": 500,
   546→            "temperature": 0.7,
   547→        }
   548→
   549→        if function_definitions:
   550→            payload["tools"] = function_definitions
   551→            payload["tool_choice"] = "auto"
   552→
   553→        async with httpx.AsyncClient(
   554→            timeout=httpx.Timeout(connect=10.0, read=30.0),
   555→        ) as client:
   556→            response = await client.post(
   557→                f"{base_url}/chat/completions",
   558→                headers=headers,
   559→                json=payload,
   560→            )
   561→
   562→        if response.status_code != 200:
   563→            logger.warning("zai_function_call_http_%d", response.status_code)
   564→            return None
   565→
   566→        data = response.json()
   567→        choices = data.get("choices", [])
   568→        if not choices:
   569→            return None
   570→
   571→        choice = choices[0]
   572→        msg = choice.get("message", {})
   573→        content = msg.get("content", "")
   574→
   575→        # Check for tool calls
   576→        function_call = None
   577→        tool_calls = msg.get("tool_calls", [])
   578→        if tool_calls:
   579→            tc = tool_calls[0]
   580→            func = tc.get("function", {})
   581→            function_call = {
   582→                "name": func.get("name", ""),
   583→                "arguments": _safe_parse_json(func.get("arguments", "{}")),
   584→            }
   585→
   586→        tokens = data.get("usage", {}).get("total_tokens", 0)
   587→
   588→        return {
   589→            "content": content,
   590→            "function_call": function_call,
   591→            "model": data.get("model", model),
   592→            "tokens_used": tokens,
   593→            "latency_ms": 0,
   594→        }
   595→
   596→    except Exception:
   597→        logger.debug("zai_function_call_failed", exc_info=True)
   598→        return None
   599→

   601→# ══════════════════════════════════════════════════════════════════
   602→# FUNCTION EXECUTION
   603→# ══════════════════════════════════════════════════════════════════
   604→
   605→
   606→async def execute_function(
   607→    db: Any,
   608→    company_id: str,

   634→    """
   635→    try:
   636→        # Route to the appropriate executor based on function name

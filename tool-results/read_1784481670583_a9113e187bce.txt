   119→    VALID_STATUSES = {
   120→        SubscriptionStatus.ACTIVE,
   121→        SubscriptionStatus.PAST_DUE,
   122→        SubscriptionStatus.PAUSED,
   123→        SubscriptionStatus.CANCELED,
   124→        SubscriptionStatus.PAYMENT_FAILED,
   125→        SubscriptionStatus.PENDING,
   126→    }

   134→    async def _get_paddle(self) -> PaddleClient:
   135→        """Get Paddle client (lazy initialization)."""

   140→    def _validate_variant(self, variant: str) -> str:
   141→        """Validate and normalize variant name."""

   145→                f"Invalid variant: {variant}. "
   146→                f"Must be one of: {', '.join(sorted(self.VALID_VARIANTS))}"
   147→            )

   156→    async def _ensure_paddle_customer(
   157→        self, company: Company, db: Session
   158→    ) -> str:
   159→        """
   160→        Ensure the company has a Paddle customer ID.
   161→        Create one in Paddle if missing, and persist it on the company.
   162→
   163→        Returns:
   164→            The paddle_customer_id string.
   165→        """
   166→        if company.paddle_customer_id:
   167→            return company.paddle_customer_id
   168→
   169→        paddle = await self._get_paddle()
   170→
   171→        # Determine email/name for the Paddle customer record.
   172→        # Fall back to company name + placeholder email if needed.
   173→        customer_email = getattr(company, "billing_email", None) or getattr(company, "email", None)
   174→        if not customer_email:
   175→            # Use a deterministic placeholder so we don't block signup
   176→            customer_email = f"billing+{company.id}@parwa.placeholder"
   177→
   178→        customer_name = company.name or f"Company {company.id[:8]}"
   179→
   180→        try:
   181→            result = await paddle.create_customer(
   182→                email=customer_email,
   183→                name=customer_name,
   184→                custom_data={"company_id": company.id},
   185→            )
   186→            paddle_customer_id = result.get("data", {}).get("id")
   187→            if not paddle_customer_id:
   188→                raise PaddleOperationError(
   189→                    "Paddle create_customer returned no customer ID"
   190→                )
   191→            company.paddle_customer_id = paddle_customer_id
   192→            db.commit()
   193→            logger.info(
   194→                "paddle_customer_created company_id=%s paddle_customer_id=%s",
   195→                company.id,
   196→                paddle_customer_id,
   197→            )
   198→            return paddle_customer_id
   199→        except PaddleError as e:
   200→            raise PaddleOperationError(
   201→                f"Failed to create Paddle customer for company {company.id}: {e}"
   202→            ) from e

   204→    # ── S-08: Helper to create a Paddle customer when no DB session is open ──

   206→    async def _ensure_paddle_customer_async(self, company_id: UUID, company_paddle_customer_id: Optional[str]) -> str:
   207→        """
   208→        Ensure the company has a Paddle customer ID (async-only version).
   209→
   210→        Used when the DB session is not held open across the Paddle call.
   211→        Reads the paddle_customer_id; if missing, creates one via Paddle
   212→        and persists it in a separate DB session.
   213→
   214→        Returns:
   215→            The paddle_customer_id string.
   216→        """
   217→        if company_paddle_customer_id:
   218→            return company_paddle_customer_id
   219→
   220→        # Read company details needed for Paddle
   221→        def _db_read_company():
   222→            with SessionLocal() as db:
   223→                company = db.query(Company).filter(
   224→                    Company.id == str(company_id)
   225→                ).first()
   226→                if not company:
   227→                    return None, None
   228→                return (
   229→                    getattr(company, "billing_email", None) or getattr(company, "email", None),
   230→                    company.name,
   231→                )
   232→
   233→        customer_email, customer_name = await asyncio.to_thread(_db_read_company)
   234→
   235→        if not customer_email:
   236→            customer_email = f"billing+{str(company_id)}@parwa.placeholder"
   237→        if not customer_name:
   238→            customer_name = f"Company {str(company_id)[:8]}"
   239→
   240→        paddle = await self._get_paddle()
   241→
   242→        try:
   243→            result = await paddle.create_customer(
   244→                email=customer_email,
   245→                name=customer_name,
   246→                custom_data={"company_id": str(company_id)},
   247→            )
   248→            paddle_customer_id = result.get("data", {}).get("id")
   249→            if not paddle_customer_id:
   250→                raise PaddleOperationError(
   251→                    "Paddle create_customer returned no customer ID"
   252→                )
   253→
   254→            # Persist the new paddle_customer_id
   255→            def _db_save_paddle_customer():
   256→                with SessionLocal() as db:
   257→                    company = db.query(Company).filter(
   258→                        Company.id == str(company_id)
   259→                    ).first()
   260→                    if company:
   261→                        company.paddle_customer_id = paddle_customer_id
   262→                        db.commit()
   263→
   264→            await asyncio.to_thread(_db_save_paddle_customer)
   265→
   266→            logger.info(
   267→                "paddle_customer_created company_id=%s paddle_customer_id=%s",
   268→                company_id,
   269→                paddle_customer_id,
   270→            )
   271→            return paddle_customer_id
   272→        except PaddleError as e:
   273→            raise PaddleOperationError(
   274→                f"Failed to create Paddle customer for company {company_id}: {e}"
   275→            ) from e

   277→    async def create_subscription(
   278→        self,
   279→        company_id: UUID,
   280→        variant: str,
   281→        payment_method_id: Optional[str] = None,
   282→    ) -> Dict[str, Any]:
   283→        """
   284→        Create a new subscription for a company.
   285→
   286→        Correct Paddle Billing flow:
   287→        ─ New customers (no payment method):
   288→          1. Ensure Paddle customer exists
   289→          2. Create a Paddle checkout transaction
   290→          3. Return checkout URL + create local subscription as PENDING
   291→          4. Frontend redirects to Paddle checkout
   292→          5. Paddle webhook confirms payment → webhook handler activates subscription
   293→
   294→        ─ Existing customers (with saved payment method):
   295→          1. Ensure Paddle customer exists
   296→          2. Create subscription directly in Paddle
   297→          3. If Paddle succeeds, create local subscription as ACTIVE
   298→          4. If Paddle fails, do NOT create the subscription (Bug 1 fix)
   299→
   300→        Args:
   301→            company_id: Company UUID
   302→            variant: Subscription variant (starter/growth/high)
   303→            payment_method_id: Paddle payment method ID (optional for new customers)
   304→
   305→        Returns:
   306→            Dict with subscription info and optional checkout_url
   307→
   308→        Raises:
   309→            SubscriptionAlreadyExistsError: If company has active subscription
   310→            InvalidVariantError: If variant is invalid
   311→            PaddleOperationError: If a Paddle call fails
   312→        """
   313→        variant = self._validate_variant(variant)
   314→
   315→        # ── S-08: DB read phase (offloaded to thread) ──
   316→        def _db_read():
   317→            with SessionLocal() as db:
   318→                # Check for existing subscription with row lock (GAP 1 fix)
   319→                existing = db.query(Subscription).filter(
   320→                    Subscription.company_id == str(company_id),
   321→                    Subscription.status.in_([
   322→                        SubscriptionStatus.ACTIVE.value,
   323→                        SubscriptionStatus.PENDING.value,
   324→                    ]),
   325→                ).with_for_update().first()
   326→
   327→                if existing:
   328→                    raise SubscriptionAlreadyExistsError(
   329→                        f"Company {company_id} already has an active or pending subscription"
   330→                    )
   331→
   332→                # Get company for paddle_customer_id
   333→                company = db.query(Company).filter(
   334→                    Company.id == str(company_id)
   335→                ).first()
   336→
   337→                if not company:
   338→                    raise SubscriptionError(f"Company {company_id} not found")
   339→
   340→                return company.paddle_customer_id
   341→
   342→        company_paddle_customer_id = await asyncio.to_thread(_db_read)
   343→
   344→        # ── Bug 3 fix: Always ensure a Paddle customer exists ──
   345→        paddle_customer_id = await self._ensure_paddle_customer_async(
   346→            company_id, company_paddle_customer_id
   347→        )
   348→
   349→        # Calculate billing period
   350→        now = datetime.now(timezone.utc)
   351→        period_end = self._calculate_period_end(now)
   352→        price_id = self._get_paddle_price_id(variant)
   353→
   354→        # ── Branch: existing customer with payment method vs new checkout ──
   355→        if payment_method_id:
   356→            # ── Existing customer: create subscription directly in Paddle ──
   357→            try:
   358→                paddle = await self._get_paddle()
   359→                result = await paddle.create_subscription(
   360→                    customer_id=paddle_customer_id,
   361→                    price_id=price_id,
   362→                )
   363→                paddle_subscription_id = result.get("data", {}).get("id")
   364→                if not paddle_subscription_id:
   365→                    raise PaddleOperationError(
   366→                        "Paddle create_subscription returned no subscription ID"
   367→                    )
   368→                logger.info(
   369→                    "subscription_created_paddle "
   370→                    "company_id=%s variant=%s paddle_sub_id=%s",
   371→                    company_id,
   372→                    variant,
   373→                    paddle_subscription_id,
   374→                )
   375→            except PaddleError as e:
   376→                # ── Bug 1 fix: If Paddle fails, do NOT create the subscription ──
   377→                logger.error(
   378→                    "subscription_paddle_failed company_id=%s error=%s",
   379→                    company_id,
   380→                    str(e),
   381→                )
   382→                raise PaddleOperationError(
   383→                    f"Failed to create subscription in Paddle: {e}"
   384→                ) from e
   385→
   386→            # ── S-08: DB write phase (offloaded to thread) ──
   387→            def _db_write_active():
   388→                with SessionLocal() as db:
   389→                    # Paddle succeeded — create ACTIVE subscription locally
   390→                    subscription = Subscription(
   391→                        company_id=str(company_id),
   392→                        tier=variant,
   393→                        status=SubscriptionStatus.ACTIVE.value,
   394→                        current_period_start=now,
   395→                        current_period_end=period_end,
   396→                        paddle_subscription_id=paddle_subscription_id,
   397→                    )
   398→
   399→                    db.add(subscription)
   400→
   401→                    # Update company subscription info
   402→                    company = db.query(Company).filter(
   403→                        Company.id == str(company_id)
   404→                    ).first()
   405→                    if company:
   406→                        company.subscription_tier = variant
   407→                        company.subscription_status = SubscriptionStatus.ACTIVE.value
   408→                        company.paddle_subscription_id = paddle_subscription_id
   409→
   410→                    db.commit()
   411→                    db.refresh(subscription)
   412→
   413→                    logger.info(
   414→                        "subscription_created company_id=%s variant=%s sub_id=%s",
   415→                        company_id,
   416→                        variant,
   417→                        subscription.id,
   418→                    )
   419→
   420→                    return {
   421→                        "subscription": self._to_subscription_info(subscription),
   422→                        "checkout_url": None,
   423→                    }
   424→
   425→            return await asyncio.to_thread(_db_write_active)
   426→
   427→        else:
   428→            # ── New customer: create Paddle checkout transaction ──
   429→            # Bug 2 fix: Return checkout URL; subscription stays PENDING
   430→            try:
   431→                paddle = await self._get_paddle()
   432→                txn_result = await paddle.create_checkout_transaction(
   433→                    customer_id=paddle_customer_id,
   434→                    price_id=price_id,
   435→                    custom_data={
   436→                        "company_id": str(company_id),
   437→                        "variant": variant,
   438→                    },
   439→                )
   440→            except PaddleError as e:
   441→                logger.error(
   442→                    "checkout_paddle_failed company_id=%s error=%s",
   443→                    company_id,
   444→                    str(e),
   445→                )
   446→                raise PaddleOperationError(
   447→                    f"Failed to create Paddle checkout transaction: {e}"
   448→                ) from e
   449→
   450→            checkout_url = txn_result.get("data", {}).get("checkout", {}).get("url")
   451→            paddle_transaction_id = txn_result.get("data", {}).get("id")
   452→
   453→            # ── S-08: DB write phase (offloaded to thread) ──
   454→            def _db_write_pending():
   455→                with SessionLocal() as db:
   456→                    # Create subscription as PENDING — webhook will activate it
   457→                    subscription = Subscription(
   458→                        company_id=str(company_id),
   459→                        tier=variant,
   460→                        status=SubscriptionStatus.PENDING.value,
   461→                        current_period_start=now,
   462→                        current_period_end=period_end,
   463→                        paddle_subscription_id=None,
   464→                    )
   465→                    # Store the transaction id in metadata so webhook can correlate
   466→                    import json as _json
   467→                    subscription.metadata_json = _json.dumps({
   468→                        "paddle_transaction_id": paddle_transaction_id,
   469→                        "checkout_url": checkout_url,
   470→                        "requested_variant": variant,
   471→                    })
   472→
   473→                    db.add(subscription)
   474→
   475→                    # Update company status to pending
   476→                    company = db.query(Company).filter(
   477→                        Company.id == str(company_id)
   478→                    ).first()
   479→                    if company:
   480→                        company.subscription_status = SubscriptionStatus.PENDING.value
   481→
   482→                    db.commit()
   483→                    db.refresh(subscription)
   484→
   485→                    logger.info(
   486→                        "subscription_pending_checkout company_id=%s variant=%s sub_id=%s txn_id=%s",
   487→                        company_id,
   488→                        variant,
   489→                        subscription.id,
   490→                        paddle_transaction_id,
   491→                    )
   492→
   493→                    return {
   494→                        "subscription": self._to_subscription_info(subscription),
   495→                        "checkout_url": checkout_url,
   496→                    }
   497→
   498→            return await asyncio.to_thread(_db_write_pending)

   500→    async def get_subscription(self, company_id: UUID) -> Optional[SubscriptionInfo]:
   501→        """
   502→        Get subscription details for a company.
   503→
   504→        Args:
   505→            company_id: Company UUID
   506→
   507→        Returns:
   508→            SubscriptionInfo or None if no subscription
   509→        """
   510→        def _db_work():
   511→            with SessionLocal() as db:
   512→                subscription = db.query(Subscription).filter(
   513→                    Subscription.company_id == str(company_id),
   514→                ).order_by(Subscription.created_at.desc()).first()
   515→
   516→                if not subscription:
   517→                    return None
   518→
   519→                return self._to_subscription_info(subscription)
   520→
   521→        return await asyncio.to_thread(_db_work)

   536→    async def upgrade_subscription(
   537→        self,
   538→        company_id: UUID,
   539→        new_variant: str,
   540→    ) -> Dict[str, Any]:
   541→        """
   542→        Upgrade subscription to a higher tier.
   543→
   544→        Immediate upgrade with proration:
   545→        1. Calculate unused amount from current variant
   546→        2. Calculate prorated charge for new variant
   547→        3. Apply credit and charge
   548→        4. Call Paddle FIRST (Bug 4 fix)
   549→        5. THEN update local DB
   550→
   551→        Args:
   552→            company_id: Company UUID
   553→            new_variant: Target variant (must be higher tier)
   554→
   555→        Returns:
   556→            Dict with subscription info and proration details
   557→
   558→        Raises:
   559→            SubscriptionNotFoundError: No active subscription
   560→            InvalidVariantError: Invalid or non-upgrade variant
   561→            PaddleOperationError: If Paddle update fails
   562→        """
   563→        new_variant = self._validate_variant(new_variant)
   564→
   565→        # ── S-08: DB read phase (offloaded to thread) ──
   566→        def _db_read():
   567→            with SessionLocal() as db:
   568→                # GAP 1 fix: Use row-level lock to prevent race conditions
   569→                subscription = db.query(Subscription).filter(
   570→                    Subscription.company_id == str(company_id),
   571→                    Subscription.status == SubscriptionStatus.ACTIVE.value,
   572→                ).with_for_update().first()
   573→
   574→                if not subscription:
   575→                    raise SubscriptionNotFoundError(
   576→                        f"No active subscription for company {company_id}"
   577→                    )
   578→
   579→                old_variant = subscription.tier
   580→
   581→                # Validate upgrade (not downgrade)
   582→                if not self._is_upgrade(old_variant, new_variant):
   583→                    raise InvalidVariantError(
   584→                        f"Cannot upgrade from {old_variant} to {new_variant}. "
   585→                        "Use downgrade_subscription for lower tiers."
   586→                    )
   587→
   588→                # Same variant - no-op
   589→                if old_variant == new_variant:
   590→                    return {
   591→                        "noop": True,
   592→                        "subscription_info": self._to_subscription_info(subscription),
   593→                    }
   594→
   595→                # Calculate proration
   596→                proration = self._calculate_proration(
   597→                    old_variant=old_variant,
   598→                    new_variant=new_variant,
   599→                    billing_cycle_start=subscription.current_period_start,
   600→                    billing_cycle_end=subscription.current_period_end,
   601→                )
   602→
   603→                return {
   604→                    "noop": False,
   605→                    "old_variant": old_variant,
   606→                    "proration": proration,
   607→                    "paddle_subscription_id": subscription.paddle_subscription_id,
   608→                    "subscription_id": subscription.id,
   609→                }
   610→
   611→        read_result = await asyncio.to_thread(_db_read)
   612→
   613→        # Handle no-op case
   614→        if read_result.get("noop"):
   615→            return {
   616→                "subscription": read_result["subscription_info"],
   617→                "proration": None,
   618→                "message": "Already on this variant",
   619→            }
   620→
   621→        old_variant = read_result["old_variant"]
   622→        proration = read_result["proration"]
   623→
   624→        # ── Bug 4 fix: Call Paddle FIRST, then update local DB ──

   705→    async def downgrade_subscription(
   706→        self,
   707→        company_id: UUID,
   708→        new_variant: str,
   709→    ) -> Dict[str, Any]:
   710→        """
   711→        Downgrade subscription to a lower tier.
   712→
   713→        Effective at next billing cycle:
   714→        - No proration needed
   715→        - Subscription continues at current tier until period end
   716→        - New tier takes effect at next billing date
   717→
   718→        Bug 6 fix: Use scheduled_change_type / scheduled_change_variant
   719→        instead of reusing cancel_at_period_end for downgrades.
   720→
   721→        Args:
   722→            company_id: Company UUID
   723→            new_variant: Target variant (must be lower tier)
   724→
   725→        Returns:
   726→            Dict with subscription info and scheduled change
   727→
   728→        Raises:
   729→            SubscriptionNotFoundError: No active subscription
   730→            InvalidVariantError: Invalid or non-downgrade variant
   731→            PaddleOperationError: If Paddle update fails
   732→        """
   733→        new_variant = self._validate_variant(new_variant)
   734→
   735→        # ── S-08: DB read phase (offloaded to thread) ──
   736→        def _db_read():
   737→            with SessionLocal() as db:
   738→                # GAP 1 fix: Use row-level lock to prevent race conditions
   739→                subscription = db.query(Subscription).filter(
   740→                    Subscription.company_id == str(company_id),
   741→                    Subscription.status == SubscriptionStatus.ACTIVE.value,
   742→                ).with_for_update().first()
   743→
   744→                if not subscription:
   745→                    raise SubscriptionNotFoundError(
   746→                        f"No active subscription for company {company_id}"
   747→                    )
   748→
   749→                old_variant = subscription.tier
   750→
   751→                # Validate downgrade (not upgrade)
   752→                if self._is_upgrade(old_variant, new_variant):
   753→                    raise InvalidVariantError(
   754→                        f"Cannot downgrade from {old_variant} to {new_variant}. "
   755→                        "Use upgrade_subscription for higher tiers."
   756→                    )
   757→
   758→                # Same variant - no-op
   759→                if old_variant == new_variant:
   760→                    return {
   761→                        "noop": True,
   762→                        "subscription_info": self._to_subscription_info(subscription),
   763→                    }
   764→
   765→                return {
   766→                    "noop": False,
   767→                    "old_variant": old_variant,
   768→                    "paddle_subscription_id": subscription.paddle_subscription_id,
   769→                    "subscription_id": subscription.id,
   770→                    "current_period_end": subscription.current_period_end,
   771→                }
   772→
   773→        read_result = await asyncio.to_thread(_db_read)
   774→
   775→        # Handle no-op case
   776→        if read_result.get("noop"):
   777→            return {
   778→                "subscription": read_result["subscription_info"],
   779→                "message": "Already on this variant",
   780→            }
   781→
   782→        old_variant = read_result["old_variant"]
   783→
   784→        # Schedule the price change in Paddle (effective next period)
   785→        if read_result["paddle_subscription_id"]:
   786→            try:
   787→                paddle = await self._get_paddle()
   788→                price_id = self._get_paddle_price_id(new_variant)
   789→                await paddle.update_subscription(
   790→                    read_result["paddle_subscription_id"],
   791→                    items=[{"price_id": price_id}],
   792→                    effective_at=read_result["current_period_end"].isoformat()
   793→                    if read_result["current_period_end"] else None,
   794→                )
   795→                logger.info(
   796→                    "subscription_downgrade_scheduled_paddle "
   797→                    "company_id=%s old=%s new=%s",
   798→                    company_id,
   799→                    old_variant,
   800→                    new_variant,
   801→                )
   802→            except PaddleError as e:
   803→                logger.error(
   804→                    "subscription_paddle_downgrade_failed company_id=%s error=%s",
   805→                    company_id,
   806→                    str(e),
   807→                )
   808→                raise PaddleOperationError(
   809→                    f"Failed to schedule downgrade in Paddle: {e}"
   810→                ) from e
   811→
   812→        # ── S-08: DB write phase (offloaded to thread) ──
   813→        def _db_write():
   814→            with SessionLocal() as db:
   815→                subscription = db.query(Subscription).filter(
   816→                    Subscription.id == read_result["subscription_id"],
   817→                ).with_for_update().first()
   818→
   819→                # ── Bug 6 fix: Use scheduled_change approach, NOT cancel_at_period_end ──
   820→                # Store the pending downgrade distinctly from a cancellation.
   821→                subscription.scheduled_change_type = "downgrade"
   822→                subscription.scheduled_change_variant = new_variant
   823→                # Do NOT set cancel_at_period_end for downgrades — that flag
   824→                # is exclusively for cancellations now.
   825→
   826→                # Also store in metadata_json for backward compatibility and
   827→                # for consumers that read the JSON blob directly.
   828→                import json as _json
   829→                meta = {}
   830→                if subscription.metadata_json:
   831→                    try:
   832→                        meta = _json.loads(subscription.metadata_json)
   833→                    except (TypeError, ValueError):
   834→                        pass
   835→                meta["scheduled_change_type"] = "downgrade"
   836→                meta["scheduled_change_variant"] = new_variant
   837→                subscription.metadata_json = _json.dumps(meta)
   838→
   839→                db.commit()
   840→                db.refresh(subscription)
   841→
   842→                logger.info(
   843→                    "subscription_downgrade_scheduled "
   844→                    "company_id=%s old=%s new=%s effective=%s",
   845→                    company_id,
   846→                    old_variant,
   847→                    new_variant,
   848→                    subscription.current_period_end,
   849→                )
   850→
   851→                return {
   852→                    "subscription": self._to_subscription_info(subscription),
   853→                    "scheduled_change": {
   854→                        "change_type": "downgrade",
   855→                        "current_variant": old_variant,
   856→                        "new_variant": new_variant,

   862→                    ),
   863→                }
   864→
   865→        return await asyncio.to_thread(_db_write)

   867→    async def cancel_subscription(
   868→        self,
   869→        company_id: UUID,
   870→        reason: Optional[str] = None,
   871→        effective_immediately: bool = False,
   872→        user_id: Optional[UUID] = None,
   873→    ) -> Dict[str, Any]:
   874→        """
   875→        Cancel a subscription.
   876→
   877→        Netflix-style cancellation:
   878→        - Default: Access until end of current billing period
   879→        - effective_immediately: Stop now (no refund)
   880→
   881→        Bug 5 fix: If Paddle cancel fails, raise an error instead of
   882→        silently continuing.
   883→
   884→        Bug 7 fix: user_id is nullable — don't fall back to company_id.
   885→
   886→        Args:
   887→            company_id: Company UUID
   888→            reason: Optional cancellation reason
   889→            effective_immediately: If True, cancel now; otherwise at period end
   890→            user_id: User who initiated cancellation (nullable)
   891→
   892→        Returns:
   893→            Dict with cancellation details
   894→
   895→        Raises:
   896→            SubscriptionNotFoundError: No active subscription
   897→            PaddleOperationError: If Paddle cancellation fails
   898→        """
   899→        # ── S-08: DB read phase (offloaded to thread) ──
   900→        def _db_read():
   901→            with SessionLocal() as db:
   902→                # GAP 1 fix: Use row-level lock to prevent race conditions
   903→                subscription = db.query(Subscription).filter(
   904→                    Subscription.company_id == str(company_id),
   905→                    Subscription.status == SubscriptionStatus.ACTIVE.value,
   906→                ).with_for_update().first()
   907→
   908→                if not subscription:
   909→                    raise SubscriptionNotFoundError(
   910→                        f"No active subscription for company {company_id}"
   911→                    )
   912→
   913→                return {
   914→                    "subscription_id": subscription.id,
   915→                    "paddle_subscription_id": subscription.paddle_subscription_id,
   916→                }
   917→
   918→        read_result = await asyncio.to_thread(_db_read)
   919→
   920→        canceled_at = datetime.now(timezone.utc)
   921→
   922→        # Cancel in Paddle
   923→        if read_result["paddle_subscription_id"]:
   924→            try:
   925→                paddle = await self._get_paddle()
   926→                await paddle.cancel_subscription(
   927→                    read_result["paddle_subscription_id"],
   928→                    effective_from="immediately" if effective_immediately else "next_billing_period",
   929→                    reason=reason,
   930→                )
   931→            except PaddleError as e:
   932→                # ── Bug 5 fix: Raise instead of silently swallowing ──
   933→                logger.error(
   934→                    "subscription_paddle_cancel_failed company_id=%s error=%s",
   935→                    company_id,
   936→                    str(e),
   937→                )
   938→                raise PaddleOperationError(
   939→                    f"Failed to {'cancel' if effective_immediately else 'schedule cancellation of'} subscription in Paddle: {e}"
   940→                ) from e
   941→
   942→        # ── S-08: DB write phase (offloaded to thread) ──
   943→        def _db_write():
   944→            with SessionLocal() as db:
   945→                subscription = db.query(Subscription).filter(
   946→                    Subscription.id == read_result["subscription_id"],
   947→                ).with_for_update().first()
   948→
   949→                if effective_immediately:
   950→                    # Immediate cancellation
   951→                    subscription.status = SubscriptionStatus.CANCELED.value
   952→                    subscription.cancel_at_period_end = False
   953→                    # Clear any scheduled change (downgrade) since we're canceling
   954→                    subscription.scheduled_change_type = None
   955→                    subscription.scheduled_change_variant = None
   956→
   957→                    # Update company status
   958→                    company = db.query(Company).filter(
   959→                        Company.id == str(company_id)
   960→                    ).first()
   961→                    if company:
   962→                        company.subscription_status = SubscriptionStatus.CANCELED.value
   963→                else:
   964→                    # Cancel at period end (Netflix style)
   965→                    subscription.cancel_at_period_end = True
   966→                    subscription.scheduled_change_type = "cancel"
   967→                    subscription.scheduled_change_variant = None
   968→                    # Status remains active until period end
   969→
   970→                # ── Bug 7 fix: user_id is nullable — no company_id fallback ──
   971→                cancellation = CancellationRequest(
   972→                    company_id=str(company_id),
   973→                    user_id=str(user_id) if user_id else None,
   974→                    reason=reason or "",
   975→                    status="completed" if effective_immediately else "scheduled",
   976→                )
   977→                db.add(cancellation)
   978→
   979→                db.commit()
   980→                db.refresh(subscription)
   981→
   982→                logger.info(
   983→                    "subscription_canceled company_id=%s immediate=%s effective=%s",
   984→                    company_id,
   985→                    effective_immediately,
   986→                    "now" if effective_immediately else "period_end",
   987→                )
   988→
   989→                return {
   990→                    "subscription": self._to_subscription_info(subscription),
   991→                    "cancellation": {
   992→                        "effective_immediately": effective_immediately,
   993→                        "access_until": (
   994→                            None if effective_immediately
   995→                            else subscription.current_period_end
   996→                        ),
   997→                        "canceled_at": canceled_at,
   998→                    },
   999→                    "message": (
  1000→                        "Subscription canceled immediately."
  1001→                        if effective_immediately else
  1002→                        f"Subscription will be canceled at end of billing period "
  1003→                        f"({subscription.current_period_end.isoformat() if subscription.current_period_end else 'period end'}). "
  1004→                        "You can continue using PARWA until then."
  1005→                    ),
  1006→                }
  1007→
  1008→        return await asyncio.to_thread(_db_write)

  1111→    def _calculate_period_end(self, start: datetime) -> datetime:
  1112→        """Calculate the end of a billing period (1 month from start)."""
  1113→        import calendar
  1114→
  1115→        month = start.month + 1
  1116→        year = start.year
  1117→        if month > 12:
  1118→            month = 1
  1119→            year += 1
  1120→        # Get last day of target month to handle 31st-day edge cases
  1121→        last_day = calendar.monthrange(year, month)[1]
  1122→        day = min(start.day, last_day)
  1123→        return start.replace(year=year, month=month, day=day)

  1125→    def _calculate_proration(
  1126→        self,
  1127→        old_variant: str,
  1128→        new_variant: str,
  1129→        billing_cycle_start: Optional[datetime],
  1130→        billing_cycle_end: Optional[datetime],
  1131→    ) -> Dict[str, Any]:
  1132→        """
  1133→        Calculate proration for variant change.
  1134→
  1135→        Formula:
  1136→        - unused_amount = (old_price / days_in_period) * days_remaining
  1137→        - new_daily_rate = new_price / days_in_period
  1138→        - new_charge = new_daily_rate * days_remaining
  1139→        - net_charge = new_charge - unused_amount
  1140→        """
  1141→        now = datetime.now(timezone.utc)
  1142→
  1143→        # Handle None dates
  1144→        if billing_cycle_start is None:
  1145→            billing_cycle_start = now.replace(day=1, hour=0, minute=0, second=0)
  1146→        if billing_cycle_end is None:
  1147→            billing_cycle_end = self._calculate_period_end(billing_cycle_start)
  1148→
  1149→        # Ensure timezone-aware datetimes
  1150→        if billing_cycle_start.tzinfo is None:
  1151→            billing_cycle_start = billing_cycle_start.replace(tzinfo=timezone.utc)
  1152→        if billing_cycle_end.tzinfo is None:
  1153→            billing_cycle_end = billing_cycle_end.replace(tzinfo=timezone.utc)
  1154→
  1155→        old_price = self._get_variant_price(old_variant)
  1156→        new_price = self._get_variant_price(new_variant)
  1157→
  1158→        # Calculate days
  1159→        total_period = billing_cycle_end - billing_cycle_start
  1160→        remaining = billing_cycle_end - now
  1161→
  1162→        days_in_period = max(total_period.days, 1)
  1163→        days_remaining = max(remaining.days, 0)
  1164→
  1165→        # Calculate amounts (BC-002: using Decimal)
  1166→        daily_rate_old = old_price / Decimal(days_in_period)
  1167→        daily_rate_new = new_price / Decimal(days_in_period)
  1168→
  1169→        unused_amount = daily_rate_old * Decimal(days_remaining)
  1170→        new_charge = daily_rate_new * Decimal(days_remaining)
  1171→        proration_credit = unused_amount
  1172→        net_charge = new_charge - proration_credit
  1173→
  1174→        return {
  1175→            "old_variant": old_variant,
  1176→            "new_variant": new_variant,
  1177→            "old_price": old_price,
  1178→            "new_price": new_price,
  1179→            "days_in_period": days_in_period,
  1180→            "days_remaining": days_remaining,
  1181→            "unused_amount": unused_amount.quantize(Decimal("0.01")),
  1182→            "proration_credit": proration_credit.quantize(Decimal("0.01")),
  1183→            "new_charge": new_charge.quantize(Decimal("0.01")),
  1184→            "net_charge": net_charge.quantize(Decimal("0.01")),
  1185→            "billing_cycle_start": billing_cycle_start.date(),
  1186→            "billing_cycle_end": billing_cycle_end.date(),
  1187→        }

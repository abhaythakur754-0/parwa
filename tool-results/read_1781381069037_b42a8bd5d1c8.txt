    60→function fmt(n: number): string {
    61→  return n.toLocaleString('en-US', {
    62→    style: 'currency',
    63→    currency: 'USD',
    64→    maximumFractionDigits: 0,
    65→  });
    66→}
    67→
    68→function fmtNum(n: number): string {
    69→  return n.toLocaleString('en-US', { maximumFractionDigits: 0 });
    70→}
    71→
    72→function fmtPct(n: number): string {
    73→  return `${Math.round(n)}%`;
    74→}

    76→// ── Usage Bar Component ────────────────────────────────────────────
    77→
    78→function UsageBar({
    79→  used,
    80→  total,
    81→  label,
    82→  unit,
    83→  color,
    84→  overageRate,
    85→}: {
    86→  used: number;
    87→  total: number;
    88→  label: string;
    89→  unit: string;
    90→  color: string;
    91→  overageRate?: string;
    92→}) {
    93→  const pct = total > 0 ? Math.min((used / total) * 100, 100) : 0;
    94→  const isNearLimit = pct >= 80;
    95→  const isOverLimit = used > total;
    96→
    97→  return (
    98→    <div className="space-y-2">
    99→      <div className="flex items-center justify-between">
   100→        <span className="text-xs text-zinc-400">{label}</span>
   101→        <span
   102→          className={cn(
   103→            'text-xs font-medium',
   104→            isOverLimit ? 'text-red-400' : isNearLimit ? 'text-amber-400' : 'text-zinc-300'
   105→          )}
   106→        >
   107→          {fmtNum(used)} / {fmtNum(total)} {unit}
   108→        </span>
   109→      </div>
   110→      <div className="h-2 bg-white/5 rounded-full overflow-hidden relative">
   111→        <div className={cn('h-full rounded-full transition-all duration-700', color)} style={{ width: `${pct}%` }} />
   112→        {isOverLimit && (
   113→          <div
   114→            className="absolute top-0 h-full bg-red-500/30 rounded-r-full"
   115→            style={{ left: '100%', width: `${Math.min(((used - total) / total) * 100, 50)}%` }}
   116→          />
   117→        )}
   118→      </div>
   119→      {isOverLimit && overageRate && (
   120→        <p className="text-[10px] text-red-400 flex items-center gap-1">
   121→          <AlertTriangle className="w-3 h-3" />
   122→          Over limit — {fmtNum(used - total)} overage tickets at {overageRate}/ticket
   123→        </p>
   124→      )}
   125→      {isNearLimit && !isOverLimit && (
   126→        <p className="text-[10px] text-amber-400 flex items-center gap-1">
   127→          <AlertTriangle className="w-3 h-3" />
   128→          Approaching limit — {fmtPct(pct)} used
   129→        </p>
   130→      )}
   131→    </div>
   132→  );
   133→}

   267→export default function CostBreakdownPage() {
   268→  const [isLoading, setIsLoading] = useState(true);
   269→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>(['starter']);
   270→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({
   271→    voice: false,
   272→    customApi: false,
   273→  });
   274→  const [billingCycle, setBillingCycle] = useState<BillingCycle>('monthly');
   275→  const [usageData, setUsageData] = useState<UsageData>({
   276→    ticketsUsed: 0,
   277→    agentsUsed: 0,
   278→    docsUsed: 0,
   279→  });
   280→
   281→  // Fetch usage data on mount
   282→  useEffect(() => {
   283→    const fetchData = async () => {
   284→      try {
   285→        // Try to get current subscription tier and usage
   286→        const subRes = await fetch('/api/billing/subscription', {
   287→          credentials: 'include',
   288→          headers: { 'Content-Type': 'application/json' },
   289→        });
   290→        if (subRes.ok) {
   291→          const subData = await subRes.json();
   292→          const tier = normalizeTier(subData.variant_tier || subData.tier || 'mini');
   293→          setActiveVariants([tier]);
   294→        }
   295→
   296→        // Try to get usage
   297→        const usageRes = await fetch('/api/billing/usage', {
   298→          credentials: 'include',
   299→          headers: { 'Content-Type': 'application/json' },
   300→        });
   301→        if (usageRes.ok) {
   302→          const data = await usageRes.json();
   303→          setUsageData({
   304→            ticketsUsed: Number(data.tickets_used ?? data.ticketsUsed ?? 0),
   305→            agentsUsed: Number(data.agents_used ?? data.agentsUsed ?? 0),
   306→            docsUsed: Number(data.docs_used ?? data.docsUsed ?? 0),
   307→          });
   308→        }
   309→
   310→        // Try localStorage for onboarding context
   311→        try {
   312→          const stored = localStorage.getItem('parwa_pricing_context');
   313→          if (stored) {
   314→            const ctx = JSON.parse(stored);
   315→            if (ctx.variant) {
   316→              const tier = normalizeTier(ctx.variant);
   317→              if (!activeVariants.includes(tier)) {
   318→                setActiveVariants([tier]);
   319→              }
   320→            }
   321→            if (ctx.addOns) {
   322→              setAddOns(ctx.addOns);
   323→            }
   324→          }
   325→        } catch {
   326→          // ignore
   327→        }
   328→      } catch {
   329→        // Use defaults
   330→      } finally {
   331→        setIsLoading(false);
   332→      }
   333→    };
   334→
   335→    fetchData();
   336→  }, []);
   337→
   338→  // Toggle variant
   339→  const toggleVariant = useCallback((tier: VariantTier) => {
   340→    setActiveVariants((prev) => {
   341→      if (prev.includes(tier)) {
   342→        // Don't allow removing the last variant
   343→        if (prev.length <= 1) {
   344→          toast.error('You must have at least one active variant');
   345→          return prev;
   346→        }
   347→        return prev.filter((t) => t !== tier);
   348→      }
   349→      return [...prev, tier];
   350→    });
   351→  }, []);
   352→
   353→  // Toggle add-on
   354→  const toggleAddOn = useCallback((key: 'voice' | 'customApi') => {
   355→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   356→  }, []);
   357→
   358→  // Calculated cost breakdown
   359→  const costBreakdown = useMemo(
   360→    () => calculateCostBreakdown(activeVariants, addOns, usageData.ticketsUsed),
   361→    [activeVariants, addOns, usageData.ticketsUsed]
   362→  );
   363→
   364→  // What-if previews for tiers not currently active
   365→  const whatIfPreviews = useMemo(() => {
   366→    const allTiers: VariantTier[] = ['starter', 'growth', 'high'];
   367→    return allTiers
   368→      .filter((tier) => !activeVariants.includes(tier))
   369→      .map((tier) => ({
   370→        tier,
   371→        preview: calculateWhatIfUpgrade(activeVariants, addOns, usageData.ticketsUsed, tier),
   372→      }));
   373→  }, [activeVariants, addOns, usageData.ticketsUsed]);
   374→
   375→  // Annual calculation
   376→  const annualTotal = costBreakdown.totalMonthly * 12;
   377→
   378→  // Active price for display
   379→  const displayTotal = billingCycle === 'annual' ? annualTotal / 12 : costBreakdown.totalMonthly;
   380→
   381→  // ── Loading State ───────────────────────────────────────────────
   382→  if (isLoading) {
   383→    return (
   384→      <div className="space-y-6">
   385→        <div className="pb-6 border-b border-white/[0.06]">
   386→          <div className="h-6 w-40 bg-white/5 rounded animate-pulse" />
   387→          <div className="h-4 w-72 bg-white/5 rounded mt-2 animate-pulse" />
   388→        </div>
   389→        <div className="h-64 bg-white/5 rounded-xl animate-pulse" />
   390→        <div className="h-48 bg-white/5 rounded-xl animate-pulse" />
   391→      </div>
   392→    );
   393→  }
   394→
   395→  // ── Render ──────────────────────────────────────────────────────
   396→
   397→  return (
   398→    <div className="space-y-6">
   399→      {/* ─── Header ─────────────────────────────────────────────── */}
   400→      <div className="flex items-center justify-between pb-6 border-b border-white/[0.06]">
   401→        <div>
   402→          <h1 className="text-xl font-bold text-white flex items-center gap-2">
   403→            <Calculator className="w-5 h-5 text-orange-400" />
   404→            Cost Breakdown
   405→          </h1>
   406→          <p className="text-sm text-zinc-500 mt-0.5">
   407→            Pure math. No hidden fees. Need more? Add another variant.
   408→          </p>
   409→        </div>
   410→        <div className="flex items-center gap-3">
   411→          {/* Billing cycle toggle */}
   412→          <div className="flex items-center gap-2 bg-white/[0.03] border border-white/[0.06] rounded-lg px-3 py-1.5">
   413→            <span
   414→              className={cn('text-xs font-medium', billingCycle === 'monthly' ? 'text-white' : 'text-zinc-500')}
   415→            >
   416→              Monthly
   417→            </span>
   418→            <button
   419→              onClick={() => setBillingCycle((prev) => (prev === 'monthly' ? 'annual' : 'monthly'))}
   420→              className={cn(
   421→                'relative w-9 h-5 rounded-full transition-colors',
   422→                billingCycle === 'annual' ? 'bg-orange-500' : 'bg-white/10'
   423→              )}
   424→              aria-label="Toggle billing cycle"
   425→            >
   426→              <span
   427→                className={cn(
   428→                  'absolute top-0.5 w-4 h-4 rounded-full bg-white shadow transition-transform',
   429→                  billingCycle === 'annual' ? 'translate-x-4' : 'translate-x-0.5'
   430→                )}
   431→              />
   432→            </button>
   433→            <span
   434→              className={cn('text-xs font-medium', billingCycle === 'annual' ? 'text-white' : 'text-zinc-500')}
   435→            >
   436→              Annual
   437→            </span>
   438→          </div>

   442→      {/* ─── 1. Variant Mixer ──────────────────────────────────── */}
   443→      <div>

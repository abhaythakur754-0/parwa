     1→"""
     2→PARWA Payment Tracking Service (Auto-Block System)
     3→
     4→Tracks payment due dates and automatically blocks accounts that miss payments.
     5→
     6→Business Rules:
     7→- Every subscription has a payment tracking record
     8→- When payment is received: update last_payment_received_at, reset counters
     9→- When payment is overdue past block_after_hours: auto-block account
    10→- Blocked accounts can't use services until payment is received
    11→- All payment history is logged for audit trail
    12→
    13→Integration Points:
    14→- subscription_service.py: Create tracking record on new subscription
    15→- webhook handlers: Record payments from Razorpay/Paddle/FlexPay
    16→- billing_tasks.py: Periodic check for overdue accounts
    17→- API middleware: Check if company is blocked before allowing access
    18→
    19→⚠️ NEW SERVICE (2025-01-19): Part of RAM-to-DB migration.
    20→   Ensures payment state survives server crashes.
    21→"""
    22→
    23→from datetime import datetime, timezone, timedelta
    24→from decimal import Decimal
    25→from typing import Any, Dict, List, Optional
    26→
    27→from sqlalchemy.orm import Session

    44→    def __init__(self, company_id: str, reason: str = None):
    45→        self.company_id = company_id
    46→        self.reason = reason or "Account blocked due to missed payment"
    47→        super().__init__(f"Company {company_id} is blocked: {self.reason}")

    68→    def create_tracking_record(
    69→        self,
    70→        company_id: str,
    71→        subscription_id: Optional[str] = None,
    72→        user_id: Optional[str] = None,
    73→        current_period_start: datetime = None,
    74→        current_period_end: datetime = None,
    75→        next_payment_due_at: datetime = None,
    76→        expected_amount: Decimal = None,
    77→        block_after_hours: int = None,
    78→    ) -> PaymentTracking:
    79→        """Create a new payment tracking record for a subscription.
    80→        
    81→        Called when:
    82→          - New subscription is created
    83→          - Subscription is upgraded/downgraded
    84→          - New billing period starts
    85→        
    86→        Args:
    87→            company_id: The company ID to track
    88→            subscription_id: Link to subscription table
    89→            user_id: Owner of the subscription
    90→            current_period_start: When this billing period starts
    91→            current_period_end: When this billing period ends
    92→            next_payment_due_at: When next payment is due
    93→            expected_amount: How much is expected this period
    94→            block_after_hours: Hours after which to auto-block
    95→        
    96→        Returns:
    97→            Created PaymentTracking object
    98→        """
    99→        db = self._get_db()
   100→        try:
   101→            # Check if record already exists
   102→            existing = db.query(PaymentTracking).filter(
   103→                PaymentTracking.company_id == company_id,
   104→            ).first()
   105→            
   106→            if existing:
   107→                # Update existing record instead of creating duplicate
   108→                existing.subscription_id = subscription_id
   109→                existing.user_id = user_id
   110→                if current_period_start:
   111→                    existing.current_period_start = current_period_start
   112→                if current_period_end:
   113→                    existing.current_period_end = current_period_end
   114→                if next_payment_due_at:
   115→                    existing.next_payment_due_at = next_payment_due_at
   116→                if expected_amount is not None:
   117→                    existing.expected_amount = expected_amount
   118→                if block_after_hours is not None:
   119→                    existing.block_after_hours = block_after_hours
   120→                existing.updated_at = datetime.now(timezone.utc)
   121→                
   122→                db.commit()
   123→                logger.info(
   124→                    "payment_tracking_updated: company=%s, due=%s",
   125→                    company_id, next_payment_due_at,
   126→                )
   127→                return existing
   128→            
   129→            # Create new record
   130→            now = datetime.now(timezone.utc)
   131→            record = PaymentTracking(
   132→                company_id=company_id,
   133→                subscription_id=subscription_id,
   134→                user_id=user_id,
   135→                current_period_start=current_period_start or now,
   136→                current_period_end=current_period_end or (now + timedelta(days=30)),
   137→                next_payment_due_at=next_payment_due_at or (now + timedelta(days=30)),
   138→                expected_amount=expected_amount or Decimal("0.00"),
   139→                received_amount=Decimal("0.00"),
   140→                outstanding_balance=expected_amount or Decimal("0.00"),
   141→                block_after_hours=block_after_hours or self.DEFAULT_BLOCK_AFTER_HOURS,
   142→                block_threshold_amount=self.DEFAULT_BLOCK_THRESHOLD,
   143→                payment_status='current',
   144→                is_blocked=False,
   145→            )
   146→            
   147→            db.add(record)
   148→            db.commit()
   149→            
   150→            logger.info(
   151→                "payment_tracking_created: company=%s, due=%s, amount=%s",
   152→                company_id, next_payment_due_at, expected_amount,
   153→            )
   154→            
   155→            return record
   156→            
   157→        except Exception as exc:
   158→            db.rollback()
   159→            logger.error(
   160→                "create_tracking_failed: company=%s, error=%s",
   161→                company_id, str(exc),
   162→            )
   163→            raise PaymentTrackingError(f"Failed to create tracking record: {exc}")
   164→        finally:
   165→            db.close()

   177→    def check_company_blocked(self, company_id: str) -> bool:
   178→        """Check if a company's account is currently blocked.
   179→        
   180→        Use this in API middleware to block requests from delinquent accounts.
   181→        
   182→        Returns True if blocked, False otherwise.
   183→        """
   184→        tracking = self.get_tracking(company_id)
   185→        if not tracking:
   186→            return False
   187→        
   188→        # Also check if they should be blocked but haven't been updated yet
   189→        self._check_and_update_block_status(tracking)
   190→        
   191→        return tracking.is_blocked

   194→        """Raise exception if company account is blocked.

   196→        Call this at the start of any paid feature to ensure
   197→        the account is in good standing.

   200→            CompanyBlockedError: If account is blocked

   213→    def record_payment_received(
   214→        self,
   215→        company_id: str,
   216→        amount: Decimal,
   217→        payment_method: str = None,
   218→        external_payment_id: str = None,
   219→        external_order_id: str = None,
   220→        metadata: Dict[str, Any] = None,
   221→    ) -> Dict[str, Any]:
   222→        """Record that a payment was received.
   223→        
   224→        Call this from:
   225→          - Razorpay webhook handler (payment.captured)
   226→          - Paddle webhook handler (payment_succeeded)
   227→          - FlexPay installment success callback
   228→        
   229→        This will:
   230→          1. Log the payment in payment_history_log
   231→          2. Update the tracking record (received_amount, outstanding_balance)
   232→          3. Reset failure counters
   233→          4. UNBLOCK the account if it was blocked
   234→        
   235→        Returns:
   236→            Dict with status and updated tracking info
   237→        """
   238→        db = self._get_db()
   239→        try:
   240→            tracking = db.query(PaymentTracking).filter(
   241→                PaymentTracking.company_id == company_id,
   242→            ).first()
   243→            
   244→            if not tracking:
   245→                raise PaymentTrackingError(f"No tracking record for company {company_id}")
   246→            
   247→            was_blocked = tracking.is_blocked
   248→            
   249→            # Log the payment in history
   250→            history_entry = PaymentHistoryLog(
   251→                company_id=company_id,
   252→                payment_tracking_id=tracking.id,
   253→                amount=amount,
   254→                currency='USD',
   255→                payment_method=payment_method,
   256→                external_payment_id=external_payment_id,
   257→                external_order_id=external_order_id,
   258→                status='success',
   259→                metadata=metadata or {},
   260→                processed_at=datetime.now(timezone.utc),
   261→            )
   262→            db.add(history_entry)
   263→            
   264→            # Update tracking record
   265→            tracking.last_payment_received_at = datetime.now(timezone.utc)
   266→            tracking.received_amount += amount
   267→            
   268→            # Reduce outstanding balance (don't go negative)
   269→            tracking.outstanding_balance = max(
   270→                tracking.outstanding_balance - amount,
   271→                Decimal("0.00")
   272→            )
   273→            
   274→            # Reset failure counters on successful payment
   275→            tracking.failed_payment_attempts = 0
   276→            tracking.last_failed_payment_at = None
   277→            tracking.next_retry_at = None
   278→            
   279→            # Update status
   280→            if tracking.outstanding_balance <= 0:
   281→                tracking.payment_status = 'current'
   282→            else:
   283→                tracking.payment_status = 'overdue'
   284→            
   285→            # UNBLOCK if was blocked (payment received!)
   286→            if was_blocked:
   287→                tracking.is_blocked = False
   288→                tracking.unblocked_at = datetime.now(timezone.utc)
   289→                tracking.block_reason = None
   290→                
   291→                logger.info(
   292→                    "account_unblocked: company=%s, payment=%s",
   293→                    company_id, amount,
   294→                )
   295→            
   296→            tracking.updated_at = datetime.now(timezone.utc)
   297→            db.commit()
   298→            
   299→            logger.info(
   300→                "payment_recorded: company=%s, amount=%s, method=%s, was_blocked=%s",
   301→                company_id, amount, payment_method, was_blocked,
   302→            )
   303→            
   304→            return {
   305→                'status': 'success',
   306→                'company_id': company_id,
   307→                'amount': float(amount),
   308→                'was_unblocked': was_blocked,
   309→                'outstanding_balance': float(tracking.outstanding_balance),
   310→                'payment_status': tracking.payment_status,
   311→            }
   312→            
   313→        except PaymentTrackingError:
   314→            raise
   315→        except Exception as exc:
   316→            db.rollback()
   317→            logger.error(
   318→                "record_payment_failed: company=%s, error=%s",
   319→                company_id, str(exc),
   320→            )
   321→            raise PaymentTrackingError(f"Failed to record payment: {exc}")
   322→        finally:
   323→            db.close()

   325→    def record_payment_failed(
   326→        self,
   327→        company_id: str,
   328→        amount: Decimal = None,
   329→        payment_method: str = None,
   330→        external_payment_id: str = None,
   331→        failure_reason: str = None,
   332→        metadata: Dict[str, Any] = None,
   333→    ) -> Dict[str, Any]:
   334→        """Record that a payment failed.
   335→        
   336→        This will:
   337→          1. Log the failed attempt in payment_history_log
   338→          2. Increment failure counter
   339→          3. Set next retry time (24 hours from now)
   340→          4. Mark as overdue if past due date
   341→        
   342→        Returns:
   343→            Dict with status and failure info
   344→        """
   345→        db = self._get_db()
   346→        try:
   347→            tracking = db.query(PaymentTracking).filter(
   348→                PaymentTracking.company_id == company_id,
   349→            ).first()
   350→            
   351→            if not tracking:
   352→                raise PaymentTrackingError(f"No tracking record for company {company_id}")
   353→            
   354→            # Log the failure in history
   355→            history_entry = PaymentHistoryLog(
   356→                company_id=company_id,
   357→                payment_tracking_id=tracking.id,
   358→                amount=amount or Decimal("0.00"),
   359→                currency='USD',
   360→                payment_method=payment_method,
   361→                external_payment_id=external_payment_id,
   362→                status='failed',
   363→                failure_reason=failure_reason,
   364→                metadata=metadata or {},
   365→                processed_at=datetime.now(timezone.utc),
   366→            )
   367→            db.add(history_entry)
   368→            
   369→            # Update tracking record
   370→            tracking.failed_payment_attempts += 1
   371→            tracking.last_failed_payment_at = datetime.now(timezone.utc)
   372→            tracking.next_retry_at = datetime.now(timezone.utc) + timedelta(hours=24)
   373→            
   374→            # Check if we're past due date → mark as overdue
   375→            if tracking.next_payment_due_at and tracking.next_payment_due_at < datetime.now(timezone.utc):
   376→                tracking.payment_status = 'overdue'
   377→            
   378→            tracking.updated_at = datetime.now(timezone.utc)
   379→            db.commit()
   380→            
   381→            logger.warning(
   382→                "payment_failed: company=%s, attempt=%s, reason=%s",
   383→                company_id, tracking.failed_payment_attempts, failure_reason,
   384→            )
   385→            
   386→            return {
   387→                'status': 'failed',
   388→                'company_id': company_id,
   389→                'failure_count': tracking.failed_payment_attempts,
   390→                'next_retry_at': tracking.next_retry_at.isoformat() if tracking.next_retry_at else None,
   391→                'is_overdue': tracking.payment_status == 'overdue',
   392→            }
   393→            
   394→        except PaymentTrackingError:
   395→            raise
   396→        except Exception as exc:
   397→            db.rollback()
   398→            logger.error(
   399→                "record_payment_failure_failed: company=%s, error=%s",
   400→                company_id, str(exc),
   401→            )
   402→            raise PaymentTrackingError(f"Failed to record payment failure: {exc}")
   403→        finally:
   404→            db.close()

   410→    def _check_and_update_block_status(self, tracking: PaymentTracking) -> bool:
   411→        """Check if a tracking record should be blocked and update if needed.
   412→        
   413→        Internal method used by other functions.
   414→        
   415→        Returns True if status changed (newly blocked).
   416→        """
   417→        if tracking.is_blocked:
   418→            return False  # Already blocked
   419→        
   420→        if tracking.payment_status != 'overdue':
   421→            return False  # Not overdue
   422→        
   423→        if not tracking.next_payment_due_at:
   424→            return False  # No due date set
   425→        
   426→        # Check if past the block threshold
   427→        block_time = tracking.next_payment_due_at + timedelta(hours=tracking.block_after_hours)
   428→        
   429→        if datetime.now(timezone.utc) >= block_time:
   430→            # Time to block!
   431→            tracking.is_blocked = True
   432→            tracking.payment_status = 'blocked'
   433→            tracking.blocked_at = datetime.now(timezone.utc)
   434→            tracking.block_reason = (
   435→                f"Auto-blocked: Payment overdue by {tracking.block_after_hours} hours. "
   436→                f"Due date was {tracking.next_payment_due_at.isoformat()}"
   437→            )
   438→            tracking.updated_at = datetime.now(timezone.utc)
   439→            
   440→            logger.warning(
   441→                "auto_blocked: company=%s, due=%s, after=%s hours",
   442→                tracking.company_id, tracking.next_payment_due_at, tracking.block_after_hours,
   443→            )
   444→            
   445→            return True
   446→        
   447→        return False

   449→    def check_and_block_overdue_accounts(self) -> Dict[str, Any]:
   450→        """Check all accounts and auto-block those that are overdue.
   451→        
   452→        Call this from a cron job (e.g., every hour).
   453→        
   454→        Returns:
   455→            Dict with count of newly blocked accounts
   456→        """
   457→        db = self._get_db()
   458→        try:
   459→            # Find all overdue accounts that aren't blocked yet
   460→            overdue_accounts = db.query(PaymentTracking).filter(
   461→                PaymentTracking.payment_status.in_(['current', 'overdue']),
   462→                PaymentTracking.is_blocked == False,
   463→                PaymentTracking.next_payment_due_at < datetime.now(timezone.utc),
   464→            ).all()
   465→            
   466→            blocked_count = 0
   467→            blocked_companies = []
   468→            
   469→            for tracking in overdue_accounts:
   470→                if self._check_and_update_block_status(tracking):
   471→                    # Log the blocking action
   472→                    history_entry = PaymentHistoryLog(
   473→                        company_id=tracking.company_id,
   474→                        payment_tracking_id=tracking.id,
   475→                        amount=Decimal("0.00"),
   476→                        status='account_blocked',
   477→                        metadata={
   478→                            'action': 'auto_block',
   479→                            'reason': 'Payment overdue',
   480→                            'due_date': tracking.next_payment_due_at.isoformat() if tracking.next_payment_due_at else None,
   481→                            'block_after_hours': tracking.block_after_hours,
   482→                        },
   483→                        created_at=datetime.now(timezone.utc),
   484→                    )
   485→                    db.add(history_entry)
   486→                    
   487→                    blocked_count += 1
   488→                    blocked_companies.append(tracking.company_id)
   489→            
   490→            if blocked_count > 0:
   491→                db.commit()
   492→                
   493→                logger.warning(
   494→                    "accounts_auto_blocked: count=%d, companies=%s",
   495→                    blocked_count, blocked_companies,
   496→                )
   497→            
   498→            return {
   499→                'checked': len(overdue_accounts),
   500→                'blocked': blocked_count,
   501→                'companies': blocked_companies,
   502→                'timestamp': datetime.now(timezone.utc).isoformat(),
   503→            }
   504→            
   505→        except Exception as exc:
   506→            db.rollback()
   507→            logger.error("check_overdue_failed: error=%s", str(exc))
   508→            return {'error': str(exc), 'checked': 0, 'blocked': 0}
   509→        finally:
   510→            db.close()

   567→    def get_payment_history(
   568→        self,
   569→        company_id: str,
   570→        limit: int = 50,
   571→        status_filter: str = None,
   572→    ) -> List[Dict[str, Any]]:
   573→        """Get payment history for a company.
   574→        
   575→        Args:
   576→            company_id: Company to get history for
   577→            limit: Max records to return
   578→            status_filter: Filter by status (success/failed/etc.)
   579→        
   580→        Returns:
   581→            List of payment history dicts
   582→        """
   583→        db = self._get_db()
   584→        try:
   585→            query = db.query(PaymentHistoryLog).filter(
   586→                PaymentHistoryLog.company_id == company_id,
   587→            )
   588→            
   589→            if status_filter:
   590→                query = query.filter(PaymentHistoryLog.status == status_filter)
   591→            
   592→            entries = query.order_by(
   593→                PaymentHistoryLog.created_at.desc()
   594→            ).limit(limit).all()
   595→            
   596→            return [
   597→                {
   598→                    'id': e.id,
   599→                    'amount': float(e.amount),
   600→                    'currency': e.currency,
   601→                    'payment_method': e.payment_method,
   602→                    'external_payment_id': e.external_payment_id,
   603→                    'status': e.status,
   604→                    'failure_reason': e.failure_reason,
   605→                    'metadata': e.metadata or {},
   606→                    'created_at': e.created_at.isoformat() if e.created_at else None,

   609→                for e in entries

   647→    def update_due_date(
   648→        self,
   649→        company_id: str,
   650→        new_due_date: datetime,
   651→        expected_amount: Decimal = None,
   652→    ) -> bool:
   653→        """Update the next payment due date.
   654→        
   655→        Call this when:
   656→          - Subscription is renewed
   657→          - Payment plan changes
   658→          - Manual adjustment needed
   659→        
   660→        Returns:
   661→            True if updated successfully
   662→        """
   663→        db = self._get_db()
   664→        try:
   665→            result = db.query(PaymentTracking).filter(
   666→                PaymentTracking.company_id == company_id,
   667→            ).update({
   668→                'next_payment_due_at': new_due_date,
   669→                'expected_amount': expected_amount if expected_amount is not None else PaymentTracking.expected_amount,
   670→                'outstanding_balance': expected_amount if expected_amount is not None else PaymentTracking.expected_amount,
   671→                'payment_status': 'current',  # Reset to current on renewal
   672→                'updated_at': datetime.now(timezone.utc),
   673→            })
   674→            db.commit()
   675→            return result > 0
   676→        except Exception as exc:
   677→            db.rollback()
   678→            logger.error("update_due_date_failed: company=%s, error=%s", company_id, str(exc))
   679→            return False
   680→        finally:
   681→            db.close()

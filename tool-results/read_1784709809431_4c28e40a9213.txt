    77→
    78→function formatRelativeDate(dateStr: string): string {
    79→  const diff = Date.now() - new Date(dateStr).getTime();
    80→  const mins = Math.floor(diff / 60000);
    81→  if (mins < 1) return 'just now';
    82→  if (mins < 60) return `${mins}m ago`;
    83→  const hrs = Math.floor(mins / 60);
    84→  if (hrs < 24) return `${hrs}h ago`;
    85→  const days = Math.floor(hrs / 24);
    86→  if (days < 7) return `${days}d ago`;
    87→  return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
    88→}
    89→
    90→function formatDate(dateStr: string): string {
    91→  return new Date(dateStr).toLocaleDateString('en-US', {
    92→    month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit',
    93→  });
    94→}

   182→// ── Customer Context Panel (shows integration data) ─────────────────
   183→
   184→function CustomerContextPanel({
   185→  ticketId,
   186→  customerEmail,
   187→}: {
   188→  ticketId: string;
   189→  customerEmail: string;
   190→}) {
   191→  const [contextData, setContextData] = useState<Record<string, unknown> | null>(null);
   192→  const [loading, setLoading] = useState(true);
   193→
   194→  useEffect(() => {
   195→    let cancelled = false;
   196→    setLoading(true);
   197→    // Fetch the ticket detail from the backend — it includes metadata_json
   198→    // which may contain CRM data, ecommerce orders, and carrier tracking
   199→    // that Node 3 fetched during pipeline processing.
   200→    fetch(`/api/v1/tickets/${ticketId}`, { credentials: 'include' })
   201→      .then((r) => r.json())
   202→      .then((data) => {
   203→        if (cancelled) return;
   204→        // The metadata_json field may contain CRM/ecommerce/carrier data
   205→        const meta = data.metadata_json || {};
   206→        setContextData(meta);
   207→      })
   208→      .catch(() => {
   209→        if (!cancelled) setContextData(null);
   210→      })
   211→      .finally(() => {
   212→        if (!cancelled) setLoading(false);
   213→      });
   214→    return () => { cancelled = true; };
   215→  }, [ticketId]);
   216→
   217→  if (loading) {
   218→    return (
   219→      <div className="px-4 py-3 border-b border-white/[0.06]">
   220→        <p className="text-[10px] text-zinc-600">Loading customer context...</p>
   221→      </div>
   222→    );
   223→  }
   224→
   225→  if (!contextData || Object.keys(contextData).length === 0) {
   226→    return null; // No integration data — don't show the panel
   227→  }
   228→
   229→  return (
   230→    <div className="px-4 py-3 border-b border-white/[0.06] bg-white/[0.01]">
   231→      <p className="text-[10px] font-semibold text-zinc-400 uppercase tracking-wide mb-2">
   232→        Customer Context
   233→      </p>
   234→      <div className="space-y-1.5 text-[11px]">
   235→        {Object.entries(contextData).slice(0, 8).map(([key, value]) => (
   236→          <div key={key} className="flex justify-between gap-2">
   237→            <span className="text-zinc-500 capitalize">{key.replace(/_/g, ' ')}</span>
   238→            <span className="text-zinc-300 text-right truncate max-w-[200px]">
   239→              {typeof value === 'object'
   240→                ? JSON.stringify(value).slice(0, 80) + '...'
   241→                : String(value).slice(0, 80)}
   242→            </span>
   243→          </div>
   244→        ))}
   245→      </div>
   246→    </div>
   247→  );
   248→}

   261→  const resumeWithGuidance = useTicketStore((s) => s.resumeWithGuidance);
   262→  const updateTicketStatus = useTicketStore((s) => s.updateTicketStatus);
   263→  const updatePriority = useTicketStore((s) => s.updatePriority);
   264→  const addMessage = useTicketStore((s) => s.addMessage);
   265→
   266→  const [replyText, setReplyText] = useState('');
   267→  const [statusDropdownOpen, setStatusDropdownOpen] = useState(false);

   269→  const messagesEndRef = useRef<HTMLDivElement>(null);
   270→  const statusRef = useRef<HTMLDivElement>(null);

   275→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });

   281→      if (statusRef.current && !statusRef.current.contains(e.target as Node)) setStatusDropdownOpen(false);
   282→      if (priorityRef.current && !priorityRef.current.contains(e.target as Node)) setPriorityDropdownOpen(false);
   283→    }
   284→    document.addEventListener('mousedown', handleClick);
   285→    return () => document.removeEventListener('mousedown', handleClick);
   286→  }, []);
   287→
   288→  const handleResolve = () => {
   289→    resolveTicket(ticket.id);
   290→  };
   291→
   292→  const handleEscalate = () => {
   293→    escalateToHuman(ticket.id);
   294→  };
   295→
   296→  const [guidanceText, setGuidanceText] = useState('');

   299→  const handleResumeWithGuidance = async () => {

   305→    const ok = await resumeWithGuidance(ticket.id, guidanceText.trim());

   307→      toast.success('Pipeline resumed with guidance. AI is processing...');
   308→      setGuidanceText('');
   309→    } else {
   310→      toast.error('Failed to resume pipeline. Please try again.');
   311→    }
   312→    setResuming(false);
   313→  };
   314→
   315→  const handleSendReply = () => {
   316→    if (!replyText.trim()) return;
   317→    addMessage(ticket.id, {
   318→      sender: 'human_agent',
   319→      sender_name: 'Human Agent',
   320→      content: replyText.trim(),
   321→      variant: ticket.assigned_variant ?? undefined,
   322→    });
   323→    setReplyText('');
   324→  };
   325→
   326→  const handleStatusChange = (status: TicketStatus) => {
   327→    updateTicketStatus(ticket.id, status);
   328→    setStatusDropdownOpen(false);
   329→  };
   330→
   331→  const handlePriorityChange = (priority: TicketPriority) => {
   332→    updatePriority(ticket.id, priority);
   333→    setPriorityDropdownOpen(false);
   334→  };
   335→
   336→  return (
   337→    <motion.div
   338→      initial={{ opacity: 0, x: 20 }}
   339→      animate={{ opacity: 1, x: 0 }}
   340→      exit={{ opacity: 0, x: 20 }}
   341→      transition={{ duration: 0.25 }}
   342→      className="w-full lg:w-[440px] xl:w-[480px] shrink-0 bg-[#1A1A1A] border border-white/[0.06] rounded-xl overflow-hidden flex flex-col max-h-[calc(100vh-180px)]"
   343→    >
   344→      {/* Header */}
   345→      <div className="p-4 border-b border-white/[0.06] flex items-start justify-between gap-3">
   346→        <div className="min-w-0">
   347→          <div className="flex items-center gap-2 mb-1">
   348→            <span className="text-xs font-mono text-orange-400">{ticket.ticket_number}</span>
   349→            <StatusBadge status={ticket.status} />
   350→            <PriorityBadge priority={ticket.priority} />
   351→          </div>
   352→          <h3 className="text-sm font-semibold text-white truncate">{ticket.subject}</h3>
   353→        </div>
   354→        <button
   355→          onClick={onClose}
   356→          className="w-7 h-7 flex items-center justify-center rounded-lg text-zinc-400 hover:text-white hover:bg-white/5 transition-colors shrink-0"

   900→          exit={{ scale: 0.96, opacity: 0, y: 8 }}
   901→          transition={{ duration: 0.18 }}
   902→          className="w-full max-w-xl rounded-2xl bg-[#161616] border border-white/[0.08] shadow-2xl"
   903→          onClick={(e) => e.stopPropagation()}
   904→        >
   905→          {/* Header */}
   906→          <div className="flex items-center justify-between px-6 py-4 border-b border-white/[0.06]">

   908→              <h2 className="text-lg font-semibold text-white">Create Ticket</h2>

   910→                The AI pipeline will resolve it automatically

   331→   830→   830→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
   332→   831→   831→        """Analyze quality_scores for drift using day-over-day comparison."""
   333→   832→   832→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
   334→   833→   833→        if len(tenant_scores) < 3:
   335→   834→   834→            return {
   336→   835→   835→                "tenant_id": tenant_id, "drift_detected": False,
   337→   836→   836→                "drift_severity": "none", "accuracy_7d": None,
   338→   837→   837→                "accuracy_yesterday": None, "accuracy_today": None,
   339→   838→   838→                "trend_direction": "stable", "trigger_reason": "insufficient_data",
   340→   839→   839→                "total_scores": len(tenant_scores),
   341→   840→   840→            }
   342→   841→   841→        # Group by date (ISO date portion)
   343→   842→   842→        by_date: Dict[str, List[float]] = {}
   344→   843→   843→        for s in tenant_scores:
   345→   844→   844→            date_str = s["created_at"][:10]  # YYYY-MM-DD
   346→   845→   845→            by_date.setdefault(date_str, []).append(s["overall_score"])
   347→   846→   846→
   348→   847→   847→        dates_sorted = sorted(by_date.keys())
   349→   848→   848→
   350→   849→   849→        def _avg(scores: List[float]) -> float:
   351→   850→   850→            return round(sum(scores) / len(scores), 4) if scores else 0.0
   352→   851→   851→
   353→   852→   852→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
   354→   853→   853→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
   355→   854→   854→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
   356→   855→   855→
   357→   856→   856→        # Drift detection: >5% drop for 3+ days
   358→   857→   857→        drift_detected = False
   359→   858→   858→        drift_severity = "none"
   360→   859→   859→        trigger_reason = None

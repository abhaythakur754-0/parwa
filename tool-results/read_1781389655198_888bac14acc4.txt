   120→// ── Savings Calculation ──────────────────────────────────────────────────
   121→
   122→function estimateSavings(ticketVolume: number, monthlyCost: number): {
   123→  agentsReplaced: number;
   124→  humanCost: number;
   125→  savings: number;
   126→  savingsPercent: number;
   127→} {
   128→  const agentsReplaced = Math.max(1, Math.round(ticketVolume / TICKETS_PER_AGENT));
   129→  const humanCost = agentsReplaced * AGENT_COST_MONTHLY;
   130→  const savings = Math.max(0, humanCost - monthlyCost);
   131→  const savingsPercent = humanCost > 0 ? Math.round((savings / humanCost) * 100) : 0;
   132→  return { agentsReplaced, humanCost, savings, savingsPercent };
   133→}

   242→// ── Component ──────────────────────────────────────────────────────────
   243→
   244→export function CostBreakdownStep({ variant, onComplete }: CostBreakdownStepProps) {
   245→  const initialTier = ONBOARDING_TO_TIER[variant] || 'growth';
   246→  const [activeVariants, setActiveVariants] = useState<VariantTier[]>([initialTier]);
   247→  const [variantQuantities, setVariantQuantities] = useState<Record<VariantTier, number>>({ starter: 1, growth: 1, high: 1 });
   248→  const [addOns, setAddOns] = useState<{ voice: boolean; customApi: boolean }>({ voice: false, customApi: false });
   249→  const [isSubmitting, setIsSubmitting] = useState(false);
   250→  const [paddleStatus, setPaddleStatus] = useState<'unknown' | 'ready' | 'unavailable'>('unknown');
   251→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
   252→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
   253→
   254→  // Restore add-ons and multi-variant selections from localStorage
   255→  useEffect(() => {
   256→    try {
   257→      const stored = localStorage.getItem('parwa_pricing_context');
   258→      if (stored) {
   259→        const ctx = JSON.parse(stored) as PricingContext & {
   260→          selectedVariants?: string[];
   261→          variantQuantities?: Record<string, number>;
   262→        };
   263→        if (ctx.addOns) {
   264→          setAddOns(ctx.addOns);
   265→        }
   266→        // Restore multiple variants if selected on Pricing page
   267→        if (ctx.selectedVariants && Array.isArray(ctx.selectedVariants) && ctx.selectedVariants.length > 0) {
   268→          const tiers: VariantTier[] = ctx.selectedVariants
   269→            .map((v: string) => {
   270→              const normalized = normalizeTier(v);
   271→              return ['starter', 'growth', 'high'].includes(normalized) ? normalized as VariantTier : null;
   272→            })
   273→            .filter(Boolean) as VariantTier[];
   274→          if (tiers.length > 0) {
   275→            setActiveVariants(tiers);
   276→          }
   277→        }
   278→        // Restore variant quantities
   279→        if (ctx.variantQuantities && typeof ctx.variantQuantities === 'object') {
   280→          const qtyMap: Record<string, number> = ctx.variantQuantities;
   281→          const newQuantities = { ...variantQuantities };
   282→          for (const [variantKey, qty] of Object.entries(qtyMap)) {
   283→            const tier = normalizeTier(variantKey);
   284→            if (['starter', 'growth', 'high'].includes(tier)) {
   285→              newQuantities[tier as VariantTier] = Math.max(1, Math.min(qty as number, 10));
   286→            }
   287→          }
   288→          setVariantQuantities(newQuantities);
   289→        }
   290→      }
   291→    } catch {
   292→      // ignore
   293→    }
   294→  }, []);
   295→
   296→  // Check if Paddle.js is available
   297→  useEffect(() => {
   298→    getPaddleInstance().then((paddle) => {
   299→      setPaddleStatus(paddle ? 'ready' : 'unavailable');
   300→    }).catch(() => {
   301→      setPaddleStatus('unavailable');
   302→    });
   303→  }, []);
   304→
   305→  // ── Calculations ──────────────────────────────────────────────────
   306→
   307→  const totalTicketLimit = useMemo(
   308→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_LIMITS[tier].monthlyTickets * (variantQuantities[tier] || 1)), 0),
   309→    [activeVariants, variantQuantities]
   310→  );
   311→
   312→  const baseSubscription = useMemo(
   313→    () => activeVariants.reduce((sum, tier) => sum + (VARIANT_PRICES[tier] * (variantQuantities[tier] || 1)), 0),
   314→    [activeVariants, variantQuantities]
   315→  );
   316→
   317→  const addOnTotal = useMemo(() => {
   318→    let total = 0;
   319→    if (addOns.voice) {
   320→      const voiceAddOn = ADD_ONS.find((a) => a.key === 'voice')!;
   321→      if (!activeVariants.some((t) => voiceAddOn.includedIn.includes(t))) {
   322→        total += voiceAddOn.price;
   323→      }
   324→    }
   325→    if (addOns.customApi) {
   326→      const customApiAddOn = ADD_ONS.find((a) => a.key === 'customApi')!;
   327→      if (!activeVariants.some((t) => customApiAddOn.includedIn.includes(t))) {
   328→        total += customApiAddOn.price;
   329→      }
   330→    }
   331→    return total;
   332→  }, [addOns, activeVariants]);
   333→
   334→  const totalMonthly = baseSubscription + addOnTotal;
   335→
   336→  const savings = estimateSavings(totalTicketLimit, totalMonthly);
   337→
   338→  const toggleVariant = (tier: VariantTier) => {
   339→    setActiveVariants((prev) => {
   340→      if (prev.includes(tier)) {
   341→        if (prev.length <= 1) {
   342→          toast.error('You must have at least one active variant');
   343→          return prev;
   344→        }
   345→        return prev.filter((t) => t !== tier);
   346→      }
   347→      return [...prev, tier];
   348→    });
   349→  };
   350→
   351→  const updateQuantity = (tier: VariantTier, qty: number) => {
   352→    const newQty = Math.max(1, Math.min(qty, 10));
   353→    setVariantQuantities((prev) => ({ ...prev, [tier]: newQty }));
   354→  };
   355→
   356→  const toggleAddOn = (key: 'voice' | 'customApi') => {
   357→    setAddOns((prev) => ({ ...prev, [key]: !prev[key] }));
   358→  };
   359→
   360→  // ── Payment success handler — ONLY called after Paddle confirms ──
   361→  const handlePaymentSuccess = () => {
   362→    setPaymentConfirmed(true);
   363→    localStorage.removeItem('parwa_payment_pending');
   364→    toast.success('Payment successful! Welcome to PARWA!');
   365→    // Only trigger onboarding completion AFTER Paddle confirms
   366→    onComplete();
   367→  };
   368→
   369→  const handleProceed = async () => {
   370→    setIsSubmitting(true);
   371→    setCheckoutError(null);
   372→
   373→    try {
   374→      // Update pricing context in localStorage
   375→      const primaryOnboardingVariant = TIER_TO_ONBOARDING[activeVariants[0]] || 'parwa';
   376→      const context: PricingContext = {
   377→        industry: 'other',
   378→        variant: primaryOnboardingVariant,
   379→        addOns,
   380→        totalMonthly,
   381→        timestamp: new Date().toISOString(),
   382→      };
   383→
   384→      // Preserve industry from existing context
   385→      try {
   386→        const stored = localStorage.getItem('parwa_pricing_context');
   387→        if (stored) {
   388→          const existing = JSON.parse(stored) as PricingContext;
   389→          context.industry = existing.industry;
   390→        }
   391→      } catch {
   392→        // ignore
   393→      }
   394→
   395→      localStorage.setItem('parwa_pricing_context', JSON.stringify(context));
   396→
   397→      // ── Paddle Checkout Flow ──────────────────────────────────────
   398→      const checkoutItems: Array<{ priceId: string; quantity: number }> = [];
   399→
   400→      // Add each active variant as a Paddle line item (with quantity)
   401→      for (const tier of activeVariants) {
   402→        const onboardingVariant = TIER_TO_ONBOARDING[tier];
   403→        const variantPriceId = VARIANT_PRICE_IDS[onboardingVariant];
   404→        const qty = variantQuantities[tier] || 1;
   405→        if (variantPriceId) {
   406→          checkoutItems.push({ priceId: variantPriceId, quantity: qty });
   407→        }
   408→      }
   409→
   410→      // Voice add-on is included in growth/high — NOT purchasable separately
   411→      // (no Paddle price ID exists for standalone add-on purchases)
   412→
   413→      // Custom API add-on is included in growth/high — NOT purchasable separately
   414→      // (no Paddle price ID exists for standalone add-on purchases)
   415→
   416→      const customData = {
   417→        source: 'parwa_onboarding',
   418→        variant: primaryOnboardingVariant,
   419→        activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   420→        variantQuantities,
   421→        addOns,
   422→        industry: context.industry,
   423→        totalMonthly,
   424→      };
   425→
   426→      // ── Step 1: Try server-side checkout (backend creates Paddle transaction) ──
   427→      let checkoutUrl: string | null = null;
   428→
   429→      try {
   430→        const res = await fetch('/api/onboarding/checkout', {
   431→          method: 'POST',
   432→          headers: { 'Content-Type': 'application/json' },
   433→          body: JSON.stringify({
   434→            variant: primaryOnboardingVariant,
   435→            activeVariants: activeVariants.map((t) => TIER_TO_ONBOARDING[t]),
   436→            variantQuantities,
   437→            addOns,
   438→            totalMonthly,
   439→            industry: context.industry,
   440→          }),
   441→        });
   442→
   443→        if (res.ok) {
   444→          const checkoutData = await res.json();
   445→          checkoutUrl = checkoutData.checkout_url || null;
   446→        }
   447→      } catch {
   448→        // API unavailable — continue with client-side Paddle checkout
   449→      }
   450→
   451→      // If we got a checkout_url from backend, redirect to it
   452→      if (checkoutUrl) {
   453→        localStorage.setItem('parwa_payment_pending', JSON.stringify({
   454→          activeVariants, addOns, totalMonthly, variantQuantities,
   455→          pendingAt: new Date().toISOString(),
   456→        }));
   457→        window.location.href = checkoutUrl;
   458→        return;
   459→      }
   460→
   461→      // ── Step 2: Client-side Paddle checkout (items-based overlay) ──
   462→      const paddle = await getPaddleInstance();
   463→
   464→      if (paddle && checkoutItems.length > 0) {
   465→        setPaddleStatus('ready');
   466→        const opened = await openCheckoutWithItems(
   467→          checkoutItems,
   468→          customData,
   469→          // onPaymentSuccess — ONLY triggers after Paddle confirms the transaction
   470→          handlePaymentSuccess,
   471→          // onCheckoutClosed — user closed without paying
   472→          () => {
   473→            toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   474→            setIsSubmitting(false);
   475→          },
   476→        );
   477→
   478→        if (opened) {
   479→          return; // Paddle overlay is showing — wait for user to complete or close
   480→        }
   481→      }
   482→
   483→      // ── Step 3: Paddle unavailable — try re-initializing ──
   484→      try {
   485→        const retryPaddle = await getPaddleInstance();
   486→        if (retryPaddle && checkoutItems.length > 0) {
   487→          setPaddleStatus('ready');
   488→          const opened = await openCheckoutWithItems(
   489→            checkoutItems,
   490→            customData,
   491→            handlePaymentSuccess,
   492→            () => {
   493→              toast('Checkout closed — payment is required to activate your plan.', { icon: '⚠️' });
   494→              setIsSubmitting(false);
   495→            },
   496→          );
   497→          if (opened) return;
   498→        }
   499→      } catch {
   500→        // Paddle truly unavailable
   501→      }
   502→
   503→      // ── Step 4: Paddle unavailable — block with error ──
   504→      setPaddleStatus('unavailable');
   505→      localStorage.setItem('parwa_payment_pending', JSON.stringify({
   506→        activeVariants, addOns, totalMonthly, variantQuantities,
   507→        pendingAt: new Date().toISOString(),
   508→      }));
   509→      setCheckoutError(
   510→        'Payment gateway is currently unavailable. Your plan configuration has been saved. ' +
   511→        'Please refresh the page and try again, or contact support@parwa.buzz to complete your subscription.'
   512→      );
   513→      toast.error('Payment gateway unavailable — please try again or contact support.');
   514→    } catch (err) {
   515→      console.error('[cost-breakdown] Error:', err);
   516→      setCheckoutError('Something went wrong. Please try again.');
   517→      toast.error('Something went wrong. Please try again.');
   518→    } finally {
   519→      setIsSubmitting(false);
   520→    }
   521→  };
   522→
   523→  return (
   524→    <div className="space-y-6">
   525→      {/* Header */}
   526→      <div className="text-center space-y-2">
   527→        <div className="w-14 h-14 mx-auto rounded-2xl bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center">
   528→          <Receipt className="w-7 h-7 text-emerald-400" />
   529→        </div>
   530→        <h2 className="text-2xl font-bold text-white">Review & Pay</h2>
   531→        <p className="text-orange-200/40 text-sm">
   532→          Configure your plan and complete payment to activate your AI agents.
   533→        </p>
   534→      </div>
   535→
   536→      {/* ── Paddle Status Indicator ─────────────────────────────────── */}
   537→      {paddleStatus === 'unavailable' && (
   538→        <div className="rounded-xl border border-amber-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(245,158,11,0.05)' }}>
   539→          <AlertCircle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
   540→          <div>
   541→            <p className="text-sm font-medium text-amber-300">Payment checkout unavailable</p>
   542→            <p className="text-[10px] text-orange-200/30 mt-1">
   543→              Paddle payment gateway is not configured. Your configuration will be saved and you can complete payment later.
   544→            </p>
   545→          </div>
   546→        </div>
   547→      )}
   548→      {paddleStatus === 'ready' && (
   549→        <div className="rounded-xl border border-emerald-500/20 p-3 flex items-center gap-2" style={{ background: 'rgba(16,185,129,0.04)' }}>
   550→          <Shield className="w-4 h-4 text-emerald-400" />
   551→          <p className="text-xs text-emerald-400">Secure checkout powered by Paddle</p>
   552→          <ExternalLink className="w-3 h-3 text-emerald-400/60 ml-auto" />
   553→        </div>
   554→      )}
   555→
   556→      {/* ── 1. Variant Mixer ──────────────────────────────────────────── */}
   557→      <div className="space-y-3">
   558→        <div className="flex items-center gap-2">
   559→          <Brain className="w-4 h-4 text-orange-400" />
   560→          <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   561→            Active Variants
   562→          </label>
   563→          <span className="text-[10px] text-orange-200/25">
   564→            Each variant adds its own ticket allocation and AI pipeline
   565→          </span>
   566→        </div>
   567→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   568→          {(['starter', 'growth', 'high'] as VariantTier[]).map((tier) => (
   569→            <VariantMixerCard
   570→              key={tier}
   571→              tier={tier}
   572→              isActive={activeVariants.includes(tier)}
   573→              quantity={variantQuantities[tier] || 1}
   574→              onToggle={() => toggleVariant(tier)}
   575→              onQuantityChange={(qty) => updateQuantity(tier, qty)}
   576→            />
   577→          ))}
   578→        </div>
   579→      </div>
   580→
   581→      {/* ── 2. Add-Ons ───────────────────────────────────────────────── */}
   582→      <div className="space-y-3">
   583→        <label className="text-xs text-orange-200/40 uppercase tracking-wider font-medium">
   584→          Add-On Features
   585→        </label>
   586→        {ADD_ONS.map((addOn) => {
   587→          const Icon = addOn.icon;
   588→          const isIncluded = activeVariants.some((t) => addOn.includedIn.includes(t));
   589→          // Add-ons are NOT purchasable separately — only included with growth/high plans
   590→          const isLocked = !isIncluded;
   591→
   592→          return (
   593→            <div
   594→              key={addOn.key}
   595→              className={cn(
   596→                'w-full text-left p-4 rounded-xl border transition-all duration-200 flex items-start gap-4',
   597→                isIncluded
   598→                  ? 'border-emerald-500/20 bg-emerald-500/[0.03]'
   599→                  : 'border-white/[0.06]'
   600→              )}
   601→              style={!isIncluded ? { background: 'rgba(255,255,255,0.03)' } : undefined}
   602→            >
   603→              <div className={cn(
   604→                'w-10 h-10 rounded-lg flex items-center justify-center shrink-0',
   605→                isIncluded ? 'bg-emerald-500/10' : 'bg-white/[0.04]'
   606→              )}>
   607→                <Icon className={cn(
   608→                  'w-5 h-5',
   609→                  isIncluded ? 'text-emerald-400' : 'text-zinc-500'
   610→                )} />
   611→              </div>
   612→              <div className="flex-1 min-w-0">
   613→                <div className="flex items-center gap-2">
   614→                  <p className={cn(
   615→                    'text-sm font-medium',
   616→                    isIncluded ? 'text-emerald-400' : 'text-white'
   617→                  )}>
   618→                    {addOn.name}
   619→                  </p>
   620→                  {isIncluded && (
   621→                    <span className="text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-full bg-emerald-500/10 text-emerald-400">
   622→                      Included
   623→                    </span>
   624→                  )}
   625→                </div>
   626→                <p className="text-[10px] text-orange-200/30 mt-0.5">
   627→                  {isIncluded ? addOn.description : `Included with PARWA and PARWA High plans`}
   628→                </p>
   629→              </div>
   630→              <div className="flex flex-col items-end shrink-0 gap-1">
   631→                {isIncluded ? (
   632→                  <div className="w-8 h-5 rounded-full flex items-center px-0.5 bg-gradient-to-r from-orange-500 to-amber-400">
   633→                    <div className="w-4 h-4 rounded-full bg-white shadow-sm translate-x-3" />
   634→                  </div>
   635→                ) : (
   636→                  <span className="text-[10px] text-orange-200/25 italic">Upgrade required</span>
   637→                )}
   638→              </div>
   639→            </div>
   640→          );
   641→        })}
   642→      </div>
   643→
   644→      {/* ── 3. Cost Summary ────────────────────────────────────────────── */}
   645→      <div
   646→        className="rounded-xl border border-white/[0.08] p-5 space-y-3"
   647→        style={{ background: 'rgba(255,255,255,0.03)' }}
   648→      >
   649→        {/* Per-variant breakdown */}
   650→        {activeVariants.map((tier) => {
   651→          const qty = variantQuantities[tier] || 1;
   652→          const price = VARIANT_PRICES[tier] * qty;
   653→
   654→          return (
   655→            <div key={tier} className="flex items-center justify-between">
   656→              <span className="text-sm text-orange-200/50 flex items-center gap-1.5">
   657→                <div className={cn(
   658→                  'w-2 h-2 rounded-full',
   659→                  tier === 'starter' ? 'bg-emerald-400' : tier === 'growth' ? 'bg-orange-400' : 'bg-purple-400'
   660→                )} />
   661→                {VARIANT_DISPLAY[tier].name}
   662→                {qty > 1 && <span className="text-orange-200/30"> × {qty}</span>}
   663→              </span>
   664→              <span className="text-sm text-white">
   665→                ${price.toLocaleString()}/mo
   666→              </span>
   667→            </div>
   668→          );
   669→        })}
   670→
   671→        {/* Add-ons — shown as included, not separately priced */}
   672→
   673→        {/* Integrations = $0 */}
   674→        <div className="flex items-center justify-between">
   675→          <span className="text-sm text-orange-200/50">Integrations</span>
   676→          <span className="text-sm text-emerald-400">$0 (free)</span>
   677→        </div>
   678→
   679→        <div className="border-t border-white/[0.06] pt-3 flex items-center justify-between">
   680→          <span className="text-sm font-semibold text-white">Total Monthly</span>
   681→          <span className="text-lg font-bold text-orange-400">
   682→            ${totalMonthly.toLocaleString()}/mo
   683→          </span>
   684→        </div>
   685→
   686→        {/* Overage rate info */}
   687→        <div className="flex items-center justify-between">
   688→          <span className="text-[10px] text-orange-200/25">Overage rate (beyond limits)</span>
   689→          <span className="text-[10px] text-orange-200/25">${OVERAGE_PRICE_PER_TICKET}/ticket</span>
   690→        </div>
   691→      </div>
   692→
   693→      {/* ── 4. Savings Comparison ──────────────────────────────────────── */}
   694→      <div
   695→        className="rounded-xl border border-emerald-500/20 p-4"
   696→        style={{ background: 'rgba(16,185,129,0.04)' }}
   697→      >
   698→        <div className="flex items-start gap-3">
   699→          <div className="w-8 h-8 rounded-lg bg-emerald-500/10 flex items-center justify-center shrink-0">
   700→            <TrendingDown className="w-4 h-4 text-emerald-400" />
   701→          </div>
   702→          <div className="flex-1">
   703→            <p className="text-sm font-semibold text-emerald-400">
   704→              Save {savings.savingsPercent}% vs human agents
   705→            </p>
   706→            <p className="text-[10px] text-orange-200/30 mt-1">
   707→              PARWA handles {totalTicketLimit.toLocaleString()} tickets/mo — equivalent to{' '}
   708→              {savings.agentsReplaced} full-time support agent{savings.agentsReplaced > 1 ? 's' : ''} at ~${AGENT_COST_MONTHLY.toLocaleString()}/mo each.
   709→              That&apos;s ${savings.humanCost.toLocaleString()}/mo in human costs vs ${totalMonthly.toLocaleString()}/mo with PARWA.
   710→            </p>
   711→          </div>
   712→        </div>
   713→      </div>
   714→
   715→      {/* ── No Hidden Fees ─────────────────────────────────────────────── */}
   716→      <div className="flex items-center justify-center gap-2 text-[10px] text-orange-200/25">
   717→        <Shield className="w-3 h-3" />
   718→        <span>No hidden fees. Need more? Add another variant.</span>
   719→      </div>
   720→
   721→      {/* ── Error Display ────────────────────────────────────────────── */}
   722→      {checkoutError && (
   723→        <div className="rounded-xl border border-red-500/20 p-4 flex items-start gap-3" style={{ background: 'rgba(239,68,68,0.05)' }}>
   724→          <AlertCircle className="w-5 h-5 text-red-400 shrink-0 mt-0.5" />
   725→          <p className="text-sm text-red-300">{checkoutError}</p>
   726→        </div>
   727→      )}
   728→
   729→      {/* ── Payment Confirmation Guard ──────────────────────────────── */}
   730→      {paymentConfirmed && (
   731→        <div className="rounded-xl border border-emerald-500/30 p-4 flex items-start gap-3" style={{ background: 'rgba(16,185,129,0.08)' }}>
   732→          <Shield className="w-5 h-5 text-emerald-400 shrink-0 mt-0.5" />
   733→          <div>
   734→            <p className="text-sm font-medium text-emerald-300">Payment confirmed by Paddle</p>
   735→            <p className="text-[10px] text-orange-200/30 mt-1">
   736→              Your subscription is now active. Redirecting to your dashboard...
   737→            </p>
   738→          </div>
   739→        </div>
   740→      )}
   741→
   742→      {/* ── Proceed Button ────────────────────────────────────────────── */}
   743→      <div className="flex justify-end">
   744→        <button
   745→          onClick={handleProceed}
   746→          disabled={isSubmitting || paymentConfirmed}
   747→          className={cn(
   748→            'px-8 py-3 font-semibold rounded-xl transition-all duration-300 shadow-lg text-sm flex items-center gap-2 disabled:cursor-not-allowed',
   749→            'bg-gradient-to-r from-orange-500 to-amber-400 hover:from-orange-400 hover:to-amber-300 text-[#1A1A1A] shadow-orange-500/25',
   750→            (isSubmitting || paymentConfirmed) && 'opacity-60'
   751→          )}
   752→        >
   753→          {isSubmitting ? (
   754→            <>
   755→              <Loader2 className="w-4 h-4 animate-spin" />
   756→              Opening Checkout...
   757→            </>
   758→          ) : paymentConfirmed ? (
   759→            <>
   760→              <Shield className="w-4 h-4" />
   761→              Activated — Redirecting...
   762→            </>
   763→          ) : (
   764→            <>
   765→              Pay ${totalMonthly.toLocaleString()}/mo & Activate
   766→              <ArrowRight className="w-4 h-4" />
   767→            </>
   768→          )}
   769→        </button>
   770→      </div>
   771→    </div>
   772→  );
   773→}

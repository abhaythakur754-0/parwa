   750→   750→    # ── Batch Queue ──────────────────────────────────────────
   751→   751→
   752→   752→    async def add_to_batch(
   753→   753→        self,
   754→   754→        tenant_id: str,
   755→   755→        batch_key: str,
   756→   756→        ticket_id: str,
   757→   757→        confidence: float,
   758→   758→    ) -> Optional[Dict[str, Any]]:
   759→   759→        BATCH_WINDOW_S = 300
   760→   760→        now = time.time()
   761→   761→
   762→   762→        if tenant_id not in self._batch_metas:
   763→   763→            self._batch_metas[tenant_id] = {}
   764→   764→
   765→   765→        # Find existing batch with same key within window
   766→   766→        existing_key = None
   767→   767→        for bk, meta in self._batch_metas[tenant_id].items():
   768→   768→            if bk == batch_key and (now - meta["batch_start"]) < BATCH_WINDOW_S:
   769→   769→                existing_key = bk
   770→   770→                break
   771→   771→
   772→   772→        if existing_key:
   773→   773→            meta = self._batch_metas[tenant_id][existing_key]
   774→   774→            meta["ticket_ids"].append(ticket_id)
   775→   775→            meta["signal_count"] = len(meta["ticket_ids"])
   776→   776→            meta["confidence_min"] = min(meta.get("confidence_min", 1.0), confidence)
   777→   777→            meta["confidence_max"] = max(meta.get("confidence_max", 0.0), confidence)
   778→   778→            # Don't flush yet — still within window
   779→   779→            return None
   780→   780→        else:
   781→   781→            meta = {
   782→   782→                "batch_key": batch_key,
   783→   783→                "tenant_id": tenant_id,
   784→   784→                "batch_start": now,
   785→   785→                "ticket_ids": [ticket_id],
   786→   786→                "signal_count": 1,
   787→   787→                "confidence_min": confidence,
   788→   788→                "confidence_max": confidence,
   789→   789→                "status": "pending",
   790→   790→            }
   791→   791→            self._batch_metas[tenant_id][batch_key] = meta
   792→   792→            return None

  1052→  1052→    async def check_quality_drift(self, tenant_id: str) -> Dict[str, Any]:
  1053→  1053→        """Analyze quality_scores for drift using day-over-day comparison."""
  1054→  1054→        tenant_scores = [s for s in self._quality_scores if s["tenant_id"] == tenant_id]
  1055→  1055→        if len(tenant_scores) < 3:
  1056→  1056→            return {
  1057→  1057→                "tenant_id": tenant_id, "drift_detected": False,

  1059→  1059→                "accuracy_yesterday": None, "accuracy_today": None,
  1060→  1060→                "trend_direction": "stable", "trigger_reason": "insufficient_data",

  1062→  1062→            }
  1063→  1063→        # Group by date (ISO date portion)
  1064→  1064→        by_date: Dict[str, List[float]] = {}
  1065→  1065→        for s in tenant_scores:
  1066→  1066→            date_str = s["created_at"][:10]  # YYYY-MM-DD
  1067→  1067→            by_date.setdefault(date_str, []).append(s["overall_score"])
  1068→  1068→
  1069→  1069→        dates_sorted = sorted(by_date.keys())
  1070→  1070→
  1071→  1071→        def _avg(scores: List[float]) -> float:
  1072→  1072→            return round(sum(scores) / len(scores), 4) if scores else 0.0
  1073→  1073→
  1074→  1074→        accuracy_7d = _avg([s for d in dates_sorted[-7:] for s in by_date[d]]) if len(dates_sorted) >= 1 else None
  1075→  1075→        accuracy_yesterday = _avg(by_date[dates_sorted[-2]]) if len(dates_sorted) >= 2 else None
  1076→  1076→        accuracy_today = _avg(by_date[dates_sorted[-1]]) if dates_sorted else None
  1077→  1077→
  1078→  1078→        # Drift detection: >5% drop for 3+ days
  1079→  1079→        drift_detected = False
  1080→  1080→        drift_severity = "none"
  1081→  1081→        trigger_reason = None
  1082→  1082→        trend_direction = "stable"

  1085→  1085→            recent_3 = dates_sorted[-3:]
  1086→  1086→            daily_avgs = [_avg(by_date[d]) for d in recent_3]
  1087→  1087→            # Check if declining 3+ consecutive days
  1088→  1088→            declining = all(daily_avgs[i] > daily_avgs[i + 1] for i in range(len(daily_avgs) - 1))
  1089→  1089→            drop_7d = accuracy_7d - accuracy_today
  1090→  1090→
  1091→  1091→            if declining and drop_7d > 0.05:
  1092→  1092→                drift_detected = True
  1093→  1093→                trend_direction = "declining"
  1094→  1094→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_over_3_days"
  1095→  1095→                drift_severity = "critical" if drop_7d > 0.10 else ("warning" if drop_7d > 0.05 else "none")
  1096→  1096→            elif drop_7d > 0.03:
  1097→  1097→                drift_detected = True
  1098→  1098→                trend_direction = "slight_decline"
  1099→  1099→                trigger_reason = f"accuracy_dropped_{drop_7d:.1%}_from_7d_avg"
  1100→  1100→                drift_severity = "warning"
  1101→  1101→            elif accuracy_today and accuracy_7d and accuracy_today > accuracy_7d + 0.03:
  1102→  1102→                trend_direction = "improving"
  1103→  1103→            else:
  1104→  1104→                trend_direction = "stable"
  1105→  1105→
  1106→  1106→        # Check for repeated errors (same low score pattern)
  1107→  1107→        if not drift_detected and tenant_scores:
  1108→  1108→            low_scores = [s for s in tenant_scores if s["overall_score"] < 0.7]
  1109→  1109→            if len(low_scores) >= 3:
  1110→  1110→                # Check if same resolution_path failing
  1111→  1111→                paths: Dict[str, int] = {}
  1112→  1112→                for s in low_scores:
  1113→  1113→                    p = s.get("resolution_path", "unknown")
  1114→  1114→                    paths[p] = paths.get(p, 0) + 1
  1115→  1115→                worst_path = max(paths.items(), key=lambda x: x[1])
  1116→  1116→                if worst_path[1] >= 3:
  1117→  1117→                    drift_detected = True
  1118→  1118→                    drift_severity = "warning"
  1119→  1119→                    trend_direction = "declining"
  1120→  1120→                    trigger_reason = f"path_{worst_path[0]}_failed_{worst_path[1]}_times"
  1121→  1121→
  1122→  1122→        return {
  1123→  1123→            "tenant_id": tenant_id,
  1124→  1124→            "drift_detected": drift_detected,
  1125→  1125→            "drift_severity": drift_severity,
  1126→  1126→            "accuracy_7d": accuracy_7d,
  1127→  1127→            "accuracy_yesterday": accuracy_yesterday,
  1128→  1128→            "accuracy_today": accuracy_today,
  1129→  1129→            "trend_direction": trend_direction,
  1130→  1130→            "trigger_reason": trigger_reason,
  1131→  1131→            "total_scores": len(tenant_scores),

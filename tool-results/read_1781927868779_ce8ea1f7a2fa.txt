   179→def _check_emergency_keywords(text: str) -> Dict[str, Any]:
   180→    """Check text for emergency signals using keyword matching.
   181→
   182→    This is a safety gate — always runs, always FREE (no LLM).
   183→
   184→    Args:
   185→        text: The text to check.
   186→
   187→    Returns:
   188→        Dict with emergency_flag, emergency_type, and matched_keywords.
   189→    """
   190→    text_lower = text.lower()
   191→    matched: Dict[str, List[str]] = {}
   192→
   193→    for emergency_type, keywords in EMERGENCY_PATTERNS.items():
   194→        for keyword in keywords:
   195→            pattern = r'\b' + re.escape(keyword) + r'\b'
   196→            if re.search(pattern, text_lower):
   197→                matched.setdefault(emergency_type, []).append(keyword)
   198→
   199→    if matched:
   200→        priority_order = ["safety", "legal_threat", "compliance", "media"]
   201→        for etype in priority_order:
   202→            if etype in matched:
   203→                return {
   204→                    "emergency_flag": True,
   205→                    "emergency_type": etype,
   206→                    "matched_keywords": matched,
   207→                }
   208→
   209→    return {
   210→        "emergency_flag": False,
   211→        "emergency_type": "",
   212→        "matched_keywords": {},
   213→    }

   244→def _keyword_empathy_check(text: str) -> Dict[str, Any]:
   245→    """Simple keyword-based empathy/sentiment analysis.
   246→
   247→    Used as a FREE fallback when LLM is unavailable.
   248→
   249→    Args:
   250→        text: The text to analyze.
   251→
   252→    Returns:
   253→        Dict with empathy_score and empathy_flags.
   254→    """
   255→    text_lower = text.lower()
   256→    flags: List[str] = []
   257→    total_matches = 0
   258→
   259→    for flag_name, keywords in EMPATHY_PATTERNS.items():
   260→        for keyword in keywords:
   261→            if " " in keyword:
   262→                if keyword in text_lower:
   263→                    flags.append(flag_name)
   264→                    total_matches += 1
   265→                    break
   266→            else:
   267→                pattern = r'\b' + re.escape(keyword) + r'\b'
   268→                if re.search(pattern, text_lower):
   269→                    flags.append(flag_name)
   270→                    total_matches += 1
   271→                    break
   272→
   273→    if not flags:
   274→        empathy_score = 0.7
   275→    elif len(flags) == 1:
   276→        empathy_score = 0.4
   277→    elif len(flags) == 2:
   278→        empathy_score = 0.25
   279→    else:
   280→        empathy_score = 0.1
   281→
   282→    return {
   283→        "empathy_score": empathy_score,
   284→        "empathy_flags": flags,
   285→    }

   373→def _apply_crp_compression(text: str) -> Dict[str, Any]:
   374→    """Apply CRP (Concise Response Protocol) compression.
   375→
   376→    Removes filler phrases, redundant sentences, and verbose
   377→    language while preserving all factual content.
   378→
   379→    Target: 30-40% token reduction with >95% information retention.
   380→
   381→    Args:
   382→        text: The response text to compress.
   383→
   384→    Returns:
   385→        Dict with compressed_text, tokens_removed, compression_ratio.
   386→    """
   387→    if not text:
   388→        return {
   389→            "compressed_text": text,
   390→            "tokens_removed": 0,
   391→            "compression_ratio": 1.0,
   392→            "phrases_removed": [],
   393→        }
   394→
   395→    original_tokens = len(text.split())
   396→    compressed = text
   397→    phrases_removed: List[str] = []
   398→
   399→    # Step 1: Remove filler phrases
   400→    for phrase in CRP_FILLER_PHRASES:
   401→        if phrase.lower() in compressed.lower():
   402→            # Case-insensitive replacement
   403→            pattern = re.compile(re.escape(phrase), re.IGNORECASE)
   404→            compressed = pattern.sub("", compressed)
   405→            phrases_removed.append(phrase)
   406→
   407→    # Step 2: Clean up double spaces and orphaned punctuation
   408→    compressed = re.sub(r'\s{2,}', ' ', compressed)
   409→    compressed = re.sub(r'\.\s*\.', '.', compressed)  # double periods
   410→    compressed = re.sub(r'^\s+|\s+$', '', compressed, flags=re.MULTILINE)
   411→    compressed = compressed.strip()
   412→
   413→    # Step 3: Remove redundant "Thank you" when it appears multiple times
   414→    thank_you_count = len(re.findall(r'\bthank you\b', compressed, re.IGNORECASE))
   415→    if thank_you_count > 1:
   416→        # Keep only the first occurrence
   417→        first = True
   418→        parts = re.split(r'(\bThank you\b|\bthank you\b)', compressed, flags=re.IGNORECASE)
   419→        result_parts = []
   420→        for part in parts:
   421→            if re.match(r'\bthank you\b', part, re.IGNORECASE):
   422→                if first:
   423→                    result_parts.append(part)
   424→                    first = False
   425→                else:
   426→                    phrases_removed.append("duplicate thank you")
   427→            else:
   428→                result_parts.append(part)
   429→        compressed = "".join(result_parts)
   430→
   431→    final_tokens = len(compressed.split())
   432→    tokens_removed = max(0, original_tokens - final_tokens)
   433→    compression_ratio = final_tokens / original_tokens if original_tokens > 0 else 1.0
   434→
   435→    return {
   436→        "compressed_text": compressed,
   437→        "tokens_removed": tokens_removed,
   438→        "compression_ratio": round(compression_ratio, 3),
   439→        "phrases_removed": phrases_removed,
   440→    }

   444→# CLARA QUALITY GATE — Response validation pipeline
   445→# ══════════════════════════════════════════════════════════════════
   446→
   447→def _run_clara_quality_gate(
   448→    response: str,
   449→    query: str,
   450→    industry: str,
   451→    tone: str,
   452→    empathy_score: float,
   453→) -> Dict[str, Any]:
   454→    """Run the CLARA quality gate pipeline on a generated response.
   455→
   456→    CLARA = Concise Logical Adaptive Response Architecture.
   457→    Validates: Structure -> Logic -> Brand -> Tone -> Delivery
   458→
   459→    For Mini: Uses keyword/rule-based checks (FREE, no LLM).
   460→    For Pro/High: Would use LLM-based validation.
   461→
   462→    Args:
   463→        response: The generated response to validate.
   464→        query: The original customer query.
   465→        industry: Industry context for brand validation.
   466→        tone: Expected brand tone.
   467→        empathy_score: Customer empathy score for tone validation.
   468→
   469→    Returns:
   470→        Dict with passed, score, issues, and adjusted_response.
   471→    """
   472→    if not response:
   473→        return {
   474→            "passed": False,
   475→            "score": 0.0,
   476→            "issues": ["empty_response"],
   477→            "adjusted_response": response,
   478→            "checks": {},
   479→        }
   480→
   481→    checks: Dict[str, Dict[str, Any]] = {}
   482→    issues: List[str] = []
   483→    score = 100.0
   484→
   485→    # Check 1: STRUCTURE — Does response have logical structure?
   486→    has_greeting = bool(re.search(
   487→        r'\b(thank you|hello|hi|dear|greetings)\b',
   488→        response, re.IGNORECASE,
   489→    ))
   490→    has_acknowledgment = bool(re.search(
   491→        r'\b(understand|sorry|apologize|acknowledge|we see)\b',
   492→        response, re.IGNORECASE,
   493→    ))
   494→    has_action = bool(re.search(
   495→        r'\b(will|can|let me|our team|we\'ll|we will)\b',
   496→        response, re.IGNORECASE,
   497→    ))
   498→
   499→    structure_score = 0
   500→    if has_greeting or has_acknowledgment:
   501→        structure_score += 40
   502→    if has_action:
   503→        structure_score += 40
   504→    if len(response.split('.')) >= 2:  # Has multiple sentences
   505→        structure_score += 20
   506→
   507→    if structure_score < 40:
   508→        issues.append("poor_structure")
   509→        score -= 20
   510→
   511→    checks["structure"] = {
   512→        "score": structure_score,
   513→        "has_greeting": has_greeting,
   514→        "has_acknowledgment": has_acknowledgment,
   515→        "has_action": has_action,
   516→    }
   517→
   518→    # Check 2: LOGIC — Does response address the query topic?
   519→    query_tokens = set(re.findall(r'\b\w{4,}\b', query.lower()))
   520→    response_tokens = set(re.findall(r'\b\w{4,}\b', response.lower()))
   521→    stop_words = {
   522→        "that", "this", "with", "have", "will", "been", "from",
   523→        "they", "would", "could", "should", "there", "their",
   524→        "about", "which", "when", "where", "your", "please",
   525→    }
   526→    query_content = query_tokens - stop_words
   527→    response_content = response_tokens - stop_words
   528→    overlap = query_content & response_content if query_content else set()
   529→    logic_score = int(len(overlap) / max(len(query_content), 1) * 100)
   530→
   531→    if logic_score < 20:
   532→        issues.append("off_topic")
   533→        score -= 25
   534→
   535→    checks["logic"] = {
   536→        "score": logic_score,
   537→        "overlap_count": len(overlap),
   538→        "query_content_count": len(query_content),
   539→    }
   540→
   541→    # Check 3: BRAND — Is the tone appropriate for the industry?
   542→    inappropriate_words = {
   543→        "dude", "bro", "lol", "lmao", "rofl", "idk", "tbh",
   544→        "ngl", "smh", "bruh", "yolo",
   545→    }
   546→    response_lower = response.lower()
   547→    brand_violations = [w for w in inappropriate_words if w in response_lower.split()]
   548→    brand_score = 100 - (len(brand_violations) * 25)
   549→
   550→    if brand_violations:
   551→        issues.append("brand_violation")
   552→        score -= len(brand_violations) * 15
   553→
   554→    checks["brand"] = {
   555→        "score": max(0, brand_score),
   556→        "violations": brand_violations,
   557→    }
   558→
   559→    # Check 4: TONE — Is the empathy level appropriate?
   560→    tone_score = 80  # Default good
   561→    if empathy_score < 0.3 and not has_acknowledgment:
   562→        # Customer is distressed but no empathy shown
   563→        tone_score = 40
   564→        issues.append("insufficient_empathy")
   565→        score -= 15
   566→    elif empathy_score > 0.6:
   567→        tone_score = 90  # Neutral customer, neutral response is fine
   568→
   569→    checks["tone"] = {
   570→        "score": tone_score,
   571→        "empathy_score": empathy_score,
   572→        "has_acknowledgment": has_acknowledgment,
   573→    }
   574→
   575→    # Check 5: DELIVERY — Is the response complete and deliverable?
   576→    delivery_score = 80
   577→    if len(response.strip()) < 20:
   578→        delivery_score = 30
   579→        issues.append("response_too_short")
   580→        score -= 20
   581→    elif len(response.strip()) > 1000:
   582→        delivery_score = 70
   583→        # Not a failure, just info
   584→
   585→    # Check for placeholder text that shouldn't be delivered
   586→    placeholders = re.findall(r'\{+\w+\}+', response)
   587→    if placeholders:
   588→        issues.append("unresolved_placeholders")
   589→        score -= 10
   590→        delivery_score -= 20
   591→
   592→    checks["delivery"] = {
   593→        "score": delivery_score,
   594→        "response_length": len(response),
   595→        "has_placeholders": bool(placeholders),
   596→    }
   597→
   598→    # Compute final score
   599→    final_score = max(0.0, min(100.0, score))
   600→    passed = final_score >= 60.0  # Mini threshold: 60 (lower than Pro's 85)
   601→
   602→    # Auto-fix: adjust response if minor issues
   603→    adjusted_response = response
   604→    if issues and "unresolved_placeholders" in issues:
   605→        # Remove any unresolved template placeholders
   606→        adjusted_response = re.sub(r'\{+\w+\}+', '', adjusted_response)
   607→        adjusted_response = re.sub(r'\s{2,}', ' ', adjusted_response).strip()
   608→
   609→    return {
   610→        "passed": passed,
   611→        "score": round(final_score, 2),
   612→        "issues": issues,
   613→        "adjusted_response": adjusted_response,
   614→        "checks": checks,
   615→    }

   931→def gsd_state_node(state: ParwaGraphState) -> Dict[str, Any]:
   932→    """Agent 4: GSD State — Track and update conversation state machine.
   933→
   934→    GSD = Guided Support Dialogue (F-053).
   935→    State machine: NEW -> GREETING -> DIAGNOSIS -> RESOLUTION -> FOLLOW-UP -> CLOSED
   936→
   937→    For Mini: Simplified state transitions with keyword-based detection.
   938→    Tracks state in SharedGSDManager for analytics and recovery.
   939→
   940→    Writes: gsd_state, current_step
   941→    """
   942→    start = time.monotonic()
   943→    try:
   944→        company_id = state.get("company_id", "")
   945→        ticket_id = state.get("ticket_id", "")
   946→        query = state.get("pii_redacted_query") or state.get("query", "")
   947→        emergency_flag = state.get("emergency_flag", False)
   948→        empathy_score = state.get("empathy_score", 0.5)
   949→        classification = state.get("classification", {})
   950→
   951→        # Determine the GSD state transition
   952→        gsd_manager = _get_gsd_manager()
   953→        current_gsd_state = "new"
   954→
   955→        if gsd_manager:
   956→            current_gsd_state = gsd_manager.get_current_state(
   957→                company_id, ticket_id,
   958→            ) or "new"
   959→
   960→        # Determine next state based on signals (code-orchestrated = FREE)
   961→        if emergency_flag:
   962→            next_state = "escalate"
   963→        elif current_gsd_state == "new":
   964→            next_state = "greeting"
   965→        elif current_gsd_state == "greeting":
   966→            # Has classification? Move to diagnosis
   967→            if classification and classification.get("intent", "general") != "general":
   968→                next_state = "diagnosis"
   969→            else:
   970→                next_state = "diagnosis"  # Default progression
   971→        elif current_gsd_state == "diagnosis":
   972→            # Has generated response? Move to resolution
   973→            if state.get("generated_response"):

   976→                next_state = "diagnosis"  # Stay in diagnosis
   977→        elif current_gsd_state == "resolution":

   980→            next_state = "closed"
   981→        else:
   982→            next_state = current_gsd_state

   987→                company_id=company_id,

   993→                    "empathy_score": empathy_score,
   994→                    "variant_tier": "mini_parwa",
   995→                },
   996→            )
   997→
   998→        # Check for stuck tickets and get recovery suggestions
   999→        recovery_suggestions = []
  1000→        if gsd_manager:
  1001→            recovery_suggestions = gsd_manager.suggest_recovery(
  1002→                company_id, ticket_id,
  1003→            )
  1004→
  1005→        duration_ms = round((time.monotonic() - start) * 1000, 2)
  1006→
  1007→        audit_entry = append_audit_entry(
  1008→            state,
  1009→            step="gsd_state",
  1010→            action="gsd_state_transition",
  1011→            duration_ms=duration_ms,
  1012→            tokens_used=0,
  1013→            details={
  1014→                "from_state": current_gsd_state,
  1015→                "to_state": next_state,
  1016→                "emergency": emergency_flag,

  1019→
  1020→        return {
  1021→            "current_step": "gsd_state",
  1022→            "step_outputs": {
  1023→                "gsd_state": {
  1024→                    "status": "completed",
  1025→                    "from_state": current_gsd_state,
  1026→                    "to_state": next_state,
  1027→                    "recovery_suggestions": recovery_suggestions,
  1028→                    "duration_ms": duration_ms,
  1029→                }
  1030→            },
  1031→            "audit_log": audit_entry["audit_log"],
  1032→        }
  1033→
  1034→    except Exception as exc:
  1035→        duration_ms = round((time.monotonic() - start) * 1000, 2)
  1036→        logger.exception("gsd_state_node_failed", error=str(exc))
  1037→        return {
  1038→            "current_step": "gsd_state",
  1039→            "errors": [f"gsd_state: {str(exc)}"],
  1040→            "step_outputs": {
  1041→                "gsd_state": {
  1042→                    "status": "error",
  1043→                    "error": str(exc),
  1044→                    "duration_ms": duration_ms,
  1045→                }
  1046→            },
  1047→            "audit_log": [append_audit_entry(
  1048→                state, "gsd_state", "error",
  1049→                duration_ms=duration_ms,
  1050→                details={"error": str(exc)},

  1055→def extract_signals_node(state: ParwaGraphState) -> Dict[str, Any]:
  1056→    """Agent 5: Extract Signals — Derive query signals for technique routing.
  1057→
  1058→    Extracts: complexity, sentiment, monetary value, intent, etc.
  1059→    These signals feed into the Technique Router (BC-013).
  1060→
  1061→    For Mini: Keyword/rule-based signal extraction (FREE, no LLM).
  1062→    Writes to state.signals which the Technique Router reads.
  1063→
  1064→    Writes: signals, technique, current_step
  1065→    """
  1066→    start = time.monotonic()
  1067→    try:
  1068→        query = state.get("pii_redacted_query") or state.get("query", "")
  1069→        empathy_score = state.get("empathy_score", 0.5)
  1070→        empathy_flags = state.get("empathy_flags", [])
  1071→        customer_tier = state.get("customer_tier", "free")
  1072→        classification = state.get("classification", {})
  1073→        emergency_flag = state.get("emergency_flag", False)
  1074→
  1075→        # ── Signal Extraction (code-orchestrated = FREE) ───────
  1076→
  1077→        # 1. Query complexity (based on word count, question marks, multi-part)
  1078→        word_count = len(query.split())
  1079→        question_marks = query.count("?")
  1080→        has_multi_part = bool(re.search(
  1081→            r'\b(and|also|furthermore|moreover|as well|additionally)\b',
  1082→            query, re.IGNORECASE,
  1083→        ))
  1084→        complexity = min(1.0, (word_count / 50) * 0.4 + question_marks * 0.15 + (0.3 if has_multi_part else 0.0))
  1085→
  1086→        # 2. Sentiment score (derived from empathy)
  1087→        sentiment = empathy_score
  1088→
  1089→        # 3. Monetary value detection
  1090→        monetary_match = re.search(
  1091→            r'\$(\d+(?:,\d+)*(?:\.\d{2})?)',
  1092→            query,
  1093→        )
  1094→        monetary_value = 0.0
  1095→        if monetary_match:
  1096→            monetary_value = float(monetary_match.group(1).replace(",", ""))
  1097→
  1098→        # 4. Intent type (from classification if available)
  1099→        intent_type = classification.get("intent", "general") if classification else "general"
  1100→
  1101→        # 5. Turn count (from conversation_id presence)
  1102→        conversation_id = state.get("conversation_id", "")
  1103→        turn_count = 1 if conversation_id else 0
  1104→
  1105→        # 6. Resolution path count (estimate from query complexity)
  1106→        resolution_path_count = 1
  1107→        if complexity > 0.5 or intent_type in ("technical", "billing"):
  1108→            resolution_path_count = 3
  1109→        elif complexity > 0.3:
  1110→            resolution_path_count = 2
  1111→
  1112→        # 7. External data required (detect order IDs, account numbers)
  1113→        external_data = bool(re.search(
  1114→            r'(order|account|invoice|ticket|tracking)\s*#?\s*\d+',
  1115→            query, re.IGNORECASE,
  1116→        ))
  1117→
  1118→        # 8. Reasoning loop detection (check step outputs for repeated steps)
  1119→        reasoning_loop = False
  1120→        step_outputs = state.get("step_outputs", {})
  1121→        if isinstance(step_outputs, dict):
  1122→            completed_steps = [k for k, v in step_outputs.items()
  1123→                             if isinstance(v, dict) and v.get("status") == "completed"]
  1124→            if completed_steps.count("classify") > 1:
  1125→                reasoning_loop = True
  1126→
  1127→        # 9. Frustration score (derived from empathy + flags)
  1128→        frustration_score = 0.0
  1129→        if empathy_flags:
  1130→            frustration_map = {
  1131→                "frustrated": 50, "angry": 80, "urgent": 60,
  1132→                "sad": 40, "confused": 30,
  1133→            }
  1134→            frustration_score = max(
  1135→                frustration_map.get(f, 20) for f in empathy_flags
  1136→            )
  1137→
  1138→        signals = {
  1139→            "query_complexity": round(complexity, 3),
  1140→            "confidence_score": 1.0,  # Will be updated by confidence scoring
  1141→            "sentiment_score": round(sentiment, 3),

  1684→                    duration_ms=round((time.monotonic() - start) * 1000, 2),
  1685→                    details={"reason": "clara_not_active_or_empty"},
  1686→                )["audit_log"][0]],
  1687→            }
  1688→
  1689→        # Run CLARA quality gate
  1690→        tone = get_industry_tone(industry)

  1702→        # Run Confidence Scoring Engine (F-059) for additional validation
  1703→        confidence_result = None

  1705→            confidence_engine = _get_confidence_engine()

  1727→

  1738→                "confidence_score": confidence_result.overall_score if confidence_result else None,
  1739→                "final_quality_score": round(quality_score, 3),

  1741→        )

  1792→
  1793→    Uses FormatterRegistry with Mini defaults:
  1794→    - token_limit, markdown, whitespace (3 formatters)
  1795→
  1796→    Writes: formatted_response, final_response, response_format,

    56→    56→# ── Optional Auth Dependency ────────────────────────────────
    57→    57→# Validates JWT/API key if provided, but does NOT block unauthenticated requests.
    58→    58→# This allows the dashboard to work in dev without forcing full auth,
    59→    59→# while still validating credentials when they are present.
    60→    60→
    61→    61→async def _optional_auth(request: Request) -> Optional[Dict[str, Any]]:
    62→    62→    """Extract auth context from request if Authorization header is present.
    63→    63→
    64→    64→    Returns None if no auth header (allows unauthenticated dev access).
    65→    65→    Raises 401 if auth header is present but invalid.
    66→    66→    """
    67→    67→    from app.core.auth.access_control import validate_request
    68→    68→
    69→    69→    auth_header = request.headers.get("Authorization", "")
    70→    70→    if not auth_header.startswith("Bearer "):
    71→    71→        return None  # No auth provided — dev mode
    72→    72→
    73→    73→    token = auth_header[7:].strip()
    74→    74→    if not token:
    75→    75→        return None
    76→    76→
    77→    77→    client_ip = request.client.host if request.client else "unknown"
    78→    78→    is_valid, context, _ = await validate_request(
    79→    79→        api_key=token if token.startswith("pk_live_") or token.startswith("pk_test_") else None,
    80→    80→        jwt_token=token if not (token.startswith("pk_live_") or token.startswith("pk_test_")) else None,
    81→    81→        client_ip=client_ip,
    82→    82→    )
    83→    83→    if is_valid:
    84→    84→        return context
    85→    85→    return None  # Invalid but don't block — middleware can enforce later

   825→   825→    response_model=Dict[str, Any],
   826→   826→)
   827→   827→async def submit_ticket(req: TicketSubmitRequest):
   828→   828→    """Run a customer ticket through the full PARWA pipeline.
   829→   829→
   830→   830→    This endpoint processes the ticket INSIDE the server process,
   831→   831→    so quality scores, notifications, and audit entries are written
   832→   832→    to the same shared DB that all API endpoints read from.
   833→   833→
   834→   834→    Flow: Ingest → Classify → Route → Knowledge → Reasoning/Simple
   835→   835→          → Act+Verify → Quality → Finalize → END
   836→   836→    """
   837→   837→    import uuid
   838→   838→    import time as _time
   839→   839→
   840→   840→    from app.core.parwa_pipeline.graph_v2 import run_parwa_pipeline
   841→   841→    from app.core.parwa_pipeline.nodes.node_2_smart_route import set_test_variant
   842→   842→
   843→   843→    tenant_id = req.tenant_id or _DEFAULT_TENANT
   844→   844→    ticket_id = f"tkt_api_{uuid.uuid4().hex[:8]}"
   845→   845→
   846→   846→    # Emit SSE events
   847→   847→    await emit_pipeline_event(tenant_id, "ticket_received", {
   848→   848→        "ticket_id": ticket_id,
   849→   849→        "channel": req.channel_type,
   850→   850→    })
   851→   851→    await emit_pipeline_event(tenant_id, "pipeline_start", {"ticket_id": ticket_id})
   852→   852→
   853→   853→    # Set variant for this ticket
   854→   854→    set_test_variant(tenant_id, req.variant_tier, 2000)
   855→   855→
   856→   856→    initial_state = {
   857→   857→        "ticket_id": ticket_id,
   858→   858→        "tenant_id": tenant_id,
   859→   859→        "query": req.query,
   860→   860→        "channel_type": req.channel_type,
   861→   861→        "customer_context": req.customer_context or {},
   862→   862→        "metadata": {
   863→   863→            "sender": req.sender or "",
   864→   864→            "timestamp": datetime.now(timezone.utc).isoformat(),
   865→   865→            "source": "api_submission",
   866→   866→        },
   867→   867→        "loop_count": 0,
   868→   868→        "total_token_usage": 0,
   869→   869→        "technique_log": [],
   870→   870→        "errors": [],
   871→   871→    }
   872→   872→
   873→   873→    try:
   874→   874→        start = _time.monotonic()
   875→   875→        # run_parwa_pipeline is sync — handles async internally
   876→   876→        result = run_parwa_pipeline(initial_state)
   877→   877→        elapsed = round(_time.monotonic() - start, 2)
   878→   878→
   879→   879→        status = result.get("status", "unknown")
   880→   880→        quality = result.get("quality_score", None)
   881→   881→        route = result.get("route_decision", result.get("current_path", "unknown"))

   883→   883→            result.get("final_response", "")
   884→   884→            or result.get("formatted_response", "")
   885→   885→            or result.get("simple_answer", "")
   886→   886→            or result.get("super_node_answer", "")
   887→   887→        )
   888→   888→
   889→   889→        # Emit completion events
   890→   890→        await emit_pipeline_event(tenant_id, "pipeline_complete", {
   891→   891→            "ticket_id": ticket_id,
   892→   892→            "status": status,
   893→   893→            "quality": quality,
   894→   894→            "elapsed": elapsed,
   895→   895→        })
   896→   896→
   897→   897→        # Write audit entry
   898→   898→        from app.core.jarvis_pipeline.jarvis_db import get_db
   899→   899→        db = get_db()
   900→   900→        await db.create_audit_entry(
   901→   901→            tenant_id=tenant_id,
   902→   902→            action="ticket_processed",
   903→   903→            actor_email="system",
   904→   904→            target_type="ticket",
   905→   905→            target_id=ticket_id,
   906→   906→            payload={
   907→   907→                "status": status,
   908→   908→                "quality_score": quality,
   909→   909→                "route": route,
   910→   910→                "elapsed": elapsed,
   911→   911→                "channel": req.channel_type,
   912→   912→            },
   913→   913→        )
   914→   914→
   915→   915→        return {

   917→   917→            "ticket_id": ticket_id,
   918→   918→            "status": status,

   922→   922→            "escalated": bool(result.get("escalation_context")),

   924→   924→            "response_preview": response[:500],
   925→   925→        }

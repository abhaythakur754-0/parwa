   282→/**
   283→ * Central event dispatcher.
   284→ * Subscribes to all backend Socket.io events and routes them to Zustand stores.
   285→ * Called internally by SocketProvider.
   286→ */
   287→export function useRealtimeEvents(): void {
   288→  // Track whether we've registered listeners to avoid duplicates
   289→  const registeredRef = useRef(false);
   290→
   291→  // ── Ticket Event Handlers ──────────────────────────────────────
   292→
   293→  const handleTicketNew = useCallback((...args: unknown[]) => {
   294→    const data = args[0] as TicketNewData;
   295→    if (!data?.ticket) return;
   296→
   297→    const store = useTicketStore.getState();
   298→
   299→    // Check if ticket already exists (avoid duplicates)
   300→    const existing = store.getTicket(data.ticket.id);
   301→    if (existing) return;
   302→
   303→    // Prepend ticket to the list
   304→    store.addTicket({
   305→      subject: data.ticket.subject,
   306→      description: data.ticket.description,
   307→      category: data.ticket.category,
   308→      priority: data.ticket.priority,
   309→      channel: data.ticket.channel,
   310→      customer_name: data.ticket.customer_name,
   311→      customer_email: data.ticket.customer_email,
   312→    });
   313→
   314→    toast.success(`New ticket: ${data.ticket.subject}`, {
   315→      duration: 4000,
   316→      id: `ticket-new-${data.ticket.id}`,
   317→    });
   318→
   319→    updateLastEventTimestamp();
   320→  }, []);
   321→
   322→  const handleTicketAssigned = useCallback((...args: unknown[]) => {
   323→    const data = args[0] as TicketAssignedData;
   324→    if (!data?.ticketId) return;
   325→
   326→    const store = useTicketStore.getState();
   327→    const ticket = store.getTicket(data.ticketId);
   328→
   329→    if (ticket) {
   330→      // Update ticket with new agent assignment
   331→      const updatedTickets = store.tickets.map((t) =>
   332→        t.id === data.ticketId
   333→          ? { ...t, assigned_agent: data.agentName, updated_at: new Date().toISOString() }
   334→          : t
   335→      );
   336→      useTicketStore.setState({ tickets: updatedTickets });
   337→    }
   338→
   339→    toast.success(`Ticket assigned to ${data.agentName}`, {
   340→      duration: 3000,
   341→      id: `ticket-assigned-${data.ticketId}`,
   342→    });
   343→
   344→    updateLastEventTimestamp();
   345→  }, []);
   346→
   347→  const handleTicketMessageNew = useCallback((...args: unknown[]) => {
   348→    const data = args[0] as TicketMessageNewData;
   349→    if (!data?.ticketId || !data?.message) return;
   350→
   351→    const store = useTicketStore.getState();
   352→
   353→    // Add message to the ticket
   354→    store.addMessage(data.ticketId, {
   355→      sender: data.message.sender,
   356→      sender_name: data.message.sender_name,
   357→      content: data.message.content,
   358→      variant: data.message.variant,
   359→    });
   360→
   361→    // Increment unread counter (could be added to ticket store in future)
   362→    toast(`New message on ticket`, {
   363→      icon: '💬',
   364→      duration: 3000,
   365→      id: `ticket-msg-${data.ticketId}-${data.message.id}`,
   366→    });
   367→
   368→    updateLastEventTimestamp();
   369→  }, []);
   370→
   371→  const handleTicketResolved = useCallback((...args: unknown[]) => {
   372→    const data = args[0] as TicketResolvedData;
   373→    if (!data?.ticketId) return;
   374→
   375→    const store = useTicketStore.getState();
   376→    store.resolveTicket(data.ticketId, data.resolution);
   377→
   378→    toast.success('Ticket resolved!', {
   379→      duration: 4000,
   380→      id: `ticket-resolved-${data.ticketId}`,
   381→    });
   382→
   383→    updateLastEventTimestamp();
   384→  }, []);
   385→
   386→  const handleTicketEscalated = useCallback((...args: unknown[]) => {
   387→    const data = args[0] as TicketEscalatedData;
   388→    if (!data?.ticketId) return;
   389→
   390→    const store = useTicketStore.getState();
   391→    store.escalateToHuman(data.ticketId);
   392→
   393→    toast.error('Ticket escalated — requires human attention', {
   394→      duration: 5000,
   395→      id: `ticket-escalated-${data.ticketId}`,
   396→    });
   397→
   398→    updateLastEventTimestamp();
   399→  }, []);
   400→
   401→  const handleTicketSlaWarning = useCallback((...args: unknown[]) => {
   402→    const data = args[0] as TicketSlaWarningData;
   403→    if (!data?.ticketId) return;
   404→
   405→    toast(
   406→      `⚠️ SLA Warning: ${data.minutesRemaining} min remaining${data.ticketNumber ? ` for ${data.ticketNumber}` : ''}`,
   407→      {
   408→        icon: '⏰',
   409→        duration: 6000,
   410→        id: `sla-warning-${data.ticketId}`,
   411→      }
   412→    );
   413→
   414→    updateLastEventTimestamp();
   415→  }, []);
   416→
   417→  const handleTicketSlaBreached = useCallback((...args: unknown[]) => {
   418→    const data = args[0] as TicketSlaBreachedData;
   419→    if (!data?.ticketId) return;
   420→
   421→    toast.error(
   422→      `🚨 SLA Breached${data.ticketNumber ? ` on ${data.ticketNumber}` : ''}! Target: ${data.slaTarget}`,
   423→      {
   424→        duration: 8000,
   425→        id: `sla-breached-${data.ticketId}`,
   426→      }
   427→    );
   428→
   429→    updateLastEventTimestamp();
   430→  }, []);
   431→
   432→  const handleTicketCollision = useCallback((...args: unknown[]) => {
   433→    const data = args[0] as TicketCollisionData;
   434→    if (!data?.ticketId) return;
   435→
   436→    // Update collision store for UI banner
   437→    if (data.activeAgents && data.currentAgent) {
   438→      for (const agent of data.activeAgents) {
   439→        if (agent !== data.currentAgent) {
   440→          useCollisionStore.getState().userEntered(data.ticketId, agent, agent, 'viewing');
   441→        }
   442→      }
   443→    }
   444→
   445→    toast(
   446→      `⚠️ Collision detected: ${data.activeAgents?.length || 0} agent(s) viewing this ticket`,
   447→      {
   448→        icon: '👥',
   449→        duration: 4000,
   450→        id: `ticket-collision-${data.ticketId}`,
   451→      }
   452→    );
   453→
   454→    updateLastEventTimestamp();
   455→  }, []);
   456→
   457→  // ── Notification Event Handlers ────────────────────────────────
   458→
   459→  const handleNotificationNew = useCallback((...args: unknown[]) => {
   460→    const data = args[0];
   461→    if (!data) return;
   462→
   463→    // Route directly to the real notification store
   464→    useNotificationStore.getState().handleNotificationNew(data);
   465→
   466→    updateLastEventTimestamp();
   467→  }, []);
   468→
   469→  const handleNotificationRead = useCallback((...args: unknown[]) => {
   470→    const data = args[0] as NotificationReadData;
   471→    if (!data?.id) return;
   472→
   473→    useNotificationStore.getState().handleNotificationRead(data);
   474→    updateLastEventTimestamp();
   475→  }, []);
   476→
   477→  const handleNotificationBulk = useCallback((...args: unknown[]) => {
   478→    const data = args[0] as NotificationBulkData;
   479→    if (!data?.notifications) return;
   480→
   481→    useNotificationStore.getState().handleNotificationBulk(data);
   482→    updateLastEventTimestamp();
   483→  }, []);
   484→
   485→  // ── Approval Event Handlers ────────────────────────────────────
   486→
   487→  const handleApprovalPending = useCallback((...args: unknown[]) => {
   488→    const data = args[0];
   489→    if (!data) return;
   490→
   491→    // Route directly to the real approval store
   492→    useApprovalStore.getState().handleApprovalPending(data);
   493→
   494→    // Also push a notification about the approval
   495→    useNotificationStore.getState().addToast({
   496→      type: 'approval',
   497→      category: 'approval',
   498→      title: 'New Approval Request',
   499→      message: String((data as Record<string, unknown>).title || 'An AI agent needs your approval'),
   500→      priority: 'high',
   501→      actionUrl: '/dashboard/monitoring',
   502→      actionLabel: 'Review',
   503→    });
   504→
   505→    updateLastEventTimestamp();
   506→  }, []);
   507→
   508→  const handleApprovalApproved = useCallback((...args: unknown[]) => {
   509→    const data = args[0] as { id: string; respondedBy?: string; reason?: string };
   510→    if (!data?.id) return;
   511→
   512→    useApprovalStore.getState().handleApprovalApproved(data);
   513→
   514→    toast.success('Approval granted', {
   515→      duration: 3000,
   516→      id: `approval-approved-${data.id}`,
   517→    });
   518→
   519→    updateLastEventTimestamp();
   520→  }, []);
   521→
   522→  const handleApprovalRejected = useCallback((...args: unknown[]) => {
   523→    const data = args[0] as { id: string; respondedBy?: string; reason?: string };
   524→    if (!data?.id) return;
   525→
   526→    useApprovalStore.getState().handleApprovalRejected(data);
   527→
   528→    toast.error('Approval rejected', {
   529→      duration: 4000,
   530→      id: `approval-rejected-${data.id}`,
   531→    });
   532→
   533→    updateLastEventTimestamp();
   534→  }, []);
   535→
   536→  const handleApprovalTimeout = useCallback((...args: unknown[]) => {
   537→    const data = args[0] as { id: string };
   538→    if (!data?.id) return;
   539→
   540→    useApprovalStore.getState().handleApprovalTimeout(data);
   541→
   542→    toast.error('Approval timed out — escalated', {
   543→      duration: 5000,
   544→      id: `approval-timeout-${data.id}`,
   545→    });
   546→
   547→    updateLastEventTimestamp();
   548→  }, []);
   549→
   550→  const handleApprovalBulk = useCallback((...args: unknown[]) => {
   551→    const data = args[0] as { approvals: unknown[] };

   553→
   554→    useApprovalStore.getState().handleApprovalBatch(data);
   555→    updateLastEventTimestamp();
   556→  }, []);
   557→
   558→  // ── System Event Handlers ──────────────────────────────────────
   559→
   560→  const handleSystemHealth = useCallback((...args: unknown[]) => {
   561→    const data = args[0];
   562→    if (!data) return;
   563→
   564→    // Route directly to the real system health store
   565→    useSystemHealthStore.getState().handleSystemHealth(data);
   566→
   567→    updateLastEventTimestamp();
   568→  }, []);
   569→
   570→  const handleSystemQueueDepth = useCallback((...args: unknown[]) => {
   571→    const data = args[0];
   572→    if (!data) return;
   573→
   574→    useSystemHealthStore.getState().handleSystemQueueDepth(data);
   575→
   576→    updateLastEventTimestamp();
   577→  }, []);
   578→
   579→  const handleSystemError = useCallback((...args: unknown[]) => {
   580→    const data = args[0];
   581→    if (!data) return;
   582→
   583→    useSystemHealthStore.getState().handleSystemError(data);
   584→
   585→    // Only show toast for high-severity errors
   586→    const severity = (data as Record<string, unknown>).severity;
   587→    const message = String((data as Record<string, unknown>).message || 'System error');
   588→    if (severity === 'critical' || severity === 'high') {
   589→      toast.error(`System Error: ${message}`, {
   590→        duration: 6000,
   591→        id: `system-error-${Date.now()}`,
   592→      });
   593→    }
   594→
   595→    updateLastEventTimestamp();
   596→  }, []);
   597→
   598→  const handleSystemMaintenance = useCallback((...args: unknown[]) => {
   599→    const data = args[0];
   600→    if (!data) return;
   601→
   602→    useSystemHealthStore.getState().handleSystemMaintenance(data);
   603→
   604→    const isActive = Boolean((data as Record<string, unknown>).active ?? (data as Record<string, unknown>).is_maintenance ?? false);
   605→    const message = String((data as Record<string, unknown>).message || '');
   606→
   607→    if (isActive) {
   608→      toast(
   609→        `Scheduled Maintenance${message ? `: ${message}` : ''}`,
   610→        {
   611→          icon: '🔧',
   612→          duration: 10000,
   613→          id: 'system-maintenance',
   614→        }
   615→      );
   616→    }
   617→
   618→    updateLastEventTimestamp();
   619→  }, []);
   620→
   621→  // ── Billing Event Handlers ─────────────────────────────────────
   622→
   623→  const handleBillingPlanChanged = useCallback((...args: unknown[]) => {
   624→    const data = args[0] as BillingPlanChangedData;
   625→    if (!data?.tier) return;
   626→
   627→    // Validate tier
   628→    const validTiers: VariantTier[] = ['mini', 'parwa', 'high'];
   629→    if (!validTiers.includes(data.tier)) return;
   630→
   631→    // Update variant store
   632→    useVariantStore.getState().setTier(data.tier);
   633→
   634→    // Update billing store
   635→    useBillingStore.getState().fetchBilling();
   636→
   637→    toast.success(`Plan updated to ${data.tier.charAt(0).toUpperCase() + data.tier.slice(1)}`, {
   638→      duration: 4000,
   639→      id: 'billing-plan-changed',
   640→    });
   641→
   642→    updateLastEventTimestamp();
   643→  }, []);
   644→
   645→  const handleBillingUsageAlert = useCallback((...args: unknown[]) => {
   646→    const data = args[0] as BillingUsageAlertData;
   647→    if (!data) return;
   648→
   649→    const pct = Math.round(data.percentage * 100);
   650→
   651→    toast(
   652→      `📊 Usage Alert: ${data.resource} at ${pct}% (${data.used}/${data.limit})`,
   653→      {
   654→        icon: pct >= 90 ? '🔴' : '🟡',
   655→        duration: pct >= 90 ? 8000 : 5000,
   656→        id: `billing-usage-${data.resource}`,
   657→      }
   658→    );
   659→

   925→    chatStatePlaceholder.setTyping({
   926→      userId: data.userId,
   927→      isTyping: data.isTyping,

   938→      messageIds: data.messageIds,
   939→      readBy: data.readBy,

   941→

   988→    if (!data) return;

   994→    toast(`📞 Call ended${durationStr}`, {
   995→      duration: 4000,
   996→      icon: '✅',
   997→      id: `call-ended-${Date.now()}`,

  1001→  }, []);

  1093→    if (process.env.NODE_ENV === 'development') {

  1099→    // Cleanup on unmount

  1101→      for (const [event, handler] of events) {
  1102→        socketClient.off(event, handler);
  1103→      }
  1104→      registeredRef.current = false;
  1105→
  1106→      if (process.env.NODE_ENV === 'development') {
  1107→        console.log('[useRealtimeEvents] Unregistered all event listeners');
  1108→      }
  1109→    };
  1110→    // We intentionally only run this effect once on mount.
  1111→    // The callbacks are stable (useCallback with []) so they won't change.
  1112→    // eslint-disable-next-line react-hooks/exhaustive-deps
  1113→  }, []);
  1114→}

  1123→  socketClient.updateLastEventTimestamp(new Date().toISOString());
  1124→}

   273→  // ── Intent Detection ──────────────────────────────────────────
   274→
   275→  private detectIntent(msg: string, session: SessionLike): IntentResult {
   276→    const lower = msg.toLowerCase();
   277→    const ctx = session.context || {};
   278→    const scores: Record<string, number> = {};
   279→
   280→    for (const [intent, def] of Object.entries(INTENT_PATTERNS)) {
   281→      let score = 0;
   282→
   283→      // Keyword matching
   284→      for (const kw of def.keywords) {
   285→        if (lower.includes(kw)) score += def.weight * 1.5;
   286→      }
   287→
   288→      // Phrase matching (higher weight for multi-word matches)
   289→      for (const phrase of def.phrases) {
   290→        if (lower.includes(phrase)) score += def.weight * 3.0;
   291→      }
   292→
   293→      if (score > 0) scores[intent] = score;
   294→    }
   295→
   296→    // Contextual boosts
   297→    if (ctx.industry) {
   298→      const industryIntent = `industry_${ctx.industry}`;
   299→      if (scores[industryIntent]) scores[industryIntent] += 0.5;
   300→    }
   301→    if (ctx.detected_stage === 'demo' && scores.demo) scores.demo += 1.0;
   302→    if (ctx.detected_stage === 'pricing' && scores.pricing) scores.pricing += 1.0;
   303→
   304→    // Sort by score descending
   305→    const sorted = Object.entries(scores).sort((a, b) => b[1] - a[1]);
   306→
   307→    if (sorted.length === 0 || sorted[0][1] < 1.0) {
   308→      return { primary: 'unknown', confidence: 0, entities: [], isDemo: false, isObjection: false };
   309→    }
   310→
   311→    const primary = sorted[0][0];
   312→    const maxScore = sorted[0][1];
   313→
   314→    return {
   315→      primary,
   316→      confidence: Math.min(maxScore / 10, 1),
   317→      entities: this.extractEntities(lower, primary),
   318→      isDemo: primary === 'demo' || lower.includes('demo') || lower.includes('roleplay'),
   319→      isObjection: primary.startsWith('objection_'),
   320→    };
   321→  }

   346→  // ── FAQ Matching ──────────────────────────────────────────────

   348→  private matchFAQ(msg: string, intent: IntentResult): { a: string; q: string } | null {
   349→    const faqs = this.kb.faqs?.faqs;
   350→    if (!faqs || !faqs.length) return null;
   351→
   352→    let bestMatch: { a: string; q: string; score: number } | null = null;
   353→
   354→    for (const faq of faqs) {
   355→      let score = 0;
   356→      const qLower = faq.q.toLowerCase();
   357→      const tagsStr = (faq.tags || []).join(' ');
   358→
   359→      // Direct question match (very high weight)
   360→      const msgWords = msg.split(/\s+/).filter(w => w.length > 2);
   361→      const qWords = qLower.split(/\s+/).filter(w => w.length > 2);
   362→      const overlap = msgWords.filter(w => qWords.includes(w) || qLower.includes(w));
   363→      score += overlap.length * 2;
   364→
   365→      // Tag matching
   366→      for (const tag of (faq.tags || [])) {
   367→        if (msg.includes(tag.toLowerCase())) score += 1.5;
   368→      }
   369→
   370→      // Intent category boost
   371→      if (intent.primary === 'pricing' && faq.category === 'pricing') score += 2;
   372→      if (intent.primary === 'security' && faq.category === 'security') score += 2;
   373→      if (intent.primary === 'features' && faq.category === 'features') score += 2;
   374→      if (intent.primary.startsWith('industry_') && faq.category === 'industries') score += 2;
   375→
   376→      if (score > 3 && (!bestMatch || score > bestMatch.score)) {
   377→        bestMatch = { a: faq.a, q: faq.q, score };
   378→      }
   379→    }
   380→
   381→    return bestMatch;
   382→  }

   384→  // ── Objection Matching ────────────────────────────────────────

   386→  private matchObjection(msg: string, intentPrimary: string): { response: string; follow_up: string } | null {
   387→    const objections = this.kb.objections?.objections;
   388→    if (!objections || !objections.length) return null;
   389→
   390→    let bestMatch: { response: string; follow_up: string; score: number } | null = null;
   391→
   392→    for (const obj of objections) {
   393→      let score = 0;
   394→      const objLower = obj.objection.toLowerCase();
   395→
   396→      // Check if the objection text matches
   397→      const msgWords = msg.split(/\s+/).filter(w => w.length > 2);
   398→      const objWords = objLower.split(/\s+/).filter(w => w.length > 2);
   399→      const overlap = msgWords.filter(w => objWords.includes(w) || objLower.includes(w));
   400→      score += overlap.length * 2;
   401→
   402→      // Exact keyword matching for specific objection types
   403→      if (intentPrimary === 'objection_expensive' && (objLower.includes('expensive') || objLower.includes('cost'))) score += 5;
   404→      if (intentPrimary === 'objection_ai_quality' && (objLower.includes('complex') || objLower.includes('wrong'))) score += 5;
   405→      if (intentPrimary === 'objection_security' && objLower.includes('security')) score += 5;
   406→      if (intentPrimary === 'objection_setup_time' && objLower.includes('setup')) score += 5;
   407→      if (intentPrimary === 'objection_competitor' && (objLower.includes('already use') || objLower.includes('competitor'))) score += 5;

   409→
   410→      if (score > 4 && (!bestMatch || score > bestMatch.score)) {
   411→        bestMatch = { response: obj.jarvis_response, follow_up: obj.follow_up, score };
   412→      }
   413→    }
   414→
   415→    return bestMatch;

   813→  private formatIndustry(ind: string): string {
   814→    const map: Record<string, string> = {
   815→      ecommerce: 'E-commerce', saas: 'SaaS', logistics: 'Logistics', healthcare: 'Healthcare',
   816→      'e_commerce': 'E-commerce', 'e-commerce': 'E-commerce',
   817→    };
   818→    return map[ind] || ind.charAt(0).toUpperCase() + ind.slice(1);
   819→  }

   821→  private pickRandom<T>(arr: T[], session: SessionLike): T {
   822→    const recent = session.messages
   823→      ?.filter(m => m.role === 'jarvis')
   824→      .slice(-3)
   825→      .map(m => m.content.toLowerCase()) || [];
   826→
   827→    // Shuffle and pick first non-repeating
   828→    const shuffled = [...arr].sort(() => Math.random() - 0.5);
   829→    for (const item of shuffled) {
   830→      const text = String(item).toLowerCase().slice(0, 60);
   831→      if (!recent.some(r => r.includes(text) || text.includes(r.slice(0, 40)))) {
   832→        return item;
   833→      }
   834→    }
   835→    return arr[0];
   836→  }

   838→  private avoidRepetition(response: string, session: SessionLike): string {
   839→    const recent = session.messages
   840→      ?.filter(m => m.role === 'jarvis')
   841→      .slice(-3)
   842→      .map(m => m.content.toLowerCase()) || [];
   843→
   844→    const responseLower = response.toLowerCase();
   845→
   846→    // Check if this response is too similar to recent ones
   847→    for (const recentMsg of recent) {
   848→      const overlap = this.calculateOverlap(responseLower, recentMsg);
   849→      if (overlap > 0.7) {
   850→        // Too repetitive — add a follow-up to make it fresh
   851→        const followUps = [
   852→          '\n\nIs there anything specific you\'d like to explore further?',
   853→          '\n\nWant me to dive deeper into any of these areas?',
   854→          '\n\nWhat other questions do you have?',
   855→          '\n\nLet me know if you\'d like to see a demo of any of this!',
   856→        ];
   857→        const freshFollowUp = followUps.find(f =>
   858→          !recentMsg.includes(f.toLowerCase().slice(0, 30))
   859→        ) || followUps[0];
   860→        return response + freshFollowUp;
   861→      }
   862→    }
   863→
   864→    return response;
   865→  }
   866→
   867→  private calculateOverlap(a: string, b: string): number {
   868→    const wordsA = new Set(a.split(/\s+/).filter(w => w.length > 3));
   869→    const wordsB = new Set(b.split(/\s+/).filter(w => w.length > 3));
   870→    if (wordsA.size === 0) return 0;
   871→    let matches = 0;
   872→    for (const w of Array.from(wordsA)) {
   873→      if (wordsB.has(w)) matches++;
   874→    }
   875→    return matches / wordsA.size;
   876→  }

   878→  private personalizeResponse(template: string, session: SessionLike): string {
   879→    const ctx = session.context || {};
   880→    let result = template;
   881→
   882→    if (ctx.industry) {
   883→      result = result.replace(/\{industry\}/g, this.formatIndustry(ctx.industry));
   884→    }
   885→    if (ctx.selected_variants && ctx.selected_variants.length > 0) {
   886→      result = result.replace(/\{variant_count\}/g, String(ctx.selected_variants.length));
   887→    }
   888→    if (ctx.roi_result) {
   889→      result = result.replace(/\{roi\}/g, JSON.stringify(ctx.roi_result));
   890→    }
   891→
   892→    return result;
   893→  }

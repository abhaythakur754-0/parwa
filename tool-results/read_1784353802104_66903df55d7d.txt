    24→
    25→function formatCurrency(n: number): string { return `$${n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`; }
    26→function formatDate(iso: string | null): string { if (!iso) return '—'; return new Date(iso).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }); }
    27→function formatRelativeDate(iso: string | null): string {
    28→  if (!iso) return '—';
    29→  const d = new Date(iso); const now = new Date(); const diffMs = d.getTime() - now.getTime();
    30→  const days = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
    31→  if (days < 0) return `${Math.abs(days)}d ago`;
    32→  if (days === 0) return 'today'; if (days === 1) return 'tomorrow';
    33→  if (days < 30) return `in ${days}d`;
    34→  return formatDate(iso);
    35→}
